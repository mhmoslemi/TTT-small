"""
Improved optimizer using Gaussian-weighted Hankel QP with FFT-accelerated CG,
adaptive peak shaving via Newton steps, asymmetric exploration, and multi-length search.
Key innovations over previous: (1) Gaussian-weighted QP for better-conditioned Hessian,
(2) adaptive sigma continuation, (3) asymmetric restarts, (4) dynamic active-set refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=800, tol=1e-15):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian H[i,j]=2*W[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Try to continue with a restarted direction
            x_new = x + (rs / max(abs(pHp), 1e-30)) * p
            r_new = r - (rs / max(abs(pHp), 1e-30)) * Hp
            rs_new = float(r_new @ r_new)
            if rs_new < rs:
                x = x_new
                r = r_new
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)

        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new

def _gaussian_qp_solve(a_init, W, n, free_mask, max_cg=600):
    """
    Solve min a^T H a / 2 s.t. sum(a)=1, a>=0 on the free set.
    H[i,j] = 2*W[i+j]. Uses CG on the KKT system.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    K = np.zeros(2 * n)
    K[:n] = 2.0 * W
    K[n:2*n-1] += 2.0 * W[::-1]

    n_free = int(free_mask.sum())
    if n_free == 0:
        return a
    if n_free == n:
        free_idx = np.arange(n)
    else:
        free_idx = np.where(free_mask)[0]

    A = np.zeros((n_free, n_free))
    for di, i in enumerate(free_idx):
        A[di, :] = K[i + free_idx]

    b_free = np.zeros(n_free)
    for di, i in enumerate(free_idx):
        for dj, j in enumerate(free_idx):
            if i != j:
                b_free[di] += A[di, dj] * a[j]

    try:
        sol = np.linalg.solve(A, b_free)
        a_new = a.copy()
        a_new[free_idx] = sol
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            return a
        a_new = a_new / s_new
        return a_new
    except np.linalg.LinAlgError:
        return a


def _cg_qp_solve(W, n, a_init, free_mask, max_cg=800, tol=1e-14):
    """
    Solve QP: min (1/2)a^T H a s.t. sum(a)=1, a_i=0 for i not in free.
    H[i,j] = 2*W[i+j]. Uses CG on the reduced KKT system.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    n_free = int(free_mask.sum())
    if n_free == 0:
        return a

    # K[i,j] = H[i,j] + H[j,i] = 2*W[i+j] + 2*W[j+i] = 4*W[i+j] (since W is the convolution weight)
    # Actually K[i,j] = 2*W[i+j] + 2*W[j+i] = 4*W[i+j] only if we define it that way
    # Let me be precise: objective = sum_{i,j} W[i+j] a[i] a[j]
    # Gradient: grad[i] = sum_j (W[i+j] + W[j+i]) a[j] = 2 * sum_j W[i+j] a[j]
    # So K[i,j] = 2*W[i+j] + 2*W[j+i] = 4*W[i+j] (since W[i+j] = W[j+i] in our construction? No!)
    # Actually W is NOT symmetric in general. W[m] = weight at convolution index m.
    # H[i,j] = 2*W[i+j]. The gradient is sum_j H[i,j] a[j] = 2*sum_j W[i+j] a[j].
    # The Hessian is H itself: H[i,j] = 2*W[i+j].
    # For the KKT: H * a + lambda*1 = 0 on free vars, sum_free(a) = 1.

    # Let's define the full K matrix for the free variables:
    # K[di, dj] = H[free[di], free[dj]] = 2 * W[free[di] + free[dj]]

    # For CG, we need to apply K to vectors. K * v = 2 * (W_hankel * v) on the free set.
    # (W_hankel * v)[i] = sum_j W[i+j] v[j]

    # Build the full padded K for FFT:
    # K_full[i,j] = 2*W[i+j] for 0<=i,j<n
    # This is a Hankel matrix. To apply it: use the same technique as _hankel_matvec.
    # _hankel_matvec(v, W, n) computes 2*conv(W, v_rev)[n-1:2n-1] which is exactly (H*v)[i] = 2*sum_j W[i+j]v[j]

    # So the Hessian matvec is just _hankel_matvec(v, W, n).

    # KKT on free set:
    # (H_ff) * a_f + lambda * 1 = 0  (gradient of quadratic = 0)
    # 1^T a_f = 1
    # where H_ff is the submatrix of H on free indices.

    # From first eq: lambda = -(1/n_free) * sum_i (H_ff a_f)[i]
    # Substitute: H_ff a_f - (lambda/n_free)*sum_i(H_ff a_f)_i * 1 = 0
    # This is: (H_ff - (1/n_free) H_ff 1 1^T) a_f = 0
    # Which means a_f is in the null space... that's only the trivial solution.

    # Actually, the KKT for min (1/2)a^T H a s.t. 1^T a = 1 is:
    # H a + lambda * 1 = 0
    # 1^T a = 1
    # Solution: a = -lambda * H^{-1} 1, and lambda = -1 / (1^T H^{-1} 1)
    # So a = H^{-1} 1 / (1^T H^{-1} 1)

    # For the reduced problem (on free set):
    # a_f = H_ff^{-1} 1 / (1^T H_ff^{-1} 1)

    # We need to solve H_ff x = 1. This is the same as solving the full system
    # with x=0 on non-free indices, which we can do with CG on the full space
    # by projecting: only update free coordinates.

    # Simpler approach: embed in full space with a_i = 0 for non-free.
    # Solve H * y = e (where e is 1 on free, 0 elsewhere) using CG,
    # then only keep the free components.

    # Actually, the cleanest approach: solve the n_free x n_free system directly
    # using CG where the matvec is: (K v)[i] = 2 * sum_j W[free[i]+free[j]] * v[j]

    # For efficiency, let's use the full-space CG with masking:
    # At each CG iteration, after computing H*p, zero out non-free components.

    rhs_full = np.zeros(n)
    rhs_full[free_mask] = 1.0

    # CG with masking
    x = np.zeros(n)
    r = rhs_full.copy()
    r[~free_mask] = 0
    p = r.copy()
    rs = float(r @ r)
    if rs < 1e-30:
        return a
    rs0 = rs

    for i in range(max_cg):
        Hp = _hankel_matvec(p, W, n)
        Hp[~free_mask] = 0
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30:
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        r_new[~free_mask] = 0
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 2:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break

    # x should approximate H^{-1} * rhs_full on the free set
    one_x = float(np.sum(x[free_mask]))
    if abs(one_x) < 1e-20 or not np.isfinite(one_x):
        # Fallback: uniform on free set
        a_new = np.zeros(n)
        a_new[free_mask] = 1.0 / n_free
        return a_new

    a_new = np.zeros(n)
    a_new[free_mask] = x[free_mask] / one_x
    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    a_new = a_new / s_new
    return a_new


def _gaussian_weight(n, center, sigma):
    """Create Gaussian weight over convolution indices 0..2n-2."""
    idx = np.arange(2 * n - 1, dtype=np.float64)
    W = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return W


def _newton_polish_gaussian(a_init, deadline, max_rounds=200):
    """
    Newton polish with Gaussian-weighted QP.
    Adaptively narrows the Gaussian to focus on the peak.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with wide Gaussian, narrow it down
    c = _full_convolve(a, a)
    c_max = float(c.max())
    k_peak = int(np.argmax(c))

    # Initial sigma based on the width of the "top" region
    sorted_c = np.sort(c)[::-1]
    n_top = max(10, n // 4)
    if n_top < len(sorted_c):
        threshold = sorted_c[min(n_top, len(sorted_c)-1)]
        above = c >= threshold
        if above.sum() > 0:
            idx_above = np.where(above)[0]
            initial_sigma = max(5.0, (idx_above.max() - idx_above.min()) / 3.0)
        else:
            initial_sigma = n / 4.0
    else:
        initial_sigma = n / 4.0

    sigmas = initial_sigma * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18,
                                        0.13, 0.09, 0.065, 0.045, 0.03,
                                        0.02, 0.014, 0.01, 0.007, 0.005,
                                        0.0035, 0.0025, 0.0018, 0.0012,
                                        0.0008, 0.0005, 0.0003, 0.0002])
    sigmas = np.maximum(sigmas, 2.0)

    for sigma in sigmas:
        if time.monotonic() > deadline:
            break

        # Find the current peak
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_peak = int(np.argmax(c))

        # Create Gaussian weight centered at peak
        W = _gaussian_weight(n, k_peak, sigma)
        W = W / W.max()  # Normalize

        # Determine free set (positive elements)
        free_mask = a > 1e-12 * a.max()
        if free_mask.sum() < 5:
            free_mask = a > 1e-15

        # Solve QP
        rounds_for_sigma = 15
        for rnd in range(rounds_for_sigma):
            if time.monotonic() > deadline:
                break

            a_qp = _cg_qp_solve(W, n, a, free_mask, max_cg=600)
            sc_qp = _score(a_qp)

            if sc_qp < best_sc - 1e-15:
                best_sc = sc_qp
                best_a = a_qp.copy()
                a = a_qp.copy()
            else:
                # Try damped step
                dx = a_qp - a
                for frac in [0.5, 0.25, 0.125, 0.0625]:
                    a_try = a + frac * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

            # Update free mask
            free_mask = a > 1e-12 * a.max()
            if free_mask.sum() < 5:
                free_mask = a > 1e-15

            # Check convergence
            c_check = _full_convolve(a, a)
            grad = _hankel_matvec(a, W, n)
            grad_proj = grad - np.mean(grad[free_mask]) if free_mask.sum() > 0 else grad
            if np.max(np.abs(grad_proj[free_mask])) < 1e-12 * c_max:
                break

    return best_a, best_sc


def _peak_shave_newton(a_init, deadline, max_iters=500):
    """
    Newton-based peak shaving: directly target the single maximum.
    Uses the exact Hessian (which is a specific Hankel matrix) for the max entry.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_peak = int(np.argmax(c))

        # The Hessian of c[k_peak] is: H[i,j] = 2 if i+j == k_peak, else 0
        # This is a "reversal" type matrix on the valid indices.
        # For the Newton step, we want to solve:
        # H * d + lambda * 1 = -g  (on free set)
        # 1^T d = 0
        # where g = gradient of c[k_peak] = 2*a[k_peak - j] (shifted)

        # Gradient
        g = np.zeros(n)
        lo = max(0, k_peak - (n - 1))
        hi = min(n - 1, k_peak)
        for j in range(lo, hi + 1):
            g[j] = 2.0 * a[k_peak - j]

        # Hessian matvec: (H*v)[i] = 2*a... no, (H*v)[i] = 2*v[k_peak-i] (if valid)
        # Actually H[i,j] = 2*delta_{i+j, k_peak}, so (H*v)[i] = 2*v[k_peak-i] if 0<=k_peak-i<n

        # For the Newton step on the full space (ignoring inequality for now):
        # H*d + lambda*1 = -g
        # 1^T d = 0
        # From first eq: d[i] = (-g[i] - lambda) / 2 (if k_peak-i is valid)
        # This gives d[i] = -g[i]/2 - lambda/2 for valid i
        # Sum: sum_valid(-g[i]/2 - lambda/2) = 0
        # => -sum_valid(g[i])/2 - lambda*n_valid/2 = 0
        # => lambda = -sum_valid(g[i]) / n_valid

        valid = (k_peak >= np.arange(n)) & (k_peak - np.arange(n) >= 0) & (k_peak - np.arange(n) < n)
        # valid[i] is True if 0 <= k_peak - i < n, i.e., the Hessian row i has a non-zero entry
        n_valid = int(valid.sum())
        if n_valid == 0:
            break
        lam = -float(np.sum(g[valid])) / n_valid

        d = np.zeros(n)
        d[valid] = -g[valid] / 2.0 - lam / 2.0

        # Project: d should have zero sum (it does by construction if we used the right lambda)
        # But let's enforce it:
        d = d - np.mean(d)

        # Line search
        alpha = 1.0
        improved = False
        for _ in range(50):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-18:
                a = a_try
                sc_try = _score(a)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not improved:
            # Try sub-gradient step
            g_proj = g - np.mean(g)
            g_norm = np.linalg.norm(g_proj)
            if g_norm < 1e-15:
                break
            step = 0.1 / (1.0 + 0.01 * it)
            for frac in [step, step*0.5, step*0.25, step*0.1]:
                a_try = a - frac * g_proj
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < c_max:
                    a = a_try
                    sc_try = _score(a)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a.copy()
                    break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _obj_wrap(z, beta_box, n_box):
    return _smoobj(z, beta_box[0], n_box[0])


def _grad_wrap(z, beta_box, n_box):
    return _smograd(z, beta_box[0], n_box[0])


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001, 0.0000005, 0.0000002, 0.000001,
                           0.0000005, 0.0000002, 0.0000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002])

    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta_state = [base_beta]
    n_state = [n]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        beta_state[0] = beta

        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 5))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.35) ** 2) + 0.02 * np.exp(-3.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_reopt(a, deadline, min_time_remaining=60.0):
    """Try zeroing smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.02, 0.05, 0.08, 0.12, 0.15, 0.20, 0.25, 0.30]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _multi_peak_newton(a_init, deadline, max_iters=300):
    """Newton step targeting the top-m peaks simultaneously."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    m_schedule = [5, 10, 20, 50, 100, 200, 500]

    for m in m_schedule:
        if time.monotonic() > deadline:
            break

        for it in range(max_iters // len(m_schedule) + 10):
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            c_max = float(c.max())

            # Find top-m peak indices
            top_m = np.argsort(c)[::-1][:m]
            # Use soft weights: higher weight for higher peaks
            peak_vals = c[top_m]
            weights = np.exp(-(peak_vals - c_max) / (0.01 * c_max + 1e-30))
            weights = weights / weights.sum()

            # Build combined weight vector
            W = np.zeros(2 * n - 1)
            for i, k in enumerate(top_m):
                W[k] = weights[i]

            # Gradient: weighted sum of individual gradients
            g = np.zeros(n)
            for i, k in enumerate(top_m):
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    g[j] += weights[i] * 2.0 * a[k - j]

            # Project gradient
            g_proj = g - np.mean(g)
            g_norm = np.linalg.norm(g_proj)
            if g_norm < 1e-15:
                break

            # Newton-like step using the Hankel structure
            # Solve H*d = -g_proj where H[i,j] = 2*W[i+j]
            d = _cg_solve(W, n, -g_proj, max_iter=300, tol=1e-14)
            d = d - np.mean(d)  # Ensure zero sum

            # Line search
            alpha = 1.0
            improved = False
            for _ in range(40):
                a_try = a + alpha * d
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < c_max - 1e-16:
                    a = a_try
                    sc_try = _score(a)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a.copy()
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-14:
                    break

            if not improved:
                # Fallback: gradient step
                step = 0.5 / (1.0 + 0.05 * it)
                a_try = a - step * g_proj / g_norm
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    if _full_convolve(a_try, a_try).max() < c_max:
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    1. Aggressive L-BFGS with log-sum-exp smoothing
    2. Gaussian-weighted Newton QP with adaptive sigma
    3. Multi-peak Newton polishing
    4. Peak shaving via direct Newton steps
    5. Shrinkage and re-optimization
    6. Asymmetric and multi-length exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS ===
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, aggressive=True)
    _update_best(a1, _score(a1), "lbfgs_aggr")

    # === Phase 2: Gaussian Newton QP ===
    print(f"\n[Phase 2] Gaussian Newton QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _newton_polish_gaussian(best_seq, min(deadline - 500, time.monotonic() + 300),
                                           max_rounds=200)
        _update_best(a2, sc2, "gauss_qp_1")

    # === Phase 3: Multi-peak Newton ===
    print(f"\n[Phase 3] Multi-peak Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _multi_peak_newton(best_seq, min(deadline - 400, time.monotonic() + 200))
        _update_best(a3, sc3, "multi_np_1")

    # === Phase 4: L-BFGS second pass (from improved point) ===
    print(f"\n[Phase 4] L-BFGS pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a4 = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0, aggressive=False)
        _update_best(a4, _score(a4), "lbfgs_p2")

    # === Phase 5: Gaussian Newton QP round 2 ===
    print(f"\n[Phase 5] Gaussian Newton QP 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _newton_polish_gaussian(best_seq, min(deadline - 250, time.monotonic() + 150),
                                           max_rounds=150)
        _update_best(a5, sc5, "gauss_qp_2")

    # === Phase 6: Peak shave Newton ===
    print(f"\n[Phase 6] Peak shave (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a6, sc6 = _peak_shave_newton(best_seq, min(deadline - 200, time.monotonic() + 100))
        _update_best(a6, sc6, "peak_shave")

    # === Phase 7: Shrinkage ===
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a7, sc7 = _shrinkage_reopt(best_seq, deadline, min_time_remaining=100.0)
        _update_best(a7, sc7, "shrinkage")

    # === Phase 8: Multi-length exploration ===
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 8192, 32768, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if abs(n_try - len(best_seq)) < 100 if best_seq is not None else False:
            continue
        if best_seq is not None:
            a_init = _resample(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0, aggressive=True)
        _update_best(a_opt, _score(a_opt), f"n_{n_try}")

        if best_seq is not None and _score(best_seq) < best_score + 0.005:
            if time.monotonic() < deadline - 60:
                a_np, sc_np = _newton_polish_gaussian(a_opt, min(deadline - 50, time.monotonic() + 80),
                                                       max_rounds=50)
                _update_best(a_np, sc_np, f"n_{n_try}_qp")

    # === Phase 9: Asymmetric restarts ===
    print(f"\n[Phase 9] Asymmetric restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_r = len(best_seq)
        asym_profiles = ["asymmetric", "ramp"]
        for i, prof in enumerate(asym_profiles):
            if time.monotonic() > deadline - 30:
                break
            p_start = _make_profile(n_r, prof)
            # Blend with best
            blend = 0.5 * best_seq + 0.5 * p_start
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()
            p_start = _perturb(blend, 0.03, rng)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            _update_best(a_r, _score(a_r), f"asym_{i}")

            if time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_polish_gaussian(a_r, min(deadline - 10, time.monotonic() + 20),
                                                       max_rounds=30)
                _update_best(a_rp, sc_rp, f"asym_{i}_qp")

    # === Phase 10: Final ultra-tight polish ===
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Gaussian QP with very tight sigma
        a_fin, sc_fin = _newton_polish_gaussian(best_seq, deadline - 2, max_rounds=100)
        _update_best(a_fin, sc_fin, "final_gauss")

        # Multi-peak with tight settings
        if time.monotonic() < deadline - 3:
            a_mp, sc_mp = _multi_peak_newton(best_seq, deadline - 2, max_iters=100)
            _update_best(a_mp, sc_mp, "final_mp")

        # Ultra-fine L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-9
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.005)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    _update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
