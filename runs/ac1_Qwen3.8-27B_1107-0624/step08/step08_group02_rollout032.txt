"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous:
- Multi-active-set Newton: tries many active set sizes, keeps best
- Robust CG with residual checking and backtracking
- Coordinate descent final polish (1D convex minimization per coordinate)
- Better L-BFGS beta schedule based on top-k gap analysis
- Adaptive time allocation favoring final polish
- Multi-n exploration with smart resampling

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=2000, tol=1e-14):
    """Conjugate gradient with residual verification."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            x = x - alpha * p
            r = rhs - _hankel_matvec(x, W, n)
            rs_new = float(r @ r)
            if not np.isfinite(rs_new) or rs_new > rs * 1.0:
                break
        if rs_new > rs * 1.1:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_set(a_init, deadline, active_sizes=None, max_rounds=200):
    """
    Newton active set polish with multiple active set sizes.
    Tries different numbers of active constraints and keeps the best solution.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if active_sizes is None:
        active_sizes = [1, 2, 3, 5, 8, 12, 20, 30, 50, 80, 120, 200]

    for target_size in active_sizes:
        if time.monotonic() > deadline:
            break

        a_work = best_a.copy()
        c = _full_convolve(a_work, a_work)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]

        # Determine active set by target size
        if target_size <= len(c_sorted):
            threshold = c_sorted[min(target_size, len(c_sorted)) - 1]
        else:
            threshold = c_sorted[-1]
        # Add small buffer
        eps = max((c_max - threshold) * 0.1, c_max * 1e-10)
        active = c >= (c_max - eps)
        n_act = int(active.sum())
        if n_act == 0:
            active[int(np.argmax(c))] = True
            n_act = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_act

        converged = False
        for rnd in range(max_rounds):
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a_work, a_work)
            c_max = float(c.max())

            # Update active set dynamically
            tol = 1e-8 * c_max
            active_new = c >= (c_max - tol)
            n_act_new = int(active_new.sum())
            if n_act_new == 0:
                active_new[int(np.argmax(c))] = True
                n_act_new = 1
            # Blend: keep at least the core active set
            if n_act_new < max(1, n_act // 2):
                active_new = active | active_new
                n_act_new = int(active_new.sum())

            W = np.zeros(2 * n - 1)
            W[active_new] = 1.0 / n_act_new

            g = _hankel_matvec(a_work, W, n)
            g_mean = np.mean(g)
            g_t = g - g_mean
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-14 * n:
                converged = True
                break

            Hg = _cg_solve(W, n, g_t, max_iter=2000, tol=1e-14)
            H1 = _cg_solve(W, n, np.ones(n), max_iter=2000, tol=1e-14)
            H1_norm = float(np.linalg.norm(H1))

            if H1_norm < 1e-20 or not np.isfinite(H1_norm):
                dx = -g_t / (gn + 1e-30)
            else:
                lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / (gn + 1e-30)

            # Line search with armijo condition
            alpha = 1.0
            step_done = False
            c_max_init = c_max
            for ls in range(100):
                a_try = a_work + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-12:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try <= c_max_init - 1e-14 or c_max_try < c_max_init:
                    a_work = a_try
                    step_done = True
                    break
                alpha *= 0.5
                if alpha < 1e-18:
                    break

            if not step_done:
                # Subgradient fallback
                for trial in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                    a_try = a_work - trial * g_t / max(gn, 1e-30)
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-12:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max_init - 1e-15:
                        a_work = a_try
                        step_done = True
                        break

            sc = _score(a_work)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_work.copy()

        sc_final = _score(a_work)
        if sc_final < best_sc:
            best_sc = sc_final
            best_a = a_work.copy()

    return best_a, best_sc


def _coordinate_polish(a_init, deadline, max_rounds=20):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D search.
    The objective (max of convolution) is convex in each coordinate.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Precompute full convolution
    c = _full_convolve(a, a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False

        # Process coordinates in random order for diversity
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            # Compute the effect of changing a[i]
            # b_k = sum_j a_j * a_{k-j}
            # db_k/da_i = a_{k-i} + a_{i-k} (with bounds)
            # So b_k(a_i + delta) = b_k + delta * w_k
            # where w_k = a_{k-i} + a_{i-k} for valid indices

            w = np.zeros(2 * n - 1)
            # For k = 0, ..., 2n-2:
            # w_k = a_{k-i} if 0 <= k-i < n, plus a_{i-k} if 0 <= i-k < n
            idx1 = np.arange(2 * n - 1) - i
            valid1 = (idx1 >= 0) & (idx1 < n)
            w[valid1] += a[idx1[valid1]]
            idx2 = i - np.arange(2 * n - 1)
            valid2 = (idx2 >= 0) & (idx2 < n)
            w[valid2] += a[idx2[valid2]]

            # Minimize max_k (c_k + delta * w_k) subject to a[i] + delta >= 0
            # i.e., delta >= -a[i]
            # This is a 1D convex piecewise linear problem
            # Use subgradient bisection

            old_val = a[i]

            def obj_at_delta(delta):
                if delta < -old_val:
                    return np.inf
                return float(np.max(c + delta * w))

            # Golden section search on [lo, hi]
            lo = -old_val  # a[i] + delta >= 0
            hi = old_val * 2 + 1.0  # upper bound

            # Find a good upper bound
            while obj_at_delta(hi) > obj_at_delta(hi / 2) and hi < old_val * 100 + 10:
                hi = hi / 2

            # Golden section
            gr = (np.sqrt(5) - 1) / 2  # ~0.618
            a_gs = hi - gr * (hi - lo)
            b_gs = lo + gr * (hi - lo)
            fa = obj_at_delta(a_gs)
            fb = obj_at_delta(b_gs)

            for _ in range(60):
                if fa < fb:
                    hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = hi - gr * (hi - lo)
                    fa = obj_at_delta(a_gs)
                else:
                    lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = lo + gr * (hi - lo)
                    fb = obj_at_delta(b_gs)
                if hi - lo < 1e-12:
                    break

            best_delta = (lo + hi) / 2
            new_val = max(0.0, old_val + best_delta)

            if abs(new_val - old_val) > 1e-15:
                a[i] = new_val
                c = _full_convolve(a, a)
                improved = True

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        elif not improved:
            break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Adaptive beta L-BFGS on the softmax parametrization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    k = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[k - 1]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    stagnant_count = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 20))))

        try:
            obj_fn = lambda z_: _smoobj(z_, beta, n)
            grad_fn = lambda z_: _smograd(z_, beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stagnant_count = 0
            else:
                stagnant_count += 1
        except Exception:
            stagnant_count += 1

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]

        if new_gap < 1e-8 * c_max_new and stagnant_count == 0:
            beta *= 0.12
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        elif stagnant_count > 3:
            beta *= 0.15
            stagnant_count = 0
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_geom":
        # Asymmetric geometric-like decay
        t = np.linspace(0, 1, n)
        a = np.exp(-1.2 * t) * (1 + 0.3 * np.sin(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "smooth_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-0.8 * t) * (0.5 + 0.5 * np.cos(np.pi * t ** 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

                                            max_rounds=80)
        update_best(a_np2, sc_np2, "newton_pass2")

    # Phase 6: Coordinate descent again
    print(f"\n[Phase 6] Coord descent 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_cd2, sc_cd2 = _coordinate_polish(best_seq, deadline - 20, max_rounds=10)
        update_best(a_cd2, sc_cd2, "coord_desc2")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        t_n_end = deadline - 30
        a_opt_n = _run_adaptive_lbfgs(a_init_n, t_n_end, min_time_remaining=30.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")

        if sc_opt_n < best_score + 0.003 and time.monotonic() < deadline - 20:
            a_np_n, sc_np_n = _newton_active_set(a_opt_n, deadline - 10,
                                                  active_sizes=[1, 3, 5, 10, 20],
                                                  max_rounds=40)
            update_best(a_np_n, sc_np_n, f"n_{n_try}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 2,
                                            active_sizes=[1, 2, 3, 4, 5],
                                            max_rounds=100)
        update_best(a_fin, sc_fin, "final_newton")

        if best_seq is not None and time.monotonic() < deadline - 1:
            a_cdf, sc_cdf = _coordinate_polish(best_seq, deadline - 0.5, max_rounds=5)
            update_best(a_cdf, sc_cdf, "final_coord")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
