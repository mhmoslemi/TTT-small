   - The convolution entry `c_{k*} = sum_j a_j * a_{k*-j}`.
   - The "contributing pairs" are `(j, k*-j)` for valid `j`.
   - For each pair, we can reduce `a_j` or `a_{k*-j}` to reduce `c_{k*}`.
   - We want to reduce `c_{k*}` while keeping the sum constant and not increasing other `c_k` too much.
   - This is a small optimization sub-problem.

Let me code this up. I'll aim for a clean, efficient implementation.

One more thought: the problem has a symmetry. If `a` is a solution, then `a[::-1]` (reversed) is also a solution with the same score (since `convolve(a[::-1], a[::-1]) = convolve(a, a)[::-1]`, so the max is the same). We can exploit this by enforcing `a = a[::-1]` (symmetric) or `a = 1 - a[::-1]` (anti-symmetric around 1/2), but that might be too restrictive.

Actually, the symmetry is: `convolve(a, a)[k] = convolve(a[::-1], a[::-1])[2n-2-k]`. So the max is the same. This means we can without loss of generality assume some symmetry property, but it's not clear which one.

Let me just write the code now. I'll focus on making it efficient and time-aware.

# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def smooth_max_obj(z, beta, n):
    a = softmax(z)
    c = fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = fft_convolve(a_init, a_init)
    max_c = c_cur.max()
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)
    
    # Adaptive betas: start large, decrease geometrically
    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                       0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                       0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                       0.0000005, 0.0000002, 0.0000001, 0.00000005, 0.00000002,
                       0.00000001])
    betas = base_beta * ratios
    
    current_a = a_init.copy()
    current_sc = score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 5))))
        
        try:
            result = minimize(
                lambda z_local: smooth_max_obj(z_local, beta, n),
                z,
                jac=lambda z_local: smooth_max_grad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = softmax(result.x)
            sc_new = score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, try smaller step
                z = z - 0.1 * smooth_max_grad(z, beta, n)
                z = z - z.mean()
                a_new = softmax(z)
                sc_new = score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def coordinate_descent_polish(a_init, deadline, max_sweeps=50):
    """
    Coordinate descent: for each coordinate i, find the optimal a_i
    that minimizes max(c) given all other coordinates fixed.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = score(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Shuffle coordinates for diversity
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i = a[i]
            
            # The convolution entries affected by a_i are c_k for k where
            # the pair (j, k-j) involves index i.
            # c_k = sum_j a_j * a_{k-j}
            # a_i appears when j=i (then a_{k-i}) or k-j=i (then j=k-i, a_{k-i})
            # So c_k gets contribution: 2*a_i*a_{k-i} (if k!=2i and 0<=k-i<n)
            #                           a_i^2 (if k=2i)
            #
            # For a range of k: k goes from i to i+n-1 (clipped to [0, 2n-2])
            k_min = max(0, i)  # = i
            k_max = min(2*n-2, i+n-1)
            
            # Current c values (recompute for efficiency in small batches, or cache)
            # For efficiency, let's compute just the affected entries
            # c_k = 2*a_i*a_{k-i} + rest_k (for k != 2i)
            # c_{2i} = a_i^2 + rest_{2i}
            
            # Compute "rest" terms: c_k - 2*a_i*a_{k-i} (or c_k - a_i^2 for k=2i)
            # This is expensive to compute for all affected k.
            # Instead, let's use a cheaper approach:
            # 
            # The objective is max_k c_k. We want to choose a_i >= 0 to minimize this.
            # Since c_k is increasing in a_i (for all affected k, since a_{k-i} >= 0),
            # the optimal a_i is either 0 or the value where the max of affected c_k
            # equals the max of unaffected c_k.
            
            # Compute the current max of unaffected entries
            # This requires knowing which entries are affected.
            # Affected entries: k in [i, i+n-1] (clipped to [0, 2n-2])
            
            # For efficiency, let's just try a few candidate values of a_i:
            # 1. a_i = 0
            # 2. a_i = current value
            # 3. a_i = value that makes the "most affected" c_k equal to the current global max
            # 4. a_i = small positive value
            
            # Actually, a simpler approach: do a few steps of "push down the peak"
            c = fft_convolve(a, a)
            c_max = c.max()
            c_max_idx = np.argmax(c)
            
            # Is the max at an affected position?
            if c_max_idx >= k_min and c_max_idx <= k_max:
                # The max is affected by a_i. Try reducing a_i.
                # Binary search for the a_i that makes c_{c_max_idx} = second_max
                # (or 0 if a_i can't reduce it further)
                
                # c_{c_max_idx} as a function of a_i:
                # If c_max_idx == 2*i: c = a_i^2 + rest
                # Else: c = 2*a_i*a_{c_max_idx - i} + rest
                
                if c_max_idx == 2 * i and 0 <= 2*i < 2*n-1:
                    # Quadratic: try a few values
                    candidates = [0.0, a_i * 0.5, a_i * 0.25, a_i * 0.1]
                else:
                    other = a[c_max_idx - i] if 0 <= c_max_idx - i < n else 0
                    if other < 1e-15:
                        candidates = [a_i]  # Can't reduce this c_k by changing a_i
                    else:
                        # c_{c_max_idx} = 2*a_i*other + rest
                        # The rest is c_{c_max_idx} - 2*a_i*other
                        rest = c[c_max_idx] - 2 * a_i * other
                        # We want 2*new_ai*other + rest <= current_second_max
                        # But this is complicated. Just try halving.
                        candidates = [0.0, a_i * 0.5, a_i * 0.75]
                
                best_ai = a_i
                best_local = c_max
                for new_ai in candidates:
                    a[i] = max(0, new_ai)
                    # Quick check: recompute only the affected part
                    # For speed, just check the specific c_max_idx
                    if c_max_idx == 2 * i:
                        new_ck = a[i]**2 + rest  # approximate
                    else:
                        other = a[c_max_idx - i] if 0 <= c_max_idx - i < n else 0
                        rest = c[c_max_idx] - 2 * a_i * other  # this is stale after changing a[i]
                        # Actually this is getting complicated. Let me just do a full recheck periodically.
                        new_ck = 2 * a[i] * other + (c[c_max_idx] - 2 * a_i * other)
                    # This is Taylor approximation, not exact. For simplicity, skip.
                
                # Simpler approach: just halve a_i and check
                a[i] = a_i * 0.7  # reduce by 30%
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = a_i
                    continue
                a = a / s_new
                sc = score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    # Revert
                    a = best_a.copy()
            else:
                # Max is not affected by a_i. We can increase a_i freely (up to normalization).
                # But this might make other c_k larger. For now, skip.
                pass
    
    return best_a

def active_set_polish(a_init, deadline, max_rounds=100):
    """Active-set QP: minimize a^T H_W a s.t. sum(a)=1, a>=0"""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, score(a)
    a = a / s
    best_a = a.copy()
    best_sc = score(a)
    
    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                    0.0000005, 0.0000002, 0.0000001, 0.00000005, 0.00000002,
                    0.00000001]
    
    tol_idx = 0
    rounds_per_tol = 5
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(a, a)
        c_max = c.max()
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = active.sum()
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        
        # Gradient: 4n * a * (h - W_val) where h = convolve(W, a_rev)[n-1:2n-1]
        a_rev = a[::-1]
        wc = fft_convolve(W, a_rev)
        h = wc[n-1:2*n-1]
        W_val = float(np.dot(W, c))
        grad = 4.0 * n * a * (h - W_val)
        
        # Newton step: solve the linear system
        # The Hessian of the smooth approximation at small beta is approximately:
        # H[i,j] = 4n * (contributions from active set)
        # For simplicity, use a steepest descent step with line search
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search steepest descent
        a_try = a - (1.0 / grad_norm) * grad
        alpha = 1.0
        improved = False
        for ls in range(30):
            a_try = a - alpha * grad / grad_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break
        
        if not improved:
            # Try with a smaller step
            alpha = 0.01
            a_try = a - alpha * grad / grad_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
        
        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1
    
    return best_a, best_sc

def resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return softmax(z)

def make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
