"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Aggressive log-sum-exp L-BFGS with extended beta continuation,
adaptive active-set Newton for peak equalization, multi-scale warm start,
and stochastic subgradient polishing. Uses global state to avoid lambdas/closures.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global state for optimizer callbacks
_G_BETA = 1.0
_G_N = 0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _obj_wrapper(z):
    return _lsobj(z, _G_BETA, _G_N)


def _grad_wrapper(z):
    return _lsgrad(z, _G_BETA, _G_N)


def _corr_a_A(a, A_vec, n):
    """Matrix-vector product for Hessian of active-set smoothed max."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Aggressive L-BFGS with extended beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap_idx = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[gap_idx]
    base_beta = max(gap * 0.2, max_c * 0.003)

    # Extended aggressive beta schedule
    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075, 0.0037,
         0.00185, 0.000925, 0.00046, 0.00023, 0.000115, 0.000058,
         0.000029, 0.000014, 0.000007, 0.000003, 0.000001])

    global _G_BETA, _G_N
    _G_N = n

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    t0 = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - min_time_remaining:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if n <= 2048:
            est_tpi = 0.002
        elif n <= 4096:
            est_tpi = 0.005
        elif n <= 8192:
            est_tpi = 0.01
        elif n <= 16384:
            est_tpi = 0.022
        elif n <= 32768:
            est_tpi = 0.05
        else:
            est_tpi = 0.1

        if i < len(betas) // 2:
            max_iter = min(3000, max(50, int(remaining / (est_tpi * 4))))
        else:
            max_iter = min(15000, max(100, int(remaining / est_tpi)))

        _G_BETA = beta
        _G_N = n
        bt0 = time.monotonic()
        try:
            result = minimize(
                _obj_wrapper,
                z,
                jac=_grad_wrapper,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 300,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            improved = sc_new < current_sc - 1e-15
            if improved:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if verbose and (i % 3 == 0 or i >= len(betas) - 3):
                print(f"      [LBFGS] beta={beta:.2e} iter={result.nit} "
                      f"score={current_sc:.10f} t={time.monotonic()-bt0:.1f}s")
        except Exception as ex:
            if verbose:
                print(f"      [LBFGS] error at beta={beta:.2e}: {ex}")
            break

    if verbose:
        print(f"    [LBFGS done] final score={current_sc:.10f} "
              f"total_time={time.monotonic()-t0:.1f}s")
    return current_a


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001, verbose=True):
    """Active-set Newton: equalize top peaks via structured quadratic subproblems."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            n_active = 4 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of smoothed max over active set
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        mu = 1e-7 * (c_max / max(m, 1)) + 1e-10

        # Conjugate gradient for the Hessian system
        def hess_mv(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve for hess_matvec(u) = g_t
        x = np.zeros(n)
        r = g_t - hess_mv(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = hess_mv(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        u = x

        # Solve for sum constraint: hess_matvec(w) = 1
        x = np.zeros(n)
        rhs = np.ones(n) * 0.5  # scale down for numerical stability
        r = rhs - hess_mv(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = hess_mv(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        w = x
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search with box constraint
        neg_mask = dx < -1e-14
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old or ls >= 40:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: gradient step
            t = 0.05
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"      [AS-Newton] iter={it} score={best_sc:.10f} "
                      f"active={n_active} cmax={c_max:.6e}")

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _stochastic_subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.02, verbose=True):
    """Stochastic subgradient: randomly sample active peaks for gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    rng = np.random.default_rng(12345)

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Random subset of active peaks
        tol = 0.01 * c_max
        active_indices = np.where(c >= c_max - tol)[0]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])
        K = min(32, len(active_indices))
        sampled = rng.choice(active_indices, size=K, replace=False)

        w = np.zeros(2 * n - 1)
        w[sampled] = 1.0 / K

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(lr_init * 0.3, 1e-4)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.arange(n)
        a = 0.5 * (1 + np.cos(2.0 * np.pi * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.arange(n)
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t / n) + 0.08 * np.cos(4.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _trim_and_refit(a, n_target, deadline):
    """Try reducing effective n by zeroing small elements."""
    n = len(a)
    if n_target >= n:
        return a, _score(a)
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    peak = c.max()
    # Elements contributing little to the peak can be zeroed
    # Compute contribution of each element to the peak
    peak_idx = int(np.argmax(c))
    contribution = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contribution[i] = a[i] * a[j]

    # Sort by contribution (ascending) and try zeroing
    sorted_idx = np.argsort(contribution)
    for frac in [0.05, 0.1, 0.15, 0.2]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            if 0.15 >= frac:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Evaluate and refine parent
    Phase 2: Aggressive L-BFGS on parent
    Phase 3: Multi-scale (small n -> large n) warm start
    Phase 4: Try n=32768
    Phase 5: Active-set Newton polishing
    Phase 6: Stochastic subgradient polish
    Phase 7: Randomized restarts
    Phase 8: Final active-set with tight tolerance
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.6e} t={time.monotonic() - t_start:.0f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "hann")
        sc_parent = _score(parent)
        print(f"No parent, using Hann n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init_hann")

    # Phase 1: Quick structured init exploration (find better starting point)
    print(f"\n[Phase 1] Structured inits (t={time.monotonic() - t_start:.0f}s)")
    for kind in ["hann", "blackman", "raised_cos", "sine"]:
        if time.monotonic() > deadline - 800:
            break
        a_test = _make_profile(n_parent, kind)
        sc_test = _score(a_test)
        print(f"  {kind}: score={sc_test:.10f}")
        if sc_test < best_score:
            update_best(a_test, sc_test, f"init_{kind}")

    # Phase 2: Main aggressive L-BFGS
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        main_deadline = deadline - 350
        a_main = _run_lbfgs(best_seq, main_deadline, min_time_remaining=300, verbose=True)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 3: Multi-scale warm start
    print(f"\n[Phase 3] Multi-scale (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        # Downsample, optimize, upsample
        for n_small in [1024, 2048]:
            if time.monotonic() > deadline - 200:
                break
            a_small = _interpolate_to_n(best_seq, n_small)
            sc_small = _score(a_small)
            print(f"  n={n_small} init: score={sc_small:.10f}")

            a_small_opt = _run_lbfgs(a_small, deadline - 150, min_time_remaining=100, verbose=False)
            sc_small_opt = _score(a_small_opt)
            print(f"  n={n_small} opt: score={sc_small_opt:.10f}")

            # Upsample back
            a_up = _interpolate_to_n(a_small_opt, n_parent)
            a_up_opt = _run_lbfgs(a_up, deadline - 100, min_time_remaining=80, verbose=False)
            update_best(a_up_opt, _score(a_up_opt), f"multiscale_{n_small}")

    # Phase 4: Active-set Newton on current best
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        as_deadline = min(deadline - 50, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=300,
                                         tol_active=0.002, verbose=True)
        update_best(a_as, sc_as, "as_newton")

        # Tighter tolerance
        if time.monotonic() < deadline - 30:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 10, time.monotonic() + 60),
                                               max_iter=200, tol_active=0.0005, verbose=False)
            update_best(a_as2, sc_as2, "as_tight")

    # Phase 5: Stochastic subgradient polish
    print(f"\n[Phase 5] Stochastic subgradient (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        sg_deadline = min(deadline - 20, time.monotonic() + 80)
        a_sg, sc_sg = _stochastic_subgrad_polish(best_seq, sg_deadline, n_steps=8000,
                                                 lr_init=0.015, verbose=False)
        update_best(a_sg, sc_sg, "stoch_subgrad")

    # Phase 6: Randomized restarts
    print(f"\n[Phase 6] Randomized restarts (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_count = 0
        for sigma in [0.01, 0.03, 0.08, 0.15]:
            if time.monotonic() > deadline - 15 or restart_count >= 4:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(len(best_seq))
            z = z - z.mean()
            a_pert = _softmax(z)
            sc_pert = _score(a_pert)
            print(f"  restart sigma={sigma}: score={sc_pert:.10f}")

            a_r = _run_lbfgs(a_pert, deadline - 10, min_time_remaining=5, verbose=False)
            update_best(a_r, _score(a_r), f"restart_{sigma}")

            # Quick active-set on the restart result
            if _score(a_r) < best_score + 0.001 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 2, time.monotonic() + 20),
                                                 max_iter=50, tol_active=0.005, verbose=False)
                update_best(a_r2, sc_r2, f"restart_{sigma}_as")
            restart_count += 1

    # Phase 7: Try n=32768 if time allows
    print(f"\n[Phase 7] n=32768 attempt (t={time.monotonic() - t_start:.0f}s)")
    if time.monotonic() < deadline - 120:
        n_large = 32768
        if best_seq is not None:
            a_large_init = _interpolate_to_n(best_seq, n_large)
        else:
            a_large_init = _make_profile(n_large, "hann")
        sc_large = _score(a_large_init)
        print(f"  n={n_large} init: score={sc_large:.10f}")
        update_best(a_large_init, sc_large, "init_32768")

        a_large = _run_lbfgs(a_large_init, deadline - 20, min_time_remaining=10, verbose=True)
        update_best(a_large, _score(a_large), "lbfgs_32768")

        if time.monotonic() < deadline - 5:
            a_large_as, sc_large_as = _active_set_newton(a_large, deadline - 2,
                                                         max_iter=50, tol_active=0.003, verbose=False)
            update_best(a_large_as, sc_large_as, "as_32768")

    # Phase 8: Final tight active-set polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1,
                                           max_iter=150, tol_active=0.0001, verbose=True)
        update_best(a_fin, sc_fin, "final_as")

        # One more subgradient pass
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _stochastic_subgrad_polish(best_seq, deadline - 0.5,
                                                         n_steps=3000, lr_init=0.005, verbose=False)
            update_best(a_fin2, sc_fin2, "final_sg")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
