def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_sum1_nonneg(x):
    """Project x onto {v: v >= 0, sum(v) = 1}."""
    n = len(x)
    # Find lambda such that sum(max(x - lambda, 0)) = 1
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    rho = np.arange(1, n + 1)
    cond = sorted_x - (cumsum - 1) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_star = int(np.max(np.where(cond)[0])) + 1
    theta = (cumsum[rho_star - 1] - 1.0) / rho_star
    return np.maximum(x - theta, 0.0)


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)

    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _topk_obj_grad(a, K):
    """Compute objective (2n*max_conv approximated by top-K mean) and gradient."""
    n = len(a)
    c = _full_convolve(a, a)
    # Find top K
    K_eff = min(K, len(c))
    top_idx = np.argpartition(c, -K_eff)[-K_eff:]
    c_max = float(c[top_idx].max())
    # Smooth approximation: weighted average of top K
    # Use softmax weights over top K for smoothness
    c_top = c[top_idx]
    c_top_shifted = (c_top - c_top.max()) / 0.01
    w_top = np.exp(c_top_shifted)
    w_top = w_top / w_top.sum()
    obj = 2.0 * n * float(np.dot(w_top, c_top))
    # Gradient: weighted average of gradients at top positions
    w = np.zeros_like(c)
    for j, idx in enumerate(top_idx):
        w[idx] = w_top[j]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad_t = grad - np.mean(grad)
    return obj, grad_t, c

            lr = min(0.04, lr * 1.02)

    return best_a, best_sc


def _lse_lbfgs(a_init, deadline, min_remaining=20.0, aggressive=True):
    """L-BFGS with LSE approximation and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.003)

    if aggressive:
        betas = base_beta * np.array(
            [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015, 0.008,
             0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001,
             0.00005, 0.00002, 0.00001]
        )
    else:
        betas = base_beta * np.array(
            [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
             0.004, 0.002, 0.001]
        )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 20))))

        class LseFunc:
            def __init__(self, b, nn):
                self.b = b
                self.nn = nn
            def __call__(self, z_in):
                return _lse_obj(z_in, self.b, self.nn)
            def grad(self, z_in):
                return _lse_grad(z_in, self.b, self.nn)

        prob = LseFunc(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        edge = 0.1
        a = np.ones(n)
        mask_left = t < edge
        mask_right = t > 1 - edge
        a[mask_left] = np.sin(np.pi * t[mask_left] / (2 * edge)) ** 0.5
        a[mask_right] = np.sin(np.pi * (1 - t[mask_right]) / (2 * edge)) ** 0.5
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "centralized":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _symmetrize(a):
    """Make sequence mirror-symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _proj_sum1_nonneg(z)


def _spectral_init(n, seed):
    """Initialize using a designed power spectrum."""
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n)
    # Design a spectrum that's approximately flat but with some structure
    freqs = np.fft.rfftfreq(N)
    # Avoid DC and Nyquist
    mask = (np.abs(freqs) > 1.0 / N) & (np.abs(freqs) < 0.5 - 1.0 / N)
    power = np.ones(N // 2 + 1)
    power[~mask] = 0
    # Add some random phase
    phase = rng.standard_normal(N // 2 + 1) * np.pi
    amplitude = np.sqrt(power) * (1 + 0.3 * rng.standard_normal(N // 2 + 1))
    amplitude = np.maximum(amplitude, 0)
    spectrum = amplitude * np.exp(1j * phase)
    spectrum[0] = 0  # No DC
    # IFFT to get time-domain signal
    a_complex = np.fft.irfft(spectrum, N)
    a = a_complex[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Diverse initialization (parent, spectral, profiles)
    2. LSE + L-BFGS coarse optimization
    3. Top-K projected gradient with BB steps
    4. Active-set KKT polish
    5. Multi-n exploration
    6. Final fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}", flush=True)
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}", flush=True)

    # === Phase 1: Try diverse initializations and pick the best ===
    print(f"\n[P1] Diverse initialization (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_main = n_parent
    if time.monotonic() < deadline - 300:
        candidates = [
            ("parent", best_a.copy()),
            ("sym_parent", _symmetrize(best_a)),
            ("tapered", _make_profile(n_main, "tapered")),
            ("flat_top", _make_profile(n_main, "flat_top")),
            ("spectral_1", _spectral_init(n_main, 12345)),
            ("spectral_2", _spectral_init(n_main, 54321)),
            ("cosine", _make_profile(n_main, "cosine")),
        ]
        # Briefly optimize each
        quick_deadline = min(t_start + 60.0, deadline - 250.0)
        for label, a_cand in candidates:
            if time.monotonic() > quick_deadline:
                break
            sc_cand = _score(a_cand)
            # Quick LSE optimization (small budget)
            remaining = quick_deadline - time.monotonic()
            if remaining > 15:
                a_opt = _lse_lbfgs(a_cand, time.monotonic() + remaining * 0.8,
                                   min_remaining=5.0, aggressive=False)
                sc_opt = _score(a_opt)
                if sc_opt < sc_cand:
                    a_cand = a_opt
                    sc_cand = sc_opt
            if sc_cand < best_score:
                best_a = a_cand.copy()
                best_score = sc_cand
                print(f"  Init {label}: {sc_cand:.10f} -> NEW BEST", flush=True)
            else:
                print(f"  Init {label}: {sc_cand:.10f}", flush=True)

    # === Phase 2: Aggressive LSE + L-BFGS on best ===
    print(f"\n[P2] LSE+L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p2_deadline = min(deadline - 900.0, t_start + 400.0)
    a2 = _lse_lbfgs(best_a, p2_deadline, min_remaining=30.0, aggressive=True)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [P2_lbfgs] t={time.monotonic()-t_start:.1f}s", flush=True)
    else:
        print(f"  Score: {sc2:.10f} (no improvement)", flush=True)

    # === Phase 3: Top-K Projected Gradient with BB steps ===
    print(f"\n[P3] Top-K PG + BB (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p3_deadline = min(deadline - 500.0, t_start + 800.0)
    a3, sc3 = _topk_pg(best_a, p3_deadline, K_init=30, K_min=1,
                       n_iters_max=30000, bb_alpha_init=0.015)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [P3_topk_pg] t={time.monotonic()-t_start:.1f}s", flush=True)
    else:
        print(f"  Score: {sc3:.10f} (no improvement)", flush=True)

    # === Phase 4: Active-set KKT polish ===
    print(f"\n[P4] KKT Polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p4_deadline = min(deadline - 300.0, t_start + 1200.0)
    a4, sc4 = _kkt_polish(best_a, p4_deadline, tol_start=0.003, tol_end=0.00001, top_k=5)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [P4_kkt] t={time.monotonic()-t_start:.1f}s", flush=True)
    else:
        print(f"  Score: {sc4:.10f} (no improvement)", flush=True)

    # Re-run LSE after KKT polish if it helped
    if sc4 < best_score + 0.002 and time.monotonic() < deadline - 250:
        p4b_deadline = min(deadline - 250.0, time.monotonic() + 200.0)
        a4b = _lse_lbfgs(best_a, p4b_deadline, min_remaining=20.0, aggressive=False)
        sc4b = _score(a4b)
        if sc4b < best_score:
            best_a = a4b.copy()
            best_score = sc4b
            print(f"  NEW BEST: {best_score:.10f} [P4b_lbfgs_post] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 5: Second Top-K round (finer K) ===
    print(f"\n[P5] Top-K PG fine (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 150.0, time.monotonic() + 200.0)
        a5, sc5 = _topk_pg(best_a, p5_deadline, K_init=10, K_min=1,
                           n_iters_max=20000, bb_alpha_init=0.008)
        if sc5 < best_score:
            best_a = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5_topk_fine] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_targets = [16384, 20480, 12288]
    if n_parent not in n_targets:
        n_targets.append(n_parent)
    n_targets = sorted(set(n_targets), reverse=False)

    for n_try in n_targets:
        if time.monotonic() > deadline - 120.0:
            break
        if n_try == len(best_a):
            continue
        a_init_n = _resample(best_a, n_try)
        sc_init = _score(a_init_n)
        print(f"  n={n_try}: init score={sc_init:.10f}", flush=True)

        if sc_init >= best_score + 0.01:
            continue  # Not promising

        remaining_for_n = min(180.0, deadline - time.monotonic() - 80.0)
        if remaining_for_n < 30:
            continue

        # LSE optimize
        a_n = _lse_lbfgs(a_init_n, time.monotonic() + remaining_for_n * 0.5,
                         min_remaining=15.0, aggressive=False)
        sc_n = _score(a_n)

        # Top-K polish
        if time.monotonic() < deadline - 100:
            a_n, sc_n = _topk_pg(a_n, min(deadline - 90.0, time.monotonic() + remaining_for_n * 0.4),
                                 K_init=15, K_min=1, n_iters_max=10000, bb_alpha_init=0.01)

        if sc_n < best_score:
            best_a = a_n.copy()
            best_score = sc_n
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P6_multi_n] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 7: Perturbation restarts ===
    print(f"\n[P7] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        perturbations = [
            ("pert_01", _perturb(best_a, 0.01, rng)),
            ("pert_03", _perturb(best_a, 0.03, rng)),
            ("pert_05", _perturb(best_a, 0.05, rng)),
            ("sym_best", _symmetrize(best_a)),
        ]
        for label, a_p in perturbations:
            if time.monotonic() > deadline - 50.0:
                break
            remaining = deadline - time.monotonic() - 40.0
            if remaining < 20:
                break
            a_opt = _lse_lbfgs(a_p, time.monotonic() + remaining * 0.6,
                               min_remaining=10.0, aggressive=False)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [P7_{label}] t={time.monotonic()-t_start:.1f}s", flush=True)
            else:
                # Try top-K on the perturbed
                if remaining > 40:
                    a_tk, sc_tk = _topk_pg(a_opt, time.monotonic() + remaining * 0.35,
                                           K_init=10, K_min=1, n_iters_max=8000, bb_alpha_init=0.01)
                    if sc_tk < best_score:
                        best_a = a_tk.copy()
                        best_score = sc_tk
                        print(f"  NEW BEST: {best_score:.10f} [P7_{label}_tk] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 5.0:
        # KKT with very tight tolerance
        a_fin, sc_fin = _kkt_polish(best_a, deadline - 3.0,
                                    tol_start=0.0005, tol_end=0.000002,
                                    top_k=3, max_inner=100)
        if sc_fin < best
