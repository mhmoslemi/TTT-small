def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * 0.002)
    min_beta = max_c * 1e-10

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        max_iter = min(5000, max(20, int(remaining / (0.005 * scale * 20))))

        try:
            obj_fn = lambda z_: _smoobj(z_, beta, n)
            grad_fn = lambda z_: _smograd(z_, beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

            beta *= 0.1
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.35

        beta = min(beta, c_max_new * 0.001)

    elif kind == "asym_taper":

    elif kind == "gaussian":

def _resample(a, n_new):

    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick Newton polish from parent
    print(f"\n[Phase 1] Banded Newton polish (t={time.monotonic() - t_start:.1f}s)")
    t_phase1_end = t_start + 120
    a1, sc1 = _banded_newton(best_seq, min(t_phase1_end, deadline),
                             bandwidths=[30, 60, 100, 150], max_rounds_per_bw=30)
    update_best(a1, sc1, "newton_fast")

    # Phase 2: Adaptive Newton with growing active set
    print(f"\n[Phase 2] Adaptive Newton (t={time.monotonic() - t_start:.1f}s)")
    t_phase2_end = t_start + 300
    a2, sc2 = _adaptive_further_newton(best_seq, min(t_phase2_end, deadline))
    update_best(a2, sc2, "adaptive_newton")

    # Phase 3: L-BFGS with smoothing
    print(f"\n[Phase 3] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    t_phase3_end = t_start + 800
    a3 = _run_lbfgs(best_seq, min(t_phase3_end, deadline), min_time_remaining=300.0)
    update_best(a3, _score(a3), "lbfgs")

    # Phase 4: Newton after L-BFGS
    print(f"\n[Phase 4] Newton after L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    t_phase4_end = t_start + 1000
    a4, sc4 = _banded_newton(best_seq, min(t_phase4_end, deadline),
                             bandwidths=[50, 100, 150, 200, 300], max_rounds_per_bw=40)
    update_best(a4, sc4, "newton_after_lbfgs")

    # Phase 5: Another L-BFGS round (finer)
    print(f"\n[Phase 5] Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a5 = _run_lbfgs(best_seq, min(t_start + 1200, deadline), min_time_remaining=100.0)
        update_best(a5, _score(a5), "lbfgs_fine")

    # Phase 6: Final Newton
    print(f"\n[Phase 6] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        a6, sc6 = _adaptive_further_newton(best_seq, min(t_start + 1350, deadline))
        update_best(a6, sc6, "final_adaptive_newton")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else 16384
    for n_try in [8192, 32768]:
        if time.monotonic() > deadline - 80:
            break
        a_init = _resample(best_seq, n_try)
        a_opt, sc_opt = _banded_newton(a_init, min(deadline - 60, time.monotonic() + 60),
                                       bandwidths=[30, 60, 100], max_rounds_per_bw=30)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 8: Perturbation + Newton (diversification)
    print(f"\n[Phase 8] Diversification (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 20:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_dp, sc_dp = _banded_newton(a_pert, min(deadline - 10, time.monotonic() + 30),
                                         bandwidths=[50, 100], max_rounds_per_bw=20)
            update_best(a_dp, sc_dp, f"perturb_{sigma}")

    # Phase 9: Ultra-fine final polish
    print(f"\n[Phase 9] Ultra-fine polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _banded_newton(best_seq, deadline - 0.5,
                                       bandwidths=[40, 80, 120, 160, 200], max_rounds_per_bw=30)
        update_best(a_fin, sc_fin, "ultra_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(16384, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {ev:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
