"""
Combined SLP + Active-Set Newton optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
- Sequential Linear Programming (SLP) with trust-region damping for large-scale moves
- Improved active-set Newton with FFT-based Hessian-vector products

- Multi-phase time-budgeted search with diverse initializations

import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s
        C_new = best_new_C

        if C_new < best_C - 1e-10:
            best_C = C_new
            best_a = a.copy()
            no_improve = 0
            alpha = min(1.0, alpha * 1.1)
        else:
            no_improve += 1
            alpha *= 0.7

        if it % 10 == 0:
            print(f"    SLP iter {it}: C={C_new:.8f}, best={best_C:.8f}, alpha={best_alpha:.3f}")

    return best_a


def _newton_polish(a_init, deadline, max_rounds=200):
    """Active-set Newton with tight tolerances and robust line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with graduated tolerance
        active = None
        for tol_frac in [5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if int(active.sum()) >= 1 and int(active.sum()) <= 200:
                break
        if active is None or int(active.sum()) == 0:
            active = np.zeros(2*n-1, dtype=bool)
            active[int(np.argmax(c))] = True

        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Gradient
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Hessian-vector product via CG
        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo condition
            if c_max_try <= c_max - 0.1 * alpha * gn * 0.1 - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done:
            if no_improve > 8:
                break
        elif rnd % 20 == 19:
            print(f"    Newton round {rnd}: score={sc:.10f}, best={best_sc:.10f}")

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Adaptive L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-7 * c_max_new and sc_improved > 1e-13:
            beta *= 0.12
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _subgrad_search(a_init, deadline, max_steps=5000):
    """Mirror descent / subgradient method with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.03)
            else:
                patience += 1
                lr *= 0.8
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization: SLP + Newton + L-BFGS, time-budgeted."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick L-BFGS to get a good starting point ===
    print(f"\n[Phase 1] Quick L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = min(deadline - 600, t_start + 300)
    a_p1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=100.0)
    update_best(a_p1, _score(a_p1), "lbfgs_1")

    # SLP from the L-BFGS result
    if time.monotonic() < deadline - 500:
        n_slp = len(a_p1)
        if n_slp <= 32768:
            print(f"  Running SLP from phase 1 result (n={n_slp})")
            a_slp1 = _slp_optimize(a_p1, min(deadline - 400, time.monotonic() + 300),
                                   max_iters=40, max_active=300)
            update_best(a_slp1, _score(a_slp1), "slp_1")

    # Newton from best
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_np1, sc_np1 = _newton_polish(best_seq, min(deadline - 250, time.monotonic() + 120),
                                       max_rounds=60)
        update_best(a_np1, sc_np1, "newton_1")

    # === Phase 2: Main SLP optimization ===
    print(f"\n[Phase 2] Main SLP (t={time.monotonic()-t_start:.1f}s)")
    start_for_slp = best_seq if best_seq is not None else parent
    slp_deadline = min(deadline - 200, time.monotonic() + 600)
    a_slp2 = _slp_optimize(start_for_slp, slp_deadline, max_iters=80, max_active=500)
    update_best(a_slp2, _score(a_slp2), "slp_2")

    # Newton after SLP
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_np2, sc_np2 = _newton_polish(best_seq, min(deadline - 60, time.monotonic() + 150),
                                       max_rounds=100)
        update_best(a_np2, sc_np2, "newton_2")

    # === Phase 3: Subgradient diversification ===
    print(f"\n[Phase 3] Subgradient diversification (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        # Try perturbed versions
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 60:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            a_sg, sc_sg = _subgrad_search(a_pert, min(deadline - 50, time.monotonic() + 80),
                                          max_steps=2000)
            update_best(a_sg, sc_sg, f"subgrad_s{sigma}")
            if sc_sg < best_score + 0.001 and time.monotonic() < deadline - 40:
                a_npx, sc_npx = _newton_polish(a_sg, min(deadline - 30, time.monotonic() + 40),
                                               max_rounds=40)
                update_best(a_npx, sc_npx, f"newton_s{sigma}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_base = len(best_seq) if best_seq is not None else 16384
    for n_try in [8192, 32768]:
        if time.monotonic() > deadline - 40:
            break
        if n_try == n_base:
            continue
        a_init_n = _resample(best_seq, n_try)
        a_opt_n = _run_lbfgs(a_init_n, min(deadline - 30, time.monotonic() + 120),
                             min_time_remaining=20.0)
        sc_n = _score(a_opt_n)
        update_best(a_opt_n, sc_n, f"n_{n_try}")
        if sc_n < best_score + 0.005 and time.monotonic() < deadline - 25:
            a_np_n, sc_np_n = _newton_polish(a_opt_n, min(deadline - 15, time.monotonic() + 30),
                                             max_rounds=30)
            update_best(a_np_n, sc_np_n, f"n_{n_try}_np")

    # === Phase 5: Final intensive polish ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Alternate SLP + Newton in tight loop
        for round_idx in range(5):
            if time.monotonic() > deadline - 8:
                break
            half_time = (deadline - 8 - time.monotonic()) / (5 - round_idx)

            # SLP burst
            a_s = _slp_optimize(best_seq, time.monotonic() + half_time * 0.6,
                                max_iters=15, max_active=200, initial_alpha=0.3)
            sc_s = _score(a_s)
            if sc_s < best_score:
                update_best(a_s, sc_s, f"final_slp_{round_idx}")
                best_seq = a_s

            # Newton burst
            if time.monotonic() < deadline - 6:
                a_np, sc_np = _newton_polish(best_seq, min(deadline - 5, time.monotonic() + half_time * 0.4),
                                             max_rounds=30)
                update_best(a_np, sc_np, f"final_np_{round_idx}")

    # === Phase 6: Ultra-final ===
    print(f"\n[Phase 6] Ultra-final (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 1, max_rounds=50)
        update_best(a_fin, sc_fin, "ultra_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
