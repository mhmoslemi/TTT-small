```python
Optimized step function search using smooth-max gradient descent with
adaptive temperature, active-set equalization, and block coordinate refinement.
Targets n=8192 with focus on peak equalization and multi-restart strategy.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

# Global best tracking
_global_best_a = None
_global_best_score = np.inf

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Assumes sum(a) = 1. Returns 2*n*max(conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_max_grad(a, c, T):

    if s < 1e-300 or not np.isfinite(s):

        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    
    # Find significant p values (top peaks)
    # For efficiency, only use top m where p is significant
    threshold = 1e-10
    significant = p > threshold
    if not np.any(significant):
        idx = np.argmax(p)
        significant = np.zeros_like(p)
        significant[idx] = 1.0
    
    sig_idx = np.where(significant)[0]
    sig_p = p[sig_idx]
    
    g = np.zeros(n)
    for i in range(len(sig_idx)):
        k = sig_idx[i]
        pk = sig_p[i]
        if pk < threshold:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        g[lo:hi+1] += 2.0 * pk * a[k-hi:k-lo+1][::-1]
    
    return g

def _smooth_max_obj(a, T):
    """Compute smooth max of convolution."""
    c = _fft_convolve(a, a)
    c_shifted = T * (c - c.max())
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c.max()
    return (1.0/T) * np.log(s) + c.max()

def _active_set_equality(a, deadline, max_iter=1000, tol=0.002, max_active=10):
    """Equalize active peaks using linear algebra."""
    global _global_best_a, _global_best_score
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    rng = np.random.default_rng(123)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > max_active:
            sorted_a = active[np.argsort(c[active])[::-1][:max_active]]
            active = sorted_a
        
        K = len(active)
        
        # Build constraint matrix: for each active peak k,
        # we want the directional derivative to be -1 (decrease by 1)
        # d_k[j] = 2 * a[k-j] for valid j
        # Constraint: sum_j d_k[j] * delta[j] = -1 for each k
        # Plus: sum_j delta[j] = 0
        
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Add normalization constraint
        M_full = np.vstack([M, np.ones(1, n)])
        b = np.concatenate([-np.ones(K), [0.0]])
        
        # Minimum norm solution: delta = M_full^T * (M_full * M_full^T)^{-1} * b
        G = M_full @ M_full.T  # (K+1) x (K+1)
        try:
            G_inv = np.linalg.inv(G + 1e-10 * np.eye(G.shape[0]))
            delta = M_full.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            delta = np.linalg.lstsq(M_full, b, rcond=None)[0]
        
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-15:
            break
        
        delta = delta / delta_norm
        
        # Try step sizes
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a + alpha * delta
            if np.any(a_cand < -1e-10):
                continue
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                if best_score < _global_best_score:
                    _global_best_score = best_score
                    _global_best_a = best_a.copy()
                break
        
        if not improved:
            # Try a random perturbation to escape
            if it % 50 == 49:
                a = a + 0.001 * rng.standard_normal(n)
                a = np.maximum(0, a)
                a /= a.sum()
            else:
                break
    
    return best_a, best_score

def _block_coordinate_descent(a, deadline, block_size=64, max_rounds=100):
    """Block coordinate descent: optimize one block at a time."""
    global _global_best_a, _global_best_score
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    rng = np.random.default_rng(456)
    num_blocks = (n + block_size - 1) // block_size
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Shuffle block order
        block_order = rng.permutation(num_blocks)
        
        for bi in block_order:
            if time.monotonic() > deadline:
                break
            
            start = bi * block_size
            end = min(start + block_size, n)
            if start >= n:
                continue
            
            # Current block values
            block = a[start:end].copy()
            rest_sum = a.sum() - block.sum()
            
            # Optimize the block: minimize max_k c[k] 
            # by adjusting block values (subject to block >= 0, rest fixed)
            # Use simple grid/targeted search
            
            # First, identify which convolution lags are affected by this block
            # c[k] = sum_j a[j]*a[k-j]
            # If we change a[j] for j in [start, end), then c[k] changes for
            # k in [start, start + end - 1) ∪ ... (it's complex)
            # Simpler: just try to reduce the overall max
            
            # Compute current max
            c = _fft_convolve(a, a)
            c_max = float(c.max())
            k_max = int(np.argmax(c))
            
            # Try to reduce c[k_max] by adjusting the block
            # c[k_max] = sum_j a[j]*a[k_max-j]
            # The contribution from our block: for j in [start,end), a[j]*a[k_max-j]
            # and for j where k_max-j in [start,end), a[j]*a[k_max-j]
            
            # Simple approach: try scaling the block down and renormalizing
            best_block = block.copy()
            best_block_score = c_max
            
            for scale in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_cand = a.copy()
                a_cand[start:end] = block * scale
                if np.any(a_cand < 0):
                    continue
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max())
                if sc < best_block_score:
                    best_block_score = sc
                    best_block = block * scale
                    a_cand_best = a_cand.copy()
            
            if best_block_score < c_max * (1 - 1e-6):
                a = a_cand_best
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if best_score < _global_best_score:
                        _global_best_score = best_score
                        _global_best_a = best_a.copy()
    
    return best_a, best_score

def _subgradient_descent(a, deadline, max_iter=50000, top_k=5, lr_init=0.002):
    """Subgradient descent on the max convolution peak."""
    global _global_best_a, _global_best_score
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Get top-k peaks
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        # Subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space
        g = g - np.sum(g) * a
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Step
        step = lr * g / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            if best_score < _global_best_score:
                _global_best_score = best_score
                _global_best_a = best_a.copy()
            lr *= 0.9999
        else:
            lr *= 0.97
            if lr < 1e-10:
                break
    
    return best_a, best_score
```

```python
def _find_descent_direction(a, active_peaks):
    """Find a direction d that reduces all active peaks and preserves sum=1, a>=0."""
    n = len(a)
    K = len(active_peaks)
    
    # Build the constraint matrix
    # For each active peak k: M[i, j] = 2 * a[k - j] (for valid j)
    M = np.zeros((K + 1, n))
    for i, k in enumerate(active_peaks):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    M[K, :] = 1.0  # sum constraint
    
    # We want: M @ d < 0 for first K rows, M[K] @ d = 0
    # Equivalently: find d such that M[:K] @ d = -1, M[K] @ d = 0 (if feasible)
    # Minimum norm solution:
    b = np.concatenate([-np.ones(K), [0.0]])
    G = M @ M.T
    try:
        d = M.T @ np.linalg.solve(G + 1e-12 * np.eye(K+1), b)
    except:
        d = np.linalg.lstsq(M, b, rcond=None)[0]
    
    # Check feasibility: a + α*d >= 0
    # For the direction to be valid, we need: for all j with d[j] < 0, a[j] + α*d[j] >= 0
    # i.e., α <= a[j] / (-d[j]) for all j with d[j] < 0
    neg_mask = d < 0
    if np.any(neg_mask):
        max_alpha = np.min(a[neg_mask] / (-d[neg_mask]))
    else:
        max_alpha = 1.0  # no restriction
    
    if max_alpha <= 0:
        return None, 0.0
    
    # Check that this direction actually reduces the peaks
    # (it should by construction, but let's verify)
    for i, k in enumerate(active_peaks):
        val = M[i] @ d
        if val >= 0:
            return None, 0.0
    
    return d, max_alpha
```

```python
"""
Optimized step function search using adaptive smooth-max gradient descent with
harmonic temperature cooling, aggressive active-set equalization, and
multi-scale random restarts. Employs a switching strategy between global
(smooth) optimization and local (active-set) equalization phases, with
occasional structural perturbations to escape local minima.

import numpy as np
import time

_global_best_a = None
_global_best_score = np.inf


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_grad(a, c, T, top_m=20):
    """Gradient of smooth max of c w.r.t. a, using top-m significant weights."""
    n = len(a)
    c_shifted = T * (c - c.max())
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s

    # Use only top-m weights for efficiency
    if len(p) > top_m:
        idx = np.argpartition(p, -top_m)[-top_m:]
        p_masked = np.zeros_like(p)
        p_masked[idx] = p[idx]
        p_masked /= p_masked.sum()
        p = p_masked

    g = np.zeros(n)
    sig = p > 1e-8
    sig_idx = np.where(sig)[0]
    for i in range(len(sig_idx)):
        k = sig_idx[i]
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return g


def _adam_smooth_max(a_init, deadline, T_start, T_end, lr_start, lr_end,
                     max_iter=100000, eval_interval=25, cooling='poly',
                     momentum_reset_patience=15):
    """Adam optimizer for smooth-max in log-space with harmonic/poly cooling."""
    global _global_best_a, _global_best_score
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)

        if cooling == 'poly':
            T = T_start * (T_end / T_start) ** (frac ** 0.6)
        else:  # harmonic
            T = T_start / (1.0 + 100.0 * frac)
            T = max(T, T_end)

        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * lr_mult

        a = _softmax(z)
        c = _fft_convolve(a, a)
        g = _smooth_max_grad(a, c, T, top_m=15)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.5:
            update *= 1.5 / update_norm

        z = z - update
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                if best_score < _global_best_score:
                    _global_best_score = best_score
                    _global_best_a = best_a.copy()
            else:
                patience += 1

            if patience >= momentum_reset_patience:
                lr_mult *= 0.6
                patience = 0
                if lr_mult < 0.04:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.003 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.4
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
        if best_score < _global_best_score:
            _global_best_score = best_score
            _global_best_a = best_a.copy()
    return best_a, best_score


def _active_set_optimize(a, deadline, max_iter=3000, tol=0.002, max_active=10):
    """Iterative active-set equalization with adaptive step size."""
    global _global_best_a, _global_best_score
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(789)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        tol_val = c_max * (1.0 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > max_active:
            active = active[np.argsort(c[active])[::-1][:max_active]]

        K = len(active)
        M = np.zeros((K + 1, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        M[K, :] = 1.0

        b = np.concatenate([-\np.ones(K), [0.0]])
        G = M @ M.T
        try:
            G_inv = np.linalg.inv(G + 1e-11 * np.eye(G.shape[0]))
            d = M.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d /= d_norm

        neg_mask = d < 0
        if np.any(neg_mask):
            max_alpha = float(np.min(a[neg_mask] / (-d[neg_mask])))
        else:
            max_alpha = 1.0

        improved = False
        for frac_alpha in [0.8, 0.5, 0.3, 0.1, 0.05, 0.02]:
            alpha = max_alpha * frac_alpha
            if alpha < 1e-10:
                continue
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1.0 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                if best_score < _global_best_score:
                    _global_best_score = best_score
                    _global_best_a = best_a.copy()
                break

        if not improved:
            if it % 30 == 29:
                a = a + 0.002 * rng.standard_normal(n)
                a = np.maximum(0, a)
                a /= a.sum()
            else:
                break

    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=50000, top_k=5, lr_init=0.001):
    """Subgradient descent for the max convolution peak."""
    global _global_best_a, _global_best_score
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

            break

        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)
        g = g - np.sum(g) * a

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        step = lr * g / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1.0 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            if best_score < _global_best_score:
                _global_best_score = best_score
                _global_best_a = best_a.copy()
            lr *= 0.9998
        else:
            lr *= 0.97
            if lr < 1e-10:
                break

    return best_a, best_score


def _try_n_candidate(n, deadline, rng):
    """Try to find a good sequence of length n."""
    global _global_best_score
    # Initialize with a cos-modified uniform
    a = np.ones(n)
    if n >= 8:
        j = np.arange(n)
        a *= 1.0 + 0.3 * np.cos(2.0 * np.pi * j / n + rng.random() * 2 * np.pi)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-6:
        a = np.ones(n)
    a /= a.sum()

    sc = _score(a)
    print(f"    n={n} init: score={sc:.10f}")

    c0 = _fft_convolve(a, a)
    max_c0 = float(c0.max())

    # Phase A: Adam smooth-max
    sub_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    a_opt, sc_opt = _adam_smooth_max(
        a_init=a, deadline=sub_dl,
        T_start=max_c0 * 0.01, T_end=max_c0 * 0.0001,
        lr_start=0.04, lr_end=0.002,
        max_iter=100000, eval_interval=30, cooling='poly'
    )
    full_sc = _score(a_opt)
    print(f"    n={n} after Adam: score={full_sc:.10f}")

    # Phase B: Active set
    sub_dl2 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    a_as, sc_as = _active_set_optimize(a_opt, sub_dl2, max_iter=1000, tol=0.003, max_active=8)
    full_sc2 = _score(a_as)
    print(f"    n={n} after active-set: score={full_sc2:.10f}")

    # Phase C: Subgradient
    sub_dl3 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    a_sg, sc_sg = _subgradient_polish(a_as, sub_dl3, max_iter=20000, top_k=4, lr_init=0.001)
    full_sc3 = _score(a_sg)
    print(f"    n={n} after subgrad: score={full_sc3:.10f}")

    return full_sc3


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining smooth-max Adam, active-set equalization,
    subgradient polish, and multi-scale restarts. Uses harmonic and polynomial
    cooling schedules with adaptive momentum resets. Attempts both n=8192
    (from parent) and n=4096 (fresh) to explore different problem sizes.
    """
    global _global_best_a, _global_best_score

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Initialize from parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("bad parent")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using uniform init")

    n = len(a)
    _global_best_a = a.copy()
    _global_best_score = _score(a)
    print(f"START: n={n}, score={_global_best_score:.10f}")

    # ===== Phase 1: Primary Adam optimization (0 - 350s) =====
    p1_end = min(deadline, t_start + 350.0)
    print(f"\n=== Phase 1: Primary Adam ({p1_end - t_start:.0f}s budget) ===")

    c0 = _fft_convolve(_global_best_a, _global_best_a)
    max_c0 = float(c0.max())

    # Multiple temperature runs, each getting a portion of the time
    temp_configs = [
        (0.015, 0.04, 'poly'),
        (0.008, 0.03, 'poly'),
        (0.004, 0.02, 'harmonic'),
        (0.002, 0.012, 'harmonic'),
        (0.001, 0.006, 'poly'),
    ]
    n_configs = len(temp_configs)
    time_per = max(15.0, (p1_end - time.monotonic()) / n_configs)

    for i, (tf, lr_s, cool) in enumerate(temp_configs):
        if time.monotonic() > p1_end - 5:
            break
        sub_dl = min(p1_end - 2, time.monotonic() + time_per)
        a_opt, sc = _adam_smooth_max(
            a_init=_global_best_a.copy(), deadline=sub_dl,
            T_start=max_c0 * tf, T_end=max_c0 * tf * 0.005,
            lr_start=lr_s, lr_end=lr_s * 0.1,
            max_iter=150000, eval_interval=20, cooling=cool
        )
        full_sc = _score(a_opt)
        print(f"  Config[{i}] T_f={tf}, cool={cool}: score={full_sc:.10f}")

    print(f"  Phase 1 best: {_global_best_score:.10f}")

    # ===== Phase 2: Active-set equalization (350 - 550s) =====
    p2_end = min(deadline, t_start + 550.0)
    if time.monotonic() < p2_end:
        print(f"\n=== Phase 2: Active-set ({p2_end - time.monotonic():.0f}s) ===")
        for tol_v, max_act in [(0.002, 8), (0.001, 6), (0.005, 12), (0.0005, 10)]:
            if time.monotonic() > p2_end - 10:
                break
            sub_dl = min(p2_end - 2, time.monotonic() + 40)
            a_as, sc_as = _active_set_optimize(
                _global_best_a.copy(), sub_dl,
                max_iter=1500, tol=tol_v, max_active=max_act
            )
            full_sc = _score(a_as)
            print(f"  tol={tol_v}, K={max_act}: score={full_sc:.10f}")

    # ===== Phase 3: Subgradient polish (550 - 700s) =====
    p3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < p3_end:
        print(f"\n=== Phase 3: Subgradient ({p3_end - time.monotonic():.0f}s) ===")
        for tk in [3, 5, 7]:
            if time.monotonic() > p3_end - 5:
                break
            sub_dl = min(p3_end - 2, time.monotonic() + 40)
            a_sg, sc_sg = _subgradient_polish(
                _global_best_a.copy(), sub_dl,
                max_iter=30000, top_k=tk, lr_init=0.0008
            )
            full_sc = _score(a_sg)
            print(f"  top_k={tk}: score={full_sc:.10f}")

    # ===== Phase 4: Low-T refinement (700 - 900s) =====
    p4_end = min(deadline, t_start + 900.0)
    if time.monotonic() < p4_end:
        print(f"\n=== Phase 4: Low-T refinement ({p4_end - time.monotonic():.0f}s) ===")
        c_best = _fft_convolve(_global_best_a, _global_best_a)
        max_cb = float(c_best.max())
        for T_f, lr_v in [(0.0005, 0.004), (0.0003, 0.003), (0.0002, 0.002)]:
            if time.monotonic() > p4_end - 5:
                break
            sub_dl = min(p4_end - 2, time.monotonic() + 55)
            a_opt, sc = _adam_smooth_max(
                a_init=_global_best_a.copy(), deadline=sub_dl,
                T_start=max_cb * T_f, T_end=max_cb * T_f * 0.01,
                lr_start=lr_v, lr_end=lr_v * 0.15,
                max_iter=80000, eval_interval=15, cooling='harmonic'
            )
            full_sc = _score(a_opt)
            print(f"  T_f={T_f}: score={full_sc:.10f}")

    # ===== Phase 5: Random restarts with diverse perturbations (900 - 1250s) =====
    p5_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < p5_end:
        print(f"\n=== Phase 5: Restarts ({p5_end - time.monotonic():.0f}s) ===")
        n_restarts = 10
        time_per_r = max(15.0, (p5_end - time.monotonic()) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > p5_end - time_per_r:
                break

            strategy = r % 5
            if strategy == 0:
                # Log-space Gaussian
                sigma = 0.008 * (1.0 + 0.3 * r)
                z = np.log(np.maximum(_global_best_a, 1e-15))
                z += sigma * rng.standard_normal(n)
                z = z - z.mean()
                a_pert = _softmax(z)
            elif strategy == 1:
                # Sinusoidal modulation
                j = np.arange(n)
                mod = 1.0 + 0.2 * np.sin(2.0 * np.pi * rng.integers(1, 8) * j / n + rng.random() * 6.28)
                a_pert = _global_best_a * mod
                a_pert = np.maximum(0, a_pert)
                a_pert /= a_pert.sum()
            elif strategy == 2:
                # Block permutation
                a_pert = _global_best_a.copy()
                bsz = n // 8
                blocks = [a_pert[i * bsz:(i + 1) * bsz] for i in range(8)]
                perm = rng.permutation(8)
                for i in range(8):
                    a_pert[i * bsz:(i + 1) * bsz] = blocks[perm[i]]
                a_pert /= a_pert.sum()
            elif strategy == 3:
                # Sparse perturbation (zero out random 10%)
                a_pert = _global_best_a.copy()
                mask = rng.random(n) < 0.1
                a_pert[mask] = 0
                s = a_pert.sum()
                if s > 0.01:
                    a_pert /= s
                else:
                    a_pert = _global_best_a.copy()
            else:
                # Mirror + noise
                a_pert = np.concatenate([_global_best_a[:n//2], _global_best_a[:n//2][::-1]])
                if n % 2 == 1:
                    a_pert = np.concatenate([a_pert, [_global_best_a[n//2]]])
                a_pert += 0.002 * rng.standard_normal(len(a_pert))
                a_pert = np.maximum(0, a_pert[:n])
                s = a_pert.sum()
                if s > 0.01:
                    a_pert /= s
                else:
                    a_pert = _global_best_a.copy()

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())

            sub_dl = min(p5_end - 3, time.monotonic() + time_per_r * 0.6)
            a_opt, sc = _adam_smooth_max(
                a_init=a_pert, deadline=sub_dl,
                T_start=max_cp * 0.003, T_end=max_cp * 0.0001,
                lr_start=0.02, lr_end=0.001,
                max_iter=40000, eval_interval=25, cooling='poly'
            )
            full_sc = _score(a_opt)

            # Quick active-set on the result
            if full_sc < _global_best_score and time.monotonic() < p5_end - 10:
                sub_dl2 = min(p5_end - 2, time.monotonic() + 10)
                a_as, sc_as = _active_set_optimize(
                    a_opt, sub_dl2, max_iter=300, tol=0.003, max_active=6
                )
                full_sc2 = _score(a_as)
                if full_sc2 < full_sc:
                    full_sc = full_sc2

            print(f"  R[{r}] strat={strategy}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ===== Phase 6: Try different n (1250 - 1450s) =====
    p6_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < p6_end - 30:
        print(f"\n=== Phase 6: Try n=4096 ===")
        sub_dl = min(p6_end - 5, time.monotonic() + min(120.0, p6_end - time.monotonic() - 10))
        sc_4096 = _try_n_candidate(4096, sub_dl, rng)
        print(f"  n=4096 best: {sc_4096:.10f}")

    # ===== Phase 7: Final ultra-fine polish (1450 - deadline) =====
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 7: Final polish ===")

        # Tight active set
        sub_dl = min(deadline - 15, time.monotonic() + 60)
        a_as, sc_as = _active_set_optimize(
            _global_best_a.copy(), sub_dl,
            max_iter=2000, tol=0.0005, max_active=15
        )
        full_sc = _score(a_as)
        print(f"  Tight active-set: score={full_sc:.10f}")

        # Subgradient tiny steps
        if time.monotonic() < deadline - 10:
            sub_dl2 = min(deadline - 5, time.monotonic() + 40)
            a_sg, sc_sg = _subgradient_polish(
                _global_best_a.copy(), sub_dl2,
                max_iter=15000, top_k=3, lr_init=0.0003
            )
            full_sc2 = _score(a_sg)
            print(f"  Fine subgrad: score={full_sc2:.10f}")

        # One more ultra-low-T Adam
        if time.monotonic() < deadline - 5:
            c_best = _fft_convolve(_global_best_a, _global_best_a)
            max_cb = float(c_best.max())
            sub_dl3 = min(deadline - 2, time.monotonic() + 30)
            a_opt, sc = _adam_smooth_max(
                a_init=_global_best_a.copy(), deadline=sub_dl3,
                T_start=max_cb * 0.0001, T_end=max_cb * 0.00001,
                lr_start=0.002, lr_end=0.0005,
                max_iter=30000, eval_interval=10, cooling='harmonic'
            )
            full_sc3 = _score(a_opt)
            print(f"  Ultra-low-T Adam: score={full_sc3:.10f}")

    # Final output
    final = np.clip(_global_best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
