"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Active-set-reduced L-BFGS with adaptive beta
2. Improved active-set Newton with KKT-based active set tracking
3. Spectral equalization (FFT-based peak reduction)
4. Multi-n exploration with interpolation
5. Weighted deflation using actual peak contributions

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj_reduced(z, beta, n, K):
    """Log-sum-exp over only top K convolution values (plus a floor term)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Use top K values for the log-sum-exp
    # Add a small floor to avoid ignoring the rest entirely
    # Weight: top K get weight 1, rest get weight 0
    # This is an approximation of the full log-sum-exp
    
    # Actually, let's use a different approach:
    # f = c_max + beta * log(sum_{k in top_K} exp((c[k]-c_max)/beta))
    # This is exact if K = 2n-1, and an approximation otherwise
    
    # Better: use the active set approach
    # f = sum_{k in S} c[k] / |S| where S is the active set
    # But that's not smooth. Let's stick with log-sum-exp but restricted.
    
    # Simplest improvement: use top K + a constant offset for the rest
    top_k_vals = np.sort(c)[::-1][:K]
    shifted = (top_k_vals - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    # The "rest" contribution is approximately (2n-1-K) * c_mean_rest
    # where c_mean_rest is the mean of the non-top values
    rest_mean = (c.sum() - top_k_vals.sum()) / max(1, (2*n-1) - K)
    rest_log = np.log(max(1, (2*n-1) - K)) + rest_mean / beta
    # Combine: log(sum_top + sum_rest) ≈ log(sum_top) + log(1 + exp(rest_log - log(sum_top)))
    # For simplicity, just use the top K:
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lsobj_active(z, beta, n, active_set):

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_active = c[active_set]
    c_max_act = float(c_active.max())

        return 2.0 * n * c_max_act
    return 2.0 * n * (c_max_act + beta * np.log(s))

def _lsgrad_active(z, beta, n, active_set):
    """Gradient of active-set-restricted log-sum-exp."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_active = c[active_set]
    c_max_act = float(c_active.max())
    shifted = (c_active - c_max_act) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        w_act = np.zeros_like(c_active)
        w_act[int(np.argmax(c_active))] = 1.0
    else:
        w_act = e / s

    w = np.zeros(2 * n - 1)
    w[active_set] = w_act
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _spectral_equalize(a, deadline, n_iters=200, eta_init=0.1):
    """FFT-based spectral equalization to flatten autocorrelation."""
    n = len(a)
    a_work = a.copy().astype(np.float64)
    a_work = np.maximum(a_work, 0)
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work / s
    best_a = a_work.copy()
    best_sc = _score(a_work)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a_work, 2 * n)
        power = np.abs(A) ** 2
        mean_power = power.mean()
        
        if mean_power < 1e-15:
            break
        
        # Damping: reduce power where it's high, boost where it's low
        # But we need to maintain |A|^2 as a valid power spectrum
        eta = eta_init * (0.97 ** it)
        
        # Geometric mean approach: make power more uniform
        # log(power) -> reduce variance
        log_power = np.log(np.maximum(power, 1e-30))
        log_power_mean = log_power.mean()
        
        # Damp the deviations
        new_log_power = log_power - eta * (log_power - log_power_mean)
        new_power = np.exp(new_log_power)
        
        # Scale back to original total power
        new_power = new_power * (power.sum() / new_power.sum())
        
        # Take square root to get new amplitude (keep phase)
        phase = np.angle(A)
        new_amplitude = np.sqrt(new_power)
        A_new = new_amplitude * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a_work = a_new.copy()
        else:
            # Try smaller eta
            eta_half = eta * 0.5
            new_log_power2 = log_power - eta_half * (log_power - log_power_mean)
            new_power2 = np.exp(new_log_power2)
            new_power2 = new_power2 * (power.sum() / new_power2.sum())
            new_amplitude2 = np.sqrt(new_power2)
            A_new2 = new_amplitude2 * np.exp(1j * phase)
            a_new2 = np.fft.irfft(A_new2, 2 * n)[:n]
            a_new2 = np.maximum(a_new2, 0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                sc_new2 = _score(a_new2)
                if sc_new2 < best_sc - 1e-14:
                    best_sc = sc_new2
                    best_a = a_new2.copy()
                    a_work = a_new2.copy()
                else:
                    break
    
    return best_a

def _active_set_qp_step(a, active_set, deadline, max_iters=50):
    """Solve the QP: min sum_{k in S} c[k] s.t. a >= 0, sum(a) = 1
    using projected gradient with FFT-based Hessian-vector products."""
    n = len(a)
    a_work = a.copy().astype(np.float64)
    a_work = np.maximum(a_work, 0)
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work / s
    best_a = a_work.copy()
    best_sc = _score(a_work)
    
    # Precompute: H*v = sum_{k in S} (d c[k]/da) * ... 
    # Actually, let's use: gradient of sum_{k in S} c[k] = 2 * corr(a, S_vec, n)
    # where S_vec is the indicator of the active set
    S_vec = np.zeros(2 * n - 1)
    S_vec[active_set] = 1.0
    
    grad = 2.0 * _corr_a_A(a_work, S_vec, n)
    
    # Projected gradient descent on the QP
    step_size = 0.1 / (1.0 + np.max(np.abs(grad)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Gradient step
        a_new = a_work - step_size * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a_work = a_new.copy()
            # Update gradient
            grad = 2.0 * _corr_a_A(a_work, S_vec, n)
            # Adaptive step size
            step_size = 0.1 / (1.0 + np.max(np.abs(grad)))
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    return best_a

def _active_set_newton_v2(a_init, deadline, max_iter=500):

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf

    best_a = a.copy()

        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < best_c_max - 1e-15:
            best_c_max = c_max
            best_a = a.copy()
            best_sc = _score(a)
        
        # Try to find a Newton direction using the tightest viable active set
        improved = False
        
        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 3 or m > 8 * n:
                continue
            
            A_vec = active.astype(np.float64)
            
            # Gradient of the "active" objective: (1/m) * sum_{k in S} c[k]
            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            
            # Hessian-vector product: H*v = (2/m) * corr(v, A_vec, n) + mu*v
            mu = 1e-8 * (c_max / max(m, 1)) + 1e-10
            
            def hess_vp(v, A_vec=A_vec, m=m, mu=mu, n=n):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v
            
            # Solve H*u = g_t using CG
            u = _cg_solve(hess_vp, g_t, max_cg=150)
            # Solve H*w = 1 (for the normalization constraint)
            w = _cg_solve(hess_vp, np.ones(n), max_cg=150)
            w_sum = float(w.sum())
            
            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w
            
            dx = dx - np.mean(dx)
            
            # Line search
            neg = dx < -1e-12
            if neg.any():

            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)
            
            t = min(1.0, t_max)
            for ls in range(30):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= c_max * (1 - 1e-12):
                    a = a_try
                    improved = True
                    break
                t *= 0.5
            
            if improved:
                break
        
        if not improved:
            # Gradient step fallback
            g_all = (2.0 / n) * _corr_a_A(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            t = 0.05
            accepted = False
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < c_max * (1 - 1e-12):
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break
        
        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break
    
    final_sc = _score(best_a)
    return best_a, final_sc

def _weighted_deflation(a, deadline, n_attempts=50):
    """Improved deflation using weighted contributions to peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        n_top = max(5, min(len(c), int(0.02 * len(c))))
        top_positions = np.argsort(c)[::-1][:n_top]
        
        # Compute contribution of each a[i] to the top peaks
        # contrib[i] = sum_{k in top} a[i] * a[k-i] / c_max
        contrib = np.zeros(n)
        for k_pos in top_positions:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = k_pos - i
                contrib[i] += a[i] * a[j]
        
        # Reduce the elements with highest contribution
        # Use soft reduction: a[i] *= (1 - eta * contrib[i] / max_contrib)
        max_contrib = contrib.max()
        if max_contrib < 1e-15:
            break
        
        eta = 0.01 * (0.95 ** attempt)  # decrease over time
        a_new = a * (1.0 - eta * contrib / max_contrib)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try half eta
            a_half = a * (1.0 - 0.5 * eta * contrib / max_contrib)
            a_half = np.maximum(a_half, 0)
            s_half = a_half.sum()
            if s_half > 1e-15:
                a_half = a_half / s_half
                sc_half = _score(a_half)
                if sc_half < best_sc - 1e-13:
                    best_sc = sc_half
                    best_a = a_half.copy()
                else:
                    break
    
    return best_a, best_sc

def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta based on active set gap."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    # Adaptive: start with beta proportional to the gap between top peaks
    if len(c_sorted) > 5:
        gap = c_sorted[0] - c_sorted[5]
        base_beta = max(gap, max_c * 0.05)
    else:
        base_beta = max_c * 0.05
    
    # More stages, with more aggressive reduction
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                
                # Update beta schedule based on new gap
                if i < len(betas) - 1:
                    c_new = _full_convolve(a_new, a_new)
                    c_s = np.sort(c_new)[::-1]
                    if len(c_s) > 5:
                        new_gap = c_s[0] - c_s[5]
                        # If gap has reduced, we can be more aggressive
                        if new_gap < betas[i+1] * 0.1:
                            betas[i+1:] = new_gap * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                                                              0.002, 0.001, 0.0005, 0.0002, 0.0001])
        except Exception:
            pass
    
    return current_a

def _active_set_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with active-set-restricted objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Determine active set size
    tol = 0.01
    active = c_cur >= (max_c - tol * max_c)
    K = max(int(active.sum()), 10)
    K = min(K, 4 * n)
    
    active_indices = np.where(active)[0]
    if len(active_indices) < 5:
        active_indices = np.argsort(c_cur)[::-1][:max(10, K)]
    
    beta_init = max_c * 0.02
    betas = beta_init * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001])
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Update active set periodically
        if i > 0 and i % 3 == 0:
            c_check = _full_convolve(current_a, current_a)
            mc = float(c_check.max())
            active = c_check >= (mc - tol * mc)
            K = max(int(active.sum()), 10)
            K = min(K, 4 * n)
            active_indices = np.where(active)[0]
            if len(active_indices) < 5:
                active_indices = np.argsort(c_check)[::-1][:K]
        
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj_active(z_local, beta, n, active_indices),
                z,
                jac=lambda z_local: _lsgrad_active(z_local, beta, n, active_indices),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a
