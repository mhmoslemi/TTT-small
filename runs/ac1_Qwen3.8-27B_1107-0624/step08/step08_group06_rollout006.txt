So we want to minimize `max_k c[k]` subject to `sum(a) = 1` and `a >= 0` (WLOG, by scale invariance).

The minimum possible value of `max_k c[k]` is `1/(2n-1)` (by averaging, since sum(c) = 1).

For n = 16384, this gives a lower bound of `1/32767 ≈ 3.05e-5`, and a score of `2 * 16384 / 32767 ≈ 1.0003`.

So the theoretical minimum is about 1, and we're at 1.5. There's still room for improvement.

But achieving the theoretical minimum would require the convolution to be perfectly uniform, which is impossible for non-negative sequences (as I discussed earlier, the DC component of the spectrum is fixed).

Let me think about what's achievable. The constraint is that `|A[0]|^2 = 1` (DC component), and we want `|A[k]|^2` to be as small as possible for k ≠ 0. By Parseval: `sum_k |A[k]|^2 = (2n-1) * sum(a^2)`.

If `sum(a^2) = S` (the "energy"), then `sum_{k≠0} |A[k]|^2 = (2n-1)S - 1`.

The minimum of `max_{k≠0} |A[k]|^2` is `((2n-1)S - 1) / (2n-2)` (by averaging over the non-DC bins).

The peak of the convolution is related to the maximum of the inverse DFT of `|A|^2`, which is bounded by `max_k |A[k]|^2` (by the triangle inequality, in the worst case).

So a lower bound on `max(c)` is approximately `1/(2n-1) * max_k |A[k]|^2 = 1/(2n-1) * max(1, ((2n-1)S - 1)/(2n-2))`.

For uniform a: S = 1/n, so `((2n-1)/n - 1)/(2n-2) = (2n-1-n)/(n(2n-2)) = (n-1)/(n(2n-2)) ≈ 1/(2n)`. And `1/(2n-1) * 1 = 1/(2n-1)`. So the bound is `1/(2n-1)`, which matches the actual value for uniform.

For a more concentrated a (larger S), the non-DC spectral energy is spread over more bins but each bin is larger. This might increase the peak of the convolution.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive active-set Newton with CG inner solver and multi-tolerance schedule
2. LP-based direction finding for robust deflation
3. L-BFGS with log-sum-exp smoothing (adaptive beta)
4. Multi-n exploration with spectral initialization
5. Strategic perturb-and-restart for escaping local minima
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = float(np.dot(r, r))
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = float(np.dot(p, Ap))
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _active_set_newton(a_init, deadline, max_iter=500, tol_schedule=None):
    """Adaptive active-set Newton with scheduled tolerance decrease."""
    if tol_schedule is None:

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    start_c_max = float(_full_convolve(a, a).max())

    tol_idx = 0
    stalled_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        if c_max < 1e-15:
            break

        tol_a = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol_a * c_max)
        if not active.any():
            active[int(np.argmax(c))] = True
        m = int(active.sum())

        if m < 2:
            active[int(np.argmax(c))] = True
            m = 1
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
            continue

        if m > 8 * n:
            top_idx = np.argsort(c)[::-1][:min(8 * n, len(c))]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = int(active.sum())

        A_vec = active.astype(np.float64)

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
            stalled_count += 1
            if stalled_count > 3:
                break
            continue

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-10

        def hess_vp(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        u = _cg_solve(hess_vp, g_t, max_cg=250, tol=1e-10)
        w = _cg_solve(hess_vp, np.ones(n), max_cg=150, tol=1e-10)
        w_sum = float(w.sum())

        if abs(w_sum) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / w_sum
            dx = -u + lam * w

        dx = dx - np.mean(dx)

        neg = dx < -1e-12
        if neg.any():
            t_max = float(np.min(a[neg] / (-dx[neg]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)
        t = min(1.0, t_max)

        f_old = c_max
        accepted = False

        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())

            if f_try <= f_old * (1.0 - 1e-10 * t) or (f_try <= f_old + 1e-12 and ls >= 30):
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try a pure gradient step as fallback
            t_g = 0.05
            for ls in range(20):
                a_try = a - t_g * g_t
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    t_g *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-12:
                    a = a_try
                    accepted = True
                    break
                t_g *= 0.5

            if not accepted:
                tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)

                if stalled_count > 5:

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

def _lp_direction(a, M_target, deadline, max_lp_iters=30):
    """LP-based direction finding: find g >= 0 such that moving a toward g
    doesn't increase peak beyond M_target."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    # Precompute for the LP: constraint matrix is conv(a, g) which is linear in g
    # (a * g)[k] = sum_i a[i] * g[k-i]
    # We want: (1-t)^2 * (a*a)[k] + 2*(1-t)*t * (a*g)[k] <= M_target
    # => (a*g)[k] <= (M_target - (1-t)^2 * (a*a)[k]) / (2*(1-t)*t)

    t_step = 0.01
    c_a = _full_convolve(a, a)
    c_max = float(c_a.max())

    for lp_iter in range(max_lp_iters):
        if time.monotonic() > deadline:
            break

        M_target = c_max * (1.0 - 0.001)  # aim to reduce peak slightly

        rhs_vals = (M_target - (1.0 - t_step) ** 2 * c_a) / (2.0 * (1.0 - t_step) * t_step)

        # Build constraint matrix A_ub: each row k gives sum_i a[i] * g[k-i] <= rhs_vals[k]
        # This is a banded matrix. For efficiency, only include rows where rhs > 0
        # (if rhs <= 0, the constraint is infeasible for non-negative g, so we skip)
        valid = rhs_vals > 0
        if not valid.any():
            break

        # For the LP, we need the constraint matrix.
        # (a*g)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * g[k-i]
        # Let j = k-i, then j ranges from max(0, k-n+1) to min(k, n-1)
        # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
        # So the k-th row of the constraint matrix has (A_ub[k,j]) = a[k-j] if 0<=k-j<=n-1

        # This is a big matrix (2n-1 x n). For n=16384, that's ~32767 x 16384.
        # Too big for standard LP solvers. Use a sparse approach or skip.
        # Instead, let's use a simpler approach: just solve a smaller LP
        # by restricting to the most constraining rows.

        n_constraints = min(200, int(valid.sum()))
        # Pick the most constraining rows (smallest rhs_vals among valid ones)
        valid_idx = np.where(valid)[0]
        if len(valid_idx) > n_constraints:
            [~, local_order] = np.sort(rhs_vals[valid_idx]), None
            # Take the smallest rhs_vals (most constraining)
            sorted_valid = valid_idx[np.argsort(rhs_vals[valid_idx])]
            tight_idx = sorted_valid[:n_constraints]
        else:
            tight_idx = valid_idx

        # Build constraint matrix for just these tight constraints
        A_ub = np.zeros((len(tight_idx), n))
        for row_i, k in enumerate(tight_idx):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                A_ub[row_i, j] = a[i]
        b_ub = rhs_vals[tight_idx]

        # Maximize sum(g) subject to A_ub @ g <= b_ub, g >= 0
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                          method='highs', options={'maxiter': 5000, 'presolve': True})
            if res.status != 0:
                break
            g = res.x
            s_g = float(g.sum())
            if s_g < 1e-10:
                break

            g = g / s_g
            a_try = (1.0 - t_step) * a + t_step * g
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                c_a = _full_convolve(a_try, a_try)
                c_max = float(c_a.max())
            else:
                # Try smaller step
                t_step *= 0.5
                if t_step < 0.001:
                    break
        except Exception:
            break

    return best_a, best_sc


def _fista_subgrad(a_init, deadline, n_steps=20000):
    """Nesterov-accelerated subgradient method for max-conv minimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    step_counter = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())

        # Use all peaks above 95% of max as subgradient
        tol = 0.02 * c_max_y
        active = c_y >= (c_max_y - tol)
        if not active.any():
            active[int(np.argmax(c_y))] = True
        w = np.zeros_like(c_y)
        w[active] = 1.0 / active.sum()

        grad = 2.0 * _corr_a_A(y, w, n)
        g_t = grad - np.mean(grad)

        # Adaptive step size
        step_size = 0.5 / (1.0 + 0.1 * step)

        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new > 1e-15:
            a_new = a_new / s_new

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = float(y.sum())
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t_k = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation for log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    beta_ratios = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,
                   0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, ratio in enumerate(beta_ratios):
        beta = base_beta * ratio
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # If no improvement, try smaller beta directly
                if i >= len(beta_ratios) // 2:
                    break
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shaped":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = n // 3
        a[:block] = 1.0
        a[2 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "block3":
        a = np.zeros(n)
        block = n // 5
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[4 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _deflation_peaks(a, deadline, n_rounds=20):
    """Iteratively deflate the top convolution peaks by reducing contributing elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks
        n_top = max(5, min(len(c), int(0.03 * len(c))))
        top_k = np.argsort(c)[::-1][:min(n_top, 50)]

        # For each top peak, find the max contributing pair and reduce it
        a_new = a.copy()
        reduction = 0.998 - 0.001 * r  # gradually less aggressive

        for k_pos in top_k[:30]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            if i_lo > i_hi:
                continue
            # Find max product pair
            idx = np.arange(i_lo, i_hi + 1)
            prods = a_new[idx] * a_new[k_pos - idx]
            max_i = idx[np.argmax(prods)]
            j = k_pos - max_i
            if max_i == j:
                a_new[max_i] *= np.sqrt(reduction)
            else:
                # Reduce the larger one
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= reduction
                else:
                    a_new[j] *= reduction

        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining active-set Newton, LP direction finding,
    L-BFGS polish, and strategic restarts to minimize the convolution peak ratio.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Active-set Newton (main optimization)
    print(f"\n[Phase 1] Active-set Newton from parent (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = min(deadline - 600.0, t_start + 720.0)
    a_newton, sc_newton = _active_set_newton(
        parent, p1_deadline, max_iter=600,
        tol_schedule=[0.05, 0.03, 0.015, 0.01, 0.005, 0.003, 0.0015, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]
    )
    update_best(a_newton, sc_newton, "newton_1")

    # Phase 2: L-BFGS polish
    print(f"\n[Phase 2] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p2_deadline = min(deadline - 300.0, time.monotonic() + 360.0)
        a_lbfgs = _run_lbfgs(best_seq, p2_deadline, min_time_remaining=180.0)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_1")

    # Phase 3: Second Newton pass (tighter tolerances)
    print(f"\n[Phase 3] Newton pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        p3_dl = min(deadline - 200, time.monotonic() + 180)
        a_n2, sc_n2 = _active_set_newton(
            best_seq, p3_dl, max_iter=400,
            tol_schedule=[0.005, 0.003, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003]
        )
        update_best(a_n2, sc_n2, "newton_2")

    # Phase 4: Deflation + L-BFGS
    print(f"\n[Phase 4] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        def_dl = min(deadline - 150, time.monotonic() + 60)
        a_def, sc_def = _deflation_peaks(best_seq, def_dl, n_rounds=30)
        update_best(a_def, sc_def, "deflation")
        if sc_def < best_score:
            a_def_lb = _run_lbfgs(a_def, min(deadline - 100, time.monotonic() + 90),
                                  min_time_remaining=60.0)
            update_best(a_def_lb, _score(a_def_lb), "deflation_lbfgs")

    # Phase 5: LP direction finding
    print(f"\n[Phase 5] LP direction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        lp_dl = min(deadline - 120, time.monotonic() + 90)
        n_lp = len(best_seq)
        # Only do LP if n is manageable
        if n_lp <= 16384:
            a_lp, sc_lp = _lp_direction(best_seq, None, lp_dl, max_lp_iters=20)
            update_best(a_lp, sc_lp, "lp_dir")

    # Phase 6: FISTA subgradient
    print(f"\n[Phase 6] FISTA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        fista_dl = min(deadline - 100, time.monotonic() + 60)
        a_fista = _fista_subgrad(best_seq, fista_dl, n_steps=5000)
        sc_fista = _score(a_fista)
        update_best(a_fista, sc_fista, "fista")
        if sc_fista < best_score:
            a_fista_n, sc_fn = _active_set_newton(
                a_fista, min(deadline - 50, time.monotonic() + 30),
                max_iter=100, tol_schedule=[0.01, 0.005, 0.002, 0.001]
            )
            update_best(a_fista_n, sc_fn, "fista_newton")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates and n_parent <= 32768:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue

        # Initialize by interpolation from best
        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        sc_init = _score(a_init)
        # Compare with a profile
        a_prof = _make_profile(n_try, "tapered")
        sc_prof = _score(a_prof)
        if sc_prof < sc_init:
            a_init = a_prof
            sc_init = sc_prof

        # Quick L-BFGS
        if time.monotonic() < deadline - 40:
            a_opt = _run_lbfgs(a_init, min(deadline - 40, time.monotonic() + 60),
                               min_time_remaining=30.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 8: Perturb and restart
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        perturbations = [
            ("small", 0.01),
            ("medium", 0.03),
            ("large", 0.08),
        ]
        for label, sigma in perturbations:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _active_set_newton(
                p_start, min(deadline - 15, time.monotonic() + 30),
                max_iter=100, tol_schedule=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
            )
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 9: Final polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        fin_dl = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(
            best_seq, fin_dl, max_iter=200,
            tol_schedule=[0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00002]
        )
        update_best(a_fin, sc_fin, "final_newton")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 6 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(100, int(remaining / 0.02)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
