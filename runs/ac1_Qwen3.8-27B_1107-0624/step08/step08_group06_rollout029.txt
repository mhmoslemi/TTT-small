import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    """Compute cross-correlation: result[i] = sum_k A_vec[k] * a[k-i] for i=0..n-1"""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    wc = _corr_a_A(a, w, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (wc - W)


def _psobj(z, p, n):
    """Power-sum objective: 2*n * (sum(c^p))^(1/p)"""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0)
    if p == 1:
        return 2.0 * n * c.sum()
    cp = c ** p
    return 2.0 * n * (cp.sum()) ** (1.0 / p)


def _psgrad(z, p, n):
    """Gradient of power-sum objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    if p == 1:
        w = np.ones_like(c) / len(c)
    else:
        cp = c_pos ** (p - 1)
        s = cp.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = cp / s
    wc = _corr_a_A(a, w, n)
    W = float(np.dot(w, c_pos))
    return 4.0 * n * a * (wc - W)


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _deflation_local(a, deadline, n_attempts=40):
    """Targeted deflation: for each peak, solve 2-variable local optimization."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        peak_pos = int(np.argmax(c))

        # Find the pair (i, j) with i+j = peak_pos that maximizes a[i]*a[j]
        i_lo = max(0, peak_pos - n + 1)
        i_hi = min(peak_pos, n - 1)
        if i_lo > i_hi:
            break

        best_pair = (i_lo, peak_pos - i_lo)
        best_prod = best_a[i_lo] * best_a[peak_pos - i_lo]
        for i in range(i_lo, i_hi + 1):
            j = peak_pos - i
            prod = best_a[i] * best_a[j]
            if prod > best_prod:
                best_prod = prod
                best_pair = (i, j)

        i_star, j_star = best_pair
        if i_star == j_star:
            # Single element at peak: reduce it and redistribute
            a_try = best_a.copy()
            reduction = 0.01 * a_try[i_star]
            a_try[i_star] -= reduction
            # Distribute to neighbors
            neighbors = [k for k in range(n) if k != i_star]
            # Give to positions that don't contribute to the peak
            safe = [k for k in neighbors if k + i_star != peak_pos and abs(k - i_star) != peak_pos]
            if safe:
                a_try[safe] += reduction / len(safe)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_a = a_try
                    best_sc = sc_try
                else:
                    break
        else:
            # Two elements: solve min a[i]*a[j] s.t. a[i]+a[j] = const
            # Optimal: a[i] = a[j] = (a[i]+a[j])/2
            total = best_a[i_star] + best_a[j_star]
            a_try = best_a.copy()
            a_try[i_star] = total / 2
            a_try[j_star] = total / 2
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try
                best_sc = sc_try
            else:
                # Try partial equalization
                a_try2 = best_a.copy()
                mid = (best_a[i_star] + best_a[j_star]) / 2
                a_try2[i_star] += 0.5 * (mid - best_a[i_star])
                a_try2[j_star] += 0.5 * (mid - best_a[j_star])
                s_try = a_try2.sum()
                if s_try > 1e-15:
                    a_try2 = a_try2 / s_try
                    sc_try2 = _score(a_try2)
                    if sc_try2 < best_sc - 1e-14:
                        best_a = a_try2
                        best_sc = sc_try2
                    else:
                        break

    return best_a, best_sc


def _run_lbfgs_multi_beta(a_init, deadline, min_time_remaining=30.0):
    """Multi-beta L-BFGS with progressive smoothing reduction."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Generate beta schedule: from moderate to very small
    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.0025,
                              0.001, 0.0005, 0.0002, 0.0001,
                              0.00005, 0.00002, 0.00001, 0.000005, 0.000002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 10:
            break

        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                _lsobj_wrapper,
                z,
                args=(beta, n),
                jac=_lsgrad_wrapper,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f} (iter {i+1}/{len(betas)})"
                      f" t={time.monotonic()-elapsed:.1f}s")
        except Exception as e:
            print(f"    L-BFGS beta={beta:.2e}: exception {e}")
            continue

    return current_a


def _lsobj_wrapper(z, beta, n):
    return _lsobj(z, beta, n)


def _lsgrad_wrapper(z, beta, n):
    return _lsgrad(z, beta, n)


def _run_power_sum(a_init, deadline, p_values=None):
    """Power-sum optimization for very flat objectives."""
    if p_values is None:
        p_values = [4, 8, 16, 32, 64, 128]

    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for p in p_values:
        elapsed = time.monotonic()
        remaining = deadline - 20 - elapsed
        if remaining < 10:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _psobj_wrapper,
                z,
                args=(p, n),
                jac=_psgrad_wrapper,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
                print(f"    PowerSum p={p}: score={sc_new:.10f} t={time.monotonic()-elapsed:.1f}s")
        except Exception as e:
            print(f"    PowerSum p={p}: exception {e}")

    return current_a


def _psobj_wrapper(z, p, n):
    return _psobj(z, p, n)


def _psgrad_wrapper(z, p, n):
    return _psgrad(z, p, n)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block3":
        a = np.zeros(n)
        block = n // 4
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[3 * block:] = 0.5
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _surge_step(a, deadline, n_surges=50):
    """Aggressive single-peak reduction: find the peak, reduce its contributors maximally."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for _ in range(n_surges):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        peak_pos = int(np.argmax(c))

        i_lo = max(0, peak_pos - n + 1)
        i_hi = min(peak_pos, n - 1)
        if i_lo > i_hi:
            break

        # Find top contributing pairs
        products = []
        for i in range(i_lo, i_hi + 1):
            j = peak_pos - i
            products.append((best_a[i] * best_a[j], i, j))
        products.sort(reverse=True)

        # Reduce top few contributors
        a_try = best_a.copy()
        for prod, i, j in products[:3]:
            if prod < 1e-15:
                break
            # Reduce the larger element by redistributing
            if i != j:
                total = a_try[i] + a_try[j]
                # Make them more equal
                a_try[i] = total / 2
                a_try[j] = total / 2
            else:
                # Reduce and spread to non-contributing positions
                reduction = 0.05 * a_try[i]
                a_try[i] -= reduction
                # Spread to positions far from i
                far_positions = [k for k in range(n) if abs(k - i) > n // 4 and k + i != peak_pos]
                if far_positions:
                    a_try[far_positions] += reduction / len(far_positions)

        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_a = a_try
            best_sc = sc_try
        else:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining:
    - Multi-beta L-BFGS (log-sum-exp smoothed)
    - Active-set Newton with KKT verification
    - Local peak deflation (2-variable equalization)
    - Surge steps for aggressive peak reduction
    - Power-sum smoothing for very flat objectives
    - Multi-n exploration
    - Strategic restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Multi-beta L-BFGS on parent ===
    print(f"\n[Phase 1] Multi-beta L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_dl = min(deadline - 600, time.monotonic() + 300)
    a_lbfgs = _run_lbfgs_multi_beta(parent, lbfgs_dl, min_time_remaining=300.0)
    sc_lbfgs = _score(a_lbfgs)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")

    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_dl = min(deadline - 300, time.monotonic() + 200)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=400,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        update_best(a_as, sc_as, "as_newton_1")

        if best_seq is not None and time.monotonic() < deadline - 250:
            as_dl2 = min(deadline - 200, time.monotonic() + 120)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=300,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            update_best(a_as2, sc_as2, "as_newton_2")

    # === Phase 3: Local deflation ===
    print(f"\n[Phase 3] Local deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        def_dl = min(deadline - 150, time.monotonic() + 60)
        a_def, sc_def = _deflation_local(best_seq, def_dl, n_attempts=60)
        update_best(a_def, sc_def, "deflation")

    # === Phase 4: Surge steps ===
    print(f"\n[Phase 4] Surge steps (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        surge_dl = min(deadline - 100, time.monotonic() + 40)
        a_surge, sc_surge = _surge_step(best_seq, surge_dl, n_surges=80)
        update_best(a_surge, sc_surge, "surge")

    # === Phase 5: Post-deflation L-BFGS ===
    print(f"\n[Phase 5] Post-optimization L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        lbfgs2_dl = min(deadline - 60, time.monotonic() + 80)
        a_lbfgs2 = _run_lbfgs_multi_beta(best_seq, lbfgs2_dl, min_time_remaining=30.0)
        sc_lbfgs2 = _score(a_lbfgs2)
        update_best(a_lbfgs2, sc_lbfgs2, "lbfgs_post")

    # === Phase 6: Power-sum optimization ===
    print(f"\n[Phase 6] Power-sum (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        ps_dl = min(deadline - 30, time.monotonic() + 40)
        a_ps = _run_power_sum(best_seq, ps_dl, p_values=[16, 32, 64, 128, 256])
        sc_ps = _score(a_ps)
        update_best(a_ps, sc_ps, "power_sum")

    # === Phase 7: Try n=32768 ===
    print(f"\n[Phase 7] n=32768 exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_32k = _interpolate_to_n(best_seq if best_seq is not None else parent, 32768)
        sc_32k_init = _score(a_32k)
        print(f"  n=32768 initial score: {sc_32k_init:.10f}")
        if sc_32k_init < 2.0:  # Only proceed if initial is reasonable
            lbfgs_32k_dl = min(deadline - 40, time.monotonic() + 60)
            a_32k_opt = _run_lbfgs_multi_beta(a_32k, lbfgs_32k_dl, min_time_remaining=20.0)
            sc_32k = _score(a_32k_opt)
            update_best(a_32k_opt, sc_32k, "n_32768")

            if sc_32k < best_score + 0.01 and time.monotonic() < deadline - 20:
                a_32k_as, sc_32k_as = _active_set_newton(a_32k_opt, min(deadline - 10, time.monotonic() + 30),
                                                         max_iter=100, tol_list=[0.002, 0.001, 0.0005])
                update_best(a_32k_as, sc_32k_as, "n_32768_as")

    # === Phase 8: Strategic restarts ===
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("perturb_0.01", 0.01),
            ("perturb_0.03", 0.03),
            ("perturb_0.08", 0.08),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs_multi_beta(p_start, min(deadline - 10, time.monotonic() + 30),
                                        min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

    # === Phase 9: Final polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight Newton
        as_final_dl = min(deadline - 2, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final_dl, max_iter=300,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_as")

        # Final deflation
        if time.monotonic() < deadline - 2:
            a_def3, sc_def3 = _deflation_local(best_seq, min(deadline - 1, time.monotonic() + 15),
                                               n_attempts=40)
            update_best(a_def3, sc_def3, "final_def")

        # Very tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj_wrapper,
                        z,
                        args=(beta_final, n),
                        jac=_lsgrad_wrapper,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
