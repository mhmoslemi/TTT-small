def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return max(2, k)

from scipy.optimize import minimize

def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    else:

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

def _active_set_step(a, n, k_active, tol=0.001):
    """One step of active-set Newton: given active set of size k, solve KKT."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if k_active == 'auto':
        tol_abs = tol * c_max
        active = c >= (c_max - tol_abs)
        k = max(1, int(active.sum()))
    else:
        k = k_active
        top_idx = np.argsort(c)[::-1][:k]
        active = np.zeros_like(c, dtype=bool)
        active[top_idx] = True
    
    # Build the "integral operator" P(a)_i = sum_{k in active} a_{k-i} / k
    # (with appropriate bounds)
    A_vec = active.astype(np.float64) / k
    
    # Compute gradient: g_i = 2 * P(a)_i
    # P(a)_i = sum_{k: 0<=k-i<n} A_vec[k] * a[k-i]
    # This is a correlation: corr(a, A_vec) restricted to valid range
    flip_a = a[::-1]
    out_len = n + 2*n - 1 - 1  # n + len(A_vec) - 1 = n + (2n-1) - 1
    N = _next_pow2(n + (2*n-1) - 1)
    fa = np.fft.rfft(flip_a, N)

    result = np.fft.irfft(fa * fA, N)[:n + (2*n-1) - 1]
    h = result[n-1 : 2*n-1]  # This gives P(a)_i for i = 0, ..., n-1
    
    g = 2.0 * h  # gradient of the "average of active peaks"
    g_centered = g - np.mean(g)
    
    return g_centered, active, k

def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_with_active(a, B, n):
    """Compute P[j] = (1/|A|) * sum_{k in A, 0<=k-j<n} a[k-j] using FFT."""
    a_flipped = a[::-1]
    out_len = n + len(B) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_flipped, N)
    fB = np.fft.rfft(B, N)
    result = np.fft.irfft(fa * fB, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _kkt_fixed_point(a_init, deadline, max_iter=500, eta=0.02):
