    """Find optimal lambda on active set."""
    p = len(active_idx)
    # Build the matrix H where H[i,j] = a[active_idx[j] - i] if valid
    # (lambda * a)(i) = sum_j lambda[j] * a[active_idx[j] - i]
    H = np.zeros((n, p))

            k = active_idx[j] - i

                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:

def _newton_active_set(a_init, deadline, max_rounds=200, tol_start=1e-4, tol_end=1e-7):
    """Newton method on the active set with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: start wide, narrow down
        progress = rnd / max(1, max_rounds)
        tol_frac = tol_start * (tol_end / tol_start) ** progress
        tol = max(tol_frac * c_max, 1e-15)
        
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > n:  # Cap active set size
            # Keep only the top n elements
            thresholds = np.sort(c)[::-1][min(n, len(c)-1)]
            active = c >= thresholds
            n_active = int(active.sum())
        
        # Build W (dual variables)
        # Option 1: Uniform (simple)
        # Option 2: Non-uniform (solve dual)
        W = np.zeros(2 * n - 1)
        
        if n_active <= 200:
            # Solve for non-uniform lambda using a few iterations
            active_idx = np.where(active)[0]
            lam = _solve_dual_lambda(active_idx, a, n, max_iter=30)
            W[active_idx] = lam
            w_sum = lam.sum()
            if w_sum > 1e-15:
                W[active_idx] = lam / w_sum
        else:
            W[active] = 1.0 / n_active
        
        # Compute gradient
        g = _hankel_matvec(a, W, n)
        g_centered = g - np.mean(g)
        g_norm = float(np.linalg.norm(g_centered))
        if g_norm < 1e-14 * n:
            break  # Converged
        
        # Newton step: solve H dx = -g_centered
        # where H is the Hessian (Hankel matrix from W)
        dx = _newton_step(W, n, g_centered, a)
        
        if dx is None or not np.all(np.isfinite(dx)):
            # Fallback: gradient step
            dx = -g_centered / g_norm
        
        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.001 * alpha) - 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        
        if not step_done:
            # Try smaller gradient steps
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_centered / g_norm
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    step_done = True
                    break
        
        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not step_done and no_improve > 10:
            break
        if no_improve > 20:
            break
    
    return best_a, best_sc

def _solve_dual_lambda(active_idx, a, n, max_iter=30):
    """Solve for optimal lambda on active set via projected gradient."""
    p = len(active_idx)
    lam = np.ones(p) / p
    
    for it in range(max_iter):
        # Compute R[i] = sum_k lam[k] * a[active_idx[k] - i]
        lam_full = np.zeros(2 * n - 1)
        lam_full[active_idx] = lam
        N = _next_pow2(3 * n)
        L = np.fft.rfft(lam_full, N)
        A = np.fft.rfft(a, N)
        R = np.fft.irfft(L * np.conj(A), N)[:n]
        
        # Gradient of ||R - mean(R)||^2 w.r.t. lam
        R_centered = R - R.mean()
        # dR[i]/dlam[j] = a[active_idx[j] - i]
        # grad[j] = 2 * sum_i R_centered[i] * a[active_idx[j] - i]
        # This is: 2 * (a * a_rev convolved with R_centered) evaluated at active_idx positions
        
        # Compute: grad_full[k] = sum_i R_centered[i] * a[k-i] (for k in active_idx)
        N2 = _next_pow2(3 * n)
        RC = np.fft.rfft(R_centered, N2)
        A2 = np.fft.rfft(a, N2)
        grad_full = np.fft.irfft(RC * np.conj(A2), N2)
        grad = 2.0 * grad_full[active_idx]
        
        # Projected gradient step
        lr = 1.0 / max(np.max(np.abs(grad)), 1e-15)
        lam = lam - lr * grad
        
        # Project onto simplex {lam >= 0, sum(lam) = 1}
        lam = np.maximum(lam, 0)
        s = lam.sum()
        if s < 1e-30:
            lam = np.ones(p) / p
        else:
            lam = lam / s
    
    return lam

def _newton_step(W, n, rhs, max_cg_iter=200):
    """Solve H dx = rhs where H is the Hankel matrix defined by W, using CG.
    Also enforce sum(dx) = 0 (for scale invariance)."""
    x = np.zeros(n)
    r = rhs.copy()
    p_vec = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return None
    
    for i in range(max_cg_iter):
        Hp = _hankel_matvec(p_vec, W, n)
        pHp = float(p_vec @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp <= 0:
            break  # Not positive definite
        alpha = rs / pHp
        x_new = x + alpha * p_vec
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p_vec = (rs_new / rs) * p_vec + r
        rs = rs_new
        if rs < 1e-25 * rs0:
            break
    
    # Enforce sum(dx) = 0
    dx = x - np.mean(x)
    return dx

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
