import numpy as np

# Core utilities

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# LSE objectives and gradients

def _lse_objective(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

def _active_gradient(a, c, n, tol_frac=0.002):

    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ============================================================
# Adaptive LSE L-BFGS optimizer
# ============================================================

def _lbfgs_lse_adaptive(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    k = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    base_beta = max(gap * 0.5, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive beta schedule: start coarse, refine gradually
    n_max_stages = 20
    for stage in range(n_max_stages):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Adaptive beta: reduce geometrically but adapt based on gap
        c_cur = _full_convolve(current_a, current_a)
        c_sorted = np.sort(c_cur)[::-1]
        k = min(50, len(c_sorted) - 1)
        gap = c_sorted[0] - c_sorted[k]
        max_c = float(c_cur.max())
        beta = max(gap * 0.4, max_c * 0.002 * (0.7 ** stage))

        if beta < max_c * 1e-6:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement, try smaller beta next iteration
                pass
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Power-sum L-BFGS
# ============================================================

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    sum_cp = np.sum(np.power(c_safe, p))
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Iterative reweighting (exponential tilting)
# ============================================================

def _iterative_reweight(a_init, deadline, n_steps=10000, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.005
    patience = 0
    best_lr = lr

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient of peak in log-space
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = h - W
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0005 * i)
        z_new = z - step * grad_z
        z_new = np.clip(z_new, -100.0, 100.0)
        z_new = z_new - z_new.mean()

        a_new = _softmax(z_new)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            patience = 0
            z = z_new
            # Try increasing lr slightly
            lr = min(lr * 1.01, 0.02)
            best_lr = lr
        else:
            z = z_new  # Still accept for exploration
            patience += 1
            if patience > 100:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                patience = 0
                lr = best_lr * 0.7
            elif patience > 30:
                lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


# ============================================================
# Fourier shaping
# ============================================================

def _fourier_flatten(a, deadline, n_steps=500, strength=0.1):
    """Try to flatten the power spectrum to reduce convolution peak."""
    n = len(a)
    a_cur = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = strength

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a_cur)
        abs_A = np.abs(A)
        # Target: make |A|^2 more uniform
        mean_abs_sq = float(np.mean(abs_A ** 2))
        # Flatten: reduce high peaks in spectrum
        target = np.full_like(abs_A, np.sqrt(mean_abs_sq))
        # Blended target
        blend = 0.5 * abs_A + 0.5 * target
        phase = np.angle(A)
        A_new = blend * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a_cur = a_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
            if lr < 0.001:
                break

    return best_a, best_sc


# ============================================================
# Subgradient polish
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.008, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.93
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


# ============================================================
# Perturbation methods
# ============================================================

def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_local(a, rng, window_size=64):
    """Perturb a random local window."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    start = int(rng.integers(0, n - window_size))
    z[start:start + window_size] += 0.05 * rng.standard_normal(window_size)
    return _softmax(z)


# ============================================================
# Profile generators
# ============================================================

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


# ============================================================
# Multi-scale optimization
# ============================================================

def _multi_scale_optimize(a_coarse, n_target, deadline, min_remaining=30.0):
    """Optimize at coarse scale, then upsample and refine."""
    n_coarse = len(a_coarse)
    n_fine = n_target

    # Optimize at coarse scale
    a_opt_coarse, sc_coarse = _lbfgs_lse_adaptive(a_coarse, deadline, min_remaining=min_remaining + 10)

    # Upsample
    a_fine = _resize_sequence(a_opt_coarse, n_fine)

    # Refine at fine scale
    a_opt_fine, sc_fine = _lbfgs_lse_adaptive(a_fine, deadline, min_remaining=min_remaining)
    return a_opt_fine, sc_fine

# Main optimization function

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer for minimizing step function autocorrelation peak.
    Uses adaptive LSE, Newton active-set, iterative reweighting, Fourier shaping,
    multi-scale optimization, and diverse restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s", flush=True)

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}", flush=True)
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse initialization (120s) ===
    print(f"\n[Phase 1] Diverse initialization", flush=True)
    expl_deadline = min(deadline, t_start + 120)
    n_work = n_parent if n_parent > 0 else 16384

    for kind in ["cos_mod", "cos_mod2"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_work, kind)
        # Quick optimization
        a_opt, sc_opt = _lbfgs_lse_adaptive(a_try, min(expl_deadline, time.monotonic() + 15), min_remaining=5)
        upd(a_opt, sc_opt, f"init_{kind}")

    # === Phase 2: Main adaptive LSE (400s) ===
    print(f"\n[Phase 2] Main adaptive LSE", flush=True)
    if time.monotonic() < deadline - 400:
        a_main, sc_main = _lbfgs_lse_adaptive(_best_seq.copy(), deadline, min_remaining=350)
        upd(a_main, sc_main, "lbfgs_main")

    # === Phase 3: Newton active-set refinement (250s) ===
    print(f"\n[Phase 3] Newton active-set", flush=True)
    if time.monotonic() < deadline - 250:
        nw_dead = min(deadline - 230, time.monotonic() + 250)
        a_nw, sc_nw = _newton_active_set(_best_seq.copy(), nw_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 4: Power-sum L-BFGS (200s) ===
    print(f"\n[Phase 4] Power-sum L-BFGS", flush=True)
    if time.monotonic() < deadline - 200:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=180)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 5: Iterative reweighting (200s) ===
    print(f"\n[Phase 5] Iterative reweighting", flush=True)
    if time.monotonic() < deadline - 200:
        ir_dead = min(deadline - 180, time.monotonic() + 200)
        a_ir, sc_ir = _iterative_reweight(_best_seq.copy(), ir_dead, n_steps=8000, tol_frac=0.002)
        upd(a_ir, sc_ir, "iter_reweight")

    # === Phase 6: Fourier shaping (100s) ===
    print(f"\n[Phase 6] Fourier shaping", flush=True)
    if time.monotonic() < deadline - 100:
        fs_dead = min(deadline - 80, time.monotonic() + 100)
        a_fs, sc_fs = _fourier_flatten(_best_seq.copy(), fs_dead, n_steps=300, strength=0.05)
        upd(a_fs, sc_fs, "fourier_flatten")

    # === Phase 7: Newton + LSE alternating (200s) ===
    print(f"\n[Phase 7] Newton + LSE alternating", flush=True)
    a_cur = _best_seq.copy()
    for round_i in range(4):
        if time.monotonic() > deadline - 80:
            break
        # Newton
        nw_dead = min(deadline - 80, time.monotonic() + 30)
        a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.001)
        if sc_nw2 < _score(a_cur):
            a_cur = a_nw2
        upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")

        # LSE with very small beta
        if time.monotonic() < deadline - 60:
            n_cur = len(a_cur)
            c_cur = _full_convolve(a_cur, a_cur)
            mb = float(c_cur.max())
            z_cur = np.log(np.maximum(a_cur, 1e-30))
            z_cur = z_cur - z_cur.mean()
            beta_small = mb * 0.0001
            lse_dead = min(deadline - 60, time.monotonic() + 25)
            remaining = lse_dead - time.monotonic()
            if remaining > 3:
                max_iter = min(1500, max(50, int(remaining / 0.005)))
                try:
                    result = minimize(
                        _lse_objective, z_cur, args=(beta_small, n_cur),
                        jac=_lse_gradient, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-15, 'maxcor': 60}
                    )
                    a_lse = _softmax(result.x)
                    sc_lse = _score(a_lse)
                    if sc_lse < _score(a_cur):
                        a_cur = a_lse
                    upd(a_cur, _score(a_cur), f"alt_lse_{round_i}")
                except Exception:
                    pass

    # === Phase 8: Diverse restarts (200s) ===
    print(f"\n[Phase 8] Diverse restarts", flush=True)
    if time.monotonic() < deadline - 60:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_log(_best_seq, 0.04, rng)
            elif i == 2:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=6, max_amp=0.15)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.05)
            elif i == 4:
                p_start = _perturb_local(_best_seq, rng, window_size=128)
            elif i == 5:
                p_start = _perturb_local(_best_seq, rng, window_size=256)
            elif i == 6:
                p_start = _perturb_log(_best_seq, 0.01, rng)
            else:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=4, max_amp=0.1)

            # Quick LSE + Newton
            a_r, sc_r = _lbfgs_lse_adaptive(p_start, deadline, min_remaining=25)
            upd(a_r, sc_r, f"restart_{i}")

            if sc_r < _best_score * 1.002 and time.monotonic() < deadline - 20:
                nw_dead = min(deadline - 15, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.002)
                upd(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 9: Multi-n exploration (150s) ===
    print(f"\n[Phase 9] Multi-n exploration", flush=True)
    n_candidates = [8192, 12288, 16384, 20480, 24576]
    if n_parent not in n_candidates and n_parent >= 4096:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        a_opt_n, sc_opt_n = _lbfgs_lse_adaptive(a_init, deadline, min_remaining=30)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if _score(a_opt_n) < _best_score * 1.003 and time.monotonic() < deadline - 20:
            nw_dead = min(deadline - 10, time.monotonic() + 30)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=60, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 10: Final aggressive polish (60s) ===
    print(f"\n[Phase 10] Final polish", flush=True)
    if _best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-low beta LSE
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 3e-5
            max_iter = min(2000, max(50, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=50, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=3000, lr_init=0.003, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

        # One more Newton on the absolute best
        if time.monotonic() < deadline - 1:
            a_final_nw, sc_final_nw = _newton_active_set(_best_seq, deadline - 0.5, max_iter=30, tol_frac=0.0005)
            upd(a_final_nw, sc_final_nw, "absolute_final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}", flush=True)
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            print(f"FALLBACK uniform n=256", flush=True)
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = np.ones(256) / 256
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
