                new_coord[idx:] = concentrated_idx + symmetry_patch[idx:]
            
            # Adjust based on score
            new_score_val = next_temp / new_coord[idx:] ** 2
            
            grad_descent_sequence(new_coord, new_score_val)
            
            # Build gradient vector
            grad = [score - old_score for score, old_score in zip(new_score_vec, scores)]
            grad = normalize_gradient(grad_val)
            
            # Update coordinates with gradient descent
            sequence[idx:] = advection_operator(sequence_val[idx:], grad_strength)
            
            # Double-check bounds
            if (sequence < 0).any() or (sequence > 1000).any():
                sequence = np.clip(sequence, 0, 1000)
                
                # Re-normalize sequence with same sum
                if total_energy > 0.01:
                    sequence = sequence * (total_energy / np.sum(sequence))
                
                # Update total_energy for next epoch
                total_energy = np.sum(sequence)
            
            iteration += 1
            
        except time.monotonic() > deadline:
            break
    
    if evaluate_sequence(best_sequence) < best_score_val:
        return best_sequence
    return sequence
