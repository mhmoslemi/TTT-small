def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smo_obj_wrapper(z):
    return _smoobj(z, _state['beta'], _state['n'])


def _smo_grad_wrapper(z):
    return _smograd(z, _state['beta'], _state['n'])


def _lbfgs_smoothmax(a_init, deadline, min_time_remaining=30.0, beta_ratios=None, label=""):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.001)

    if beta_ratios is None:
        beta_ratios = np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                                0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                                0.000001, 0.0000005, 0.0000001, 0.00000002,
                                0.00000001, 0.000000005])
    betas = base_beta * beta_ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_iter_time * 3))))

        _state['beta'] = float(beta)
        _state['n'] = n

        try:
            result = minimize(
                _smo_obj_wrapper,
                z,
                jac=_smo_grad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.6g}: score={sc_new:.10f} [{label}]")
        except Exception:
            pass

    return current_a


def _peak_deflation(a_init, deadline, m=8, max_rounds=50, label="peak_defl"):
    """
    Peak deflation: directly reduce the top-m autocorrelation peaks
    using minimum-norm Jacobian steps.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top-m peaks (exclude very small ones)
        threshold = c_max * 0.9
        top_mask = c >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(c))])
        if len(top_idx) > m:
            top_idx = top_idx[np.argsort(c[top_idx])[::-1][:m]]

        n_top = len(top_idx)

        # Build B (n × n_top): B[p, j] = a[k_j - p] if 0 <= k_j - p < n, else 0
        # B[p, j] = a[k_j - p] for valid (k_j - p)
        B = np.zeros((n, n_top))
        for j, k in enumerate(top_idx):
            # p ranges such that 0 <= k-p < n => k-(n-1) <= p <= k
            p_lo = max(0, k - n + 1)
            p_hi = min(k, n - 1)
            if p_hi >= p_lo:
                B[p_lo:p_hi + 1, j] = a[k - p_hi:k - p_lo + 1][::-1]

        # Solve B * delta = -eps * 1 (minimum norm)
        # delta = -eps * B^T (B B^T + lam*I)^{-1} 1
        BBT = B.T @ B  # n_top × n_top
        BT_1 = B.T @ np.ones(n)  # n_top × 1

        # Regularize
        lam = 1e-10 * np.max(np.abs(np.diag(BBT)))
        if lam < 1e-30:
            lam = 1e-30
        BBT_reg = BBT + lam * np.eye(n_top)

        try:
            sol = np.linalg.solve(BBT_reg, BT_1)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(BBT_reg, BT_1, rcond=None)[0]

        # delta = -eps * B^T * sol
        # Choose eps adaptively
        eps = 0.01 * c_max
        delta = -eps * (B @ sol)

        # Line search
        best_alpha = 0
        best_trial_sc = best_sc
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015]:
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_trial_sc:
                best_trial_sc = sc_try
                best_alpha = alpha
        # Hmm, this line search might be slow. Let me use a simpler approach.
        # Actually for n=16384 and m=8, the score computation is one FFT, which is fast.
        # 7 evaluations per round, 50 rounds = 350 evaluations. Should be fine.

        if best_alpha > 0:
            a_new = a + best_alpha * delta
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new
                    if rnd % 10 == 0:
                        print(f"    PeakDefl round={rnd}: score={sc_new:.10f} m={n_top} [{label}]")
        else:
            # Try with smaller eps
            eps = 0.001 * c_max
            delta = -eps * (B @ sol)
            for alpha in [1.0, 0.5, 0.25]:
                a_try = a + alpha * delta
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                if rnd > 5:
                    break  # No progress, stop

    return best_a, best_sc


def _iterative_refine(a_init, deadline, min_time_remaining=60.0, label="refine"):
    """
    Alternating L-BFGS and peak deflation for iterative refinement.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for phase in range(6):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # L-BFGS with finer betas each phase
        remaining_for_lbfgs = min(180.0, deadline - min_time_remaining - time.monotonic())
        if remaining_for_lbfgs < 10:
            break

        ratios = [1.0 / (2.0 ** (phase + 1))] * 1 + [0.5 * r / (2.0 ** (phase + 1)) for r in [1, 0.5, 0.25, 0.1, 0.05, 0.01, 0.001, 0.0001, 0.00001]]
        ratios = np.array(ratios)

        a_lb = _lbfgs_smoothmax(a, deadline, min_time_remaining=min_time_remaining,
                                 beta_ratios=ratios, label=f"{label}_p{phase}_lbfgs")
        sc_lb = _score(a_lb)
        if sc_lb < best_sc:
            best_sc = sc_lb
            best_a = a_lb.copy()
            a = a_lb

        # Peak deflation
        if time.monotonic() + min_time_remaining > deadline:
            break
        pd_deadline = min(time.monotonic() + 60.0, deadline - min_time_remaining)
        a_pd, sc_pd = _peak_deflation(a, pd_deadline, m=8, max_rounds=30,
                                       label=f"{label}_p{phase}_pd")
        if sc_pd < best_sc:
            best_sc = sc_pd
            best_a = a_pd.copy()
            a = a_pd

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.7 + 0.01
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.001
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.001
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _spectral_shaping(a_init, deadline, max_iters=100, label="spectral"):
    """
    Iteratively shape the power spectrum to flatten the autocorrelation.
    Uses Gerchberg-Saxton style approach: alternate between time-domain
    projection (a >= 0, sum=1) and frequency-domain shaping.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Target: flat autocorrelation
    c_target = np.full(2 * n - 1, 1.0 / (2 * n - 1))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)

        # Compute error in autocorrelation domain
        # Weight the error by how much each lag contributes to the max
        c_max = float(c.max())
        # Use a "soft max" weight: w[k] = c[k]^p / sum(c^p) for large p
        p = 50
        c_p = c ** p
        c_p_sum = c_p.sum()
        if c_p_sum < 1e-300:
            break
        w = c_p / c_p_sum

        # Gradient of (1/2) * sum_k w[k] * (c[k] - c_target[k])^2 w.r.t. a
        # = sum_k w[k] * (c[k] - c_target[k]) * dc[k]/da
        # dc[k]/da = 2 * (a * w_shifted) where w_shifted is w convolved with a
        error = w * (c - c_target)
        a_rev = a[::-1]
        ec = _full_convolve(error, a_rev)
        grad = 2.0 * ec[n - 1:2 * n - 1]

        # Normalized gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break

        # Adaptive step size
        alpha = 0.1 / (1.0 + 0.01 * it)

        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            a_new = a - 0.3 * alpha * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new

        if it % 20 == 0:
            print(f"    Spectral iter={it}: score={best_sc:.10f} [{label}]")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    1. L-BFGS smoothed-max (adaptive beta)
    2. Peak deflation (null steering)
    3. Spectral shaping
    4. Iterative refinement (alternating 1-3)
    5. Multi-start and multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent (generous time)
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    a1 = _lbfgs_smoothmax(parent, deadline, min_time_remaining=600.0, label="p1_lbfgs")
    update_best(a1, _score(a1), "p1_lbfgs")

    # Phase 2: Peak deflation
    print(f"\n[Phase 2] Peak deflation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        pd_deadline = min(time.monotonic() + 200, deadline - 500)
        a2, sc2 = _peak_deflation(best_seq, pd_deadline, m=10, max_rounds=50, label="p2_pd")
        update_best(a2, sc2, "p2_pd")

    # Phase 3: Iterative refinement (alternating L-BFGS + peak deflation)
    print(f"\n[Phase 3] Iterative refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3, sc3 = _iterative_refine(best_seq, deadline, min_time_remaining=300.0, label="p3_refine")
        update_best(a3, sc3, "p3_refine")

    # Phase 4: Spectral shaping
    print(f"\n[Phase 4] Spectral shaping (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        sp_deadline = min(time.monotonic() + 80, deadline - 200)
        a4, sc4 = _spectral_shaping(best_seq, sp_deadline, max_iters=100, label="p4_spec")
        update_best(a4, sc4, "p4_spec")

    # Phase 5: More peak deflation (tighter)
    print(f"\n[Phase 5] Tight peak deflation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        pd2_deadline = min(time.monotonic() + 100, deadline - 150)
        a5, sc5 = _peak_deflation(best_seq, pd2_deadline, m=15, max_rounds=50, label="p5_pd_tight")
        update_best(a5, sc5, "p5_pd_tight")

    # Phase 6: Final L-BFGS with ultra-fine beta

                               0.00000005, 0.00000001, 0.000000001])
        a6 = _lbfgs_smoothmax(best_seq, deadline - 2, min_time_remaining=10.0,
                               beta_ratios=ratios_fin, label="p6_final_lbfgs")
        update_best(a6, _score(a6), "p6_final")
