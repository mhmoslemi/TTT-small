        current_time = time.time()
        if current_time - start_time > constraint_b_max / 1e6:  # Check for time constraint
            print(f"Time constraint violated, ending early at iteration {it}")
            break
        
        # Calculate temperature using capped polynomial cooling
        t_fraction = min(current_time - start_time, constraint_b_max)
        t_max_iter = max_iter * cooling_rate
        current_T = T_start * (T_end / T_start) ** (t_fraction / t_max_iter)
        
        # Calculate responsibilities for stable weights
        conv = _fft_convolve(sequence, sequence)
        responsibilities = np.exp((conv - current_T) / current_T)  # Avoid zero by poling
        if np.sum(responsibilities) == 0.0 or np.min(responsibilities) <= 0.0:
            responsibilities = weights * (np.abs(np.log(weights)) / 0.2)  # Heuristic centering
        responsibilities = responsibilities / np.sum(responsibilities)
        
        # Compute gradient
        grad = np.zeros(n)
        for lag_idx, lag in enumerate(range(0, 2*n-2, 3)):  # Sample every 3rd lag for efficiency
            if lag >= len(conv):
                continue
            if responsibilities[lag_idx] < 1e-10:
                continue
            lag_arr = conv[lag]
            lag_arr = lag_arr - np.max(lag_arr)  # Scale for variance reduction
            lag_arr = lag_arr / ((np.std(lag_arr) if len(lag_arr) > 1 else 1.0) + 1e-10)
            grad += responsibilities[lag_idx] * lag_arr
        
        # Project gradient to preserve the sum constraint
        grad_projected = grad - (np.sum(grad) / n) * sequence
        
        # Learning rate smoothing based on temperature
        learning_rate = learning_rate_max * (current_decay ** (current_T / 1000.0))
        
        # Gradient descent update but with stabilized learning
        sequence = sequence - learning_rate * grad_projected
        
        # Clip values to prevent explosion
        sequence = [max(0.0, min(x, 1000.0)) for x in sequence]
        
        # Maintain constant sum
        if np.sum(sequence) < 0.05:
            sequence = [0.01 if x < 0 else x + 0.01 for x in sequence]
        
        # Calculate new score
        score_val = evaluate_sequence_direct(sequence)
        
        # Update if improvement
        if score_val < best_score:
            best_score = score_val
            best_candidate = sequence.copy()
            current_decay *= 0.999
        
        time_elas = current_time - start_time
        if time_elas > t_step * (max_iter/1000)) {
            break
        }
    return best_candidate, best_score

    for n_val in n_vals:
        initial_guess = np.ones(n_val) * 0.01 / n_val  # Small fixed sum

        cand, score_val = adam_optimizer(initial_guess, max_iter,
