"""
Optimized stepper for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:

3. Adaptive beta-continuation L-BFGS with more aggressive schedule

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=400, tol_active=0.001, verbose=False):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: start loose, tighten over iterations
        adapt_tol = tol_active * max(0.1, 1.0 / (1.0 + 0.05 * it))
        tol = adapt_tol * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 4 * n

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient: (1/m) * sum_{k in active} dc_k/da_i = (1/m) * 2 * (A * a_flip)_i
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)

        # Project gradient to sum-zero subspace
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        # Regularization for numerical stability
        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            rs0 = rs
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1.0 + np.sqrt(rs0)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search with more aggressive strategy
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.98
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        best_t = 0
        best_f = f_old

        for ls in range(60):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < best_f:
                best_f = f_try
                best_t = t
            if f_try < f_old * (1.0 - 1e-10 * t * gnorm) or ls >= 50:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted and best_t > 0:
            a_try = a + best_t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < f_old:
                    a = a_try
                    accepted = True

        if not accepted:
            # Fallback: steepest descent with short step
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-14:
                    a = a_try
                    break
                t *= 0.5
            else:
                if it > 5:
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print(f"    AS iter {it}: score={sc:.10f}")

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _peak_equalize(a_init, deadline, max_rounds=80):
    """Directly try to equalize the top peaks by adjusting a."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find peaks above 99.5% of max
        thresh = c_max * 0.995
        peak_mask = c >= thresh
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            peak_mask[int(np.argmax(c))] = True
            n_peaks = 1
        if n_peaks > 20:
            top_idx = np.argsort(c)[::-1][:20]
            peak_mask = np.zeros_like(c, dtype=bool)
            peak_mask[top_idx] = True
            n_peaks = 20

        # For each peak k, the derivative dc_k/da_i = 2*a_{k-i}
        # We want to reduce all peaks simultaneously
        # Direction: for each i, sum over peaks of a_{k-i}
        # This is: h_i = sum_{k in peaks} a_{k-i} = (peak_mask * a_flip) convolved appropriately
        A_vec = peak_mask.astype(np.float64)
        h = _corr_a_A(a, A_vec, n)
        h = h / n_peaks
        W = float(np.dot(peak_mask.astype(float), c)) / n_peaks

        # Gradient of the "average peak" w.r.t. a
        grad = 2.0 * a * (h - W)
        grad = grad - np.mean(grad)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Line search
        f_old = c_max
        step = 1.0 / (1.0 + 0.1 * rnd)
        accepted = False
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            if rnd > 10:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                                      0.08, 0.05, 0.03, 0.02, 0.01, 0.005,
                                      0.0025, 0.001, 0.0005, 0.0002, 0.0001])
    else:
        betas = base_beta * np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04,
                                      0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if n <= 4096:
            est_time_per_iter = 0.004
        elif n <= 8192:
            est_time_per_iter = 0.008
        elif n <= 16384:
            est_time_per_iter = 0.02
        elif n <= 32768:
            est_time_per_iter = 0.06
        else:
            est_time_per_iter = 0.12

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                if verbose_lbfgs:
                    print(f"    L-BFGS beta_idx={i}: score={sc_new:.10f}")
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i > 3:
                # If no improvement for several stages, stop
                break
        except Exception:
            pass

    return current_a


def _spectral_init(n, rng, n_bands=100):
    """Create a sequence with approximately flat power spectrum."""
    # Start with random phases (like a CAZAC sequence but real-valued)
    # Use the idea: a = |FFT^-1(random_complex)| won't work directly
    # Instead: create a sequence, compute FFT, flatten the magnitude, take IFFT, clip to positive
    
    a = np.ones(n) / n
    for _ in range(5):
        A = np.fft.rfft(a, 2 * n)
        # Flatten: replace magnitude with average magnitude
        mag = np.abs(A)
        avg_mag = mag.mean()
        phase = A / np.maximum(mag, 1e-30)
        A_flat = avg_mag * phase
        a_new = np.fft.irfft(A_flat, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    return a


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.exp(-((t - 0.3) ** 2) / 0.01) + 0.8 * np.exp(-((t - 0.7) ** 2) / 0.01)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spectral":
        if rng is None:
            rng = np.random.default_rng(42)
        return _spectral_init(n, rng)
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.standard_normal(n)
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):

                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)

    return best_a, best_sc

    if time.monotonic() < deadline - 200:
        if best_seq is not None:
            a_large_init = _interpolate_to_n(best_seq, n_large)
        else:
            a_large_init = _make_profile(n_large, "uniform")
        sc_large_init = _score(a_large_init)
        print(f"  n={n_large} init score: {sc_large_init:.10f}")
        update_best(a_large_init, sc_large_init, f"init_n{n_large}")

        # L-BFGS on large
        a_large = _run_lbfgs(a_large_init, deadline - 100, min_time_remaining=50.0, aggressive=True)
        update_best(a_large, _score(a_large), f"lbfgs_n{n_large}")

        # Active-set on large
        if time.monotonic() < deadline - 50:
            a_large_as, sc_large_as = _active_set_newton(a_large, deadline - 30, max_iter=100, tol_active=0.001)
            update_best(a_large_as, sc_large_as, f"active_n{n_large}")

    # Phase 5: Try different n values near parent
    print(f"\n[Phase 5] Try n variants (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [8192, 24576]:
        if time.monotonic() > deadline - 60:
            break
        if best_seq is not None:
            a_try = _interpolate_to_n(best_seq, n_try)
        else:
            a_try = _make_profile(n_try, "uniform")
        sc_try = _score(a_try)
        update_best(a_try, sc_try, f"init_n{n_try}")

        if sc_try < best_score + 0.05 and time.monotonic() < deadline - 30:
            a_try_opt = _run_lbfgs(a_try, deadline - 10, min_time_remaining=5.0, aggressive=True)
            update_best(a_try_opt, _score(a_try_opt), f"lbfgs_n{n_try}")
            if time.monotonic() < deadline - 10:
                a_try_as, sc_try_as = _active_set_newton(a_try_opt, deadline - 5, max_iter=80, tol_active=0.001)
                update_best(a_try_as, sc_try_as, f"as_n{n_try}")

    # Phase 6: Spectral initialization at n=16384
    print(f"\n[Phase 6] Spectral init (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        for trial in range(3):
            if time.monotonic() > deadline - 80:
                break
            rng_sub = np.random.default_rng(int(seed) * 100 + trial)
            a_spec = _spectral_init(n_parent, rng_sub)
            sc_spec = _score(a_spec)
            print(f"  spectral trial {trial}: score={sc_spec:.10f}")
            if sc_spec < best_score + 0.1:
                a_spec_opt = _run_lbfgs(a_spec, deadline - 10, min_time_remaining=5.0, aggressive=True)
                update_best(a_spec_opt, _score(a_spec_opt), f"spec_{trial}")

    # Phase 7: More structured initializations
    print(f"\n[Phase 7] Structured inits (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        for kind in ["bimodal", "exp_ramp", "random_smooth"]:
            if time.monotonic() > deadline - 30:
                break
            a_struct = _make_profile(n_parent, kind, rng)
            sc_struct = _score(a_struct)
            if sc_struct < best_score + 0.05:
                a_struct_opt = _run_lbfgs(a_struct, deadline - 5, min_time_remaining=5.0, aggressive=True)
                update_best(a_struct_opt, _score(a_struct_opt), f"struct_{kind}")

    # Phase 8: Diverse restarts from best
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score + 0.02:
                a_r = _run_lbfgs(p_start, deadline - 5, min_time_remaining=3.0)
                update_best(a_r, _score(a_r), f"restart_{sigma}")

    # Phase 9: Final intensive active-set Newton
    print(f"\n[Phase 9] Final AS Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_fin_as, sc_fin_as = _active_set_newton(
            best_seq, deadline - 5, max_iter=300, tol_active=0.0005, verbose=False)
        update_best(a_fin_as, sc_fin_as, "final_as")

    # Phase 10: Peak equalization final
    print(f"\n[Phase 10] Final peak eq (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pe2, sc_pe2 = _peak_equalize(best_seq, deadline - 3, max_rounds=40)
        update_best(a_pe2, sc_pe2, "final_peak_eq")

    # Phase 11: Subgradient polish
    print(f"\n[Phase 11] Final subgrad (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_max_steps=3000, lr_init=0.005)
        update_best(a_sg, sc_sg, "final_sg")

    # Final tight active set
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_final_as, sc_final_as = _active_set_newton(
            best_seq, deadline - 1, max_iter=200, tol_active=0.0001)
        update_best(a_final_as, sc_final_as, "final_as_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
