      The first-order change: `delta c_k = 2 * sum_i d_i * a_{k-i}`.
      We want: `2 * sum_i d_i * a_{k-i} = -e_k` (for all k in the active set).
      This is a linear system: `(2 * P) @ d = -e` where P is the matrix with `P_{ki} = a_{k-i}`.
   d. Solve for d (least squares if overdetermined).
   e. Update: `a_new = a + alpha * d`, clip to [0, inf), renormalize.

This is essentially a Gauss-Newton method applied to the "convolution matching" problem.

The advantage of this approach is that it directly targets flattening the convolution, rather than minimizing the max.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _corr_vec(a, vec, n):
    """Compute h_j = sum_k vec_k * a_{k-j} for j=0..n-1."""

    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_rev, N)
    fv = np.fft.rfft(vec, N)
    result = np.fft.irfft(fa * fv, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _cg_solve(matvec_fn, rhs, max_iter=300, tol=1e-14):
    """CG solver. matvec_fn(v) returns M @ v."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs.copy()  # rhs - M @ 0 = rhs
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Mp = matvec_fn(p)
        pMp = np.dot(p, Mp)
        if pMp < 1e-30:
            break
        alpha = rs / pMp
        x += alpha * p
        r -= alpha * Mp
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x
