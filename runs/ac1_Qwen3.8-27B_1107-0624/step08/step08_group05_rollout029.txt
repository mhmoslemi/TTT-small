"""
Multi-objective optimizer for step function autocorrelation peak minimization.
Combines L2-autocorrelation smoothing, adaptive LSE-Newton, variance-aware
objectives, top-k averaging, and diversified restarts to beat 1.5027.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax(z):
    z = z - np.max(z)
    return z - np.log(np.sum(np.exp(z)))


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _l2_autocorr_obj(z, n):
    """Minimize ||autocorrelation||_2^2 (smooth, exact)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(c * c))


def _l2_autocorr_grad(z, n):
    """Gradient of ||c||_2^2 w.r.t. z where a = softmax(z)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    # grad_a = 4 * [convolve(c, a_rev)](n-1 : 2n-1)
    a_rev = a[::-1]
    conv_ca = _full_convolve(c, a_rev)
    grad_a = 4.0 * conv_ca[n - 1:2 * n - 1]
    return a * (grad_a - np.dot(a, grad_a))


def _centered_l2_obj(z, n):
    """Minimize variance of autocorrelation (centered L2)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    return float(np.sum(centered * centered))


def _centered_l2_grad(z, n):
    """Gradient of centered L2 norm of autocorrelation."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    a_rev = a[::-1]
    conv_cat = _full_convolve(centered, a_rev)
    grad_centered = 2.0 * conv_cat[n - 1:2 * n - 1]
    # Subtract mean contribution: d(mean_c)/da = ones/n roughly
    # mean_c = (1/(2n-1)) sum_l c(l), d(mean_c)/da_i = (2/(2n-1)) * [convolve(a_rev, ones)](i+n-1)/(2n-1)
    # Simplified: the gradient of mean(c) w.r.t. a_i = 2 * (sum_j a(j) * indicator) / (2n-1)
    # = 2 * sum_j a(j) / (2n-1) for all i (by symmetry of convolution with ones)
    # Actually: mean(c) = (1/(2n-1)) sum_l c(l) = (1/(2n-1)) sum_l sum_j a(j)a(l-j)
    # = (1/(2n-1)) sum_j a(j) sum_l a(l-j) = (1/(2n-1)) sum_j a(j) * (sum of a)
    # So d(mean_c)/da_i = (1/(2n-1)) * sum(a) * 2 (from product rule) -- wait
    # sum_l c(l) = sum_l sum_j a(j) a(l-j) = sum_j a(j) sum_m a(m) = (sum a)^2
    # So mean_c = (sum a)^2 / (2n-1)
    # d(mean_c)/da_i = 2 * (sum a) / (2n-1)
    norm = 2.0 * (2 * float(a.sum()) / (2 * n - 1))
    grad_a = grad_centered - norm
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = float(np.sum(cp))
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _topk_obj(z, k, n):
    """Average of top-k autocorrelation values."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:k]
    return float(np.mean(c[idx]))


def _topk_grad(z, k, n):
    """Subgradient of top-k average."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = h - W
    return a * (grad_a - np.dot(a, grad_a))


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _adaptive_lse_optimize(a_init, deadline, min_remaining=30.0, n_stages=16):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Finer beta schedule with more stages
    ratios = [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
              0.03, 0.02, 0.01, 0.006, 0.003, 0.001, 0.0005]
    betas = base_beta * np.array(ratios[:n_stages])

    for stage, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    # Adaptive tol: start tighter
    tol_frac_cur = min(tol_frac, 0.001 * c_max / max(c_max, 1e-10))

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac_cur * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_fourier_phase(a, rng, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_antipodal(a, rng, n_positions=8):
    """Flip signs at random positions (in log domain, this means large changes)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    for _ in range(n_positions):
        i = int(rng.integers(0, n))
        z[i] = -z[i] + rng.normal(0, 0.1)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 8)
        a[:taper] *= np.linspace(0.5, 1, taper)
        a[-taper:] *= np.linspace(1, 0.5, taper)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _composite_smooth_phase(a_init, deadline, min_remaining=30.0):
    """
    Multi-objective smooth phase: L2 -> centered L2 -> power sum -> top-k -> LSE.
    Each stage uses the output of the previous as initialization.
    """
    current_a = a_init.copy()
    current_sc = _score(current_a)

    stages = [
        ("L2", _l2_optimize),
        ("CenteredL2", _centered_l2_optimize),
        ("PowerSum", _power_sum_optimize),
        ("TopK", _topk_optimize),
        ("LSE", _adaptive_lse_optimize),
    ]

    for name, func in stages:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 30:
            break
        stage_deadline = min(deadline - min_remaining, time.monotonic() + max(30, remaining * 0.5))
        try:
            a_new, sc_new = func(current_a, stage_deadline, min_remaining=min(10, min_remaining))
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                print(f"    {name}: score improved to {sc_new:.10f}")
            else:
                print(f"    {name}: no improvement ({sc_new:.10f} >= {current_sc:.10f})")
        except Exception as e:
            print(f"    {name}: error {e}")

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-objective optimizer combining L2-autocorrelation smoothing,
    centered L2 (variance) minimization, top-k averaging, adaptive LSE,
    Newton active-set refinement, subgradient polish, and diverse restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    best_n = len(_best_seq) if _best_seq is not None else 16384

    # === Phase 1: L2 autocorrelation minimization (smooth, exact gradient) ===
    print(f"\n[Phase 1] L2 autocorrelation minimization")
    if time.monotonic() < deadline - 600:
        a_l2, sc_l2 = _l2_optimize(_best_seq.copy(), deadline, min_remaining=500)
        upd(a_l2, sc_l2, "L2_phase1")

    # === Phase 2: Centered L2 (variance) minimization ===
    print(f"\n[Phase 2] Centered L2 (variance) minimization")
    if time.monotonic() < deadline - 500:
        a_c, sc_c = _centered_l2_optimize(_best_seq.copy(), deadline, min_remaining=400)
        upd(a_c, sc_c, "CenteredL2_phase2")

    # === Phase 3: Composite smooth phase ===
    print(f"\n[Phase 3] Composite smooth phase")
    if time.monotonic() < deadline - 400:
        a_comp, sc_comp = _composite_smooth_phase(_best_seq.copy(), deadline, min_remaining=300)
        upd(a_comp, sc_comp, "composite_smooth")

    # === Phase 4: Power-sum optimization ===
    print(f"\n[Phase 4] Power-sum L-BFGS")
    if time.monotonic() < deadline - 300:
        a_ps, sc_ps = _power_sum_optimize(_best_seq.copy(), deadline, min_remaining=200)
        upd(a_ps, sc_ps, "powersum")

    # === Phase 5: Top-k averaging ===
    print(f"\n[Phase 5] Top-k averaging")
    if time.monotonic() < deadline - 250:
        a_tk, sc_tk = _topk_optimize(_best_seq.copy(), deadline, min_remaining=180)
        upd(a_tk, sc_tk, "topk")

    # === Phase 6: Main LSE L-BFGS ===
    print(f"\n[Phase 6] Main adaptive LSE L-BFGS")
    if time.monotonic() < deadline - 200:
        a_lse, sc_lse = _adaptive_lse_optimize(_best_seq.copy(), deadline, min_remaining=150)
        upd(a_lse, sc_lse, "lse_main")

    # === Phase 7: Newton active-set (primary refinement) ===
    print(f"\n[Phase 7] Newton active-set")
    if time.monotonic() < deadline - 150:
        nw_dead = min(deadline - 100, time.monotonic() + 200)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 8: Subgradient polish ===
    print(f"\n[Phase 8] Subgradient polish")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 70, time.monotonic() + 130)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=8000, lr_init=0.008, tol_frac=0.0008)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 9: Newton + SG alternating ===
    print(f"\n[Phase 9] Newton + SG alternating")
    if time.monotonic() < deadline - 100:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > deadline - 80:
                break
            nw_dead = min(deadline - 80, time.monotonic() + 30)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.0015)
            sc_cur = _score(a_cur)
            if sc_nw2 < sc_cur:
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")

            sg_dead2 = min(deadline - 80, time.monotonic() + 20)
            a_sg2, sc_sg2 = _subgrad_polish(a_cur, sg_dead2, n_steps=3000, lr_init=0.005, tol_frac=0.001)
            sc_cur = _score(a_cur)
            if sc_sg2 < sc_cur:
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 10: Multi-n exploration ===
    print(f"\n[Phase 10] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384]
    if best_n not in n_candidates and best_n >= 2048:
        n_candidates.append(best_n)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        # Quick LSE + Newton
        a_opt_n, sc_opt_n = _adaptive_lse_optimize(a_init, deadline, min_remaining=40)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if _score(a_opt_n) < _best_score * 1.005 and time.monotonic() < deadline - 40:
            nw_dead = min(deadline - 30, time.monotonic() + 40)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=60, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 11: Diverse restarts ===
    print(f"\n[Phase 11] Diverse restarts")
    if time.monotonic() < deadline - 40:
        perturbations = [
            ("log_003", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("log_005", lambda: _perturb_log(_best_seq, 0.05, rng)),
            ("fourier", lambda: _perturb_fourier_phase(_best_seq, rng, max_amp=0.3)),
            ("amp", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.08)),
            ("block", lambda: _perturb_block_swap(_best_seq, rng, n_swaps=4)),
            ("antipodal", lambda: _perturb_antipodal(_best_seq, rng, n_positions=6)),
        ]

        for i, (name, func) in enumerate(perturbations):
            if time.monotonic() > deadline - 25:
                break
            try:
                p_start = func()
            except Exception:
                continue

            # Quick LSE then Newton
            a_r, sc_r = _adaptive_lse_optimize(p_start, deadline, min_remaining=20)
            upd(a_r, sc_r, f"restart_{name}")

            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 15:
                nw_dead = min(deadline - 10, time.monotonic() + 15)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{name}_newton")

    # === Phase 12: Final aggressive polish ===
    print(f"\n[Phase 12] Final aggressive polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()

        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            # Ultra-low beta LSE
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_obj, z_fin, args=(beta_final, n_fin),
                    jac=_lse_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=100, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton_tight")

        # Final subgradient with very tight active set
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=5000, lr_init=0.004, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg_tight")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
