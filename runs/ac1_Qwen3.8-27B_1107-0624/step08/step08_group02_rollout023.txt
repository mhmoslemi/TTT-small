"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative (exponential gradient) updates as primary optimizer,
combined with L-BFGS smoothing, active-set Newton polish, spectral perturbations,
and systematic multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 1.4) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay2":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 3.0 + 0.005
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 3.0 + 0.005
        return a / a.sum()
    elif kind == "step_like":
        a = np.ones(n)
        third = n // 3
        a[:third] = 1.5
        a[third:2*third] = 1.0
        a[2*third:] = 0.4
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "watermelon":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) * (1.0 + 0.5 * np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _multiplicative_optimize(a_init, deadline, verbose=True, label=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_bmax = float(_full_convolve(a, a).max())
    best_score = _score(a)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    vel = np.zeros(n)
    momentum = 0.7
    eta = 0.08
    patience = 0
    max_patience = 40
    convergence_count = 0

    while time.monotonic() < deadline:
        a = _softmax(z)
        b = _full_convolve(a, a)
        b_max = float(b.max())

        # Adaptive active set
        active_count = 0
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
            active = b >= b_max - tol_frac * b_max
            active_count = int(active.sum())
            if active_count >= 2:
                break
        if active_count < 1:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(b))] = True
            active_count = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active_count

        # Compute gradient in log-domain
        h = _hankel_matvec(a, w, n)
        h_bar = float(a @ h)
        grad = h - h_bar

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Momentum update
        vel = momentum * vel + grad
        vel_norm = float(np.linalg.norm(vel))
        if vel_norm < 1e-30:
            vel = grad
            vel_norm = grad_norm

        # Adaptive learning rate with backtracking
        alpha = eta / max(vel_norm, 1e-30)
        step_done = False
        for ls in range(20):
            z_try = z - alpha * vel
            a_try = _softmax(z_try)
            b_try = _full_convolve(a_try, a_try)
            b_max_try = float(b_try.max())
            if b_max_try < b_max * (1.0 - 1e-8 * alpha * vel_norm) - 1e-16:
                z = z_try
                a = a_try
                step_done = True
                break
            alpha *= 0.5

        if not step_done:
            # Try pure gradient step without momentum
            alpha2 = eta / max(grad_norm, 1e-30)
            for ls in range(10):
                z_try = z - alpha2 * grad
                a_try = _softmax(z_try)
                b_try = _full_convolve(a_try, a_try)
                b_max_try = float(b_try.max())
                if b_max_try < b_max - 1e-16:
                    z = z_try
                    a = a_try
                    step_done = True
                    break
                alpha2 *= 0.5

        if step_done:
            patience = 0
            eta = min(eta * 1.01, 0.3)
            momentum = min(momentum + 0.001, 0.9)
        else:
            patience += 1
            eta *= 0.6
            vel = 0.5 * vel
            if patience > max_patience:
                # Reset to best
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                vel = np.zeros(n)
                eta = 0.05
                patience = 0
                momentum = 0.5

        # Track best
        sc = 2.0 * n * b_max / (1.0)  # since sum(a)=1
        if b_max < best_bmax - 1e-15:
            best_bmax = b_max
            best_a = a.copy()
            if verbose and convergence_count % 50 == 0:
                print(f"    {label}: bmax={b_max:.12f} score={sc:.10f} "
                      f"active={active_count} eta={eta:.4f} t={time.monotonic():.0f}")
            convergence_count = 0
        else:
            convergence_count += 1
            if convergence_count > 200:
                # Try smaller active set (tighten)
                pass  # Already handled by adaptive tolerance

        if convergence_count > 500:
            break

    # Final score
    final_sc = _score(best_a)
    if final_sc < best_score:
        best_score = final_sc

    return best_a, best_score


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _lbfgs_optimize(a_init, deadline, min_time=10.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * 0.002)
    min_beta = max_c * 1e-9

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(6000, max(20, int(remaining / (est_iter_time * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-8 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-3 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _newton_active_polish(a_init, deadline, max_rounds=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Tight active set with progressive loosening
        for tol_frac in [1e-5, 5e-5, 1e-4, 3e-4, 1e-3]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 2:
                break
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        # Solve H dx = -g_t using CG on the Hankel system
        # H v = _hankel_matvec(v, W, n) ... but we need the actual H matrix application
        # Use CG directly
        x = np.zeros(n)
        r = -g_t.copy()
        p = r.copy()
        rs = float(r @ r)
        rs0 = rs
        if rs0 < 1e-30:
            dx = -g_t / gn
        else:
            for i in range(300):
                Hp = _hankel_matvec(p, W, n)
                # Need to project onto the subspace where sum(dx)=0
                Hp_centered = Hp - np.mean(Hp)
                pHp = float(p @ Hp_centered)
                if abs(pHp) < 1e-30 or not np.isfinite(pHp):
                    break
                alpha = rs / pHp
                x_new = x + alpha * p
                r_new = r - alpha * Hp_centered
                rs_new = float(r_new @ r_new)
                if not np.isfinite(rs_new) or rs_new > rs * 1.1:
                    break
                x = x_new
                r = r_new
                p = (rs_new / rs) * p + r
                rs = rs_new
                if rs < 1e-26 * rs0:
                    break
            dx = x - np.mean(x)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / gn

        # Line search
        alpha_ls = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha_ls * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha_ls *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.003 * alpha_ls) - 1e-15:
                a = a_try
                step_done = True
                break
            alpha_ls *= 0.5
            if alpha_ls < 1e-15:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.1, 0.05, 0.02, 0.01, 0.005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _spectral_perturb(a, rng, mode="phase"):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    if mode == "phase":
        # Add small random phase
        phase = rng.uniform(-0.1, 0.1, len(A))
        A_pert = A * np.exp(1j * phase)
        # Enforce Hermitian symmetry for real signal
        A_pert[0] = np.abs(A_pert[0])
        if N % 2 == 0:
            A_pert[-1] = np.abs(A_pert[-1])
        a_pert = np.fft.irfft(A_pert, N)[:n]
    elif mode == "amplitude":
        # Add small random amplitude perturbation
        amp_pert = 1.0 + rng.uniform(-0.05, 0.05, len(A))
        A_pert = A * amp_pert
        A_pert[0] = np.abs(A_pert[0])
        if N % 2 == 0:
            A_pert[-1] = np.abs(A_pert[-1])
        a_pert = np.fft.irfft(A_pert, N)[:n]
    else:
        a_pert = a.copy()

    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    quick_candidates = []
    for pk in profile_kinds:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_parent, pk)
        sc_p = _score(a_p)
        quick_candidates.append((sc_p, a_p, pk))

    # Sort by score and take top ones
    quick_candidates.sort(key=lambda x: x[0])
    print(f"  Best profiles: {[f'{k}={s:.6f}' for s, _, k in quick_candidates[:3]]}")

    # Also add perturbed parent
    for sigma in [0.03, 0.08, 0.15]:
        a_pert = _perturb_log(parent, sigma, rng)
        sc_pert = _score(a_pert)
        quick_candidates.append((sc_pert, a_pert, f"pert_{sigma}"))

    # Spectral perturbations of parent
    for mode in ["phase", "amplitude"]:
        a_spec = _spectral_perturb(parent, rng, mode)
        sc_spec = _score(a_spec)
        quick_candidates.append((sc_spec, a_spec, f"spec_{mode}"))

    quick_candidates.sort(key=lambda x: x[0])

    # Quick multiplicative optimization on top candidates
    top_n_quick = min(5, len(quick_candidates))
    quick_deadline = min(deadline - 500, t_start + 120)
    for i in range(top_n_quick):
        if time.monotonic() > quick_deadline:
            break
        sc_init, a_init, label = quick_candidates[i]
        a_opt, sc_opt = _multiplicative_optimize(a_init, min(quick_deadline, time.monotonic() + 20),
                                                  verbose=False, label=label)
        update_best(a_opt, sc_opt)

    print(f"\n{'='*60}")
    print(f"Phase 2: Deep multiplicative optimization")
    print(f"{'='*60}")

    # Take best from phase 1 and optimize deeply
    base_a = best_seq.copy()
    deep_deadline = min(deadline - 300, time.monotonic() + 500)
    a_deep, sc_deep = _multiplicative_optimize(base_a, deep_deadline,
                                                verbose=True, label="deep")
    update_best(a_deep, sc_deep)

    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS refinement")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 200:
        lbfgs_deadline = min(deadline - 200, time.monotonic() + 150)
        a_lb = _lbfgs_optimize(best_seq, lbfgs_deadline, min_time=30)
        update_best(a_lb, _score(a_lb))

    print(f"\n{'='*60}")
    print(f"Phase 4: Newton active-set polish")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 100:
        np_deadline = min(deadline - 100, time.monotonic() + 80)
        a_np, sc_np = _newton_active_polish(best_seq, np_deadline, max_rounds=60)
        update_best(a_np, sc_np)

    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-n exploration")
    print(f"{'='*60}")

    # Try different n values
    n_tries = sorted(set([4096, 8192, 12288, 16384, 20480, 24576, 32768, 36864, 49152]))
    n_tries = [n for n in n_tries if n != n_parent]  # Skip already tried
    # Also include parent n for refinement
    n_tries = [n_parent] + n_tries

    for n_try in n_tries:
        if time.monotonic() > deadline - 60:
            break
        if n_try > 60000:
            continue  # Too slow for remaining time

        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        remaining_for_n = deadline - 60 - time.monotonic()
        if remaining_for_n < 30:
            break

        # Allocate time based on n (larger n gets less per-iteration budget)
        n_budget = min(80.0, max(20.0, remaining_for_n * 0.3))
        a_opt_n, sc_opt_n = _multiplicative_optimize(a_init_n, time.monotonic() + n_budget,
                                                      verbose=False, label=f"n_{n_try}")
        update_best(a_opt_n, sc_opt_n)

        # If competitive, do L-BFGS + Newton
        if best_seq is not None and _score(best_seq) < best_score + 0.005 and \
           len(best_seq) == n_try and time.monotonic() < deadline - 30:
            a_lb_n = _lbfgs_optimize(best_seq, min(deadline - 40, time.monotonic() + 30), min_time=10)
            update_best(a_lb_n, _score(a_lb_n))

            if time.monotonic() < deadline - 20:
                a_np_n, sc_np_n = _newton_active_polish(best_seq, min(deadline - 10, time.monotonic() + 20),
                                                         max_rounds=40)
                update_best(a_np_n, sc_np_n)

    print(f"\n{'='*60}")
    print(f"Phase 6: Diversification restarts")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restart_sigs = [0.02, 0.05, 0.1]

        for i, sig in enumerate(restart_sigs):
            if time.monotonic() > deadline - 20:
                break
            a_restart = _perturb_log(best_seq, sig, rng)
            # Also try spectral perturbation
            if i == 1:
                a_restart = _spectral_perturb(best_seq, rng, "phase")
            elif i == 2:
                a_restart = _spectral_perturb(best_seq, rng, "amplitude")

            rem = deadline - 20 - time.monotonic()
            if rem < 10:
                break
            a_opt_r, sc_opt_r = _multiplicative_optimize(a_restart, time.monotonic() + min(40, rem * 0.5),
                                                          verbose=False, label=f"restart_{i}")
            update_best(a_opt_r, sc_opt_r)

            if best_score < 1.506 and time.monotonic() < deadline - 15:
                a_np_r, sc_np_r = _newton_active_polish(best_seq, min(deadline - 5, time.monotonic() + 15),
                                                         max_rounds=30)
                update_best(a_np_r, sc_np_r)

    print(f"\n{'='*60}")
    print(f"Phase 7: Final polishing")
    print(f"{'='*60}")

    if best_seq is not None:
        # Final multiplicative push
        if time.monotonic() < deadline - 10:
            a_final, sc_final = _multiplicative_optimize(best_seq, min(deadline - 5, time.monotonic() + 30),
                                                         verbose=False, label="final")
            update_best(a_final, sc_final)

        # Final Newton
        if best_seq is not None and time.monotonic() < deadline - 3:
            a_fin_np, sc_fin_np = _newton_active_polish(best_seq, deadline - 1, max_rounds=50)
            update_best(a_fin_np, sc_fin_np)

        # Final L-BFGS with tiny beta
        if best_seq is not None and time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 2 - time.monotonic()
            if remaining > 3:
                max_iter = min(1500, max(20, int(remaining / 0.02)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 50, 'maxls': 150}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb))
                except Exception:
                    pass

    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"RESULT: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(8192, "asym_taper")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
