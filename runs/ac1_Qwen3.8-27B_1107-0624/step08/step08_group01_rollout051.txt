        if improved:
            alpha = min(alpha * 1.01, 0.3)
            # Gradually widen active set
            if no_improve < 100:
                top_k = min(max_top_k, top_k + 1)
        else:
            no_improve += 1
            alpha *= 0.5
            if no_improve % 30 == 0:
                top_k = min(max_top_k, int(top_k * 1.3))
                momentum_coef = min(0.7, momentum_coef * 1.05)
            if no_improve > 400:
                break

    return best_a, best_sc


def _peak_equalize(a_init, deadline, n_passes=5000):
    """
    Direct peak equalization: iteratively reduce the single highest peak
    by transferring mass from its dominant contributors to low-contribution
    positions. Very targeted for the final bits of improvement.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99999)

    for it in range(n_passes):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Find contributors to the peak: a[i] * a[k_star - i]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_lo > i_hi:
            continue

        i_range = np.arange(i_lo, i_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-18:
            continue

        # Identify top contributors
        n_top = max(1, int(len(contrib) * 0.05))
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_i = i_range[top_local]

        # Amount to transfer
        frac = rng.uniform(0.02, 0.06)
        amounts = contrib[top_local] * frac
        total_transfer = amounts.sum()
        if total_transfer < 1e-18:
            continue

        # Where to put the transferred mass: positions with low contribution
        # to ALL top peaks
        c_top5 = np.sort(c)[::-1][:5]
        top5_indices = np.argsort(c)[::-1][:5]

        # For each position i, compute its total contribution to top peaks
        contrib_all = np.zeros(n)
        for k_t in top5_indices:
            lo = max(0, k_t - n + 1)
            hi = min(n - 1, k_t)
            if lo <= hi:
                rr = np.arange(lo, hi + 1)
                jj = k_t - rr
                contrib_all[rr] += a[rr] * a[jj]

        # Exclude top contributors from receiving mass
        exclude = set(int(x) for x in top_i)
        for k_t in top5_indices:
            for ti in top_i:
                j_val = k_t - ti
                if 0 <= j_val < n:
                    exclude.add(int(j_val))

        receive_mask = ~np.isin(np.arange(n), list(exclude))
        if receive_mask.sum() == 0:
            receive_mask = np.ones(n, dtype=bool)
            receive_mask[list(exclude)] = False
            if receive_mask.sum() == 0:
                continue

        # Distribute mass inversely proportional to contribution
        weights = 1.0 / (contrib_all[receive_mask] + total / (2 * n))
        weights = weights / weights.sum()

        a_try = a.copy()
        for idx, amt in zip(top_i, amounts):
            a_try[int(idx)] -= amt
        a_try = np.maximum(a_try, 0.0)

        recv_positions = np.where(receive_mask)[0]
        a_try[recv_positions] += total_transfer * weights

        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < sc - 1e-14:
            a = a_try

        if it % 500 == 0 and it > 0:
            print(f"    PeakEq iter {it}: sc={sc:.10f}, best={best_sc:.10f}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Quick LSE to get into basin
    P2: MGD polishing (fast, many iterations)
    P3: Wide active-set polish (complementary)
    P4: Peak equalization (targeted)
    P5: Multi-n exploration
    P6: Final fine-tuning
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Quick LSE (150s) ===
    print(f"\n[P1] LSE aggressive (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200.0, t_start + 150.0)
    a1 = _run_lbfgs(best_a, p1_deadline, min_remaining=10.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: MGD Polishing (400s) ===
    print(f"\n[P2] MGD polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 700.0, t_start + 500.0)
    a2, sc2 = _mgd_polish(best_a, p2_deadline, top_k=50, alpha_init=0.015, momentum=0.6)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [mgd] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Wide active-set polish (300s) ===
    print(f"\n[P3] Wide active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400.0, t_start + 750.0)
    a3, sc3 = _wide_active_polish(best_a, p3_deadline, max_top_k=150, alpha_init=0.01)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [wide_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak equalization (200s) ===
    print(f"\n[P4] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 200.0, t_start + 900.0)
    a4, sc4 = _peak_equalize(best_a, p4_deadline, n_passes=10000)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [peak_eq] t={time.monotonic()-t_start:.1f}s")

    # Quick LSE after peak equalization
    if time.monotonic() < deadline - 100.0:
        p4b_deadline = min(deadline - 150.0, time.monotonic() + 100.0)
        a4b = _run_lbfgs(best_a, p4b_deadline, min_remaining=10.0)
        sc4b = _score(a4b)
        if sc4b < best_score:
            best_a = a4b.copy()
            best_score = sc4b
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_p4b] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_candidates = set()
    # Try sizes near and larger than current best
    for factor in [1.25, 1.5, 2.0]:
        n_try = int(n_best * factor)
        # Round to power of 2 for FFT efficiency
        n_try = int(2 ** np.ceil(np.log2(n_try)))
        n_candidates.add(n_try)
    # Also try specific sizes
    for n_fix in [16384, 20480, 24576, 32768]:
        if n_fix != n_best:
            n_candidates.add(n_fix)

    n_candidates = sorted([x for x in n_candidates if x <= 49152 and x != n_best])

    for n_try in n_candidates[:4]:  # Limit to 4 candidates
        if time.monotonic() > deadline - 80.0:
            break

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init_score={sc_init:.10f}")

        if sc_init >= best_score * 1.001:
            # Only worth trying if resampling doesn't hurt too much
            continue

        # Quick LSE on new n
        p5_deadline = min(deadline - 60.0, time.monotonic() + 120.0)
        a_opt = _run_lbfgs(a_init, p5_deadline, min_remaining=5.0)
        sc_opt = _score(a_opt)

        # Quick MGD
        if time.monotonic() < deadline - 50.0:
            mgd_dl = min(deadline - 50.0, time.monotonic() + 40.0)
            a_opt, sc_opt = _mgd_polish(a_opt, mgd_dl, top_k=30, alpha_init=0.01, momentum=0.5)

        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final intensive polishing (remaining) ===
    print(f"\n[P6] Final polishing (t={time.monotonic()-t_start:.1f}s)")
    final_deadline = deadline - 3.0

    # MGD with very small alpha
    if time.monotonic() < final_deadline - 30.0:
        mgd_dl = min(final_deadline - 30.0, time.monotonic() + 200.0)
        a6, sc6 = _mgd_polish(best_a, mgd_dl, top_k=20, alpha_init=0.005, momentum=0.4)
        if sc6 < best_score:
            best_a = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [mgd_final] t={time.monotonic()-t_start:.1f}s")

    # Peak equalize with more passes
    if time.monotonic() < final_deadline - 20.0:
        pe_dl = min(final_deadline - 20.0, time.monotonic() + 100.0)
        a7, sc7 = _peak_equalize(best_a, pe_dl, n_passes=20000)
        if sc7 < best_score:
            best_a = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [peak_eq_final] t={time.monotonic()-t_start:.1f}s")

    # Tiny LSE
    if time.monotonic() < final_deadline - 5.0:
        n = len(best_a)
        c_best = _full_convolve(best_a, best_a)
        mb = float(c_best.max())
        beta_fin = mb * 0.000005
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()

        def obj_fin(z):
            return _lse_obj(z, beta_fin, n)

        def grad_fin(z):
            return _lse_grad(z, beta_fin, n)

        remaining = final_deadline - 5.0 - time.monotonic()
        if remaining > 3.0:
            max_iter = min(2000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(obj_fin, z, jac=grad_fin, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 50})
                a_fin = _softmax(result.x)
                a_fin = np.maximum(a_fin, 0.0)
                s_fin = a_fin.sum()
                if s_fin > 1e-10:
                    a_fin = a_fin / s_fin
                    sc_fin = _score(a_fin)
                    if sc_fin < best_score:
                        best_a = a_fin.copy()
                        best_score = sc_fin
                        print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
