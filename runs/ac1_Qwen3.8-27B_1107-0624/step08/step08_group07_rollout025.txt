def _solve_lp_cutting_plane(a, M, max_active=30, max_iters=50, time_limit=5.0):
    """Solve max sum(b) s.t. b>=0, (a*b)[k]<=M using cutting plane."""
    n = len(a)
    b = a.copy()
    active = []  # list of active constraint indices
    t0 = time.monotonic()
    
    for cp_iter in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = _fft_convolve(a, b)
        
        # Find violated constraints
        tol = M * (1 + 1e-10)
        violated_mask = c > tol
        violated = np.where(violated_mask)[0]
        
        if len(violated) == 0:
            break
        
        # Add violated to active set
        for k in violated:
            k_int = int(k)
            if k_int not in active:
                active.append(k_int)
        
        # Limit active set
        if len(active) > max_active:
            # Keep the most violated
            c_at_active = np.array([c[k] for k in active])
            order = np.argsort(c_at_active)[::-1]
            active = [active[i] for i in order[:max_active]]
        
        # Build constraint matrix
        A = np.zeros((len(active), n))
        for i, k in enumerate(active):
            # A[i, j] = a[k-j] if 0 <= k-j < n
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[i, j] = a[idx]
        
        # Solve LP
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
            if res.success:
                b = res.x
            else:
                # Try with the violated constraints added
                b = np.maximum(b, 0)
        except Exception:
            break
    
    return np.maximum(b, 0.0)

"""
LP-based alternating minimization for step function autocorrelation optimization.
Primary method: Cutting-plane LP to find optimal descent directions in the space
of non-negative sequences, combined with subgradient polish and coordinate descent.
Key innovation: Each LP iteration finds the best direction to reduce peak autocorrelation,
enabling larger and more informed steps than pure gradient methods.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _solve_lp_cutting(a, M, max_active=40, max_iters=80, time_limit=3.0, bias=0.001):
    """
    Solve: max sum(b) + bias*dot(b,a) s.t. b>=0, (a*b)[k] <= M for all k.
    Uses cutting-plane: start with no constraints, add violated ones.
    The bias term breaks degeneracy and biases toward current iterate.
    """
    n = len(a)
    b = a.copy()
    active = []
    t0 = time.monotonic()

    for cp_iter in range(max_iters):
        if time.monotonic() - t0 > time_limit:

        c = _fft_convolve(a, b)
        tol = M * (1 + 1e-12)
        violated = np.where(c > tol)[0]

            order = np.argsort(c_at)[::-1]

        A = np.zeros((len(active), n))

        c_obj = -(1.0 + bias * a)

        # Solve LP for optimal direction
        b = _solve_lp_cutting(a, M, max_active=max_active,
                              max_iters=60, time_limit=lp_time_limit, bias=0.001)

        b_sum = b.sum()
        if b_sum < 1e-14:
            b = a.copy()
            b_sum = 1.0

        g = b / b_sum

        # Check: does g improve?
        # Try several step sizes
        improved = False
        step_sizes = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]
        for t in step_sizes:
            a_cand = (1.0 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s < 1e-14:
                continue
            a_cand = a_cand / s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            # Try the LP solution directly
            sc_g = _score_norm(g)
            if sc_g < best_score * (1 - 1e-12):
                best_score = sc_g
                best_a = g.copy()
                a = g
                improved = True
            else:
                stuck_count += 1
                # After being stuck, perturb
                if stuck_count > 30:
                    sigma = 0.002 * (1 + stuck_count / 100.0)
                    z = np.log(np.maximum(a, 1e-15))
                    z += sigma * rng.standard_normal(n)
                    z = z - z.mean()
                    a = _softmax_stable(z)
                    stuck_count = 0

        if it % score_interval == 0 and it > 0:
            elapsed = time.monotonic() - it_t0
            print(f"    LP[{it}]: score={best_score:.10f}, M={M:.8f}, "
                  f"sum_b={b_sum:.4f}, stuck={stuck_count}, dt={elapsed:.2f}s")

    return best_a.copy(), best_score


def _subgradient_polish(a_init, deadline, max_iter=100000, top_k=5, initial_lr=0.003):
    """Subgradient method on the max convolution peak with adaptive learning rate."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline - 3:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Get active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = [int(sorted_idx[0])]
        for idx in sorted_idx[1:min(top_k, len(sorted_idx))]:
            if c[int(idx)] > c_max * 0.998:
                active.append(int(idx))

        # Subgradient: average of gradient at active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        # Project to tangent space (preserve sum)
        g_tangent = g - np.sum(g) * a
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break

        # Adaptive step
        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-14:
            lr *= 0.9
            continue
        a_new /= s

        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-13):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 0.9999, initial_lr)
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            if lr < 1e-10:
                break
            if no_improve > 200:
                break

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_rounds=500):
    """Targeted coordinate descent on elements affecting the peak most."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline - 3:
            break
        round_num += 1
        improved = False

        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        # Try modifying contributing elements
        candidates = list(range(lo, hi + 1))
        np.random.shuffle(candidates)
        for j in candidates[:200]:
            old_val = a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_cand = a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-14:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-14):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Broader search
            any_improve = False
            step = max(1, n // 500)
            for j in range(0, n, step):
                old_val = a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.5, 0.8, 1.25, 1.5]:
                    a_cand = a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-14:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-14):
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
                        c = c_cand
                        any_improve = True
                        improved = True
                        break
                if improved:
                    break
            if not any_improve:
                break

    return best_a, best_score


def _adam_logspace(a_init, deadline, T_start, lr_start, max_iter=100000,
                   active_k=5, eval_interval=50):
    """Adam optimizer in log-space with adaptive temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0

    while t < max_iter:
        if time.monotonic() > deadline - 5:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (0.001) ** (frac ** 0.5)
        lr = lr_start * (0.05) ** (frac ** 0.7)

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Active set gradient
        c_max = float(c.max())
        tol = c_max * 0.99
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > active_k:
            top_idx = np.argsort(c[active])[::-1][:active_k]
            active = active[top_idx]

        g = np.zeros(n)
        for k in active:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[ki - hi:ki - lo + 1][::-1]
        g /= len(active)

        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            if patience > 20 and t % 200 == 0:
                # Restart
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z += 0.01 * np.random.randn(n)
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining LP-based alternating minimization (primary),
    subgradient polish, Adam in log-space, and coordinate descent.
    
    Phase 1: LP alternating (0-500s) - main optimization
    Phase 2: Subgradient polish (500-800s) - refine peaks
    Phase 3: Second LP round with tighter params (800-1100s)
    Phase 4: Adam log-space with low temperature (1100-1400s)
    Phase 5: Coordinate descent + final subgradient (1400-1750s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent or default
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP Alternating (primary method)
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: LP Alternating (budget={phase1_end-t_start:.0f}s) ===")
    a_lp, sc_lp = _lp_alternating(a, phase1_end, max_iter=5000,
                                   max_active=40, lp_time_limit=1.5,
                                   score_interval=100)
    full_sc = _score(a_lp)
    print(f"Phase 1 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if full_sc < best_score:
        best_score = full_sc
        best_a = a_lp.copy()

    # Phase 2: Subgradient polish
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end - 20:
        print(f"\n=== Phase 2: Subgradient Polish (budget={phase2_end-t_start:.0f}s) ===")
        a_sg, sc_sg = _subgradient_polish(best_a, phase2_end, max_iter=200000,
                                           top_k=5, initial_lr=0.004)
        full_sc_sg = _score(a_sg)
        print(f"Phase 2 done: score={full_sc_sg:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_sg < best_score:
            best_score = full_sc_sg
            best_a = a_sg.copy()

    # Phase 3: Second LP round
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 20:
        print(f"\n=== Phase 3: LP Round 2 (budget={phase3_end-t_start:.0f}s) ===")
        # Use slightly different parameters
        a_lp2, sc_lp2 = _lp_alternating(best_a, phase3_end, max_iter=5000,
                                         max_active=50, lp_time_limit=1.0,
                                         score_interval=100)
        full_sc2 = _score(a_lp2)
        print(f"Phase 3 done: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_lp2.copy()

    # Phase 4: Adam in log-space (low temperature fine-tuning)
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end - 20:
        print(f"\n=== Phase 4: Adam Log-Space (budget={phase4_end-t_start:.0f}s) ===")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        T_init = max_cb * 0.002
        a_adam, sc_adam = _adam_logspace(best_a, phase4_end, T_start=T_init,
                                         lr_start=0.02, max_iter=200000,
                                         active_k=5, eval_interval=30)
        full_sc_adam = _score(a_adam)
        print(f"Phase 4 done: score={full_sc_adam:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_adam < best_score:
            best_score = full_sc_adam
            best_a = a_adam.copy()

    # Phase 5: Coordinate descent + final subgradient
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 20:
        print(f"\n=== Phase 5: Coordinate Descent (budget={phase5_end-t_start:.0f}s) ===")
        a_cd, sc_cd = _coordinate_descent(best_a, phase5_end, max_rounds=1000)
        full_sc_cd = _score(a_cd)
        print(f"Phase 5a done: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 6: Final ultra-fine subgradient
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 6: Final Polish ===")
        a_fin, sc_fin = _subgradient_polish(best_a, deadline - 10, max_iter=100000,
                                             top_k=3, initial_lr=0.001)
        full_sc_fin = _score(a_fin)
        print(f"Phase 6 done: score={full_sc_fin:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_fin < best_score:
            best_score = full_sc_fin
            best_a = a_fin.copy()

    # Final validation and output
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"\nFINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
