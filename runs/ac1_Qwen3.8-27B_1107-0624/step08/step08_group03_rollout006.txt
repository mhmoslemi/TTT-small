    # The objective is 2n * max_c / sum^2
    # In log-space: we minimize 2n * LSE(c) / sum^2
    # grad of LSE w.r.t. a is (1/sum_w) * sum_k (e_k/sum_e) * grad_c_k
    # But we already computed the weighted sum of grad_c_k
    # The LSE grad is: (1/m) * sum_k exp((c_k-c_max)/beta) * dC[k]/da / sum_k exp(...)
    # = sum_k w_k * dC[k]/da (which is what we computed above, times 2)
    # Actually the LSE is c_max + beta * log(sum exp((c_k-c_max)/beta))
    # d LSE / da = beta * (1/sum) * sum exp((c_k-c_max)/beta) * (1/beta) * dC[k]/da
    # = (1/sum_e) * sum e_k * dC[k]/da
    # = sum_k w_k * dC[k]/da
    # And dC[k]/da[j] = 2*a[k-j]
    # So grad_LSE[j] = 2 * sum_k w_k * a[k-j]
    # And grad_objective = 2n * grad_LSE / sum^2 (by chain rule, since obj = 2n*LSE/sum^2)
    # But in log-space, we don't need the 2n/sum^2 factor for the direction (it's a positive scalar)
    
    return grad, c_max, top_k

def _subgrad_method(z_init, deadline, n_peaks=20, lr_init=0.02, max_steps=5000):
    """
    Subgradient method in log-space with adaptive step size.
    """
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    
    best_a = _softmax(z)
    best_sc = _score(best_a)
    
    lr = lr_init
    patience = 0
    best_patience = 0
    
    for step in range(max_steps):
        if time.monotonic() > deadline - 5:
            break
        
        a = _softmax(z)
        grad, c_max, _ = _grad_objective(a, n_peaks)
        
        # Project gradient to sum-zero subspace (to maintain normalization in log-space)
        grad = grad - grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step_size = lr / (1.0 + 0.0005 * step)
        
        z_new = z - step_size * grad
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        
        a_new = _softmax(z_new)
        sc_new = _score(a_new)
        
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            patience = 0
            lr = min(lr * 1.02, lr_init)
        else:
            patience += 1
            if patience % 10 == 0:
                lr *= 0.9
            if patience > 100:
                z = np.log(np.maximum(best_a, 1e-30)) - np.log(np.maximum(best_a, 1e-30)).mean()
                patience = 0
                lr = lr_init * 0.5
                best_patience += 1
                if best_patience > 5:
                    break
        
        z = z_new
    
    return best_a, best_sc

def _lbfgs_smooth(z_init, deadline, beta_init, n_peaks=20):
    """
    L-BFGS optimization of the smoothed objective in log-space.
    """
    n = len(z_init)
    
    def _lsobj(z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        top_k = np.argsort(c)[::-1][:n_peaks]
        shifted = (c[top_k] - c_max) / beta_init
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta_init * np.log(s))
    
    def _lsgrad(z):
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        top_k = np.argsort(c)[::-1][:n_peaks]
        shifted = (c[top_k] - c_max) / beta_init
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(n_peaks)
            w[0] = 1.0
        else:
            w = e / s
        
        grad = np.zeros(n)
        for idx, k in enumerate(top_k):
            wk = w[idx]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            kmj = k - j_range
            grad[j_range] += 2.0 * wk * a[kmj]
        
        return 2.0 * n * grad
    
    # Optimize with L-BFGS-B
    result = minimize(
        _lsobj,
        z_init,
        jac=_lsgrad,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50}
    )
    
    a_new = _softmax(result.x)
    sc_new = _score(a_new)
    return a_new, sc_new

def _block_coordinate_descent(a_init, deadline, block_size=64, max_rounds=100):
    """
    Block coordinate descent: optimize blocks of coordinates one at a time.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    offsets = list(range(0, n, block_size))
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 10:
            break
        
        # Shuffle offsets
        np.random.shuffle(offsets)
        
        improved = False
        for off in offsets:
            if time.monotonic() > deadline - 10:
                break
            
            end = min(off + block_size, n)
            j_range = np.arange(off, end)
            a_block = a[j_range].copy()
            
            if a_block.sum() < 1e-15:
                continue
            
            # Optimize this block to minimize max(conv)
            # Use a small L-BFGS
            def _block_obj(z_b):
                a_try = a.copy()
                a_try[j_range] = _softmax(z_b) * a_block.sum()
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    return np.inf
                a_try = a_try / s_try
                c = _fft_convolve(a_try, a_try)
                return c.max()
            
            def _block_grad(z_b):
                a_try = a.copy()
                a_try[j_range] = _softmax(z_b) * a_block.sum()
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                a_try = a_try / s_try
                c = _fft_convolve(a_try, a_try)
                c_max = c.max()
                k_max = int(np.argmax(c))
                
                # Gradient of c[k_max] w.r.t. a_try[j]
                lo = max(0, k_max - (n - 1))
                hi = min(n - 1, k_max)
                j_all = np.arange(lo, hi + 1)
                kmj = k_max - j_all
                grad_all = 2.0 * a_try[kmj]
                
                grad_block = grad_all[j_range]
                return grad_block * (a_block.sum() / s_try)  # chain rule through normalization
            
            z_b = np.log(np.maximum(a_block, 1e-30))
            z_b = z_b - z_b.mean()
            
            # Use a simple gradient step (L-BFGS might be too slow for small blocks)
            for inner in range(20):
                a_try = a.copy()
                a_try[j_range] = _softmax(z_b) * a_block.sum()
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                a_try = a_try / s_try
                c = _fft_convolve(a_try, a_try)
                c_now = c.max()
                
                k_max = int(np.argmax(c))
                lo = max(0, k_max - (n - 1))
                hi = min(n - 1, k_max)
                j_all = np.arange(lo, hi + 1)
                kmj = k_max - j_all
                grad_all = 2.0 * a_try[kmj]
                grad_b = grad_all[j_range]
                
                gnorm = np.linalg.norm(grad_b)
                if gnorm < 1e-14:
                    break
                
                step = 0.01 / (1 + 0.01 * inner)
                z_b = z_b - step * grad_b / gnorm
                z_b = z_b - z_b.mean()
            
            # Update
            a_new_block = _softmax(z_b) * a_block.sum()
            a[j_range] = a_new_block
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s
            
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _swap_search(a_init, deadline, n_swaps=5000):
    """
    Local search: try small transfers between pairs of positions.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(123)
    
    c = _fft_convolve(a, a)
    c_max = c.max()
    
    for i in range(n_swaps):
        if time.monotonic() > deadline - 5:
            break
        
        # Pick two random positions
        j1 = rng.integers(0, n)
        j2 = rng.integers(0, n)
        if j1 == j2:
            continue
        
        # Try a small transfer from j1 to j2
        eps = a[j1] * 0.01
        if eps < 1e-15:
            continue
        
        a[j1] -= eps
        a[j2] += eps
        
        if a[j1] < 0:
            a[j1] = 0
            a[j2] -= a[j1]  # undo
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        else:
            a[j1] += eps
            a[j2] -= eps
    
    return best_a, best_sc
