"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Spectral equalization for diverse initializations
2. Top-k smoothed max with FISTA (Nesterov accelerated projected gradient)
3. Exact 1D coordinate descent with adaptive candidate generation
4. Improved active-set Newton with CG and adaptive tolerance
5. Multi-n exploration and multi-start strategy

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _spectral_equalize(a, alpha, n_iter=10):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    a = a.copy()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        new_mag = np.power(np.maximum(mag, 1e-10), alpha)
        A_new = phase * new_mag
        a_new = np.fft.irfft(A_new, N)[:n]
        a = _normalize(a_new)
    return a


def _topk_smoothed_max(c, k):
    """Compute the average of the top-k values of c."""
    if k >= len(c):
        return float(c.max())
    # Use the fact that we only need the threshold
    sorted_c = np.sort(c)[::-1]
    return float(sorted_c[k - 1])


def _topk_obj_and_grad(a, k, n):
    """Compute top-k smoothed max objective and its gradient."""
    c = _full_convolve(a, a)
    # Find top-k threshold
    if k >= len(c):
        threshold = c.max()
    else:
        # Use partition for efficiency
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[k - 1]
    
    # Active set: indices where c >= threshold
    active = c >= (threshold - 1e-10 * threshold)
    n_active = active.sum()
    if n_active == 0:
        active[np.argmax(c)] = True
        n_active = 1
    
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    
    # Objective: (1/n_active) * sum of active c_k
    obj = float(np.dot(w, c))
    
    # Gradient: (1/n_active) * sum of active grad c_k
    # grad c_k / grad a_i = a[k-i] + a[i-k] (by symmetry, = 2*a[|k-i|] for dx)
    # Actually for c_k = sum_i a[i]*a[k-i], dc_k/da_j = a[k-j] + a[j-k] (if both valid)
    # By symmetry of c, dc_k/da_j = a[k-j] (if 0<=k-j<n) + a[j-k+2*(n-1)] ... 
    # More precisely: dc_k/da_j = a[(k-j) mod (2n-1)] in circular sense, but for linear:
    # dc_k/da_j = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, i.e., k<j)
    # For k=j: dc_k/da_j = a[0] (the k-j=0 term) + (j-k=0 term is same) = 2*a[0]? No...
    # Let me just use the correlation trick:
    # The gradient of c_k w.r.t. a is: at position i, it's a[k-i] + a[i-k] (when valid)
    # Summing over active k with weight w_k:
    # grad_i = sum_k w_k * (a[k-i] + a[i-k]) = 2 * sum_k w_k * a[k-i] (by symmetry)
    # Wait, that's not quite right for linear convolution.
    
    # Using the _corr_a_A trick:
    # _corr_a_A(a, w, n) gives the convolution of a with w, restricted to the "middle" part
    # which is exactly the gradient of sum_k w_k * c_k w.r.t. a
    # Actually let me verify: c_k = sum_i a[i]*a[k-i]
    # d/dj (sum_k w_k * c_k) = sum_k w_k * (a[k-j] + a[j-k]) for valid indices
    # = sum_k w_k * a[k-j] + sum_k w_k * a[j-k]
    # By change of variable in the second sum (let m = j-k, so k = j-m):
    # = sum_k w_k * a[k-j] + sum_m w_{j-m} * a[m]
    # The first sum is (w * a)[j] (cross-correlation of w with a)
    # The second sum is (w_rev * a)[j] where w_rev is reversed
    # For the "full" (not banded) convolution, this equals 2 * (a * w_rev)[j] approximately
    # But _corr_a_A handles this correctly. Let me use it.
    
    grad = 2.0 * _corr_a_A(a, w, n)
    
    # Project gradient onto simplex tangent (remove mean)
    grad_t = grad - np.mean(grad)
    
    return obj, grad_t, c, w


def _fista_topk(a_init, deadline, k=100, n_max_iter=50000, beta_init=0.01):
    """FISTA with top-k smoothed max."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = _score(a)
    
    # Estimate Lipschitz constant
    # The gradient of c_k w.r.t. a has norm at most 2*||a||_inf * ||w||_1
    # For top-k, ||w||_1 = 1, so L <= 2*||a||_inf
    # Use adaptive step size
    step = beta_init
    c_prev = _full_convolve(a, a).max()
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        
        obj, grad_t, c, w = _topk_obj_and_grad(y, k, n)
        
        # Adaptive step size (Lipschitz estimation)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break
        
        # Try the step
        a_new = y - step * grad_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            step *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check if step is good (objective decreased)
        c_new_max = _full_convolve(a_new, a_new).max()
        if c_new_max > c_prev + 1e-10:
            step *= 0.5
            if step < 1e-10:
                break
            continue
        
        c_prev = c_new_max
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t = t_new
        
        # Gradually increase k (start with small k, increase for finer optimization)
        if it % 1000 == 0:
            k = min(len(c), k + len(c) // 200)
        
        if it % 200 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Slowly increase step size
        if it % 500 == 0:
            step = min(step * 1.01, 0.1)
    
    sc_final = _score(a)
    if sc_final < best_score:
        best_a = a.copy()
        best_score = sc_final
    return best_a, best_score


def _logsumexp_obj_grad(z, beta, n):
    """Log-sum-exp smoothed max objective and gradient in log-parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    obj = 2.0 * n * (c_max + beta * np.log(max(s, 1e-300)))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation (generalized cross-entropy style)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    c_sorted = np.sort(c_cur)[::-1]
    n_c = len(c_sorted)
    # Determine initial beta from the gap
    if n_c > 20:
        gap = max_c - c_sorted[min(20, n_c - 1)]
    else:
        gap = max_c * 0.1
    base_beta = max(gap, max_c * 0.02)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                   0.015, 0.007, 0.003, 0.001, 0.0005,
                                   0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(4000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                lambda z_local: _logsumexp_obj_grad(z_local, beta, n)[0],
                z,
                jac=lambda z_local: _logsumexp_obj_grad(z_local, beta, n)[1],
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _cg_solve(matvec, rhs, max_cg=200, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    """Active-set Newton with multi-tolerance and adaptive step."""
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        improvement = False
        
        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 3 * n:
                if m > 3 * n:
                    top_idx = np.argsort(c)[::-1][:3 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue
            
            A_vec = active.astype(np.float64)
            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            
            # Hessian + regularization
            mu = 1e-8 * (c_max / max(m, 1)) + 1e-10
            
            def hess_vp(v):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v
            
            u = _cg_solve(hess_vp, g_t, max_cg=150)
            w = _cg_solve(hess_vp, np.ones(n), max_cg=150)
            w_sum = float(w.sum())
            
            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w
            
            dx = dx - np.mean(dx)
            
            # Line search
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)
            t = min(1.0, t_max)
            
            for ls in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= c_max * (1 - 1e-10) or ls >= 30:
                    if f_try <= c_max + 1e-10:
                        a = a_try
                        improvement = True
                        break
                t *= 0.5
            
            if improvement:
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                # Check convergence
                c_new = _full_convolve(a, a)
                if abs(float(c_new.max()) - c_max) < 1e-13:
                    return best_a, best_sc
                break
        
        if not improvement:
            # Try a smaller gradient step

            g_t = g_all - np.mean(g_all)

                a_try = np.maximum(a_try, 0)

                if f_try < c_max - 1e-12:
                    a = a_try
                    sc = _score(a)

                        best_sc = sc

def _coordinate_descent(a_init, deadline, n_passes=50, n_cands=5):
    """Coordinate descent with 1D optimization per coordinate."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Process coordinates in random order
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            s_rest = 1.0 - a_old
            
            # Try several candidate values
            candidates = [0.0, a_old]
            # Perturb around current
            for frac in [0.5, 0.75, 1.25, 1.5, 0.1, 2.0]:
                candidates.append(a_old * frac)
            # Try to equalize with the current max peak
            c = _full_convolve(a, a)
            c_max = c.max()
            # The peak at lag j involves a[j]^2, try reducing
            if a_old > 0:
                candidates.append(a_old * 0.8)
                candidates.append(a_old * 0.9)
            
            best_val = a_old
            best_local = np.inf
            
            for val in candidates:
                val = max(0.0, val)
                if val > 1.0:
                    continue
                a[j] = val
                s = a.sum()
                if s < 1e-15:
                    a[j] = a_old
                    continue
                a_norm = a / s
                # Quick score: just check the max of convolution
                c_check = _full_convolve(a_norm, a_norm)
                local_score = 2.0 * n * c_check.max()
                if local_score < best_local:
                    best_local = local_score
                    best_val = val
            
            a[j] = best_val
            a = _normalize(a)
            
            if best_val != a_old:
                improved = True
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.1 ** 2))
        return a / a.sum()
    elif kind == "multi_freq":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _deflation(a, deadline, n_attempts=30):
    """Targeted deflation of top convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        top_k = np.argsort(c)[::-1]
        n_top = max(3, min(len(c), int(0.03 * len(c))))
        top_positions = top_k[:n_top]
        
        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 30)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.99
                else:
                    a_new[j] *= 0.99
        
        a_new = _normalize(a_new)
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try smaller reduction
            a_small = a.copy()
            for k_pos in top_positions[:10]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    jj = k_pos - max_i
                    if a_small[max_i] >= a_small[jj]:
                        a_small[max_i] *= 0.998
                    else:
                        a_small[jj] *= 0.998
            a_small = _normalize(a_small)
            sc_small = _score(a_small)
            if sc_small < best_sc - 1e-12:
                best_sc = sc_small
                best_a = a_small.copy()
            else:
                break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining spectral equalization, FISTA,
    coordinate descent, active-set Newton, and L-BFGS with adaptive beta.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # Phase 1: Spectral equalization to get diverse starting points
    print(f"\n[Phase 1] Spectral equalization (t={time.monotonic() - t_start:.1f}s)")
    spectral_candidates = []
    for alpha in [0.5, 0.7, 1.0, 1.3, 1.5]:
        if time.monotonic() > deadline - 200:
            break
        a_spec = _spectral_equalize(parent, alpha, n_iter=5)
        sc_spec = _score(a_spec)
        spectral_candidates.append((a_spec, sc_spec))
        update_best(a_spec, sc_spec, f"spectral_a{alpha}")
    
    # Also try different profiles
    for kind in ["tapered", "cos", "cos2", "multi_freq"]:
        if time.monotonic() > deadline - 200:
            break
        a_prof = _make_profile(n_parent, kind)
        sc_prof = _score(a_prof)
        if sc_prof < best_score + 0.1:
            a_spec = _spectral_equalize(a_prof, 1.0, n_iter=3)
            sc_spec = _score(a_spec)
            spectral_candidates.append((a_spec, sc_spec))
            update_best(a_spec, sc_spec, f"spec_{kind}")
    
    # Pick the best starting point for main optimization
    start_candidates = [parent] + [c[0] for c in spectral_candidates]
    start_scores = [_score(c) for c in start_candidates]
    best_start_idx = int(np.argmin(start_scores))
    a_main = start_candidates[best_start_idx].copy()
    print(f"  Best start: score={start_scores[best_start_idx]:.10f}")
    
    # Phase 2: FISTA with top-k smoothed max
    print(f"\n[Phase 2] FISTA top-k (t={time.monotonic() - t_start:.1f}s)")
    fista_deadline = min(deadline - 600, time.monotonic() + 240)
    a_fista, sc_fista = _fista_topk(a_main, fista_deadline, k=50, n_max_iter=50000)
    update_best(a_fista, sc_fista, "fista_main")
    a_main = a_fista
    
    # Phase 3: L-BFGS with adaptive beta
    print(f"\n[Phase 3] L-BFGS smoothed (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 400, time.monotonic() + 240)
    a_lbfgs, sc_lbfgs = _lbfgs_smoothed(a_main, lbfgs_deadline, min_time_remaining=200.0)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")
    a_main = a_lbfgs
    
    # Phase 4: Active-set Newton
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        as_dl = min(deadline - 300, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(a_main, as_dl, max_iter=300,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        update_best(a_as, sc_as, "as_newton_1")
        a_main = a_as
    
    # Phase 5: Coordinate descent
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        cd_dl = min(deadline - 200, time.monotonic() + 120)
        a_cd, sc_cd = _coordinate_descent(a_main, cd_dl, n_passes=30, n_cands=5)
        update_best(a_cd, sc_cd, "coord_descent")
        a_main = a_cd
    
    # Phase 6: Deflation
    print(f"\n[Phase 6] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        def_dl = min(deadline - 150, time.monotonic() + 60)
        a_def, sc_def = _deflation(a_main, def_dl, n_attempts=30)
        update_best(a_def, sc_def, "deflation")
        if sc_def < best_score:
            a_main = a_def
    
    # Phase 7: Second round of optimization on best
    print(f"\n[Phase 7] Second round (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        # L-BFGS with very small beta
        lbfgs2_deadline = min(deadline - 80, time.monotonic() + 120)
        a_lbfgs2, sc_lbfgs2 = _lbfgs_smoothed(best_seq, lbfgs2_deadline, min_time_remaining=50.0)
        update_best(a_lbfgs2, sc_lbfgs2, "lbfgs_round2")
        
        if time.monotonic() < deadline - 60:
            as2_dl = min(deadline - 60, time.monotonic() + 60)
            a_as2, sc_as2 = _active_set_newton(best_seq, as2_dl, max_iter=200,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_as2, sc_as2, "as_round2")
    
    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    
    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        
        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")
        
        a_opt, sc_opt = _lbfgs_smoothed(a_init, min(deadline - 30, time.monotonic() + 90),
                                         min_time_remaining=20.0)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 20:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 10, time.monotonic() + 30),
                                                  max_iter=100, tol_list=[0.002, 0.001, 0.0005])
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")
    
    # Phase 9: Diverse restarts
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _lbfgs_smoothed(p_start, min(deadline - 10, time.monotonic() + 40),
                                         min_time_remaining=10.0)
            update_best(a_r, sc_r, f"restart_s{sigma}")
    
    # Phase 10: Final polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight active-set Newton
        as_final = min(deadline - 2, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=200,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_final, "final_as")
        
        # Coordinate descent polish
        if time.monotonic() < deadline - 3:
            a_cd_final, sc_cd_final = _coordinate_descent(best_seq, min(deadline - 1, time.monotonic() + 20),
                                                          n_passes=15, n_cands=4)
            update_best(a_cd_final, sc_cd_final, "final_cd")
        
        # Final deflation
        if time.monotonic() < deadline - 2:
            a_def_final, sc_def_final = _deflation(best_seq, min(deadline - 1, time.monotonic() + 10),
                                                   n_attempts=20)
            update_best(a_def_final, sc_def_final, "final_def")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
