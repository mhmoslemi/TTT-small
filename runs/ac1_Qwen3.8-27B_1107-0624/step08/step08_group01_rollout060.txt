"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated projected subgradient descent, LP-based direction
finding (when CVXPY is available), coordinate descent with 1D convex minimization,
and multi-n exploration. Uses FFT-accelerated subgradient computation.
"""
import numpy as np
import time
from scipy.optimize import minimize
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _compute_subgrad_fft(a, s_active):
    """
    Compute subgradient of max_k c[k] at point a, given active set indicator s_active.
    Uses FFT for O(n log n) computation.
    grad[i] = (2/|S|) * sum_{k in S} a[k-i] (with a[j]=0 for j outside [0,n))
    """
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    S = np.fft.rfft(s_active.astype(np.float64), N)
    # cross-correlation: ifft(S * conj(A)) gives sum_k s[k] * a[k-i]
    xc = np.fft.irfft(S * np.conj(A), N)
    count = s_active.sum()
    if count < 1e-15:
        return np.zeros(n)
    return (2.0 / count) * xc[:n]


def _active_set_mask(c, epsilon_frac=0.01):
    """Get indicator vector for lags within epsilon_frac*max of the max."""
    c_max = c.max()
    threshold = c_max - epsilon_frac * c_max
    return (c >= threshold).astype(np.float64)


def _nesterov_subgradient(a_init, deadline, min_remaining=5.0,
                          epsilon_start=0.005, epsilon_end=0.0005,
                          base_lr=0.5, name="nesterov"):
    """
    Nesterov-accelerated projected subgradient method.
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = a_init.copy()
    a = _project(a)
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_cmax = float(c.max())
    best_score = 2.0 * n * best_cmax

    x_prev = a.copy()
    eps = epsilon_start
    t = 0
    stall = 0
    max_stall = 200

    while time.monotonic() < deadline - min_remaining:
        t += 1
        c = _full_convolve(a, a)
        cmax = float(c.max())
        curr_score = 2.0 * n * cmax

        if curr_score < best_score - 1e-14:
            best_score = curr_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break

        # Adaptive epsilon: widen if stalled
        if stall > 50:
            eps = min(epsilon_start, eps * 1.1)
        elif stall < 20:
            eps = max(epsilon_end, eps * 0.995)

        # Active set
        s_active = _active_set_mask(c, eps)
        if s_active.sum() == 0:
            idx = int(np.argmax(c))
            s_active = np.zeros(2 * n - 1)
            s_active[idx] = 1.0

        # Subgradient
        g = _compute_subgrad_fft(a, s_active)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (beta - 1.0) / beta
        y = a + momentum * (a - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        # Compute gradient at extrapolated point
        c_y = _full_convolve(y, y)
        s_y = _active_set_mask(c_y, eps)
        if s_y.sum() == 0:
            idx = int(np.argmax(c_y))
            s_y = np.zeros(2 * n - 1)
            s_y[idx] = 1.0
        g_y = _compute_subgrad_fft(y, s_y)
        gnorm_y = np.linalg.norm(g_y)
        if gnorm_y < 1e-16:
            break

        # Adaptive step size
        gap = cmax - float(c.mean())
        if gap < 1e-15:
            gap = cmax * 0.001
        step = base_lr * gap / (gnorm_y * gnorm_y)
        step = min(step, 1.0 / gnorm_y)

        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(30):
            a_new = a - alpha * g_y
            a_new = _project(a_new)
            c_new = _full_convolve(a_new, a_new)
            sc_new = float(c_new.max())
            if sc_new < cmax - 1e-15:
                a = a_new
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller fixed step
            for trial_alpha in [1e-4, 5e-5, 1e-5]:
                a_new = _project(a - trial_alpha * g_y / (gnorm_y + 1e-15))
                c_new = _full_convolve(a_new, a_new)
                if float(c_new.max()) < cmax - 1e-15:
                    a = a_new
                    improved = True
                    break

        x_prev = a.copy()

    return best_a, best_score


def _lp_direction(f, deadline, min_remaining=5.0):
    """
    Solve LP: max sum(b) s.t. conv(f,b)[k] <= M, b>=0.
    Returns normalized b or None if too slow/failed.
    """
    if not HAS_CVXPY:
        return None

    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())

    if time.monotonic() > deadline - min_remaining - 30:
        return None

    b = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b))

    # Build constraint: conv(f, b)[k] <= M for all k
    # conv(f,b)[k] = sum_i f[i]*b[k-i]
    # This is a linear function of b.
    # For efficiency, use the structure: it's a Toeplitz matrix times b.
    # Build the constraint matrix A (2n-1 x n) where A[k,i] = f[i] if 0<=k-i<n else 0
    # This is essentially a "convolution matrix" (anti-Toeplitz).

    # For large n, building the full matrix might use too much memory.
    # n=12288 gives matrix of size 24575 x 12288 ≈ 300M entries ≈ 2.4 GB. Too large!
    # Use a sparser approach or skip for large n.
    if n > 8192:
        # Use a block approach: only enforce constraints for "important" lags
        c_sorted = np.sort(c)[::-1]
        n_active = min(200, 2 * n - 1)
        active_indices = np.argsort(c)[::-1][:n_active]
        # Build only these constraints
        constraints = []
        for k in active_indices:
            # conv(f,b)[k] = sum_i f[i]*b[k-i] for valid i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            coeffs = f[lo:hi + 1]  # these multiply b[k-lo], ..., b[k-hi]
            b_cols = np.arange(k - hi, k - lo + 1)
            # constraint: sum(coeffs[j] * b[b_cols[j]]) <= M
            lin_expr = coeffs @ b[b_cols]
            constraints.append(lin_expr <= M)
        constraints.append(b >= 0)
        constraints.append(cp.sum(b) <= 10.0)  # bound to keep LP bounded

    else:
        # For smaller n, build full constraint matrix
        conv_matrix = np.zeros((2 * n - 1, n))
        for i in range(n):
            for k in range(i, i + n):
                conv_matrix[k, i] = f[i]
        constraints = [conv_matrix @ b <= M, b >= 0, cp.sum(b) <= 10.0]

    prob = cp.Problem(objective, constraints)

    t0 = time.monotonic()
    try:
        # Try ECOS first (fast for this type), then SCS
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
        except Exception:
            prob.solve(solver=cp.SCS, max_iters=10000)
    except Exception:
        return None

    elapsed = time.monotonic() - t0
    if elapsed > 60:
        print(f"  [LP] Too slow ({elapsed:.1f}s), skipping")
        return None

    if b.value is None:
        return None

    b_val = np.maximum(b.value, 0.0)
    s = b_val.sum()
    if s < 1e-10:
        return None
    b_val = b_val / s
    print(f"  [LP] Solved in {elapsed:.1f}s, obj={prob.value:.4f}")
    return b_val


def _blend(f, g, t):
    """Blend two sequences: (1-t)*f + t*g, then project."""
    new = (1.0 - t) * f + t * g
    return _project(new)


def _lp_iterate(a_init, deadline, min_remaining=10.0, n_rounds=30):
    """Iterative LP-based improvement."""
    n = len(a_init)
    a = _project(a_init.copy())
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_cmax = float(c.max())
    best_score = 2.0 * n * best_cmax

    rng = np.random.default_rng(42)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline - min_remaining:
            break

        b = _lp_direction(a, deadline, min_remaining + 5.0)
        if b is None:
            break

        # Try several blend factors
        tvals = [0.05, 0.1, 0.2, 0.3, 0.5]
        improved = False
        for t in tvals:
            a_try = _blend(a, b, t)
            c_try = _full_convolve(a_try, a_try)
            cmax_try = float(c_try.max())
            sc_try = 2.0 * n * cmax_try
            if sc_try < best_score - 1e-12:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try a small perturbation + LP
            a_pert = _project(a + 0.01 * rng.standard_normal(n))
            b2 = _lp_direction(a_pert, deadline, min_remaining + 5.0)
            if b2 is not None:
                for t in [0.1, 0.2, 0.3]:
                    a_try = _blend(a, b2, t)
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) * 2.0 * n < best_score - 1e-12:
                        best_score = 2.0 * n * float(c_try.max())
                        best_a = a_try.copy()
                        a = a_try
                        break

    return best_a, best_score


def _coord_descent_1d(a_init, deadline, min_remaining=5.0, max_sweeps=50):
    """
    Coordinate descent: for each coordinate i, find optimal value via 1D minimization.
    The 1D problem: minimize max_k (A_k + B_k*x + C_k*x^2) subject to x >= 0.
    """
    n = len(a_init)
    a = _project(a_init.copy())
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_cmax = float(c.max())
    best_score = 2.0 * n * best_cmax

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - min_remaining:
            break

        improved_sweep = False
        for i in range(n):
            if time.monotonic() > deadline - min_remaining:
                break

            # Compute coefficients for c[k] as function of a[i] = x
            # c[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]:
            #   j=i: a[i]*a[k-i] = x * a[k-i] (if 0<=k-i<n)
            #   k-j=i => j=k-i: a[k-i]*a[i] = a[k-i] * x (if 0<=k-i<n and k-i!=i)
            #   j=i and k-j=i (k=2i): a[i]^2 = x^2

            # So c[k] = A_k + B_k * x + C_k * x^2
            # where:
            #   C_k = 1 if k == 2i, else 0
            #   B_k = a[k-i] + a[k-i] (if k!=2i) or 2*a[i] (if k=2i)... 

            # Actually: for k where 0<=k-i<n:
            #   If k != 2i: B_k = 2*a[k-i], C_k = 0
            #   If k == 2i: B_k = 0, C_k = 1 (the x^2 term)
            #   But also the "other" terms: when j = k-i (which is != i if k != 2i):
            #     a[k-i]*x contributes to B_k
            #   Wait, I already counted both. Let me redo.

            # c[k] = sum_{j valid} a[j] * a[k-j]
            # Expand in terms of x = a[i]:
            # For each j:
            #   If j == i: term = x * a[k-i] (contributes x to c[k], coeff = a[k-i])
            #   If k-j == i (j = k-i, and j != i): term = a[k-i] * x (contributes x, coeff = a[k-i])
            #   If j == i and k-j == i (k = 2i): term = x * x = x^2
            #   Otherwise: term doesn't involve x

            # So:
            #   A_k = c[k] - (terms involving x)
            #   B_k = a[k-i] + (a[k-i] if k-i != i and 0<=k-i<n else 0) = 2*a[k-i] (if 0<=k-i<n, k!=2i)
            #         or 0 (if k=2i, since the x^2 term is separate)
            #   Wait, for k=2i: B_k = a[i] + a[i] = 2*a[i]? No...
            #   For k=2i: the only terms involving x are x*a[i] (j=i, k-j=i) and a[i]*x (j=i, k-j=i)
            #   But these are the SAME term: a[i]*a[i] = x^2. So there's no linear term.
            #   B_k = 0, C_k = 1 for k = 2i.

            # Let me just compute numerically:
            a_save = a[i]
            # Set a[i] = 0 to get A_k (constant term)
            a[i] = 0.0
            c_base = _full_convolve(a, a)

            # Set a[i] = 1, others same as original (with a[i]=0 base)
            # Actually, let's compute B_k and C_k directly:
            # c[k] with a[i]=x: c[k] = c_base[k] + x * f1[k] + x^2 * f2[k]
            # where f1[k] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n and k-i!=i)
            #             = 2*a[k-i] (if 0<=k-i<n and k!=2i), or 2*a[i_save] (if k=2i, but we set a[i]=0)...

            # This is getting confusing. Let me just do it numerically:
            a[i] = 0.0
            c0 = _full_convolve(a, a)  # c with a[i]=0
            a[i] = 1.0
            c1 = _full_convolve(a, a)  # c with a[i]=1
            a[i] = 2.0
            c2 = _full_convolve(a, a)  # c with a[i]=2

            # c(x) = c0 + x*(c1-c0) + x^2*(c2 - 2*c1 + c0)
            B = c1 - c0
            C = c2 - 2 * c1 + c0
            a[i] = a_save

            # Now minimize max_k (c0[k] + B[k]*x + C[k]*x^2) over x >= 0
            # This is a convex 1D problem (max of convex quadratics with C[k]>=0)
            # Use golden section search

            # Bounds: x in [0, a_save * 3] (reasonable range)
            lo = 0.0
            hi = max(a_save * 3.0, 0.1)

            def obj_1d(x):
                vals = c0 + B * x + C * x * x
                return float(np.max(vals))

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = obj_1d(x1)
            f2 = obj_1d(x2)

            for _ in range(40):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = obj_1d(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = obj_1d(x2)

            x_opt = (lo + hi) / 2.0
            f_opt = obj_1d(x_opt)

            # Check if this improves
            if f_opt < best_cmax - 1e-12:
                a[i] = x_opt
                # Verify
                c_check = _full_convolve(a, a)
                cm = float(c_check.max())
                if cm < best_cmax - 1e-13:
                    best_cmax = cm
                    best_score = 2.0 * n * cm
                    best_a = a.copy()
                    improved_sweep = True

            a[i] = a_save  # restore if not improved... wait, we already set it

        if not improved_sweep and sweep > 5:
            break

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.arange(n)
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t / max(n - 1, 1)) + 0.08 * np.cos(4.0 * np.pi * t / max(n - 1, 1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.sqrt(t)
        return a / a.sum()
    elif kind == "sinusoidal_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _peak_shave_targeted(a, deadline, min_remaining=5.0, n_trials=100):
    """
    Targeted peak shaving: for the top peak, find the best linear combination
    of nearby coordinates to reduce it.
    """
    n = len(a)
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_cmax = float(c.max())
    best_score = 2.0 * n * best_cmax
    rng = np.random.default_rng(99)

    for trial in range(n_trials):
        if time.monotonic() > deadline - min_remaining:
            break

        c = _full_convolve(a, a)
        cmax = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributors to peak at k_star
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = a[lo:hi + 1] * a[k_star - hi:k_star - lo + 1]
        total = contrib.sum()
        if total < 1e-15:
            continue

        # Strategy: reduce top contributors and add to bottom ones
        n_top = max(1, len(contrib) // 10)
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:n_top]

        a_try = a.copy()
        for idx in top_idx:
            i = lo + idx
            a_try[i] *= 0.85
        reduction = (a[i] * 0.15).sum() if len(top_idx) > 0 else 0.0
        # Better: compute actual reduction
        red = 0.0
        for idx in top_idx:
            i = lo + idx
            red += a[i] * 0.15
        for idx in bot_idx:
            i = lo + idx
            a_try[i] += red / max(len(bot_idx), 1)

        a_try = _project(a_try)
        c_try = _full_convolve(a_try, a_try)
        if float(c_try.max()) < best_cmax - 1e-12:
            best_cmax = float(c_try.max())
            best_score = 2.0 * n * best_cmax
            best_a = a_try.copy()
            a = a_try

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase approach:
    1. Nesterov subgradient on parent
    2. LP-based direction finding
    3. Coordinate descent polishing
    4. Multi-n exploration
    5. Final targeted polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    print(f"CVXPY available: {HAS_CVXPY}")
    print(f"Budget: {budget_s}s, deadline at t={deadline - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}, score={2.0*len(parent)*float(_full_convolve(parent,parent).max()):.10f}")
    except (NameError, Exception) as e:
        print(f"No parent: {e}")

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    def check_update(a_new, sc_new, label):
        nonlocal best_a, best_score
        if sc_new < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc_new
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [{label}] t={time.monotonic()-t_start:.1f}s")
            return True
        return False

    # ============ Phase 1: Nesterov subgradient ============
    print(f"\n{'='*60}")
    print(f"[P1] Nesterov subgradient (n={len(best_a)})")
    p1_deadline = min(deadline - 900.0, t_start + 400.0)
    a1, sc1 = _nesterov_subgradient(best_a, p1_deadline, min_remaining=10.0,
                                     epsilon_start=0.003, epsilon_end=0.0003,
                                     base_lr=0.8)
    check_update(a1, sc1, "nesterov_p1")

    # ============ Phase 2: Second Nesterov pass with different params ============
    print(f"\n[P2] Nesterov pass 2 (n={len(best_a)})")
    p2_deadline = min(deadline - 700.0, t_start + 600.0)
    a2, sc2 = _nesterov_subgradient(best_a, p2_deadline, min_remaining=10.0,
                                     epsilon_start=0.001, epsilon_end=0.0001,
                                     base_lr=0.4)
    check_update(a2, sc2, "nesterov_p2")

    # ============ Phase 3: LP-based (if available) ============
    if HAS_CVXPY and len(best_a) <= 65536:
        print(f"\n[P3] LP-based refinement (n={len(best_a)})")
        p3_deadline = min(deadline - 500.0, t_start + 800.0)
        a3, sc3 = _lp_iterate(best_a, p3_deadline, min_remaining=10.0, n_rounds=20)
        check_update(a3, sc3, "lp_p3")
        if sc3 < best_score + 0.001:
            # Follow up with Nesterov
            p3b_deadline = min(deadline - 400.0, t_start + 900.0)
            a3b, sc3b = _nesterov_subgradient(best_a, p3b_deadline, min_remaining=10.0,
                                               epsilon_start=0.001, epsilon_end=0.0001, base_lr=0.3)
            check_update(a3b, sc3b, "nesterov_post_lp")
    else:
        print(f"\n[P3] LP skipped (CVXPY={HAS_CVXPY}, n={len(best_a)})")

    # ============ Phase 4: Coordinate descent ============
    print(f"\n[P4] Coordinate descent (n={len(best_a)})")
    if time.monotonic() < deadline - 200.0:
        p4_deadline = min(deadline - 200.0, t_start + 1100.0)
        a4, sc4 = _coord_descent_1d(best_a, p4_deadline, min_remaining=10.0, max_sweeps=20)
        check_update(a4, sc4, "coord_descent_p4")

    # ============ Phase 5: Peak shaving ============
    print(f"\n[P5] Peak shaving (n={len(best_a)})")
    if time.monotonic() < deadline - 150.0:
        p5_deadline = min(deadline - 150.0, t_start + 1250.0)
        a5, sc5 = _peak_shave_targeted(best_a, p5_deadline, min_remaining=5.0, n_trials=150)
        check_update(a5, sc5, "peak_shave_p5")

    # ============ Phase 6: Multi-n exploration ============
    print(f"\n[P6] Multi-n exploration")
    n_candidates = [16384, 12288, 20480, 9216]
    n_candidates = [x for x in n_candidates if x != len(best_a)]
    n_candidates.sort(key=lambda x: abs(x - len(best_a)))

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 100.0:
            break
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # Quick Nesterov
        rem = min(150.0, deadline - time.monotonic() - 60.0)
        if rem < 30:
            break
        a_opt, sc_opt = _nesterov_subgradient(a_init, time.monotonic() + rem,
                                               min_remaining=5.0,
                                               epsilon_start=0.003, epsilon_end=0.0003, base_lr=0.6)
        check_update(a_opt, sc_opt, f"multi_n_{n_try}")

        # Quick coord descent
        if time.monotonic() < deadline - 80.0 and sc_opt < best_score + 0.01:
            rem2 = min(60.0, deadline - time.monotonic() - 40.0)
            if rem2 > 15:
                a_cd, sc_cd = _coord_descent_1d(a_opt, time.monotonic() + rem2,
                                                 min_remaining=3.0, max_sweeps=5)
                check_update(a_cd, sc_cd, f"coord_n_{n_try}")

    # ============ Phase 7: Diverse restarts ============
    print(f"\n[P7] Diverse restarts")
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        restarts = [
            ("tapered", _make_profile(n_best, "tapered")),
            ("raised_cos", _make_profile(n_best, "raised_cosine")),
            ("blackman", _make_profile(n_best, "blackman")),
            ("sin_mod", _make_profile(n_best, "sinusoidal_mod")),
            ("perturbed", _perturb(best_a, 0.03, rng)),
            ("perturbed2", _perturb(best_a, 0.08, rng)),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 40.0:
                break
            rem = min(80.0, deadline - time.monotonic() - 25.0)
            if rem < 20:
                break
            a_opt, sc_opt = _nesterov_subgradient(a_r, time.monotonic() + rem,
                                                   min_remaining=3.0,
                                                   epsilon_start=0.005, epsilon_end=0.0005, base_lr=0.5)
            check_update(a_opt, sc_opt, f"restart_{label}")

    # ============ Phase 8: Final ultra-fine polishing ============
    print(f"\n[P8] Final polishing")
    if time.monotonic() < deadline - 5.0:
        # Nesterov with very tight epsilon
        a_fin, sc_fin = _nesterov_subgradient(best_a, deadline - 3.0,
                                              min_remaining=2.0,
                                              epsilon_start=0.0005, epsilon_end=0.00005,
                                              base_lr=0.2)
        check_update(a_fin, sc_fin, "final_nesterov")

        # Brief coordinate descent
        if time.monotonic() < deadline - 2.0:
            a_cd, sc_cd = _coord_descent_1d(best_a, deadline - 1.0,
                                             min_remaining=1.0, max_sweeps=3)
            check_update(a_cd, sc_cd, "final_coord")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    # Scale to have a reasonable sum (avoid numerical issues)
    final = final / final.sum()
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    return [float(x) for x in final]
