"""
Optimized step-function search using adaptive L-BFGS with simulated annealing,
multi-scale optimization, and structured initializations.
Key innovations: (1) iterative explore-exploit loop with temperature annealing,
(2) multi-scale optimization starting from coarse to fine,
(3) chirp-based structured initializations with known good autocorrelation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_and_grad(z, beta, n):
    """Compute LSE-smoothed objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2*n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        return _lse_obj_and_grad(z, self.beta, self.n)[0]
    
    def grad(self, z):
        return _lse_obj_and_grad(z, self.beta, self.n)[1]


def _lbfgs_optimize(a_init, deadline, n_betas=12, min_remaining=15.0):
    """L-BFGS with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Adaptive beta schedule based on the gap between top peaks
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    # Geometric decay of beta
    ratios = np.array([1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04,
                       0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005])
    betas = base_beta * ratios[:n_betas]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    scale = max(1.0, n / 4096.0)
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break
        
        est_iter_time = 0.003 * scale
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 20))))
        
        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxcor': 50, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _projected_gradient_descent(a_init, deadline, lr=None, momentum=0.9, min_remaining=10.0):
    """Project
ed gradient descent with Nesterov momentum on the LSE-smoothed objective."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    if lr is None:
        c = _full_convolve(a, a)
        max_c = float(c.max())
        lr = max_c / (n + 1) * 0.1
    
    v = np.zeros(n)
    lr_decay = 0.999
    lr_min = lr * 0.01
    stale_count = 0
    max_stale = 500
    
    while time.monotonic() < deadline - min_remaining:
        # Nesterov step
        a_nv = a + momentum * v
        a_nv = np.maximum(a_nv, 0)
        s_nv = a_nv.sum()
        if s_nv < 1e-15:

def _directed_perturb(a, magnitude=0.01):
    """Perturb by reducing top contributors to the peak and increasing others."""
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    
    # Contributions to c[k_star]
    k_lo = max(0, k_star - n + 1)
    k_hi = min(n - 1, k_star)
    i_range = np.arange(k_lo, k_hi + 1)
    j_range = k_star - i_range
    contrib = a[i_range] * a[j_range]
    
    # Reduce top 20% contributors, increase bottom 20%
    n_reduce = max(1, len(contrib) // 5)
    top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
    bottom_idx = i_range[np.argsort(contrib)[:n_reduce]]
    
    a_new = a.copy()
    total_reduce = 0
    for idx in top_idx:
        red = a_new[idx] * magnitude
        a_new[idx] -= red
        total_reduce += red
    
    # Distribute to non-peak elements
    mask = np.ones(n, dtype=bool)
    mask[top_idx] = False
    for idx in top_idx:
        j = k_star - idx
        if 0 <= j < n:
            mask[j] = False
    
    if mask.sum() > 0:
        a_new[mask] += total_reduce / mask.sum()
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s

Step-function optimization using targeted directed perturbation with spectral
shaping, adaptive L-BFGS annealing, and multi-scale search.
Key innovations over previous approach:
1. Directed perturbation targeting the peak lag's top contributors
2. Spectral domain perturbation (partial spectrum flattening)
3. Explore-exploit loop with simulated annealing acceptance
4. Multi-scale optimization (coarse-to-fine)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    problem = _LseProblem(beta, n)
    return problem

        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W)


def _lbfgs_optimize(a_init, deadline, min_remaining=15.0, n_betas=None):
    """L-BFGS with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    if n_betas is None:
        n_betas = min(14, max(6, int((deadline - time.monotonic() - min_remaining) / 30)))

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.005)

    ratios = np.array([1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04,
                       0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005])
    betas = base_beta * ratios[:n_betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    scale = max(1.0, n / 4096.0)

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        est_iter_time = 0.003 * scale
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 20))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxcor': 50, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _quick_lbfgs(a_init, max_time=30.0):
    """Quick L-BFGS optimization with limited time budget."""
    n = len(a_init)

def _directed_perturb(a, magnitude=0.02, rng=None):
    """Perturb by reducing top contributors to the peak and increasing others."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))

    k_lo = max(0, k_star - n + 1)
    k_hi = min(n - 1, k_star)
    if k_lo >= k_hi:
        return a.copy()

    i_range = np.arange(k_lo, k_hi + 1)
    j_range = k_star - i_range
    contrib = a[i_range] * a[j_range]

    frac = rng.uniform(0.05, 0.25)
    n_reduce = max(1, int(len(contrib) * frac))
    top_idx_local = np.argsort(contrib)[::-1][:n_reduce]
    top_idx = i_range[top_idx_local]

    a_new = a.copy()
    total_reduce = 0.0
    for idx in top_idx:
        red = a_new[idx] * magnitude
        a_new[idx] -= red
        total_reduce += red

    mask = np.ones(n, dtype=bool)
    for idx in top_idx:
        mask[idx] = False
        j = k_star - idx
        if 0 <= j < n:
            mask[j] = False

    if mask.sum() > 0:
        a_new[mask] += total_reduce / mask.sum()

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _spectral_perturb(a, flatness=0.3):
    """Perturb by partially flattening the spectrum."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    mean_mag = mag.mean()
    new_mag = (1.0 - flatness) * mag + flatness * mean_mag

    new_A = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(new_A, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _random_perturb(a, sigma, rng):
    """Random perturbation in log-space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _chirp_sequence(n, f, phase, taper=0.5, rng=None):
    """Generate a chirp-based sequence with taper."""
    t = np.arange(n, dtype=np.float64)
    chirp = np.cos(2.0 * np.pi * f * t * t / (2.0 * n) + phase)
    shape = 1.0 + 0.3 * chirp
    # Apply taper
    if taper > 0:
        edge = np.linspace(0, 1, n // 4 + 1)
        taper_arr = np.ones(n)
        taper_arr[:n // 4 + 1] = 1.0 - taper * (1.0 - edge)
        taper_arr[n // 4 + 1:] = 1.0
        # Mirror for the other end
        taper_arr[n // 4 + 1:] = 1.0 - taper * (1.0 - np.linspace(1, 0, n - n // 4 - 1))
        shape = shape * taper_arr
    shape = np.maximum(shape, 1e-10)
    return shape / shape.sum()

Hmm, this is getting long. Let me streamline and focus on the most impactful changes.

Let me reconsider. The previous algorithm already has a lot of components. The key question is: what single change would most likely improve the result?

I think the answer is: **better exploration to escape the current local minimum**. The previous algorithm spends most of its time on exploitation (L-BFGS, active-set, peak shaving) and only a small amount on exploration (diverse restarts at the end).

My proposal: **Dedicate 60% of the time to an explore-exploit loop** where we:
1. Perturb the current best (using directed or spectral perturbation)
2. Run quick L-BFGS
3. Accept if improved

This is essentially simulated annealing with a "lazy" acceptance criterion (always accept improvements, and use the temperature only to sometimes accept worsenings).

Let me write a clean, focused implementation:
