```python
def autocorr_matvec(p, w, n):
    """Compute sum_j w[|i-j|] * p[j] for all i."""
    # Build the centered kernel
    K = np.zeros(2 * n - 1)
    K[n-1] = w[0]
    for d in range(1, n):
        K[n-1-d] = w[d]
        K[n-1+d] = w[d]
    
    # Linear convolution of K and p
    out_len = len(K) + n - 1  # = 3n - 2
    N = 1
    while N < out_len:
        N <<= 1
    FK = np.fft.rfft(K, N)
    FP = np.fft.rfft(p, N)
    conv = np.fft.irfft(FK * FP, N)[:out_len]
    
    # Extract elements at indices n-1, n, ..., 2n-2
    return conv[n-1:2*n-1]
```
