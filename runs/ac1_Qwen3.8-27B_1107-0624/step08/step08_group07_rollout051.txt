```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)

    fb = np.fft.rfft(b, N)
```

```python
"""
Optimized step function search using GS initialization, multi-phase Adam
optimization, active-set equalization, and coordinate descent polish.
Combines frequency-domain (GS) and time-domain (Adam/Newton) methods.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_fft(a, w_full):
    """Compute gradient = 2 * cross_correlation(w_full, a) via FFT.
    w_full has length 2n-1, a has length n.
    Returns gradient of length n.
    """
    n = len(a)
    N = _next_pow2(3 * n)  # for linear cross-correlation
    W = np.fft.rfft(w_full, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A, N)
    # cross[i] = sum_k w_full[k] * a[k-i] for valid indices
    # We want indices 0 to n-1
    return 2.0 * cross[:n]
```
