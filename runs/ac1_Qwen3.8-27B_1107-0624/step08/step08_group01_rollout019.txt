"""
Optimized step-function search combining Adam optimization with power-mean
smoothing, adaptive active-set polishing, peak-splitting heuristics, and
systematic multi-length exploration. Uses fundamentally different optimization
dynamics (Adam + momentum) compared to the previous L-BFGS approach, with

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < int(n):
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0.0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _power_mean_obj(a, p, n):
    """Compute power mean of order p of the convolution, times 2n (the score if sum=1)."""
    c = _full_convolve(a, a)
    K = len(c)
    # Power mean: (sum c^p / K)^(1/p)
    c_clipped = np.maximum(c, 1e-30)
    pm = (np.sum(c_clipped ** p) / K) ** (1.0 / p)
    return 2.0 * n * pm


def _lse_obj_and_grad(z, beta, n):
    """LSE objective and gradient in softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    # Gradient of obj wrt z
    # dL/dz_j = 2n * (w @ grad_c) * (a_j - a_j * sum_i w_i * dc_i/dz_j ... )
    # More carefully: LSE = 2n * c_max + 2n*beta*log(sum exp((c-c_max)/beta))
    # dL/dz_j = 2n * (1/beta) * sum_k w_k * dc_k/dz_j
    # dc_k/dz_j = dc_k/da_i * da_i/dz_j = dc_k/da_i * a_i*(delta_ij - a_j)
    # dc_k/da_i = 2*a_{k-i} (if 0<=k-i<n)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # wc[j] = sum_k w[k] * a[k-j] for valid k-j, which is sum_k w_k * dc_k/da_j / 2
    # So sum_k w_k * dc_k/da_j = 2 * wc[j]
    # But we need the middle part: wc[n-1:2n-1] gives us the relevant range
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # dL/dz_j = 2n/beta * (2*h_j * a_j - a_j * 2*W) ... let me redo
    # dL/dz_j = (2n/beta) * sum_k w_k * (dc_k/da_i)(da_i/dz_j) summed over i
    # = (2n/beta) * sum_k w_k * sum_i (2*a_{k-i} * a_i * (delta_ij - a_j))
    # = (2n/beta) * [sum_k w_k * 2*a_{k-j} * a_j - a_j * sum_k w_k * sum_i 2*a_{k-i}*a_i]
    # = (2n/beta) * [2*a_j * (sum_k w_k * a_{k-j}) - a_j * 2 * sum_k w_k * c_k]
    # = (2n/beta) * a_j * [2*h_j - 2*W]
    # where h_j = sum_k w_k * a_{k-j} (for valid range)
    grad = (4.0 * n / beta) * a * (h - W)
    return obj, grad


def _lse_gradient_direct(a, beta, n):
    """Compute LSE gradient directly w.r.t. a (not z). Useful for Adam on a."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # d(LSE)/da_j = (2n/beta) * (2*h_j - 2*W*a_j) ... wait
    # L = 2n*(c_max + beta*log(s))
    # dL/da_j = 2n/beta * sum_k w_k * dc_k/da_j = 2n/beta * 2*(h_j - W*a_j)
    # Wait: dc_k/da_j = 2*a_{k-j} (if valid). So sum_k w_k * dc_k/da_j = 2*sum_k w_k*a_{k-j} = 2*h_j
    # But we also need to account for the c_max term... actually c_max is the max, treated as constant
    # in the LSE formula (we use c - c_max in the exp, and c_max is added back).
    # So dL/da_j = 2n/beta * 2*h_j ... no that's not right either.
    # Let me redo: L = 2n * c_max + 2n*beta * log(sum_k exp((c_k - c_max)/beta))
    # dL/d(c_k) = 2n * (1/beta) * exp((c_k-c_max)/beta) / sum exp = 2n * w_k
    # dL/da_j = sum_k w_k * (2n) * dc_k/da_j = 2n * sum_k w_k * 2*a_{k-j} = 4n*h_j
    # (treating c_max as constant, which is valid since we're differentiating the smooth part)
    grad = 4.0 * n * h
    return c, w, grad


def _adam_step(params, grads, state, lr, t, beta1=0.9, beta2=0.999, eps=1e-8):
    """Single Adam step. Returns updated params and state."""
    if state['m'] is None:
        state['m'] = [np.zeros_like(p) for p in params]
        state['v'] = [np.zeros_like(p) for p in params]
    for i in range(len(params)):
        bc1 = 1 - beta1 ** t
        bc2 = 1 - beta2 ** t
        state['m'][i] = beta1 * state['m'][i] + (1 - beta1) * grads[i]
        state['v'][i] = beta2 * state['v'][i] + (1 - beta2) * grads[i] ** 2
        m_hat = state['m'][i] / bc1
        v_hat = state['v'][i] / bc2
        params[i] = params[i] - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, state


def _optimization_adam_lse(a_init, deadline, n_beta_steps=200):
    """Adam optimizer on log(a) with LSE objective and cosine annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c_init = _full_convolve(a, a)
    c_max = float(c_init.max())
    c_sorted = np.sort(c_init)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    beta_start = max(gap * 2, c_max * 0.05)
    beta_end = c_max * 0.0001

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    state = {'m': None, 'v': None}
    t = 0
    lr_start = 0.5
    lr_end = 0.001

    # Pre-compute timing
    _ = _lse_obj_and_grad(z, beta_start, n)
    t_test = time.monotonic()
    for _ in range(3):
        _lse_obj_and_grad(z, beta_start, n)
    iter_time = (time.monotonic() - t_test) / 3.0

    max_iter = min(50000, max(100, int((deadline - time.monotonic() - 5.0) / iter_time)))

    for step in range(max_iter):
        if time.monotonic() > deadline - 3.0:
            break

        # Cosine annealing for beta
        progress = step / max(1, max_iter - 1)
        beta = beta_start + (beta_end - beta_start) * 0.5 * (1 + np.cos(np.pi * progress))

        # Cosine annealing for learning rate
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))

        obj, grad = _lse_obj_and_grad(z, beta, n)
        t += 1
        z, state = _adam_step([z], [grad], state, lr, t)

        # Evaluate periodically
        if step % 50 == 0 or step == max_iter - 1:
            a_new = _softmax(z)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = _softmax(z - np.log(n))
                s_new = a_new.sum()
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()

    return best_a, best_sc


def _optimization_adam_lse_v2(a_init, deadline):
    """Improved Adam with gradient clipping, warmup, and periodic re-parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    if gap < c_max * 0.001:
        gap = c_max * 0.001
    beta_start = max(gap, c_max * 0.02)
    beta_end = c_max * 0.00005

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Warmup + cosine decay
    warmup_steps = 100
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_peak = 1.0
    grad_clip = 10.0

    # Estimate iteration time
    _, _ = _lse_obj_and_grad(z, beta_start, n)
    t0 = time.monotonic()
    for _ in range(5):
        _lse_obj_and_grad(z, beta_start, n)
    iter_time = max(1e-4, (time.monotonic() - t0) / 5.0)

    total_budget = max(50, int((deadline - time.monotonic() - 3.0) / iter_time))

    for step in range(total_budget):
        if time.monotonic() > deadline - 2.0:
            break

        progress = step / max(1, total_budget - 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Learning rate: warmup then cosine decay
        if step < warmup_steps:
            lr = lr_peak * (step + 1) / warmup_steps
        else:
            lr = lr_peak * 0.5 * (1 + np.cos(np.pi * (step - warmup_steps) / max(1, total_budget - warmup_steps)))

        obj, grad = _lse_obj_and_grad(z, beta, n)

        # Gradient clipping
        gnorm = np.linalg.norm(grad)
        if gnorm > grad_clip:
            grad = grad * (grad_clip / gnorm)

        t += 1
        bc1 = 1 - beta1 ** t
        bc2 = 1 - beta2 ** t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / bc1
        v_hat = v / bc2
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation and re-parameterization
        if step % 20 == 0:
            a_new = _softmax(z)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_a = a_new.copy()
                # Re-center z to avoid floating point issues
                z = z - z.mean()

    # Final evaluation
    a_final = _softmax(z)
    a_final = np.maximum(a_final, 0.0)
    s_final = a_final.sum()
    if s_final > 1e-10:
        a_final = a_final / s_final
        sc_final = _score(a_final)
        if sc_final < best_sc:
            best_sc = sc_final
            best_a = a_final.copy()

    return best_a, best_sc

    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    tol = tol_start
    lr = 0.05
    no_improve = 0
    stall_count = 0

    for outer in range(max_rounds):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive tolerance: tighten when improving, loosen when stuck
        if no_improve > 50:
            tol = max(tol_end, tol * 0.5)
            lr *= 0.7
            no_improve = 0
            if tol <= tol_end * 1.1 and lr < 0.001:
                # Try perturbation to escape
                rng = np.random.default_rng(outer * 12345)
                a = a + 0.001 * rng.standard_normal(n) * c_max
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count += 1
                if stall_count > 20:
                    break
            continue

        # Identify active set
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        elif n_active > 50:
            # Use top-k for efficiency
            top_k = np.argsort(c)[::-1][:20]
            active = np.zeros_like(c, dtype=bool)
            active[top_k] = True
            n_active = 20

        # Compute gradient
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        # Center the gradient (removes component in direction of 1, which doesn't change score)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Line search with quadratic interpolation
        alpha = min(lr, 2.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False
        best_alpha = 0.0
        best_trial_sc = base_sc

        for ls in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_trial_sc:
                best_trial_sc = sc_try
                best_alpha = alpha
            if sc_try < base_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved and best_alpha > 0:
            # Accept the best trial even if not strictly improving (for exploration)
            a_try = a - best_alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
                improved = sc_try < base_sc

        if improved:
            tol = max(tol_end, tol * 0.97)
            lr = min(0.1, lr * 1.02)
        else:
            if no_improve % 10 == 0:
                tol = max(tol_end, tol * 0.8)
                lr *= 0.8

    return best_a, best_sc


def _peak_equalization(a_init, deadline, n_rounds=200):
    """
    Novel peak equalization: explicitly try to equalize the top peaks
    by adjusting the sequence to balance their values.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find top peaks
        top_indices = np.argsort(c)[::-1][:10]
        top_vals = c[top_indices]

        # If the top peak dominates, try to reduce it
        if top_vals[0] > top_vals[-1] * 1.5:
            # Single dominant peak: use gradient from that peak only
            k_star = top_indices[0]
            k_lo = max(0, k_star - n + 1)
            k_hi = min(n - 1, k_star)

            if k_lo < k_hi:
                i_range = np.arange(k_lo, k_hi + 1)
                j_range = k_star - i_range
                contrib = a[i_range] * a[j_range]
                total = contrib.sum()
                if total > 1e-15:
                    # Reduce top contributors, distribute to non-contributors
                    n_top = max(1, int(len(contrib) * 0.3))
                    top_idx = i_range[np.argsort(contrib)[::-1][:n_top]]
                    reduction = 0.0
                    for idx in top_idx:
                        red = a[idx] * 0.05
                        a[idx] -= red
                        reduction += red
                    # Distribute to elements not contributing to this peak
                    low_mask = np.ones(n, dtype=bool)
                    low_mask[i_range] = False
                    low_mask[j_range] = False
                    if low_mask.sum() > 0:
                        a[low_mask] += reduction / low_mask.sum()
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                else:
                    break
            else:
                # Peak is at endpoint, just the square of a[0] or a[n-1]
                idx = int(np.argmax(a))
                a[idx] *= 0.98
                others = np.ones(n, dtype=bool)
                others[idx] = False
                a[others] += a[idx] * 0.02 / others.sum()
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            # Multiple peaks at similar levels: use averaging gradient
            active_w = np.zeros_like(c)
            for idx in top_indices:
                active_w[idx] = 1.0
            active_w = active_w / active_w.sum()
            a_rev = a[::-1]
            wc = _full_convolve(active_w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad = 2.0 * h
            grad_t = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_t)
            if gnorm < 1e-18:
                break
            alpha = 0.02 / (gnorm + 1e-15)
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shaped":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "w_shaped":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.1) ** 2) + np.exp(-30.0 * (t - 0.9) ** 2) + 0.01
        return a / a.sum()
    elif kind == "modulated":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fractal":
        t = np.arange(n, dtype=float)
        a = 1.0
        freq = 1.0
        amp = 1.0
        for octave in range(8):
            a = a + amp * np.cos(2.0 * np.pi * freq * t / n)
            freq *= 2.0
            amp *= 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_flatten(a_init, deadline, n_steps=5000):
    """
    Novel approach: work in the Fourier domain to flatten the power spectrum.
    The goal is to make |DFT(a)|^2 as flat as possible (CAZAC-like),
    which minimizes the peak of the circular autocorrelation.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Work with the DFT
    F = np.fft.rfft(a, 2 * n)
    # Power spectrum
    P = np.abs(F) ** 2
    # We want P to be as flat as possible
    # Gradient of variance(P) wrt a:
    # P_k = |sum a_j exp(-2pi i jk/N)|^2
    # This is complex; let's use a simpler approach.

    # Alternative: iterative "spectral equalization"
    for step in range(n_steps):
        if time.monotonic() > deadline - 2.0:
            break

        F = np.fft.rfft(a, 2 * n)
        P = np.abs(F) ** 2
        P_mean = P.mean()
        P_max = P.max()
        P_min = P.min()

        if P_min < 1e-30:
            P_min = 1e-30

        # Target: make P proportional to 1 (flat)
        # Update: a_new = IDFT(sqrt(P_target) * phase(F))
        # P_target: compress the dynamic range
        P_compressed = P ** 0.3  # Compress dynamic range
        P_compressed = P_compressed / P_compressed.max() * P_max

        # New complex spectrum: keep phase, adjust magnitude
        phase_F = F / np.maximum(np.abs(F), 1e-30)
        F_new = np.sqrt(P_compressed) * phase_F
        a_new = np.fft.irfft(F_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    Phase 1: Adam + LSE on parent (main workhorse)
    Phase 2: Active-set polish v3 (refinement)
    Phase 3: Peak equalization (novel)
    Phase 4: Fourier flattening (novel spectral approach)
    Phase 5: Multi-n exploration with resampling
    Phase 6: Diverse restarts
    Phase 7: Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Adam + LSE (main optimization) ===
    print(f"\n[P1a] Adam+LSE v2 on parent (n={len(best_a)})")
    p1_deadline = min(deadline - 1000.0, t_start + 400.0)
    a1, sc1 = _optimization_adam_lse_v2(best_a, p1_deadline)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [adam_lse_v2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish v3 ===
    print(f"\n[P2] Active-set polish v3 (n={len(best_a)})")
    p2_deadline = min(deadline - 700.0, t_start + 600.0)
    a2, sc2 = _active_set_polish_v3(best_a, p2_deadline, tol_start=0.003, tol_end=0.00001)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [polish_v3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak equalization (novel) ===
    print(f"\n[P3] Peak equalization (n={len(best_a)})")
    p3_deadline = min(deadline - 500.0, t_start + 750.0)
    a3, sc3 = _peak_equalization(best_a, p3_deadline, n_rounds=500)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [peak_equal] t={time.monotonic()-t_start:.1f}s")

    # Re-polish after peak equalization
    if sc3 < best_score + 0.001 and time.monotonic() < deadline - 300:
        p3b_deadline = min(deadline - 300.0, t_start + 820.0)
        a3b, sc3b = _active_set_polish_v3(best_a, p3b_deadline, tol_start=0.001, tol_end=0.000005)
        if sc3b < best_score:
            best_a = a3b.copy()
            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [polish_after_eq] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Fourier flattening (novel) ===
    print(f"\n[P4] Fourier flattening (n={len(best_a)})")
    p4_deadline = min(deadline - 300.0, t_start + 900.0)
    a4, sc4 = _fourier_flatten(best_a, p4_deadline, n_steps=3000)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [fourier] t={time.monotonic()-t_start:.1f}s")
        # Polish after fourier
        if time.monotonic() < deadline - 200:
            a4b, sc4b = _active_set_polish_v3(best_a, min(deadline - 200.0, time.monotonic() + 60.0),
                                              tol_start=0.001, tol_end=0.00001)
            if sc4b < best_score:
                best_a = a4b.copy()
                best_score = sc4b
                print(f"  NEW BEST: {best_score:.10f} [polish_after_fourier] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration")
    n_best = len(best_a)
    n_targets = sorted(set([n_best, n_best * 2, n_best // 2, 4096, 16384, 24576]))

    for n_try in n_targets[:4]:  # Limit to 4 different n values
        if time.monotonic() > deadline - 120.0:
            break
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        p5_dl = min(deadline - 100.0, time.monotonic() + 120.0)

            if sc_pol < best_score:
                best_a = a_pol.copy()

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (n={len(best_a)})")
    n_r = len(best_a)
    restarts = [
        ("pert_small", _perturb(best_a, 0.01, rng)),
        ("pert_med", _perturb(best_a, 0.03, rng)),
        ("pert_large", _perturb(best_a, 0.08, rng)),
        ("u_shaped", _make_profile(n_r, "u_shaped")),
        ("w_shaped", _make_profile(n_r, "w_shaped")),
        ("modulated", _make_profile(n_r, "modulated")),
        ("fractal", _make_profile(n_r, "fractal")),
    ]

    for label, a_r in restarts:
        if time.monotonic() > deadline - 60.0:
            break
        remaining = deadline - time.monotonic() - 30.0
        if remaining < 20.0:
            break
        a_opt, sc_opt = _optimization_adam_lse_v2(a_r, deadline - 30.0)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[P7] Final ultra-fine polish (n={len(best_a)})")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_polish_v3(best_a, deadline - 2.0,
                                              tol_start=0.0005, tol_end=0.000001,
                                              max_rounds=3000)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # Final micro-step: try tiny perturbations in the active-set gradient direction
    if time.monotonic() < deadline - 1.0:
        c_fin = _full_convolve(best_a, best_a)
        c_max = float(c_fin.max())
        active = c_fin >= (c_max - c_max * 0.0001)
        n_act = max(1, int(active.sum()))
        w = np.zeros_like(c_fin)
        w[active] = 1.0 / n_act
        a_rev = best_a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[len(best_a) - 1:2 * len(best_a) - 1]
        grad = 2.0 * h - 2.0 * np.mean(h)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-18:
            for tiny_lr in [0.001, 0.0005, 0.0002, 0.0001]:
                a_try = best_a - tiny_lr / gnorm * grad
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_score:
                        best_a = a_try
                        best_score = sc_try
                        print(f"  NEW BEST: {best_score:.10f} [micro_step_{tiny_lr}] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
        s = float(final.sum())
    final = final / s
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
