def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-15:
        return np.inf

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

        return np.inf

    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute B_p @ v efficiently using FFT.
    (B_p)_{ij} = p_{i+j}/2 for i != j, p_{2i} for i=j (if 2i < 2n-1)

    P_rev = p[::-1]  # length 2n-1

        a = x_new
        t_val = t_new
        
        # Periodic check
        # (skip for speed)

            best_a = a_new.copy()

"""
Minimax alternating optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: uses the minimax duality max_k (a*a)_k = max_p sum_k p_k (a*a)_k
to create a sequence of QP subproblems on the simplex, solved by Nesterov
accelerated projected gradient with FFT-based Hankel matrix-vector products.
Temperature continuation drives convergence. Combined with active-set Newton
refinement and diverse restarts for final polish.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _Bp_matvec(p, v, n):
    """Compute B_p @ v where (B_p)_{ij} = p_{i+j}/2 for i!=j, p_{2i} for i=j.
    Uses FFT-based Hankel matrix-vector product in O(n log n)."""
    P_rev = p[::-1]
    out_len = 3 * n - 2
    N = _next_pow2(out_len)
    fP = np.fft.rfft(P_rev, N)
    fv = np.fft.rfft(v, N)
    conv = np.fft.irfft(fP * fv, N)[:out_len]
    g = conv[n - 1:2 * n - 1][::-1]
    idx = np.arange(n)
    valid = 2 * idx < (2 * n - 1)
    diag = np.zeros(n)
    diag[valid] = p[2 * idx[valid]] * v[valid]
    return 0.5 * (g + diag)


def _Bp_quad(p, a, n):
    """Compute a^T B_p a."""
    return float(np.dot(a, _Bp_matvec(p, a, n)))


def _nesterov_inner(a_init, p, n, deadline, max_iter=3000, step=0.45):
    """Nesterov accelerated projected gradient for min a^T B_p a on simplex."""
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_mom = 1.0

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad_y = 2.0 * _Bp_matvec(p, y, n)
        x_new = _project_simplex(y - step * grad_y)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        y = x_new + beta * (x_new - a)
        if np.min(y) < -1e-10:
            y = _project_simplex(y)
        a = x_new
        t_mom = t_new
    return a


def _fw_inner(a_init, p, n, deadline, max_iter=5000):
    """Frank-Wolfe for min a^T B_p a on simplex."""
    a = a_init.copy()
    a = np.maximum(a, 0.0)

        a = np.ones(n) / n
    else:

    best_a = a.copy()
    best_val = _Bp_quad(p, a, n)

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad = 2.0 * _Bp_matvec(p, a, n)
        s_idx = int(np.argmin(grad))
        e_s = np.zeros(n)
        e_s[s_idx] = 1.0
        gamma = 2.0 / (i + 2)
        a_new = (1.0 - gamma) * a + gamma * e_s
        val_new = _Bp_quad(p, a_new, n)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:

    return a_new / s


def _softmax_p(c, beta):
    """Compute p_k proportional to exp(c_k / beta)."""
    c_shift = c - c.max()
    e = np.exp(np.clip(c_shift / beta, -500, 500))
    s = e.sum()
    if s < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s

        if time.monotonic() > deadline:
            break
        a = _softmax_z(z, n)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        p_norm = w / w.sum()

        grad = 2.0 * _Bp_matvec(p_norm, a, n)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax_z(z, n)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax_z(z, n)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _softmax_z(z, n):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return e / s


def _peak_refine(a_init, deadline, max_rounds=50):
    """Peak-targeted coordinate refinement."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        n_peaks = min(20, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        w = np.zeros_like(c)
        w[top_peaks] = 1.0 / n_peaks
        p_norm = w / w.sum()

        grad = 2.0 * _Bp_matvec(p_norm, a, n)
        grad = grad - np.mean(grad)

        f_old = c_max
        step = 0.5 / (1.0 + 0.05 * rnd)
        for ls in range(25):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using minimax duality:
    Phase 1: Minimax alternating with temperature continuation (main engine)
    Phase 2: Active-set Newton refinement
    Phase 3: Peak-targeted refinement
    Phase 4: Subgradient polish
    Phase 5: Diverse restarts + final active-set
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # === Phase 1: Minimax alternating (main engine) ===
    print(f"\n[Phase 1] Minimax alternating (t={time.monotonic() - t_start:.1f}s)")
    n_main = n_parent if n_parent >= 4096 else 8192

    if time.monotonic() < deadline - 300:
        phase1_deadline = deadline - 300
        a_mm, sc_mm = _minimax_optimize(parent, n_main, phase1_deadline, n_outer=60, n_inner=4000)
        _update_best(a_mm, sc_mm, "minimax_main")

    # === Phase 2: Active-set Newton refinement ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_as, sc_as = _active_set_newton(best_seq, deadline - 120, max_iter=200, tol_active=0.001)
        _update_best(a_as, sc_as, "active_set")

        if time.monotonic() < deadline - 60:
            a_as2, sc_as2 = _active_set_newton(best_seq, deadline - 60, max_iter=100, tol_active=0.0003)
            _update_best(a_as2, sc_as2, "active_set_tight")

    # === Phase 3: Peak-targeted refinement ===
    print(f"\n[Phase 3] Peak refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_pr, sc_pr = _peak_refine(best_seq, deadline - 60, max_rounds=50)
        _update_best(a_pr, sc_pr, "peak_refine")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 30, n_max_steps=2000, lr_init=0.015)
        _update_best(a_sg, sc_sg, "subgrad")

    # === Phase 5: Minimax from perturbed start ===
    print(f"\n[Phase 5] Minimax from perturbed start (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_pert = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z + 0.05 * rng.standard_normal(n_pert)
        z = z - z.mean()
        a_pert = _softmax_z(z, n_pert)
        sc_pert = _score(a_pert)
        print(f"  Perturbed start score: {sc_pert:.10f}")

        if sc_pert < best_score + 0.05:
            a_mm2, sc_mm2 = _minimax_optimize(a_pert, n_pert, deadline - 30, n_outer=30, n_inner=2000)
            _update_best(a_mm2, sc_mm2, "minimax_pert")
            if time.monotonic() < deadline - 10:
                a_as3, sc_as3 = _active_set_newton(a_mm2, deadline - 10, max_iter=50, tol_active=0.001)
                _update_best(a_as3, sc_as3, "as_pert")

    # === Phase 6: Try different n ===
    print(f"\n[Phase 6] Different n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_current = len(best_seq) if best_seq is not None else 8192
    alt_ns = [16384, 32768]
    for n_alt in alt_ns:
        if n_alt == n_current:
            continue
        if time.monotonic() > deadline - 60:
            break
        a_interp = _interpolate_to_n(best_seq, n_alt)
        sc_interp = _score(a_interp)
        print(f"  n={n_alt} init score: {sc_interp:.10f}")
        if sc_interp < best_score + 0.05:
            a_mm3, sc_mm3 = _minimax_optimize(a_interp, n_alt, deadline - 30, n_outer=20, n_inner=1500)
            _update_best(a_mm3, sc_mm3, f"minimax_n{n_alt}")
            if time.monotonic() < deadline - 10:
                a_as4, sc_as4 = _active_set_newton(a_mm3, deadline - 10, max_iter=50, tol_active=0.001)
                _update_best(a_as4, sc_as4, f"as_n{n_alt}")

    # === Phase 7: Final active-set with very tight tolerance ===
    print(f"\n[Phase 7] Final tight active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 3, max_iter=80, tol_active=0.0001)
        _update_best(a_fin, sc_fin, "final_as_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
