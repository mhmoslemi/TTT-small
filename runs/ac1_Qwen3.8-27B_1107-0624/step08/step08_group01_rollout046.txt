def next_pow2(n):
    """Next power of 2 >= n"""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):

    return r

"""
Optimized step-function search algorithm.
Uses POCS initialization, adaptive L-BFGS with log-sum-exp smoothing,
Nesterov-accelerated correlation peak reduction (CPR), and multi-scale
exploration to minimize the autocorrelation peak metric.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Full linear convolution of two sequences using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    """Gradient of LSE smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

class _LseProblem:
    """Wrapper for scipy.optimize.minimize with LSE objective."""
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, beta_init, n_betas, beta_ratio, min_remaining=20.0):
    """
    Run L-BFGS with geometric beta annealing.
    
    Args:
        a_init: initial sequence (normalized, sum=1)
        deadline: absolute time deadline
        beta_init: initial beta
        n_betas: number of beta levels
        beta_ratio: ratio between consecutive betas
        min_remaining: minimum seconds to leave before deadline
    """
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    betas = beta_init * (beta_ratio ** np.arange(n_betas))
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break
        
        # Scale max_iter based on available time and n
        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.003 * scale
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 10))))
        
        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 60,
                    'iprint': -1,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, _score(current_a)

        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    momentum = momentum_init
    
    while time.monotonic() < deadline - 1.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Compute gradient (subgradient of max)
        # Use top-k peaks for a smoother gradient
        k_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros_like(c)
        for k in k_idx:
            weights[k] = 1.0
        weights = weights / weights.sum()
        
        # Compute h = cross_correlation(weights, a)
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(weights, c))
        
        # Gradient in a-space (for the objective 2n*max_c with sum=1 constraint)
        # We want to minimize 2n*c_max, so gradient direction is +h (increase c_max)
        # We subtract from a: a_new = a - lr * (h - W/n) (projected to sum=0 changes)
        grad = h - W / n  # Centered gradient (tangent to simplex)
        
        # Nesterov momentum
        # v = momentum * v + grad
        # a_new = a - lr * v
        # For simplicity, use simple momentum
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
        
        if sc_try < current_sc - 1e-15:
            a = a_try
            # Increase learning rate slightly
            lr = min(0.1, lr * 1.01)
        else:
            # Backtracking
            for _ in range(20):
                lr *= 0.5
                a_try = a - lr * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    break
            else:
                # No improvement, reduce momentum and restart
                lr = max(0.001, lr * 0.5)
                # If stuck for too long, break
                if lr < 0.001:
                    break
    
    return best_a, _score(best_a)


def _pocs_init(n, n_iter=200, seed=42, soft=True):
    """
    POCS-based initialization for approximately flat-spectrum non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    a = a / a.sum()
    
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        # Forward transform
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Project to flat magnitude
        mag = np.abs(A)
        mean_pow = np.mean(mag**2)
        target_mag = np.sqrt(mean_pow)
        mag_safe = np.maximum(mag, 1e-15)
        phase = A / mag_safe
        A_new = phase * target_mag
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, N)[:n].real
        
        # Project to non-negative simplex (soft or hard)
        if soft:
            a_new = np.log1p(np.maximum(a_new, 0.0))
            a_new = np.maximum(a_new, 1e-10)
        else:
            a_new = np.maximum(a_new, 0.0)
        
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a_new / s
    
    return a


def _make_profile(n, kind="uniform"):
    """Create a deterministic profile of length n."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n, dtype=float)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t * 3.0 / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp2":
        t = np.arange(n, dtype=float)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t * 7.0 / n) + 0.1 * np.sin(2.0 * np.pi * t * 13.0 / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    """Perturb a sequence by adding Gaussian noise in log-space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
