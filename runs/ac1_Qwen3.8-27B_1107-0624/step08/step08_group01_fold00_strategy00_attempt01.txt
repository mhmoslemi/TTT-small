def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a non-negative sequence of coefficients. Normalizes and computes the score."""
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    s = np.sum(sequence)
    if s < 1e-15:
        return np.inf
    conv = _full_convolve_with_FFT(sequence)
    if len(conv) == 0 or np.max(conv) < 1e-15:
        return np.inf
    return float((2.0 * len(sequence) * np.max(conv)) / (s ** 2))

def _full_convolve_with_FFT(a):
    n_val = len(a)
    if n_val == 0:
        return np.array([], dtype=float)
    # choose FFT size as next power of 2 of (2*n_val-1)
    s = n_val * 2 - 1
    next_power = 2**np.ceil(np.log2(s)) if hasattr(np, 'log2') else 2**np.ceil(np.log(s))
    N = int(next_power) + 64  # Adding some headroom
    zeros = N - s
    extended = np.zeros(N - n_val + 1, dtype=float)
    padded = np.concatenate([np.array(a, copy=True), extended])
    fa = np.fft.rfft(padded, N)
    if np.any(np.isnan(fa)):
        fa = np.nan_to_num(fa, nan=0.0, posinf=0.0, neginf=0.0)
    fb = np.fft.rfft(np.array(a, copy=True), N)
    prod = fa * fb
    # Note: the FFT length N is chosen dynamically per sequence
    result = np.fft.irfft(prod, N)[:2*n_val-1]
    if np.size(result) > 0 and np.max(result) > 10000000.0: # Security: bound values
        # This is unusual, but note the FFT might be unstable for very high values
        # Scale the result to be reasonable
        max_val = np.max(result)
        if max_val > 0.0:
            result = result / max_val * 1000.0
    return result
