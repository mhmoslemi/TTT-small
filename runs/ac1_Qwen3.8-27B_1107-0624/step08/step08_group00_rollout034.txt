Optimized step function search using Iterative Peak Reduction (IPR) with
multi-scale refinement, parameterized initialization, and active-set polishing.
Key innovations: (1) two-timescale peak equalization, (2) coarse-to-fine
multi-resolution search, (3) adaptive active-set size based on peak structure.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _ipr_step(a, k_peaks, lr):
    """One IPR step: reduce top-k peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Select top-k peaks
    k = min(k_peaks, 2*n-1)
    top_idx = np.argsort(c)[-k:]
    
    # Build weight vector
    w = np.zeros(2*n-1)
    w[top_idx] = 1.0 / k
    
    # Compute gradient: grad[j] = 2 * sum_p w[p] * a[p-j]
    v = w[::-1]
    conv_w = _full_convolve(a, v)
    grad = 2.0 * conv_w[n-1:2*n-1][::-1]
    
    # Center the gradient (to preserve sum)
    grad = grad - grad.mean()
    
    # Take step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, c_max
    a_new = a_new / s
    
    return a_new, c_max

def _ipr_optimize(a_init, deadline, time_budget, k_start=100, k_end=5, 
                  lr_start=0.1, lr_end=0.001):
    """IPR optimization with adaptive k and lr."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()
    t_end = t_start + time_budget
    
    no_improve = 0
    max_stall = 200
    
    while time.monotonic() < min(deadline, t_end):
        elapsed = time.monotonic() - t_start
        frac = elapsed / time_budget
        
        # Adaptive k: start large, shrink over time
        k = max(k_end, int(np.exp(np.log(k_start) + frac * np.log(k_end / k_start))))
        
        # Adaptive lr
        lr = np.exp(np.log(lr_start) + frac * np.log(lr_end / lr_start))
        
        a_new, c_max = _ipr_step(a, k, lr)
        sc = 2.0 * n * c_max  # since sum(a)=1
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve >= max_stall:
            # Restart with smaller lr
            lr_start *= 0.5
            lr = lr_start * 0.5
            no_improve = 0
            if lr_start < 1e-6:
                # Try a random perturbation
                a = best_a.copy()
                noise = np.random.default_rng(int(time.monotonic()*1000) % 10000).standard_normal(n) * 0.01
                a = np.maximum(a + noise, 0)
                a = a / a.sum()
    
    return best_a, best_sc

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) Fourier-parametrized global search, (2) Iterative Peak Reduction (IPR)
with adaptive FFT-based gradients, (3) active-set polish with dual-gap diagnostics,
(4) L-BFGS log-sum-exp smoothing, (5) random restarts to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _ipr_gradient(a, w):
    """Compute grad[j] = 2*sum_p w[p]*a[p-j] using FFT."""
    n = len(a)
    v = w[::-1]
    conv_w = _full_convolve(a, v)
    return 2.0 * conv_w[n - 1:2 * n - 1][::-1]


def _ipr_step(a, k_peaks, lr, rng=None):
    """One IPR step: reduce top-k peaks with adaptive weighting."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a, c_max

    # Select top-k peaks with soft weighting
    k = min(k_peaks, 2 * n - 1)
    # Use a soft threshold: weight peaks above (1-1/k)*c_max
    threshold = c_max * (1.0 - 1.0 / max(k, 2))
    w = np.zeros(2 * n - 1)
    above = c > threshold
    if above.sum() == 0:
        idx = int(np.argmax(c))
        w[idx] = 1.0
    else:
        # Soft weight based on peak height
        w[above] = (c[above] - threshold)
        ws = w.sum()
        if ws > 1e-30:
            w /= ws
        else:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0

    # Compute gradient
    grad = _ipr_gradient(a, w)
    # Center to preserve sum
    grad = grad - grad.mean()
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, c_max

    # Step with line search
    step = lr * grad / gnorm
    a_old_sc = c_max
    best_a = a.copy()
    best_sc = a_old_sc
    alpha = 1.0
    for _ in range(30):
        a_try = a - alpha * step
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            alpha *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        sc_try = float(c_try.max())
        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
        alpha *= 0.7
        if alpha < 1e-10:
            break

    return best_a, best_sc


def _ipr_optimize(a_init, deadline, time_budget, k_start=200, k_end=5,
                  lr_start=0.5, lr_end=0.01, verbose=True):
    """IPR optimization with adaptive k and lr, line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)

    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < t_end:
        elapsed = time.monotonic() - t_start
        frac = min(1.0, elapsed / max(1e-6, time_budget))

        # Adaptive k: start large, shrink
        k = max(k_end, int(np.exp(np.log(max(k_start, 2)) * (1 - frac) + np.log(max(k_end, 1)) * frac)))
        # Adaptive lr
        lr = np.exp(np.log(max(lr_start, 1e-8)) * (1 - frac) + np.log(max(lr_end, 1e-8)) * frac)

        a_new, c_max = _ipr_step(a, k, lr)
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
            if verbose and no_improve == 0 and best_sc < 1.51:
                print(f"    IPR: score={best_sc:.10f} t={time.monotonic()-t_start:.1f}s k={k}")
        else:
            no_improve += 1

        a = a_new

        if no_improve >= max_stall:
            no_improve = 0
            stall_count += 1
            # Try random perturbation from best
            if stall_count % 3 == 0:
                sigma = 0.005 * (stall_count // 3 + 1)
                rng = np.random.default_rng(stall_count * 12345 + 678)
                a = best_a + sigma * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            else:
                # Try smaller k and lr
                k_start = max(k_end, k_start * 0.5)
                lr_start = lr_start * 0.5
                a = best_a.copy()
            if stall_count > 30:
                break

    return best_a, best_sc


def _fourier_param_search(n, K, deadline, n_trials=20):
    """Search for good shape using K Fourier modes with softmax parameterization."""
    j = np.arange(n)
    best_a = None
    best_val = np.inf
    rng = np.random.default_rng(12345)

    def a_from_coeffs(coeffs):
        """Reconstruct a from Fourier coefficients using softmax."""
        b = np.zeros(n)
        for k_idx in range(min(K + 1, n // 2)):
            ck = coeffs[2 * k_idx]
            sk = coeffs[2 * k_idx + 1] if 2 * k_idx + 1 < len(coeffs) else 0.0
            freq = 2.0 * np.pi * (k_idx + 1) * j / n
            if k_idx == 0:
                b += ck * np.cos(freq)
            else:
                b += ck * np.cos(freq) + sk * np.sin(freq)
        a = np.exp(b - b.max())
        a = a / a.sum()
        return a, b

    def objective(coeffs):
        a, _ = a_from_coeffs(coeffs)
        c = _full_convolve(a, a)
        return c.max()

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Random initialization: small coefficients
        scale = 0.1 / (trial + 1)
        x0 = rng.standard_normal(2 * (K + 1)) * scale

        try:
            result = minimize(objective, x0, method='Nelder-Mead',
                             options={'maxiter': 2000, 'xatol': 1e-12,
                                      'fatol': 1e-14, 'adapt': True})
            a, _ = a_from_coeffs(result.x)
            val = float(_full_convolve(a, a).max())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                if verbose:
                    print(f"    Fourier trial {trial}: score={2*n*val:.8f}")
        except Exception:
            continue

    if best_a is None:
        return None
    return best_a


def _fourier_param_search_fast(n, K, deadline, verbose=False):
    """Faster version: use gradient-based optimization in Fourier space."""
    j = np.arange(n)
    best_a = None
    best_val = np.inf
    rng = np.random.default_rng(99999)

    def a_from_coeffs(coeffs):
        b = np.zeros(n)
        for k_idx in range(K + 1):
            freq = 2.0 * np.pi * (k_idx + 1) * j / n
            b += coeffs[2 * k_idx] * np.cos(freq)
            if 2 * k_idx + 1 < len(coeffs):
                b += coeffs[2 * k_idx + 1] * np.sin(freq)
        a = np.exp(b - b.max())
        a = a / a.sum()
        return a

    class FObj:
        def __init__(self, n, K, beta):
            self.n = n
            self.K = K
            self.beta = beta
            self.j = np.arange(n)

        def __call__(self, coeffs):
            b = np.zeros(self.n)
            for k_idx in range(self.K + 1):
                freq = 2.0 * np.pi * (k_idx + 1) * self.j / self.n
                b += coeffs[2 * k_idx] * np.cos(freq)
                if 2 * k_idx + 1 < len(coeffs):
                    b += coeffs[2 * k_idx + 1] * np.sin(freq)
            a = np.exp(b - b.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / self.beta
            e = np.exp(np.clip(shifted, -50, 0))
            s = e.sum()
            if s < 1e-30:
                return 2.0 * self.n * c_max
            return 2.0 * self.n * (c_max + self.beta * np.log(s))

        def grad(self, coeffs):
            b = np.zeros(self.n)
            for k_idx in range(self.K + 1):
                freq = 2.0 * np.pi * (k_idx + 1) * self.j / self.n
                b += coeffs[2 * k_idx] * np.cos(freq)
                if 2 * k_idx + 1 < len(coeffs):
                    b += coeffs[2 * k_idx + 1] * np.sin(freq)
            a = np.exp(b - b.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / self.beta
            e = np.exp(np.clip(shifted, -50, 0))
            s = e.sum()
            if s < 1e-30:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            # Gradient of w @ c w.r.t. a: grad_a = _ipr_gradient(a, w)
            grad_a = _ipr_gradient(a, w)
            grad_a = grad_a - grad_a.mean()  # center for normalization invariance
            W = float(np.dot(w, c))
            # d(2n*(c_max + beta*log(s)))/d(a) through softmax
            # da/db = a (for softmax)
            # db/dcoeffs: need Jacobian
            # Simplified: use finite differences for the Fourier coefficients
            # Actually, let's just use the chain rule approximately
            # The gradient w.r.t. b is: grad_b = a * (grad_a - W*a) ... 
            # Hmm, this is getting complicated. Let's just use Nelder-Mead.
            return None

    # For reliability, use Nelder-Mead with multiple starts
    for trial in range(15):
        if time.monotonic() > deadline:
            break
        scale = 0.05 / np.sqrt(trial + 1)
        x0 = rng.standard_normal(2 * (K + 1)) * scale

        def obj(x):
            b = np.zeros(n)
            for k_idx in range(K + 1):
                freq = 2.0 * np.pi * (k_idx + 1) * j / n
                b += x[2 * k_idx] * np.cos(freq)
                if 2 * k_idx + 1 < len(x):
                    b += x[2 * k_idx + 1] * np.sin(freq)
            a = np.exp(b - b.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            return c.max()

        try:
            result = minimize(obj, x0, method='Nelder-Mead',
                             options={'maxiter': 3000, 'xatol': 1e-11, 'fatol': 1e-13})
            b = np.zeros(n)
            for k_idx in range(K + 1):
                freq = 2.0 * np.pi * (k_idx + 1) * j / n
                b += result.x[2 * k_idx] * np.cos(freq)
                if 2 * k_idx + 1 < len(result.x):
                    b += result.x[2 * k_idx + 1] * np.sin(freq)
            a = np.exp(b - b.max())
            a = a / a.sum()
            val = float(_full_convolve(a, a).max())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                if verbose:
                    print(f"    Fourier L trial {trial}: score={2*n*val:.10f}")
        except Exception:
            continue

    return best_a


def _active_set_polish(a_init, deadline, time_budget, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05, verbose=False):
    """Active-set gradient polishing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)

    while time.monotonic() < t_end:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            if verbose and best_sc < 1.51:
                print(f"    Polish: score={best_sc:.10f} t={time.monotonic()-t_start:.1f}s")

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        grad = _ipr_gradient(a, w)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        a_old_sc = current_sc
        for _ in range(40):
            a_try = a - alpha * grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
