"""
Multi-strategy optimizer combining LP-based SOTA iteration, adaptive LSE annealing,
Newton active-set polish, multi-n exploration, and aggressive final refinement
for minimizing the step function autocorrelation peak ratio.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _lp_solve_topk(f, K=50):
    n = len(f)
    c = _full_convolve(f, f)
    c_max = float(c.max())
    if c_max < 1e-30:
        return None
    top_k = np.argsort(c)[::-1][:K]
    c_obj = -np.ones(n)
    A_mat = np.zeros((min(K, len(top_k)), n))
    b_mat = np.full(len(top_k), c_max)
    for k_idx in range(len(top_k)):
        k = int(top_k[k_idx])
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            A_mat[k_idx, js] = f[k - js]
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_mat,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _lp_solve_full(f):
    n = len(f)
    c = _full_convolve(f, f)
    c_max = float(c.max())
    if c_max < 1e-30:
        return None
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    A_mat = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    b_mat = np.full(num_constraints, c_max)
    try:
        res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_mat,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

            return _hess_matvec(v, active_f, n) + mu * v
        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)

def _perturb_fourier_phase(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)

        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.4) ** 2) / (2 * 0.18 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):

    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:

    return a_target / s
