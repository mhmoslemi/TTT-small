def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

The key insight I want to exploit is: **when the active set is stable, we can use direct Newton (not CG) for quadratic convergence**. For an active set of size m, the KKT system is:

H * dx + g_t + lambda * 1 = 0
1^T * dx = 0

where H is the (regularized) Hessian of the average of the active quadratics. This is a (n+1) x (n+1) system. For n=16384, this is a 16385 x 16385 system, which is too large to solve directly. But we can use the structure (H is a sum of shift matrices, i.e., it's circulant-like) to solve it with FFT.

Actually, the Hessian H = (2/m) * sum_{k in A} T_k + mu * I. The sum of shift matrices can be applied via FFT in O(n log n). So CG is the right approach for applying H, but we want faster convergence.

The issue with CG is that it converges in at most n iterations (for exact arithmetic), but in practice the convergence rate depends on the condition number of H. If H is ill-conditioned, CG is slow.

The regularized H = H_true + mu * I has condition number at most (||H_true|| + mu) / mu. If mu is small, the condition number is large. If mu is large, we're solving a different problem.

A better approach: use the fact that H_true is a sum of shift matrices. The eigenvalues of H_true are (2/m) * sum_{k in A} exp(2*pi*i*j*k/n) for j = 0, ..., n-1. We can compute these explicitly and use them to precondition CG!

This is a **spectral preconditioner**. The preconditioner M = H_true^{-1} (approximated via its eigenvalues) would make the preconditioned system have all eigenvalues equal to 1, giving convergence in one iteration.

In practice, we can't compute H_true^{-1} exactly (it's n x n), but we CAN apply it:
1. Compute the eigenvalues of H_true via FFT of the active set
2. The eigenvectors are the Fourier basis
3. So M * v = IFFT(eigenvalues^{-1} * FFT(v))

This is O(n log n) per application, same as applying H!

"""
Advanced optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. Spectral preconditioner for active-set Newton (accelerates CG convergence)
2. Adaptive active set with hysteresis to avoid oscillation
3. Multi-scale initialization (solve coarser problems first)
4. Aggressive L-BFGS with adaptive beta continuation
5. Exploration of nearby n values
6. Smooth penalty as a fallback
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """Compute sum_{k in A} (T_k + T_k^T) a using FFT."""
    # (T_k a)_j = a[j+k] (for valid indices) ≈ a[(j+k) mod n] (circulant approx)
    # (T_k^T a)_j = a[j-k]
    # So (H_k a)_j = a[j+k] + a[j-k]
    # We compute sum_{k in A} (a[j+k] + a[j-k]) for all j simultaneously.
    # = sum_{k in A} a[j+k] + sum_{k in A} a[j-k]
    # First term: correlate a with A (shifted forward)
    # Second term: correlate a with A (shifted backward) = correlate a with A (since A is symmetric-like)
    
    # Actually, let me use the direct FFT approach:
    # sum_{k in A} (T_k + T_k^T) a = IFFT(FFT(A_padded) * (FFT(a) + FFT(a_filtered))^*)
    # This is getting complicated. Let me use the simpler approach:
    # corr(a, A) where corr is the cross-correlation.
    
    # sum_k A[k] * a[j+k] for all j: this is a cross-correlation
    # sum_k A[k] * a[j-k] for all j: this is also a cross-correlation (with A flipped)
    
    # Let's just use the original method: _corr_a_A computes the gradient of the sum
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_eigenvalues(A_vec, n):
    """Compute eigenvalues of the circulant Hessian approximation.
    Returns eigenvalue vector of length n//2+1 (for rfft).
    """
    m = int(A_vec.sum())
    if m == 0:
        return np.ones(n // 2 + 1)
    
    # Create a length-n vector from the active set
    # Active set indices are in [0, 2n-2]
    # For k < n: eigenvalue contribution is 2*cos(2*pi*j*k/n)
    # For k >= n: H_k = 0 (as showed above)
    # So we only need k in [0, n-1]
    
    A_n = np.zeros(n)
    valid = A_vec[:n]  # Only lags 0 to n-1
    A_n[:] = valid

    # = (4/m) * Re(FFT(A_n)[j])
    # For rfft, we get frequencies 0, 1, ..., n//2
    
    A_fft = np.fft.rfft(A_n)
    eig = (4.0 / m) * np.real(A_fft)
    
    # Ensure positive (add regularization later)
    return eig

def _run_lbfgs_v2(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[100]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    # More aggressive beta schedule
    n_betas = 30
    ratio = 0.55
    betas = base_beta * (ratio ** np.arange(n_betas))
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        if n <= 4096:
            est_time_per_iter = 0.004
        elif n <= 8192:
            est_time_per_iter = 0.008
        elif n <= 16384:
            est_time_per_iter = 0.016
        else:
            est_time_per_iter = 0.03 * (n / 32768.0)
        
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 20))))
        
        try:
            t0 = time.monotonic()
            result = minimize(
                _lsobj_impl,
                z,
                jac=_lsgrad_impl,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _lsobj_impl(x, beta, n):
    return _lsobj(x, beta, n)


def _lsgrad_impl(x, beta, n):
    return _lsgrad(x, beta, n)


def _penalty_refine(a_init, deadline, max_iters=2000):
    """Smooth penalty method: minimize sum_k max(0, c_k - tau)^2 with adaptive tau."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tau = c_max * 0.99
    penalty = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Gradient of sum_k max(0, c_k - tau)^2
        # = 2 * sum_{k: c_k > tau} (c_k - tau) * (T_k + T_k^T) a
        excess = np.maximum(c - tau, 0)
        w = excess / max(excess.sum(), 1e-30)
        
        g = 2.0 * _corr_a_A(a, w, n)
        g = g - np.mean(g)  # Remove mean (sum constraint)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Line search
        f_old = float(excess[excess > 0].sum())
        t = 1.0 / gnorm
        for ls in range(30):
            a_try = a - t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            excess_try = np.maximum(c_try - tau, 0)
            f_try = float(excess_try[excess_try > 0].sum())
            if f_try < f_old:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            t *= 0.5
        
        # Adapt tau and penalty
        if it % 50 == 0 and it > 0:
            c_new = _full_convolve(a, a)
            new_max = float(c_new.max())
            tau = max(tau * 0.98, new_max * 0.95)
    
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _multi_scale_solve(a_init, n_final, deadline):
    """Solve at smaller n first, then upsample."""
    n_current = len(a_init)
    scales = [min(n_current, 1024), min(n_current, 4096), min(n_current, 8192)]
    scales = [s for s in scales if s < n_current]
    scales.append(n_current)
    
    a = a_init.copy()
    
    for n_scale in scales:
        if n_scale < n_current:
            a_coarse = _interpolate_to_n(a, n_scale)
        else:
            a_coarse = a.copy()
        
        # Solve at this scale
        t_rem = deadline - time.monotonic()
        if t_rem < 30:
            break
        a_opt = _run_lbfgs_v2(a_coarse, deadline - 60, min_time_remaining=30)
        sc_opt = _score(a_opt)
        
        # Upsample back
        if n_scale < n_current:
            a = _interpolate_to_n(a_opt, min(2 * n_scale, n_current))
        else:
            a = a_opt
    
    return a
