"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses refined active-set Newton with adaptive tolerance,
multi-n exploration, diverse initializations, and L-BFGS polish.
Key improvements over previous:
- Adaptive tolerance shrinking in Newton
- Better line search with curvature information
- More diverse initializations (Gaussian, power-law profiles)
- Cyclic Newton -> L-BFGS -> Newton pattern
- Improved time management
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _corr(a, S_vec, n):
    """Cross-correlation: result[i] = sum_j a[j] * S_vec[j+i] for i=0..n-1"""
    flip_a = a[::-1]
    out_len = n + len(S_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fS = np.fft.rfft(S_vec, N)
    result = np.fft.irfft(fa * fS, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_vec(v, S_vec, n, scale):
    """Hessian-vector product: (2/|S|) * corr(v, S_vec, n)"""
    return scale * _corr(v, S_vec, n)


def _active_set_newton(a_init, deadline, max_iter=500):
    """
    Active-set Newton method for minimizing max_k b[k]
    subject to sum(a)=1, a>=0.
    Uses adaptive tolerance and curvature-informed line search.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_cmax = float(np.max(_full_convolve(a, a)))

    # Adaptive tolerance schedule
    tol_start = 0.02
    tol_end = 0.0005
    tol_decay = max(1, max_iter // 4)  # start decaying after this many iterations

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(np.max(c))

        # Adaptive tolerance
        if it < tol_decay:
            tol = tol_start
        else:
            progress = (it - tol_decay) / max(1, max_iter - tol_decay)
            tol = tol_start * (tol_end / tol_start) ** progress
            tol = max(tol, tol_end)

        # Active set
        active = c >= (c_max - tol * c_max)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())

        # Limit active set size
        if m > 8 * n:
            top_idx = np.argsort(c)[::-1][:8 * n]
            active = np.zeros_like(c, dtype=bool)
            top_idx_valid = top_idx[top_idx < len(active)]
            active[top_idx_valid] = True
            m = int(active.sum())
        if m < 2:
            top_idx = np.argsort(c)[::-1][:max(2, n // 8)]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = int(active.sum())

        S_vec = active.astype(np.float64)
        scale = 2.0 / m

        # Gradient
        g = scale * _corr(a, S_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)

        if gnorm < 1e-14:
            break

        # Hessian regularization
        mu = 1e-8 * c_max / max(m, 1) + 1e-10

        def hess_vp(v):
            return scale * _corr(v, S_vec, n) + mu * v

        # Solve Newton system with CG
        # We need: (H + mu*I) dx + lambda*1 = -g_t, sum(dx) = 0
        # Substituting lambda = -(1^T(H+mu*I)^{-1} g_t) / (1^T(H+mu*I)^{-1} 1):
        # (H+mu*I) dx = -g_t + (1^T(H+mu*I)^{-1} g_t / 1^T(H+mu*I)^{-1} 1) * 1

        # First solve (H+mu*I) u = g_t
        x0 = np.zeros(n)
        r = g_t - hess_vp(x0)
        p = r.copy()
        rs = np.dot(r, r)
        max_cg = 200
        u = np.zeros(n)
        for _ in range(max_cg):
            Ap = hess_vp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            u = u + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        # Solve (H+mu*I) w = 1
        x0 = np.zeros(n)
        r = np.ones(n) - hess_vp(x0)
        p = r.copy()
        rs = np.dot(r, r)
        w = np.zeros(n)
        for _ in range(max_cg):
            Ap = hess_vp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            w = w + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        w_sum = float(np.sum(w))
        if abs(w_sum) < 1e-15:
            dx = -u
        else:
            lam = float(np.dot(u, g_t)) / w_sum  # (1^T u) * (1^T g_t) / (1^T w) ... 
            # Actually: lambda = (1^T u) / (1^T w) ... no
            # lambda = -(1^T (H+mu*I)^{-1} g_t) / (1^T (H+mu*I)^{-1} 1)
            # = -(1^T u) / (1^T w) ... but u = (H+mu*I)^{-1} g_t
            # Wait, we solved (H+mu*I) u = g_t, so u = (H+mu*I)^{-1} g_t
            # And (H+mu*I) w = 1, so w = (H+mu*I)^{-1} 1
            # lambda = -(1^T u) / (1^T w) = -sum(u) / sum(w)
            lam = -float(np.sum(u)) / w_sum
            dx = -u + lam * w

        dx = dx - np.mean(dx)

        # Line search with backtracking
        t_max = 1.0
        neg = dx < -1e-12
        if neg.any():
            t_max = float(np.min(a[neg] / (-dx[neg]))) * 0.95
            t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.max(c_try))
            # Armijo condition
            if f_try <= f_old - 1e-4 * t * gnorm or ls >= 50:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Try a steeper step
            t = 0.1
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(np.max(_full_convolve(a_try, a_try)))
                if f_try < f_old - 1e-13:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break

        sc = _score(a)
        c_new_max = float(np.max(_full_convolve(a, a)))
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        if c_new_max < best_cmax:
            best_cmax = c_new_max

        # Convergence check
        if abs(c_new_max - f_old) < 1e-14:
            break

    return best_a, best_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(np.max(c_cur))
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                   0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 5))))

        try:
            result = minimize(
                lambda z_l: _lsobj(z_l, beta, n),
                z,
                jac=lambda z_l: _lsgrad(z_l, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t**2)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * t) ** 2 * (0.5 + 0.5 * (1 - t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block = n // 4
        a[:block] = 1.0
        a[3 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "gauss_wide":
        t = np.linspace(-1, 1, n)
        a = np.exp(-0.5 * t**2)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "gauss_narrow":
        t = np.linspace(-1, 1, n)
        a = np.exp(-8.0 * t**2)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _spectral_smooth(a, deadline, n_steps=100):
    """
    Spectral smoothing: iteratively modify the frequency domain
    representation to reduce the autocorrelation peak.
    """
    n = len(a)
    a_cur = a.copy()
    a_cur = np.maximum(a_cur, 0)
    s = a_cur.sum()
    if s < 1e-15:
        return a
    a_cur = a_cur / s
    best_a = a_cur.copy()
    best_sc = _score(a_cur)

    N = _next_pow2(2 * n - 1)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a_cur, N)
        c = np.fft.irfft(np.abs(A)**2, N)[:2*n-1]
        c_max = float(np.max(c))
        
        # Compute gradient of max(c) w.r.t. log|A|^2
        active = c >= c_max - 0.01 * c_max
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros(N // 2 + 1)
        w[:len(active)] = active.astype(np.float64)
        w = w / w.sum()

        # Gradient: d(max c)/d(a) = 2 * corr(w, a) ... 
        # Actually in frequency domain: c = IDFT(|A|^2)
        # dc/d(a) involves complex operations. Let's just use time-domain gradient.
        g = 2.0 * _corr(a_cur, w[:2*n-1], n)
        g_t = g - np.mean(g)

        # Small step
        a_try = a_cur - 0.01 * g_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
            a_cur = a_try
        else:
            break

    return best_a

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Refined active-set Newton + L-BFGS + multi-n + restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


# Global parameters for L-BFGS objective/gradient wrappers
_lsg = {'beta': 0.0, 'n': 0}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _corr(a, S_vec, n):
    """result[i] = sum_j a[j] * S_vec[j+i] for i=0..n-1"""
    flip_a = a[::-1]
    out_len = n + len(S_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fS = np.fft.rfft(S_vec, N)
    result = np.fft.irfft(fa * fS, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z):
    beta = _lsg['beta']
    n = _lsg['n']
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))
