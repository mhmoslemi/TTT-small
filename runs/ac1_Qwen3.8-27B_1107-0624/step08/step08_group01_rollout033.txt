def spectral_grad_descent(a_init, deadline, n_max_steps=1000000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Adaptive threshold
    c = _full_convolve(a, a)
    c_mean = float(c.mean())  # = 1/(2n-1) approximately
    c_max = float(c.max())
    
    # Schedule: M goes from c_max * 0.99 down to c_mean
    # Use cosine annealing over total steps
    M_start = c_max * 0.99
    M_end = c_mean * 1.05
    
    prev_step = None
    prev_grad = None
    
    for step in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        if step % 10000 == 0:
            print(f"  SGD step {step}, score={_score(a):.10f}")
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
        
        # Compute threshold
        progress = min(1.0, step / 50000)  # 50000 step cosine schedule
        M = M_end + (M_start - M_end) * (1 + np.cos(np.pi * progress)) / 2
        
        # Compute penalty weights
        w = np.maximum(0.0, c - M)
        w = w * w  # quadratic penalty -> w[k] = (c[k]-M)^2 for c[k]>M
        
        # Compute gradient
        N = 1
        while N < 3 * n:
            N *= 2
        W = np.fft.rfft(w, N)
        A = np.fft.rfft(a, N)
        grad = np.fft.irfft(np.conj(W) * A, N)[:n]
        grad = 2.0 * grad
        
        # Project gradient onto tangent space (sum constraint)
        grad = grad - grad.mean()
        
        # Barzilai-Borwein step size
        if prev_step is not None and np.dot(prev_step, prev_step) > 1e-30:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if yy > 1e-15 and sy > 0:
                alpha = ss / sy
            else:
                alpha = 0.1 / (np.linalg.norm(grad) + 1e-15)
        else:
            alpha = 1.0 / (np.linalg.norm(grad) + 1e-15)
        
        alpha = min(alpha, 1.0)  # cap the step size
        
        # Backtracking line search
        best_alpha = 0.0
        for bt in range(20):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_score - 1e-15:
                best_alpha = alpha
                a = a_try
                break
            alpha *= 0.5
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        prev_step = step
    
    return best_a, best_score

"""
Spectral gradient descent + L-BFGS annealing + active-set polishing for minimizing
2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences a of length n.
New: FFT-based spectral gradient with L1 and L2 excess penalties, adaptive threshold
scheduling, Barzilai-Borwein step sizes, and multi-scale multi-init search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_l1(w, a, n):
    """Gradient of (1/2)*sum_k max(0,c[k]-M)^2 * 0 + sum_k max(0,c[k]-M) w.r.t. a
    which is 2*sum_k w[k]*a[k-i] where w[k]=1 for c[k]>M."""
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)[:n]
    return 2.0 * corr


def _grad_l2(w, a, n):
    """Gradient of (1/2)*sum_k (max(0,c[k]-M))^2 w.r.t. a_i = 2*sum_k w[k]^2 * a[k-i]"""
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w * w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(W) * A, N)[:n]
    return 2.0 * corr


def _spectral_gd(a_init, deadline, penalty="l2", verbose=True):
    """Spectral gradient descent with adaptive threshold."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    c = _full_convolve(a, a)
    c_mean = 1.0 / (2 * n - 1)
    c_max = float(c.max())

    # Threshold schedule
    M_start = c_max * 0.995
    M_end = c_mean * 1.02

    lr_base = 0.8 / (n + 1.0)
    prev_a = None
    prev_grad = None

    total_steps = 0
    no_improve = 0

    while time.monotonic() < deadline:
        batch_end = min(deadline, time.monotonic() + 5.0)
        while time.monotonic() < batch_end:
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max

            if current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 200000:
                break

            # Adaptive threshold: cosine schedule
            progress = min(1.0, total_steps / 30000.0)
            M = M_end + (M_start - M_end) * (1.0 + np.cos(np.pi * progress)) / 2.0

            if penalty == "l1":
                w = (c > M).astype(np.float64)
                grad = _grad_l1(w, a, n)
            else:
                excess = np.maximum(0.0, c - M)
                w = excess
                grad = _grad_l2(w, a, n)

            grad = grad - grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-20:
                break

            # Step size
            if prev_a is not None:
                s_vec = a - prev_a
                y_vec = grad - prev_grad
                sy = float(np.dot(s_vec, y_vec))
                ss = float(np.dot(s_vec, s_vec))
                if sy > 1e-15:
                    alpha = ss / sy
                else:
                    alpha = lr_base
            else:
                alpha = lr_base

            alpha = min(alpha, 2.0 / (gnorm + 1e-15))
            alpha = max(alpha, 1e-10)

            # Try step with limited backtracking
            improved = False
            for bt in range(8):
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-12:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                # Try a smaller fixed step
                a_try = a - 0.1 * alpha * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-12:
                    a = a_try / s_try

            prev_a = a.copy()
            prev_grad = grad.copy()
            total_steps += 1

        if verbose and total_steps % 50000 == 0:
            print(f"    SGD({penalty}): step={total_steps}, score={best_sc:.10f}, t={time.monotonic()-0:.0f}s elapsed")

        if total_steps > 200 and (no_improve > 50000):
            break

    return best_a, best_sc


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_remaining=20.0):
    """L-BFGS with log-sum-exp annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
         0.00005, 0.00002, 0.00001, 0.000005]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 15))))

        def f(z_val):
            return _lse_obj(z_val, beta, n)

        def g(z_val):
            return _lse_grad(z_val, beta, n)

        try:
            result = minimize(
                f, z, jac=g, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_start=0.004, tol_end=0.00001, lr_start=0.025):
    """Active-set GD with top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0

    while time.monotonic() < deadline - 1.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > 20:
            top_indices = np.argsort(c)[::-1][:5]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = 5

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False

        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve % 50 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= 5000:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.5
                no_improve = 0
                if tol <= tol_end * 1.1 and lr < lr_start * 0.1:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.005)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.1) ** 2) + np.exp(-30.0 * (t - 0.9) ** 2) + 0.01
        return a / a.sum()
    elif kind == "trimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.1) ** 2) + np.exp(-40.0 * (t - 0.5) ** 2) + np.exp(-40.0 * (t - 0.9) ** 2) + 0.005
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(999)
        a = rng.random(n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: spectral GD + L-BFGS + active-set, multi-n, multi-init."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Spectral Gradient Descent with multiple inits ===
    print(f"\n[P1] Spectral GD (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600.0, t_start + 700.0)

    n_candidates = [n_parent]
    if n_parent != 16384:
        n_candidates.append(16384)
    if n_parent != 32768:
        n_candidates.append(32768)

    n_init = 12288
    inits = [
        ("uniform", _make_profile(n_init, "uniform")),
        ("tapered", _make_profile(n_init, "tapered")),
        ("tapered2", _make_profile(n_init, "tapered2")),
        ("bimodal", _make_profile(n_init, "bimodal")),
        ("cos_mod", _make_profile(n_init, "cosine_mod")),
    ]
    if parent is not None:
        inits.append(("parent_rs", _resample(parent, n_init)))

    time_per_init = (p1_deadline - t_start) / len(inits)

    for label, a_init in inits:
        if time.monotonic() > deadline - 200.0:
            break
        init_dl = min(p1_deadline, time.monotonic() + time_per_init)
        if init_dl - time.monotonic() < 10.0:
            continue

        print(f"  [{label}] n={n_init}, init_sc={_score(a_init):.10f}")

        # L1 spectral GD
        a_l1, sc_l1 = _spectral_gd(a_init, init_dl, penalty="l1", verbose=False)
        if sc_l1 < best_score:
            best_a = a_l1.copy()
            best_score = sc_l1
            print(f"    NEW BEST L1: {best_score:.10f} [{label}]")

        # L2 spectral GD from L1 result
        remaining = min(p1_deadline, time.monotonic() + 15.0)
        if remaining > time.monotonic() + 5.0:
            a_l2, sc_l2 = _spectral_gd(a_l1, remaining, penalty="l2", verbose=False)
            if sc_l2 < best_score:
                best_a = a_l2.copy()
                best_score = sc_l2
                print(f"    NEW BEST L2: {best_score:.10f} [{label}]")

    # === Phase 2: L-BFGS polishing on best ===
    print(f"\n[P2] L-BFGS polish (t={time.monotonic()-t_start:.1f}s), score={best_score:.10f}")
    p2_deadline = min(deadline - 300.0, t_start + 1100.0)
    a2 = _run_lbfgs(best_a, p2_deadline, min_remaining=30.0)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_p2]")

    # === Phase 3: Active-set polish ===
    print(f"\n[P3] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 150.0, t_start + 1350.0)
    a3, sc3 = _active_set_polish(best_a, p3_deadline, tol_start=0.003, tol_end=0.00001, lr_start=0.02)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [polish_p3]")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 20480, 32768]
    n_best = len(best_a)
    if n_best not in n_targets:
        n_targets.insert(0, n_best + 1)  # avoid skip

    for n_try in sorted(set(n_targets)):
        if time.monotonic() > deadline - 120.0:
            break
        if n_try == n_best:
            continue
        if n_try > 65536:
            continue

        a_init_n = _resample(best_a, n_try)
        sc_init = _score(a_init_n)
        print(f"  n={n_try}, init={sc_init:.10f}")

        # Quick spectral GD
        gd_dl = min(deadline - 100.0, time.monotonic() + 80.0)
        a_gd, sc_gd = _spectral_gd(a_init_n, gd_dl, penalty="l2", verbose=False)
        if sc_gd < best_score:
            best_a = a_gd.copy()
            best_score = sc_gd
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_gd]")

        # Quick L-BFGS
        if time.monotonic() < deadline - 60.0:
            a_lb = _run_lbfgs(a_gd, min(deadline - 50.0, time.monotonic() + 60.0), min_remaining=10.0)
            sc_lb = _score(a_lb)
            if sc_lb < best_score:
                best_a = a_lb.copy()
                best_score = sc_lb
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_lbfgs]")

    # === Phase 5: Diversified restarts around best ===
    print(f"\n[P5] Diversified restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60.0:
        n_best = len(best_a)
        restarts = [
            ("pert1", _perturb(best_a, 0.03, rng)),
            ("pert2", _perturb(best_a, 0.08, rng)),
            ("tapered", _make_profile(n_best, "tapered")),
            ("cos_mod", _make_profile(n_best, "cosine_mod")),
            ("trimodal", _make_profile(n_best, "trimodal")),
        ]

        for label, a_r in restarts:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 15.0:
                break
            a_opt = _run_lbfgs(a_r, deadline - 20.0, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}]")

    # === Phase 6: Final ultra-fine polishing ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 2.0,
                                            tol_start=0.0005, tol_end=0.000005,
                                            lr_start=0.008)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

        # One more spectral GD with very tight tolerance
        if time.monotonic() < deadline - 2.0:
            a_gd2, sc_gd2 = _spectral_gd(best_a, deadline - 1.0, penalty="l2", verbose=False)
            if sc_gd2 < best_score:
                best_a = a_gd2.copy()
                best_score = sc_gd2
                print(f"  NEW BEST: {best_score:.10f} [final_gd]")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (0.01 / 256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
