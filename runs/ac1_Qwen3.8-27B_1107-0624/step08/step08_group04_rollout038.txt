"""
Multi-phase optimizer for non-negative step function minimization.
Uses: (1) custom L-BFGS with log-sum-exp continuation, (2) active-set Newton polish,
(3) coarse-to-fine multi-resolution optimization, (4) n=32768 exploration,
(5) peak-deflation heuristic, and (6) diverse restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smo(z, beta, n):
    """Compute smoothed objective value and gradient at z."""
    a = _softmax(z)
    c = _full_conv(a, a)
    c_max = float(c.max())

    """Compute H*v where H is the Hankel operator defined by W."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)

        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp

            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new

        d = d - d.mean()  # ensure sum(d) = 0
        
        # Line search
        best_alpha = 0.0
        best_obj = np.inf
        alpha = 1.0
        for ls in range(30):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_conv(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < best_obj:
                best_obj = c_max_try
                best_alpha = alpha
            if c_max_try < c_max * (1 - 1e-6):
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break
        
        if best_alpha > 0:
            a_new = a + best_alpha * d
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-15:
                    best_sc = sc_new
                    best_a = a_new.copy()
                a = a_new
        
        # Advance tolerance
        if (rnd + 1) % rounds_per_tol == 0 and tol_idx < len(tol_schedule) - 1:
            tol_idx += 1
        
        # Check convergence
        c = _full_conv(a, a)
        c_flat = c.max() / (c.sum() / len(c))
        if c_flat < 1.0001:
            break
    
    return best_a, best_sc


def _peak_deflate(a_init, deadline, max_iters=500):
    """
    Direct peak deflation: identify the peak autocorrelation bin
    and adjust the sequence to reduce it.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_conv(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))
        
        # The peak autocorrelation is at lag peak_idx
        # dc_peak/da_i involves a at positions related to peak_idx
        # Reduce the contributions to the peak
        
        # Compute gradient of c[peak_idx] w.r.t. a
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i] (with bounds)
        # dc/da_i = a[peak_idx - i] + a[i - peak_idx] (where valid)
        peak_grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = peak_idx - i
            if 0 <= j1 < n:
                val += a[j1]
            j2 = i - peak_idx
            if 0 <= j2 < n:
                val += a[j2]
            peak_grad[i] = val
        
        # Don't reduce indices where peak_grad is 0
        pg_norm = np.linalg.norm(peak_grad)
        if pg_norm < 1e-15:
            break
        
        # Step to reduce peak while increasing mean (to maintain sum)
        step = 0.05 / (1.0 + 0.05 * it)
        d = -peak_grad / pg_norm + peak_grad.mean() / pg_norm
        # This reduces the peak proportionally
        
        a_new = a + step * d
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a + 0.25 * step * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
    
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "flat_key":
        # Flat with slight key-hole pattern
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(4 * np.pi * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

    b1 = np.zeros(n)
    valid1 = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)

    b2[valid2] = a[np.arange(n)[valid2] - k]
    
    return b1 + b2

"""
Multi-phase optimizer for non-negative step function autocorrelation minimization.
Combines: (1) aggressive L-BFGS continuation with log-sum-exp smoothing,
(2) active-set Newton KKT solves, (3) spectral peak deflation,
(4) n=32768 multi-resolution exploration, and (5) diverse restarts.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smo(z, beta, n):
    """Compute smoothed objective and gradient at z using global beta, n."""
    a = _softmax(z)
    c = _full_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    a_rev = a[::-1]
    wc = _full_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    return obj, grad


def _hankel_matvec(v, W, n):
    """Compute H*v where H is defined by the weight vector W on autocorrelation lags."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """CG solve for H*x = rhs with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        if not np.isfinite(alpha):
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 2.0:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _lbfgs_smo(z, beta, n, max_iter=5000, deadline=None, min_time_remaining=10.0, gtol=1e-11):
    """Custom L-BFGS on the smoothed objective with softmax parameterization."""
    m = min(20, max(5, n // 100))
    s_list = []
    y_list = []
    rho_list = []

    z = z - z.mean()
    f_val, g = _compute_smo(z, beta, n)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break

        g_norm = float(np.linalg.norm(g))
        if g_norm < gtol * (1.0 + float(np.linalg.norm(z))):
            break

        # L-BFGS two-loop recursion
        q = g.copy()
        alphas = []
        for i in range(len(s_list) - 1, -1, -1):
            alpha_i = rho_list[i] * float(s_list[i] @ q)
            alphas.append(alpha_i)
            q = q - alpha_i * y_list[i]
        alphas.reverse()

        if len(s_list) == 0:
            r = q.copy()
        else:
            sy_last = float(s_list[-1] @ y_list[-1])
            yy_last = float(y_list[-1] @ y_list[-1])
            gamma = sy_last / max(yy_last, 1e-30)
            r = gamma * q

        for i in range(len(s_list)):
            beta_i = rho_list[i] * float(y_list[i] @ r)
            r = r + (alphas[i] - beta_i) * s_list[i]

        # Ensure descent direction
        g_dir = float(g @ r)
        if g_dir > -1e-30:
            r = -g.copy()
            g_dir = -float(g @ g)
            if g_dir > -1e-30:
                break

        # Backtracking line search (Armijo + curvature)
        alpha = 1.0
        c1 = 1e-4
        c2 = 0.7
        success = False
        for ls in range(50):
            z_new = z + alpha * r
            z_new = z_new - z_new.mean()
            f_new, g_new = _compute_smo(z_new, beta, n)
            if not np.isfinite(f_new):
                alpha *= 0.5
                continue
            if f_new <= f_val + c1 * alpha * g_dir:
                g_new_dir = float(g_new @ r)
                if g_new_dir >= c2 * g_dir:
                    success = True
                    break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not success:
            break

        # Update L-BFGS memory
        s_new = alpha * r
        y_new = g_new - g
        sy = float(s_new @ y_new)
        if sy > 1e-18:
            s_list.append(s_new)
            y_list.append(y_new)
            rho_list.append(1.0 / sy)
            if len(s_list) > m:
                s_list.pop(0)
                y_list.pop(0)
                rho_list.pop(0)

        z = z_new
        g = g_new
        f_val = f_new

    return z


def _active_set_newton(a_init, deadline, max_rounds=200, min_time_remaining=5.0):
    """
    Active-set Newton: identify active autocorrelation bins,
    solve KKT system via CG, line search.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001,
                    0.00000005, 0.00000002]
    tol_idx = 0
    rounds_per_tol = 6

    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_conv(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]

        # Identify active set
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute gradient of smoothed max over active set
        grad = _hankel_matvec(a, W, n)
        lam = float(grad.mean())
        g_t = grad - lam

        g_norm = float(np.linalg.norm(g_t))
        if g_norm < 1e-14:
            # Check non-negativity
            if np.all(a >= -1e-15):
                break

        # Newton direction: solve H_W * d = -(g_t) s.t. 1^T d = 0
        Hg = _cg_solve(W, n, g_t, max_iter=400)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=400)

        sum_Hg = float(np.sum(Hg))
        sum_H1 = float(np.sum(H1))
        if abs(sum_H1) > 1e-20 and np.isfinite(sum_H1):
            lam2 = sum_Hg / sum_H1
            d = -Hg + lam2 * H1
        else:
            d = -g_t

        d = d - d.mean()

        # Damped line search
        alpha = 1.0
        c_max_orig = c_max
        improved = False
        for ls in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_conv(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_orig * (1.0 - 1e-10):
                improved = True
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                a = a_try
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not improved and rnd > 10:
            # Try a gradient step as fallback
            step = 0.1 / (1.0 + 0.1 * rnd)
            a_try = a - step * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try

        # Advance tolerance
        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _spectral_deflate(a_init, deadline, max_iters=200):
    """
    Spectral domain peak reduction: identify frequency bins contributing
    most to the autocorrelation peak and reduce their contribution.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_conv(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Compute gradient of c[peak_idx] w.r.t. a
        # dc[k]/da[i] = a[k-i] + a[i-k] (with 0-padding)
        k = peak_idx
        i_arr = np.arange(n)
        b1 = np.zeros(n)
        valid1 = (k - i_arr >= 0) & (k - i_arr < n)
        b1[valid1] = a[k - i_arr[valid1]]
        b2 = np.zeros(n)
        valid2 = (i_arr - k >= 0) & (i_arr - k < n)
        b2[valid2] = a[i_arr[valid2] - k]
        peak_grad = b1 + b2

        # Also compute gradient of mean autocorrelation (= constant for normalized a)
        # So we want to reduce peak while maintaining sum
        pg_mean = peak_grad.mean()
        d = peak_grad - pg_mean  # zero-mean direction that reduces peak

        pg_norm = float(np.linalg.norm(d))
        if pg_norm < 1e-15:
            break

        # Line search
        improved = False
        alpha = 0.1
        for ls in range(20):
            a_try = a - alpha * d / pg_norm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.7
                continue
            a_try = a_try / s_try
            c_try = _full_conv(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 1e-8):
                improved = True
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                a = a_try
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break

        if not improved:
            break

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.05 * np.cos(2 * np.pi * 3 * t) + 0.03 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()

def _next_pow2(n):
    k = 1
    while k < n:

def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1

    fa = np.fft.rfft(a, N)
