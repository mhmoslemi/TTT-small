"""
Step function optimization using Multiplicative Weight Exponentiated Gradient (MWEG)
method with temperature annealing, combined with L-BFGS polishing for fine-tuning.
Key innovation: direct multiplicative updates that naturally preserve non-negativity
while targeting the max of the auto-convolution via soft-minimax weights.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    ss = e.sum()
    if ss < 1e-300:
        return np.ones_like(z) / len(z)
    return e / ss


def _mse_obj_and_grad(z, beta, n):
    """Log-sum-exp smoothed objective and gradient for softmax-parameterized a."""
    a = _softmax(z)
    c = _fft_conv(a, a)
    shifted = c / beta
    m = shifted.max()
    e = np.exp(shifted - m)
    s = e.sum()
    if s < 1e-300:
        obj = n * beta * (m + 0)  # just the max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = n * beta * (m + np.log(s))
        w = e / s
    
    # Gradient computation
    N = _next_pow2(len(w) + n)
    W = np.fft.rfft(w, N)
    A_flip = np.fft.rfft(a[::-1], N)
    g_full = np.fft.irfft(W * A_flip, N)
    g_a = g_full[n-1:2*n-1] / beta
    
    W_tot = np.dot(g_a, a)
    grad = a * (g_a - W_tot)
    return obj, grad


def _lbfgs_optimize(a_init, deadline, beta_init=None):
    """L-BFGS optimization on softmax parameterization with log-sum-exp."""
    n = len(a_init)
    c = _fft_conv(a_init, a_init)
    c_max = float(c.max())
    if beta_init is None:
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted)-1)]
        beta_init = max(gap * 0.5, c_max * 0.01)
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    # Anneal beta
    betas = beta_init * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        
        def obj(z):
            val, _ = _mse_obj_and_grad(z, beta, n)
            return val
        
        def grad(z):
            _, g = _mse_obj_and_grad(z, beta, n)
            return g
        
        remaining = deadline - time.monotonic() - 5
        # Estimate iterations
        scale = max(1.0, n / 4096.0)
        est_time_per_iter = 0.003 * scale
        max_iter = max(20, min(3000, int(remaining / (est_time_per_iter * 10))))
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50})
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _mweg_optimize(a_init, deadline, T_start=0.1, T_end=0.001, eta_start=0.5, eta_min=0.01):
    """Multiplicative Weight Exponentiated Gradient optimization.
    
    Directly optimizes the max of the convolution using multiplicative updates
    with temperature-annealed soft-minimax weights.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = eta_start
    no_improve = 0
    max_no_improve = 50
    
    # Precompute FFT size
    N = _next_pow2(3 * n - 1)
    
    total_iters = 0
    while time.monotonic() < deadline - 3:
        c = _fft_conv(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature based on progress
        progress = min(1.0, no_improve / max_no_improve)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, c_max * 0.001)
        
        # Soft weights
        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Compute sensitivity g[i] = sum_j w[i+j] * a[j]
        W = np.fft.rfft(w, N)
        A_flip = np.fft.rfft(a[::-1], N)
        g_full = np.fft.irfft(W * A_flip, N)
        g = g_full[n-1:2*n-1]
        
        # Center and scale g
        g_mean = g.mean()
        g_range = g.max() - g.min()
        if g_range < 1e-15:
            break
        g_norm = (g - g_mean) / g_range  # in [-0.5, 0.5] roughly
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g_norm * 2.0)  # factor of 2 for units
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        total_iters += 1
        
        if sc_new < sc - 1e-14:
            a = a_new
            eta = min(eta * 1.01, 2.0)
        else:
            # Try smaller step
            improved = False
            for factor in [0.5, 0.25, 0.1]:
                a_try = a * np.exp(-eta * factor * g_norm * 2.0)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < sc - 1e-14:
                    a = a_try
                    eta *= 0.8
                    improved = True
                    break
            if not improved:
                eta *= 0.7
        
        if no_improve >= max_no_improve:
            # Reset with perturbation
            a = best_a + 0.01 * np.random.default_rng(total_iters).standard_normal(n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            eta = eta_start * 0.5
    
    return best_a, best_sc


def _local_search(a_init, deadline, n_perturb=50, sigma=0.05):
    """Local perturbation search around the current best."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)
    rng = np.random.default_rng(42)
    
    for _ in range(n_perturb):
        if time.monotonic() > deadline:
            break
        # Random perturbation in log-space
        z = np.log(np.maximum(best_a, 1e-30))
        z += sigma * rng.standard_normal(n)
        a_try = _softmax(z)
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_a = a_try
            best_sc = sc_try
            sigma = min(sigma * 1.1, 0.5)
        else:
            sigma *= 0.9
    
    return best_a, best_sc


def _peak_targeted_update(a, deadline, n_trials=500):
    """Targeted peak reduction: identify top contributors to each peak and reduce them."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(best_a, best_a)
        # Find top few peaks
        top_k = min(5, len(c))
        peak_idxs = np.argsort(c)[::-1][:top_k]
        
        for k in peak_idxs:
            # Find contributors to c[k]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            contrib = best_a[i_lo:i_hi+1] * best_a[k-i_hi:k-i_lo+1][::-1]
            # Wait, let me be more careful
            # c[k] = sum_i a[i]*a[k-i] for i in [i_lo, i_hi]
            # contrib[i] = a[i] * a[k-i]
            i_range = np.arange(i_lo, i_hi + 1)
            j_range = k - i_range
            contrib = best_a[i_range] * best_a[j_range]
            
            if contrib.sum() < 1e-15:
                continue
            
            # Reduce top contributors
            n_reduce = max(1, len(contrib) // 5)
            top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
            
            a_try = best_a.copy()
            total_red = 0
            for idx in top_idx:
                red = a_try[idx] * 0.05
                a_try[idx] -= red
                total_red += red
            
            # Distribute to low contributors
            low_set = set(range(n)) - set(top_idx.tolist()) - set(j_range[top_idx - i_lo].tolist())
            if len(low_set) == 0:
                low_set = set(range(n)) - set(top_idx.tolist())
            low_arr = np.array(list(low_set))
            
            if len(low_arr) == 0:
                continue
            
            a_try[low_arr] += total_red / len(low_arr)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try
                best_sc = sc_try
                break  # restart from new best
    
    return best_a, best_sc


def _make_init(n, kind="asym_cos", rng=None):
    """Create diverse initial sequences."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "asym_cos":
        # Asymmetric cosine modulation
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * t / n + 0.5) + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "ramp":
        t = np.arange(n)
        a = 0.5 + 0.5 * t / (n - 1)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.arange(n)
        a = np.exp(0.5 * t / n)
        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.arange(n) / n
        a = np.exp(-30.0 * (t - 0.3) ** 2) + 0.7 * np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        a = np.exp(rng.standard_normal(n) * 0.3)
        return a / a.sum()
    elif kind == "high_f_cos":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * 11.0 * t / n + 1.0)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n

    """
    Main optimization function combining MWEG, L-BFGS, and peak-targeted methods.
    
    Strategy:
    1. Try multiple initializations (asymmetric profiles)
    2. For each, run MWEG (multiplicative weight exponentiated gradient)
    3. Polish with L-BFGS
    4. Final peak-targeted refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        n_best = len(parent)
        best_a = parent.copy()
        best_score = _score(best_a)
        print(f"Parent loaded: n={n_best}, score={best_score:.10f}")
    else:
        n_best = 8192
        best_a = _make_init(n_best, "asym_cos")

        print(f"No parent: n={n_best}, score={best_score:.10f}")
    
    # Phase 1: MWEG on parent
    print(f"\n[P1] MWEG on parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600.0, t_start + 400.0)
    a1, sc1 = _mweg_optimize(best_a, p1_deadline, T_start=0.1, T_end=0.005, eta_start=0.3)
    if sc1 < best_score:
        best_a = a1
        best_score = sc1
        print(f"  MWEG improved: {best_score:.10f}")
    
    # Phase 2: L-BFGS polish
    print(f"\n[P2] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400.0, t_start + 700.0)
    a2, sc2 = _lbfgs_optimize(best_a, p2_deadline)
    if
