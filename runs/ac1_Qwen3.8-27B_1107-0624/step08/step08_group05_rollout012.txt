    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgrad_active(a, n, tol_frac):
    """Compute exact subgradient of max(c) w.r.t. a using active set."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n), 1
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    # g[j] = 2 * sum_m w[j+m] * a[m], m=0..n-1
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    g_full = np.fft.irfft(W * A, N)
    g = 2.0 * g_full[:n]
    return g, int(active.sum())


def _subgrad_balanced(a, n, top_k=5, tol_frac=0.001):
    """Subgradient using top-k lags weighted by their values."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    # Weight by proximity to max
    w = np.zeros_like(c)
    w[active] = (c[active] / c_max) ** 4
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    g_full = np.fft.irfft(W * A, N)
    return 2.0 * g_full[:n]


def _hess_matvec(v, w, n):
    """Compute H*v where H comes from active set w on lags."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(w.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b_vec, tol=1e-10, max_iter=150):
    n = len(b_vec)
    x = np.zeros(n)
    r = b_vec.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b_vec))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad = _hess_matvec(a, w, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8 + 1e-5 * max(0, 10 - it)

        def _hv(v):
            return _hess_matvec(v, w, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=150)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    h = 2.0 * np.fft.irfft(W * A, N)[:n]
    W_dot_c = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W_dot_c)
    return a * (grad_a - np.dot(a, grad_a))


def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=16):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(80, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.002)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # More aggressive beta schedule
    ratios = [1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01,
              0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
              0.00005, 0.00002]
    betas = base_beta * np.array(ratios[:n_stages])

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_descent(a_init, deadline, n_steps=10000, lr_init=0.005, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    non_improve = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = max(tol_frac * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        N = _next_pow2(2 * n)
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        g = 2.0 * np.fft.irfft(W * A, N)[:n]

        grad_z = a * (g - np.dot(a, g))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                if non_improve > 5:
                    z = np.log(np.maximum(a_check, 1e-30))
                    z = z - z.mean()
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                non_improve = 0
            else:
                non_improve += 1
                patience += 1
                if patience > 15:
                    lr *= 0.9
                    if lr < 1e-6:
                        z = np.log(np.maximum(best_a, 1e-30))
                        z = z - z.mean()
                        patience = 0
                        lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _balance_step(a, n, n_iters=50):
    """Iteratively balance the top autocorrelation values."""
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        idx_max = int(np.argmax(c))

        # Compute gradient direction to reduce c[idx_max]
        w = np.zeros_like(c)
        w[idx_max] = 1.0
        N = _next_pow2(2 * n)
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        g = 2.0 * np.fft.irfft(W * A, N)[:n]
        g = g - np.mean(g)

        # Line search
        alpha = 0.01
        for ls in range(30):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if c_try[idx_max] < c_max - 1e-15:
                a = a_try
                break
            alpha *= 0.7

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_bump":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "w_triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t) + 0.1 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_seq(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    sum_cp = float(np.sum(np.power(c_safe, p)))
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    w = w / w_sum
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    h = 2.0 * np.fft.irfft(W * A, N)[:n]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0)
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _slsqp_minmax(a_init, deadline, min_remaining=20.0):
    """Use SLSQP to directly minimize the max autocorrelation."""
    n = len(a_init)
    a0 = a_init.copy()
    s0 = a0.sum()
    if s0 < 1e-10:
        return a_init, _score(a_init)
    a0 = a0 / s0

    def _obj(x):
        a_x = np.maximum(x, 0)
        s_x = a_x.sum()
        if s_x < 1e-10:
            return 1e10
        a_x = a_x / s_x
        c = _full_convolve(a_x, a_x)
        return float(c.max())

    def _jac(x):
        a_x = np.maximum(x, 0)
        s_x = a_x.sum()
        if s_x < 1e-10:
            return np.zeros(n)
        a_x = a_x / s_x
        c = _full_convolve(a_x, a_x)
        c_max = float(c.max())
        active = c >= (c_max - 1e-4 * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        N = _next_pow2(2 * n)
        A = np.fft.rfft(a_x, N)
        W = np.fft.rfft(w, N)
        g = 2.0 * np.fft.irfft(W * A, N)[:n]
        return g - np.mean(g)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n

    try:
        result = minimize(
            _obj, a0, jac=_jac, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-14}
        )
        if result.success or _score(np.maximum(result.x, 0) / max(np.maximum(result.x, 0).sum(), 1e-10)) < _score(a0):
            a_new = np.maximum(result.x, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                return a_new, _score(a_new)
    except Exception:
        pass

    return a0, _score(a0)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining exact subgradient descent, L-BFGS with
    aggressive LSE/power-sum smoothing, Newton active-set, SLSQP min-max,
    multi-n exploration, and diverse restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (40s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 40)
    n_expl = max(2048, n_parent // 4) if n_parent > 4096 else 2048
    for kind in ["uniform", "tapered", "cos_bump", "gauss", "bimodal", "ramp", "asym", "w_triangle"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Main L-BFGS LSE on best (300s) ===
    print(f"\n[Phase 2] Main L-BFGS LSE")
    if time.monotonic() < deadline - 400:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=350)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 3: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 3] Power-sum L-BFGS")
    if time.monotonic() < deadline - 300:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=280)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 4: Newton active-set (200s) ===
    print(f"\n[Phase 4] Newton active-set")
    if time.monotonic() < deadline - 250:
        nw_dead = min(deadline - 230, time.monotonic() + 200)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=400, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 5: Subgradient descent (200s) ===
    print(f"\n[Phase 5] Subgradient descent")
    if time.monotonic() < deadline - 200:
        sg_dead = min(deadline - 180, time.monotonic() + 200)
        a_sg, sc_sg = _subgrad_descent(_best_seq, sg_dead, n_steps=20000, lr_init=0.004, tol_frac=0.0008)
        upd(a_sg, sc_sg, "subgrad_main")

    # === Phase 6: Alternating Newton + Subgradient (150s) ===
    print(f"\n[Phase 6] Alternating Newton + Subgrad")
    a_cur = _best_seq.copy()
    for round_i in range(4):
        if time.monotonic() > deadline - 130:
            break
        nw_dead = min(deadline - 130, time.monotonic() + 30)
        a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=100, tol_frac=0.0015)
        if sc_nw2 < _score(a_cur):
            a_cur = a_nw2
        upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")

        sg_dead2 = min(deadline - 130, time.monotonic() + 25)
        a_sg2, sc_sg2 = _subgrad_descent(a_cur, sg_dead2, n_steps=8000, lr_init=0.003, tol_frac=0.0005)
        if sc_sg2 < _score(a_cur):
            a_cur = a_sg2
        upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 7: SLSQP min-max (60s) ===
    print(f"\n[Phase 7] SLSQP min-max")
    if time.monotonic() < deadline - 80:
        sl_dead = min(deadline - 60, time.monotonic() + 60)
        a_sl, sc_sl = _slsqp_minmax(_best_seq, sl_dead, min_remaining=10)
        upd(a_sl, sc_sl, "slsqp")

    # === Phase 8: Multi-n exploration (250s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384, 20480, 24576]
    if n_parent not in n_candidates and 2048 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        a_init = _best_seq.copy() if len(_best_seq) == n_try else _resize_seq(_best_seq, n_try)
        sc_init = _score(a_init)
        upd(a_init, sc_init, f"n_{n_try}_init")

        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=40)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        if sc_opt_n < _best_score * 1.002 and time.monotonic() < deadline - 30:
            nw_dead = min(deadline - 20, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=100, tol_frac=0.001)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 9: Diverse restarts (200s) ===
    print(f"\n[Phase 9] Diverse restarts")
    if time.monotonic() < deadline - 50:
        restart_strategies = [
            ("log_perturb_0.03", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("log_perturb_0.05", lambda: _perturb_log(_best_seq, 0.05, rng)),
            ("fourier_phase", lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)),
            ("amplitude_perturb", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.06)),
            ("log_perturb_0.01", lambda: _perturb_log(_best_seq, 0.01, rng)),
            ("fourier_phase_small", lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=4, max_amp=0.15)),
            ("amplitude_small", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.03)),
        ]
        for i, (label_name, gen_fn) in enumerate(restart_strategies):
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=20)
            upd(a_r, sc_r, f"restart_{label_name}")

            if sc_r < _best_score * 1.002 and time.monotonic() < deadline - 15:
                nw_dead = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=60, tol_frac=0.002)
                upd(a_r2, sc_r2, f"restart_{label_name}_newton")

    # === Phase 10: Final aggressive polish (80s) ===
    print(f"\n[Phase 10] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()

        if remaining > 10:
            # Ultra-low beta LSE
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.008)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-16, 'maxcor': 80, 'maxls': 300}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 4, max_iter=200, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton_tight")

        # Final subgradient with very small tol
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _subgrad_descent(_best_seq, deadline - 2,
                                               n_steps=20000, lr_init=0.002, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg_tight")

        # Balance step
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _balance_step(_best_seq, len(_best_seq), n_iters=100)
            upd(a_fin4, sc_fin4, "final_balance")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
