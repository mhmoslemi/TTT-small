import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0


def _lsobj(z, beta, n):

    a = _symmetrize(a)

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)

        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _subgradient(a, active_frac=0.02):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    threshold = c_max * (1 - active_frac)
    active = c >= threshold
    if not active.any():
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0
    w /= w.sum()
    grad = 2.0 * _corr_a_A(a, w, n)
    grad = _symmetrize(grad)
    return grad, c


def _multiplicative_weight_descent(a_init, deadline, n_iters=50000, active_frac=0.02):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    a = _symmetrize(a)
    best_a = a.copy()
    best_sc = _score(a)

    y = a.copy()
    t_nest = 1.0
    step_size = 0.5

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        grad, c = _subgradient(y, active_frac)
        c_max = float(c.max())

        # Center gradient (projection onto simplex tangent)
        g_t = grad - np.mean(grad)

        # Adaptive step size
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Mirror descent step: a_new ∝ y * exp(-eta * grad)
        eta = step_size / (gnorm * np.max(a) + 1e-15)
        eta = min(eta, 2.0)

        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad
        log_a_new = log_a_new - np.mean(log_a_new)
        a_new = _softmax(log_a_new)
        a_new = _symmetrize(a_new)

        # Nesterov update for y
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y_new = a_new + ((t_nest - 1.0) / t_new) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y
        y = y_new
        a = a_new
        t_nest = t_new

        # Adaptive step size adjustment
        if it % 100 == 99:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()
                step_size = min(step_size * 1.1, 3.0)
            elif sc > best_sc + 0.001:
                step_size *= 0.7

        if it % 500 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            print(f"    MWD iter {it}: score={sc:.8f}, best={best_sc:.8f}, eta={eta:.4f}")

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def _pairwise_exchange(a, deadline, n_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(best_a, best_a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())

        # Find the worst position
        worst_k = int(np.argmax(c))

        # Find contributing pairs
        i_lo = max(0, worst_k - n + 1)
        i_hi = min(worst_k, n - 1)

        # Find the pair with largest product
        max_prod = 0
        max_i = -1
        for i in range(i_lo, i_hi + 1):
            j = worst_k - i
            prod = best_a[i] * best_a[j]
            if prod > max_prod:
                max_prod = prod
                max_i = i

        if max_i < 0:
            break

        j = worst_k - max_i

        # Try to transfer mass from the larger to the smaller
        if max_i == j:
            # Diagonal: reduce a[i] and distribute to neighbors
            delta = best_a[max_i] * 0.01
            # Reduce a[max_i] and add to a[0] and a[n-1] (which don't affect c[worst_k] much)
            alt_i = 0 if max_i != 0 else 1
            alt_j = n - 1 if max_i != n - 1 else n - 2
            a_try = best_a.copy()
            a_try[max_i] -= delta
            a_try[alt_i] += delta / 2
            a_try[alt_j] += delta / 2
        else:
            # Transfer from larger to smaller
            if best_a[max_i] >= best_a[j]:
                src, dst = max_i, j
            else:
                src, dst = j, max_i
            delta = best_a[src] * 0.005
            a_try = best_a.copy()
            a_try[src] -= delta
            a_try[dst] += delta

        a_try = np.maximum(a_try, 0)
        a_try = _symmetrize(a_try)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
        else:
            # Try smaller delta
            if max_i != j:
                if best_a[max_i] >= best_a[j]:
                    src, dst = max_i, j
                else:
                    src, dst = j, max_i
                delta = best_a[src] * 0.001
                a_try = best_a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                a_try = np.maximum(a_try, 0)
                a_try = _symmetrize(a_try)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-12:
                        best_sc = sc_try
                        best_a = a_try.copy()
                    else:
                        break
            else:
                break

    return best_a


def _lp_direction_step(a, deadline, top_k=200):
    """Compute a descent direction using the dual of the linearized LP."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Select top-k active positions
    top_indices = np.argsort(c)[::-1][:top_k]
    active_mask = np.zeros(len(c), dtype=np.float64)
    active_mask[top_indices] = 1.0

    # The linearized constraint gradients are: 2 * a_shifted[k] for each active k
    # We solve the dual: find u_k >= 0, sum u_k = 1, minimizing max_i (sum_k u_k * a[k-i])
    # This is a fractional programming problem; solve with projected gradient on the dual

    # Compute the "gradient matrix" G[i,k] = a[k-i] for active k
    # We want: minimize max_i sum_k u_k * G[i,k]
    # Dual of dual: this is equivalent to finding the minimax of a bilinear form

    # Simple approach: iterative proportional fitting
    u = np.ones(top_k) / top_k
    a_sel = a  # full sequence

    for it in range(500):
        if time.monotonic() > deadline:
            break

        # Compute G @ u for each i
        # G[i,k] = a[k - i] (with zero padding)
        # sum_k u_k * a[k-i] = (w * a_rev)[n-1-i] where w is active_mask * u
        w = active_mask.copy()
        for idx, ki in enumerate(top_indices):
            w[ki] = u[idx]

        # Compute the "effective gradient" for each i
        flip_a = a[::-1]
        out_len = n + len(w) - 1
        N = _next_pow2(out_len)
        fa = np.fft.rfft(flip_a, N)
        fw = np.fft.rfft(w, N)
        result = np.fft.irfft(fa * fw, N)[:out_len]
        G_u = result[n - 1:2 * n - 1]

        # Find the max coordinate
        i_max = int(np.argmax(G_u))
        g_max = G_u[i_max]

        if g_max < 1e-15:
            break

        # Update: increase u_k for k that contribute most to G_u[i_max]
        # G[i_max, k] = a[k - i_max]
        contrib = np.zeros(top_k)
        for idx, ki in enumerate(top_indices):
            j = ki - i_max
            if 0 <= j < n:
                contrib[idx] = a[j]

        # Multiplicative weight update
        contrib_scaled = contrib / (contrib.max() + 1e-15)
        u = u * (1 + 0.1 * contrib_scaled)
        u = u / u.sum()

    # Compute the final direction: d = -sum_k u_k * grad_k = -2 * sum_k u_k * a_shifted[k]
    w_final = active_mask.copy()
    for idx, ki in enumerate(top_indices):
        w_final[ki] = u[idx]

    direction = -2.0 * _corr_a_A(a, w_final, n)
    direction = _symmetrize(direction)

    # Project onto tangent space of simplex
    direction = direction - np.mean(direction)

    # Ensure we don't go negative
    neg = direction < -1e-12
    if neg.any():
        t_max = np.min(a[neg] / (-direction[neg])) * 0.99
    else:
        t_max = 1.0
    t_max = max(t_max, 1e-10)

    # Line search
    best_t = 0
    best_sc = _score(a)
    t = min(1.0, t_max)
    for _ in range(30):
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        a_try = _symmetrize(a_try)
        s_try = a_try.sum()
        if s_try < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_t = t
        t *= 0.5

    if best_t > 0:
        a_result = a + best_t * direction
        a_result = np.maximum(a_result, 0)
        a_result = _symmetrize(a_result)
        a_result = a_result / a_result.sum()
        return a_result

    return a


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    if tol_list is None:
        tol_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    a = _symmetrize(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_dir_score = np.inf
        best_t = 0.0

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[int(np.argmax(c))] = True
            m = int(active.sum())
            if m < 2 or m > 8 * n:
                if m > 8 * n:
                    top_idx = np.argsort(c)[::-1][:min(4 * n, len(c))]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue

            A_vec = active.astype(np.float64)

            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g = _symmetrize(g)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            mu = 1e-6 * (c_max / max(m, 1)) + 1e-9

            def hess_vp(v, A_vec=A_vec, m=m, mu=mu, n=n):
                r = (2.0 / m) * _corr_a_A(v, A_vec, n)
                r = _symmetrize(r)
                return r + mu * v

            u = _cg_solve(hess_vp, g_t, max_cg=200)
            w = _cg_solve(hess_vp, np.ones(n), max_cg=200)
            w = _symmetrize(w)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = _symmetrize(dx)
            dx = dx - np.mean(dx)

            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            t = min(1.0, t_max)
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            a_try = _symmetrize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx
                best_t = t

        if best_dir is None:
            # Fallback gradient step
            g_all = (2.0 / n) * _corr_a_A(a, np.ones(2 * n - 1), n)
            g_all = _symmetrize(g_all)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break
            t = 0.05
            for _ in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                a_try = _symmetrize(a_try)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        # Full line search
        f_old = c_max
        t = best_t * 1.5
        accepted = False
        for _ in range(40):
            if t < 1e-15:
                break
            a_try = a + t * best_dir
            a_try = np.maximum(a_try, 0)
            a_try = _symmetrize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old - 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Smaller step
            t = 0.01
            for _ in range(20):
                a_try = a + t * best_dir
                a_try = np.maximum(a_try, 0)
                a_try = _symmetrize(a_try)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-13:
                    a = a_try
                    break
                t *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

        if it % 50 == 0:
            print(f"    AS-Newton iter {it}: score={sc:.10f}, c_max={c_max:.6e}")

    return best_a, best_sc


def _deflation(a, deadline, n_attempts=30):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())

        top_k = np.argsort(c)[::-1]
        n_top = max(5, min(len(c), int(0.03 * len(c))))
        top_positions = top_k[:n_top]

        a_new = best_a.copy()
        for k_pos in top_positions[:15]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(int(k_pos), n - 1)
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                j = int(k_pos) - i
                if 0 <= j < n:
                    prod = a_new[i] * a_new[j]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
            if max_i >= 0 and max_prod > 0:
                j = int(k_pos) - max_i
                if max_i == j:
                    a_new[max_i] *= 0.99
                else:
                    if a_new[max_i] >= a_new[j]:
                        a_new[max_i] *= 0.995
                    else:
                        a_new[j] *= 0.995

        a_new = np.maximum(a_new, 0)
        a_new = _symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_sc


def _run_smoothed_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing and aggressive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Aggressive beta schedule
    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
             0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    betas = [b * max_c for b in betas]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.005 * max(1, n / 4096.0)
        max_iter = min(4000, max(50, int(remaining / (est_time_per_iter * 8))))
        if max_iter < 50:
            break

        def obj_fn(z_local):
            return _lsobj(z_local, beta, n)

        def grad_fn(z_local):
            return _lsgrad(z_local, beta, n)

        try:
            result = minimize(
                obj_fn,
                z,
                jac=grad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            a_new = _symmetrize(a_new)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_sc = sc_new
                current_a = a_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception as e:
            print(f"    L-BFGS beta={beta:.2e} failed: {e}")
            break

    return current_a


def _multi_n_optimize(a_best, n_new, deadline, min_time=30.0):
    """Restrict best sequence to a different n and re-optimize."""
    n_old = len(a_best)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_best)
    a_new = np.maximum(a_new, 0)
    a_new = _symmetrize(a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a_best
    a_new = a_new / s
    return _run_smoothed_lbfgs(a_new, deadline, min_time)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses symmetry, multiplicative weight descent, smoothed L-BFGS,
    LP directions, pairwise exchange, active-set Newton, and deflation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        # Symmetrize parent
        parent_sym = _symmetrize(parent)
        sc_parent_sym = _score(parent_sym)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}, sym_score={sc_parent_sym:.10f}")
        if sc_parent_sym <= sc_parent:
            update_best(parent_sym, sc_parent_sym, "parent_sym")
        else:
            update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        update_best(parent, _score(parent), "uniform_init")
        print(f"No parent, using uniform n={n_parent}, score={_score(parent):.6f}")

    a_work = best_seq.copy() if best_seq is not None else parent.copy()

    # Phase 1: Multiplicative Weight Descent (fast exploration)
    print(f"\n[Phase 1] MWD (t={time.monotonic() - t_start:.1f}s)")
    mwd_dl = min(deadline - 600, time.monotonic() + 120)
    a_mwd = _multiplicative_weight_descent(a_work, mwd_dl, n_iters=30000, active_frac=0.02)
    sc_mwd = _score(a_mwd)
    update_best(a_mwd, sc_mwd, "mwd")

    # Phase 2: Smoothed L-BFGS (main optimization)
    print(f"\n[Phase 2] Smoothed L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_lbfgs = _run_smoothed_lbfgs(best_seq, deadline, min_time_remaining=300.0)
    sc_lbfgs = _score(a_lbfgs)
