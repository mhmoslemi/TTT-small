"""
Multi-phase optimizer for minimizing step function autocorrelation peak.
Uses homotopy-based Newton active-set, ultra-low-temperature LSE,
exact active set refinement, and structured perturbations.
Target: score < 1.5027 (previous: 1.505675).
"""
import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE objective and gradient
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

# Newton active-set with improved CG
# ============================================================

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-12, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002, cg_tol=1e-12):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 0.0)
        if tol_frac <= 0:
            # Exact: only the max lag(s)
            tol = 0.0
            active = c >= (c_max - 1e-14)
        else:
            active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-15:
            break

        mu = 1e-10 * active.sum() / n

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=cg_tol, max_iter=300)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        best_alpha = 0.0
        best_f = f_old
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < best_f:
                best_f = f_try
                best_alpha = alpha
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if best_alpha > 0:
                a_try = a + best_alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_new = a_try / s_try
            else:
                a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_shift(a, rng, max_shift=5):
    n = len(a)
    shift = int(rng.integers(-max_shift, max_shift + 1))
    a_new = np.roll(a, shift)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_mixed(a, rng):
    a1 = _perturb_log(a, 0.02, rng)
    a2 = _perturb_fourier_phase(a1, rng, max_amp=0.15)
    return a2

import time
from scipy.optimize import minimize

_best_seq = None
_best_score = np.inf


def _update_best(p, sc=None):
    global _best_seq, _best_score
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _variance_obj(z, n):

    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _variance_grad(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    a_rev = a[::-1]
    dcd_a = _full_convolve(a_rev, centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    return a * (g - np.dot(a, g))


def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(active_f, b, n, mu, tol=1e-12, max_iter=300):
    x = np.zeros_like(b)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hess_matvec(p, active_f, n) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if not np.isfinite(alpha) or alpha <= 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002, cg_tol=1e-12):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if tol_frac <= 0:
            active = c >= (c_max - 1e-14)
        else:
            tol = max(tol_frac * c_max, 1e-15)
            active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-15:
            break

        mu = 1e-10 * active.sum() / n
        dx = _cg_solve(active_f, -grad_t, n, mu, tol=cg_tol, max_iter=300)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        best_alpha = 0.0
        best_f = f_old
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < best_f:
                best_f = f_try
                best_alpha = alpha
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5

                a_try = a + best_alpha * dx
                a_try = np.maximum(a_try, 0)

                    a_new = a_try / s_try

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:

                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_shift(a, rng, max_shift=5):
    shift = int(rng.integers(-max_shift, max_shift + 1))
    a_new = np.roll(a, shift)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_mixed(a, rng):
    a1 = _perturb_log(a, 0.02, rng)
    a2 = _perturb_fourier_phase(a1, rng, max_amp=0.15)
    return a2


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=20):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.4, max_c * 0.004)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * (0.62 ** np.arange(n_stages))

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_variance(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    try:
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            return current_a, current_sc
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(200, int(remaining / (est_time * 10))))
        result = minimize(
            _variance_obj, z, args=(n,),
            jac=_variance_grad, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-18,
                     'gtol': 1e-14, 'maxcor': 50, 'maxls': 100}
        )
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new
    except Exception:
        pass

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800.0):
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent)
        print(f"  BEST: {_best_score:.10f} [parent] t=0.0s")
    else:
        parent = np.ones(16384) / 16384
        _update_best(parent, _score(parent))
        print(f"  BEST: {_best_score:.10f} [init_uniform] t=0.0s")

    a = _best_seq.copy()
    n = len(a)

    # === Phase 1: Variance minimization (new approach, 150s) ===
    print(f"\n[Phase 1] Variance minimization")
    p1_deadline = min(deadline, t_start + 150)
    a_var, sc_var = _lbfgs_variance(a, p1_deadline, min_remaining=10)
    if _score(a_var) < _score(a):
        _update_best(a_var, _score(a_var))
        print(f"  BEST: {_best_score:.10f} [variance] t={time.monotonic()-t_start:.1f}s")
        a = a_var

    # === Phase 2: Homotopy Newton (250s) ===
    print(f"\n[Phase 2] Homotopy Newton active-set")
    p2_deadline = min(deadline, t_start + 400)
    tol_fracs = [0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0007, 0.0004, 0.0002, 0.0001]
    for tf in tol_fracs:
        if time.monotonic() > p2_deadline:
            break
        remaining = p2_deadline - time.monotonic()
        if remaining < 10:
            break
        est_per_iter = 0.03 * max(1.0, n / 8192.0)
        n_it = min(30, max(5, int(remaining / (est_per_iter * 3))))
        cg_t = max(1e-13, 1e-11 / (tf * 1000))
        a_new, sc_new = _newton_active_set(a, p2_deadline, max_iter=n_it, tol_frac=tf, cg_tol=cg_t)
        sc_curr = _score(a)
        if sc_new < sc_curr - 1e-15:
            a = a_new
        _update_best(a, None)
        print(f"  tf={tf:.4f}: score={_score(a):.10f}, best={_best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: LSE refinement (350s) ===
    print(f"\n[Phase 3] LSE refinement")
    p3_deadline = min(deadline, t_start + 750)
    a_lse, sc_lse = _lbfgs_lse(a, p3_deadline, min_remaining=150, n_stages=22)
    if _score(a_lse) < _score(a):
        _update_best(a_lse, _score(a_lse))
        print(f"  BEST: {_best_score:.10f} [LSE] t={time.monotonic()-t_start:.1f}s")
        a = a_lse

    # === Phase 4: Variance + Newton (150s) ===
    print(f"\n[Phase 4] Variance + Newton")
    p4_deadline = min(deadline, t_start + 900)
    a_var2, sc_var2 = _lbfgs_variance(a, p4_deadline, min_remaining=50)
    if _score(a_var2) < _score(a):
        a = a_var2
    a_new, sc_new = _newton_active_set(a, p4_deadline, max_iter=100, tol_frac=0.0005, cg_tol=1e-13)
    if sc_new < _score(a):
        a = a_new
    _update_best(a, None)
    print(f"  After phase 4: score={_score(a):.10f}, best={_best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Exact active set Newton (200s) ===
    print(f"\n[Phase 5] Exact active set Newton")
    p5_deadline = min(deadline, t_start + 1100)
    for cg_t in [1e-12, 1e-13, 1e-14]:
        if time.monotonic() > p5_deadline:
            break
        a_new, sc_new = _newton_active_set(a, p5_deadline, max_iter=150, tol_frac=0.0, cg_tol=cg_t)
        if sc_new < _score(a) - 1e-15:
            a = a_new
        _update_best(a, sc_new)
        print(f"  cg_tol={cg_t:.0e}: score={_score(a):.10f}, best={_best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: LSE + Newton alternating (200s) ===
    print(f"\n[Phase 6] LSE + Newton alternating")
    p6_deadline = min(deadline, t_start + 1300)
    for rnd in range(4):
        if time.monotonic() > p6_deadline - 20:
            break
        c_cur = _full_convolve(a, a)
        mb = float(c_cur.max())
        beta = max(mb * 0.0001, 1e-12)
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        remaining = p6_deadline - time.monotonic()
        if remaining > 30:
            max_iter = min(2000, max(100, int(remaining / 0.06)))
            try:
                result = minimize(
                    _lse_objective, z, args=(beta, n),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 50}
                )
                a_lse = _softmax(result.x)
                if _score(a_lse) < _score(a):
                    a = a_lse
            except Exception:
                pass
        if time.monotonic() < p6_deadline - 10:
            a_new, sc_new = _newton_active_set(a, p6_deadline - 5, max_iter=80, tol_frac=0.0001, cg_tol=1e-13)
            if sc_new < _score(a) - 1e-15:

        _update_best(a, None)
        print(f"  round {rnd}: score={_score(a):.10f}, best={_best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Structured perturbations + re-optimize (200s) ===

    n_perts = 6

        if i == 0:
            p_start = _perturb_log(a, 0.01, rng)
        elif i == 1:

        elif i == 3:

        elif i == 4:

        nw_dead = min(p7_deadline - 10, time.monotonic() + 50)

        if _score(a_r) < _score(a) - 1e-15:

        _update_best(a, None)
        print(f"  perturb {i}: score={_score(a):.10f}, best={_best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Multi-n exploration (150s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    p8_deadline = min(deadline, t_start + 1650)

    if time.monotonic() < deadline - 10:

            mb = float(c_cur.max())
            beta_final = max(mb * 1e-6, 1e-14)
