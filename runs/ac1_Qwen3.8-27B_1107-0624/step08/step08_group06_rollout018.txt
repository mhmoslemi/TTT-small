import numpy as np
import time
from scipy.optimize import minimize

def _nfft(n):
    """Next power of 2 >= n"""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    """Full convolution using FFT"""
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _nfft(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]

def _score(a):
    """Compute the objective value"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr(a, w, n):
    """Compute correlation: (w * a_reversed)[n-1:2n-1]"""
    a_rev = a[::-1]
    out_len = n + len(w) - 1
    N = _nfft(out_len)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n-1:2*n-1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    """Log-sum-exp smoothed objective"""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective"""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _spectral_shaping(a, n_iter=500, alpha_start=0.3, alpha_end=0.8):
    """
    Iterative spectral shaping: flatten the power spectrum of a.
    At each step: FFT, raise amplitude to power alpha, IFFT, project to simplex.
    """
    n = len(a)
    a_work = a.copy()
    a_work = np.maximum(a_work, 0)
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work / s
    
    best_a = a_work.copy()
    best_score = _score(a_work)
    N = _nfft(2 * n - 1)
    
    for it in range(n_iter):
        # Adaptive alpha: start with more flattening, end with less
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iter - 1))
        
        A = np.fft.rfft(a_work, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: raise to power alpha (alpha < 1 flattens)
        mag_new = mag ** alpha
        # Normalize to preserve energy
        mag_new = mag_new * (np.sum(mag**2) / max(1e-30, np.sum(mag_new**2))) ** 0.5
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a_work = a_new
        
        if it % 50 == 0:
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_a = a_work.copy()
    
    sc = _score(a_work)
    if sc < best_score:
        best_a = a_work.copy()
    return best_a

def _subgradient_momentum(a, deadline, n_steps=10000):
    """
    Subgradient method with momentum for min max_k c_k s.t. sum a = 1, a >= 0.
    Uses the "active set" of near-max positions for the subgradient.
    """

    a_work = a.copy()
    a_work = np.maximum(a_work, 0)
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work / s
    
    best_a = a_work.copy()
    best_score = _score(a_work)
    
    v = np.zeros(n)  # momentum
    beta_mom = 0.9
    
    # Adaptive step size
    step = 0.01
    best_f = _score(a_work)
    
    for step_it in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a_work, a_work)
        c_max = float(c.max())
        
        # Active set: positions where c_k is within 2% of max
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Subgradient: 2 * correlation of w with reversed a
        grad = 2.0 * _corr(a_work, w, n)
        
        # Project gradient to tangent space of simplex
        g_t = grad - np.mean(grad)
        
        # Momentum update
        v = beta_mom * v - step * g_t
        
        # Update
        a_new = a_work + v
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            step *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check improvement
        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        
        a_work = a_new
        
        # Adaptive step size
        if sc < best_f - 1e-10:
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.8
            v = beta_mom * v  # rollback momentum partially
            a_work = best_a.copy()
            best_f = sc if sc < best_f else best_f
    
    return best_a

            break

        tol = tol_frac * c_max

        if not active.any():
            active[np.argmax(c)] = True
        
        m = int(active.sum())
        if m < 2:
            active[np.argmax(c)] = True
            m = int(active.sum())

        if m > 8 * n:

            active[top_idx] = True

        # where T_k is the Toeplitz matrix for lag k
        # We apply H to a vector v as: (2/m) * _corr(v, A_vec, n) + mu * v

        # We solve: minimize (1/2) x^T H x + g_t^T x s.t. sum(x) = 0
        # KKT: H x + g_t + lambda * 1 = 0, sum(x) = 0
        
        def hess_matvec(v):
            return (2.0 / m) * _corr(v, A_vec, n) + mu * v

def _lbfgs_smooth(a, deadline, min_time_remaining=30.0):

    max_c = float(c_cur.max())

        return a

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:

    else:
        gap = c_sorted[0] * 0.01
    base_beta = max(gap * 0.5, max_c * 0.02)
    
    # Beta schedule: geometric decay
    n_betas = 12
    betas = base_beta * (0.5 ** np.arange(n_betas))
    
    current_a = a.copy()
    current_score = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - np.mean(z)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2:
1. Spectral shaping (FFT-based amplitude flattening)
2. Log-sum-exp smoothed L-BFGS with adaptive beta cooling
3. Active-set Newton with CG-solved KKT system
4. Subgradient method with Nesterov momentum
5. Iterative peak equalization
6. Final polish with very small beta L-BFGS
"""
import numpy as np
import time
from scipy.optimize import minimize


def _nfft(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _nfft(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_vec(a, w, n):
    """Compute (w * reversed(a))[n-1 : 2n-1] - the 'correlation' vector."""
    a_rev = a[::-1]
    out_len = n + len(w) - 1
    N = _nfft(out_len)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    """Log-sum-exp smoothed objective: 2*n*(c_max + beta*log(sum(exp((c-c_max)/beta))))."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700.0, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700.0, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    wc = _full_convolve(w, a[::-1])
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _spectral_shaping(a, n_iter=400, alpha_start=0.15, alpha_end=0.65):
    """
    Iterative spectral flattening: FFT -> raise magnitude to alpha -> IFFT -> project to simplex.
    Fast pre-conditioner that drives the spectrum toward flatness.
    """
    n = len(a)
    a_work = a.copy()
    a_work = np.maximum(a_work, 0.0)
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work / s

    best_a = a_work.copy()
    best_score = _score(a_work)
    N = _nfft(2 * n - 1)

    for it in range(n_iter):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iter - 1))

        A = np.fft.rfft(a_work, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten spectrum by raising magnitude to power < 1
        mag_new = mag ** alpha
        energy_orig = np.dot(mag, mag)
        energy_new = np.dot(mag_new, mag_new)
        if energy_new > 1e-30:
            mag_new = mag_new * (energy_orig / energy_new) ** 0.5

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project to probability simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_work = a_new

        if it % 50 == 0:
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_a = a_work.copy()

def _active_set_newton(a_init, deadline, max_iter=300):
    """
    Active-set Newton: at each step, identify near-max convolution positions,
    form the Hessian as sum of Toeplitz matrices over active set, solve KKT with CG.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Tightening tolerance
        tol_frac = max(0.0005, 0.01 * (1.0 - it / max(1, max_iter)))
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())

        # Cap active set
        if m > 10 * n:
            top_idx = np.argsort(c)[::-1][:10 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = int(active.sum())
        if m < 2:
            continue

        A_vec = active.astype(np.float64)

        # Gradient averaged over active set
        g = (2.0 / m) * _corr_vec(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Regularization
        mu = 1e-7 * (c_max / max(m, 1)) + 1e-9

        # CG to solve: [H  1; 1^T  0] [x; lam] = [-g; 0]
        # where H v = (2/m) * _corr_vec(v, A_vec, n) + mu * v
        # We solve in tangent space: find dx with sum(dx)=0 minimizing (1/2)dx^T H dx + g_t^T dx

        rhs = -g_t
        x = np.zeros(n)
        r = rhs.copy()
        p = r - np.mean(r)
        rs = np.dot(p, p)
        rhs_norm = np.linalg.norm(rhs)

        for _ in range(250):
            Hp = (2.0 / m) * _corr_vec(p, A_vec, n) + mu * p
            Hp = Hp - np.mean(Hp)
            pHp = np.dot(p, Hp)
            if pHp < 1e-30:
                break
            alpha_cg = rs / pHp
            x = x + alpha_cg * p
            r = rhs - ((2.0 / m) * _corr_vec(x, A_vec, n) + mu * x)
            r = r - np.mean(r)
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12 * (1.0 + rhs_norm):
                break
            p = r + (rs_new / rs) * p
            p = p - np.mean(p)
            rs = rs_new

        dx = x - np.mean(x)

        # Backtracking line search
        t = 1.0
        f_old = c_max
        accepted = False
        for ls in range(40):
            a_try = a + t * dx
            neg = a_try < 0
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
                t = min(t, max(t_max, 1e-10))
            a_try = np.maximum(a + t * dx, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            f_try = float(_full_convolve(a_try, a_try).max())
            if f_try <= f_old - 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Fallback: negative gradient step
            t = 0.02
            for ls in range(20):
                a_try = np.maximum(a - t * g_t, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old - 1e-12:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _subgradient_momentum(a_init, deadline, n_steps=8000):
    """
    Subgradient method with Nesterov momentum for min max_k c_k on simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    v = np.zeros(n)
    beta_m = 0.9
    step = 0.005
    stall_count = 0
    best_step_at = step

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with fixed relative tolerance
        tol = 0.015 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Subgradient
        grad = 2.0 * _corr_vec(a, w, n)
        g_t = grad - np.mean(grad)

        # Nesterov momentum
        v = beta_m * v - step * g_t

        y = a + ((it + 1) / (it + 1 + 1.0)) * 0  # simplified; use standard momentum
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            step *= 0.5
            continue
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            stall_count = 0
            step = min(step * 1.05, 0.02)
            best_step_at = step
        else:
            stall_count += 1
            if stall_count % 50 == 0:
                step *= 0.85

        a = a_new

    return best_a


def _peak_equalize(a_init, deadline, n_rounds=40):
    """
    Iteratively identify top convolution peaks and reduce their dominant contributors.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top peaks
        n_peaks = max(3, min(15, int(0.003 * len(c))))
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # For each peak, find and reduce the dominant pair
        a_red = a.copy()
        for k_pos in top_idx:
            k_pos = int(k_pos)
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                j = k_pos - i
                if j < n:
                    prod = a[i] * a[j]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a[max_i] >= a[j]:
                    a_red[max_i] -= 0.002 * max_prod / (a[max_i] + 1e-30) * a[max_i]
                else:
                    a_red[j] -= 0.002 * max_prod / (a[j] + 1e-30) * a[j]

        a_red = np.maximum(a_red, 0.0)
        s_red = a_red.sum()
        if s_red < 1e-15:
            break
        a_red = a_red / s_red

        sc = _score(a_red)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_red.copy()
            a = a_red
        else:
            # Try with smaller reduction
            a_small = a.copy()
            for k_pos in top_idx[:5]:
                k_pos = int(k_pos)
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0.0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    j = k_pos - i
                    if j < n:
                        prod = a_small[i] * a_small[j]
                        if prod > max_prod:
                            max_prod = prod
                            max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    if a_small[max_i] >= a_small[j]:
                        a_small[max_i] *= 0.9995
                    else:
                        a_small[j] *= 0.9995
            s_sm = a_small.sum()
            if s_sm > 1e-15:
                a_small = a_small / s_sm
                sc_sm = _score(a_small)
                if sc_sm < best_score - 1e-13:
                    best_score = sc_sm
                    best_a = a_small.copy()
                    a = a_small
                else:
                    break

    return best_a


def _perturb(a, sigma, rng):
    """Perturb in log-space and renormalize."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - np.mean(z)
    return _softmax(z)

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral shaping, log-sum-exp L-BFGS,
    active-set Newton, subgradient momentum, and peak equalization.
    """
    global t_start
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s_val = float(p_arr.sum())
        if s_val < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, uniform n={n_parent}")
        update_best(parent, _score(parent), "uniform")

    # === Phase 1: Spectral Shaping (fast, ~60-120s) ===
    print(f"\n[Phase 1] Spectral Shaping (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline, t_start + 90.0)
    a_spec = _spectral_shaping(parent, n_iter=300, alpha_start=0.12, alpha_end=0.6)
    sc_spec = _score(a_spec)
    update_best(a_spec, sc_spec, "spectral")

    # === Phase 2: L-BFGS main (300-500s) ===
    print(f"\n[Phase 2] L-BFGS Main (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p2_dl = min(deadline, t_start + 650.0)
        a_lb1 = _lbfgs_smooth(best_seq, p2_dl, min_time_remaining=250.0)
        sc_lb1 = _score(a_lb1)
        update_best(a_lb1, sc_lb1, "lbfgs_1")

    # === Phase 3: Active-set Newton (~150s) ===
    print(f"\n[Phase 3] Active-Set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        p3_dl = min(deadline - 80.0, time.monotonic() + 150.0)
        a_as, sc_as = _active_set_newton(best_seq, p3_dl, max_iter=250)
        update_best(a_as, sc_as, "as_newton")

    # === Phase 4: Subgradient momentum (~100s) ===
    print(f"\n[Phase 4] Subgradient Momentum (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        p4_dl = min(deadline - 60.0, time.monotonic() + 100.0)
        a_sg = _subgradient_momentum(best_seq, p4_dl, n_steps=6000)
        sc_sg = _score(a_sg)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 5: Peak equalization (~60s) ===
    print(f"\n[Phase 5] Peak Equalization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        p5_dl = min(deadline - 40.0, time.monotonic() + 60.0)
        a_pe = _peak_equalize(best_seq, p5_dl, n_rounds=50)
        sc_pe = _score(a_pe)
        update_best(a_pe, sc_pe, "peak_eq")

    # === Phase 6: L-BFGS round 2 (~150s) ===
    print(f"\n[Phase 6] L-BFGS Round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        p6_dl = min(deadline - 20.0, time.monotonic() + 150.0)
        a_lb2 = _lbfgs_smooth(best_seq, p6_dl, min_time_remaining=25.0)
        sc_lb2 = _score(a_lb2)
        update_best(a_lb2, sc_lb2, "lbfgs_2")

    # === Phase 7: Active-set Newton round 2 (~60s) ===
    print(f"\n[Phase 7] Active-Set Newton 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        p7_dl = min(deadline - 8.0, time.monotonic() + 60.0)
        a_as2, sc_as2 = _active_set_newton(best_seq, p7_dl, max_iter=100)
        update_best(a_as2, sc_as2, "as_newton_2")

    # === Phase 8: Perturbation + quick L-BFGS (~40s) ===
    print(f"\n[Phase 8] Perturb + Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        for tag, sig in [("small", 0.01), ("med", 0.03)]:
            if time.monotonic() > deadline - 8:
                break
            p_pt = _perturb(best_seq, sig, rng)
            p8_dl = min(deadline - 3.0, time.monotonic() + 25.0)
            a_pt = _lbfgs_smooth(p_pt, p8_dl, min_time_remaining=5.0)
            sc_pt = _score(a_pt)
            update_best(a_pt, sc_pt, f"perturb_{tag}")

    # === Final: subgradient polish on very best ===
    print(f"\n[Final] Subgradient Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin = _subgradient_momentum(best_seq, deadline - 1.0, n_steps=3000)
        sc_fin = _score(a_fin)
        update_best(a_fin, sc_fin, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        eval_sc = evaluate_sequence([float(x) for x in final])
        print(f"Evaluated: {eval_sc:.10f}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
