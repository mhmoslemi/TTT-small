    # Gradient of c[k] w.r.t. a: g_k[i] = 2*a[k-i] (for valid k-i)
    # We need: grad_z = a * (grad_a - sum(grad_a * a))
    # where grad_a[i] = sum_k p[k] * 2 * a[k-i]
    
    grad_a = np.zeros(n)
    # For efficiency, use the top lags
    # Actually, let's use all lags with significant weight
    threshold = p.max() * 1e-6
    active = p > threshold
    if not np.any(active):
        active[np.argmax(p)] = True
    
    for k in np.where(active)[0]:
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad_a[lo:hi+1] += pk * 2.0 * a[k-hi:k-lo+1][::-1]
    
    # Center: grad_z = a * (grad_a - dot(a, grad_a))
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    
    return obj, grad_z

def _lbfgs_optimize(a_init, T, deadline, max_iter=500, tol=1e-10):
    """Run L-BFGS on the log-space parameterization."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    state = {'count': 0}
    
    def fun(z):
        state['count'] += 1
        obj, grad = _smoothed_obj_and_grad(z, T)
        return obj, grad
    
    try:
        result = minimize(fun, z0, method='L-BFGS-B', 
                         options={'maxiter': max_iter, 'ftol': tol, 
                                  'gtol': 1e-8, 'maxfun': max_iter * 3,
                                  'maxls': 20})
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return a_init, _score_norm(a_init)
    
    a_final = _softmax_stable(result.x)
    sc = _score_norm(a_final)
    print(f"  L-BFGS: {result.niter} iters, {state['count']} fun evals, score={sc:.10f}")
    return a_final, sc

def _adam_optimize(a_init, T_start, T_end, lr, deadline, max_iter=100000):
    """Adam optimizer in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        cur_lr = lr * (0.1 ** frac)
        
        obj, grad_z = _smoothed_obj_and_grad(z, T)
        
        m_buf = b1 * m_buf + (1 - b1) * grad_z
        v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 200 == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
    
    a_final = _softmax_stable(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score

def _active_set_polish(a, deadline, max_iter=2000):
    """Polish using active-set equalization."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks (within 0.5% of max)
        tol_val = c_max * 0.9995
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 10:
            sorted_a = active[np.argsort(c[active])[::-1][:10]]
            active = sorted_a
        else:
            active = active[:10]
        
        K = len(active)
        # Build constraint matrix
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Add sum-zero constraint
        M_full = np.vstack([M, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])
        
        G = M_full @ M_full.T
        try:
            d = M_full.T @ np.linalg.solve(G + 1e-10 * np.eye(G.shape[0]), b)
        except:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        improved = False
        for alpha in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = np.maximum(0, a + alpha * d)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score

"""
Optimized step function search using FFT-accelerated gradients, L-BFGS with
temperature continuation, active-set equalization, and diverse restarts.
Target: minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed_logspace(z, T):
    """Compute (objective, gradient) of smoothed max-conv in log-space.
    
    obj = T * log(sum_k exp(c[k]/T)) where c = conv(softmax(z), softmax(z))
    grad = d(obj)/dz = a * (grad_a - <a, grad_a>)
    grad_a[i] = 2 * sum_k p[k] * a[k-i] where p[k] = exp(c[k]/T) / sum exp
    """
    a = _softmax_stable(z)
    n = len(a)
    c = _fft_convolve(a, a)
    
    # Softmax weights
    c_max = c.max()
    e = np.exp((c - c_max) / T)
    s = e.sum()
    p = e / s
    
    # Objective (up to constant)
    obj = c_max + T * np.log(s)
    
    # Compute grad_a via FFT cross-correlation
    # grad_a[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * cross_correlation(p, a)[i]
    # Using FFT: cross = IDFT(conj(DFT(p)) * DFT(a)), first n elements
    N = _next_pow2(2 * n - 1 + n - 1)  # >= 3n-2
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(P) * A, N)
    grad_a = 2.0 * cross[:n]
    
    # Center for log-space
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    
    return obj, grad_z


def _lbfgs_run(a_init, T, max_iter=300, gtol=1e-7):
    """Run L-BFGS on log-space parameterization with given temperature."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    counter = [0]
    
    def fun(z):
        counter[0] += 1
        return _grad_smoothed_logspace(z, T)
    
    try:
        res = minimize(fun, z0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': gtol,
                                'maxfun': max_iter * 5, 'maxls': 30})
    except Exception as ex:
        print(f"    L-BFGS exception: {ex}")
        return a_init.copy()
    
    a_out = _softmax_stable(res.x)
    return a_out


def _adam_logspace(a_init, T_start, T_end, lr_start, lr_end, deadline, max_iter=100000):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.8)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        obj, grad_z = _grad_smoothed_logspace(z, T)
        
        m_buf = b1 * m_buf + (1 - b1) * grad_z
        v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % 500 == 0:
            a_cur = _softmax_stable(z)
            sc = _score(a_cur)
            if sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()
    
    a_final = _softmax_stable(z)
    sc = _score(a_final)
    if sc < best_sc:
        best_sc = sc
        best_a = a_final.copy()
    return best_a


def _active_set_polish(a, deadline, max_iter=10000, peak_tol=0.999):
    """Polish by equalizing the top peaks using linear algebra."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks

        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 12:
            idx = np.argsort(c[active])[::-1][:12]
            active = active[idx]
        else:
            active = active[:12]
        
        K = len(active)
        if K < 2:
            # Only one peak; try to reduce it by moving mass away
            k0 = int(active[0])
            lo = max(0, k0 - n + 1)
            hi = min(k0, n - 1)
            # Direction: reduce a[lo:hi+1] proportionally, increase others
            d = np.zeros(n)
            mid = (lo + hi) // 2
            d[lo:hi+1] = -a[lo:hi+1]
            other = np.zeros(n)
            other[lo:hi+1] = 0

                mask[lo:hi+1] = False
                d[mask] = a[mask] * 0.5
            d = d - d.sum() * a  # project to sum-zero
            dn = np.linalg.norm(d)
            if dn < 1e-15:

            d = d / dn

            M = np.zeros((K, n))
            for i, k in enumerate(active):
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo <= hi:
                    M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
            
            # Add sum-zero constraint
            M_full = np.vstack([M, np.ones(n)])
            b = np.concatenate([-np.ones(K), [0.0]])

            try:

                d = M_full.T @ (G_inv @ b)
            except np.linalg.LinAlgError:
                d = np.linalg.lstsq(M_full, b, rcond=None)[0]
            
            dn = np.linalg.norm(d)
            if dn < 1e-15:
                break
            d = d / dn
        
        # Try step sizes
        improved = False
        for alpha in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            sc2 = a_cand.sum()
            if sc2 < 1e-15:
                continue
            a_cand = a_cand / sc2
            sc = _score(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            # Try to escape: small random perturbation
            if it < max_iter - 100:
                d_rand = np.random.randn(n)
                d_rand -= d_rand.sum() * a
                dn = np.linalg.norm(d_rand)
                if dn > 1e-15:
                    d_rand /= dn
                    for alpha in [0.001, 0.0005]:
                        a_cand = np.maximum(0, a + alpha * d_rand)
                        sc2 = a_cand.sum()
                        if sc2 < 1e-15:
                            continue
                        a_cand /= sc2
                        sc = _score(a_cand)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a_cand.copy()
                            a = a_cand
                            improved = True
                            break
            if not improved and it > 200:
                break
