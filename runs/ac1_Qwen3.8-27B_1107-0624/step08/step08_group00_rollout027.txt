    return e / s


def _compute_smoothed_grad(a, beta, n):
    """Compute gradient of 2n * beta * log(sum exp(c/beta)) where c = conv(a,a).
    Returns (value, grad) where grad is w.r.t. a."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        val = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        val = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    # grad[j] = 2 * sum_k w[k] * a[k-j] (a[x]=0 for x outside [0,n-1])
    # = 2 * np.correlate(w, a, 'full')[n-1 : 2n-1]
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return val, grad, w


def _subgradient_max(a, n, top_k=10):
    """Compute subgradient of 2n*max(conv(a,a)) using top-k peaks."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Get indices of top peaks
    idx = np.argpartition(c, -top_k)[-top_k:]
    vals = c[idx]
    weights = vals / vals.sum()
    w = np.zeros_like(c)
    w[idx] = weights
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[n - 1: 2 * n - 1]
    return 2.0 * n * c_max, grad


def _nesterov_smoothed_grad(a_init, deadline, beta_start=None, gamma=0.9, min_time=30.0):
    """Nesterov-accelerated smoothed gradient descent with adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_nn_sum1(a)
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    t = 1.0
    t_prev = 1.0

    if beta_start is None:
        c = _full_convolve(a, a)
        cs = np.sort(c)[::-1]
        gap = cs[0] - cs[min(20, len(cs) - 1)]
        beta_start = max(gap, cs[0] * 0.01)

    beta = beta_start
    alpha = 0.01
    stall = 0
    max_stall = 50

    # Initialize momentum with Nesterov
    mu = t  # Nesterov momentum parameter
    y = a.copy()  # lookahead point
    a_prev = a.copy()

    while time.monotonic() + min_time < deadline:
        # Compute gradient at lookahead point y
        val, grad, w = _compute_smoothed_grad(y, beta, n)

        # Project gradient to sum=0 subspace (maintain sum constraint)
        grad = grad - np.mean(grad)

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / (t_new + 1.0)
        y = a + momentum * (a - a_prev) - alpha * grad

        # Project to feasible set
        y = _project_nn_sum1(y)

        # Check improvement
        sc_y = _score(y)
        if sc_y < best_sc - 1e-14:
            best_sc = sc_y
            best_a = y.copy()
            stall = 0
            # Increase learning rate slightly
            alpha = min(0.1, alpha * 1.02)
        else:
            stall += 1
            alpha *= 0.8
            if stall > max_stall:
                # Reset and reduce beta
                beta *= 0.5
                alpha = 0.01
                stall = 0
                a = best_a.copy()
                a_prev = a.copy()
                y = a.copy()
                t = 1.0
                if beta < best_sc * 1e-5:
                    break

        # Adapt beta: reduce effective active set size over time
        eff_active = 1.0 / np.sum(w * w)
        target_eff = max(5.0, n * 0.001)
        if eff_active > target_eff * 2:
            beta *= 0.7
        elif eff_active < target_eff * 0.5:
            beta *= 1.3
        beta = max(beta, best_sc * 1e-6)

        a_prev = a
        a = y
        t = t_new

        if beta < best_sc * 1e-5 and stall > 30:
            break

    return best_a, best_sc


def _freq_domain_flatten(a_init, deadline, min_time=30.0):
    """Frequency-domain spectral shaping: flatten |FFT(a)| to reduce autocorrelation peak."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_nn_sum1(a)
    best_a = a.copy()
    best_sc = _score(a)

    alpha_power = 0.8  # Start with moderate compression
    gamma = 0.1  # Blend factor

    for iteration in range(2000):
        if time.monotonic() + min_time > deadline:
            break

        A = np.fft.fft(a)
        abs_A = np.abs(A)
        phase_A = np.angle(A)

        # Compress dynamic range of |A|
        abs_A_mean = abs_A.mean()
        if abs_A_mean < 1e-15:
            break
        abs_A_norm = abs_A / abs_A_mean

        # Interpolate between original and flattened spectrum
        abs_A_new = abs_A_norm ** alpha_power
        abs_A_blend = (1.0 - gamma) * abs_A_norm + gamma * abs_A_new
        abs_A_final = abs_A_blend * abs_A_mean

        A_new = abs_A_final * np.exp(1j * phase_A)
        a_new = np.fft.ifft(A_new).real
        a_new = _project_nn_sum1(a_new)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            # Increase compression
            alpha_power = max(0.3, alpha_power * 0.97)
            gamma = min(0.5, gamma * 1.01)
        else:
            # Decrease perturbation
            gamma *= 0.9
            alpha_power = min(0.95, alpha_power * 1.02)
            if alpha_power > 0.94 and gamma < 0.02:
                break

    return best_a, best_sc


def _subgrad_nesterov(a_init, deadline, min_time=30.0, top_k=5):
    """Nesterov-accelerated subgradient method for the non-smooth max."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_nn_sum1(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_prev = a.copy()
    t = 1.0
    alpha = 1.0 / np.linalg.norm(a) ** 2 * 0.01
    stall = 0

    while time.monotonic() + min_time < deadline:
        val, grad = _subgradient_max(a, n, top_k=top_k)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / (t_new + 1.0)

        # Step
        step = alpha / t_new
        y = a + momentum * (a - a_prev) - step * grad
        y = _project_nn_sum1(y)

        sc_y = _score(y)
        if sc_y < best_sc - 1e-14:
            best_sc = sc_y
            best_a = y.copy()
            stall = 0
            alpha = min(alpha * 1.05, 1.0)
        else:
            stall += 1
            alpha *= 0.9
            if stall > 80:
                break

        a_prev = a
        a = y
        t = t_new

    return best_a, best_sc


def _active_set_gradient(a_init, deadline, min_time=30.0):
    """Active-set gradient: focus on the top peaks with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_nn_sum1(a)
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.01
    lr = 0.02
    stall = 0

    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
            tol_frac = max(0.001, tol_frac * 0.95)
        else:
            stall += 1
            if stall % 20 == 0:
                tol_frac = max(0.0005, tol_frac * 0.8)
                lr *= 0.7
            if stall > 300:
                break

        # Active set: peaks within tol_frac of max
        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Uniform weights on active set
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        corr = np.correlate(w, a, mode='full')
        grad = 2.0 * corr[n - 1: 2 * n - 1]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        a_old_sc = current_sc
        step = lr / gnorm
        improved = False
        for _ in range(40):
            a_try = a - step * grad
            a_try = _project_nn_sum1(a_try)
            sc_try = _score(a_try)
            if sc_try < a_old_sc * (1.0 - 1e-6):
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            lr *= 0.9

    return best_a, best_sc


def _lbfgs_smoothed(a_init, deadline, beta=None, min_time=30.0):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a_init)

    if beta is None:
        c = _full_convolve(a_init, a_init)
        cs = np.sort(c)[::-1]
        gap = cs[0] - cs[min(20, len(cs) - 1)]
        beta = max(gap, cs[0] * 0.005)

    class _Prob:
        def __call__(self, z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad(self, z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            corr = np.correlate(w, a, mode='full')
            h = corr[n - 1: 2 * n - 1]
            return 2.0 * h - 2.0 * np.mean(h)

    prob = _Prob()
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    remaining = deadline - min_time - time.monotonic()
    if remaining < 5:
        return a_init, _score(a_init)

    max_iter = min(3000, max(50, int(remaining / 0.02)))
    try:
        result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxcor': 50, 'maxls': 30})
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        return a_new, sc_new
    except Exception:
        return a_init, _score(a_init)


def _structured_perturb(a, rng, n_perturbs=5):
    """Apply structured frequency-domain perturbations to escape local optima."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    candidates = []

    # Add cosine perturbations at various frequencies
    for _ in range(n_perturbs):
        freq = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        pert = amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
        a_try = _project_nn_sum1(a + pert)
        sc_try = _score(a_try)
        candidates.append((sc_try, a_try))

    # Add random smooth perturbation
    for _ in range(3):
        sigma = rng.uniform(0.05, 0.15)
        z = np.log(np.maximum(a, 1e-30))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_try = _softmax(z)
        sc_try = _score(a_try)
        candidates.append((sc_try, a_try))

    candidates.sort(key=lambda x: x[0])
    if candidates[0][0] < best_sc:
        best_a = candidates[0][1]
        best_sc = candidates[0][0]

    return best_a, best_sc


def _shrinkage_optimize(a, deadline, min_time=10.0):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() + min_time > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        a_try = _project_nn_sum1(a_try)
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return _project_nn_sum1(a_new)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: Nesterov smoothed gradient -> freq-domain shaping ->
    subgradient -> active-set -> multi-restart -> L-BFGS polish.
    """
    t_start = time.monotonic()

    best_score = np.inf

    parent = None

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        n_main = n_parent
        a_main = parent.copy()
        best_score = _score(a_main)
        best_seq = a_main.copy()
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_main = 16384
        a_main = _make_profile(n_main, "uniform")
        best_score = _score(a_main)
        best_seq = a_main.copy()
        print(f"No parent, uniform n={n_main}, score={best_score:.10f}")

    # === Phase 1: Nesterov smoothed gradient (main optimization) ===
    print(f"\n[P1] Nesterov smoothed gradient (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, time.monotonic() + 700)
    a1, sc1 = _nesterov_smoothed_grad(a_main, p1_deadline, min_time=100)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  NEW BEST: {best_score:.10f} [nesterov] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS with multi-beta ===
    print(f"\n[P2] L-BFGS multi-beta (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        for beta_frac in [0.01, 0.005, 0.002, 0.001]:
            if time.monotonic() > deadline - 400:
                break
            beta = mb * beta_frac
            p2_deadline = min(deadline - 400, time.monotonic() + 120)
            a2, sc2 = _lbfgs_smoothed(best_seq, p2_deadline, beta=beta, min_time=60)
            if sc2 < best_score:

                best_seq = a2.copy()
                print(f"  NEW BEST: {best_score:.10f} [lbfgs beta={beta:.6f}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Frequency-domain spectral shaping ===
    print(f"\n[P3] Frequency-domain flattening (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, time.monotonic() + 200)
        a3, sc3 = _freq_domain_flatten(best_seq, p3_deadline, min_time=60)
        if sc3 < best_score:

            best_seq = a3.copy()
            print(f"  NEW BEST: {best_score:.10f} [freq_flat] t={time.monotonic()-t_start:.1f}s")
        else:

    # === Phase 4: Active-set gradient ===

            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
