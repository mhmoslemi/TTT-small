"""
Primal-dual convex optimization for minimizing the self-convolution peak.
Uses FFT-accelerated Hankel matrix operations, adaptive L-BFGS with log-sum-exp
smoothing, multiplicative-weights dual acceleration, and active-set QP polishing.
Key insight: the gradient of (a*a)[k] is 2*shifted_reversed(a), enabling
fast gradient computation. The problem is convex, so primal-dual methods
with multiplicative weights guarantee convergence.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """Compute H*v where H_{i,j} = W[i+j], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-12):
    """Solve H*x = rhs using CG where H is Hankel with entries W[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_worker(args):
    z, beta, n = args
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _smograd(z, beta, n):
    return _smograd_worker((z, beta, n))


def _smoobj_noargs_wrapper(z, beta_holder, n_holder):
    return _smoobj(z, beta_holder[0], n_holder[0])


def _smograd_wrapper(z, beta_holder, n_holder):
    return _smograd(z, beta_holder[0], n_holder[0])


def _run_lbfgs(a_init, deadline, min_time_remaining=100.0, aggressive=True):
    """L-BFGS with adaptive beta continuation on log-parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.001)

    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                  0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                  0.0000005, 0.0000002, 0.0000001]
    else:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                  0.00002, 0.00001, 0.000005, 0.000002, 0.000001]

    betas = [base_beta * r for r in ratios]
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(100, int(remaining / (est_time_per_iter * 5))))

        beta_ref = [beta]
        n_ref = [n]

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.clip(a_new, 0, None)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _solve_qp_hankel(W, n, a_init, max_cg_iter=800):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 where H is Hankel(W).
    Returns the solution a."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    # Try interior solution first
    ones = np.ones(n)
    x = _cg_solve_hankel(W, n, ones, max_iter=max_cg_iter)
    sx = float(np.sum(x))
    if abs(sx) < 1e-20 or not np.isfinite(sx):
        return a

    a_int = x / sx
    # Check non-negativity
    if np.min(a_int) > -1e-10:
        a_int = np.maximum(a_int, 0)
        s_int = a_int.sum()
        if s_int > 1e-15:
            a_int = a_int / s_int
            return a_int

    # Active set: iteratively fix negative components to 0
    a_as = a_int.copy()
    fixed = np.zeros(n, dtype=bool)
    for _ in range(min(50, n)):
        if np.min(a_as[~fixed]) > -1e-12:
            break
        worst = np.argmin(np.where(fixed, np.inf, a_as))
        fixed[worst] = True
        a_as[worst] = 0.0
        free = ~fixed
        n_free = int(free.sum())
        if n_free < 2:
            break
        # Solve reduced system
        rhs_f = np.zeros(n)
        rhs_f[free] = 1.0
        x_f = _cg_solve_hankel(W, n, rhs_f, max_iter=max_cg_iter // 2)
        x_f[fixed] = 0.0
        sx_f = float(np.sum(x_f))
        if abs(sx_f) < 1e-20:
            break
        a_as = x_f / sx_f
        a_as[fixed] = 0.0

    a_as = np.maximum(a_as, 0)
    s_as = a_as.sum()
    if s_as < 1e-15:
        return a
    a_as = a_as / s_as
    return a_as


def _primal_dual_optimize(a_init, deadline, initial_eta=0.01, max_iters=5000):
    """Primal-dual multiplicative weights optimization.
    Primal: min_a max_k (a*a)[k] s.t. a in simplex
    Dual:   max_p min_a sum_k p_k (a*a)[k]
    Uses multiplicative weight updates on p and QP solves on a."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    # Initialize dual distribution p
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol_init = 0.05 * c_max
    active = c >= (c_max - tol_init)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / active.sum()

    # Store W = p (the Hankel matrix parameters)
    W = p.copy()

    eta = initial_eta
    t = 0
    patience = 0
    best_gap = np.inf

    while t < max_iters:
        if time.monotonic() > deadline:
            break

        # --- Primal step: solve QP min a^T H_W a s.t. a in simplex ---
        a_qp = _solve_qp_hankel(W, n, a, max_cg_iter=300)
        a = a_qp

        # Compute convolution and score
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # --- Compute duality gap ---
        weighted_c = float(np.dot(p, c))
        gap = c_max - weighted_c
        if gap < best_gap:
            best_gap = gap
        if gap < 1e-10 * c_max:
            break

        # --- Dual step: multiplicative weights update ---
        # p_k *= exp(eta * c_k), normalize
        eta_scaled = eta / (c_max + 1e-30)
        log_p = np.full(2 * n - 1, -1000.0)
        valid = p > 1e-30
        log_p[valid] = np.log(p[valid]) + eta_scaled * c
        log_p -= np.max(log_p[valid])
        p_new = np.zeros(2 * n - 1)
        p_new[valid] = np.exp(np.clip(log_p[valid], -700, 0))
        s_p = p_new.sum()
        if s_p < 1e-300:
            break
        p_new = p_new / s_p
        p = p_new
        W = p.copy()

        # Adaptive eta: decrease if gap not decreasing
        if patience > 20:
            eta *= 0.8
            patience = 0

        t += 1
        if t % 200 == 0:
            print(f"    PD iter {t}: score={sc:.8f}, gap={gap:.2e}, "
                  f"eta={eta:.4f}, active_frac={np.sum(p>1e-6)/(2*n-1):.4f}")

    return best_a, best_sc


def _active_set_polish(a_init, deadline, max_rounds=200):
    """Progressive tolerance active-set QP polishing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                    0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                    0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 5

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        a_qp = _solve_qp_hankel(W, n, a, max_cg_iter=500)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()

        # Check convergence
        c_check = _full_convolve(a, a)
        c_max_check = float(c_check.max())
        violation = c_max_check - float(np.dot(W, c_check))
        if violation < tol * c_max_check * 0.1:
            tol_idx += 1
        else:
            # Add most violated constraint
            viol_idx = int(np.argmax(c_check - float(np.dot(W, c_check))))
            active[viol_idx] = True
            n_active += 1
            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_active

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)

    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=500):
    """Direct gradient-based flattening of the convolution peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top-m active set
        m = max(10, min(200, int(0.02 * (2 * n - 1))))
        top_idx = np.argsort(c)[-m:]
        W = np.zeros(2 * n - 1)
        W[top_idx] = 1.0 / m

        # Gradient: 2 * sum_k W[k] * (shifted_reversed a by k)
        # This is 2 * (W * a_reversed)[n-1 : 2n-1]
        a_rev = a[::-1]
        Wa = _full_convolve(W, a_rev)
        grad = 2.0 * Wa[n - 1:2 * n - 1]

        # Project gradient onto tangent space of simplex
        grad_t = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_t)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        alpha = min(0.1, 1.0 / grad_norm)
        improved = False
        for ls in range(30):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1 - 1e-8):
                a = a_try
                sc_try = _score(a)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not improved:
            # Try smaller step
            alpha = 0.01 / (it + 1)
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0.01, 1.0, n)
        a = t ** (-0.3) + 0.1
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _nesterov_subgrad(a_init, deadline, max_steps=10000, lr=0.02, momentum=0.9):
    """Nesterov-accelerated subgradient on log-space with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    v = np.zeros(n)  # momentum
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        tol = max(0.001 * c_max, 1e-8)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        m = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / m

        # Gradient: 2 * a * ((W * a_rev)[n-1:2n-1] - E_W[c])
        a_rev = a[::-1]
        Wa = _full_convolve(W, a_rev)
        h = Wa[n - 1:2 * n - 1]
        W_c = float(np.dot(W, c))
        grad = 2.0 * n * a * (h - W_c)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        v = momentum * v - step * grad
        z_new = z + v
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    v = np.zeros(n)
                    patience = 0
                    lr *= 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimizer combining L-BFGS smoothing,
    primal-dual multiplicative weights, active-set QP, and
    Nesterov subgradient for minimizing 2n*max(conv(a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def update_best(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score_val:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, "init")

    # === Phase 1: L-BFGS coarse optimization (budget: ~500s) ===
    print(f"\n[Phase 1] L-BFGS coarse (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = min(deadline, t_start + 500)
    a1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=50.0, aggressive=True)
    update_best(a1, "lbfgs_p1")

    # Try a different starting point too
    a1b_init = _make_profile(n_parent, "asym_taper")
    if n_parent > 0:
        a1b_init = 0.7 * parent + 0.3 * a1b_init
        a1b_init = np.maximum(a1b_init, 0)
        a1b_init = a1b_init / a1b_init.sum()
    a1b = _run_lbfgs(a1b_init, min(deadline, t_start + 550), min_time_remaining=30.0, aggressive=True)
    update_best(a1b, "lbfgs_p1b")

    # === Phase 2: Primal-dual multiplicative weights (budget: ~600s) ===
    print(f"\n[Phase 2] Primal-dual MW (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(deadline, t_start + 1200)
    if best_seq is not None:
        a_pd, sc_pd = _primal_dual_optimize(best_seq, phase2_deadline,
                                            initial_eta=0.005, max_iters=3000)
        update_best(a_pd, "primal_dual")

    # === Phase 3: Active-set QP polish (budget: ~200s) ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic()-t_start:.1f}s)")
    phase3_deadline = min(deadline, t_start + 1500)
    if best_seq is not None:
        a_qp, sc_qp = _active_set_polish(best_seq, phase3_deadline, max_rounds=300)
        update_best(a_qp, "active_qp")

    # === Phase 4: Flatten polish (budget: ~100s) ===
    print(f"\n[Phase 4] Flatten polish (t={time.monotonic()-t_start:.1f}s)")
    phase4_deadline = min(deadline, t_start + 1600)
    if best_seq is not None:
        a_fl, sc_fl = _flatten_polish(best_seq, phase4_deadline, max_iters=500)
        update_best(a_fl, "flatten")

    # === Phase 5: Nesterov subgradient (budget: ~80s) ===
    print(f"\n[Phase 5] Nesterov subgrad (t={time.monotonic()-t_start:.1f}s)")
    phase5_deadline = min(deadline, t_start + 1700)
    if best_seq is not None:
        a_ns, sc_ns = _nesterov_subgrad(best_seq, phase5_deadline,
                                        max_steps=5000, lr=0.015, momentum=0.85)
        update_best(a_ns, "nesterov")

    # === Phase 6: Final tight polishing (budget: ~80s) ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 5, max_rounds=200)
        update_best(a_fin, "final_qp")

        if time.monotonic() < deadline - 5:
            a_fl2, sc_fl2 = _flatten_polish(best_seq, deadline - 3, max_iters=300)
            update_best(a_fl2, "final_flatten")

    # === Phase 7: Try different n values ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        for n_try in [32768, 8192]:
            if time.monotonic() > deadline - 20:
                break
            if n_try == n_parent:
                continue
            if best_seq is not None:
                a_try = _resample(best_seq, n_try)
            else:
                a_try = _make_profile(n_try, "tapered")
            a_opt = _run_lbfgs(a_try, min(deadline - 10, time.monotonic() + 60),
                               min_time_remaining=5.0, aggressive=True)
            update_best(a_opt, f"n_{n_try}")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        for i, prof in enumerate(["asym_taper", "power_law", "cos"]):
            if time.monotonic() > deadline - 10:
                break
            n_r = len(best_seq)
            p_start = _make_profile(n_r, prof)
            p_start = 0.6 * best_seq + 0.4 * p_start
            p_start = np.maximum(p_start, 0)
            p_start = p_start / p_start.sum()
            a_r = _run_lbfgs(p_start, min(deadline - 5, time.monotonic() + 25),
                             min_time_remaining=3.0)
            update_best(a_r, f"restart_{i}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
