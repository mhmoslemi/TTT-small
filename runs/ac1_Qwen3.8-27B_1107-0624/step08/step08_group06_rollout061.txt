                a_try = (1.0 - t) * a + t * a_qp
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_new_sc:
                    best_new_sc = sc_try
                    best_new_a = a_try.copy()

        if best_new_a is None:
            break

        a = best_new_a
        if best_new_sc < best_sc - 1e-15:
            best_sc = best_new_sc
            best_a = a.copy()
        elif best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()
        else:
            # No improvement, try smaller tolerance
            pass

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with aggressive multi-stage beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    elif len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = max_c * 0.1
    base_beta = max(gap * 0.3, max_c * 0.02)

    # Aggressive schedule: more stages at small beta
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.004, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations
        scale = max(1.0, n / 4096.0)
        if beta < base_beta * 0.005:
            est_per_iter = 0.004 * scale * 5
            max_iter = min(10000, max(200, int(remaining / est_per_iter)))
        elif beta < base_beta * 0.05:
            est_per_iter = 0.004 * scale * 4
            max_iter = min(8000, max(100, int(remaining / est_per_iter)))
        else:
            est_per_iter = 0.004 * scale * 3
            max_iter = min(5000, max(50, int(remaining / est_per_iter)))

        try:
            result = minimize(
                _lsobj,
                z,
                jac=_lsgrad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif sc_new < current_sc:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_shaving(a_init, deadline, n_steps=5000):
    """Direct peak shaving: iteratively reduce the top peak using subgradient."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    step_size = 0.02
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use average of top M peaks for smoother gradient
        M = max(3, min(20, int(0.02 * len(c))))
        top_idx = np.argsort(c)[::-1][:M]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / M

        # Gradient of (1/M) * sum_{k in top} c_k
        grad = (2.0 / M) * _corr_a_A(a, w, n)
        # Project onto tangent space (sum = 0)
        g_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        # Line search
        accepted = False
        t = step_size
        for _ in range(20):
            a_try = a - t * g_t / gnorm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc:
                a = a_try
                best_sc = sc_try
                best_a = a.copy()
                accepted = True
                step_size = min(0.05, t * 1.1)
                break
            t *= 0.5

        if not accepted:
            step_size *= 0.7
            if step_size < 1e-8:
                break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_structured(a, mode, rng):
    """Structured perturbation: add a low-frequency component."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    if mode == "cos1":
        t = np.linspace(0, 2 * np.pi, n)
        z += 0.05 * np.cos(t)
    elif mode == "cos2":
        t = np.linspace(0, 4 * np.pi, n)
        z += 0.05 * np.cos(t)
    elif mode == "bathtub":
        t = np.linspace(0, 1, n)
        z += 0.1 * np.cos(np.pi * t)
    elif mode == "rand_small":
        z += 0.02 * rng.standard_normal(n)
    elif mode == "rand_mid":
        z += 0.05 * rng.standard_normal(n)
    else:
        z += 0.03 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining aggressive L-BFGS, active-set QP,
    peak shaving, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent (most of the time)
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = deadline - 400.0
    a_main = _run_lbfgs(parent, lbfgs_deadline, min_time_remaining=400.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set QP refinement
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        as_deadline = min(deadline - 300, t_start + 1500)
        a_as, sc_as = _active_set_refine(best_seq, as_deadline, max_iter=200)
        update_best(a_as, sc_as, "as_qp_1")

    # Phase 3: Peak shaving
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        ps_deadline = min(deadline - 200, t_start + 1600)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_deadline, n_steps=3000)
        update_best(a_ps, sc_ps, "peak_shave")

    # Phase 4: L-BFGS polish after peak shaving
    print(f"\n[Phase 4] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_pol = _run_lbfgs(best_seq, min(deadline - 150, t_start + 1650), min_time_remaining=150.0)
        sc_pol = _score(a_pol)
        update_best(a_pol, sc_pol, "lbfgs_polish")

    # Phase 5: Active-set QP again
    print(f"\n[Phase 5] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_as2, sc_as2 = _active_set_refine(best_seq, min(deadline - 100, t_start + 1680), max_iter=150)
        update_best(a_as2, sc_as2, "as_qp_2")

    # Phase 6: Try n=32768
    print(f"\n[Phase 6] Larger n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_large = 32768
    if time.monotonic() < deadline - 80:
        a_large = _interpolate_to_n(best_seq, n_large)
        sc_large = _score(a_large)
        print(f"  Interpolated to n={n_large}: score={sc_large:.10f}")
        update_best(a_large, sc_large, "interp_32768")

        # Quick L-BFGS on larger n
        if time.monotonic() < deadline - 60:
            a_large_opt = _run_lbfgs(a_large, min(deadline - 60, t_start + 1720), min_time_remaining=60.0)
            sc_large_opt = _score(a_large_opt)
            update_best(a_large_opt, sc_large_opt, "lbfgs_32768")

    # Phase 7: Structured restarts
    print(f"\n[Phase 7] Structured restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for mode in ["cos1", "bathtub", "rand_small"]:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb_structured(best_seq, mode, rng)
            a_r = _run_lbfgs(p_start, min(deadline - 20, t_start + 1750), min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{mode}")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00002
        remaining = deadline - 4 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            try:
                result = minimize(
                    _lsobj,
                    z,
                    jac=_lsgrad,
                    args=(beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra")
            except Exception:
                pass

        # Final active-set
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_refine(best_seq, deadline - 1, max_iter=100)
            update_best(a_fin2, sc_fin2, "final_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
