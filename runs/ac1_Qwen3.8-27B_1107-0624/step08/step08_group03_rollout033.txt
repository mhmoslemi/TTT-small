def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-300:

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Hankel-structure-aware Hessian with precomputed FFT for O(n log n) mat-vec
- Adaptive active-set tolerance with progressive tightening
- Diagonal-preconditioned CG for Newton solves
- Aggressive beta-continuation L-BFGS
- n-sweep exploring 16384 and 32768
- Symmetric sequence exploitation
- Multi-phase strategy: coarse L-BFGS -> fine Newton -> ultra-tight polish
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global state for scipy callbacks
_opt_state = {}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_mv(fw, N_hess, v, n):
    """H @ v where H[i,j] = 2*w[i+j]. Uses precomputed FFT of w."""
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N_hess)
    result = np.fft.irfft(fw * fv, N_hess)
    return 2.0 * result[n - 1:2 * n - 1]


def _grad_from_w(fw, N_hess, a, n):
    """Compute gradient = 2 * (w cross-correlate a) using precomputed FFT of w."""
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N_hess)
    result = np.fft.irfft(fw * fa_rev, N_hess)
    return 2.0 * result[n - 1:2 * n - 1]


def _cg_solve(fw, N_hess, mu, rhs, n, max_cg=400, diag_precond=None):
    """Solve (H + mu*I) x = rhs where H[i,j]=2*w[i+j]. With optional diagonal preconditioning."""
    x = np.zeros(n)
    if diag_precond is not None:
        z0 = rhs / diag_precond
    else:
        z0 = rhs.copy()
    r = rhs - (_hankel_mv(fw, N_hess, x, n) + mu * x)
    if diag_precond is not None:
        z = r / diag_precond
        p = z.copy()
    else:
        z = r.copy()
        p = r.copy()
    rz = np.dot(r, z)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x

    for cg_it in range(max_cg):
        Hp = _hankel_mv(fw, N_hess, p, n) + mu * p
        if diag_precond is not None:
            zHp = Hp / diag_precond
            pHzHp = np.dot(p, zHp)
        else:
            pHzHp = np.dot(p, Hp)
        if pHzHp < 1e-30:
            break
        alpha = rz / pHzHp
        x += alpha * p
        r -= alpha * Hp
        if diag_precond is not None:
            z = r / diag_precond
            rz_new = np.dot(r, z)
        else:
            rz_new = np.dot(r, r)
        if np.sqrt(abs(rz_new)) < 1e-12 * rhs_norm:
            break
        if diag_precond is not None:
            p = z + (rz_new / rz) * p
        else:
            p = r + (rz_new / rz) * p
        rz = rz_new

    return x


def _lsobj(z):
    beta = _opt_state['beta']
    n = _opt_state['n']
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z):
    beta = _opt_state['beta']
    n = _opt_state['n']
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_window":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        t = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.5 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.4 * np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 0.3 + 1.7 * t ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_symmetric(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_sym / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS coarse optimization with beta continuation
    Phase 2: Active-set Newton with adaptive tolerance
    Phase 3: Peak-targeted refinement
    Phase 4: Ultra-tight Newton polish
    Phase 5: Multi-start and n-sweep
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS coarse optimization on parent ===
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_main = _run_lbfgs(parent, deadline - 500, min_time_remaining=300.0, aggressive=True)
        update_best(a_main, _score(a_main), "lbfgs_parent")

    # Try larger n
    print(f"\n[Phase 1b] Larger n exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        n_large = 32768
        a_large_init = _interpolate_to_n(best_seq if best_seq is not None else parent, n_large)
        sc_large_init = _score(a_large_init)
        print(f"  n={n_large} init score: {sc_large_init:.10f}")
        if sc_large_init < best_score + 0.01:
            a_large = _run_lbfgs(a_large_init, deadline - 200, min_time_remaining=100.0, aggressive=True)
            update_best(a_large, _score(a_large), f"lbfgs_n{n_large}")

    # Try structured initializations
    print(f"\n[Phase 1c] Structured initializations (t={time.monotonic() - t_start:.1f}s)")
    for kind in ["cos_window", "hamming", "bimodal"]:
        if time.monotonic() > deadline - 150:
            break
        a_struct = _make_profile(n_parent, kind)
        sc_struct = _score(a_struct)
        if sc_struct < best_score + 0.05:
            a_struct_opt = _run_lbfgs(a_struct, min(deadline - 100, time.monotonic() + 60),
                                      min_time_remaining=20.0, aggressive=True)
            update_best(a_struct_opt, _score(a_struct_opt), f"struct_{kind}")

    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 150, time.monotonic() + 300),
                                         max_iter=400, tol_start=0.005, tol_end=0.0001)
        update_best(a_as, sc_as, "as_newton_1")

    # === Phase 3: Peak-targeted refinement ===
    print(f"\n[Phase 3] Peak-targeted refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_pt, sc_pt = _peak_targeted_refine(best_seq, min(deadline - 80, time.monotonic() + 120),
                                            max_rounds=30)
        update_best(a_pt, sc_pt, "peak_refine")

    # === Phase 4: Ultra-tight Newton polish ===
    print(f"\n[Phase 4] Ultra-tight Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_ult, sc_ult = _active_set_newton(best_seq, min(deadline - 20, time.monotonic() + 200),
                                           max_iter=300, tol_start=0.001, tol_end=0.00002)
        update_best(a_ult, sc_ult, "as_newton_tight")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 10, time.monotonic() + 100),
                                      n_max_steps=2000, lr_init=0.005)
        update_best(a_sg, sc_sg, "subgrad_polish")

    # === Phase 5: Multi-start ===
    print(f"\n[Phase 5] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for sigma in [0.03, 0.01, 0.08]:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score + 0.005:
                a_r, sc_r = _active_set_newton(p_start, min(deadline - 15, time.monotonic() + 80),
                                               max_iter=200, tol_start=0.002, tol_end=0.00005)
                update_best(a_r, sc_r, f"restart_sig{sigma}")

    # Try symmetric version
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sym = _make_symmetric(best_seq)
        sc_sym = _score(a_sym)
        print(f"  Symmetric score: {sc_sym:.10f}")
        if sc_sym < best_score + 0.001:
            a_sym_opt, sc_sym_opt = _active_set_newton(a_sym, min(deadline - 5, time.monotonic() + 60),
                                                       max_iter=200, tol_start=0.003, tol_end=0.00005)
            update_best(a_sym_opt, sc_sym_opt, "symmetric")

    # Final tight Newton on absolute best
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1,
                                           max_iter=200, tol_start=0.0005, tol_end=0.00001)
        update_best(a_fin, sc_fin, "final_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
