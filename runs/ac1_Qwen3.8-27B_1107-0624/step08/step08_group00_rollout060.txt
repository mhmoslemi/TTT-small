    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_fft(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_weighted_conv(a, lam):
    """Compute gradient of sum_k lam[k]*c(a)[k] with respect to a.
    Result: g[i] = 2 * sum_j a[j] * lam[i+j]
    Computed as: g[i]/2 = (lam * a_rev)[i+n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    # convolve lam (length 2n-1) with a_rev (length n)
    # result has length (2n-1) + n - 1 = 3n-2
    total_len = (2*n - 1) + n - 1  # 3n-2
    N = _next_pow2(total_len)
    fl = np.fft.rfft(lam, N)
    fr = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(fl * fr, N)[:total_len]
    # g[i]/2 = conv[i + n - 1] for i = 0, ..., n-1
    g = 2.0 * conv[n-1:2*n-1]
    return g
