def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _cg_solve(hess_func, b, mu, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha <= 0 or not np.isfinite(alpha):
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

# LSE optimization with fine temperature schedule
# ============================================================

def _lbfgs_lse_fine(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Fine geometric schedule
    ratios = [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03,
              0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for ratio in ratios:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time_per_iter = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128, 256, 512]

        remaining = deadline - min_remaining - time.monotonic()

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new

# Subgradient polish
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=8000, lr_init=0.006, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = active.astype(np.float64)
        w = w / w.sum()

        w_rev = w[::-1]
        L = _next_pow2(n + len(w) - 1)
        A = np.fft.rfft(a, L)
        W = np.fft.rfft(w_rev, L)
        conv_result = np.fft.irfft(A * W, L)[:n + len(w) - 1]
        grad_a = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
        W_val = float(np.dot(w, c))
        grad_a = grad_a - W_val
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.94
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
                    tol_frac *= 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# Perturbation methods
# ============================================================

def _perturb_fourier_phase(a, rng, max_amp=0.4):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.08):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(8, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - block_size)))
        j = int(rng.integers(0, max(1, n - block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_reverse_mix(a, rng, alpha=0.1):
    a_rev = a[::-1]
    a_new = (1 - alpha) * a + alpha * a_rev
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return _perturb_log(a_new, 0.01, rng)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining top-K subgradient, LSE with fine
    temperature schedule, Newton active-set refinement, power-sum smoothing,
    and multi-n exploration. Targets score < 1.5027.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick LSE from parent (180s) ===
    print(f"\n[Phase 1] Quick LSE optimization")
    p1_dead = min(deadline, t_start + 180)
    if time.monotonic() < p1_dead:
        a1, sc1 = _lbfgs_lse_fine(_best_seq.copy(), p1_dead, min_remaining=20)
        upd(a1, sc1, "lse_quick")

    # === Phase 2: Top-K subgradient (300s) ===
    print(f"\n[Phase 2] Top-K subgradient")
    p2_dead = min(deadline, time.monotonic() + 300)
    if time.monotonic() < p2_dead:
        a2, sc2 = _topk_subgrad_optimize(_best_seq.copy(), p2_dead, K_max=300, min_K=1)
        upd(a2, sc2, "topk_sg")

    # === Phase 3: Fine LSE (300s) ===
    print(f"\n[Phase 3] Fine LSE")
    p3_dead = min(deadline, time.monotonic() + 300)
    if time.monotonic() < p3_dead:
        a3, sc3 = _lbfgs_lse_fine(_best_seq.copy(), p3_dead, min_remaining=20)
        upd(a3, sc3, "lse_fine")

    # === Phase 4: Newton active-set (250s) ===
    print(f"\n[Phase 4] Newton active-set")
    p4_dead = min(deadline, time.monotonic() + 250)
    if time.monotonic() < p4_dead:
        a4, sc4 = _newton_active_set(_best_seq, p4_dead, max_iter=500, tol_frac=0.0005)
        upd(a4, sc4, "newton")

    # === Phase 5: Subgradient polish (200s) ===
    print(f"\n[Phase 5] Subgradient polish")
    p5_dead = min(deadline, time.monotonic() + 200)
    if time.monotonic() < p5_dead:
        a5, sc5 = _subgrad_polish(_best_seq, p5_dead, n_steps=10000, lr_init=0.005, tol_frac=0.001)
        upd(a5, sc5, "sg_polish")

    # === Phase 6: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 6] Power-sum L-BFGS")
    p6_dead = min(deadline, time.monotonic() + 150)
    if time.monotonic() < p6_dead:
        a6, sc6 = _lbfgs_power_sum(_best_seq.copy(), p6_dead, min_remaining=15)
        upd(a6, sc6, "power_sum")

    # === Phase 7: Newton + SG alternating (150s) ===
    print(f"\n[Phase 7] Newton + SG alternating")
    p7_dead = min(deadline, time.monotonic() + 150)
    if time.monotonic() < p7_dead:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > p7_dead - 30:
                break
            nw_dead = min(p7_dead - 25, time.monotonic() + 30)
            a_nw, sc_nw = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.001)
            if sc_nw < _score(a_cur) - 1e-12:
                a_cur = a_nw
            upd(a_cur, _score(a_cur), f"alt_nw_{round_i}")

            if time.monotonic() > p7_dead - 15:
                break
            sg_dead = min(p7_dead - 10, time.monotonic() + 20)
            a_sg, sc_sg = _subgrad_polish(a_cur, sg_dead, n_steps=3000, lr_init=0.004, tol_frac=0.001)
            if sc_sg < _score(a_cur) - 1e-12:
                a_cur = a_sg
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 8: Multi-n exploration (250s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    p8_dead = min(deadline, time.monotonic() + 250)
    n_candidates = [8192, 12288, 16384, 24576, 32768]
    if n_parent not in n_candidates and n_parent >= 4096:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > p8_dead - 30:
            break
        if len(_best_seq) == n_try:
            a_init_n = _best_seq.copy()
        else:
            a_init_n = _resize_sequence(_best_seq, n_try)

        # LSE at this n
        a_n, sc_n = _lbfgs_lse_fine(a_init_n, p8_dead, min_remaining=15)
        upd(a_n, sc_n, f"n{n_try}_lse")

        # Newton polish
        if _score(a_n) < _best_score * 1.002 and time.monotonic() < p8_dead - 15:
            nw_dead_n = min(p8_dead - 10, time.monotonic() + 40)
            a_n2, sc_n2 = _newton_active_set(a_n, nw_dead_n, max_iter=60, tol_frac=0.002)
            upd(a_n2, sc_n2, f"n{n_try}_newton")

    # === Phase 9: Diverse restarts (200s) ===
    print(f"\n[Phase 9] Diverse restarts")
    p9_dead = min(deadline, time.monotonic() + 200)
    perturbations = [
        ("phase", lambda: _perturb_fourier_phase(_best_seq, rng, max_amp=0.3)),
        ("amp", lambda: _perturb_amplitude(_best_seq, rng, sigma=0.06)),
        ("log", lambda: _perturb_log(_best_seq, 0.03, rng)),
        ("block", lambda: _perturb_block_swap(_best_seq, rng, n_swaps=6)),
        ("rev_mix", lambda: _perturb_reverse_mix(_best_seq, rng, alpha=0.05)),
        ("log_big", lambda: _perturb_log(_best_seq, 0.06, rng)),
    ]

    for i, (name, fn) in enumerate(perturbations):
        if time.monotonic() > p9_dead - 20:
            break
        try:
            p_start = fn()
        except Exception:
            continue

        a_r, sc_r = _lbfgs_lse_fine(p_start, p9_dead, min_remaining=10)
        upd(a_r, sc_r, f"restart_{name}")

        if sc_r < _best_score * 1.002 and time.monotonic() < p9_dead - 10:
            nw_dead_r = min(p9_dead - 5, time.monotonic() + 20)
            a_r2, sc_r2 = _newton_active_set(a_r, nw_dead_r, max_iter=40, tol_frac=0.003)
            upd(a_r2, sc_r2, f"restart_{name}_nw")

    # === Phase 10: Final ultra-polish (180s) ===
    print(f"\n[Phase 10] Final ultra-polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-low beta LSE
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.003)))
            try:
                result = minimize(
                    _lse_obj, z_fin, args=(beta_final, n_fin),
                    jac=_lse_grad, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 3,
                                                  max_iter=100, tol_frac=0.0003)
            upd(a_fin2, sc_fin2, "final_newton_tight")

        # Final subgradient with very small tol_frac
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 2,
                                              n_steps=5000, lr_init=0.003, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg_tight")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
