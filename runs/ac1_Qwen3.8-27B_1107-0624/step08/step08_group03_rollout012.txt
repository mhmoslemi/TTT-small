The idea is:
1. Start with weights w_j on each convolution lag
2. Solve: min sum_j w_j * c_j s.t. sum(a) = 1, a >= 0

4. Repeat

Step 2 is a convex QP because:
- sum_j w_j * c_j = sum_j w_j * sum_i a_i * a_{j-i} = a^T (sum_j w_j * T_j) a
  where T_j is the matrix such that c_j = a^T T_j a
- This is a quadratic form, and if sum_j w_j * T_j is positive semi-definite, it's convex.

Actually, T_j is the "convolution matrix" for lag j, which is symmetric. The sum of PSD matrices is PSD, so sum_j w_j * T_j is PSD for w_j >= 0.

So the subproblem is:
min a^T Q a s.t. sum(a) = 1, a >= 0
where Q = sum_j w_j * T_j is PSD.

This is a convex QP! We can solve it with standard methods (e.g., projected gradient, active set, or even just Newton since the Hessian is Q + lambda*I for the equality constraint).

The solution to this QP gives the "weighted" optimal a. By iteratively updating the weights to emphasize the peak lags, we converge to a solution that minimizes the max.

This is essentially a **Dykstra-like** or **iterative reweighting** approach.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Spectral pre-initialization via FFT autocorrelation flattening
2. L-BFGS with log-sum-exp smoothing (aggressive multi-restart)
3. Multi-scale active-set Newton with direct solution
4. Adaptive peak weighting
5. Multi-n exploration (8192, 16384, 32768)
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _convolve_with_ones(a, ones_vec, n):
    """Compute (a * ones_vec)[n-1:2n-1] using FFT."""
    a_flip = a[::-1]
    out_len = n + len(ones_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_flip, N)
    fo = np.fft.rfft(ones_vec, N)
    result = np.fft.irfft(fa * fo, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _spectral_init(a_init, deadline, n_iters=20, alpha=0.5):
    """FFT-based autocorrelation flattening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        P = np.abs(A) ** 2
        r = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = float(r.max())
        
        # Water-filling: reduce peaks
        target = c_max * (1.0 - alpha * 0.1)
        r_mod = r.copy()
        above = r_mod > target
        if above.any():
            # Redistribute: reduce peaks, increase troughs proportionally
            excess = (r_mod[above] - target).sum()
            below = ~above
            if below.any():
                r_mod[above] = target + (r_mod[above][None] - target) * 0  # set to target
                # Actually just clip
                r_mod[r_mod > target] = target
                # Don't change below-threshold values (to preserve the DC component)
        
        # Go back to frequency domain
        r_pad = np.zeros(N)
        r_pad[:2 * n - 1] = r_mod
        P_new = np.fft.rfft(r_pad)
        # P_new should be >= 0 for a valid power spectrum, but might have small negatives
        P_new = np.maximum(P_new, 0)
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

        accepted = False
        t = 1.0
        neg_mask = dx < -1e-12
        if neg_mask.any():
            t = min(t, 0.99 * np.min(a[neg_mask] / (-dx[neg_mask])))
        
        for ls in range(50):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5
        
        if not accepted:
            # Fallback: steepest descent
            t = 0.1
            for ls in range(20):
                a_try = np.maximum(a - t * g_t, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    f_try = float(c_try.max())
                    if f_try < f_old:
                        a = a_try
                        break
                t *= 0.5
            else:
                pass
        
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        
        it += 1
    
    # Final iterations with the smallest tolerance
    if time.monotonic() < deadline - 5:
        tol_final = 0.00001
        for it2 in range(100):
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            active = c >= (c_max - tol_final * c_max)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            A_vec = active.astype(np.float64)
            m = int(active.sum())
            
            g = (2.0 / m) * _convolve_with_ones(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-13:
                break
            
            mu = max(1e-8, 1e-6 * c_max / max(m, 1))
            
            def hess_matvec2(v):
                return (2.0 / m) * _convolve_with_ones(v, A_vec, n) + mu * v
            
            def cg_solve2(rhs, max_cg=200):
                x = np.zeros_like(rhs)
                r = rhs - hess_matvec2(x)
                p = r.copy()
                rs = float(np.dot(r, r))
                for cg_it in range(max_cg):
                    Ap = hess_matvec2(p)
                    pAp = float(np.dot(p, Ap))
                    if pAp < 1e-30:
                        break
                    alpha = rs / pAp
                    x = x + alpha * p
