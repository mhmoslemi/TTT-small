2*n*max(conv(a,a))/sum(a)^2. Uses FFT-based convolution, softmax
parameterization, multi-scale beta cooling, and random restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


class _LSProblem:
    """Log-sum-exp smoothed minimization problem."""
    
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    
    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))
    
    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W)

    base_beta = max(gap, c_max0 * 0.01)
    
    # Geometric cooling schedule
    ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
              0.015625, 0.0078, 0.004, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
              0.00001, 0.000005, 0.000002, 0.000001]
    
    # Estimate time per iteration
    t0 = time.monotonic()
    _full_convolve(a, a)  # warm up
    conv_time = (time.monotonic() - t0) / max(1, n / 16384)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for i, ratio in enumerate(ratios):
        beta = base_beta * ratio
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        
        # Estimate max iterations based on remaining time
        # Each iteration: 1-3 convolutions + overhead
        time_per_iter = max(0.001, 0.003 * (n / 16384.0))
        max_iter = min(5000, max(50, int(remaining / (time_per_iter * 15))))
        
        prob = _LSProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:

                best_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 4 == 0:
                    print(f"    beta={beta:.2e}: score={best_sc:.10f}, iter={result.nit}")
        except Exception as e:
            print(f"    beta={beta:.2e}: L-BFGS failed: {e}")
    
    return best_a


def _gradient_descent_polish(a, deadline, lr=0.01, min_time=5.0):
    """Simple gradient descent on the max (subgradient) for final polish."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_a = a.copy()
            best_sc = current_sc
        
        # Subgradient: average of gradients of all peaks within 1% of max
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            break
        
        # Compute h = (w * a_rev)[n-1:2n-1] where w = active/n_active
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)  # gradient of the smoothed max wrt a
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        alpha = lr / gnorm
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.8
            if lr < 1e-10:
                break
    
    return best_a


def _perturb_and_optimize(a_best, deadline, min_time=30.0, sigma_values=None):
    """Try random perturbations of the best solution and re-optimize."""
    if sigma_values is None:
        sigma_values = [0.1, 0.05, 0.02, 0.01, 0.005]
    
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)
    rng = np.random.default_rng(12345)
    
    for sigma in sigma_values:
        if time.monotonic() + min_time > deadline:
            break
        
        # Generate a perturbation
        z = np.log(np.maximum(a_best, 1e-30))
        z = z - z.mean()
        z_pert = z + sigma * rng.standard_normal(n)
        a_pert = _softmax(z_pert)
        
        # Optimize from perturbed point
        a_opt = _optimize_lbfgs(a_pert, deadline, min_time=min_time)
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc - 1e-12:
            best_a = a_opt.copy()
            best_sc = sc_opt
            print(f"  Perturb sigma={sigma}: NEW BEST {best_sc:.10f}")
    
    return best_a


def _try_different_n(a_best, n_target, deadline, min_time=30.0):
    """Resample to a different n and re-optimize."""
    old_n = len(a_best)
    if old_n == n_target:
        return a_best
    
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_best)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    
    a_opt = _optimize_lbfgs(a_new, deadline, min_time=min_time)
    return a_opt
