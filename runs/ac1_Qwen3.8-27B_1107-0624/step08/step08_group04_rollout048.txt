def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = a.sum()
    if s < 1e-15:

    c = full_convolve(a, a)
