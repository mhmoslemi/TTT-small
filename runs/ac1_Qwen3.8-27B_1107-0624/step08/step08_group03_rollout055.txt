"""
Optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Multi-scale n exploration (up to 65536)
2. p-norm smoothed L-BFGS optimization in log-space
3. Active-set Newton with CG linear solves
4. Coordinate descent polish
5. Diverse structured initializations
6. Variance-based frequency domain initialization
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _pnorm_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)
    cp = np.sum(c ** p)
    if cp < 1e-300:
        return 2.0 * n
    return float(2.0 * n * cp ** (1.0 / p))


def _pnorm_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)
    cp = np.sum(c ** p)
    if cp < 1e-300:
        return np.zeros_like(z)
    w = c ** (p - 1.0)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_p = float(np.dot(w, c))
    return 4.0 * n * (cp ** (1.0 / p - 1.0)) * a * (h - W_p)


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:

        c_max = float(c.max())
        tol = tol_active * c_max

        if n_active == 0:

            n_active = 1
        if n_active > 3 * n:

            top_idx = np.argsort(c)[::-1][:3 * n]
            active[top_idx] = True

        m = n_active

                p = r_new + (rs_new / rs) * p

            return x

            t *= 0.5

            else:

        sc = _score(a)
        if sc < best_sc - 1e-14:

                print(f"    p={p}: score={sc_new:.10f}")
        except Exception as e:
            print(f"    p={p}: error {e}")

    return current_a


def _run_lbfgs_lse(a_init, deadline, min_time_remaining=20.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[30]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                  0.015, 0.0075, 0.0037, 0.00185, 0.0009,
                                  0.00045, 0.00022, 0.00011, 0.00005,
                                  0.000025, 0.0000125, 0.000006])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 8192:
            est_time_per_iter = 0.003
        elif n <= 16384:
            est_time_per_iter = 0.008
        elif n <= 32768:
            est_time_per_iter = 0.02
        else:
            est_time_per_iter = 0.05

        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda zz: _lsobj(zz, beta, n),
                z,
                jac=lambda zz: _lsgrad(zz, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-10,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, n_sweeps=20):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    block_size = max(1, n // 50)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - 0.01 * c_max)
        A_vec = active.astype(np.float64)

        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            idx = slice(start, end)
            a_block = a[idx].copy()
            a_fix = a.copy()

            for inner in range(30):
                if time.monotonic() > deadline:
                    break
                c = _full_convolve(a_fix, a_fix)
                c_max = float(c.max())
                active = c >= (c_max - 0.01 * c_max)
                A_vec = active.astype(np.float64)
                m = max(1, int(np.sum(active)))

                g_block = (2.0 / m) * _corr_a_A(a_fix, A_vec, n)[idx]
                g_full = (2.0 / m) * _corr_a_A(a_fix, A_vec, n)
                g_t_full = g_full - np.mean(g_full)

                H_ii = (2.0 / m) * len(A_vec) * 2 + 1e-8
                step = g_t_full[idx] / H_ii
                step = np.maximum(step, -a_block / H_ii + 1e-15)

                a_try = a_fix.copy()
                a_try[idx] = a_block - step
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                sc_try = _score(a_try)

                if sc_try < _score(a_fix) - 1e-14:
                    a_fix = a_try
                    a_block = a_try[idx].copy()
                else:
                    step *= 0.5
                    a_try = a_fix.copy()
                    a_try[idx] = a_block - step
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        break
                    a_try = a_try / s_try
                    if _score(a_try) < _score(a_fix) - 1e-14:
                        a_fix = a_try
                        a_block = a_try[idx].copy()
                    else:
                        break

            a = a_fix

        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t ** 2 / (2.0 * 0.5 ** 2))
        return a / a.sum()
    elif kind == "hamming":
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif kind == "blackman":
        a = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) + \
            0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1))
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        a = rng.standard_normal(n)
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "random_smooth2":
        a = rng.standard_normal(n)
        k = max(2, n // 20)
        kernel = np.exp(-np.arange(k) ** 2 / (2.0 * (k // 3) ** 2))
        kernel = kernel / kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "plateau":
        a = np.ones(n)
        a[n // 4:3 * n // 4] *= 1.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _freq_init(n, rng):
    """Initialize by trying to make |FFT(a)| as flat as possible."""
    a = rng.standard_normal(n)
    a = np.maximum(a, 0.1)
    a = a / a.sum()
    for _ in range(50):
        A = np.fft.fft(a, 2 * n - 1)
        mag = np.abs(A) ** 2
        target = np.mean(mag)
        # Want to reduce peaks in |A|^2
        # Gradient of sum |A|^4 w.r.t. a
        grad = np.real(np.fft.ifft(mag * A, 2 * n - 1))[:n]
        a = a - 0.01 * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining p-norm smoothing, L-BFGS,
    active-set Newton, and coordinate descent across multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        ps = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={ps:.10f}")
        update_best(parent, ps, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # Define n values to try
    n_values = [8192, 16384, 32768, 65536]
    # Time allocation (leave 100s for final polish)
    time_frac = [0.10, 0.20, 0.30, 0.30]
    total_frac = sum(time_frac)
    time_frac = [f / total_frac for f in time_frac]

    for ni, n_try in enumerate(n_values):
        t_phase_start = time.monotonic()
        time_budget = (deadline - t_phase_start) * time_frac[ni]
        phase_deadline = t_phase_start + time_budget

        print(f"\n{'='*60}")
        print(f"Phase n={n_try}: budget={time_budget:.0f}s, remaining_until={deadline - t_phase_start:.0f}s")
        print(f"{'='*60}")

        if time.monotonic() > deadline - 50:
            break

        # Create initializations
        inits = []

        # Parent upsampled
        if parent is not None:
            a_p = _interpolate_to_n(parent, n_try)
            inits.append(("parent", a_p))

        # Structured profiles
        for kind in ["uniform", "gaussian", "hamming", "sine", "ramp"]:
            a_k = _make_profile(n_try, kind, rng)
            inits.append((kind, a_k))

        # Random smooth
        for r in range(2):
            a_r = _make_profile(n_try, "random_smooth", rng)
            inits.append((f"rand_{r}", a_r))

        # Frequency-domain init
        if n_try <= 32768:
            a_f = _freq_init(n_try, rng)
            inits.append(("freq", a_f))

        # Evaluate all initializations
        init_scores = []
        for label, a_init in inits:
            sc = _score(a_init)
            init_scores.append((sc, label, a_init))
            print(f"  init {label}: score={sc:.10f}")

        init_scores.sort(key=lambda x: x[0])

        # Optimize top initializations
        n_opt = min(3, len(init_scores))
        for oi in range(n_opt):
            if time.monotonic() > phase_deadline - 10:
                break
            sc_i, label_i, a_i = init_scores[oi]
            remaining_in_phase = phase_deadline - time.monotonic()
            if remaining_in_phase < 10:
                break

            # Allocate time among optimizers
            t_pnorm = remaining_in_phase * 0.4
            t_lse = remaining_in_phase * 0.3
            t_asn = remaining_in_phase * 0.2
            t_cd = remaining_in_phase * 0.1

            # p-norm L-BFGS
            if t_pnorm > 15:
                print(f"  Optimizing {label_i} with p-norm (t={t_pnorm:.0f}s)")
                a_opt1 = _run_lbfgs_pnorm(a_i, time.monotonic() + t_pnorm, min_time_remaining=5.0)
                sc_opt1 = _score(a_opt1)
                update_best(a_opt1, sc_opt1, f"{label_i}_pnorm")

            # LSE L-BFGS from p-norm result
            if time.monotonic() < phase_deadline - 10:
                print(f"  Optimizing {label_i} with LSE (t={t_lse:.0f}s)")
                a_opt2 = _run_lbfgs_lse(a_opt1, time.monotonic() + t_lse, min_time_remaining=5.0)
                sc_opt2 = _score(a_opt2)
                update_best(a_opt2, sc_opt2, f"{label_i}_lse")

            # Active-set Newton
            if time.monotonic() < phase_deadline - 10:
                a_opt3, sc_opt3 = _active_set_newton(a_opt2, time.monotonic() + t_asn,
                                                     max_iter=100, tol_active=0.002)
                update_best(a_opt3, sc_opt3, f"{label_i}_asn")

            # Coordinate descent
            if time.monotonic() < phase_deadline - 5:
                a_opt4, sc_opt4 = _coordinate_descent(a_opt3, time.monotonic() + t_cd, n_sweeps=10)
                update_best(a_opt4, sc_opt4, f"{label_i}_cd")

    # Final polish phases on the global best
    print(f"\n{'='*60}")
    print(f"Final Polish Phase")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 10:
        # Tighter active-set Newton
        a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 5, time.monotonic() + 60),
                                           max_iter=150, tol_active=0.0005)
        update_best(a_fin, sc_fin, "final_asn_tight")

        # Very tight active-set
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _active_set_newton(a_fin, min(deadline - 3, time.monotonic() + 30),
                                                 max_iter=100, tol_active=0.0001)
            update_best(a_fin2, sc_fin2, "final_asn_vtight")

    # Perturbation restarts on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        for sigma in [0.01, 0.03, 0.05]:
            if time.monotonic() > deadline - 8:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score + 0.01:
                a_r = _run_lbfgs_pnorm(p_start, min(deadline - 5, time.monotonic() + 30),
                                       min_time_remaining=3.0)
                update_best(a_r, _score(a_r), f"perturb_{sigma}")

                if time.monotonic() < deadline - 5:
                    a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 3, time.monotonic() + 15),
                                                     max_iter=50, tol_active=0.001)
                    update_best(a_r2, sc_r2, f"perturb_{sigma}_asn")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
