warm start, and targeted spectral perturbations to beat 1.5027.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _active_set_gradient(a, c, top_k=5, tol_frac=0.01):
    """Compute gradient using active set of lags near the max."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-15:
        return np.zeros(n)
    # Find active lags: those within tol_frac of max
    threshold = c_max * (1.0 - tol_frac)
    active = c >= threshold
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 1
    if n_active > top_k:
        top_indices = np.argsort(c)[::-1][:top_k]
        active = np.zeros_like(c, dtype=bool)
        active[top_indices] = True
        n_active = top_k
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m] for valid indices
    # This is a cross-correlation
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # wc has length len(w) + len(a_rev) - 1 = (2n-1) + n - 1 = 3n-2
    # The valid range for the cross-correlation that gives g[m] for m in [0,n-1]:
    # g[m] = sum_k w[k] * a[k-m] where 0<=k-m<n means k in [m, m+n-1]
    # In the convolution w * a_rev: (w * a_rev)[p] = sum_k w[k] * a_rev[p-k]
    # a_rev[j] = a[n-1-j], so a_rev[p-k] = a[n-1-(p-k)] = a[n-1-p+k]
    # We want sum_k w[k]*a[k-m], so we need n-1-p+k = k-m, i.e., n-1-p = -m, p = n-1+m
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad


def _soft_weight(c, beta):
    """Compute soft weights for LSE approximation."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _soft_gradient(a, c, beta):
    """Gradient using soft (LSE) active set."""
    n = len(a)
    w = _soft_weight(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W) / (a.sum() ** 2)


def _lse_obj_and_grad(a, beta):
    """Objective and gradient for LSE formulation (working directly on a)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _adam_optimize(a, deadline, lr_init=0.05, top_k=5, tol_frac=0.005,
                   beta1=0.8, beta2=0.99, epsilon=1e-8, min_time=2.0):
    """Adam optimizer with exact subgradient on the simplex."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    lr = lr_init
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() < deadline - min_time:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max  # since sum(a)=1
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= max_no_improve:
            # Restart with perturbation
            a = _normalize(best_a)
            c = _full_convolve(a, a)
            k_star = int(np.argmax(c))
            # Perturb at frequency related to k_star
            freq = 2.0 * np.pi * k_star / n
            pert = 0.01 * np.cos(freq * np.arange(n))
            a = _normalize(a + pert * a)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            max_no_improve = max(20, max_no_improve // 2)
            continue
        
        # Compute gradient
        grad = _active_set_gradient(a, c, top_k=top_k, tol_frac=tol_frac)
        # Center gradient (to preserve sum)
        grad = grad - np.mean(grad)
        
        # Adam update
        lr_t = lr / (1.0 + 0.01 * t * 0.001)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check if this is an improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < sc + 1e-14:
            a = a_new
        else:
            # Line search: try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = a - frac * lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < sc - 1e-15:
                    a = a_try
                    break
        
        # Decay learning rate
        if t % 200 == 0:
            lr *= 0.9
        if lr < 1e-6:
            lr = 1e-6
            # Also tighten active set
            top_k = max(1, top_k - 1)
            tol_frac = max(0.001, tol_frac * 0.5)
    
    return best_a, best_sc


def _lbfgs_lse(a_init, deadline, beta_start=None, min_remaining=10.0):
    """L-BFGS on log-softmax parametrization with LSE objective."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a, _score(a)
    
    if beta_start is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        beta_start = max(gap, max_c * 0.003)
    
    current_a = a.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Beta schedule
    n_betas = 12
    betas = beta_start * (0.55 ** np.arange(n_betas))
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3.0:
            break
        
        scale = max(1.0, n / 4096.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 20))))
        
        def obj(z_in):
            a_in = _softmax(z_in)
            return _lse_obj_and_grad(a_in, beta)[0]
        
        def grad(z_in):
            a_in = _softmax(z_in)
            return _lse_obj_and_grad(a_in, beta)[1]
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _newton_polish(a, deadline, top_k=3, max_iter=500, min_remaining=2.0):
    """Newton-like polishing using the structured Hessian."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    no_improve = 0
    
    for _ in range(max_iter):
        if time.monotonic() > deadline - min_remaining:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
        
        # Get active set
        threshold = c_max * (1.0 - 0.001)
        active = np.where(c >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(c)[::-1][:top_k]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Gradient from active set
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: use 1/sqrt(gnorm) as a rough adaptive step
        step = lr / (gnorm + 1e-10)
        
        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        
        if sc_try < sc - 1e-15:
            a = a_try
            lr = min(0.05, lr * 1.1)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _spectral_perturb(a, k_star, amplitude=0.02):
    """Add a perturbation in frequency domain targeting a specific peak lag."""
    n = len(a)
    a_work = a.copy()
    # DFT
    A = np.fft.rfft(a_work)
    # Add a small real perturbation at frequency related to k_star
    # The peak at lag k_star is related to low-frequency components
    freq_idx = min(k_star // 2, len(A) - 1)
    if freq_idx > 0:
        A[1] += amplitude * np.exp(1j * np.pi * k_star / n)
    a_new = np.fft.irfft(A, n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _intensive_adam(a, deadline, phases=None):
    """Multi-phase Adam with progressively tighter active sets."""
    if phases is None:
        phases = [
            {'lr': 0.08, 'top_k': 10, 'tol_frac': 0.02, 'time_frac': 0.3},
            {'lr': 0.04, 'top_k': 5, 'tol_frac': 0.01, 'time_frac': 0.25},
            {'lr': 0.02, 'top_k': 3, 'tol_frac': 0.005, 'time_frac': 0.2},
            {'lr': 0.01, 'top_k': 2, 'tol_frac': 0.002, 'time_frac': 0.15},
            {'lr': 0.005, 'top_k': 1, 'tol_frac': 0.001, 'time_frac': 0.1},
        ]
    
    total_time = deadline - time.monotonic()
    a_cur = a.copy()
    for phase in phases:
        phase_end = time.monotonic() + phase['time_frac'] * total_time
        if phase_end > deadline - 1.0:
            phase_end = deadline - 1.0
        if time.monotonic() >= phase_end:
            continue
        print(f"    Adam phase: lr={phase['lr']}, top_k={phase['top_k']}, tol={phase['tol_frac']}")
        a_next, sc = _adam_optimize(a_cur, phase_end, lr_init=phase['lr'],
                                     top_k=phase['top_k'], tol_frac=phase['tol_frac'])
        sc_cur = _score(a_cur)
        if sc < sc_cur:
            a_cur = a_next
            print(f"    Phase done: score={sc:.10f}")
        else:
            print(f"    Phase done: no improvement (best={sc_cur:.10f})")
    return a_cur


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization combining Adam with exact
    subgradients, L-BFGS LSE polishing, spectral perturbations, and
    multi-resolution exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    n_best = len(best_a)
    
    # === Phase 1: Intensive multi-phase Adam (parent n) ===
    print(f"\n[P1] Intensive Adam on n={n_best} (t={time.monotonic()-t_start:.1f}s)")
    p1_end = min(deadline - 700.0, t_start + 500.0)
    a1, sc1 = _intensive_adam(best_a, p1_end)
    if sc1 < best_score:
        best_a = a1
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [adam_p1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement from Adam: {best_score:.10f}")
    
    # === Phase 2: L-BFGS LSE polish ===
    print(f"\n[P2] L-BFGS LSE polish (t={time.monotonic()-t_start:.1f}s)")
    p2_end = min(deadline - 500.0, t_start + 750.0)
    a2, sc2 = _lbfgs_lse(best_a, p2_end)
    if sc2 < best_score:
        best_a = a2
        best_score = sc2
        n_best = len(best_a)
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_p2] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement from L-BFGS: {best_score:.10f}")
    
    # === Phase 3: Newton polish ===
    print(f"\n[P3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    p3_end = min(deadline - 400.0, t_start + 900.0)
    a3, sc3 = _newton_polish(best_a, p3_end, top_k=3)
    if sc3 < best_score:
        best_a = a3
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [newton_p3] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Spectral perturbations + re-optimization ===
    print(f"\n[P4] Spectral perturbations (t={time.monotonic()-t_start:.1f}s)")
    c_best = _full_convolve(best_a, best_a)
    k_star = int(np.argmax(c_best))
    
    p4_end = min(deadline - 250.0, t_start + 1100.0)
    n_attempts = 5
    for attempt in range(n_attempts):
        if time.monotonic() > p4_end - 30.0:
            break
        # Create perturbation
        a_pert = _spectral_perturb(best_a, k_star, amplitude=0.01 * (attempt + 1))
        # Also try random perturbation
        if attempt % 2 == 1:
            z = np.log(np.maximum(a_pert, 1e-30))
            z += 0.03 * rng.standard_normal(len(z))
            z -= z.mean()
            a_pert = _softmax(z)
        
        # Quick Adam
        sub_end = min(p4_end, time.monotonic() + (p4_end - time.monotonic()) / (n_attempts - attempt))
        a_opt, sc_opt = _adam_optimize(a_pert, sub_end, lr_init=0.03, top_k=5, tol_frac=0.01)
        if sc_opt < best_score:
            best_a = a_opt
            best_score = sc_opt
            k_star = int(np.argmax(_full_convolve(best_a, best_a)))
            print(f"  NEW BEST: {best_score:.10f} [perturb_{attempt}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [int(n_best * 1.25), int(n_best * 0.8), n_best + 1024]
    n_targets = [x for x in n_targets if x != n_best and 2000 <= x <= 30000]
    if not n_targets:
        n_targets = [n_best + 2048]
    
    p5_end = min(deadline - 200.0, t_start + 1400.0)
    for n_try in sorted(set(n_targets))[:3]:
        if time.monotonic() > p5_end - 60.0:
            break
        a_init = _resample(best_a, n_try)
        print(f"  Trying n={n_try}")
        sub_end = min(p5_end, time.monotonic() + 120.0)
        a_opt, sc_opt = _adam_optimize(a_init, sub_end, lr_init=0.05, top_k=8, tol_frac=0.02)
        if sc_opt < best_score:
            best_a = a_opt
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Final intensive polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    p6_end = deadline - 5.0
    
    # Tight Adam
    p6a_end = min(p6_end - 60.0, time.monotonic() + 200.0)
    a6a, sc6a = _adam_optimize(best_a, p6a_end, lr_init=0.005, top_k=1, tol_frac=0.001)
    if sc6a < best_score:
        best_a = a6a
        best_score = sc6a
        print(f"  NEW BEST: {best_score:.10f} [final_adam] t={time.monotonic()-t_start:.1f}s")
    
    # L-BFGS with tiny beta
    if time.monotonic() < p6_end - 30.0:
        c_fin = _full_convolve(best_a, best_a)
        mb = float(c_fin.max())
        a6b, sc6b = _lbfgs_lse(best_a, p6_end - 5.0, beta_start=mb * 0.001)
        if sc6b < best_score:
            best_a = a6b
            best_score = sc6b
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # Newton micro-polish
    if time.monotonic() < p6_end - 5.0:
        a6c, sc6c = _newton_polish(best_a, p6_end - 1.0, top_k=1, max_iter=200)
        if sc6c < best_score:
            best_a = a6c
            best_score = sc6c
            print(f"  NEW BEST: {best_score:.10f} [final_newton] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
