"""
Power-sum objective optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: uses sum(c^p) as a smooth surrogate for max(c), with FFT-based
gradient computation. Adaptive p schedule transitions from energy balancing to
peak reduction. Combined with active-set sub-gradient polish and multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


class _PowerProblem:
    """L-BFGS problem for minimizing sum(conv(a,a)^p) with softmax parameterization."""
    
    def __init__(self, p, n):
        self.p = p
        self.n = n
    
    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        return float(np.sum(c ** self.p))
    
    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        w = c ** (self.p - 1)
        # G[j] = 2p * sum_k w[k] * a[k-j]
        # = 2p * conv(w_rev, a)[n-1-j] for j=0,...,n-1
        w_rev = w[::-1]
        conv_wa = _full_convolve(w_rev, a)
        # conv_wa has length (2n-1) + n - 1 = 3n-2
        # We need indices (2n-2-j) for j=0,...,n-1, i.e., indices 2n-2, 2n-3, ..., n-1
        # That's conv_wa[n-1 : 2*n-1] reversed
        G = 2.0 * self.p * conv_wa[n-1 : 2*n-1][::-1]
        # Softmax reparameterization
        G_mean = np.dot(a, G)
        return a * (G - G_mean)


def _run_power_lbfgs(a_init, deadline, min_time_remaining=100.0):
    """Run L-BFGS with increasing power-sum objective."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Power schedule
    powers = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192]
    
    for p in powers:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        prob = _PowerProblem(p, n)
        
        # Estimate max iterations based on remaining time
        est_per_iter = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_per_iter * 15))))
        
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                a = a_new
            elif sc_new < best_sc + 0.001:
                # Accept if not much worse (for exploration)
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                a = a_new
            print(f"    p={p}: score={sc_new:.10f}, best={best_sc:.10f}, iters={result.niter}")
        except Exception as e:
            print(f"    p={p}: L-BFGS error: {e}")
            continue
    
    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_frac_start=0.005, tol_frac_end=0.00001, lr_init=0.05):
    """Active-set sub-gradient method for direct peak reduction."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 200
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        
        # Compute gradient: G[j] = 2 * sum_k w[k] * a[k-j]
        w_rev = w[::-1]
        conv_wa = _full_convolve(w_rev, a)
        G = 2.0 * conv_wa[n-1 : 2*n-1][::-1]
        
        G_proj = G - np.dot(a, G)
        gnorm = np.linalg.norm(G_proj)
        
        if gnorm < 1e-15:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(80):
            a_try = a - alpha * G_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.2
                no_improve = 0
                if tol_frac <= tol_frac_end * 1.1 and lr <= lr_init * 0.05:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.95)
            lr = min(lr_init, lr * 1.03)
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod2":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 13.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _multi_peak_polish(a_init, deadline, n_peaks=5, lr_init=0.03):
    """Polish that targets multiple peaks simultaneously with weighted sub-gradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    no_improve = 0

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Weight the top n_peaks by their excess over the median
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:n_peaks]
        w = np.zeros_like(c)
        for i, k in enumerate(top_k):
            w[k] = 1.0 / (i + 1)
        w = w / w.sum()
        
        w_rev = w[::-1]
        conv_wa = _full_convolve(w_rev, a)
        G = 2.0 * conv_wa[n-1 : 2*n-1][::-1]
        G_proj = G - np.dot(a, G)
        gnorm = np.linalg.norm(G_proj)
        
        if gnorm < 1e-15:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(80):
            a_try = a - alpha * G_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue

            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.8

            lr = min(lr_init, lr * 1.02)
    
    return best_a, best_sc
