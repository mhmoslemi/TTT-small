"""
Optimized step-function search combining momentum-based exact-max subgradient
descent, aggressive LSE annealing, spectral (Fourier-flat) initialization,
and multi-n exploration. Targets score below 1.5027 by using a more direct
optimization on the true max objective supplemented by smooth LSE refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _project_simplex(x):
    """Project x onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / rho > 0
    rho_star = int(np.sum(cond))
    if rho_star == 0:
        return np.ones(n) / n
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(x - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_subgradient(a, c, top_k=5, epsilon=0.005):
    """Compute weighted subgradient of max(c) using peaks above threshold."""
    n = len(a)
    c_max = float(c.max())
    threshold = c_max * (1.0 - epsilon)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > top_k:
        active = active[np.argsort(c[active])[::-1][:top_k]]
    # Weights proportional to excess
    excess = c[active] - threshold
    if float(excess.sum()) < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = excess / excess.sum()
    grad = np.zeros(n)
    for i in range(len(active)):
        k = int(active[i])
        wi = float(w[i]) * 2.0
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # grad[lo..hi] += wi * a[k-lo], a[k-lo-1], ..., a[k-hi]
        seg = a[k - lo: max(k - hi - 1, -1): -1]
        grad[lo:hi + 1] += wi * seg
    return grad


def _exact_subgradient(a, c):
    """Compute exact subgradient at the unique maximizer (or average of tied maximizers)."""
    n = len(a)
    c_max = float(c.max())
    active = np.where(c >= c_max * (1 - 1e-12))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - lo: max(k - hi - 1, -1): -1]
            grad[lo:hi + 1] += 2.0 * seg
    grad /= len(active)
    return grad


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


class _PNormProblem:
    """Minimize p-norm of autocorrelation (smooth approximation to max)."""
    def __init__(self, p, n):
        self.p = p
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c = np.maximum(c, 1e-30)
        return float(np.sum(c ** self.p) ** (1.0 / self.p))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c = np.maximum(c, 1e-30)
        p = self.p
        c_norm = float(np.sum(c ** p) ** (1.0 / p))
        # d||c||_p / da_j = (1/p) * ||c||_p^(1-p) * sum_k p * c_k^(p-1) * dc_k/dj
        # = ||c||_p^(1-p) * sum_k c_k^(p-1) * 2*a[k-j]
        # = 2 * ||c||_p^(1-p) * (c^(p-1) * a) convolved appropriately
        weights = c ** (p - 1)
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[self.n - 1: 2 * self.n - 1]
        return 2.0 * (c_norm ** (1.0 - p)) * a * h

            no_improve += 1
            patience += 1

        # Adaptive momentum
        momentum = min(momentum_end, momentum + 0.002)
        # Decay alpha
        if t % 50 == 0:
            alpha *= 0.998

        # Compute gradient (soft subgradient)
        if no_improve < 100:
            grad = _soft_subgradient(a, c, top_k=top_k, epsilon=epsilon)
        else:
            grad = _exact_subgradient(a, c)

        # Center gradient (stay on simplex)
        grad_mean = np.mean(grad)
        grad = grad - grad_mean

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-18:
            v = grad - np.mean(grad)
            v_norm = np.linalg.norm(v)
            if v_norm < 1e-18:
                break

        # Step size: scale by gradient magnitude for stability
        scale = 2.0 / (n * (best_max + 1e-30))
        step = alpha * scale / (v_norm * 0.1 + 1e-10)
        step = min(step, 5.0)

        # Try step with limited backtracking
        improved = False
        for bt in range(8):
            a_try = a - step * v
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(c_try.max())
            if sc_try < c_max - 1e-20:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            # Reset momentum if stuck
            if no_improve > 2000:
                v = np.zeros(n)
                alpha = max(alpha, 2.0 / (n * (best_max + 1e-30)) * 0.5)
                no_improve = 0
                patience = 0
                # Small random perturbation
                z = np.log(np.maximum(a, 1e-30)) + 0.01 * np.random.default_rng(t).standard_normal(n)
                a = _softmax(z)
            if patience > 10000:
                break

    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_start=0.005, tol_end=0.00001,
                       lr_start=0.05, top_k=5):
    """Projected gradient with adaptive top-k active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_max = float(c0.max())
    best_sc = 2.0 * n * best_max

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Select active set
        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = np.where(c >= (c_max - tol_val))[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > k_eff * 5:
            active = active[np.argsort(c[active])[::-1][:k_eff]]

        w = np.ones(len(active)) / len(active)
        grad = np.zeros(n)
        for i in range(len(active)):
            k = int(active[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                seg = a[k - lo: max(k - hi - 1, -1): -1]
                grad[lo:hi + 1] += 2.0 * float(w[i]) * seg

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        alpha = min(lr, 2.0 / (gnorm * gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(c_try.max())
            if sc_try * 2.0 * n < best_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.65
            if no_improve >= 500:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.3
                no_improve = 0
                stall += 1
                if stall > 15:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.02)

    return best_a, best_sc


def _peak_shave(a, deadline, n_trials=200, rng=None):
    """Direct peak shaving heuristic."""
    if rng is None:
        rng = np.random.default_rng(999)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            continue
        frac = rng.uniform(0.02, 0.10)
        n_reduce = max(1, int(len(contrib) * frac))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.15
            a_try[idx] -= red
            reduction += red
        if reduction < 1e-15:
            continue
        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            jv = k_star - int(idx)
            if 0 <= jv < n:
                low_mask[jv] = False
        if low_mask.sum() == 0:
            low_mask = np.ones(n, dtype=bool)
            for idx in top_idx:
                low_mask[idx] = False
            if low_mask.sum() == 0:
                continue
        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
    return best_a, best_sc


def _fourier_flat_init(n, n_iter=200, seed=123):
    """Spectral initialization: iterative amplitude flattening in Fourier domain."""
    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag2 = np.abs(A) ** 2
        mean_mag2 = np.mean(mag2)
        target_mag = np.sqrt(mean_mag2 + 1e-10)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        if np.linalg.norm(a_new - a) < 1e-14:
            break
        a = a_new
    return a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spectral":
        return _fourier_flat_init(n, n_iter=300, seed=777)
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, seed):
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: LSE L-BFGS with aggressive annealing
    P2: Momentum subgradient on exact max (new)
    P3: LSE very fine refinement
    P4: Multi-n exploration
    P5: Active-set polish + peak shave
    P6: Final combined polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    total_n = len(best_a)

    # === Phase 1: LSE L-BFGS aggressive (400s) ===
    print(f"\n[P1] LSE L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900.0, t_start + 400.0)
    a1 = _run_lbfgs_lse(best_a, p1_deadline, min_remaining=30.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  P1 NEW BEST: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P1 no improvement: {sc1:.10f} vs {best_score:.10f}")

    # === Phase 2: Momentum subgradient on exact max (350s) ===
    print(f"\n[P2] Momentum subgradient on exact max (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 600.0, t_start + 750.0)
    a2, sc2 = _run_momentum_subgrad(best_a, p2_deadline, top_k=5, epsilon=0.005,
                                     momentum_init=0.3, momentum_end=0.9)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  P2 NEW BEST: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P2 no improvement: {sc2:.10f} vs {best_score:.10f}")

    # === Phase 3: LSE very fine (200s) ===
    print(f"\n[P3] LSE very fine refinement (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400.0, t_start + 950.0)
    a3 = _run_lbfgs_lse(best_a, p3_deadline, min_remaining=30.0)
    sc3 = _score(a3)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  P3 NEW BEST: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  P3 no improvement: {sc3:.10f} vs {best_score:.10f}")

    # === Phase 4: Multi-n exploration (400s) ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [8192, 16384, 24576]
    if total_n not in n_targets:
        n_targets.append(total_n)
    n_targets = sorted([x for x in n_targets if x != len(best_a)])

    for n_try in n_targets:
        if time.monotonic() > deadline - 200.0:
            break
        remaining_for_n = min(300.0, deadline - time.monotonic() - 100.0)
        if remaining_for_n < 60:
            continue
        print(f"  Trying n={n_try} (remaining~{remaining_for_n:.0f}s)")
        a_init = _resample(best_a, n_try)
        # Quick LSE
        a_opt = _run_lbfgs_lse(a_init, deadline - 100.0, min_remaining=30.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  P4 NEW BEST: {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.1f}s")
        # Brief subgradient on this n
        if time.monotonic() < deadline - 200.0:
            a_sg, sc_sg = _run_momentum_subgrad(a_opt, min(deadline - 100.0, time.monotonic() + 80.0),
                                                 top_k=3, epsilon=0.01)
            if sc_sg < best_score:
                best_a = a_sg.copy()
                best_score = sc_sg
                print(f"  P4 NEW BEST (sg): {best_score:.10f} (n={n_try}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Active-set polish + peak shave ===
    print(f"\n[P5] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60.0:
        p5_deadline = min(deadline - 60.0, time.monotonic() + 150.0)
        a5, sc5 = _active_set_polish(best_a, p5_deadline, tol_start=0.003,
                                      tol_end=0.00003, lr_start=0.03, top_k=5)
        if sc5 < best_score:
            best_a = a5.copy()
            best_score = sc5
            print(f"  P5 NEW BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  P5 no improvement: {sc5:.10f}")

    print(f"\n[P5b] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40.0:
        a_ps, sc_ps = _peak_shave(best_a, min(deadline - 40.0, time.monotonic() + 80.0),
                                   n_trials=200, rng=rng)
        if sc_ps < best_score:
            best_a = a_ps.copy()
            best_score = sc_ps
            print(f"  P5b NEW BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final combined polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5.0:
        # One more subgradient round
        if time.monotonic() < deadline - 30.0:
            a_sg2, sc_sg2 = _run_momentum_subgrad(best_a, min(deadline - 30.0, time.monotonic() + 100.0),
                                                   top_k=3, epsilon=0.003,
                                                   momentum_init=0.5, momentum_end=0.95)
            if sc_sg2 < best_score:
                best_a = a_sg2.copy()
                best_score = sc_sg2
                print(f"  P6 sg NEW BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

        # Final LSE squeeze
        if time.monotonic() < deadline - 10.0:
            n_fin = len(best_a)
            c_fin = _full_convolve(best_a, best_a)
            mb = float(c_fin.max())
            beta_fin = max(mb * 0.00001, 1e-15)
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            prob = _LseProblem(beta_fin, n_fin)
            remaining = deadline - 10.0 - time.monotonic()
            max_iter = min(3000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-22,
                                           'gtol': 1e-18, 'maxcor': 60})
                a_fin = _softmax(result.x)
                a_fin = np.maximum(a_fin, 0.0)
                s_fin = a_fin.sum()
                if s_fin > 1e-10:
                    a_fin = a_fin / s_fin
                    sc_fin = _score(a_fin)
                    if sc_fin < best_score:
                        best_a = a_fin.copy()
                        best_score = sc_fin
                        print(f"  P6 lbfgs NEW BEST: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # === Diverse restarts if time allows ===
    print(f"\n[P7] Quick diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30.0:
        n_best = len(best_a)
        restarts = [
            ("perturb", _perturb(best_a, 0.03, 42)),
            ("spectral", _fourier_flat_init(n_best, 100, seed=777)),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 20.0:
                break
            remaining = deadline - 20.0 - time.monotonic()
            if remaining < 10.0:
                break
            a_opt = _run_lbfgs_lse(a_r, deadline - 15.0, min_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  P7 NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
        s = float(final.sum())
    if s > 0.01:
        # Normalize to have sum=1 for consistency (score is scale-invariant anyway)
        final = final / s
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
