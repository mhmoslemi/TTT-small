"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp, subgradient with simplex projection,
and adaptive QP-based peak reduction. Key improvements: proper simplex
projection, more aggressive beta schedule, and larger QP active sets.
"""
import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_score(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - 1.0) / ind
    if not cond.any():
        return np.ones(n) / n
    i = int(np.max(np.where(cond)[0]))
    theta = (css[i] - 1.0) / (i + 1)
    return np.maximum(v - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with geometric beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Estimate good initial beta
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.002)

    # More aggressive beta schedule
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
        0.015, 0.0075, 0.0038, 0.0019, 0.001,
        0.0005, 0.00025, 0.00012, 0.00006,
        0.00003, 0.000015, 0.000008, 0.000004, 0.000002
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if sc_new < 1.506:
                    print(f"    L-BFGS beta_idx={i}, beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a


def _subgradient_phase(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method with simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _conv_score(a)

    # Initial step size
    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    lr = c_max0 * 0.1

    # Tolerance for active set
    tol_frac = 0.01

    no_improve = 0
    max_stall = 100

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Active set: peaks within tol_frac of max
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute subgradient
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h
        g = g - np.mean(g)  # Remove mean to maintain sum constraint approximately
        gnorm = np.linalg.norm(g)

        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = lr / (gnorm + 1e-15)
        a_old_sc = current_sc
        improved = False
        for _ in range(80):
            a_try = a - alpha * g
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc * (1 - 1e-12):
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:
            no_improve += 1
            lr *= 0.85
            if no_improve % 20 == 0:
                tol_frac = max(0.001, tol_frac * 0.8)
            if no_improve >= max_stall:
                a = best_a.copy()
                lr = c_max0 * 0.05
                tol_frac = max(0.001, tol_frac * 0.5)
                no_improve = 0
                if lr < c_max0 * 1e-4:
                    break
        else:
            if tol_frac > 0.001:
                tol_frac = max(0.001, tol_frac * 0.98)
            lr = min(lr * 1.01, c_max0 * 0.05)

    return best_a, _score(best_a)


def _solve_qp_peaks(a, active_indices, target, deadline):
    """Solve QP to reduce peaks at active_indices to below target."""
    n = len(a)
    c = _full_convolve(a, a)

    m = len(active_indices)
    B_rows = []
    rhs = []
    for k in active_indices:
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        j_lo = k - i_hi
        j_hi = k - i_lo
        row = np.zeros(n)
        # B[k, j] = a[k-j] for j in [j_lo, j_hi]
        # a[k-j] for j from j_lo to j_hi: j_lo=k-i_hi, k-j_lo=i_hi; j_hi=k-i_lo, k-j_hi=i_lo
        # So a[k-j] for j in [j_lo, j_hi] = a[i_hi], a[i_hi-1], ..., a[i_lo]
        row[j_lo:j_hi + 1] = a[i_hi:i_lo - 1:-1] if i_hi > i_lo else a[i_hi:i_lo]
        B_rows.append(row)
        rhs.append((target - c[k]) / 2.0)

    B = np.array(B_rows)
    rhs = np.array(rhs)

    if not HAS_CVXPY:
        return None

    d_var = cp.Variable(n)
    constraints = [cp.sum(d_var) == 0]
    constraints.append(B @ d_var <= rhs)

    # Non-negativity for near-zero elements
    zero_tol = 1e-8 * a.max()
    neg_mask = a < zero_tol
    if neg_mask.any():
        constraints.append(d_var[neg_mask] >= -a[neg_mask] + zero_tol * 0.1)

    objective = cp.Minimize(cp.sum_squares(d_var))
    prob = cp.Problem(objective, constraints)

    for solver in ['ECOS', 'SCS', 'MOSEK', 'CLARABEL']:
        if time.monotonic() > deadline:
            return None
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    else:
        try:
            prob.solve()
        except Exception:
            return None

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None

    return np.asarray(d_var.value, dtype=np.float64)


def _qp_phase(a_init, deadline, min_time_remaining=30.0):
    """QP-based peak reduction with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _conv_score(a)

    active_frac = 0.03
    target_frac = 0.99
    consecutive_failures = 0
    max_failures = 15

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            consecutive_failures = 0
        else:
            consecutive_failures += 1

        if consecutive_failures >= max_failures:
            active_frac = min(0.3, active_frac * 1.8)
            target_frac = min(0.9999, target_frac + 0.005)
            consecutive_failures = 0
            if active_frac >= 0.25 and target_frac >= 0.999:
                break

        n_active = max(10, int(n * active_frac))
        c_sorted_idx = np.argsort(c)[::-1]
        active_indices = c_sorted_idx[:n_active]
        target = c_max * target_frac

        d = _solve_qp_peaks(a, active_indices, target, deadline - min_time_remaining)
        if d is None:
            consecutive_failures += 3
            continue

        # Line search
        a_old_sc = current_sc
        alpha = 1.0
        improved = False
        for _ in range(50):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            target_frac = min(0.9999, target_frac + 0.003)
            active_frac = max(0.01, active_frac * 0.7)

        # Gradually tighten
        elapsed_frac = (time.monotonic() - (deadline - min_time_remaining)) / max(1e-10, min_time_remaining)
        # Actually let's just use a simple schedule
        target_frac = max(0.90, target_frac * 0.997)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
