def _npo2(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    N = _npo2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]


def _score(a):
    """Compute the objective score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hmatvec(v, w, n):
    """Compute H_w @ v where (H_w)_{ij} = 2*w[i+j], w has length 2n-1."""
    vf = v[::-1]
    out_len = (2 * n - 1) + n - 1
    N = _npo2(out_len)
    fw = np.fft.rfft(w, N)
    fv = np.fft.rfft(vf, N)
    conv = np.fft.irfft(fw * fv, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(w, n, rhs, max_iter=500, tol=1e-12):
    """Solve H_w @ x = rhs using preconditioned CG."""
    x = np.zeros(n)
    r = rhs.copy()
    # Diagonal preconditioner: D[i] = 2*w[2i]
    diag = 2.0 * w[0::2]  # w[0], w[2], w[4], ... length n
    diag = np.maximum(diag, 1e-30)
    z = r / diag
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hmatvec(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        z_new = r / diag
        rz_new = float(r @ z_new)
        if not np.isfinite(rz_new) or rz_new < 0:
            rz_new = 0.0
        if rz_new < tol * tol * rz0:
            break
        beta = rz_new / max(rz, 1e-30)
        p = z_new + beta * p
        rz = rz_new
    return x

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _sobj(z, beta, n):
    """Smoothed objective: 2n * (max_c + beta * log(sum(exp((c-max_c)/beta))))"""
    a = _softmax(z)
    c = _conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    a = _softmax(z)
    c = _conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    wc = _conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """L-BFGS with geometric beta continuation on softmax parameterization."""
    n = len(a_init)
    c_cur = _conv(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # Aggressive geometric schedule
    n_steps = 30
    ratios = np.exp(np.linspace(0, np.log(1e-7), n_steps))
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 5))))

        obj_fn = _SmoObj(beta, n)
        grad_fn = _SmoGrad(beta, n)

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 80, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _dual_ascent(a_init, deadline, min_remaining=30.0):
    """
    Dual ascent: iterate between QP solve (primal) and weight update (dual).
    Adaptive active set with geometric tightening.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Active set fraction: start wide, tighten geometrically
    frac_schedule = np.array([1.0, 0.8, 0.6, 0.5, 0.4, 0.35, 0.3, 0.25,
                              0.2, 0.18, 0.15, 0.12, 0.10, 0.08, 0.06,
                              0.05, 0.04, 0.03, 0.025, 0.02, 0.015,
                              0.01, 0.008, 0.006, 0.005, 0.004, 0.003,
                              0.002, 0.0015, 0.001])
    
    # Each frac gets multiple rounds
    rounds_per_frac = 3
    total_rounds = len(frac_schedule) * rounds_per_frac

    for rnd in range(total_rounds):
        if time.monotonic() + min_remaining > deadline:
            break
        frac = frac_schedule[min(rnd // rounds_per_frac, len(frac_schedule) - 1)]

        c = _conv(a, a)
        c_max = float(c.max())
        n_active = max(int(2 * n - 1), int(frac * (2 * n - 1)))
        threshold = np.sort(c)[::-1][min(n_active, len(c) - 1)]

        active = c >= threshold
        if not active.any():
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_act

        # Solve QP: min a^T H_w a s.t. sum(a)=1, a>=0
        # Try CG for the KKT system
        rhs = np.ones(n)
        y = _cg_solve(w, n, rhs, max_iter=300, tol=1e-12)
        oney = float(y.sum())
        if abs(oney) > 1e-20 and np.isfinite(oney):
            a_qp = y / oney
            n_neg = int((a_qp < -1e-10).sum())
            if n_neg < 0.01 * n:
                # Project small negatives
                a_qp = np.maximum(a_qp, 0.0)
                s_qp = a_qp.sum()
                if s_qp > 1e-15:
                    a_qp = a_qp / s_qp
            else:
                # Significant negatives: use damped step from current a
                a_qp = a.copy()

        if np.isfinite(_score(a_qp)):
            sc_qp = _score(a_qp)
            if sc_qp < best_sc - 1e-14:
                best_sc = sc_qp
                best_a = a_qp.copy()
                a = a_qp.copy()
            else:
                # Damped linear interpolation
                for t in [0.5, 0.3, 0.1]:
                    a_try = (1 - t) * a + t * a_qp
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

        # Convergence check
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Early termination if improvement is tiny
        if rnd > 10:
            c_check = _conv(a, a)
            c_max_check = float(c_check.max())
            c_mean_check = float(c_check.mean())
            if c_max_check / c_mean_check < 1.001:
                break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=5000):
    """Subgradient method with Nesterov acceleration on the smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    lr = 0.05
    momentum = 0.9
    patience = 0
    no_improve = 0

    for i in range(max_steps):
        if time.monotonic() + 10 > deadline:
            break
        a = _softmax(z)
        c = _conv(a, a)
        c_max = float(c.max())

        # Adaptive tolerance for active set
        tol = max(0.001, 0.01 / (1 + i * 0.01)) * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        # Nesterov
        z_new = z_prev + momentum * (z_new - z) + (1 - momentum) * (-step * grad)
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 50 == 0:
            sc = _score(_softmax(z))
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = _softmax(z).copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    lr *= 0.8
                    no_improve = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=200):
    """Push down high-convolution entries via gradient of sum of excess^2."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() + 5 > deadline:
            break
        c = _conv(a, a)
        c_target = float(c.mean())
        c_max = float(c.max())
        if c_max < c_target * 1.0001:
            break

        excess = np.maximum(c - c_target, 0.0)
        if excess.max() < 1e-15:
            break

        a_rev = a[::-1]
        ec = _conv(excess, a_rev)
        grad_part = ec[n - 1:2 * n - 1]

        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "slepian_like":
        t = np.linspace(-1, 1, n)
        a = np.sinc(0.8 * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimizer combining L-BFGS, dual ascent, and polish phases.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent if available
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS initialization
    print(f"\n[Phase 1] L-BFGS init (t={time.monotonic() - t_start:.1f}s)")
    t_alloc_1 = min(400.0, (deadline - t_start) * 0.25)
    a1 = _run_lbfgs(parent, t_start + t_alloc_1, min_remaining=200.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Dual ascent (main workhorse)
    print(f"\n[Phase 2] Dual ascent (t={time.monotonic() - t_start:.1f}s)")
    t_alloc_2 = min(800.0, (deadline - t_start) * 0.45)
    if best_seq is not None and time.monotonic() < t_start + t_alloc_2 - 100:
        a2, sc2 = _dual_ascent(best_seq, t_start + t_alloc_2, min_remaining=50.0)
        update_best(a2, sc2, "dual_1")

    # Phase 3: Flatten polish
    print(f"\n[Phase 3] Flatten polish (t={time.monotonic() - t_start:.1f}s)")
    t_alloc_3 = min(150.0, (deadline - t_start) * 0.1)
    if best_seq is not None and time.monotonic() < t_start + t_alloc_3 - 80:
        a3, sc3 = _flatten_polish(best_seq, t_start + t_alloc_3, max_iters=200)
        update_best(a3, sc3, "flatten_1")

    # Phase 4: Subgradient with momentum
    print(f"\n[Phase 4] Subgradient+Nesterov (t={time.monotonic() - t_start:.1f}s)")
    t_alloc_4 = min(200.0, (deadline - t_start) * 0.12)
    if best_seq is not None and time.monotonic() < t_start + t_alloc_4 - 60:
        a4, sc4 = _subgrad_polish(best_seq, t_start + t_alloc_4, max_steps=5000)
        update_best(a4, sc4, "subgrad_1")

    # Phase 5: Second dual ascent (tighter)
    print(f"\n[Phase 5] Dual ascent round 2 (t={time.monotonic() - t_start:.1f}s)")
    t_alloc_5 = min(400.0, (deadline - t_start) * 0.2)
    if best_seq is not None and time.monotonic() < t_start + t_alloc_5 - 80:
        a5, sc5 = _dual_ascent(best_seq, t_start + t_alloc_5, min_remaining=60.0)
        update_best(a5, sc5, "dual_2")

    # Phase 6: L-BFGS refinement
    print(f"\n[Phase 6] L-BFGS refine (t={time.monotonic() - t_start:.1f}s)")
    t_alloc_6 = min(200.0, (deadline - t_start) * 0.12)
    if best_seq is not None and time.monotonic() < t_start + t_alloc_6 - 50:
        a6 = _run_lbfgs(best_seq, t_start + t_alloc_6, min_remaining=50.0)
        update_best(a6, _score(a6), "lbfgs_2")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq) if best_seq is not None else True:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        remaining = deadline - time.monotonic() - 80
        if remaining < 30:
            break
        a_opt = _run_lbfgs(a_init, deadline - 80, min_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 60:
            a_dp, sc_dp = _dual_ascent(a_opt, min(deadline - 40, time.monotonic() + 120),
                                       min_remaining=40.0)
            update_best(a_dp, sc_dp, f"n_{n_try}_dual")

    # Phase 8: Diverse restarts
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    restart_profiles = ["raised_cos", "slepian_like", "cos"]
    for i, prof in enumerate(restart_profiles):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic() - 20
        if remaining < 15:
            break
        n_r = len(best_seq)
        if i == 0:
            p_start = _perturb(best_seq, 0.03, rng)
        else:
            p_start = _make_profile(n_r, prof)

        a_r = _run_lbfgs(p_start, deadline - 20, min_remaining=10.0)
        sc_r = _score(a_r)
        update_best(a_r, sc_r, f"restart_{i}")

    # Phase 9: Final ultra-tight polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _flatten_polish(best_seq, deadline - 3, max_iters=100)
        update_best(a_fin, sc_fin, "final_flatten")

        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _dual_ascent(best_seq, deadline - 3, min_remaining=3.0)
            update_best(a_fin2, sc_fin2, "final_dual")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        # Scale to ensure evaluate_sequence gives same result
        # The score is scale-invariant, so any positive scaling works.
        # Keep sum in a reasonable range.
        target_sum = np.sqrt(2.0 * len(final))
        final = final * (target_sum / s)
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
