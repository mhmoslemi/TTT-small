"""
Optimized multi-phase optimizer for minimizing step function autocorrelation peak.
Uses adaptive LSE smoothing, Riemannian subgradient with BB step size,
CG-based Newton active-set, and spectral perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Core utilities - all top-level, no closures

def get_next_pow2(n):
    k = 4
    while k < n:
        k <<= 1
    return k

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = get_next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def lse_value(a, beta, n):
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def lse_grad_z(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # dLSE/da = 2n * (w * a_reversed) centered
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def lse_objective(z, beta, n):
    a = softmax(z)
    return lse_value(a, beta, n)

def active_set_mask(c, tol_frac):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(c, dtype=bool)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    return active

def hess_matvec(v, active, n):

    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n-1:2*n-1]
    return 2.0 * result

def cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):

        pH = float(np.dot(p, Hp))

            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = score(a_init)
    
    z_prev = z.copy()
    g_prev = None
    a_prev = None
    lr = 0.01

        if time.monotonic() > deadline:
            break
        a = softmax(z)
        c = full_convolve(a, a)
        active = active_set_mask(c, tol_frac)
        active_f = active.astype(np.float64)
        
        # Subgradient of max (approximated by average over active set)
        grad_a = hess_matvec(a, active_f, n)
        grad_z = a * (grad_a - np.dot(a, grad_a))

def perturb_fourier(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def perturb_amplitude(a, rng, sigma=0.05):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Multi-phase optimizer for minimizing step function autocorrelation peak.
Combines adaptive LSE smoothing, Riemannian subgradient with BB step size,
CG-based Newton active-set, and diversified spectral restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def get_next_pow2(n):
    k = 4
    while k < n:
        k <<= 1
    return k


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = get_next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def lse_objective_z(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def lse_grad_z(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

    if c_max < 1e-30:
        return np.zeros_like(c, dtype=bool)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    return active


def hess_matvec(v, active_f, n):
    N = get_next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n_dim = len(b)
    x = np.zeros(n_dim)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = active_set_mask(c, tol)
        active_f = active.astype(np.float64)

        grad = hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8

        def _hv(v):
            return hess_matvec(v, active_f, n) + mu * v

        dx = cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(np.dot(active_f, c))
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = full_convolve(a_try, a_try)
            f_try = float(np.dot(active_f, c_try))
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def subgrad_bb(a_init, deadline, n_steps=5000, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = score(a_init)

    z_prev = z.copy()
    g_prev = None
    lr = 0.01
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = softmax_fn(z)
        c = full_convolve(a, a)
        active = active_set_mask(c, tol_frac)
        active_f = active.astype(np.float64)

        grad_a = hess_matvec(a, active_f, n)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        if g_prev is not None:
            s_vec = z - z_prev
            y_vec = grad_z - g_prev
            sS = float(np.dot(s_vec, s_vec))
            sY = float(np.dot(s_vec, y_vec))
            if sS > 1e-30 and abs(sY) > 1e-30:
                lr_bb = sS / sY
                if 0 < lr_bb < 1.0:
                    lr = 0.5 * lr + 0.5 * lr_bb

        lr = min(max(lr, 1e-7), 0.05)
        z_prev = z.copy()
        g_prev = grad_z.copy()
        z = z - lr * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = softmax_fn(z)
            sc = score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 10:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.005

    a_final = softmax_fn(z)
    sc_final = score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def lbfgs_lse_adaptive(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = score(current_a)

    c_cur = full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    k = min(200, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    beta = max(gap * 0.5, max_c * 0.008)
    min_beta = max_c * 0.000005

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    no_improve = 0
    iter_count = 0

    while time.monotonic() + min_remaining < deadline and beta > min_beta:
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 12))))

        try:
            result = minimize(
                lse_objective_z, z, args=(beta, n),
                jac=lse_grad_z, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 50, 'maxls': 100}
            )
            a_new = softmax_fn(result.x)
            sc_new = score(a_new)
            iter_count += 1

            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 2:
                beta *= 0.35
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
                no_improve = 0
            else:
                beta *= 0.55

        except Exception:
            beta *= 0.4

    return current_a, current_sc


def perturb_log_fn(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return softmax_fn(z)


def perturb_fourier_fn(a, rng, max_amp=0.2):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def perturb_amplitude_fn(a, rng, sigma=0.04):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def perturb_smooth_fn(a, rng, bandwidth=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.standard_normal(n)
    kernel_size = max(3, int(n * bandwidth))
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (bandwidth * n * 0.3)) ** 2)
    kernel = kernel / kernel.sum()
    noise = np.convolve(noise, kernel, mode='same')
    z = z + 0.03 * noise
    z = z - z.mean()
    return softmax_fn(z)


def make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


_best_seq = None
_best_score = np.inf


def update_best(p, sc=None, label=""):
    global _best_seq, _best_score
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = score(p)
    if np.isfinite(sc) and sc < _best_score:
        _best_seq = p.copy()
        _best_score = float(sc)
        print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}]")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: adaptive LSE, Riemannian subgradient with BB step size,
    Newton active-set, spectral perturbations, and diversified restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = make_profile(16384, "uniform")
        update_best(parent, score(parent), "init")

    # === Phase 1: Primary adaptive LSE (350s) ===
    print(f"\n[Phase 1] Adaptive LSE (primary)")
    p1_deadline = min(deadline, t_start + 350)
    a1, sc1 = lbfgs_lse_adaptive(_best_seq.copy(), p1_deadline, min_remaining=30)
    update_best(a1, sc1, "lse_primary")

    # === Phase 2: Newton active-set (150s) ===
    print(f"\n[Phase 2] Newton active-set")
    p2_deadline = min(deadline, t_start + 500)
    a2, sc2 = newton_active_set(_best_seq, p2_deadline, max_iter=300, tol_frac=0.002)
    update_best(a2, sc2, "newton_p2")

    # === Phase 3: Subgradient BB (200s) ===
    print(f"\n[Phase 3] Subgradient BB")
    p3_deadline = min(deadline, t_start + 700)
    a3, sc3 = subgrad_bb(_best_seq, p3_deadline, n_steps=10000, tol_frac=0.001)
    update_best(a3, sc3, "sg_bb_p3")

    # === Phase 4: Second LSE pass (200s) ===
    print(f"\n[Phase 4] LSE second pass")
    p4_deadline = min(deadline, t_start + 900)
    a4, sc4 = lbfgs_lse_adaptive(_best_seq.copy(), p4_deadline, min_remaining=25)
    update_best(a4, sc4, "lse_p4")

    # === Phase 5: Newton + SG alternating (200s) ===
    print(f"\n[Phase 5] Newton + SG alternating")
    a_cur = _best_seq.copy()
    p5_end = min(deadline, t_start + 1100)
    for rnd in range(5):
        if time.monotonic() > p5_end - 30:
            break
        nd = min(p5_end - 30, time.monotonic() + 40)
        a_n, sc_n = newton_active_set(a_cur, nd, max_iter=100, tol_frac=0.002)
        if sc_n < score(a_cur):
            a_cur = a_n
        update_best(a_cur, score(a_cur), f"alt_n_{rnd}")
        sg_d = min(p5_end - 15, time.monotonic() + 35)
        a_s, sc_s = subgrad_bb(a_cur, sg_d, n_steps=3000, tol_frac=0.001)
        if sc_s < score(a_cur):
            a_cur = a_s
        update_best(a_cur, score(a_cur), f"alt_s_{rnd}")

    # === Phase 6: Diversified restarts (250s) ===
    print(f"\n[Phase 6] Diversified restarts")
    p6_end = min(deadline, t_start + 1350)
    restart_funcs = [
        ("log_002", 0.02, "log"),
        ("log_004", 0.04, "log"),
        ("fourier", 0.0, "fourier"),
        ("amp", 0.0, "amp"),
        ("smooth", 0.0, "smooth"),
    ]
    for name, sig, kind in restart_funcs:
        if time.monotonic() > p6_end - 25:
            break
        if kind == "log":
            p_start = perturb_log_fn(_best_seq, sig, rng)
        elif kind == "fourier":
            p_start = perturb_fourier_fn(_best_seq, rng, max_amp=0.15)
        elif kind == "amp":
            p_start = perturb_amplitude_fn(_best_seq, rng, sigma=0.03)
        else:
            p_start = perturb_smooth_fn(_best_seq, rng)

        a_r, sc_r = lbfgs_lse_adaptive(p_start, p6_end, min_remaining=15)
        update_best(a_r, sc_r, f"restart_{name}")
        if score(a_r) < _best_score * 1.003 and time.monotonic() < p6_end - 8:
            nd2 = min(p6_end - 5, time.monotonic() + 15)
            a_r2, sc_r2 = newton_active_set(a_r, nd2, max_iter=50, tol_frac=0.003)
            update_best(a_r2, sc_r2, f"restart_{name}_n")

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print(f"\n[Phase 7] Final ultra-fine polish")
    # Tight Newton
    a_fn, sc_fn = newton_active_set(_best_seq, deadline - 80, max_iter=200, tol_frac=0.0005)
    update_best(a_fn, sc_fn, "final_newton_tight")

    # Very tight SG
    a_fs, sc_fs = subgrad_bb(_best_seq, deadline - 30, n_steps=5000, tol_frac=0.0003)
    update_best(a_fs, sc_fs, "final_sg_tight")

    # One more Newton
    a_fn2, sc_fn2 = newton_active_set(_best_seq, deadline - 5, max_iter=100, tol_frac=0.0005)
    update_best(a_fn2, sc_fn2, "final_newton2")

    # Final tiny LSE
    if _best_seq is not None and time.monotonic() < deadline - 3:
        n_fin = len(_best_seq)
        c_fin = full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = max(2, deadline - 3 - time.monotonic())
        if remaining > 3:
            beta_final = mb * 0.00001
            max_iter = min(2000, max(50, int(remaining / 0.005)))
            try:
                result = minimize(
                    lse_objective_z, z_fin, args=(beta_final, n_fin),
                    jac=lse_grad_z, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 50}
                )
                a_fin = softmax_fn(result.x)
                sc_fin = score(a_fin)
                update_best(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = make_profile(256, "uniform")
            print("FALLBACK uniform n=256")
        else:
            sc_final = score(final)
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={score(fallback):.10f}")
    return [float(x) for x in fallback]
