"""
Minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. Flattening pre-condition (minimize convolution variance)
2. Log-sum-exp smoothed max minimization via L-BFGS
3. Active-set Newton refinement
4. Multi-profile exploration with adaptive time budget
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_a_flipa(a, A_vec, n):
    """Compute sum_i a[i] * A_vec[j+i] for j=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n-1:2*n-1]

def _flatten_objective(z, n):
    """Objective: sum_k (c[k] - 1/(2n-1))^2 with a=softmax(z)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    target = 1.0 / (2*n - 1)
    diff = c - target
    return float(np.dot(diff, diff))

def _flatten_gradient(z, n):
    """Gradient of flattening objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    target = 1.0 / (2*n - 1)
    f = c - target
    h = _corr_a_flipa(a, f, n)
    ah = float(np.dot(a, h))
    grad_a = 2.0 * h
    grad_z = a * (grad_a - ah)
    return grad_z

def _ls_objective(z, beta, n):
    """Log-sum-exp smoothed max objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _ls_gradient(z, beta, n):
    """Gradient of log-sum-exp smoothed max objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _corr_a_flipa(a, w, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

"""
Minimizer for 2*n*max(conv(a,a))/sum(a)^2 using multi-phase optimization:
1. Flattening pre-condition (minimize convolution variance via L-BFGS)
2. Aggressive log-sum-exp smoothed max minimization
3. Active-set Newton refinement
4. Asymmetric profile exploration
5. Peak-targeted coordinate descent
Key innovation: variance-flattening pre-condition exploits the fact that
asymmetric sequences can have max(c) < S^2/n (beating the symmetric bound of 2).

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_a_flipa(a, A_vec, n):
    """Compute sum_i a[i]*A_vec[j+i] for j=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _flatten_objective(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    target = 1.0 / (2 * n - 1)
    diff = c - target
    return float(np.dot(diff, diff))


def _flatten_gradient(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    target = 1.0 / (2 * n - 1)
    f_vec = c - target
    h = _corr_a_flipa(a, f_vec, n)
    ah = float(np.dot(a, h))
    grad_a = 2.0 * h
    grad_z = a * (grad_a - ah)
    return grad_z


def _ls_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
