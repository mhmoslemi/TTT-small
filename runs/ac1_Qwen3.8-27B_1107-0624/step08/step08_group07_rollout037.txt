```python
"""
Optimized step function search using L_p-norm smoothing with adaptive
Newton active-set refinement. Key innovations:
1. Clean L_p surrogate with FFT-based gradient computation
2. Newton-based active-set equalization for final convergence
3. Multi-phase optimization with increasing p
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
```

```python
"""
Step function optimization using L_p-norm smooth surrogate with FFT-based
gradients, multi-phase temperature scheduling, and Newton active-set refinement.
Key improvements over prior approach: (1) numerically stable L_p gradient via
log-domain weight computation, (2) Newton-based active-set equalization for
final convergence, (3) adaptive p-schedule, (4) symmetry exploitation.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _log_lp_norm(c, p):
    """Compute log(||c||_p) in a numerically stable way."""
    c_safe = np.maximum(c, 1e-300)
    log_c = np.log(c_safe)
    p_log_c = p * log_c
    M = p_log_c.max()
    log_sum_exp = M + np.log(np.sum(np.exp(p_log_c - M)))
    return log_sum_exp / p


def _lp_gradient(a, c, p):
    """Compute gradient of ||c||_p w.r.t. a, where c = conv(a, a).
    Returns (g_a, Fp) where g_a[j] = d(||c||_p)/da[j].
    """
    n = len(a)
    c_safe = np.maximum(c, 1e-300)
    log_c = np.log(c_safe)
    log_Fp = _log_lp_norm(c, p)

    # w[k] = exp((p-1)*(log_c[k] - log_Fp)) = (c[k]/Fp)^(p-1)
    # Since c[k] <= Fp, we have w[k] <= 1, no overflow
    log_w = (p - 1) * (log_c - log_Fp)
    w = np.exp(np.minimum(log_w, 0.0))

    # g_a[j] = 2 * cross_corr(w, a)[j] = 2 * sum_k w[k]*a[k-j]
    # Use convolution: cross_corr(w,a)[j] = conv(w, a_rev)[n-1+j]
    a_rev = a[::-1]
    N = _next_pow2(len(c) + n - 1)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(W * Ar, N)
    g_a = 2.0 * conv_result[n - 1: n - 1 + n]

    Fp = float(np.exp(log_Fp))
    return g_a, Fp


def _active_lag_gradient(a, c, lags):
    """Compute subgradient of max(c[k] for k in lags) w.r.t. a.
    Uses softmax weighting over the active lags for smoothing."""
    n = len(a)
    c_lags = c[lags]
    T = max((c_lags.max() - c_lags.min()) * 0.1, 1e-15)
    w = np.exp((c_lags - c_lags.max()) / T)
    w = w / w.sum()

    g = np.zeros(n)
    for idx, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += w[idx] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _adam_optimize_lp(a_init, deadline, p, lr_start, lr_end,
                      max_iter=500000, eval_interval=50,
                      backtrack_interval=200, active_k=5,
                      switch_to_active_at=0.7):
    """Adam optimization of L_p objective in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0
    last_improve_t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = _softmax(z)
        c = _fft_convolve(a, a)

        # Choose gradient: L_p or active-set
        if frac > switch_to_active_at:
            c_max = float(c.max())
            tol = c_max * 0.998
            active = np.where(c >= tol)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            elif len(active) > active_k:
                top_idx = np.argsort(c[active])[::-1][:active_k]
                active = active[top_idx]
            g_a = _active_lag_gradient(a, c, active)
        else:
            g_a, _ = _lp_gradient(a, c, p)

        if not np.all(np.isfinite(g_a)):
            g_a = np.zeros(n)

        # Log-space gradient: grad_z = a * (g_a - dot(a, g_a))
        W = float(np.dot(a, g_a))
        grad_z = a * (g_a - W)

        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        # Backtracking check
        if t % backtrack_interval == 0:
            a_cand = _softmax(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z.copy()
                m_buf *= 0.5
                v_buf *= 0.5
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve_t = t
            else:
                patience += 1

            if patience > 6 and t - last_improve_t > 500:
                current_lr_mult *= 0.6
                patience = 0
                if current_lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    current_lr_mult = 0.2

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _newton_active_set(a, deadline, max_iter=500, top_m=4, tol=0.002):
    """Newton-based active-set equalization.
    Solves linearized system to equalize top peaks while preserving sum.
    """
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Find active peaks
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            # Expand tolerance
            tol_val = c_max * (1 - tol * 5)
            active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > top_m:
            sorted_a = active[np.argsort(c[active])[::-1][:top_m]]
            active = sorted_a

        K = len(active)
        # Build gradient matrix G (K x n): G[i, j] = 2*a[active[i]-j] (valid)
        G = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                G[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        # Add sum constraint row
        G_full = np.vstack([G, np.ones(n)])

        # Target: equalize peaks (reduce each by its excess over mean)
        c_active = c[active]
        c_mean = c_active.mean()
        b = np.concatenate([-(c_active - c_mean), [0.0]])

        # Minimum norm solution: d = G_full^T @ inv(G_full @ G_full^T) @ b
        K1 = K + 1
        Gram = G_full @ G_full.T
        try:
            Gram_inv = np.linalg.inv(Gram + 1e-14 * np.eye(K1))
            d = G_full.T @ (Gram_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(G_full, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Line search
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            # Try smaller steps
            for alpha in [0.0001, 0.00005, 0.00001]:
                a_cand = a + alpha * d
                a_cand = np.maximum(0, a_cand)
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                sc = _score_norm(a_cand)
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=50000, top_k=3, initial_lr=0.002):
    """Subgradient method with decreasing step size."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.9995:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        # Project to tangent space
        W = float(np.dot(a, g))
        g_tan = a * (g - W)
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-18:
            break

        step = lr * g_tan / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9998
        else:
            lr *= 0.97
            if lr < 1e-10:
                break

    return best_a, _score_norm(best_a)


def _coordinate_descent_polish(a, deadline, max_rounds=500):
    """Coordinate descent on the top-contributing elements."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        # Focus on elements that most affect the peak
        for j in range(lo, hi + 1):
            old_val = best_a[j]
            if old_val < 1e-18:
                continue
            for scale in [0.85, 0.92, 0.97, 1.03, 1.08, 1.15]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Sweep all coordinates sparsely
            any_improve = False
            step = max(1, n // 500)
            for j in range(0, n, step):
                old_val = best_a[j]
                if old_val < 1e-18:
                    continue
                for scale in [0.8, 1.2]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        any_improve = True
                        improved = True
                        break
                if improved:
                    break
            if not any_improve:
                break

    return best_a, best_score


def _try_symmetry(a):
    """Try enforcing palindromic symmetry and see if it helps."""
    n = len(a)
    n_half = n // 2
    a_sym = np.zeros(n)
    a_sym[:n_half] = (a[:n_half] + a[n - 1:n_half - 1:-1]) / 2.0
    if n % 2 == 1:
        a_sym[n_half] = a[n_half]
    a_sym[n_half:] = a_sym[n - 1:n_half - 1:-1]
    s = a_sym.sum()
    if s < 1e-15:
        return a
    a_sym /= s
    return a_sym
```
