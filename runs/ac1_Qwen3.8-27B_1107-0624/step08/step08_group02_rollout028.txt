"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Smoothed L-BFGS with adaptive continuation on log-sum-exp approximation
2. Active-set Newton method with FFT-based Hessian application
3. LP-inspired subgradient direction for diversification
4. Multi-n exploration for length optimization
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute (sum_k W_k H_k) v efficiently using FFT."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Conjugate gradient for the Hankel system Hx = rhs."""
    n = n
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=200):
    """Active set Newton method for the convex problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    prev_cmax = None

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set tolerance: start loose, tighten over iterations
        base_tol = max(0.001, 0.01 / (1.0 + rnd * 0.5))
        for tol_frac in [base_tol, base_tol * 0.3, base_tol * 0.1, 0.001, 0.0003, 0.0001, 1e-5]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 2:
                break
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))

        if gn < 1e-14 * n:
            break

        # Solve H dx = -g_t with sum(dx) = 0
        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with backtracking
        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo condition
            if c_max_try < c_max - 0.001 * alpha * gn / (2.0 * n) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-18:
                break

        if not step_done:
            # Fallback: subgradient step with varying magnitudes
            best_trial_sc = np.inf
            best_trial_a = None
            for trial in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(c_try.max())
                if sc_try < best_trial_sc:
                    best_trial_sc = sc_try
                    best_trial_a = a_try
            if best_trial_a is not None and best_trial_sc < c_max - 1e-15:
                a = best_trial_a
                step_done = True

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break
        if abs(prev_cmax - c_max) < 1e-14 * c_max if prev_cmax is not None else False:
            break
        prev_cmax = c_max

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """Adaptive L-BFGS with continuation on smoothing parameter."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-11
    prev_sc = current_sc
    n_phases = 0
    max_phases = 30

    while beta > min_beta and n_phases < max_phases:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.004 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, a_init if n_phases == 0 else current_a)
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc
        n_phases += 1

        # Decide beta reduction based on convergence
        if new_gap < 1e-10 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new and sc_improved > 1e-13:
            beta *= 0.2
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.3
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _lp_direction(a, M, n_subset=None):
    """
    Compute LP-inspired direction: find g >= 0 maximizing sum(g)
    subject to (a*g)_k <= M for a subset of constraints.
    Falls back to subgradient if LP is too expensive.
    """
    n = len(a)
    if n > 8192:
        # Too large for LP, use subgradient
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        # Return negative gradient direction projected to simplex
        d = -grad
        d = d - np.mean(d)
        return d

    # For smaller n, try to solve the LP
    # Constraint: (a * g)_k <= M for selected k
    # Variable: g >= 0
    # Objective: maximize sum(g)
    try:
        # Build constraint matrix A where A[k,j] = a[k-j] if valid
        # Only use a subset of constraints for speed
        c = _full_convolve(a, a)
        # Use constraints where (a*a)_k is in the top half
        threshold = np.median(c)
        constraint_indices = np.where(c >= threshold)[0]
        if len(constraint_indices) > 4096:
            # Subsample
            constraint_indices = constraint_indices[::len(constraint_indices)//4096]

        A = np.zeros((len(constraint_indices), n))
        for idx, k in enumerate(constraint_indices):
            j_start = max(0, k - n + 1)
            j_end = min(k + 1, n)
            for j in range(j_start, j_end):
                A[idx, j] = a[k - j]

        # LP: max 1^T g s.t. A g <= M, g >= 0
        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(len(constraint_indices), M),
                      bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            g = res.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
                d = g - a / a.sum()
                return d
    except Exception:
        pass

    # Fallback: subgradient
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 0.01 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    d = -grad
    d = d - np.mean(d)
    return d


def _subgrad_polish(a_init, deadline, max_steps=2000):
    """Subgradient method with adaptive step size on the simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set tolerance
        tol = max(0.001, 0.01 / (1.0 + i * 0.1)) * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.06)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    elif kind == "fejer_like":
        # Approximate Fejer kernel shape
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _cycle_optimize(a_init, deadline, min_time_remaining=60.0):
    """Alternate between L-BFGS bursts and Newton polish."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    cycle_time = min(45.0, max(10.0, (deadline - min_time_remaining - time.monotonic()) / 8.0))

    for cycle in range(10):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t_cycle_start = time.monotonic()
        t_cycle_end = t_cycle_start + cycle_time

        # L-BFGS burst
        if time.monotonic() < t_cycle_end - 5:
            a_lb = _run_adaptive_lbfgs(
                current_a,
                min(t_cycle_end - 5, deadline - min_time_remaining),
                min_time_remaining=5.0
            )
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb

        # Newton polish
        if time.monotonic() + 10 < min(t_cycle_end, deadline - min_time_remaining):
            a_np, sc_np = _newton_active_polish(
                current_a,
                min(t_cycle_end, deadline - min_time_remaining),
                max_rounds=50
            )
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np

        cycle_time *= 1.15
        if cycle_time > 90:
            cycle_time = 90

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization driver."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick profile exploration ===
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic()-t_start:.1f}s)")
    profiles = ["tapered", "fejer_like", "asym_taper", "left_heavy", "bimodal",
                "exp_decay", "raised_cos", "cos"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f} (improved)")

    # Try perturbed parent
    for sigma in [0.03, 0.08, 0.15]:
        if time.monotonic() > deadline - 700:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_profile_sc:
            best_profile_sc = sc_pert
            best_profile_a = a_pert
            print(f"  Perturbed parent (sigma={sigma}): score={sc_pert:.10f}")

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 700:
        a_quick = _run_adaptive_lbfgs(
            best_profile_a,
            min(deadline - 700, time.monotonic() + 120),
            min_time_remaining=600.0
        )
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

    # === Phase 2: Main L-BFGS optimization ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_profile_a
    a_main = _run_adaptive_lbfgs(start_a, deadline, min_time_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Cycle optimization (alternating methods) ===
    print(f"\n[Phase 3] Cycle optimization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_cyc = _cycle_optimize(best_seq, deadline, min_time_remaining=350.0)
        update_best(a_cyc, _score(a_cyc), "cycle")

    # === Phase 4: Newton polish ===
    print(f"\n[Phase 4] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_np, sc_np = _newton_active_polish(
            best_seq,
            min(deadline - 250, time.monotonic() + 200),
            max_rounds=200
        )
        update_best(a_np, sc_np, "newton_1")

    # === Phase 5: Subgradient diversification ===
    print(f"\n[Phase 5] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        # Perturb and subgradient
        a_pert = _perturb(best_seq, 0.05, rng)
        a_sg, sc_sg = _subgrad_polish(
            a_pert,
            min(deadline - 150, time.monotonic() + 80)
        )
        update_best(a_sg, sc_sg, "subgrad_pert")

        # LP-direction based step
        if time.monotonic() < deadline - 120:
            c_cur = _full_convolve(best_seq, best_seq)
            M_cur = float(c_cur.max())
            d = _lp_direction(best_seq, M_cur, n_subset=4096)
            for step_size in [0.1, 0.05, 0.02, 0.01]:
                a_try = best_seq + step_size * d
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_score:
                    update_best(a_try, sc_try, f"lp_dir_{step_size}")
                    best_seq = a_try
                    break

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    base_n = len(best_seq) if best_seq is not None else n_parent
    n_candidates = [base_n * 2, base_n, base_n // 2]
    n_candidates = [x for x in n_candidates if 1024 <= x <= 65536]
    # Remove duplicates
    n_candidates = list(dict.fromkeys(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        print(f"  Trying n={n_try}")
        a_init = _resample(best_seq, n_try)
        a_opt = _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_np3, sc_np3 = _newton_active_polish(
                a_opt,
                min(deadline - 25, time.monotonic() + 25),
                max_rounds=50
            )
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # === Phase 7: Final intensive Newton ===
    print(f"\n[Phase 7] Final Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(
            best_seq,
            deadline - 1,
            max_rounds=300
        )
        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS at ultra-small beta
        if best_seq is not None and time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-12
            remaining = deadline - 3 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.015)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-18,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_vf, sc_vf = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=100)
            update_best(a_vf, sc_vf, "very_final")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
