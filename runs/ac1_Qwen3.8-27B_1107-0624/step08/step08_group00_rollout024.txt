"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, accelerated mirror descent,
and multi-scale exploration. Key innovations: proper gradient computation
via cross-correlation, adaptive beta scheduling, and Nesterov-accelerated
subgradient for the non-smooth max component.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(a):
    """Project vector onto the probability simplex."""
    n = len(a)
    if n == 0:
        return a
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)


def _smooth_weighted_corr(a, w, n):
    """Compute h[j] = sum_m a[m]*w[j+m] for j=0..n-1 using FFT cross-correlation."""
    N_fft = _next_pow2(n + len(w) - 1)
    A = np.fft.fft(a, N_fft)
    W = np.fft.fft(w, N_fft)
    H = np.fft.ifft(A * np.conj(W))[:n].real
    return H


def _smooth_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _autocorr(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _autocorr(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _smooth_weighted_corr(a, w, n)
    grad_a = 4.0 * n * h
    W_dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W_dot)
    return grad_z


class _SmoothProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def func(self, z):
        return _smooth_obj(z, self.beta, self.n)

    def grad(self, z):
        return _smooth_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=60.0):
    """Run L-BFGS with adaptive beta schedule on the log-sum-exp smoothed objective."""
    n = len(a_init)
    current_a = a_init.copy()
    s = current_a.sum()
    if s < 1e-15:
        return current_a
    current_a = current_a / s
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _autocorr(current_a)
    c_max = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.002)

    # Adaptive beta schedule: start moderate, decrease gradually
    n_betas = 12
    betas = base_beta * (0.7 ** np.arange(n_betas))
    betas = np.maximum(betas, c_max * 0.00001)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_te = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_te * 8))))

        prob = _SmoothProblem(beta, n)
        try:
            result = minimize(
                prob.func, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,

            a_new = _softmax_fn(result.x)

                current_sc = sc_new

                z = z - z.mean()
                if i % 3 == 0:

def _mirror_descent(a_init, deadline, min_remaining=30.0, n_iter_cap=500000):
    """Nesterov-accelerated mirror descent (exponential weights) for peak reduction."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Initialize
    t = 1.0
    a_prev = a.copy()
    eta = 1.0 / (4.0 * n * float(np.max(np.abs(a))))
    no_improve = 0
    max_no_improve = 5000

    for it in range(n_iter_cap):
        if time.monotonic() + min_remaining > deadline:
            break

        # Nesterov extrapolation
        coeff = (t - 1) / (t + 1)
        a_early = a + coeff * (a - a_prev)
        a_early = np.maximum(a_early, 1e-30)
        s_early = a_early.sum()
        a_early = a_early / s_early

        # Compute gradient
        c = _autocorr(a_early)
        c_max = float(c.max())
        # Use top peaks with softmax weighting
        beta_md = c_max * 0.01
        shifted = (c - c_max) / beta_md
        e = np.exp(np.clip(shifted, -700, 0))
        w = e / e.sum()

        h = _smooth_weighted_corr(a_early, w, n)
        grad_a = 4.0 * n * h
        grad_a = grad_a - float(np.mean(grad_a))  # Center for sum constraint

        # Mirror descent step
        a_prev = a.copy()
        a_new = a_early * np.exp(-eta * grad_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
            eta = min(eta * 1.01, 1.0 / (n * 0.1))
        else:
            no_improve += 1
            if no_improve % 1000 == 0:
                eta *= 0.8
                if no_improve >= max_no_improve:
                    a = best_a.copy()
                    a_prev = a.copy()
                    no_improve = 0
                    max_no_improve = int(max_no_improve * 0.7)
                    if max_no_improve < 500:
                        break

        a = a_new
        t += 1

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_remaining=20.0):
    """Direct subgradient method on the max (no smoothing) with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    alpha = 1.0 / (n * 0.5)
    momentum = 0.7
    v = np.zeros(n)
    no_improve = 0

    for it in range(200000):
        if time.monotonic() + min_remaining > deadline:
            break

        c = _autocorr(a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Subgradient: top peaks with soft weighting
        top_k = max(1, int(n * 0.001))
        c_sorted_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        w[c_sorted_idx] = 1.0 / top_k

        h = _smooth_weighted_corr(a, w, n)
        grad = 4.0 * n * h
        grad = grad - float(np.mean(grad))

        # Momentum update
        v = momentum * v - alpha * grad
        a_new = a + v
        a_new = _project_simplex(a_new)

        sc_new = _score(a_new)
        if sc_new < current_sc:
            a = a_new
            alpha = min(alpha * 1.005, 2.0 / n)
        else:
            alpha *= 0.7
            v *= 0.5
            if no_improve > 10000:
                a = best_a.copy()
                v = np.zeros(n)
                alpha = 1.0 / n
                no_improve = 0
                break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _random_init(n, kind, rng):
    """Generate a random initial sequence of given type."""

    elif kind == "gaussian":

        return a / a.sum()
    elif kind == "periodic":

    elif kind == "bimodal":

    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** rng.uniform(0.3, 2.0)

    elif kind == "sparse":

    all_ns.sort(key=lambda x: abs(x - len(best_seq)))

    for n_try in all_ns:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        remaining = deadline - time.monotonic() - 60
        if remaining < 60:
            break
        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_n, sc_n = _try_different_n(best_seq, n_try, deadline, min_remaining=30.0)
        print(f"  n={n_try}: score={sc_n:.10f}")
        if sc_n < best_score:
            best_seq = a_n.copy()
            best_score = sc_n
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

    # === Phase 5: Diverse restarts on best n ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    init_kinds = ["periodic", "bimodal", "tapered", "sparse", "gaussian"]
    for kind in init_kinds:
        if time.monotonic() > deadline - 80:
            break
        a_init = _random_init(n_best, kind, rng)
        # Blend with best
        for mix in [0.7, 0.5]:
            if time.monotonic() > deadline - 40:
                break
            a_blend = mix * a_init + (1 - mix) * best_seq
            a_blend = _project_simplex(a_blend)
            a_opt = _run_lbfgs(a_blend, deadline, min_remaining=40.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart {kind}, mix={mix}]")

    # === Phase 6: Final L-BFGS with very small beta ===
    print(f"\n[P6] Final fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        n = len(best_seq)
        c_best = _autocorr(best_seq)
        mb = float(c_best.max())
        # Use very small beta for near-exact max
        beta_final = mb * 0.00001
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _SmoothProblem(beta_final, n)
        remaining = deadline - time.monotonic() - 10
        max_iter = min(5000, max(50, int(remaining / 0.01)))
        try:
            result = minimize(prob.func, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 100, 'maxls': 60})
            a_fin = _softmax_fn(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
        except Exception:
            pass

    # === Phase 7: Tiny mirror descent polish ===
    print(f"\n[P7] Final mirror polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_mp, sc_mp = _mirror_descent(best_seq, deadline, min_remaining=3.0, n_iter_cap=100000)
        if sc_mp < best_score:
            best_seq = a_mp.copy()
            best_score = sc_mp
            print(f"  NEW BEST: {best_score:.10f} [final_mirror]")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
