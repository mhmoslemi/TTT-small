"""
Optimized active-set Newton with FFT-based Hessian application for minimizing
f(a) = 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: structured O(n log n) Hessian matvec via FFT, adaptive active set
identification, multi-scale initialization, CG solve with regularization,
and population-based perturbation/restart for escaping local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hess_matvec(v, s_vec, m, n):
    """
    Apply Hessian H to vector v where H[j1][j2] = (2/m) * 1[j1+j2 in S].
    Uses FFT-based convolution.
    s_vec: indicator vector of active set S, length 2n-1.
    """
    # (Hv)[j1] = (2/m) * sum_{j2} s_vec[j1+j2] * v[j2]
    # = (2/m) * (v * s_rev)[2n-2-j1]
    s_rev = s_vec[::-1]
    out_len = n + len(s_rev) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(v, N)
    fs = np.fft.rfft(s_rev, N)
    conv = np.fft.irfft(fv * fs, N)[:out_len]
    # We need conv[2n-2-j1] for j1 = 0, ..., n-1
    # That's conv[n-1], conv[n], ..., conv[2n-2]
    result = conv[n - 1:2 * n - 1][::-1]
    return (2.0 / m) * result


def _cg_solve(matvec_fn, rhs, n, max_iter=200, tol=1e-10, reg=1e-8):
    """Conjugate gradient solver with regularization for PSD systems."""
    x = np.zeros(n)
    r = rhs.copy()
    # Add regularization if needed
    r_reg = r.copy()
    p = r_reg.copy()
    rs_prev = np.dot(r_reg, r_reg)
    if rs_prev < 1e-30:
        return x
    for i in range(max_iter):
        Ap = matvec_fn(p) + reg * p
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs_prev / pAp
        x += alpha * p
        r_reg -= alpha * Ap
        rs_new = np.dot(r_reg, r_reg)
        if np.sqrt(rs_new) < tol * (1.0 + np.sqrt(rs_prev)):
            break
        p = r_reg + (rs_new / rs_prev) * p
        rs_prev = rs_new
    return x

    Active-set Newton method for minimizing max(c) over active set S,
    subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)

            n_active = 1
        # Cap active set size
        if n_active > 3 * n:
            active = np.zeros(2 * n - 1, dtype=bool)
            top_idx = np.argsort(c)[::-1][:3 * n]
            active[top_idx] = True
            n_active = 3 * n

        m = n_active
        s_vec = active.astype(np.float64)

        # Gradient: g[j] = (2/m) * sum_{k in S} a[k-j] (for valid k-j)
        g = _hess_matvec(a, s_vec, m, n)

        # Center gradient (remove mean for constraint sum(a)=1)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Solve H * u = g_t using CG
        def hess_mv(v):
            return _hess_matvec(v, s_vec, m, n) + 1e-10 * v

        u = _cg_solve(hess_mv, g_t, n, max_iter=150, tol=1e-10, reg=1e-10)

        # Solve H * w = 1 for the constraint
        ones = np.ones(n)
        w = _cg_solve(hess_mv, ones, n, max_iter=100, tol=1e-10, reg=1e-10)
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-15:
            dx = -u
            dx -= np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx -= np.mean(dx)

        # Ensure non-negativity: compute max step
        neg_mask = dx < -1e-12
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        # Line search
        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(30):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent
            t = min(0.05, t_max)
            improved = False
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    improved = True
                    break
                t *= 0.5
            if not improved:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12:
            break

    return best_a, best_sc


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with log-sum-exp smoothing and beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap_idx = min(20, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[gap_idx]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001, 0.00005, 0.00002,
                                      0.00001, 0.000005])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                      0.015, 0.007, 0.003, 0.001, 0.0005,
                                      0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.002
        elif n <= 8192:
            est_time_per_iter = 0.005
        elif n <= 16384:
            est_time_per_iter = 0.012
        else:
            est_time_per_iter = 0.025 * (n / 32768.0)

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            obj_fn = lambda z_local: _lsobj(z_local, beta, n)
            grad_fn = lambda z_local: _lsgrad(z_local, beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_refine(a_init, deadline, max_steps=5000, lr_init=0.02):
    """Subgradient method with adaptive learning rate, targeting top peaks."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top-k peaks
        k = min(15, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / k

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 15 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.6

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="ramp", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "ramp_steep":
        t = np.linspace(0, 1, n)
        a = 0.01 + 0.99 * t ** 2
        return a / a.sum()
    elif kind == "ramp_cube":
        t = np.linspace(0, 1, n)
        a = 0.01 + 0.99 * t ** 3
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.2) ** 2) / 0.01) + 0.5 * np.exp(-((t - 0.8) ** 2) / 0.01)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * t
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "concave":
        t = np.linspace(0, 1, n)
        a = np.sqrt(t + 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _peak_targeted_step(a, deadline, max_rounds=80):
    """Target the single highest peak specifically."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Gradient from single peak
        w = np.zeros(2 * n - 1)
        w[peak_idx] = 1.0
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = c[peak_idx]
        grad = 2.0 * a * (h - W)

        # Line search
        f_old = c_max
        improved = False
        for step_mult in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_try = a - step_mult * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                improved = True
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break

        if not improved or abs(f_try - f_old) < 1e-13:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Active-set Newton from parent (FFT-based Hessian)
    2. L-BFGS with log-sum-exp smoothing (beta continuation)
    3. Multi-n exploration with interpolation
    4. Population-based perturbation + local search
    5. Final tight active-set polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "ramp")
        sc_parent = _score(parent)
        print(f"No parent, using ramp n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init_ramp")

    # Phase 1: Active-set Newton on parent
    print(f"\n[Phase 1] Active-set Newton on parent (t={time.monotonic() - t_start:.1f}s)")
    time_budget_1 = min(300, (deadline - t_start) * 0.2)
    a_as, sc_as = _active_set_newton(parent, t_start + time_budget_1, max_iter=150, tol_active=0.002)
    update_best(a_as, sc_as, "as_newton_parent")

    # Phase 2: L-BFGS with aggressive beta continuation
    print(f"\n[Phase 2] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    time_budget_2 = min(400, (deadline - t_start) * 0.25)
    a_lbfgs = _run_lbfgs(parent, t_start + time_budget_2, min_time_remaining=100.0, aggressive=True)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_main")

    # Active-set on the L-BFGS result
    if time.monotonic() < deadline - 60:
        a_as2, sc_as2 = _active_set_newton(a_lbfgs, min(deadline - 30, time.monotonic() + 60),
                                           max_iter=100, tol_active=0.001)
        update_best(a_as2, sc_as2, "as_after_lbfgs")

    # Phase 3: Multi-n exploration
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        for n_try in [8192, 16384, 32768]:
            if time.monotonic() > deadline - 120:
                break
            if n_try == n_parent:
                continue
            a_interp = _interpolate_to_n(best_seq, n_try)
            sc_interp = _score(a_interp)
            # Quick L-BFGS
            time_left = deadline - time.monotonic() - 30
            if time_left < 30:
                break
            a_opt = _run_lbfgs(a_interp, t_start + time_left, min_time_remaining=20, aggressive=False)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n{n_try}_lbfgs")
            # Quick active-set
            if time.monotonic() < deadline - 20:
                a_as3, sc_as3 = _active_set_newton(a_opt, min(deadline - 10, time.monotonic() + 30),
                                                   max_iter=50, tol_active=0.002)
                update_best(a_as3, sc_as3, f"n{n_try}_as")

    # Phase 4: Structured initializations at best n
    print(f"\n[Phase 4] Structured initializations (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else n_parent
    profiles = ["ramp_steep", "exp_ramp", "ramp_cube", "concave", "sine_mod"]
    for kind in profiles:
        if time.monotonic() > deadline - 60:
            break
        a_struct = _make_profile(n_best, kind, rng)
        sc_struct = _score(a_struct)
        if sc_struct < best_score + 0.05:
            time_left = deadline - time.monotonic() - 20
            if time_left > 20:
                a_opt = _run_lbfgs(a_struct, t_start + time_left, min_time_remaining=10, aggressive=False)
                update_best(a_opt, _score(a_opt), f"struct_{kind}")

    # Phase 5: Population-based perturbation + local search
    print(f"\n[Phase 5] Population-based search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            sigma = 0.01 * (1 + i * 0.5)
            a_pert = _perturb(best_seq, sigma, rng)
            sc_pert = _score(a_pert)
            time_left = deadline - time.monotonic() - 15
            if time_left < 10:
                break
            # Quick subgradient refinement
            a_sg, sc_sg = _subgrad_refine(a_pert, t_start + time_left, max_steps=1000, lr_init=0.01)
            update_best(a_sg, sc_sg, f"perturb_{i}_sg")
            # Quick active-set
            if time.monotonic() < deadline - 10:
                a_as4, sc_as4 = _active_set_newton(a_sg, min(deadline - 5, time.monotonic() + 20),
                                                   max_iter=40, tol_active=0.002)
                update_best(a_as4, sc_as4, f"perturb_{i}_as")

    # Phase 6: Peak-targeted refinement
    print(f"\n[Phase 6] Peak-targeted refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pt, sc_pt = _peak_targeted_step(best_seq, min(deadline - 10, time.monotonic() + 30), max_rounds=60)
        update_best(a_pt, sc_pt, "peak_target")

    # Phase 7: Final tight active-set Newton
    print(f"\n[Phase 7] Final active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Very tight tolerance
        a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 5, time.monotonic() + 40),
                                           max_iter=100, tol_active=0.0005)
        update_best(a_fin, sc_fin, "final_as_tight0.0005")

        # Even tighter
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_newton(best_seq, deadline - 2,
                                                 max_iter=80, tol_active=0.0001)
            update_best(a_fin2, sc_fin2, "final_as_tight0.0001")

    # Phase 8: Final subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin3, sc_fin3 = _subgrad_refine(best_seq, deadline - 2, max_steps=1000, lr_init=0.003)
        update_best(a_fin3, sc_fin3, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "ramp")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "ramp")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
