def _fista_subgrad(a_init, n_iters, deadline, min_time_remaining=30.0):

    n = len(a_init)

    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    step = 1.0 / (2.0 * a.max() + 1e-10)  # Lipschitz constant estimate
    
    # We'll use a fixed step or adaptive step
    # The function f(a) = max_k c[k] has Lipschitz gradient (where differentiable) bounded by
    # L = max_k ||dc[k]/da||_inf = max_k max_j 2*|a[k-j]| <= 2*max(a)
    # So step <= 1/L = 1/(2*max(a))
    
    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Evaluate at y
        c_y = _full_convolve(y, y)
        c_max_y = c_y.max()
        active = c_y >= c_max_y - 1e-10 * c_max_y
        n_active = int(active.sum())
        if n_active == 0:
            active_idx = np.array([int(np.argmax(c_y))])
        else:
            active_idx = np.where(active)[0]
        
        # Subgradient at y
        g = _peak_subgrad_vectorized(y, n, active_idx)

        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t_old - 1.0) / t

        z = y - step * g
        # Project onto simplex
        x_new = _project_simplex(z)
        
        # Check improvement
        sc_new = _score_normalized(x_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = x_new.copy()

        y = _project_simplex(y)  # Keep y in the feasible set

        def grad(z_val):
            a_val = _softmax(z_val)
            c_val = _full_convolve(a_val, a_val)
            c_max = float(c_val.max())
            shifted = (c_val - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros(len(c_val))
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s
            a_rev = a_val[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_val))
            return 4.0 * n * a_val * (h - W)
        
        try:
            result = minimize(fun, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score_normalized(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
        except Exception:
            pass
    
    return best_a, best_sc


def _spectral_shaping(a_init, n_iters, deadline, min_time_remaining=30.0):
    """Fienup-style spectral shaping to flatten the power spectrum."""
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    N = _next_pow2(2 * n)  # FFT size
    
    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Compute FFT
        A = rfft(a, N)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum (geometric mean of power)
        log_power = np.log(power + 1e-30)
        target_log = np.mean(log_power)
        target_power = np.exp(target_log)
        
        # Update: gradually flatten the spectrum
        # New amplitude: blend current with target
        t_blend = 0.3  # blending factor
        current_amp = np.sqrt(power)
        target_amp = np.sqrt(target_power) * current_amp / (np.mean(current_amp) + 1e-30)
        new_amp = (1 - t_blend) * current_amp + t_blend * target_amp
        
        # Preserve phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = new_amp * phase
        
        # Inverse FFT
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc_new = _score_normalized(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Blend with previous to avoid oscillation
        a = 0.7 * a + 0.3 * a_new
        a = _project_simplex(a)
    
    return best_a, best_sc


def _kkt_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-guided refinement: adjust active variables to equalize 
    the subgradient, and determine which variables should be at 0."""
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)
    max_iter = 500
    
    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 1e-6 * c_max
        active_peak = c >= c_max - tol
        n_active = int(active_peak.sum())
        if n_active == 0:
            active_idx = np.array([int(np.argmax(c))])
        else:
            active_idx = np.where(active_peak)[0]
        
        # Compute subgradient
        g = _peak_subgrad_vectorized(a, n, active_idx)
        
        # KKT condition: for j where a[j] > 0, g[j] should be equal to mu
        # For j where a[j] = 0, g[j] should be >= mu
        active_vars = a > 1e-12
        if active_vars.sum() < 2:
            break
        
        mu = np.mean(g[active_vars])
        residual = g[active_vars] - mu
        
        # Variance of the subgradient over active variables
        var = np.var(residual)
        if var < 1e-12:
            break
        
        # Move variables to reduce variance
        # Variables with g[j] > mu should be decreased
        # Variables with g[j] < mu should be increased
        # (since increasing a[j] when g[j] < mu reduces the peak faster)
        
        # Small adjustment
        lr = 0.01 / (np.max(np.abs(g)) + 1e-10)
        delta = -lr * (g - mu)
        delta = np.where(active_vars, delta, 0.0)
        # Don't let variables go negative
        delta = np.where(a > 0, np.maximum(delta, -a * 0.5), 0.0)
        
        a_try = a + delta
        a_try = np.maximum(a_try, 0.0)
        a_try = _project_simplex(a_try)
        
        sc_try = _score_normalized(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            lr *= 0.5
            delta = -lr * (g - mu)
            delta = np.where(active_vars, delta, 0.0)
            delta = np.where(a > 0, np.maximum(delta, -a * 0.5), 0.0)
            a_try = a + delta
            a_try = np.maximum(a_try, 0.0)
            a_try = _project_simplex(a_try)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            else:
                # No improvement, try to move some variables to 0
                # Variables with smallest a and largest g should go to 0
                if it % 50 == 49:
                    candidates = np.where(a < 1e-4 * a.max())[0]
                    if len(candidates) > 0:
                        # Sort by g (largest g first - these contribute most to peak per unit)
                        candidates = candidates[np.argsort(g[candidates])[::-1]]
                        n_remove = max(1, len(candidates) // 10)
                        for idx in candidates[:n_remove]:
                            a[idx] = 0.0
                        a = _project_simplex(a)
    
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "sine_power":
        t = np.linspace(0, 1, n)
        p = 2.0
        a = (np.sin(np.pi * t)) ** p
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.standard_normal(n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
