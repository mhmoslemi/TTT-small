"""
Refined optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: adaptive log-sum-exp L-BFGS, exact active-set KKT solver,
adaptive beta continuation, and multi-start with structured perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):
    """Compute the correlation: for each position i in a, sum over k where A_vec[k]=1 of a_contributing."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _ls_obj(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _ls_grad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _cg_solve(matvec, rhs, max_cg=500, tol=1e-13):
    """Conjugate gradient solver."""
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _kkt_solve(a_init, active_mask, n, max_iter=200):
    """
    Solve KKT conditions for the active set.
    At optimum: gradient of L = 0 where L = sum_{k in active} lambda_k * ((a*a)[k] - M) + mu * (sum a - 1)
    This gives: 2 * sum_{k in active} lambda_k * (C_k * a) + mu * 1 = 0
    And: sum a = 1, a >= 0
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    active = active_mask.astype(np.float64)
    m = int(active.sum())
    
    for it in range(max_iter):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Weight by proximity to max (soft active set)
        tol = 0.001 * c_max
        weights = np.zeros_like(c)
        weights[active] = 1.0 / m
        
        # Gradient of the objective (minimize max, approximated by weighted sum)
        g = 2.0 * _corr_a_A(a, weights, n)
        g_t = g - np.mean(g)  # Project onto sum=0 subspace

        # H*v = 2 * (C_weights * v) where C_weights*v = _corr_a_A(v, weights, n)
        # Plus regularization for positive definiteness
        mu = 1e-8 * c_max
        
        def hess_vp(v):
            return 2.0 * _corr_a_A(v, weights, n) + mu * v
        
        # Solve H * dx = -g_t
        u = _cg_solve(hess_vp, -g_t, max_cg=100)
        
        # Solve H * w_vec = ones for the normalization constraint
        w_vec = _cg_solve(hess_vp, np.ones(n), max_cg=100)
        w_sum = float(w_vec.sum())
        
        if abs(w_sum) < 1e-15:
            dx = u
        else:
            lam = float(u.sum()) / w_sum
            dx = u - lam * w_vec

        t = 1.0

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr(a, A_vec, n):
    """Compute gradient-related correlation: (a * A_vec) restricted to valid indices."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _ls_obj_wrapped(z, beta, n):
    return _ls_obj(z, beta, n)


def _ls_grad_wrapped(z, beta, n):
    return _ls_grad(z, beta, n)


def _cg_solve(matvec_fn, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            x = x_new
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
        x = x_new
    return x


class _HessContext:
    """Context for Hessian-vector products in active-set Newton."""
    def __init__(self, A_vec, n, mu):
        self.A_vec = A_vec
        self.n = n
        self.mu = mu

    def matvec(self, v):
        return (2.0 / self.A_vec.sum()) * _corr(v, self.A_vec, self.n) + self.mu * v


def _active_set_newton_precise(a_init, deadline, max_outer=80):

    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Decreasing tolerance schedule
    tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003, 0.0002,
                    0.0001, 0.00005, 0.00003, 0.00002, 0.00001, 0.000005]
    tol_idx = 0

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Determine active set with current tolerance
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        if not active.any():
            active[int(np.argmax(c))] = True
        m = int(active.sum())

        if m < 2:
            top2 = np.argsort(c)[::-1][:2]
            active = np.zeros_like(c, dtype=bool)
            active[top2] = True
            m = 2

        if m > 8 * n:
            # Too many active; use top 8n
            top_idx = np.argsort(c)[::-1][:8 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            m = 8 * n

        A_vec = active.astype(np.float64)
        mu = 1e-9 * (c_max / max(m, 1)) + 1e-12

        # Compute gradient projected onto tangent space (sum=0)
        g = (2.0 / m) * _corr(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)

        if gnorm < 1e-15:
            # Converged
            break

        # Newton step via CG
        hc = _HessContext(A_vec, n, mu)
        u = _cg_solve(hc.matvec, -g_t, max_cg=150, tol=1e-13)
        w_vec = _cg_solve(hc.matvec, np.ones(n), max_cg=150, tol=1e-13)
        w_sum = float(w_vec.sum())

        if abs(w_sum) < 1e-15:
            dx = u
        else:
            lam = float(u.sum()) / w_sum
            dx = u - lam * w_vec

        # Ensure dx is in tangent space
        dx = dx - np.mean(dx)

        f_old = c_max
        accepted = False

            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)

                continue

            # Sufficient decrease

                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try smaller fixed step
            for t_try in [0.1, 0.05, 0.01, 0.005]:
                a_try = a + t_try * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < f_old:
                    a = a_try
                    break
            else:
                # Stagnation: tighten tolerance and continue
                if tol_idx < len(tol_schedule) - 1:
                    tol_idx += 1
                continue

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-14:
            # Try tightening tolerance
            if tol_idx < len(tol_schedule) - 1:

            else:

def _multi_beta_lbfgs(a_init, deadline, min_remaining=20.0):
    """
    Multi-stage L-BFGS with geometrically decreasing beta.
    Adaptively allocates iterations based on remaining time.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Determine initial beta from correlation spread
    c_sorted = np.sort(c_cur)[::-1]
    k = min(100, len(c_sorted))
    gap = c_sorted[0] - c_sorted[k - 1] if k > 1 else c_sorted[0]
    initial_beta = max(gap * 0.3, max_c * 0.01)

    # Create fine schedule: 30 stages from initial_beta down to very small
    n_stages = 30
    ratios = np.geomspace(1.0, 1e-6, n_stages)
    betas = initial_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration
        est_tpi = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(10000, max(30, int(remaining / (est_tpi * 4))))

        try:
            result = minimize(
                _ls_obj_wrapped,
                z,
                jac=_ls_grad_wrapped,
                method='L-BFGS-B',
                args=(beta, n),
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            improved = sc_new < current_sc - 1e-15
            if improved:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0 or improved:
                print(f"    stage {i}/{n_stages} beta={beta:.2e} score={current_sc:.10f} {'***' if improved else ''}")
        except Exception as e:
            print(f"    stage {i} error: {e}")

    return current_a

def _interpolate(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

            old = best_score
            best_seq = arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(arr)}) [{label}] improve={old-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_p = float(parent_raw.sum())
            if s_p > 0.01:
                parent = parent_raw / s_p
    except Exception:
        pass

    if parent is not None:
        sc_p = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.10f}")
        _try_update(parent, "parent")
        n_main = len(parent)
    else:
        n_main = 16384
        parent = np.ones(n_main) / n_main
        _try_update(parent, "uniform_init")

    # === Main pipeline on parent ===
    print(f"\n=== Main pipeline (n={n_main}) t={time.monotonic()-t_start:.1f}s ===")
    a_main, sc_main = _run_pipeline(parent, deadline, min_remaining=180, label="main")
    _try_update(a_main, "main_pipeline")

    # === Try n = 2*n_main ===
    n_alt = 2 * n_main
    if n_alt <= 65536 and time.monotonic() < deadline - 120:
        print(f"\n=== Alternative n={n_alt} t={time.monotonic()-t_start:.1f}s ===")
        a_alt = _interpolate(parent, n_alt)
        a_alt_sc = _score(a_alt)
        print(f"  Interpolated to n={n_alt}: score={a_alt_sc:.10f}")
        a_alt_opt, sc_alt = _run_pipeline(a_alt, deadline, min_remaining=60, label=f"n{n_alt}")
        _try_update(a_alt_opt, f"n{n_alt}_pipeline")

    # === Diverse restarts from best ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Restarts t={time.monotonic()-t_start:.1f}s ===")
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _run_pipeline(p_start, deadline, min_remaining=10, label=f"perturb_{sigma}")
            _try_update(a_r, f"perturb_{sigma}")

    # === Final ultra-precise polish ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish t={time.monotonic()-t_start:.1f}s ===")
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())

        # Very small beta L-BFGS
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        for beta_f in [mb * 2e-5, mb * 1e-5, mb * 5e-6, mb * 2e-6, mb * 1e-6]:
            if time.monotonic() > deadline - 3:
                break
            remaining = deadline - 3 - time.monotonic()
            if remaining < 2:
                break
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(
                    _ls_obj_wrapped,
                    z,
                    jac=_ls_grad_wrapped,
                    method='L-BFGS-B',
                    args=(beta_f, n),
                    options={'maxiter': max_iter, 'ftol': 1e-19, 'gtol': 1e-15, 'maxcor': 100, 'maxls': 300}
                )
                a_fin = _softmax(result.x)
                _try_update(a_fin, f"beta_{beta_f:.0e}")
                z = np.log(np.maximum(a_fin, 1e-30))
                z = z - z.mean()
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 2:
            a_fin2 = _active_set_newton_precise(best_seq, deadline - 1, max_outer=30)
            _try_update(a_fin2, "final_newton")

    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
