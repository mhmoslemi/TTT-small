This is interesting! The update direction is a "blurred" version of a itself (superposition of reflected copies). This means the method is essentially trying to make a "self-smoothing" - it's pushing a toward a state where superpositions of its reflected copies are as flat as possible.

    """Fine active-set refinement with exact line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_unit(a)
    
    lr = 0.01
    tol_frac = 0.001
    min_tol = 0.00001
    stall = 0
    max_stall = 200
    
    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            a = best_a.copy()
            lr *= 0.7
            tol_frac = max(min_tol, tol_frac * 0.5)
            stall = 0
            if lr < 0.0001:
                break
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_act = int(active.sum())
        if n_act < 1:
            active[np.argmax(c)] = True
            n_act = 1
        active_idx = np.where(active)[0]
        
        g = _grad_at_peaks(a, active_idx)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break
        
        # Line search
        improved = False
        alpha = min(lr, 1.0 / gnorm)
        for _ in range(50):
            a_try = a - alpha * g_t
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            if tol_frac > min_tol:
                tol_frac *= 0.99
            lr = min(0.02, lr * 1.01)
        else:
            lr *= 0.8
    
    return best_a, _score_unit(best_a)

def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine5":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine7":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.sin(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_like":
        t = np.linspace(-3, 3, n)
        a = np.sinc(t) ** 2 + 1e-10
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-2, 2, n)
        a = np.exp(-t**2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n
