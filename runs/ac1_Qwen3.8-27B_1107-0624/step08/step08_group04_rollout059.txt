"""
Multi-phase optimizer minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FISTA with exact subgradient, active-set QP via CG, L-BFGS with
adaptive smoothing, multi-resolution refinement, and adaptive active-set
sizing. Key improvements: (1) FISTA with true subgradient of the max
function, (2) multi-resolution warm-start, (3) adaptive active-set width,
(4) n=32768 exploration, (5) primal-dual alternating refinement.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = rfft(V, N)
    F_W = rfft(W, N)
    conv = irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-13):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _fista_optimize(a_init, deadline, max_iters=5000, step=None,
                    active_tol=0.001, momentum_decay=False):
    """FISTA with exact subgradient of f(a)=max_k conv(a,a)[k]."""
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_cmax = float(c.max())
    best_sc = _score(a)

    if step is None:
        step = 0.8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Adaptive active set: entries within active_tol * c_max of the peak
        threshold = c_max - active_tol * c_max
        active = c >= threshold
        if active.sum() < 1:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Subgradient: g = 2 * H_W * y
        g = _hankel_matvec(y, W, n)

        # FISTA step
        a_new = _project_simplex(y - step * g)

        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y > 1e-30:
            y_new = y_new / s_y
        else:
            y_new = a_new.copy()

        a = a_new
        y = y_new
        t = t_new

        # Check improvement
        if it % 20 == 0 or it < 5:
            c_check = _full_convolve(a, a)
            cmax_check = float(c_check.max())
            if cmax_check < best_cmax:
                best_cmax = cmax_check
                best_a = a.copy()
                best_sc = _score(a)

        # Adaptive step: increase if we're making progress slowly
        if it % 100 == 0 and active_tol > 1e-6:
            active_tol *= 0.7

    c_final = _full_convolve(a, a)
    if float(c_final.max()) < best_cmax:
        best_a = a.copy()
        best_sc = _score(a)

    return best_a, best_sc


def _active_set_qp(W, n, a_init, max_outer=50, cg_iter=400):
    """Solve min a^T H_W a s.t. sum(a)=1, a>=0 via active-set."""
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) / 2.0)

    for outer in range(max_outer):
        support = a > 1e-12
        n_sup = int(support.sum())
        if n_sup == 0:
            break
        if n_sup == n:
            y = _cg_solve(W, n, np.ones(n), max_iter=cg_iter)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            grad = _hankel_matvec(a, W, n)
            lam = grad[support].mean()
            # Projected gradient step
            step_size = 1.0 / (np.max(np.abs(grad)) + 1e-15)
            step_size = min(step_size, 0.5)
            a_new = a - step_size * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

        val_new = float(a_new @ _hankel_matvec(a_new, W, n) / 2.0)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

        # Convergence check
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-12
        if support.any():
            lam = grad[support].mean()
            resid = np.max(np.abs(grad[support] - lam))
            off_s = ~support
            if off_s.any():
                resid = max(resid, np.max(lam - grad[off_s]))
            if resid < 1e-9 * (np.max(np.abs(grad)) + 1e-15):
                break

    return best_a


def _newton_polish(a_init, deadline, max_rounds=80):
    """Active-set QP with progressive tightening."""
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001]
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        tol = tol_schedule[min((rnd // rounds_per_tol), len(tol_schedule) - 1)]

        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        a_qp = _active_set_qp(W, n, a, max_outer=40, cg_iter=300)
        sc_qp = _score(a_qp)
        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped Newton
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-15:
                Hg = _cg_solve(W, n, g, max_iter=200)
                H1 = _cg_solve(W, n, np.ones(n), max_iter=200)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20:
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)
                alpha = 1.0
                for ls in range(30):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max:
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        break
                    alpha *= 0.5
                    if alpha < 1e-14:
                        break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _lbfgs_smooth(a_init, deadline, aggressive=True):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    if aggressive:
        ratios = np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                           0.05, 0.03, 0.02, 0.012, 0.007, 0.004,
                           0.002, 0.001, 0.0005, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
                           0.0075, 0.004, 0.002, 0.001, 0.0005, 0.00025,
                           0.0001, 0.00005, 0.00002, 0.00001])

    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 15))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + t
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _flatten_polish(a_init, deadline, max_iters=200):

            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        # Tighten active set gradually
        if it % 200 == 0:
            active_tol *= 0.6
            if active_tol < 1e-8:
                active_tol = 1e-8

        # Adjust step size
        if it % 500 == 0 and it > 0:
            step *= 0.9

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining FISTA, active-set QP, L-BFGS smoothing,
    and flattening. Uses multi-resolution warm-start and adaptive active sets.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: FISTA with exact subgradient (broad active set)
    print(f"\n[Phase 1] FISTA broad (t={time.monotonic() - t_start:.1f}s)")
    d1 = min(deadline - 800, time.monotonic() + 400)
    a1, sc1 = _fista_optimize(parent, d1, max_iters=3000, step=0.6,
                               active_tol=0.01)
    update_best(a1, sc1, "fista_broad")

    # Phase 2: Active-set QP polish
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    d2 = min(deadline - 600, time.monotonic() + 350)
    a2, sc2 = _newton_polish(best_seq, d2, max_rounds=60)
    update_best(a2, sc2, "qp_1")

    # Phase 3: L-BFGS with smoothing
    print(f"\n[Phase 3] L-BFGS smooth (t={time.monotonic() - t_start:.1f}s)")
    d3 = min(deadline - 400, time.monotonic() + 300)
    a3 = _lbfgs_smooth(best_seq, d3, aggressive=True)
    update_best(a3, _score(a3), "lbfgs_1")

    # Phase 4: FISTA with tighter active set
    print(f"\n[Phase 4] FISTA tight (t={time.monotonic() - t_start:.1f}s)")
    d4 = min(deadline - 300, time.monotonic() + 250)
    a4, sc4 = _fista_tight(best_seq, d4, max_iters=4000, start_tol=0.001)
    update_best(a4, sc4, "fista_tight_1")

    # Phase 5: Another QP round
    print(f"\n[Phase 5] QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    d5 = min(deadline - 200, time.monotonic() + 150)
    a5, sc5 = _newton_polish(best_seq, d5, max_rounds=70)
    update_best(a5, sc5, "qp_2")

    # Phase 6: Flatten polish
    print(f"\n[Phase 6] Flatten (t={time.monotonic() - t_start:.1f}s)")
    d6 = min(deadline - 150, time.monotonic() + 80)
    a6, sc6 = _flatten_polish(best_seq, d6, max_iters=150)
    update_best(a6, sc6, "flatten_1")

    # Phase 7: Final FISTA ultra-tight
    print(f"\n[Phase 7] FISTA ultra-tight (t={time.monotonic() - t_start:.1f}s)")
    d7 = min(deadline - 100, time.monotonic() + 100)
    a7, sc7 = _fista_tight(best_seq, d7, max_iters=5000, start_tol=0.0001)
    update_best(a7, sc7, "fista_ultra")

    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [32768, 8192]:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try)
        # Quick FISTA at new n
        d8 = min(deadline - 60, time.monotonic() + 50)
        a8, sc8 = _fista_optimize(a_init, d8, max_iters=2000, step=0.5,
                                   active_tol=0.005)
        update_best(a8, sc8, f"fista_n{n_try}")

        # Quick QP
        if time.monotonic() < deadline - 30:
            d8b = min(deadline - 25, time.monotonic() + 25)
            a8b, sc8b = _newton_polish(a8, d8b, max_rounds=30)
            update_best(a8b, sc8b, f"qp_n{n_try}")

    # Phase 9: Restarts with different profiles
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        n_cur = len(best_seq)
        for prof in ["ramp", "exp_decay", "cos"]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _make_profile(n_cur, prof)
            a_r = _lbfgs_smooth(p_start, min(deadline - 10, time.monotonic() + 15))
            update_best(a_r, _score(a_r), f"restart_{prof}")

    # Phase 10: Final combined polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Ultra-tight FISTA
        d10 = deadline - 3
        a10, sc10 = _fista_tight(best_seq, d10, max_iters=3000, start_tol=5e-6)
        update_best(a10, sc10, "final_fista")

        # Quick QP
        if time.monotonic() < deadline - 3:
            a10b, sc10b = _newton_polish(best_seq, deadline - 2, max_rounds=30)
            update_best(a10b, sc10b, "final_qp")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
