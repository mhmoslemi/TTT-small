Iteratively Reweighted QP optimizer with p-norm continuation,
symmetry exploitation, FFT-based Hankel matvec, preconditioned CG,
and multi-n multi-init exploration.
Key improvements over prior: (1) p-norm reweighting instead of fixed-tolerance
active set, (2) symmetric optimization mode, (3) diagonal preconditioned CG,
(4) n=32768 exploration, (5) flat-top window initialization.

import numpy as np
import time
from scipy.optimize import minimize

T_START = 0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_mv(w, v, n):
    """Compute H*v where H[i,j] = w[i+j], w has length 2n-1, v has length n."""
    # H*v[i] = sum_j w[i+j] * v[j]
    # This is a "anti-diagonal" convolution. Use FFT trick:
    # If we reverse v: V = v[::-1], then (conv(V, w))[n-1+i] = sum_j V[j] * w[n-1+i-j]
    # = sum_j v[n-1-j] * w[n-1+i-j]
    # Let k = n-1-j: = sum_k v[k] * w[i+k]  ... wait that's not right.
    
    # Actually: (V * w)[m] = sum_j V[j] * w[m-j] = sum_j v[n-1-j] * w[m-j]
    # Let k = n-1-j: j = n-1-k, m-j = m-n+1+k
    # = sum_k v[k] * w[m-n+1+k]
    # For m = n-1+i: = sum_k v[k] * w[i+k]
    # So (V * w)[n-1+i] = sum_k v[k] * w[i+k] = (H*v)[i]
    
    V = v[::-1].copy()
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n-1:2*n-1]

def _pcg_solve(w, rhs, n, max_iter=500, tol=1e-12):
    """Preconditioned CG to solve H*x = rhs where H[i,j] = w[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    # Diagonal preconditioner: D[i] = w[2*i]
    D = w[0::2]  # w[0], w[2], w[4], ..., w[2*(n-1)]
    # Make sure D is bounded away from zero
    D = np.maximum(D, 1e-30)
    z = r / D
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    
    for i in range(max_iter):
        Hp = _hankel_mv(w, p, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        z = r / D
        rz_new = float(r @ z)
        if not np.isfinite(rz_new):
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        beta = rz_new / rz
        p = z + beta * p
        rz = rz_new
    
    return x

def _cg_solve_unprecond(w, rhs, n, max_iter=500, tol=1e-12):
    """Basic CG to solve H*x = rhs where H[i,j] = w[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    
    for i in range(max_iter):
        Hp = _hankel_mv(w, p, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        beta = rs_new / rs
        p = r + beta * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    
    return x

def _solve_qp_equality(w, n):
    """Solve min a^T H a s.t. 1^T a = 1 (no inequality constraints).
    Returns a (length n) with sum 1."""
    # H a = lambda * 1, 1^T a = 1
    # a = lambda * H^{-1} 1, lambda = 1 / (1^T H^{-1} 1)
    y = _pcg_solve(w, np.ones(n), n, max_iter=800, tol=1e-13)
    oney = float(np.sum(y))
    if abs(oney) < 1e-20 or not np.isfinite(oney):
        return np.ones(n) / n
    a = y / oney
    # Clamp tiny negatives
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s


def _hankel_mv(w, v, n):
    """Compute H*v where H[i,j] = w[i+j], w has length 2n-1, v has length n."""
    V = v[::-1].copy()
    out_len = n + len(w) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _pcg_solve(w, rhs, n, max_iter=800, tol=1e-13):
    """Preconditioned CG to solve H*x = rhs where H[i,j] = w[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    D = w[0::2].copy()
    D = np.maximum(D, 1e-30)
    z = r / D
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_mv(w, p, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        z = r / D
        rz_new = float(r @ z)
        if not np.isfinite(rz_new):
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        beta = rz_new / rz
        p = z + beta * p
        rz = rz_new
    return x

    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flattop":
        # Flat-top window shape: 1 - a1*cos(2*pi*t) + a2*cos(4*pi*t) - a3*cos(6*pi*t)
        t = np.linspace(-1, 1, n)
        a1, a2, a3 = 0.21557895, 0.41663158, 0.277263158
        a = 1 - a1 * np.cos(np.pi * t) + a2 * np.cos(2 * np.pi * t) - a3 * np.cos(3 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sech":
        t = np.linspace(0, 1, n)
        a = 1.0 / np.cosh(3.0 * (t - 0.5)) + 0.005
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining IRW-QP, active-set polish, L-BFGS,
    flattening, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}", flush=True)
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}", flush=True)
        update_best(parent, _score(parent), "init")

    # --- Phase 1: IRW-QP at parent n ---
    print(f"\n[Phase 1] IRW-QP at n={len(best_seq)} (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p_schedule_1 = [2, 4, 8, 16, 32, 64, 128, 256, 512]
    a1 = _irw_qp_optimize(best_seq, p_schedule_1, deadline, min_time_remaining=600)
    update_best(a1, _score(a1), "irw_qp_1")

    # --- Phase 2: Active-set polish ---
    print(f"\n[Phase 2] Active-set polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    a2 = _active_set_polish(best_seq, deadline, min_time_remaining=500)
    update_best(a2, _score(a2), "active_set_1")

    # --- Phase 3: L-BFGS ---
    print(f"\n[Phase 3] L-BFGS (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 400:
        a3 = _run_lbfgs(best_seq, deadline, min_time_remaining=350)
        update_best(a3, _score(a3), "lbfgs_1")

    # --- Phase 4: Second IRW-QP round (higher p) ---
    print(f"\n[Phase 4] IRW-QP round 2 (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 350:
        p_schedule_2 = [64, 128, 256, 512, 1024, 2048, 4096]
        a4 = _irw_qp_optimize(best_seq, p_schedule_2, deadline, min_time_remaining=300)
        update_best(a4, _score(a4), "irw_qp_2")

    # --- Phase 5: Flatten polish ---
    print(f"\n[Phase 5] Flatten polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 280:
        a5 = _flatten_polish(best_seq, min(deadline - 250, time.monotonic() + 80))
        update_best(a5, _score(a5), "flatten_1")

    # --- Phase 6: Final active-set polish (tighter) ---
    print(f"\n[Phase 6] Active-set polish tight (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 200:
        a6 = _active_set_polish(best_seq, deadline, min_time_remaining=150,
                                tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                                          0.00002, 0.00001, 0.000005, 0.000002,
                                          0.000001, 0.0000005, 0.0000002, 0.0000001])
        update_best(a6, _score(a6), "active_set_2")

    # --- Phase 7: Large n exploration ---
    print(f"\n[Phase 7] Large n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    large_n_list = [32768, 16384]
    for n_try in large_n_list:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        print(f"  Trying n={n_try}...", flush=True)
        a_init = _resample(best_seq, n_try)
        # Quick IRW-QP
        p_sched = [4, 16, 64, 256, 1024]
        a_ln = _irw_qp_optimize(a_init, p_sched, deadline, min_time_remaining=80)
        update_best(a_ln, _score(a_ln), f"n_{n_try}_irw")
        # Active set polish
        if time.monotonic() < deadline - 60:
            a_lp = _active_set_polish(a_ln, deadline, min_time_remaining=40,
                                      tol_list=[0.005, 0.002, 0.001, 0.0005,
                                                0.0002, 0.0001, 0.00005, 0.00001])
            update_best(a_lp, _score(a_lp), f"n_{n_try}_qp")

    # --- Phase 8: Diverse restarts ---
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 60:
        profiles = ["flattop", "gaussian", "sech", "cos"]
        n_r = len(best_seq)
        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _make_profile(n_r, prof)
            elif i == 1:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 2:
                p_start = _make_profile(n_r, prof)
            else:
                p_shrunk = best_seq.copy()
                idx = np.argsort(p_shrunk)
                p_shrunk[idx[:len(idx) // 4]] = 0
                p_shrunk = p_shrunk / (p_shrunk.sum() + 1e-30)
                p_start = _perturb(p_shrunk, 0.08, rng)

            a_r = _irw_qp_optimize(p_start, [4, 16, 64, 256], deadline, min_time_remaining=25)
            update_best(a_r, _score(a_r), f"restart_{prof}")
            if time.monotonic() < deadline - 20:
                a_rp = _active_set_polish(a_r, deadline, min_time_remaining=10,
                                          tol_list=[0.002, 0.001, 0.0005, 0.0001])
                update_best(a_rp, _score(a_rp), f"restart_{prof}_qp")

    # --- Phase 9: Final ultra-tight polish ---
    print(f"\n[Phase 9] Final polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _active_set_polish(best_seq, deadline - 2, min_time_remaining=2,
                                   tol_list=[0.00001, 0.000005, 0.000002, 0.000001,
                                             0.0000005, 0.0000002, 0.0000001])
        update_best(a_fin, _score(a_fin), "final_qp")

        if time.monotonic() < deadline - 3:
            a_fl = _flatten_polish(best_seq, deadline - 2, max_iters=80)
            update_best(a_fl, _score(a_fl), "final_flatten")

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL SCORE: {sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK SCORE: {_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
