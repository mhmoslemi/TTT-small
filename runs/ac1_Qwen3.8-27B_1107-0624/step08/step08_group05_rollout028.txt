def _next_pow2(n):
    k = 4
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _cross_corr(w, a):
    """Compute g[i] = sum_j w[j+i] * a[j] for i=0,...,len(a)-1 via FFT."""
    n = len(a)
    N = _next_pow2(len(w) + n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fw * np.conj(fa), N)
    return result[:n]


def _score(a):
    n = len(a)
    s = float(a.sum())

    c = _full_convolve(a, a)

def _power_sum_grad_direct(a, p):

        max_pat = 200

            g = g / gnorm

            a_new = np.maximum(a_new, 0)

            if s_new < 1e-15:

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = _cross_corr(w, a)
    grad_a = 2.0 * n * g
    return a * (grad_a - np.dot(a, grad_a))

    base_beta = max(gap * 0.3, max_c * 0.005)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    for beta in betas:

            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad, method='L-BFGS-B',

def _hess_matvec(v, w, n):

def _cg_solve(hv_func, b, tol=1e-10, max_iter=500):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p_vec = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hv_func(p_vec)
        pH = float(np.dot(p_vec, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha <= 0:
            break
        x = x + alpha * p_vec
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p_vec = r + beta_cg * p_vec
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=500, tol_frac=0.002):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad = _hess_matvec(a, w, n)
        grad_proj = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_proj))
        if gnorm < 1e-14:
            break

        mu = 1e-7
        w_f = w

        def _hv(v):
            return _hess_matvec(v, w_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_proj, tol=1e-10, max_iter=300)
        dx = dx - np.mean(dx)

        # Line search
        f_old = 2.0 * float(np.dot(w, c))
        a_new = a.copy()
        alpha = 1.0
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = 2.0 * float(np.dot(w, c_try))
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc
