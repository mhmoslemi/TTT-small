def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

"""
Step function optimization using active-set mirror descent with structured
initialization, multi-resolution search, FW hybrid steps, and coordinate polish.
Improvements over previous: better init diversity, adaptive active-set growth,
Frank-Wolfe hybrid updates, and more aggressive polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient(a, active):
    """Compute averaged subgradient 2*Q_k*a over active lags."""
    n = len(a)
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    return g


def _project_simplex(y):
    """Project y onto {a >= 0, sum(a) = 1}."""
    n = len(y)
    if n == 0:
        return y
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)


def _mirror_descend(a_init, deadline, lr_init, max_iter, active_tol_init,
                    max_active, eval_every, fw_interval):
    """Mirror descent in log-space with adaptive active set and FW hybrid."""
    n = len(a_init)
    a = a_init / a_init.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    lr = lr_init
    active_tol = active_tol_init

    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    no_improve_iter = 0
    last_best = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Active set with growing tolerance
        threshold = c_max * active_tol
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > max_active:
            order = np.argsort(c[active])[::-1][:max_active]
            active = active[order]

        # Subgradient
        g = _subgradient(a, active)

        # Mix with FW step periodically
        if fw_interval > 0 and (it % fw_interval == 0) and it > 0:
            # FW: move toward argmin(g)
            j_min = int(np.argmin(g))
            v = np.zeros(n)
            v[j_min] = 1.0
            t_fw = 2.0 / (it + 2)
            a_fw = (1 - t_fw) * a + t_fw * v
            # Blend: 30% FW, 70% mirror descent
            z_grad = a * (g - float(np.dot(a, g)))
            z_md = z - lr * z_grad
            a_md = _softmax(z_md - z_md.mean())
            a = 0.7 * a_md + 0.3 * a_fw
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
        else:
            z_grad = a * (g - float(np.dot(a, g)))
            z = z - lr * z_grad
            z = np.clip(z, -100.0, 100.0)
            z = z - z.mean()

        # Periodic evaluation
        if (it + 1) % eval_every == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                no_improve_iter = 0
            else:
                no_improve_iter += eval_every

            # Adaptive LR
            if no_improve_iter > 500:
                lr *= 0.7
                no_improve_iter = 0
                if lr < 1e-5:
                    break

        # Grow active set tolerance slowly
        if (it + 1) % 2000 == 0:
            active_tol = min(0.9999, active_tol + 0.0005)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _projected_gradient(a_init, deadline, lr_init, max_iter, active_tol_init,
                        max_active, eval_every):
    """Projected gradient descent (Euclidean) with active set."""
    n = len(a_init)
    a = a_init / a_init.sum()
    lr = lr_init
    active_tol = active_tol_init
    best_a = a.copy()
    best_score = _score_norm(a)
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        threshold = c_max * active_tol
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > max_active:
            order = np.argsort(c[active])[::-1][:max_active]
            active = active[order]

        g = _subgradient(a, active)
        a_new = _project_simplex(a - lr * g)

        if (it + 1) % eval_every == 0:
            sc = _score_norm(a_new)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                a = a_new
                no_improve += eval_every
                if no_improve > 300:
                    lr *= 0.6
                    no_improve = 0
                    if lr < 1e-7:
                        break

        if (it + 1) % 2000 == 0:
            active_tol = min(0.9999, active_tol + 0.0005)

    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=200):

def _refined_adam(a_init, deadline, lr_init, max_iter, eval_every):
    """Adam optimizer with careful active-set gradient and restarts."""
    n = len(a_init)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr_eff = lr * (0.05 ** (frac ** 0.6)) * current_mult

        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Active set
        tol = c_max * (0.99 + 0.009 * frac)
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > 10:
            order = np.argsort(c[active])[::-1][:10]
            active = active[order]

        g = _subgradient(a, active)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr_eff * mh / (np.sqrt(vh) + eps)
        upd_norm = np.linalg.norm(update)
        if upd_norm > 1.0:
            update = update / upd_norm
        z = z - update
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if (t + 1) % eval_every == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            if patience > 30:
                current_mult *= 0.6
                patience = 0
                if current_mult < 0.02:
                    # Restart from best with perturbation
                    rng = np.random.default_rng(t)
                    z = np.log(np.maximum(best_a, 1e-15))
                    z += 0.02 * rng.standard_normal(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    current_mult = 0.5

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score
