    Returns gradient vector of length n.

    and take spectral "square root" to update a.

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_flat = np.full(out_n, 1.0 / out_n)  # flat target (sum = 1)
        
        # Blend: as iterations progress, blend more toward flat
        alpha = blend_start * (blend_end / blend_start) ** (it / max(1, n_iters - 1))
        c_target = alpha * c + (1.0 - alpha) * c_flat
        
        # Try spectral square root
        C = np.fft.fft(c_target, N)
        # C should be "close to" A^2 for some A
        # Take sqrt (element-wise)
        # Handle: if C[k] is close to real and positive, sqrt is real
        # If C[k] has negative real part, sqrt gives imaginary component
        A = np.sqrt(C + 0j)
        a_new = np.fft.ifft(A)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Check if this improved things
        sc_old = _score(a)
        sc_new = _score(a_new)
        if sc_new < sc_old:
            a = a_new
        # If not improved, stop (we've diverged)
        else:
            # Try with more blending toward current c
            alpha2 = min(1.0, alpha * 2)
            c_target2 = alpha2 * c + (1.0 - alpha2) * c_flat
            C2 = np.fft.fft(c_target2, N)
            A2 = np.sqrt(C2 + 0j)
            a_new2 = np.fft.ifft(A2)[:n].real
            a_new2 = np.maximum(a_new2, 0.0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-10:
                a_new2 = a_new2 / s_new2
                sc_new2 = _score(a_new2)
                if sc_new2 < sc_old:
                    a = a_new2
                else:
                    break
            else:
                break
    
    return a


def _adam_lbfgs(z_init, beta_schedule, var_weight_schedule, n, deadline, 
                lr_base=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                iters_per_beta=200):
    """
    Adam optimizer on the combined LSE + variance objective.
    z_init: initial log-parameter
    beta_schedule: list of beta values (temperature for LSE)
    var_weight_schedule: list of variance weights (same length as beta_schedule)
    """
    z = z_init.copy()
    m = np.zeros_like(z)  # Adam first moment
    v = np.zeros_like(z)  # Adam second moment
    t = 0
    best_sc = _score(_softmax(z))
    best_z = z.copy()
    
    for bi, (beta, vw) in enumerate(zip(beta_schedule, var_weight_schedule)):
        if time.monotonic() > deadline - 10:
            break
        for it in range(iters_per_beta):
            if time.monotonic() > deadline - 5:
                break
            t += 1
            obj, grad = _combined_obj_grad(z, beta, n, vw)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            # Learning rate with warmup and decay
            lr = lr_base * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
            # Decay learning rate over the schedule
            progress = (bi * iters_per_beta + it) / max(1, len(beta_schedule) * iters_per_beta)
            lr *= (1.0 - 0.5 * progress)
            
            z = z - lr * (m_hat / (np.sqrt(v_hat) + eps))
            z = z - z.mean()  # Re-center to maintain invariance
            
            # Periodic score check
            if it % 50 == 0 or it == iters_per_beta - 1:
                a_cur = _softmax(z)
                sc_cur = _score(a_cur)
                if sc_cur < best_sc - 1e-12:
                    best_sc = sc_cur
                    best_z = z.copy()
    
    return best_z, best_sc


def _subgrad_peak_reduce(a, deadline, n_steps=5000, top_k=20, lr_start=0.01, lr_end=0.0001):
    """
    Subgradient method: at each step, compute the subgradient of max(c)
    (averaged over top-k elements) and take a projected gradient step.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
        
        # Top-k subgradient
        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        top_c = c[top_idx]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / k
        
        # Gradient of max(c) wrt a: for peak at position p, grad[i] = 2*a[p-i]
        # For top-k average: grad[i] = 2 * sum_p w[p] * a[p-i]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        
        # Learning rate with cosine decay
        progress = step / max(1, n_steps - 1)
        lr = lr_start * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, lr_end)
        
        # Projected gradient step (maintain a >= 0, sum = 1)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
        a_new = a_new / s_new
        a = a_new
    
    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_start=0.003, tol_end=0.00001,
                       lr_start=0.02, top_k=10):
    """Active-set gradient descent with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0
    
    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: elements within tol of max
        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / len(top_idx)
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-16:
            break
        
        # Backtracking
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.5
            if no_improve >= 200:
                a = best_a.copy()
                lr = lr_start * 0.3
                no_improve = 0
                stall += 1
                if stall > 10:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.97)
            lr = min(lr_start, lr * 1.02)
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.1) ** 2) + np.exp(-15.0 * (t - 0.9) ** 2) + 0.01
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = rng.random(n)
        a = a + 0.1  # Ensure no zeros
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_optimization_pipeline(a_init, deadline, label=""):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_score = _score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Step 1: Spectral flattening
    if time.monotonic() < deadline - 30:
        a_spec = _spectral_flatten(a, n_iters=30, blend_start=0.6, blend_end=0.1)
        sc_spec = _score(a_spec)
        if sc_spec < best_score:
            best_a = a_spec.copy()
            best_score = sc_spec
            print(f"    [Spectral] {sc_spec:.10f} (n={n}) {label} t={time.monotonic()-t0:.1f}s")
    
    # Step 2: Adam on LSE + variance
    if time.monotonic() < deadline - 10:
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()
        
        # Compute initial c to set beta
        c = _full_convolve(best_a, best_a)
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted)-1)]
        base_beta = max(gap, float(c.max()) * 0.005)
        
        # Beta schedule: from smooth to sharp
        n_betas = 8
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005])
        var_weights = np.array([0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005])
        
        # Estimate time per iteration
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        available_time = deadline - time.monotonic() - 5
        total_iters = int(available_time / (est_iter_time * 3))
        iters_per_beta = max(50, total_iters // n_betas)
        
        z_opt, sc_adam = _adam_lbfgs(z, betas, var_weights, n, time.monotonic() + available_time,
                                      lr_base=0.05, iters_per_beta=iters_per_beta)
        a_adam = _softmax(z_opt)
        sc_adam = _score(a_adam)
        if sc_adam < best_score:
            best_a = a_adam.copy()
            best_score = sc_adam
            print(f"    [Adam] {sc_adam:.10f} (n={n}) {label} t={time.monotonic()-t0:.1f}s")
    
    # Step 3: Subgradient peak reduction
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        n_steps = int(remaining / (0.005 * max(1.0, n/4096.0)))
        n_steps = min(n_steps, 10000)
        a_sg, sc_sg = _subgrad_peak_reduce(best_a, time.monotonic() + remaining * 0.5,
                                            n_steps=n_steps, top_k=30)
        if sc_sg < best_score:
            best_a = a_sg.copy()
            best_score = sc_sg
            print(f"    [Subgrad] {sc_sg:.10f} (n={n}) {label} t={time.monotonic()-t0:.1f}s")
    
    # Step 4: Active-set polish
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 3
        a_pol, sc_pol = _active_set_polish(best_a, time.monotonic() + remaining,
                                            tol_start=0.002, tol_end=0.00001,
                                            lr_start=0.015, top_k=5)
        if sc_pol < best_score:
            best_a = a_pol.copy()
            best_score = sc_pol
            print(f"    [Polish] {sc_pol:.10f} (n={n}) {label} t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: spectral flattening, Adam LSE curriculum,
    subgradient peak reduction, active-set polish, multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Optimize parent ===
    print(f"\n[P1] Optimizing parent (n={len(best_a)})...")
    p1_deadline = min(deadline - 900.0, t_start + 400.0)
    a1, sc1 = _run_optimization_pipeline(best_a, p1_deadline, "parent")
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  P1 result: {best_score:.10f} (n={len(best_a)})")
    
    # === Phase 2: Multi-n exploration ===
    n_candidates = [12288, 16384, 20480, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    n_candidates = sorted(set(n_candidates))
    
    print(f"\n[P2] Multi-n exploration: {n_candidates}")
    time_per_n = max(60.0, (deadline - time.monotonic() - 100.0) / len(n_candidates))
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60.0:
            break
        if n_try == len(best_a):
            # Already optimized at this n, skip
            continue
        
        # Try multiple initializations for this n
        inits = [
            ("resample", _resample(best_a, n_try)),
            ("uniform", _make_profile(n_try, "uniform")),
            ("tapered", _make_profile(n_try, "tapered")),
            ("wedge", _make_profile(n_try, "wedge")),
        ]
        
        for init_label, a_init in inits:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = min(time_per_n, deadline - time.monotonic() - 20.0)
            if remaining < 30:
                break
            
            a_opt, sc_opt = _run_optimization_pipeline(a_init, time.monotonic() + remaining, 
                                                        f"n={n_try}_{init_label}")
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}, {init_label}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Perturbation & restart ===
    print(f"\n[P3] Perturbation restarts...")
    if time.monotonic() < deadline - 60.0:
        n_best = len(best_a)
        perturbations = [
            ("pert_01", _perturb(best_a, 0.01, rng)),
            ("pert_03", _perturb(best_a, 0.03, rng)),
            ("pert_10", _perturb(best_a, 0.10, rng)),
            ("pert_30", _perturb(best_a, 0.30, rng)),
        ]
        
        remaining = deadline - time.monotonic() - 20.0
        time_per_pert = max(20.0, remaining / len(perturbations))
        
        for label, a_r in perturbations:
            if time.monotonic() > deadline - 15.0:
                break
            remaining = min(time_per_pert, deadline - time.monotonic() - 10.0)
            if remaining < 15:
                break
            a_opt, sc_opt = _run_optimization_pipeline(a_r, time.monotonic() + remaining, label)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [{label}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Final ultra-fine polish ===
    print(f"\n[P4] Final polish...")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 2.0,
                                            tol_start=0.0005, tol_end=0.000003,
                                            lr_start=0.005, top_k=3)
        if sc_fin < best_score:
