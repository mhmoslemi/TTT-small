"""
Optimized step function search using active-set gradient methods,
LSE-smoothed L-BFGS with adaptive beta, Newton refinement with CG inner solver,
and structural perturbations for minimizing peak autocorrelation of non-negative sequences.
Key improvements: custom gradient loops (no scipy overhead for inner iterations),
tighter active-set Newton, peak-pairing perturbations, and palindromic-structure exploitation.
"""
import numpy as np
import time
from scipy.optimize import minimize

_best_seq = None
_best_score = np.inf


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_and_grad(a, n, weights):
    """Compute convolution and gradient of dot(weights, conv(a,a)) w.r.t. a.
    grad_i = 2 * sum_k weights[k] * a[k-i] (with validity)
    Computed as: grad = 2 * conv(weights_padded, a_rev)[n-1 : 2n-1]
    """
    c = _full_convolve(a, a)
    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return c, grad


def _lse_obj_and_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return obj, grad_z


def _active_set_grad(a, n, tol_frac=0.002, top_k=None):
    """Compute gradient of sum of top-k or tol-fract active peaks."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if top_k is not None:
        idx = np.argpartition(c, -top_k)[-top_k:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    else:
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 2.0 * n * (h - W)
    return c, grad, active


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    return 2.0 * r[n - 1:2 * n - 1]


def _cg_solve(hess_func, b, tol=1e-10, max_iter=150):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active(a_init, deadline, max_iter=200, tol_frac=0.001, top_k=None):
    """Newton method on the active set with CG inner solver."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if top_k is not None:
            idx = np.argpartition(c, -top_k)[-top_k:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        else:
            tol = max(tol_frac * c_max, 1e-14)
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-6
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=150)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _grad_descent_smoothed(a_init, deadline, min_remaining=60.0, n_stages=10):
    """Custom gradient descent with LSE smoothing and adaptive beta.
    Faster than scipy for inner loop since no function evaluation overhead.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.7, 0.4, 0.25, 0.15, 0.09, 0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005])
    betas = betas[:n_stages]

    # Adaptive step size (Barzilai-Borwein style)
    step = 0.1
    prev_grad = None
    prev_z = None

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        est_per_iter = 0.004 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(200, int(remaining / est_per_iter)))

        for it in range(max_iter):
            if time.monotonic() + min_remaining > deadline:
                break

            obj, grad_z = _lse_obj_and_grad(z, beta, n)
            gnorm = float(np.linalg.norm(grad_z))
            if gnorm < 1e-12:
                break

            # Barzilai-Borwein step size
            if prev_grad is not None and prev_z is not None:
                sz = z - prev_z
                sy = grad_z - prev_grad
                sz_norm = float(np.dot(sz, sz))
                sy_norm = float(np.dot(sy, sy))
                if sz_norm > 1e-20 and sy_norm > 1e-20:
                    s_sz = float(np.dot(sz, sy))
                    if abs(s_sz) > 1e-15:
                        if s_sz > 0:
                            step_candidate = sz_norm / s_sz
                        else:
                            step_candidate = s_sz / sy_norm
                        step = np.clip(step_candidate, 1e-6, 1.0)
            prev_grad = grad_z.copy()
            prev_z = z.copy()

            z = z - step * grad_z
            z = np.clip(z, -100.0, 100.0)
            z = z - z.mean()

            if it % 500 == 0:
                a_check = _softmax(z)
                sc = _score(a_check)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_check.copy()
                    # Reset BB memory
                    prev_grad = None
                    prev_z = None
                    step = 0.1

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))

    return _softmax(z)

def _resize_sequence(a, n_target):

    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:

    return a_target / s


def _peak_shaving(a, deadline, n_rounds=100, n_peaks=3):
    """Directly target and reduce the top peaks in the convolution."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002

    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        # Find top peaks
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        top_val = c[top_idx]
        # Weight peaks by how much they exceed the mean
        mean_c = np.mean(c)
        weights = np.zeros_like(c)
        for idx in top_idx:
            excess = max(0, c[idx] - mean_c)
            weights[idx] = excess
        w_sum = weights.sum()
        if w_sum < 1e-20:
            break
        weights = weights / w_sum

        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        grad_a = 2.0 * wc[n - 1:2 * n - 1]
        grad_a = grad_a - np.mean(grad_a)

        a_new = a - lr * grad_a
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            a = a_new
            best_sc = sc_new
            best_a = a_new.copy()
            lr = min(lr * 1.05, 0.01)
        else:
            lr *= 0.9

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-stage optimizer: custom gradient descent, L-BFGS LSE, Newton active-set,
    peak shaving, subgradient polish, and diverse restarts.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init")

    n_main = n_parent if n_parent > 1024 else 16384

    # === Phase 1: Quick diverse exploration (30s) ===
    print(f"\n[Phase 1] Diverse exploration")
    for kind in ["blackman", "hann", "tapered", "cos2", "gauss"]:
        if time.monotonic() > deadline - 1500:
            break
        a_try = _make_profile(n_main, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Custom gradient descent with LSE (300s) ===
    print(f"\n[Phase 2] Custom GD + LSE")
    p2_dead = min(deadline - 1200, time.monotonic() + 300)
    a_p2, sc_p2 = _grad_descent_smoothed(_best_seq.copy(), p2_dead, min_remaining=100, n_stages=10)
    upd(a_p2, sc_p2, "gd_lse")

    # === Phase 3: L-BFGS LSE (350s) ===
    print(f"\n[Phase 3] L-BFGS LSE")
    p3_dead = min(deadline - 850, time.monotonic() + 350)
    a_p3, sc_p3 = _lbfgs_lse(_best_seq.copy(), p3_dead, min_remaining=100, n_stages=12)
    upd(a_p3, sc_p3, "lbfgs_lse")

    # === Phase 4: Newton active-set (150s) ===
    print(f"\n[Phase 4] Newton active-set")
    p4_dead = min(deadline - 700, time.monotonic() + 150)
    a_p4, sc_p4 = _newton_active(_best_seq, p4_dead, max_iter=200, tol_frac=0.001)
    upd(a_p4, sc_p4, "newton")

    # === Phase 5: Peak shaving (100s) ===
    print(f"\n[Phase 5] Peak shaving")
    p5_dead = min(deadline - 600, time.monotonic() + 100)
    a_p5, sc_p5 = _peak_shaving(_best_seq, p5_dead, n_rounds=200, n_peaks=3)
    upd(a_p5, sc_p5, "peak_shave")

    # === Phase 6: Subgradient polish (150s) ===
    print(f"\n[Phase 6] Subgradient polish")
    p6_dead = min(deadline - 450, time.monotonic() + 150)
    a_p6, sc_p6 = _subgrad_polish(_best_seq, p6_dead, n_steps=6000, lr_init=0.004, tol_frac=0.001)
    upd(a_p6, sc_p6, "sg_polish")

    # === Phase 7: Newton + SGD alternating (150s) ===
    print(f"\n[Phase 7] Newton + SGD alternating")
    a_cur = _best_seq.copy()
    for rnd in range(4):
        if time.monotonic() > deadline - 300:
            break
        # Newton
        nw_dead = min(deadline - 300, time.monotonic() + 35)
        a_nw, sc_nw = _newton_active(a_cur, nw_dead, max_iter=80, tol_frac=0.002)
        if sc_nw <= _score(a_cur) + 1e-10:
            a_cur = a_nw
        upd(a_cur, _score(a_cur), f"alt_nw_{rnd}")
        # Peak shave
        ps_dead = min(deadline - 300, time.monotonic() + 20)
        a_ps, sc_ps = _peak_shaving(a_cur, ps_dead, n_rounds=50, n_peaks=2)
        if sc_ps <= _score(a_cur) + 1e-10:
            a_cur = a_ps
        upd(a_cur, _score(a_cur), f"alt_ps_{rnd}")

    # === Phase 8: Diverse restarts (300s) ===
    print(f"\n[Phase 8] Diverse restarts")
    n_restarts = 8
    for i in range(n_restarts):
        if time.monotonic() > deadline - 100:
            break
        if i == 0:
            p_start = _perturb_log(_best_seq, 0.02, rng)
        elif i == 1:
            p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)
        elif i == 2:
            p_start = _perturb_palindromic(_best_seq, 0.02, rng)
        elif i == 3:
            p_start = _perturb_amplitude(_best_seq, rng, sigma=0.05)
        elif i == 4:
            p_start = _perturb_block_swap(_best_seq, rng, n_swaps=3)
        elif i == 5:
            p_start = _perturb_log(_best_seq, 0.04, rng)
        elif i == 6:
            p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=15, max_amp=0.2)
        else:
            p_start = _perturb_log(_best_seq, 0.01, rng)

        # Quick optimization
        gd_dead = min(deadline - 100, time.monotonic() + 60)
        a_r, sc_r = _grad_descent_smoothed(p_start, gd_dead, min_remaining=30, n_stages=6)
        upd(a_r, sc_r, f"restart_{i}_gd")

        # Newton polish if promising
        if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 40:
            nw_dead = min(deadline - 30, time.monotonic() + 30)
            a_r2, sc_r2 = _newton_active(a_r, nw_dead, max_iter=60, tol_frac=0.003)
            upd(a_r2, sc_r2, f"restart_{i}_nw")

    # === Phase 9: Multi-n exploration (200s) ===
    print(f"\n[Phase 9] Multi-n exploration")
    n_candidates = sorted(set([4096, 8192, 16384, 32768, n_main]))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        gd_dead = min(deadline - 60, time.monotonic() + 50)
        a_opt, sc_opt = _grad_descent_smoothed(a_init, gd_dead, min_remaining=30, n_stages=8)
        upd(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < _best_score * 1.003 and time.monotonic() < deadline - 30:
            nw_dead = min(deadline - 20, time.monotonic() + 30)
            a_opt2, sc_opt2 = _newton_active(a_opt, nw_dead, max_iter=50, tol_frac=0.002)
            upd(a_opt2, sc_opt2, f"n_{n_try}_nw")

    # === Phase 10: Final aggressive polish (remaining time) ===
    print(f"\n[Phase 10] Final polish")
    remaining = deadline - time.monotonic()
    if _best_seq is not None and remaining > 10:
        # Ultra-tight Newton
        a_fin, sc_fin = _newton_active(_best_seq, deadline - 3, max_iter=100, tol_frac=0.0005, top_k=2)
        upd(a_fin, sc_fin, "final_newton_tight")

        # Peak shave with more peaks
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _peak_shaving(_best_seq, deadline - 3, n_rounds=100, n_peaks=5)
            upd(a_fin2, sc_fin2, "final_peak5")

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=3000, lr_init=0.003, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

        # One more Newton
        if time.monotonic() < deadline - 1:
            a_fin4, sc_fin4 = _newton_active(_best_seq, deadline - 0.5, max_iter=30, tol_frac=0.0003, top_k=3)
            upd(a_fin4, sc_fin4, "final_newton2")

    # Final validation
    print(f"\n{'='*60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
