"""
Optimized step-function search using Newton-type linearized convolution with
conjugate gradient inner solver, projected constraints, and multi-scale
refinement. Combines a Gauss-Newton approach (linearizing the quadratic
convolution) with efficient FFT-based matrix-vector products for the inner
CG loop, enabling fast convergence for large sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _full_convolve(a, b):
    return _fft_convolve(a, b)


def _cross_corr(a, g):
    """Compute (a cross g)[i] = sum_m a[m] * g[m+i] for i=0..len(a)-1."""
    n = len(a)
    m = len(g)
    N = _next_pow2(n + m)
    fa = np.fft.rfft(a, N)
    fg = np.fft.rfft(g, N)
    result = np.fft.irfft(np.conj(fa) * fg, N)[:n]
    return result


def _L_apply(a_fixed, delta):
    """L(delta) = conv(a_fixed, delta), result has length 2n-1."""
    return _fft_convolve(a_fixed, delta)


def _LT_apply(a_fixed, gamma):
    """L^T(gamma) where L(delta) = conv(a_fixed, delta). Result has length n."""
    n = len(a_fixed)
    return _cross_corr(a_fixed, gamma)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_box_sum_zero(delta, a):
    """Project delta onto {d: a + d >= 0, sum(d) = 0}."""
    # Find t such that sum_j max(delta_j - t, -a_j) = 0
    def f(t):
        return float(np.sum(np.maximum(delta - t, -a)))

    d_min = float(np.min(delta))
    a_max = float(np.max(a))
    lo = d_min - a_max - 2.0
    hi = float(np.max(delta)) + 2.0
    if f(lo) < 0:
        lo -= abs(f(lo)) + 1
    if f(hi) > 0:
        hi += abs(f(hi)) + 1

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid

    t = 0.5 * (lo + hi)
    return np.maximum(delta - t, -a)


def _cg_solve_lstm(a_fixed, b, eps, max_iter=300, tol=1e-8):
    """
    Solve (4*L^T*L + eps*I) * delta = 2*L^T*b using conjugate gradient.
    L(delta) = conv(a_fixed, delta), so L^T is cross-correlation.
    """
    n = len(a_fixed)
    rhs = 2.0 * _LT_apply(a_fixed, b)

    def matvec(x):
        Lx = _L_apply(a_fixed, x)
        LT_Lx = _LT_apply(a_fixed, Lx)
        return 4.0 * LT_Lx + eps * x

    x = np.zeros(n)
    r = rhs - matvec(x)
    p = r.copy()
    rs_old = float(np.dot(r, r))
    rhs_norm = float(np.linalg.norm(rhs))
    if rhs_norm < 1e-30:
        return x

    for _ in range(max_iter):
        Ap = matvec(p)
        pAp = float(np.dot(p, Ap))
        if abs(pAp) < 1e-30:
            break
        alpha = rs_old / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * rhs_norm:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new

    return x


def _newton_cg_refine(a_init, deadline, eps_start=0.5, max_steps=500):
    """Newton-type method with linearized convolution and CG inner solver."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eps = eps_start

    for it in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Target: reduce all c_k toward c_max (flatten the convolution)
        # Newton step: minimize ||c + 2*L(delta) - c_max*1||^2 + eps*||delta||^2
        b_target = c_max * np.ones(2 * n - 1) - c

        delta = _cg_solve_lstm(a, b_target, eps, max_iter=200, tol=1e-7)

        if np.linalg.norm(delta) < 1e-12:
            eps *= 0.3
            continue

        # Project onto feasible directions
        delta = _project_box_sum_zero(delta, a)

        if np.linalg.norm(delta) < 1e-14:
            eps *= 0.3
            continue

        # Line search
        best_alpha = 0.0
        best_sc_ls = sc
        alphas = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]
        for alpha in alphas:
            a_try = a + alpha * delta
            if (a_try < -1e-12).any():
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc_ls - 1e-14:
                best_sc_ls = sc_try
                best_alpha = alpha

        if best_alpha > 0:
            a = a + best_alpha * delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            # Succeed
            eps = max(0.01, eps * 0.95)
        else:
            eps *= 0.5
            if eps < 1e-6:
                # Try a different strategy: reduce top peaks directly
                c = _full_convolve(a, a)
                top_idx = np.argsort(c)[::-1][:max(5, n // 100)]
                for idx in top_idx:
                    k_val = int(idx)
                    # Reduce contributors to this peak
                    k_lo = max(0, k_val - n + 1)
                    k_hi = min(n - 1, k_val)
                    if k_lo < k_hi:
                        contrib = a[k_lo:k_hi+1] * a[k_val - (k_lo:k_hi+1)][::-1]
                        top_contrib = np.argsort(contrib)[::-1][:3]
                        for tc in top_contrib:
                            a[k_lo + tc] *= 0.99
                        # Redistribute
                        reduction = 0.0
                        for tc in top_contrib:
                            reduction += a[k_lo + tc] * 0.01
                        a += reduction / n
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s > 1e-10:
                            a = a / s
                        sc_new = _score(a)
                        if sc_new < best_sc:
                            best_sc = sc_new
                            best_a = a.copy()
                eps = 0.5
                continue
            break

    return best_a, best_sc


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=15.0):
    """L-BFGS with aggressive beta annealing."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
         0.00005, 0.00002, 0.00001, 0.000005]
    )

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(3000, max(30, int(remaining / (est_iter_time * 20))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_start=0.004, tol_end=0.00002,
                       lr_start=0.02, top_k=5):
    """Active-set gradient descent with top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 10:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= 200:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.4
                no_improve = 0
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.985)
            lr = min(lr_start, lr * 1.01)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_conc":
        t = np.linspace(0, 1, n)
        a = t**3 + (1 - t)**3 + 0.01
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.1) ** 2) + np.exp(-30.0 * (t - 0.9) ** 2) + 0.005
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2.0 * np.abs(t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

        best_a = a_pol.copy()
        best_score = sc_pol
        print(f"  NEW BEST: {best_score:.10f} [polish_p3] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 4: Newton-CG again after polish ===
    print(f"\n[P4] Newton-CG round 2 (t={time.monotonic()-t_start:.1f}s)", flush=True)
    p4_deadline = min(deadline - 300.0, t_start + 950.0)
    a_newton2, sc_newton2 = _newton_cg_refine(best_a, p4_deadline, eps_start=0.3, max_steps=200)
    if sc_newton2 < best_score:
        best_a = a_newton2.copy()
        best_score = sc_newton2
        print(f"  NEW BEST: {best_score:.10f} [newton_cg_2] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_targets = [16384, 20480, 24576]
    if n_parent not in n_targets and n_parent > 4096:
        n_targets.append(n_parent)

    for n_try in sorted(n_targets):
        if time.monotonic() > deadline - 200.0:
            break
        if n_try == len(best_a):
            continue

        remaining_for_n = deadline - time.monotonic() - 150.0
        if remaining_for_n < 60.0:
            break

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  Trying n={n_try}, init score={sc_init:.10f}", flush=True)

        # Newton-CG on new n
        if time.monotonic() < deadline - 100.0:
            a_opt, sc_opt = _newton_cg_refine(a_init, deadline - 100.0,
                                              eps_start=0.5, max_steps=150)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [newton_multi_n] "
                      f"t={time.monotonic()-t_start:.1f}s", flush=True)
            else:
                a_opt = a_init

        # Quick L-BFGS on new n
        if time.monotonic() < deadline - 60.0:
            a_fast = _run_lbfgs(a_opt, deadline - 60.0, min_remaining=10.0)
            sc_fast = _score(a_fast)
            if sc_fast < best_score:
                best_a = a_fast.copy()
                best_score = sc_fast
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [lbfgs_multi_n] "
                      f"t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 100.0:
        n_best = len(best_a)
        restarts = [
            ("edge_conc", _make_profile(n_best, "edge_conc")),
            ("bimodal", _make_profile(n_best, "bimodal")),
            ("cos_mod", _make_profile(n_best, "cos_mod")),
            ("wedge", _make_profile(n_best, "wedge")),
            ("tapered", _make_profile(n_best, "tapered")),
        ]
        # Perturbed best
        z_pert = np.log(np.maximum(best_a, 1e-30)) + 0.03 * rng.standard_normal(n_best)
        z_pert = z_pert - z_pert.mean()
        restarts.append(("perturbed", _softmax(z_pert)))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 50.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 30.0:
                break
            # Quick Newton on restart
            a_opt, sc_opt = _newton_cg_refine(a_r, t_start + remaining - 5.0,
                                              eps_start=1.0, max_steps=50)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] "
                      f"t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 7: Final ultra-fine polishing ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 5.0:
        # Very fine active-set
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 3.0,
                                            tol_start=0.001, tol_end=0.000001,
                                            lr_start=0.005, top_k=3)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] "
                  f"t={time.monotonic()-t_start:.1f}s", flush=True)

        # Final Newton step
        if time.monotonic() < deadline - 3.0:
            a_fin2, sc_fin2 = _newton_cg_refine(best_a, deadline - 2.0,
                                                eps_start=0.1, max_steps=50)
            if sc_fin2 < best_score:
                best_a = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_newton] "
                      f"t={time.monotonic()-t_start:.1f}s", flush=True)

    # Final validation
    print(f"\n{'='*60}", flush=True)
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final]
