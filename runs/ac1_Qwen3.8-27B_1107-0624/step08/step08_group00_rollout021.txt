    n = len(a)
    a = a / a.sum()
    
    for it in range(max_iter):
        c = conv(a, a)
        M = c.max()
        
        # Select active constraints: top K values of c
        top_idx = np.argsort(c)[::-1][:K]
        
        # Build constraint matrix A (K x n)
        # A[k, j] = a[k-j] if 0 <= k-j < n
        A = np.zeros((K, n))
        for i, k in enumerate(top_idx):

            j_hi = min(n - 1, k)
            A[i, j_lo:j_hi+1] = a[j_lo:k-j_lo+1][::-1]
        
        # Solve LP: max sum(b) s.t. A @ b <= M, b >= 0
        # Use cvxpy or scipy
        
        # Step size
        t = 2.0 / (it + 2)
        
        # Update
        b_norm = b / b.sum()
        a = (1 - t) * a + t * b_norm
    
    return a

    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):

    out_n = n + m - 1

        for _ in range(40):
            a_try = a - alpha * g_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _conv_score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.85
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
            if no_improve >= max_stall:
                break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.01)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10
