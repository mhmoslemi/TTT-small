    a = np.exp(z)  
    a_n = (a / a.sum()).clip(0, 1000) * np.sqrt(2*n/np.pi)  # Rayleigh-normalized  
    c = np.convolve(a, a)  
    max_c = c.max() / max(a_n) if a_n.any() else 0  
    return score_eval(a, n)

def score_eval(a, n):  
    s = a.sum()  
    a_norm = a / s * np.sqrt(2*n)  # Adjusted normalization  
    conv = np.convolve(a_norm, a_norm)  
    max_val = conv.max()  
    return 2.0 * n * max_val / (np.sqrt(2*n) ** 2)
