Optimized step function search using minimax multiplicative weights,
KKT active-set solving, and diverse structural initializations.
Key innovation: treats the problem as a minimax game and uses
multiplicative weight updates to find the optimal active set,
then solves the KKT conditions exactly for that active set.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _compute_c_gradient(a, c, k):
    """Compute gradient of c[k] with respect to a.
    c[k] = sum_i a[i]*a[k-i], grad_i = 2*a[k-i] for valid i."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        # g[i] = 2*a[k-i] for i in [lo, hi]
        # a[k-i] for i in [lo,hi] = a[k-hi : k-lo+1] reversed
        g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return g

    The idea: we want to find a that minimizes max_k c[k].
    This is equivalent to finding a saddle point of:
      min_a max_w sum_k w[k]*c[k]  where w is a probability vector.
    
    We use the multiplicative weights algorithm:
    - Update w to put more weight on the currently largest c[k]
    - Update a to minimize the weighted sum
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Initialize weights
    c = _fft_convolve(a, a)
    T = c.max() * 0.1
    w = np.exp((c - c.max()) / T)
    w = w / w.sum()
    
    # Learning rate schedule
    base_lr = 0.5
    best_lr = base_lr

        if time.monotonic() > deadline:
            break
        
        if it % 100 == 0:

            T = max(c.max() * 0.05, 1e-10)
            w = np.exp((c - c.max()) / T)
            w = w / w.sum()
        
        # Compute weighted gradient: g = sum_k w[k] * grad_k(c[k])
        # For efficiency, only use the top weights
        top_k_idx = np.argsort(w)[::-1]
        num_active = 0
        for idx in top_k_idx:
            if w[idx] < 1e-8:
                break
            num_active = idx + 1  # not quite right for arbitrary indices

            active = np.array([np.argmax(w)])

        for k in active:
            g += w[k] * _compute_c_gradient(a, c, k)

            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:

            a_new /= s
            sc = _score_norm(a_new)

                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

                a = a_new
                improved = True

def _kkt_solve(a_init, active_lags, deadline, max_iter=5000):
    """Solve the KKT conditions for a given active set.
    
    At the optimum with active set A:
    - For each k in A: c[k] = t (equalized)
    - sum(a) = 1
    - a[i] >= 0
    - grad(sum_{k in A} c[k]/|A|) = lambda (for interior a[i] > 0)
    
    We iterate:
    1. Given active set, equalize c[k] for k in active
    2. Check if any other c[k] exceeds the equalized value
    3. If yes, add to active set
    4. If some active c[k] is no longer at max, remove it
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    active = list(active_lags)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Update active set
        tol = c_max * 0.999
        new_active = set(np.where(c >= tol)[0].tolist())
        
        # Also ensure we have at least 2 active
        if len(new_active) < 2:
            sorted_idx = np.argsort(c)[::-1]
            new_active = set(sorted_idx[:3].tolist())
        
        if len(new_active) > 20:
            # Keep only top 20
            sorted_by_val = sorted(new_active, key=lambda k: -c[k])
            new_active = set(sorted_by_val[:20])
        
        if new_active == set(active):
            # Same active set, try to solve
            pass
        else:
            active = list(new_active)
        
        # Solve: minimize max_{k in active} c[k]
        # Use the approach: find direction d that equalizes active c[k]
        # and reduces the common value
        
        K = len(active)
        # Build Jacobian: J[i,j] = d(c[i])/d(a[j]) for i in active
        J = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                J[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # We want: J @ d = -alpha * ones(K) (reduce all active c by same amount)
        # Subject to: sum(d) = 0 (preserve normalization)
        # Minimum norm solution:
        # d = -J^T @ (J @ J^T + eps*I)^{-1} @ ones / (ones^T @ (J @ J^T + eps*I)^{-1} @ ones)
        
        ones_K = np.ones(K)
        G = J @ J.T + 1e-12 * np.eye(K)
        try:
            G_inv_ones = np.linalg.solve(G, ones_K)
        except np.linalg.LinAlgError:
            G_inv_ones = np.linalg.lstsq(G, ones_K, rcond=None)[0]
        
        d = -J.T @ G_inv_ones
        # Project to sum-zero: d = d - mean(d)
        d = d - np.mean(d)
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Line search
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            a_new = a + alpha * d
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            # Check if we're at a local minimum
            c_new = _fft_convolve(a, a)
            if c_new.max() < c_max * (1 + 1e-10):
                break
    
    return best_a, best_score

def _legendre_sequence(n, rng):
    """Generate a pseudorandom sequence with low autocorrelation properties."""
    # Use a simple quadratic residue sequence (Legendre sequence)

"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _smoothed_obj_and_grad(z, T, n):
    """Compute smoothed objective and gradient in log-space.
    z is the log-parameter, a = softmax-like (exp(z)/sum(exp(z))).
    We minimize 2*n*max(c) with sum(a)=1 constraint handled by param.
    Actually, let's work directly with a on the simplex.
    """
    a = np.exp(z - np.max(z))
    s = a.sum()
    a = a / s
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    obj = 2.0 * n * c_max  # score with s=1
    
    # Smoothed max using log-sum-exp
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    logsumexp = c_max + T * np.log(e.sum())
    obj_smooth = 2.0 * n * logsumexp
    
    # Gradient of logsumexp w.r.t. a:
    # d(logsumexp)/d(c[k]) = exp(c[k]/T - logsumexp/T) = e[k]/sum(e)
    p = e / e.sum()
    # d(c[k])/d(a[i]) = 2*a[k-i] for valid i
    g_a = np.zeros(n)
    # Efficiently: g_a[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * (cross-correlation of p and a)
    # Use FFT: let p_pad and a_pad be zero-padded to size N
    N = _next_pow2(2 * n)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # cross[i] = sum_k p[k] * a[k-i] = ifft(conj(P) * A)[i + offset]
    cross = np.fft.irfft(np.conj(P) * A, N)
    # cross[i] = sum_k p[k] * a[(k-i) mod N]
    # For linear cross-correlation with proper padding:
    # We want sum_k p[k] * a[k-i] for i=0,...,n-1, with a[m]=0 for m<0 or m>=n
    # cross[i] for i in [0, N-1] gives the circular correlation
    # With N >= 2n-1, the linear correlation is in cross[n-1 : 2n-1] reversed? 
    # Actually: ifft(conj(P)*A)[i] = sum_k P[k] * A[(k-i) mod N]
    # = sum_k p[k] * a[(k-i) mod N]
    # For i in range(n), and N >= len(p) + n - 1 = (2n-1) + n - 1 = 3n-2... 
    # Hmm, let me just do it the safe way
    
    # Safer: direct computation for top active lags
    top_lags = np.argsort(p)[::-1]
    num_active = min(50, len(top_lags))
    g_a = np.zeros(n)
    for idx in range(num_active):
        k = top_lags[idx]
        if p[k] < 1e-12:
            break
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g_a[lo:hi+1] += p[k] * 2.0 * a[k-hi:k-lo+1][::-1]
    
    # Chain rule: d(obj_smooth)/d(z) = d(obj_smooth)/d(a) * da/dz
    # da/dz: a[i] = exp(z[i]) / sum(exp(z[j]))
    # da[i]/dz[j] = a[i]*(delta_ij - a[j])
    # So dZ[j] = sum_i g_a[i] * a[i]*(delta_ij - a[j]) = g_a[j]*a[j] - a[j]*sum_i g_a[i]*a[i]
    W = float(np.dot(g_a, a))
    g_z = a * (g_a - W)
    g_z *= 2.0 * n  # scale by 2*n
    
    return obj_smooth, g_z


def _lbfgs_optimize(a_init, deadline, T_init=0.01, T_min=0.0001, max_iter=50000):
    """L-BFGS optimization in log-space with annealing temperature."""
    n = len(a_init)
    best_a = _normalize(a_init.copy())
    best_score = _score_norm(best_a)
    
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()
    
    class Closure:
        def __init__(self):
            self.T = T_init
            self.grad = None
        
        def fun(self, z_in):
            obj, g = _smoothed_obj_and_grad(z_in, self.T, n)
            self.grad = g
            return obj
        
        def fun_and_grad(self, z_in):
            obj, g = _smoothed_obj_and_grad(z_in, self.T, n)
            self.grad = g
            return obj, g
    
    # Multi-phase temperature annealing
    phases = [
        (T_init, T_init * 0.1, max_iter // 3),
        (T_init * 0.1, T_init * 0.01, max_iter // 3),
        (T_init * 0.01, T_min, max_iter // 3 + max_iter % 3),
    ]
    
    z_cur = z.copy()
    for T_hi, T_lo, n_iter in phases:
        phase_deadline = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) * n_iter / max_iter))
        
       closure_obj = 0.0
        closure_grad = None
        call_count = [0]
        
        def objective(z_in):
            call_count[0] += 1
            # Linearly interpolate T within this phase
            frac = min(1.0, call_count[0] / max(1, n_iter))
            T_cur = T_hi * (T_lo / T_hi) ** frac
            obj, g = _smoothed_obj_and_grad(z_in, T_cur, n)
            return obj, g
        
        try:
            result = scipy_minimize(
                objective, z_cur, method='L-BFGS-B',
                options={'maxiter': n_iter, 'maxfun': n_iter, 'gtol': 1e-8}
            )
            if result.success or result.fun < _score_norm(_normalize(np.exp(z_cur - np.max(z_cur)))):
                z_cur = result.x
        except Exception as e:
            print(f"  L-BFGS error: {e}")
        
        a_cur = _normalize(np.exp(z_cur - np.max(z_cur)))
        sc = _score_norm(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        print(f"    Phase T={T_hi:.5f}->{T_lo:.5f}: score={sc:.10f}, t={time.monotonic()-time.monotonic()+0:.0f}s")
        
        if time.monotonic() > deadline:
            break
    
    return best_a, best_score


def _kkt_newton(a_init, deadline, max_iter=10000, active_tol=0.999):
    """KKT-based Newton method for active set equalization.
    
    At optimum with active set A = {k_1, ..., k_m}:
    - c[k_j] = t for all j (equalization)
    - sum(a) = 1
    - For interior: sum_j (1/m) * 2*a[k_j - i] = lambda for all i
    
    We use Newton's method on the system:
    F(a, t) = [c[k_1] - t, c[k_2] - t, ..., c[k_m] - t, sum(a) - 1] = 0
    
    The Jacobian is (m+1) x (n+1), but we solve for the minimum-norm
    update using the structure.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set
        tol = c_max * active_tol
        active = np.where(c >= tol)[0]
        if len(active) < 2:
            active = np.argsort(c)[::-1][:3]
        elif len(active) > 25:
            active = active[np.argsort(c[active])[::-1][:25]]
        
        m = len(active)
        
        # Build the system:
        # For each active k: c[k] - t = 0
        # d(c[k])/da[j] = 2*a[k-j] for valid j
        # d(c[k])/dt = -1
        # sum(a) - 1 = 0: d/da[j] = 1, d/dt = 0
        
        # Newton step: solve J @ [da; dt] = -F
        # where J is (m+1) x (n+1)
        # F = [c[k_1]-t, ..., c[k_m]-t, sum(a)-1]
        
        # We want minimum-norm solution, so:
        # [da; dt] = -J^T @ (J @ J^T)^{-1} @ F
        # But J @ J^T is (m+1) x (m+1), which is manageable.
        
        # J @ J^T[i, j] for i,j <= m: sum_k J[i,k] * J[j,k] = sum_k J[i,k]*J[j,k] + J[i,n+1]*J[j,n+1]
        # For i,j <= m: J[i, 0:n] = grad of c[active[i]] w.r.t. a
        #                J[i, n] = -1
        # J[m, 0:n] = ones(n), J[m, n] = 0
        
        # Let J_i = grad of c[active[i]] w.r.t. a (length n)
        # Then J@J^T[i,j] = J_i . J_j + (i==j ? 1 : 0) for i,j < m (since -1 * -1 = 1 for diagonal)
        # Wait: J[i,n] = -1 for i < m, J[m,n] = 0
        # So J@J^T[i,j] = J_i . J_j for i,j < m
        #                   + (-1)(-1) = +1 for i==j < m (from the t-column)
        #                   Actually: J@J^T[i,j] = sum_l J[i,l]*J[j,l]
        #   = sum_{l=0}^{n-1} J[i,l]*J[j,l] + J[i,n]*J[j,n]
        #   = J_i . J_j + (-1)*(-1) if i,j < m = J_i . J_j + 1 for i==j
        #   Wait no: J[i,n] = -1 for all i < m. So J[i,n]*J[j,n] = 1 for all i,j < m.
        #   Hmm that's not right either. Let me be more careful.
        
        # J is (m+1) x (n+1):
        # Row i (i < m): [2*a[k_i - j] for valid j, -1]
        # Row m: [1, 1, ..., 1, 0]
        
        # G = J @ J^T is (m+1) x (m+1):
        # G[i,j] for i,j < m: sum_{l=0}^{n-1} (2*a[k_i-l])*(2*a[k_j-l]) + (-1)*(-1)
        #   = 4 * sum_l a[k_i-l]*a[k_j-l] + 1
        #   = 4 * c[k_i - k_j + ...] hmm this is getting complicated
        
        # Let me just compute it directly for small m
        if m > 20:
            active = active[:20]
            m = 20
        
        J = np.zeros((m + 1, n + 1))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                J[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
            J[i, n] = -1.0
        J[m, 0:n] = 1.0
        # J[m, n] = 0 already
        
        F = np.zeros(m + 1)
        t_cur = c[active[0]]  # current "t" (should be ~c_max)
        for i, k in enumerate(active):
            F[i] = c[k] - t_cur
        F[m] = a.sum() - 1.0
        
        G = J @ J.T
        G += 1e-10 * np.eye(m + 1)
        
        try:
            G_inv_F = np.linalg.solve(G, F)
        except np.linalg.LinAlgError:
            G_inv_F = np.linalg.lstsq(G, F, rcond=None)[0]
        
        step = -J.T @ G_inv_F  # (n+1,) vector
        
        # Extract da and dt
        da = step[0:n]
        dt = step[n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_new = a + alpha * da
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            elif sc < _score_norm(a) + 1e-10:
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try smaller active tolerance or break
            active_tol = max(0.99, active_tol - 0.001)
            if it > 100 and not improved:
                break
    
    return best_a, best_score


def _momentum_subgrad(a_init, deadline, max_iter=100000, initial_lr=0.01):
    """Momentum-based subgradient descent with adaptive learning rate."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    
    momentum = np.zeros(n)
    beta = 0.8
    lr = initial_lr
    best_lr = lr
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Active set with smooth weighting
        tol = c_max * 0.999
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > 15:
            active = active[np.argsort(c[active])[::-1][:15]]
        
        # Weighted average gradient
        w = c[active] / c[active].sum()
        g = np.zeros(n)
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += w[i] * 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Project to sum-zero (tangent to simplex)
        g = g - np.mean(g)
        
        # Momentum
        momentum = beta * momentum + (1.0 - beta) * g
        direction = momentum - np.mean(momentum)
        
        # Line search
        a_new = a - lr * direction
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_new /= s
        sc_new = _score_norm(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            best_lr = lr
            lr = min(0.05, lr * 1.005)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 200:
                # Restart momentum
                momentum *= 0.1
                lr = max(lr, 0.001)
                no_improve = 0
            if lr < 1e-9:
                break
    
    return best_a, best_score


def _diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n)
    
    if parent is not None:
        inits.append(_normalize(parent.copy()))
    
    # Uniform with noise
    inits.append(_normalize(np.ones(n) + 0.3 * rng.standard_normal(n)))
    
    # Pure random
    inits.append(_normalize(rng.random(n)))
    
    # Raised cosine
    inits.append(_normalize(0.5 + 0.5 * np.cos(2 * np.pi * x / n)))
    
    # Gaussian
    center = n * 0.5
    sigma = n * 0.2
    inits.append(_normalize(np.exp(-0.5 * (x - center) ** 2 / sigma ** 2)))
    
    # Asymmetric (shifted Gaussian)
    center = n * 0.35
    inits.append(_normalize(np.exp(-0.5 * (x - center) ** 2 / (n * 0.15) ** 2)))
    
    # Multi-frequency sinusoid
    inits.append(_normalize(np.maximum(0.01, 0.5 + 0.3 * np.sin(2*np.pi*2*x/n) + 0.2 * np.sin(2*np.pi*5*x/n + 0.7))))
    
    # Chirp
    inits.append(_normalize(np.maximum(0.01, 0.5 + 0.5 * np.cos(2*np.pi * (0.5*x/n + 0.5*(x/n)**2)))))
    
    # Two bumps
    inits.append(_normalize(np.exp(-0.5*((x-0.3*n)/(0.08*n))**2) + np.exp(-0.5*((x-0.7*n)/(0.08*n))**2)))
    
    return inits

        time_per = max(30.0, (phase3_end - time.monotonic()) / max(1, min(3, len(scored_inits))))
        
        for i, (a_init, sc_init) in enumerate(scored_inits[:3]):
            if time.monotonic() > phase3_end - 10:
                break
            sub_dl = min(phase3_end - 5, time.monotonic() + time_per * 0.8)
            a_opt_i, sc_i = _lbfgs_optimize(a_init, sub_dl, T_init=0.003, T_min=0.00003, max_iter=15000)
            full_sc_i = _score(a_opt_i)
            print(f"  Init[{i+1}] (init_score={sc_init:.6f}): L-BFGS={full_sc_i:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc_i < best_score:
                best_score = full_sc_i
                best_a = a_opt_i.copy()
                print(f"  *** NEW BEST ***")
    
    # Phase 4: Momentum subgradient (1100 - 1450s)
    phase4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase4_end:
        print(f"\n=== Phase 4: Momentum subgradient ===")
        a_mom, sc_mom = _momentum_subgrad(best_a, phase4_end, max_iter=80000, initial_lr=0.005)
        full_sc_mom = _score(a_mom)
        print(f"  Momentum result: {full_sc_mom:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_mom < best_score:
            best_score = full_sc_mom
            best_a = a_mom.copy()
    
    # Phase 5: Second KKT pass with wider active set (1450 - 1650s)
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end:
        print(f"\n=== Phase 5: KKT with wider active set ===")
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        active = np.where(c >= c_max * 0.995)[0].tolist()
        if len(active) < 3:
            active = np.argsort(c)[::-1][:5].tolist()
        if len(active) > 30:
            active = sorted(active, key=lambda k: -c[k])[:30]
        a_kkt2, sc_kkt2 = _kkt_newton(best_a, phase5_end, max_iter=5000, active_tol=0.995)
        full_sc_kkt2 = _score(a_kkt2)
        print(f"  KKT2 result: {full_sc_kkt2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_kkt2 < best_score:
            best_score = full_sc_kkt2
            best_a = a_kkt2.copy()
    
    # Phase 6: Multi-restart with perturbation (1650 - 1770s)
    phase6_end = min(deadline, t_start + 1770.0)
    if time.monotonic() < phase6_end:
        print(f"\n=== Phase 6: Multi-restart ===")
        n_restarts = 4
        time_per = max(15.0, (phase6_end - time.monotonic()) / n_restarts)
        
        for r in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _normalize(np.exp(z - np.max(z)))
            
            sub_dl = min(phase6_end - 5, time.monotonic() + time_per * 0.7)
            a_opt_r, sc_r = _momentum_subgrad(a_pert, sub_dl, max_iter=20000, initial_lr=0.003)
            full_sc_r = _score(a_opt_r)
            print(f"  Restart[{r}] sigma={sigma:.4f}: {full_sc_r:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc_r < best_score:
                best_score = full_sc_r
                best_a = a_opt_r.copy()
                print(f"  *** NEW BEST ***")
    
    # Phase 7: Final fine polish (1770 - deadline)
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 7: Final polish ===")
        a_final, sc_final = _momentum_subgrad(best_a, deadline - 10, max_iter=30000, initial_lr=0.001)
        full_sc_final = _score(a_final)
        print(f"  Final polish: {full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()
        
        # Ultra-fine KKT
        if time.monotonic() < deadline - 5:
            c = _fft_convolve(best_a, best_a)
            c_max = float(c.max())
            active = np.where(c >= c_max * 0.9999)[0].tolist()
            if len(active) < 2:
                active = np.argsort(c)[::-1][:2].tolist()
            a_kkt_f, sc_kkt_f = _kkt_newton(best_a, deadline - 3, max_iter=2000, active_tol=0.9999)
            full_sc_kkt_f = _score(a_kkt_f)
            if full_sc_kkt_f < best_score:
                best_score = full_sc_kkt_f
                best_a = a_kkt_f.copy()
    
    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(n) * 0.01
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
