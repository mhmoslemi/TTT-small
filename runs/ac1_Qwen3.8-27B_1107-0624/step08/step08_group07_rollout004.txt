Wait, I need to be more careful with the subgradient.

`(a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]`

`d(a*a)[k] / da[m] = a[k-m] * I(0<=k-m<n) + a[m-k] * I(0<=m-k<n)`

For the first term `a[k-m] * I(0<=k-m<n)`:
- Valid when `0 <= k-m < n`, i.e., `k-n < m <= k`, i.e., `max(0, k-n+1) <= m <= min(n-1, k)`.
- The value is `a[k-m]`.

For the second term `a[m-k] * I(0<=m-k<n)`:
- Valid when `0 <= m-k < n`, i.e., `k <= m < k+n`, i.e., `max(0, k) <= m <= min(n-1, k+n-1)`.
- Since k >= 0 (convolution indices are non-negative), this is `k <= m <= min(n-1, k+n-1)`.
- The value is `a[m-k]`.

Note: when `m = k`, both terms give `a[0]`, so the total contribution at m=k is `2*a[0]` (if k is in [0, n-1]).

```python
"""
Optimized step function search using multi-phase optimization:
1. Adam in log-space with adaptive-temperature smoothed gradient
2. Exact subgradient method with Armijo backtracking
3. Pair-swap local search for escaping 1D local minima
4. Final ultra-fine polish
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1: 2*n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smoothed_grad(a, c, T):
    """Compute smoothed gradient of max(c) using softmax temperature T."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    
    # g[m] = 2 * sum_k p[k] * dc[k]/da[m] / 2
    # Actually: g[m] = sum_k p[k] * dc[k]/da[m]
    # where dc[k]/da[m] = a[k-m]*I(0<=k-m<n) + a[m-k]*I(0<=m-k<n)
    
    # For efficiency, compute via the top contributing lags
    # Find lags with significant probability
    threshold = p.max() * 0.001
    active_mask = p > threshold
    active_lags = np.where(active_mask)[0]
    
    g = np.zeros(n)
    for k in active_lags:
        w = p[k]
        # First part: a[k-m] for valid m
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += w * a[k-hi:k-lo+1][::-1]
        # Second part: a[m-k] for valid m
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += w * a[lo2-k:hi2-k+1]
    
    return g


def _compute_subgrad(a, c, n_active):
    """Compute exact subgradient: average gradient over top n_active lags."""
    n = len(a)
    c_max = float(c.max())
    threshold = c_max * 0.999
    active = np.where(c >= threshold)[0]
    if len(active) > n_active:
        # Keep only the top n_active
        order = np.argsort(c[active])[::-1][:n_active]
        active = active[order]
    
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += a[lo2-k:hi2-k+1]
    
    g /= len(active)
    return g, active


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, eval_interval=50, n_active_cap=10):
    """Adam optimizer in log-space with adaptive temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    no_improve = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        
        # Adaptive schedules
        T = T_start * (T_end / T_start) ** (frac ** 0.5)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7)
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Compute gradient
        if T < T_start * 0.005:
            # Sharp: use near-exact subgradient
            g, _ = _compute_subgrad(a, c, n_active_cap)
        else:
            g = _compute_smoothed_grad(a, c, T)
        
        if not np.all(np.isfinite(g)):
            continue
        
        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad ** 2)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update *= 0.5 / update_norm
        
        z_new = z - update
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        
        z = z_new
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive restart
            if no_improve > 20 and t > max_iter * 0.3:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z += 0.005 * np.random.randn(n)
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                no_improve = 0
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_iter=50000, n_active=5, 
                         initial_lr=0.002):
    """Exact subgradient method with Armijo backtracking."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    c = _fft_convolve(a, a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_max = float(c.max())
        
        # Get active lags
        threshold = c_max * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > n_active:
            order = np.argsort(c[active])[::-1][:n_active]
            active = active[order]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                g[lo2:hi2+1] += a[lo2-k:hi2-k+1]
        g /= len(active)
        
        # Project to tangent space of simplex (sum preserved)
        g_tan = g - g @ a  # Riemannian projection (approximate)
        # Or simpler: g_tan = g - mean(g)
        # Actually for the constraint sum(a)=1, the tangent projection is:
        # g_tan = g - (sum(g * a) * 1) — no, that's not right either
        # For the simplex {a >= 0, sum(a) = 1}, the projection of g onto
        # the tangent space is: g - (sum(g)/n) * 1 (Euclidean)
        # or g - g@sum(a) * 1 = g - g (since sum(a)=1)... 
        # Actually the tangent space is {v : sum(v) = 0}, so:
        # g_tan = g - mean(g) * ones
        
        g_tan = g - np.mean(g)
        
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break
        
        # Armijo backtracking
        alpha = lr
        improved = False
        for _ in range(10):
            a_new = a - alpha * g_tan
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-10:
                alpha *= 0.5
                continue
            a_new /= s
            sc_new = _score_norm(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                c = _fft_convolve(a, a)
                improved = True
                lr = min(lr * 1.01, 0.01)
                break
            else:
                alpha *= 0.5
        
        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
    
    return best_a, _score_norm(best_a)


def _pair_swap_search(a_init, deadline, max_rounds=200):
    """Local search that swaps mass between pairs of coordinates."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    
    # Identify which coordinates affect the peak most
    c_max = float(c.max())
    peak_lags = np.where(c >= c_max * 0.999)[0]
    
    # For each peak lag k, the contributing coordinates are
    # j in [max(0,k-n+1), min(k,n-1)] and k-j in [0,n-1]
    # So coordinates that appear in many peak lag decompositions are most important
    
    importance = np.zeros(n)
    for k in peak_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        importance[lo:hi+1] += 1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        importance[lo2:hi2+1] += 1
    
    # Sort by importance
    important_idx = np.argsort(importance)[::-1]
    # Take top candidates for swapping
    candidates = important_idx[:min(200, n)]
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        any_improve = False
        
        # Try swapping mass between pairs of important coordinates
        for i in range(0, len(candidates), max(1, len(candidates) // 50)):
            j1 = candidates[i]
            # Try a few j2 values
            for offset in [1, 2, 5, 10, 50, 100]:
                j2 = (j1 + offset) % n
                if j2 == j1:
                    continue
                
                # Try moving mass from j1 to j2 (and vice versa)
                for frac in [0.05, 0.1, 0.2, 0.5]:
                    delta = a[j1] * frac
                    if delta < 1e-15:
                        continue
                    a_cand = best_a.copy()
                    a_cand[j1] -= delta
                    a_cand[j2] += delta
                    if a_cand[j1] < 0:
                        continue
                    s = a_cand.sum()
                    if s < 1e-10:
                        continue
                    a_cand /= s
                    sc = _score_norm(a_cand)
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = _fft_convolve(best_a, best_a)
                        any_improve = True
                        break
                if any_improve:
                    break
            
            if time.monotonic() > deadline:
                break
        
        if not any_improve:
            # Try less important coordinates
            candidates = important_idx[min(200, n):min(500, n)]
            if len(candidates) < 10:
                break
    
    return best_a, best_score
```

```python
"""
Optimized step function search using corrected subgradient computation,
multi-phase Adam in log-space with adaptive temperature, Armijo backtracking,
and guided pair-swap local search. Key improvement: correct two-term
subgradient (a[k-m] + a[m-k]) instead of the approximate 2*a[k-m].
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _correct_subgrad_at_lags(a, lags):
    """Compute exact subgradient: average over lags of dc[k]/da[m].
    dc[k]/da[m] = a[k-m]*I(0<=k-m<n) + a[m-k]*I(0<=m-k<n)
    """
    n = len(a)
    g = np.zeros(n)
    for k in lags:
        # Term 1: a[k-m] valid for m in [max(0,k-n+1), min(n-1,k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            g[lo1:hi1+1] += a[k-hi1:k-lo1+1][::-1]
        # Term 2: a[m-k] valid for m in [max(0,k), min(n-1,k+n-1)]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += a[lo2-k:hi2-k+1]
    g /= len(lags)
    return g


def _smoothed_grad(a, c, T, max_lags=50):
    """Compute smoothed gradient using softmax over all lags, but only
    process lags with significant weight for efficiency."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s

    # Only process lags with p[k] > threshold
    threshold = p.max() * 1e-4
    active_mask = p > threshold
    active_lags = np.where(active_mask)[0]

    g = np.zeros(n)
    for k in active_lags:
        w = p[k]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            g[lo1:hi1+1] += w * a[k-hi1:k-lo1+1][::-1]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += w * a[lo2-k:hi2-k+1]
    return g


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,

    n = len(a_init)

    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    no_improve = 0
    best_t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)

        # Schedules
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8)

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Compute gradient (correct two-term version)

def _subgrad_line_search(a_init, deadline, max_iter=30000, n_active=5,
                         initial_lr=0.005):
    """Exact subgradient with Armijo backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Active lags
        thr = c_max * 0.9995
        active = np.where(c >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > n_active:
            order = np.argsort(c[active])[::-1][:n_active]
            active = active[order]

        # Correct subgradient
        g = _correct_subgrad_at_lags(a, active)

        # Project to tangent space of {sum=1}: g_tan = g - mean(g)
        g_tan = g - np.mean(g)
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break

        # Armijo backtracking
        alpha = lr
        improved = False
        for _ in range(12):
            a_new = a - alpha * g_tan
            a_new = np.maximum(0.0, a_new)
            s_new = a_new.sum()
            if s_new < 1e-10:
                alpha *= 0.5
                continue
            a_new /= s_new
            sc_new = _score_norm(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                lr = min(lr * 1.005, 0.05)
                break
            else:
                alpha *= 0.5

        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break

    return best_a, _score_norm(best_a)


def _directed_local_search(a_init, deadline, max_rounds=300):
    """Guided local search: move mass from peak-contributing coords to
    coords that minimize the increase in other peaks."""
    n = len(a_init)

            hi2 = min(n - 1, k + n - 1)

        # Top contributing source coordinates

        for src in top_sources:

                # Target: try coords that are far from src (less impact on same peak)

def _active_set_bisection(a_init, deadline, max_iter=2000):
    """Iteratively bisect the peak value and project onto feasibility.
    For target M, find a that minimizes max(0, c[k]-M)^2 subject to
    a>=0, sum(a)=1. Use gradient descent on this smoothed penalty."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)

    # Start with M slightly above current max
    c = _fft_convolve(a, a)
    M = float(c.max()) * 1.01

    lr = 0.01
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Update M: move toward c_max if feasible, else expand
        if c_max <= M:
            M = max(c_max * 1.001, M * 0.98)
        else:
            M = c_max * 1.01

        # Penalty: sum_k max(0, c[k]-M)^2
        # Gradient: 2 * sum_k max(0, c[k]-M) * dc[k]/da
        excess = np.maximum(0, c - M)
        if excess.max() < 1e-12:
            # All feasible, try to reduce M
            M *= 0.995
            continue

        active = np.where(excess > 0)[0]
        if len(active) > 20:
            order = np.argsort(excess[active])[::-1][:20]
            active = active[order]

        # Gradient of penalty w.r.t. a
        g = np.zeros(n)
        for k in active:
            w = excess[k]
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                g[lo1:hi1+1] += w * a[k-hi1:k-lo1+1][::-1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                g[lo2:hi2+1] += w * a[lo2-k:hi2-k+1]

        g_tan = g - np.mean(g)
        a_new = a - lr * g_tan
        a_new = np.maximum(0.0, a_new)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new /= s_new
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
        lr *= 0.999

    return best_a, best_score

        print(f"  P1: {s1_full:.10f} (t={time.monotonic()-t_start:.0f}s)")
        if s1_full < best_score:
            best_score = s1_full
            best_a = a1.copy()

    # Phase 2: Adam log-space, low temperature (400 - 750s)
    p2_end = min(deadline, t_start + 750.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Adam (sharp) [{p2_end - time.monotonic():.0f}s]")
        c1 = _fft_convolve(best_a, best_a)
        max1 = float(c1.max())
        a2, s2 = _adam_logspace(best_a, p2_end,
                                T_start=max1 * 0.005, T_end=max1 * 0.00001,
                                lr_start=0.025, lr_end=0.0005,
                                max_iter=400000)
        s2_full = _score(a2)
        print(f"  P2: {s2_full:.10f} (t={time.monotonic()-t_start:.0f}s)")
        if s2_full < best_score:
            best_score = s2_full
            best_a = a2.copy()

    # Phase 3: Subgradient with line search (750 - 1050s)
    p3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Subgrad+LS [{p3_end - time.monotonic():.0f}s]")
        a3, s3 = _subgrad_line_search(best_a, p3_end,
                                       max_iter=40000, n_active=5,
                                       initial_lr=0.005)
        s3_full = _score(a3)
        print(f"  P3: {s3_full:.10f} (t={time.monotonic()-t_start:.0f}s)")
        if s3_full < best_score:
            best_score = s3_full
            best_a = a3.copy()

    # Phase 4: Active-set bisection (1050 - 1300s)
    p4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Active-set bisection [{p4_end - time.monotonic():.0f}s]")
        a4, s4 = _active_set_bisection(best_a, p4_end, max_iter=3000)
        s4_full = _score(a4)
        print(f"  P4: {s4_full:.10f} (t={time.monotonic()-t_start:.0f}s)")
        if s4_full < best_score:
            best_score = s4_full
            best_a = a4.copy()

    # Phase 5: Directed local search (1300 - 1550s)
    p5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: Directed local search [{p5_end - time.monotonic():.0f}s]")
        a5, s5 = _directed_local_search(best_a, p5_end, max_rounds=200)
        s5_full = _score(a5)
        print(f"  P5: {s5_full:.10f} (t={time.monotonic()-t_start:.0f}s)")
        if s5_full < best_score:
            best_score = s5_full
            best_a = a5.copy()

    # Phase 6: Final Adam polish (1550 - 1700s)
    p6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < p6_end:
        print(f"Phase 6: Final Adam polish [{p6_end - time.monotonic():.0f}s]")
        c6 = _fft_convolve(best_a, best_a)
        max6 = float(c6.max())
        a6, s6 = _adam_logspace(best_a, p6_end,
                                T_start=max6 * 0.001, T_end=max6 * 0.000001,
                                lr_start=0.008, lr_end=0.0001,
                                max_iter=200000)
        s6_full = _score(a6)
        print(f"  P6: {s6_full:.10f} (t={time.monotonic()-t_start:.0f}s)")
        if s6_full < best_score:
            best_score = s6_full
            best_a = a6.copy()

    # Phase 7: Final subgradient micro-steps (1700 - deadline-15)
    if time.monotonic() < deadline - 15:
        print("Phase 7: Final subgrad micro-steps")
        a7, s7 = _subgrad_line_search(best_a, deadline - 10,
                                       max_iter=10000, n_active=3,
                                       initial_lr=0.001)
        s7_full = _score(a7)
        if s7_full < best_score:
            best_score = s7_full
            best_a = a7.copy()
        print(f"  P7: {s7_full:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
