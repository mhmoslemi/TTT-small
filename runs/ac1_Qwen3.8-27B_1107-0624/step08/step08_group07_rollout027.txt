Here's another idea: **use the structure of the problem more directly**.

The objective is 2n * max_k C[k] where C = conv(a, a). The maximum of C often occurs at k = n-1 (the central coefficient) for "spread out" sequences. If we know which k achieves the max, we can focus on reducing that specific C[k].

C[k] = sum_j a[j] * a[k-j] is a bilinear form. To reduce it while keeping sum a = 1, we want to "spread" the mass of a away from the pairs (j, k-j) that contribute most.

This suggests an iterative algorithm:
1. Compute C, find k* = argmax
2. For each j, compute the "contribution" a[j]*a[k*-j]
3. Reduce a[j] for j with high contribution, increase a[j] for j with low contribution
4. Normalize
5. Repeat

This is essentially the "water-filling" or "proportional fairness" approach. Let me formalize it:

At each step, for the active lag k*:
- Define contribution[j] = a[j] * a[k*-j] (for valid k*-j, else 0)
- The "pressure" on a[j] is proportional to a[k*-j] (the other factor in the product)
- Reduce a[j] by an amount proportional to a[k*-j], increase by amount proportional to (mean - a[k*-j])

This is like a multiplicative weight update:
a[j] <- a[j] * (1 - eta * a[k*-j]) for j with high contribution
a[j] <- a[j] * (1 + eta * (mean - a[k*-j])) for j with low contribution

def _lbfgs_iteration(z, g, s_list, y_list, m):
    """One L-BFGS iteration: compute search direction."""
    n = len(z)
    r = len(s_list)
    if r == 0:
        return -g
    alpha = [0.0] * r
    q = g.copy()
    # Two-loop recursion (forward)
    for i in range(r-1, -1, -1):
        s = s_list[i]
        y = y_list[i]
        yy = np.dot(y, y)
        if yy < 1e-30:
            alpha[i] = 0.0
        else:
            alpha[i] = np.dot(s, q) / yy
        q -= alpha[i] * y
    # Initial inverse Hessian approximation (use identity or last curvature)
    if r > 0:
        s_last = s_list[-1]
        y_last = y_list[-1]
        yy_last = np.dot(y_last, y_last)
        gamma = np.dot(s_last, y_last) / yy_last if yy_last > 1e-30 else 1.0
    else:
        gamma = 1.0
    r_vec = gamma * q
    # Two-loop recursion (backward)
    for i in range(r):
        s = s_list[i]
        y = y_list[i]
        yy = np.dot(y, y)
        beta = np.dot(y, r_vec) / yy if yy > 1e-30 else 0.0
        r_vec += (alpha[i] - beta) * s
    return -r_vec

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    last_improve = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.7) * lr_mult
        a = _softmax_stable(z)

        if T > T_start * 0.01:
            _, g = _smoothed_obj_grad(z, T)
        else:
            c = _fft_convolve(a, a)
            c_max = float(c.max())
            tol = c_max * 0.997
            active = np.where(c >= tol)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            elif len(active) > active_k:
                active = active[np.argsort(c[active])[::-1][:active_k]]
            g = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo <= hi:
                    g[lo:hi + 1] += (2.0 / len(active)) * a[k - hi:k - lo + 1][::-1]

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 1.0:
            update = update / un
        z_new = z - update
        z_new = np.clip(z_new, -70.0, 70.0)
        z_new = z_new - z_new.mean()

        if t % 200 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z.copy()
                m_buf *= 0.5
                v_buf *= 0.5
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve = t
            else:
                patience += 1
            if patience > 10 and t - last_improve > 800:
                lr_mult *= 0.6
                patience = 0
                if lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.3

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score
