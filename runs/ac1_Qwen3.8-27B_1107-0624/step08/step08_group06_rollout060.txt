import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

# Global state for L-BFGS callbacks
_GLOBAL_N = 0
_GLOBAL_BETA = 0.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):
    """Compute the 'correlation' of a with A_vec, extracting the valid n-length slice."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = _GLOBAL_BETA
    n = _GLOBAL_N
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

    wc = _full_convolve(w, a_rev)

    r = rhs - matvec_func(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:

        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):

        p = r + (rs_new / rs) * p

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive active-set Newton with continuation on tolerance
2. LP-based block coordinate descent (dual LP via cvxpy)
3. Multi-n exploration and diverse restarts
4. Final L-BFGS polish with very small beta
5. Theoretical flat-autocorrelation initialization

import numpy as np

from scipy.optimize import minimize

_GLOBAL_N = 0
_GLOBAL_BETA = 0.0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = _GLOBAL_BETA
    n = _GLOBAL_N
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z):

    c_max = float(c.max())
    beta = _GLOBAL_BETA

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _newton_direction(a, A_vec, m, mu, n):
    g = (2.0 / m) * _corr_a_A(a, A_vec, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-14:
        return None

    # CG to solve H @ u = g_t, where H @ v = (2/m)*corr(v, A) + mu*v
    u = np.zeros(n)
    r = g_t.copy()
    p = r.copy()
    rs = np.dot(r, r)
    gs_norm = np.linalg.norm(g_t)
    for _ in range(400):
        Hp = (2.0 / m) * _corr_a_A(p, A_vec, n) + mu * p
        pHp = np.dot(p, Hp)
        if pHp < 1e-30:
            break
        alpha = rs / pHp
        u = u + alpha * p
        r = r - alpha * Hp
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-13 * (1.0 + gs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

    # Solve H @ w_vec = ones for sum constraint handling
    ones = np.ones(n)
    w_vec = np.zeros(n)
    r = ones.copy()
    p = r.copy()
    rs = np.dot(r, r)
    on_norm = np.linalg.norm(ones)
    for _ in range(400):
        Hp = (2.0 / m) * _corr_a_A(p, A_vec, n) + mu * p
        pHp = np.dot(p, Hp)
        if pHp < 1e-30:
            break
        alpha = rs / pHp
        w_vec = w_vec + alpha * p
        r = r - alpha * Hp
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < 1e-13 * (1.0 + on_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

    w_sum = float(w_vec.sum())
    if abs(w_sum) < 1e-15:
        dx = -u
    else:
        lam = float(u.sum()) / w_sum
        dx = -u + lam * w_vec
    dx = dx - np.mean(dx)
    return dx

def _solve_dual_lp(f, M, active_indices, deadline):
    """Solve the dual LP: min M*sum(w) s.t. (corr(w,f))_j >= 1, w >= 0.
    Returns the optimal w (on active indices) and the optimal value."""
    n = len(f)
    k = len(active_indices)
    if k < 1:
        return None, None

    # Build constraint matrix G: G[j, i] = f[active_indices[i] - j] if valid
    # To save memory, build column by column using the structure
    # G[j, i] = f[active_indices[i] - j] if 0 <= active_indices[i] - j < n

    # For efficiency, use the fact that each column is a shifted version of f
    w_var = cp.Variable(k, nonneg=True)

    # Build G as sparse-ish (but cvxpy needs explicit)
    # G has shape (n, k)
    # G[j, i] = f[active_indices[i] - j] if valid
    # Let's build it row by row using slicing

    G_rows = []
    for j in range(n):
        # For column i: G[j,i] = f[active_indices[i] - j] if 0 <= active_indices[i]-j < n
        # i.e., active_indices[i] >= j and active_indices[i] - j < n
        # i.e., j <= active_indices[i] < j + n
        col_vals = np.zeros(k)
        for i in range(k):
            ai = active_indices[i]
            if j <= ai < j + n:
                col_vals[i] = f[ai - j]
        G_rows.append(col_vals)

    G = np.array(G_rows)

    constraints = [G @ w_var >= 1.0]
    obj = cp.Minimize(M * cp.sum(w_var))
    prob = cp.Problem(obj, constraints)

    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None, None

    if w_var.value is None:
        return None, None

    w_opt = np.asarray(w_var.value, dtype=np.float64)
    w_opt = np.maximum(w_opt, 0)
    return w_opt, float(prob.value) if prob.value is not None else float(M * np.sum(w_opt))


def _lp_iteration(a, deadline, n_lp_iters=30, tol_active=0.01):
    """LP-based block coordinate descent iteration."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_lp_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set
        active_thresh = c_max - tol_active * c_max
        active_indices = np.where(c >= active_thresh)[0]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])
        # Limit active set size for LP efficiency
        if len(active_indices) > 800:
            top = np.argsort(c)[::-1][:800]
            active_indices = top

        # Solve dual LP
        w_opt, lp_val = _solve_dual_lp(a, c_max, active_indices, deadline)
        if w_opt is None:
            break

        # Get primal solution from complementary slackness
        # g_j > 0 where (G^T w)_j < 1 (dual constraint not tight means primal var can be 0)
        # Actually: g_j > 0 => (G^T w)_j = 1 (complementary slackness)
        # So g is non-zero only where the dual constraint is tight
        # Let's reconstruct g from the LP: we need the primal solution

        # Alternative: solve the primal directly
        # Primal: max sum(g) s.t. (f*g)_k <= M for k in active, g >= 0
        # Let's solve this instead (it has n variables, |active| constraints)

        g_var = cp.Variable(n, nonneg=True)
        # Build constraint: for each k in active_indices, (a*g)_k <= c_max
        constraints = []
        for k in active_indices:
            # (a*g)_k = sum_i a[i] * g[k-i] for valid i
            # = sum_{i: 0<=k-i<n} a[i] * g[k-i]
            # Let j = k-i, so i = k-j, need 0<=k-j<n => max(0,k-n+1) <= j <= min(k, n-1)
            # (a*g)_k = sum_{j} a[k-j] * g[j] for valid j
            coeff = np.zeros(n)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                coeff[j] = a[k - j]
            constraints.append(coeff @ g_var <= c_max)

        obj = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS)
                except Exception:
                    break

        if g_var.value is None:
            break

        g_opt = np.asarray(g_var.value, dtype=np.float64)
        g_opt = np.maximum(g_opt, 0)
        s_g = float(g_opt.sum())
        if s_g < 1e-15:
            break

        g_opt = g_opt / s_g

        # Line search: a_new = (1-t)*a + t*g_opt
        accepted = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.01]:
            a_try = (1.0 - t) * a + t * g_opt
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                a = a_try
                best_sc = sc_try
                best_a = a_try.copy()
                accepted = True
                break
            elif sc_try < _score(a):
                a = a_try
                accepted = True
                break

        if not accepted:
            break

        # Check convergence
        if abs(_score(a) - best_sc) < 1e-14:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = max(1, n // 4)
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = max(1, n // 6)
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Active-set Newton from parent (broad to narrow)
    print(f"\n[Phase 1] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 400)
    a_main, sc_main = _active_set_newton(parent, p1_deadline, max_iter=600,
                                          tol_list=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    update_best(a_main, sc_main, "as_newton_p1")

    # Phase 2: L-BFGS continuation
    print(f"\n[Phase 2] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_lbfgs = _run_lbfgs(best_seq, deadline, min_time_remaining=500.0)
        sc_lbfgs = _score(a_lbfgs)
        update_best(a_lbfgs, sc_lbfgs, "lbfgs_p2")

    # Phase 3: LP-based iteration (different geometry)
    print(f"\n[Phase 3] LP-based BCD (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        lp_deadline = min(deadline - 300, time.monotonic() + 250)
        a_lp, sc_lp = _lp_iteration(best_seq, lp_deadline, n_lp_iters=40, tol_active=0.015)
        update_best(a_lp, sc_lp, "lp_bcd")
        if sc_lp < best_score:
            # Polish after LP
            a_poly = _run_lbfgs(a_lp, deadline, min_time_remaining=200.0)
            update_best(a_poly, _score(a_poly), "lbfgs_post_lp")

    # Phase 4: Tight active-set Newton
    print(f"\n[Phase 4] Tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        tight_dl = min(deadline - 200, time.monotonic() + 200)
        a_tight, sc_tight = _active_set_newton(best_seq, tight_dl, max_iter=400,
                                                tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        update_best(a_tight, sc_tight, "tight_newton")

    # Phase 5: Second LP pass (with smaller active set tolerance)
    print(f"\n[Phase 5] LP BCD tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        lp2_dl = min(deadline - 150, time.monotonic() + 120)
        a_lp2, sc_lp2 = _lp_iteration(best_seq, lp2_dl, n_lp_iters=20, tol_active=0.008)
        update_best(a_lp2, sc_lp2, "lp_bcd_tight")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            ("perturb_002", 0.002),
            ("perturb_005", 0.005),
            ("perturb_01", 0.01),
            ("perturb_02", 0.02),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _active_set_newton(p_start, min(deadline - 20, time.monotonic() + 40),
                                            max_iter=100, tol_list=[0.005, 0.002, 0.001, 0.0005])
            update_best(a_r, sc_r, f"restart_{label}")
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 10:
                a_r2 = _run_lbfgs(a_r, deadline, min_time_remaining=5.0)
                update_best(a_r2, _score(a_r2), f"restart_{label}_lbfgs")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_candidates = [8192, 16384, 32768]
        if n_parent in n_candidates:
            n_candidates.remove(n_parent)
        for n_try in sorted(n_candidates):
            if time.monotonic() > deadline - 50:
                break
            if best_seq is not None:
                a_init = _interpolate_to_n(best_seq, n_try)
            else:
                a_init = _make_profile(n_try, "uniform")
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")
            if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 20:
                a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 10, time.monotonic() + 25),
                                                      max_iter=80, tol_list=[0.002, 0.001, 0.0005, 0.0002])
                update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 8: Final ultra-tight polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight Newton
        ultra_dl = min(deadline - 3, time.monotonic() + 40)
        a_ultra, sc_ultra = _active_set_newton(best_seq, ultra_dl, max_iter=200,
                                                tol_list=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        update_best(a_ultra, sc_ultra, "ultra_newton")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 4:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            _GLOBAL_N = n
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            for beta_f in [mb * 0.00003, mb * 0.00001, mb * 0.000003]:
                if time.monotonic() > deadline - 3:
                    break
                _GLOBAL_BETA = beta_f
                remaining = deadline - 4 - time.monotonic()
                if remaining < 3:
                    break
                max_iter = min(3000, max(100, int(remaining / 0.05)))
                try:
                    result = minimize(
                        _lsobj, z, jac=_lsgrad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 100}
                    )
                    a_fin = _softmax(result.x)
                    update_best(a_fin, _score(a_fin), f"final_lbfgs_b{beta_f:.2e}")
                    z = np.log(np.maximum(a_fin, 1e-30))
                    z = z - z.mean()
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
