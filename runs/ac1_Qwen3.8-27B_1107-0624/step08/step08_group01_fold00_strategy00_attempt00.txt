def make_profile(length, kind="uniform"):
    """Create a test profile for the sequence, with the given kind"""
    if kind == "uniform":
        return np.ones(length) / length * length**0.5 * (2.0 * length)**0.5
    elif kind == "tapered":
        t = np.linspace(0, 1, length)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1.0e-10)
        return a / np.sqrt(np.sum(a**2))
    else:
        return np.ones(length) / length

def perturb(a, factor=0.1):
    """Small perturbation to increase exploration"""
    if len(a) < 10000:
        # Very small perturbation for sequences of moderate size
        return a + factor * np.random.standard_normal(len(a))
    else:
        # Large sequences: weird shift perturbation
        return a * (1 + factor * (np.random.random(len(a)) - 0.5))

def resample(a, new_n):
    """Resample the sequence to a desired length"""
    if new_n > len(a):
        # Up-sample by adding zero-padding or interpolation
        a = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, len(a)), a)
    elif new_n < len(a):
        # Down-sample by averaging
        a = np.convolve(a, np.ones(10)/10, mode='valid')
    else:
        # No change needed
        pass
    return a
