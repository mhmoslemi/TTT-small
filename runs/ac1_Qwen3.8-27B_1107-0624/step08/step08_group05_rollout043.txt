def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = e.sum()

        return np.ones_like(z) / len(z)

    """Compute A_w * v where (A_w * v)[m] = sum_j w_full[m+j] * v[j]."""
    n = len(v)

    a_rev = v[::-1]
    Fw = np.fft.rfft(w_full, N)
    Fa = np.fft.rfft(a_rev, N)
    result = np.fft.irfft(Fw * Fa, N)
    return result[n - 1:2 * n - 1]

        return _aw_matvec(w_full, v, n) + mu_reg * v

    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))

        pAp = float(np.dot(p, Ap))
        if abs(pAp) < 1e-30:

        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap

        if np.sqrt(r_new) < 1e-12 * b_norm:

        p = r + beta_cg * p
        rs = r_new

    a = x / s
    return a

        # grad_i = 2/n * sum_m (Aw_a[m] - c_mean) * (a[S[i]-m] - mean_j a[S[i]-m])
        # Actually, let me just use the full residual approach
        # d/dw_i ||Aw_a - c_mean||^2 = 2 * dot(residual, v_i) where v_i is the contribution of weight i
        # But c_mean also depends on w... Let me just do it properly.
        # Aw_a = sum_i w_i * v_i, where v_i[m] = a[S[i]-m] (zero-padded)
        # c_mean = (1/n) * sum_m Aw_a[m] = (1/n) * sum_i w_i * sum_m v_i[m]
        # Let s_i = sum_m v_i[m] = sum_j a[j] (for valid range) ... actually this varies with k
        
        # For simplicity and correctness, use finite differences for gradient
        # (only feasible for small p)
        if p <= 50:
            grad = np.zeros(p)
            eps = 1e-7
            for i in range(p):
                w_plus = w.copy()
                w_plus[i] += eps
                w_plus = np.maximum(w_plus, 0)
                w_plus = w_plus / w_plus.sum()
                w_full_plus = np.zeros(2 * n - 1)
                w_full_plus[S] = w_plus
                Aw_a_plus = _aw_matvec(w_full_plus, a, n)
                val_plus = float(np.var(Aw_a_plus))
                grad[i] = (val_plus - val) / eps
            return val, grad
        else:
            # Use analytic gradient
            # f(w) = (1/n) * ||V w - ones * (1/n) * 1^T V w||^2
            # = (1/n) * (V w)^T (I - (1/n) J) (V w) where J = ones*ones^T
            # grad = (2/n) * V^T (I - (1/n) J) V w
            # V^T (I - (1/n) J) V w = V^T V w - (1/n) V^T J V w
            # V^T J V w = V^T (ones * (ones^T V w)) = V^T (ones * s) where s = ones^T V w
            # = (V^T ones) * s
            # This is getting complex, let's just use the simple approach
            
            # Actually for the QP, let me just use scipy
            return val, None
    
    # Use scipy minimize for the QP
    w0 = np.ones(p) / p
    
    def qp_obj(w):
        w = np.maximum(w, 0)
        w = w / w.sum()
        w_full = np.zeros(2 * n - 1)
        w_full[S] = w
        Aw_a = _aw_matvec(w_full, a, n)
        return float(np.var(Aw_a))
    
    # Constraints: w >= 0, sum(w) = 1
    bounds = [(0, None)] * p
    constraints = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0}]
    
    if p <= 60:
        try:
            result = minimize(qp_obj, w0, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-14})
            w_opt = np.maximum(result.x, 0)
            w_opt = w_opt / w_opt.sum()
            return w_opt
        except Exception:
            pass
    
    # Fallback: projected gradient
    w = w0.copy()
    lr = 0.1
    for it in range(500):
        w_full = np.zeros(2 * n - 1)
        w_full[S] = w
        Aw_a = _aw_matvec(w_full, a, n)
        c_mean = Aw_a.mean()
        resid = Aw_a - c_mean
        
        # Gradient: d/dw_i = 2 * dot(resid, v_i) / n where v_i[m] = a[S[i]-m]
        grad = np.zeros(p)
        for i in range(p):
            k = S[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[i] = 2.0 * float(np.dot(resid[lo:hi+1], a[k-hi:k-lo+1][::-1])) / n
            else:
                grad[i] = 0.0
        
        w = w - lr * grad
        w = np.maximum(w, 0)
        s = w.sum()
        if s < 1e-15:
            w = w0.copy()
        else:
            w = w / s
        lr *= 0.999
    
    return w


def _kkt_refine(a_init, deadline, max_iter=200):
    """
    KKT-based active set refinement.
    Alternates between:
    1. Finding optimal weights w for current active set S (QP)
    2. Solving for a given w (CG linear system)
    3. Updating active set S
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify active set
        # Use adaptive tolerance
        tol = max(1e-6 * c_max, 1e-15)
        S = np.where(c >= c_max - tol)[0]
        
        if len(S) == 0:
            S = np.array([int(np.argmax(c))])
        
        if len(S) > 200:
            # Too many active points; keep only top 200
            c_sorted = np.argsort(c)[::-1]
            S = c_sorted[:200]
        
        # Step 1: Find optimal weights
        w = _find_optimal_weights(a, S, n)
        if w is None:
            break
        
        # Step 2: Build w_full and solve for a
        w_full = np.zeros(2 * n - 1)
        w_full[S] = w
        
        # Try multiple regularization levels
        a_new = None
        for mu in [1e-12, 1e-10, 1e-8, 1e-6, 1e-4]:
            a_try = _solve_kkt_a(w_full, n, mu_reg=mu, max_cg_iter=300)
            if a_try is not None:
                neg_count = int(np.sum(a_try < -1e-10))
                if neg_count == 0:
                    a_new = a_try
                    break
                elif mu == 1e-4:
                    # Accept even with some negatives (will clip)
                    a_new = a_try
        
        if a_new is None:
            # Fallback: use subgradient step
            grad = _aw_matvec(w_full, a, n)
            grad_t = grad - float(np.mean(grad))
            a_new = a - 0.01 * grad_t
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        
        # Ensure non-negativity
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Update
        a = a_new
        
        # Convergence check
        if outer > 10:
            c_check = _full_convolve(a, a)
            c_max_check = float(c_check.max())
            if abs(c_max_check - c_max) < 1e-14 * c_max:
                break
    
    return best_a, best_sc

# LSE L-BFGS (improved version)
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _lbfgs_lse_custom(a_init, deadline, min_remaining=30.0, n_stages=16):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.7, 0.45, 0.3, 0.2, 0.12, 0.07,
                                   0.04, 0.025, 0.015, 0.008, 0.004,
                                   0.002, 0.001, 0.0005, 0.0002])
    betas = betas[:n_stages]
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))
        
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _subgrad_polish_custom(a_init, deadline, n_steps=10000, lr_init=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_tol = 0.001
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        
        # Adaptive tolerance
        tol_frac = best_tol * (1.0 + 0.5 * np.cos(0.01 * i))
        grad_a = _active_gradient(a, c, n, tol_frac=max(tol_frac, 0.0005))
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                best_tol = max(0.0005, best_tol * 0.9)
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
                    best_tol = 0.001
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve_hess(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break
        
        mu = 1e-8
        active_f = active.astype(np.float64)
        
        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v
        
        dx = _cg_solve_hess(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)
        
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a
        
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800.0):
    """
    KKT-based optimizer with LSE L-BFGS global refinement,
    active-set KKT refinement, Newton polish, and multi-n exploration.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")
        n_parent = 8192

    # === Phase 1: LSE L-BFGS refinement (500s) ===
    print(f"\n[Phase 1] LSE L-BFGS refinement")
    if time.monotonic() < deadline - 500:
        a_opt, sc_opt = _lbfgs_lse_custom(_best_seq.copy(), deadline, min_remaining=480, n_stages=14)
        upd(a_opt, sc_opt, "lse_main")

    # === Phase 2: KKT-based refinement (300s) ===
    print(f"\n[Phase 2] KKT-based refinement")
    if time.monotonic() < deadline - 280:
        kkt_dead = min(deadline - 260, time.monotonic() + 300)
        a_kkt, sc_kkt = _kkt_refine(_best_seq.copy(), kkt_dead, max_iter=100)
        upd(a_kkt, sc_kkt, "kkt_main")

    # === Phase 3: Subgradient polish (150s) ===
    print(f"\n[Phase 3] Subgradient polish")
    if time.monotonic() < deadline - 140:
        sg_dead = min(deadline - 130, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish_custom(_best_seq.copy(), sg_dead, n_steps=8000, lr_init=0.006)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 4: Newton active-set (150s) ===
    print(f"\n[Phase 4] Newton active-set")
    if time.monotonic() < deadline - 130:
        nw_dead = min(deadline - 120, time.monotonic() + 150)
        a_nw, sc_nw = _newton_active_set(_best_seq.copy(), nw_dead, max_iter=200, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 5: KKT + Newton alternating (200s) ===
    print(f"\n[Phase 5] KKT + Newton alternating")
    a_cur = _best_seq.copy()
    for round_i in range(4):
        if time.monotonic() > deadline - 100:
            break
        # KKT step
        if time.monotonic() < deadline - 80:
            kkt_dead = min(deadline - 80, time.monotonic() + 50)
            a_kkt2, sc_kkt2 = _kkt_refine(a_cur.copy(), kkt_dead, max_iter=50)
            if sc_kkt2 < _score(a_cur):
                a_cur = a_kkt2
            upd(a_cur, _score(a_cur), f"alt_kkt_{round_i}")
        # Newton step
        if time.monotonic() < deadline - 60:
            nw_dead = min(deadline - 60, time.monotonic() + 40)
            a_nw2, sc_nw2 = _newton_active_set(a_cur.copy(), nw_dead, max_iter=100, tol_frac=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")
        # SG step
        if time.monotonic() < deadline - 40:
            sg_dead = min(deadline - 40, time.monotonic() + 30)
            a_sg2, sc_sg2 = _subgrad_polish_custom(a_cur.copy(), sg_dead, n_steps=2000, lr_init=0.004)
            if sc_sg2 < _score(a_cur):
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 6: Multi-n exploration (300s) ===
    print(f"\n[Phase 6] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384, 20480, 24576]
    if n_parent not in n_candidates and 2048 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        
        if _best_seq is not None and len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        elif _best_seq is not None:
            a_init = _resize_sequence(_best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")
        
        # LSE L-BFGS
        a_opt_n, sc_opt_n = _lbfgs_lse_custom(a_init, deadline, min_remaining=40, n_stages=10)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")
        
        # KKT polish
        if time.monotonic() < deadline - 30:
            kkt_dead = min(deadline - 25, time.monotonic() + 60)
            a_kkt_n, sc_kkt_n = _kkt_refine(a_opt_n.copy(), kkt_dead, max_iter=50)
            upd(a_kkt_n, sc_kkt_n, f"n_{n_try}_kkt")
        
        # Newton polish
        if time.monotonic() < deadline - 20:
            nw_dead = min(deadline - 18, time.monotonic() + 30)
            a_nw_n, sc_nw_n = _newton_active_set(a_opt_n.copy(), nw_dead, max_iter=80, tol_frac=0.002)
            upd(a_nw_n, sc_nw_n, f"n_{n_try}_newton")

    # === Phase 7: Diverse restarts (150s) ===
    print(f"\n[Phase 7] Diverse restarts")
    if time.monotonic() < deadline - 40:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)
            elif i == 2:
                p_start = _perturb_log(_best_seq, 0.04, rng)
            elif i == 3:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=5, max_amp=0.5)
            else:
                p_start = _perturb_log(_best_seq, 0.06, rng)
            
            a_r, sc_r = _lbfgs_lse_custom(p_start, deadline, min_remaining=20, n_stages=8)
            upd(a_r, sc_r, f"restart_{i}")
            
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 15:
                kkt_dead = min(deadline - 12, time.monotonic() + 25)
                a_r2, sc_r2 = _kkt_refine(a_r.copy(), kkt_dead, max_iter=30)
                upd(a_r2, sc_r2, f"restart_{i}_kkt")

    # === Phase 8: Final aggressive polish (80s) ===
    print(f"\n[Phase 8] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        
        # Ultra-low beta LSE
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.006)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 100}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass
        
        # Final KKT
        if time.monotonic() < deadline - 4:
            a_fin2, sc_fin2 = _kkt_refine(_best_seq.copy(), deadline - 3, max_iter=30)
            upd(a_fin2, sc_fin2, "final_kkt")
        
        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _newton_active_set(_best_seq.copy(), deadline - 2, max_iter=50, tol_frac=0.001)
            upd(a_fin3, sc_fin3, "final_newton")
        
        # Final SG with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _subgrad_polish_custom(_best_seq.copy(), deadline - 1, n_steps=5000, lr_init=0.003)
            upd(a_fin4, sc_fin4, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
