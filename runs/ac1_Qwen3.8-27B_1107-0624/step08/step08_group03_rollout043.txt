```
def _select_indices(c, K, tol_frac=0.004):
    K=int(min(max(1,K), len(c)))
    if len(c)==0: return np.array([],dtype=np.int64)
    cmax=float(c.max())
    cmean=float(c.mean())
    rngval = cmax - cmean
    if rngval < 0: rngval = abs(cmax)
    tol = max(1e-15, tol_frac*max(rngval, 1e-30))
    active = np.flatnonzero(c >= cmax - tol)
    if active.size == 0:
        active=np.array([int(np.argmax(c))], dtype=np.int64)
    if active.size <= K:
        return active.astype(np.int64)
    # sample evenly
    pos=active.astype(np.float64)
    ts=np.linspace(0.0,1.0,K)
    vals=pos[ts]
    sampled=np.rint(vals).astype(np.int64)
    argmax=int(np.argmax(c))
    # ensure argmax included
    if argmax not in sampled: # O(K), for K up to 8192 okay
       sampled[-1]=argmax
    S=np.unique(sampled).astype(np.int64)
```

```
def _newton_solve(rhs, use_fft, params, invK, fA,N4,n, mu, max_cg):
    n=len(rhs)
    x=np.zeros(n)
    r=np.asarray(rhs,dtype=np.float64).copy()
    p=r.copy()
    rs=float(np.dot(r,r))
    base_norm=float(np.dot(rhs,rhs))
    for _ in range(max_cg):
       v=p
       if use_fft:
          hv=_h_fft(v,fA,N4,n)
       else:
          hv=_h_params(v,params,invK)
       hv = hv + mu*v
       pAp=float(np.dot(p,hv))
       if pAp <= 1e-300 or not np.isfinite(pAp): break
       alpha=rs/pAp
       x = x + alpha*p
       r = r - alpha*hv
       rs_new=float(np.dot(r,r))
       if not np.isfinite(rs_new): break
       if rs_new <= 1e-24*(1.0+base_norm): break
       p = r + (rs_new/rs)*p
       rs = rs_new
    return x
```

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over nonnegative step functions.

Strategy:
- Keep several length scales (mainly parent length, 32768, and optionally 65536).
- Use a fixed active-set formulation: for a chosen set of convolution peaks S,
  minimize the average of exactly those peaks in log-space with L-BFGS-B,
  then polish with a projected Newton step for the same fixed set.
- Periodically reselect S from the current full convolution, sampling evenly
  from the active plateau when many peaks are near the maximum.
- Include an auxiliary log-sum-exp L-BFGS method as a secondary smoother.
"""

import numpy as np
import time
from scipy.optimize import minimize

# ------------------------------------------------------------------
# Global state used by top-level scipy callbacks (no closures/lambdas)
# ------------------------------------------------------------------
_G_N = 0
_G_INVK = 1.0
_G_PARAMS = None
_G_FA = None
_G_N4 = 0
_G_USEFFT = False

_G_LO_N = 0
_G_LO_BETA = 1.0


# ------------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------------
def _next_pow2(m):
    k = 1
    while k < m:
        k <<= 1
    return max(2, k)


def _clean_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def _normalize(a):
    a = _clean_array(a)
    if a.size == 0:
        return a
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def _full_a_square(a):
    a = _clean_array(a)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if out.min() < 0.0:
        out = np.maximum(out, 0.0)
    return out


def _score(a):
    a = _clean_array(a)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    max_b = float(_full_a_square(a).max())
    if not np.isfinite(max_b):
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _interp_linear(a, n_new):
    a = _clean_array(a)
    n_old = len(a)
    if n_old == n_new:
        a = a.copy()
    else:
        x0 = np.linspace(0.0, 1.0, n_old)
        x1 = np.linspace(0.0, 1.0, n_new)
        a = np.interp(x1, x0, a)
    a = _clean_array(a)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    return a / s


def _interp_log(a, n_new):
    a = _clean_array(a)
    n_old = len(a)
    if n_old == n_new:
        z = np.log(np.maximum(a, 1e-300))
        z = z - z.mean()
        return _softmax(z)
    z_old = np.log(np.maximum(a, 1e-300))
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    z = np.interp(x1, x0, z_old)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind, rng=None, param=1.0):
    n = int(max(1, n))
    t = np.linspace(0.0, 1.0, n)

    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "inv_sqrt":
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    elif kind == "sqrt_half":
        if n == 1:
            a = np.ones(1, dtype=np.float64)
        else:
            idx = np.arange(1, n, dtype=np.float64)
            ratios = (2.0 * idx - 1.0) / (2.0 * idx)
            a = np.empty(n, dtype=np.float64)
            a[0] = 1.0
            a[1:] = np.cumprod(ratios)
    elif kind == "exp_decay":
        a = np.exp(-float(param) * t)
    elif kind == "ramp":
        a = 0.5 + t
    elif kind == "sine":
        a = 1.0 + 0.3 * np.sin(np.pi * t)
    elif kind == "cos_asym":
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
    elif kind == "bimodal":
        a = 1.0
        a += 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02)
        a += 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.standard_normal(n)
        k = max(2, n // 100)
        kernel = np.ones(k, dtype=np.float64) / float(k)
        a = np.convolve(a, kernel, mode="same")
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-15)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _select_indices(c, K, tol_frac=0.004):
    c = _clean_array(c)
    if c.size == 0:
        return np.array([], dtype=np.int64)
    K = int(min(max(1, K), c.size))

    cmax = float(c.max())
    cmean = float(c.mean())
    spread = cmax - cmean
    if spread < 0.0:
        spread = abs(cmax)
    tol = max(1e-15, float(tol_frac) * max(spread, 1e-30))

    active = np.flatnonzero(c >= cmax - tol)
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int64)

    if active.size <= K:
        return active.astype(np.int64)

    pos = active.astype(np.float64)
    ts = np.linspace(0.0, 1.0, K)
    sampled = np.rint(pos[ts]).astype(np.int64)

    argmax = int(np.argmax(c))
    if not np.any(sampled == argmax):
        sampled[-1] = argmax

    S = np.unique(sampled).astype(np.int64)

    if S.size < K and c.size > 0:
        ntop = min(K, c.size)
        top = np.argpartition(c, ntop - 1)[:ntop]
        S = np.union1d(S, top).astype(np.int64)
        if S.size > K:
            S = S[np.argsort(-c[S])[:K]]

    return S


def _choose_fft_mode(K, n):
    return (K > 128) or (n > 4096 and K * n > 500000)


def _precompute_direct_params(S, n):
    params = []
    S = np.asarray(S, dtype=np.int64).ravel()
    for idx in S:
        idx = int(idx)
        lo = idx - n + 1
        if lo < 0:
            lo = 0
        hi = idx + 1
        if hi > n:
            hi = n
        if lo < hi:
            js = idx - lo
            je = idx - (hi - 1)
            params.append((lo, hi, js, je))
    return params


def _make_freq_for_S(S, n):
    N4 = _next_pow2(3 * n - 2)
    A = np.zeros(2 * n - 1, dtype=np.float64)
    S = np.asarray(S, dtype=np.int64).ravel()
    if S.size > 0:
        A[S] = 1.0 / float(S.size)
    fA = np.fft.rfft(A, N4)
    return fA, N4


def _h_params(a, params, invK):
    n = len(a)
    h = np.zeros(n, dtype=np.float64)
    if invK == 0.0 or len(params) == 0:
        return h
    for lo, hi, js, je in params:
        if je == 0:
            vals = a[js::-1]
        else:
            vals = a[js:je - 1:-1]
        h[lo:hi] += invK * vals
    return h


def _h_fft(a, fA, N4, n):
    fa = np.fft.rfft(a[::-1], N4)
    out = np.fft.irfft(fa * fA, N4)[:2 * n - 1]
    return out[n - 1:2 * n - 1]


def _h_pre(a, use_fft, params, invK, fA, N4, n):
    if use_fft:
        return _h_fft(a, fA, N4, n)
    return _h_params(a, params, invK)


def _prepare_h_pre(S, n):
    S = np.unique(np.asarray(S, dtype=np.int64).ravel())
    m = int(S.size)
    if m == 0:
        return (False, [], 1.0, None, 0)
    invK = 1.0 / float(m)
    if _choose_fft_mode(m, n):
        fA, N4 = _make_freq_for_S(S, n)
        return (True, None, invK, fA, N4)
    params = _precompute_direct_params(S, n)
    return (False, params, invK, None, 0)

def _set_fixed_state(S, n):
    global _G_N, _G_INVK, _G_PARAMS, _G_FA, _G_N4, _G_USEFFT

    S = np.unique(np.asarray(S, dtype=np.int64).ravel())
    n = int(n)
    K = int(S.size)
    _G_N = n
    _G_UsetFF = False

    if K == 0:
        _G_INVK = 1.0
        _G_PARAMS = []
        _G_FA = None
        _G_N4 = 0
        _G_USEFFT = False
        return

    _G_INVK = 1.0 / float(K)

    if _choose_fft_mode(K, n):
        _G_FA, _G_N4 = _make_freq_for_S(S, n)
        _G_USEFFT = True
        _G_PARAMS = None
    else:
        _G_PARAMS = _precompute_direct_params(S, n)
        _G_USEFFT = False
        _G_FA = None
        _G_N4 = 0


def _h_state(a):
    n = _G_N
    if _G_USEFFT:
        return _h_fft(a, _G_FA, _G_N4, n)
    return _h_params(a, _G_PARAMS, _G_INVK)


def _fixed_calc(z):
    n = int(_G_N)
    if n <= 0:
        return (np.inf, np.array([], dtype=np.float64))

    a = _softmax(z)
    h = _h_state(a)
    W = float(np.dot(a, h))
    if not np.isfinite(W):
        return (np.inf, np.zeros(n, dtype=np.float64))

    f = 2.0 * float(n) * W
    g = 4.0 * float(n) * a * (h - W)
    gm = float(np.mean(g))
    if np.isfinite(gm):
        g = g - gm
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

    return (float(f), g)


def _fixed_fun(z):
    f, g = _fixed_calc(z)
    return (f, g)


def _fixed_fun_obj(z):
    f, g = _fixed_calc(z)
    return f


def _fixed_fun_jac(z):
    f, g = _fixed_calc(z)
    return g


def _fixed_lbfgs_static_maxiter(n):
    if n <= 2048:
        return 2500
    if n <= 8192:
        return 1200
    if n <= 16384:
        return 700
    if n <= 32768:
        return 350
    if n <= 65536:
        return 200
    return 100


def _run_fixed_lbfgs(z, S, budget_s, deadline):
    z = _clean_array(z)
    n = len(z)
    if n == 0 or budget_s < 3.0:
        return z

    _set_fixed_state(S, n)

    t0 = time.monotonic()
    try:
        obj, grad = _fixed_calc(z)
    except Exception:
        return z
    tcall = max(1e-6, time.monotonic() - t0)

    if not np.isfinite(obj):
        return z

    max_iter_static = _fixed_lbfgs_static_maxiter(n)
    remaining = max(0.0, float(deadline) - time.monotonic())
    limit = min(float(budget_s), remaining)
    max_iter = min(int(max_iter_static), max(10, int(limit / max(tcall * 4.0, 1e-9))))
    if max_iter < 10:
        return z

    options = {
        "maxiter": int(max_iter),
        "ftol": 1e-16,
        "gtol": 1e-8,
        "maxcor": 80,
        "maxls": 50,
    }

    newz = None
    try:
        res = minimize(_fixed_fun, z, method="L-BFGS-B", jac=True, options=options)
        if res.x is not None:
            newz = res.x
    except Exception:
        try:
            res = minimize(_fixed_fun_obj, z, jac=_fixed_fun_jac,
                           method="L-BFGS-B", options=options)
            if res.x is not None:
                newz = res.x
        except Exception:
            newz = None

    if newz is None:
        return z

    newz = _clean_array(newz)
    if not np.all(np.isfinite(newz)):
        return z
    return newz

def _newton_solve(rhs, use_fft, params, invK, fA, N4, n, mu, max_cg):
    rhs = _clean_array(rhs)
    if rhs.size == 0:
        return rhs
    rs0 = float(np.dot(rhs, rhs))
    if not np.isfinite(rs0) or rs0 <= 0.0:
        return np.zeros_like(rhs)

    x = np.zeros(n, dtype=np.float64)
    r = rhs.copy()
    p = r.copy()
    rs = rs0

    tol2 = 1e-24 * (1.0 + rs0)

    for _ in range(int(max_cg)):
        hv = _h_pre(p, use_fft, params, invK, fA, N4, n)
        hv = hv + mu * p
        pAp = float(np.dot(p, hv))
        if not np.isfinite(pAp) or pAp <= 1e-300:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * hv
        rs_new = float(np.dot(r, r))
        if not np.isfinite(rs_new):
            break
        if rs_new <= tol2:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new

    return x


def _newton_max_iter(n):
    if n <= 4096:
        return 90
    if n <= 8192:
        return 60
    if n <= 16384:
        return 45
    if n <= 32768:
        return 30
    if n <= 65536:
        return 18
    return 12


def _newton_cg_limits(n):
    if n <= 4096:
        return 80, 50
    if n <= 16384:
        return 55, 35
    if n <= 32768:
        return 35, 25
    return 25, 20


def _newton_fixed(a_init, S, deadline, max_iter=50, tol=1e-5):
    a = _normalize(a_init)
    n = len(a)
    S = np.unique(np.asarray(S, dtype=np.int64).ravel())
    if S.size == 0 or n == 0:
        return a

    use_fft, params, invK, fA, N4 = _prepare_h_pre(S, n)
    best_a = a.copy()
    best_sc = _score(a)

    max_iter = int(max(1, max_iter))
    cg_main, cg_sum = _newton_cg_limits(n)
    ones = np.ones(n, dtype=np.float64)

    for it in range(max_iter):
        if time.monotonic() > float(deadline):
            break

        h = _h_pre(a, use_fft, params, invK, fA, N4, n)
        f = float(np.dot(a, h))
        if not np.isfinite(f):
            break

        g = h - h.mean()
        gnorm = float(np.linalg.norm(g))
        scale = max(1.0, float(np.max(np.abs(h))) if h.size else 1.0)
        if gnorm < 1e-12 * scale:
            break

        mu = max(1e-8, 1e-6 * max(1e-12, f) / float(S.size))

        x = _newton_solve(g, use_fft, params, invK, fA, N4, n, mu, cg_main)
        y = _newton_solve(ones, use_fft, params, invK, fA, N4, n, mu, cg_sum)

        sx = float(x.sum())
        sy = float(y.sum())
        if abs(sy) < 1e-15:
            dx = -x
        else:
            dx = -x + (sx / sy) * y
        dx = dx - dx.mean()

        neg = dx < -1e-14
        t = 1.0
        if neg.any():
            denom = -dx[neg]
            ratios = a[neg] / denom
            ratios = ratios[np.isfinite(ratios) & (ratios > 0.0)]
            if ratios.size > 0:
                t = min(t, 0.99 * float(np.min(ratios)))
        t = max(t, 1e-10)

        accepted = False
        for _ls in range(25):
            at = a + t * dx
            if np.min(at) < -1e-9:
                t *= 0.5
                continue
            at = np.maximum(at, 0.0)
            s = float(at.sum())
            if not np.isfinite(s) or s < 1e-15:
                t *= 0.5
                continue
            at = at / s
            ht = _h_pre(at, use_fft, params, invK, fA, N4, n)
            ft = float(np.dot(at, ht))

            if (ft < f - 1e-14) or (_ls == 24):
                if ft <= f + 1e-10:
                    a = at
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = 0.1
            for _ls in range(15):
                at = a - t * g
                if np.min(at) < -1e-9:
                    t *= 0.5
                    continue
                at = np.maximum(at, 0.0)
                s = float(at.sum())
                if not np.isfinite(s) or s < 1e-15:
                    t *= 0.5
                    continue
                at = at / s
                ht = _h_pre(at, use_fft, params, invK, fA, N4, n)
                ft = float(np.dot(at, ht))
                if ft < f - 1e-15:
                    a = at
                    accepted = True
                    break
                t *= 0.5

            if not accepted:
                break

        if it % 5 == 0 or it == max_iter - 1:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()

    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        return a
    return best_a

# Auxiliary log-sum-exp L-BFGS
# ------------------------------------------------------------------
def _set_lo_state(n, beta):
    global _G_LO_N, _G_LO_BETA
    _G_LO_N = int(n)
    _G_LO_BETA = float(beta)


def _lo_calc(z):
    n = int(_G_LO_N)
    beta = float(_G_LO_BETA)
    a = _softmax(z)
    c = _full_a_square(a)
    cmax = float(c.max())

    if beta <= 0.0 or not np.isfinite(beta):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * float(n) * cmax
    else:
        shifted = (c - cmax) / beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s = float(e.sum())
        if not np.isfinite(s) or s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            obj = 2.0 * float(n) * cmax
        else:
            w = e / s
            obj = 2.0 * float(n) * (cmax + beta * float(np.log(s)))

    N4 = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N4)
    fa = np.fft.rfft(a[::-1], N4)
    out = np.fft.irfft(fw * fa, N4)[:2 * n - 1]
    h = out[n - 1:2 * n - 1]

    W = float(np.dot(w, c))
    g = 4.0 * float(n) * a * (h - W)
    gm = float(np.mean(g))
    if np.isfinite(gm):
        g = g - gm
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

    if not np.isfinite(obj):
        return (np.inf, np.zeros(n, dtype=np.float64))
    return (float(obj), g)


def _lo_fun(z):
    f, g = _lo_calc(z)
    return (f, g)


def _lo_fun_obj(z):
    f, g = _lo_calc(z)
    return f


def _lo_fun_jac(z):
    f, g = _lo_calc(z)
    return g


def _logsum_static_maxiter(n):
    if n <= 4096:
        return 2500
    if n <= 16384:
        return 1200
    if n <= 32768:
        return 500
    if n <= 65536:
        return 250
    return 120


def _run_logsum_lbfgs(a_init, deadline, reserve=30.0):
    a = _normalize(a_init)
    n = len(a)
    if n == 0:
        return a

    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()

    c = _full_a_square(a)
    cmax = float(c.max())
    cmean = float(c.mean())
    base = max(cmax * 0.05, (cmax - cmean) * 0.25, cmax * 1e-4, 1e-12)
    betas = base * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])

    for beta in betas:
        if time.monotonic() > float(deadline) - float(reserve):
            break

        _set_lo_state(n, beta)
        t0 = time.monotonic()
        try:
            obj, grad = _lo_calc(z)
        except Exception:
            break
        tcall = max(1e-6, time.monotonic() - t0)

        if not np.isfinite(obj):
            break

        max_iter_static = _logsum_static_maxiter(n)
        remaining = max(0.0, float(deadline) - float(reserve) - time.monotonic())
        max_iter = min(int(max_iter_static), max(20, int(remaining / max(tcall * 3.0, 1e-9))))
        if max_iter < 10:
            break

        options = {
            "maxiter": int(max_iter),
            "ftol": 1e-16,
            "gtol": 1e-10,
            "maxcor": 70,
            "maxls": 60,
        }

        newz = None
        try:
            res = minimize(_lo_fun, z, method="L-BFGS-B", jac=True, options=options)
            if res.x is not None:
                newz = res.x
        except Exception:
            try:
                res = minimize(_lo_fun_obj, z, jac=_lo_fun_jac,
                               method="L-BFGS-B", options=options)
                if res.x is not None:
                    newz = res.x
            except Exception:
                newz = None

        if newz is None or not np.all(np.isfinite(newz)):
            break

        a_new = _softmax(newz)
        sc_new = _score(a_new)

        if np.isfinite(sc_new) and sc_new < best_sc - 1e-14:
            z = newz
            a = a_new
            best_a = a.copy()
            best_sc = sc_new
        else:
            # A smaller beta may still help in rare plateaus, but normally stop.
            break

    return best_a

# Main fixed-active-set optimizer

def _k_schedule(n):
    if n <= 2048:
        Ks = [8, 32, 128, 512]
    elif n <= 8192:
        Ks = [16, 64, 256, 1024]
    elif n <= 32768:
        Ks = [32, 128, 512, 2048]
    elif n <= 65536:
        Ks = [64, 256, 1024, 4096]
    else:
        Ks = [128, 512, 2048, 8192]

    c_len = max(1, 2 * n - 1)
    Ks = [max(1, min(int(k), c_len)) for k in Ks]

    uniq = []
    for k in Ks:
        if k not in uniq:
            uniq.append(k)
    return uniq


def _optimize_fixed(a_init, deadline, reserve=30.0):
    a = _normalize(a_init)
    n = len(a)
    if n == 0:
        return a

    best_a = a.copy()
    best_sc = _score(a)
    cur_a = a.copy()
    cur_sc = best_sc

    z = np.log(np.maximum(cur_a, 1e-300))
    z = z - z.mean()

    Ks = _k_schedule(n)
    stage = 0
    max_cycles = 3
    no_improve = 0

    while True:
        now = time.monotonic()
        if now > float(deadline) - float(reserve):
            break
        total = float(deadline) - float(reserve) - now
        if total < 12.0:
            break

        if stage >= max_cycles * len(Ks):
            break

        K = Ks[min(stage, len(Ks) - 1)]
        c = _full_a_square(cur_a)
        S = _select_indices(c, K, 0.004)

        if stage < len(Ks):
            stages_left = len(Ks) - stage
        else:
            stages_left = 1
        stage_budget = total / float(max(1, stages_left))
        if stage_budget < 8.0:
            break

        print(f"    fixed-set stage={stage}, n={n}, K={K}, budget={stage_budget:.0f}s")

        improved_this_stage = False

        lb_budget = 0.5 * stage_budget
        if lb_budget > 8.0 and time.monotonic() + lb_budget < float(deadline) - float(reserve):
            zb = _run_fixed_lbfgs(z, S, lb_budget, float(deadline) - float(reserve))
            if zb is not None and np.all(np.isfinite(zb)):
                a_new = _softmax(zb)
                sc_new = _score(a_new)
                if np.isfinite(sc_new):
                    if sc_new < best_sc - 1e-15:
                        best_a = a_new.copy()
                        best_sc = sc_new
                        improved_this_stage = True
                    if sc_new < cur_sc - 1e-12 or (sc_new < cur_sc + 0.0005 and sc_new < best_sc + 0.005):
                        cur_a = a_new.copy()
                        cur_sc = sc_new
                        z = np.log(np.maximum(cur_a, 1e-300))
                        z = z - z.mean()
                        if sc_new < best_sc + 0.005:
                            improved_this_stage = True

        newton_deadline = now + stage_budget - 4.0
        if newton_deadline > time.monotonic() + 10.0:
            c_now = _full_a_square(cur_a)
            S2 = _select_indices(c_now, K, 0.003)
            a_new = _newton_fixed(cur_a, S2, newton_deadline,
                                  _newton_max_iter(n), 1e-5)
            sc_new = _score(a_new)
            if np.isfinite(sc_new):
                if sc_new < best_sc - 1e-15:
                    best_a = a_new.copy()
                    best_sc = sc_new
                    improved_this_stage = True
                if sc_new < cur_sc - 1e-12 or (sc_new < cur_sc + 0.0005 and sc_new < best_sc + 0.005):
                    cur_a = a_new.copy()
                    cur_sc = sc_new
                    z = np.log(np.maximum(cur_a, 1e-300))
                    z = z - z.mean()
                    improved_this_stage = True

        if improved_this_stage:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break

        stage += 1

    sc_best = _score(best_a)
    if np.isfinite(sc_best) and sc_best < best_sc:
        best_sc = sc_best
    return _normalize(best_a)

def _record(state, cand, label, t0):
    if cand is None:
        return
    seq = _clean_array(cand)
    if seq.size == 0:
        return
    s = float(seq.sum())
    if not np.isfinite(s) or s < 0.01:
        return
    sc = _score(seq)
    if np.isfinite(sc) and sc < state["score"]:
        prev = state["score"]
        state["score"] = float(sc)
        state["seq"] = seq.copy()
        print(f"  NEW BEST {sc:.10f} n={seq.size} [{label}] "
              f"prev={prev:.10f} improve={prev - sc:.3e} t={time.monotonic() - t0:.1f}s")


def _compute_phase_end(deadline, target_budget, reserve_after):
    now = time.monotonic()
    rem = max(0.0, float(deadline) - now)
    budget = min(float(target_budget), max(0.0, rem - float(reserve_after)))
    if budget < 25.0:
        return now
    return now + budget

def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    state = {"seq": None, "score": np.inf}

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = _clean_array(parent_raw)
            if float(parent_raw.sum()) > 0.01:
                _record(state, parent_raw, "parent", t0)
                if state["seq"] is not None:
                    parent = _normalize(state["seq"])
    except Exception:
        parent = None

    if parent is None:
        parent = _make_profile(16384, "inv_sqrt", rng)
        _record(state, parent, "default_init", t0)

    # Use a manageable parent length for the first medium-scale phase.
    n_parent = len(parent)
    if n_parent > 16384:
        base = _interp_linear(parent, 16384)
    elif n_parent < 2048:
        base = _interp_linear(parent, 2048)
    else:
        base = parent

    print(f"Base length={len(base)}, base score={_score(base):.10f}, t={time.monotonic() - t0:.1f}s")

    # Phase 1: optimize at base length.
    phase_end = _compute_phase_end(deadline, 350.0, 1000.0)
    if phase_end > time.monotonic() + 20.0:
        print(f"[Phase 1] fixed-set optimization, n={len(base)}, budget={phase_end - time.monotonic():.0f}s")
        a_opt = _optimize_fixed(base, phase_end, reserve=5.0)
        _record(state, a_opt, "fixed_base", t0)

    # Phase 2: secondary log-sum-exp polish at base or medium length.
    if state["seq"] is not None and len(state["seq"]) <= 16384:
        phase_end = _compute_phase_end(deadline, 220.0, 900.0)
        if phase_end > time.monotonic() + 20.0:
            print(f"[Phase 2] log-sum-exp, n={len(state['seq'])}, budget={phase_end - time.monotonic():.0f}s")
            a_lo = _run_logsum_lbfgs(state["seq"], phase_end, reserve=10.0)
            _record(state, a_lo, "logsum_base", t0)

    # Phase 3: upscale to 32768 and optimize aggressively.
    if state["seq"] is not None and time.monotonic() < float(deadline) - 700.0:
        n_large = 32768
        cands = [
            ("linear", _interp_linear(state["seq"], n_large)),
            ("log", _interp_log(state["seq"], n_large)),
        ]
        best_c = None
        best_csc = np.inf
        for lab, c in cands:
            sc = _score(c)
            print(f"  32768 init {lab}: {sc:.10f}")
            if np.isfinite(sc) and sc < best_csc:
                best_csc = sc
                best_c = c

        if best_c is not None:
            phase_end = _compute_phase_end(deadline, 950.0, 350.0)
            if phase_end > time.monotonic() + 40.0:
                print(f"[Phase 3] fixed-set optimization, n=32768, budget={phase_end - time.monotonic():.0f}s")
                a_large = _optimize_fixed(best_c, phase_end, reserve=20.0)
                _record(state, a_large, "fixed_32768", t0)

    # Phase 4: optionally upscale to 65536 if enough time remains.
    if state["seq"] is not None and time.monotonic() < float(deadline) - 350.0:
        n_xl = 65536
        c1 = _interp_linear(state["seq"], n_xl)
        sc1 = _score(c1)
        c2 = _interp_log(state["seq"], n_xl)
        sc2 = _score(c2)
        print(f"  65536 init linear={sc1:.10f}, log={sc2:.10f}")
        cand = c1 if np.isfinite(sc1) and sc1 <= sc2 else c2

        phase_end = _compute_phase_end(deadline, 450.0, 180.0)
        if phase_end > time.monotonic() + 40.0:
            print(f"[Phase 4] fixed-set optimization, n=65536, budget={phase_end - time.monotonic():.0f}s")
            a_xl = _optimize_fixed(cand, phase_end, reserve=15.0)
            _record(state, a_xl, "fixed_65536", t0)

    # Final polish on the current best at its current length.
    if state["seq"] is not None and time.monotonic() < float(deadline) - 20.0:
        phase_end = float(deadline) - 10.0
        if phase_end > time.monotonic() + 10.0:
            print(f"[Final] polish n={len(state['seq'])}, budget={phase_end - time.monotonic():.0f}s")
            a_fin = _optimize_fixed(state["seq"], phase_end, reserve=3.0)
            _record(state, a_fin, "final_polish", t0)

    if state["seq"] is not None:
        final = _clean_array(state["seq"])
        s = float(final.sum())
        if not np.isfinite(s) or s < 0.01:
            final = _make_profile(256, "uniform", rng)
        print(f"FINAL score={_score(final):.10f}, n={len(final)}, t={time.monotonic() - t0:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform", rng)
    print(f"FALLBACK score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
