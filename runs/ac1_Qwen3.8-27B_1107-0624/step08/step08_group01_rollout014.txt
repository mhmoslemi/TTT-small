import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _topk_grad(a, K=10):
    """Compute gradient for reducing top-K peaks simultaneously."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Select top-K peaks (within a tolerance of the max)
    tol = 0.01 * c_max
    active = c >= (c_max - tol)
    active_count = int(active.sum())
    if active_count == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        active_count = 1
    # Cap at K
    if active_count > K:
        top_indices = np.argsort(c)[::-1][:K]
        active = np.zeros_like(c, dtype=bool)
        active[top_indices] = True
        active_count = K
    
    w = np.zeros_like(c)
    w[active] = 1.0 / active_count
    
    # Gradient: d(sum w[k]*c[k])/d(a[i]) = sum_k w[k]*a[k-i] (with appropriate bounds)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[len(a)-1:2*len(a)-1]
    return 2.0 * h, c, c_max

def _make_freq_init(n, rng, N=None):
    """Frequency-domain inspired initialization."""
    if N is None:
        N = _next_pow2(2*n)
    # Random flat spectrum with Hermitian symmetry
    X = np.zeros(N, dtype=complex)
    for k in range(N//2 + 1):
        phase = rng.random() * 2 * np.pi
        mag = 1.0
        # Ensure Hermitian: X[k] and X[N-k] are conjugates
        X[k] = mag * np.exp(1j * phase)
        if k > 0 and k < N//2:
            X[N-k] = np.conj(X[k])
    # IFFT to get time domain
    a_complex = np.fft.ifft(X)
    a = np.real(a_complex[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _make_sine_init(n, rng):

def _peak_shave_direct(a, deadline, n_trials=100):
    """Direct peak shaving by reducing top contributors to the peak."""
    n = len(a)
    best_a = a.copy()

    rng = np.random.default_rng(777)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Find contributors to peak
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_lo >= i_hi:
            continue
        
        i_range = np.arange(i_lo, i_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            continue
        
        # Reduce top contributors
        n_reduce = max(1, int(len(contrib) * 0.05))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
        
        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.05
            a_try[idx] -= red
            reduction += red
        
        # Distribute to non-contributing positions
        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False
        
        if low_mask.sum() == 0:
            continue
        
        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
    
    return best_a, best_sc

"""
Multi-start optimization for flat auto-convolution sequences.
Uses diverse initializations (flat-top windows, frequency-domain random phase,
binary patterns, parent), momentum gradient descent with adaptive top-K peak
reduction, L-BFGS log-sum-exp refinement, and direct peak shaving.
Explores multiple sequence lengths to avoid local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_gradient(a, top_k=10, tol_frac=0.01):
    """Compute the gradient direction for reducing top-K peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    active_count = int(active.sum())
    if active_count == 0:
        idx = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        active_count = 1
    if active_count > top_k:
        top_indices = np.argsort(c)[::-1][:top_k]
        active = np.zeros_like(c, dtype=bool)
        active[top_indices] = True
        active_count = top_k
    w = np.zeros_like(c)
    w[active] = 1.0 / active_count
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h, c, c_max


def _momentum_gd(a_init, deadline, max_iter=5000, top_k=12, lr=0.03, momentum=0.7):

        return a_init, _score(a_init)

    no_improve = 0

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_refine(a_init, deadline, min_remaining=10.0):

    n = len(a_init)
    current_a = a_init.copy()

    z = np.log(np.maximum(current_a, 1e-30))

    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.003)
    betas = base_beta * np.array([1.0, 0.4, 0.15, 0.06, 0.025, 0.01, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

                return _lse_obj(x, beta, n)
            def gfun(x):
                return _lse_grad(x, beta, n)

            if s_new < 1e-15:

def _make_flat_top_window(n, alpha=0.2156):
    """Flat-top window (Nuttall-like) for initialization."""
    t = np.arange(n)
    a = (1.0 - alpha) - alpha * np.cos(2.0 * np.pi * t / n) \
        - 0.1141 * np.cos(4.0 * np.pi * t / n) + 0.0115 * np.cos(6.0 * np.pi * t / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_blackman_harris(n):
    """Blackman-Harris window."""
    t = np.arange(n)
    w = 0.35875 - 0.48829 * np.cos(2 * np.pi * t / n) \
        + 0.14128 * np.cos(4 * np.pi * t / n) - 0.01168 * np.cos(6 * np.pi * t / n)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def _make_binary_init(n, rng, density=0.5):
    """Two-level initialization with smoothing."""
    binary = (rng.random(n) < density).astype(float)
    kernel = np.ones(5) / 5.0
    smooth = np.convolve(binary, kernel, mode='same')
    smooth = np.maximum(smooth, 0.005)
    return smooth / smooth.sum()


def _make_freq_init(n, rng):
    """Frequency-domain random phase with flat magnitude."""
    N = _next_pow2(2 * n)
    X = np.zeros(N, dtype=complex)
    phases = rng.random(N // 2 + 1) * 2 * np.pi
    for k in range(N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        if k > 0 and k < N // 2:
            X[N - k] = np.conj(X[k])
    a_complex = np.fft.ifft(X)
    a = np.real(a_complex[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_modulated_init(n, rng):
    """Cosine-modulated initialization with random frequencies."""
    n_freqs = rng.integers(2, 6)
    a = np.ones(n)
    for _ in range(n_freqs):
        f = rng.integers(1, n // 4)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.random() * 2 * np.pi
        a += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _peak_shave(a, deadline, n_trials=150):
    """Direct peak shaving targeting top contributors to the maximum."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Target multiple peaks
        top_peaks = np.argsort(c)[::-1][:3]
        for k_star in top_peaks:
            k_star = int(k_star)
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            if i_lo >= i_hi:
                continue
            i_range = np.arange(i_lo, i_hi + 1)
            j_range = k_star - i_range
            contrib = a[i_range] * a[j_range]
            total = contrib.sum()
            if total < 1e-15:
                continue
            n_reduce = max(1, int(len(contrib) * 0.04))
            top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
            a_try = a.copy()
            reduction = 0.0
            for idx in top_idx:
                red = a_try[idx] * 0.04
                a_try[idx] -= red
                reduction += red
            low_mask = np.ones(n, dtype=bool)
            for idx in top_idx:
                low_mask[idx] = False
                jv = k_star - idx
                if 0 <= jv < n:
                    low_mask[jv] = False
            if low_mask.sum() == 0:
                continue
            a_try[low_mask] += reduction / low_mask.sum()
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start flat auto-convolution optimizer.
    - Diverse initializations: flat-top windows, Blackman-Harris, frequency-domain,
      binary patterns, cosine-modulated, and parent.
    - Momentum GD with adaptive top-K for rapid convergence.
    - L-BFGS log-sum-exp refinement for fine polishing.
    - Direct peak shaving for final improvements.
    - Multiple sequence lengths explored.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={len(parent)}, score={best_score:.10f}")

    # Sequence lengths to explore
    n_list = [12288, 8192, 16384]
    if parent is not None and len(parent) not in n_list:
        n_list.insert(0, len(parent))

    # === Phase 1: Optimize parent with momentum GD ===
    if parent is not None and time.monotonic() < deadline - 700:
        print(f"\n[P1] Momentum GD on parent (n={len(parent)})")
        p1_dl = min(deadline - 700, t_start + 250)
        a1, sc1 = _momentum_gd(parent, p1_dl, max_iter=5000, top_k=15, lr=0.035)
        if sc1 < best_score:
            best_a = a1.copy()
            best_score = sc1
            print(f"  BEST: {best_score:.10f} [mom_gd_parent] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS refinement on current best ===
    if best_a is not None and time.monotonic() < deadline - 500:
        print(f"\n[P2] L-BFGS refine (n={len(best_a)})")
        p2_dl = min(deadline - 500, t_start + 450)
        a2 = _lbfgs_refine(best_a, p2_dl, min_remaining=15.0)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_a = a2.copy()
            best_score = sc2
            print(f"  BEST: {best_score:.10f} [lbfgs_refine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Diverse initializations and optimization ===
    print(f"\n[P3] Diverse initializations")
    for n in sorted(n_list):
        if time.monotonic() > deadline - 150:
            break
        if best_a is not None and len(best_a) == n:
            continue

        inits = []
        inits.append(("flat_top", _make_flat_top_window(n)))
        inits.append(("bh", _make_blackman_harris(n)))
        inits.append(("binary1", _make_binary_init(n, rng, 0.5)))
        inits.append(("freq1", _make_freq_init(n, rng)))
        inits.append(("mod1", _make_modulated_init(n, rng)))
        inits.append(("binary2", _make_binary_init(n, rng, 0.4)))
        inits.append(("freq2", _make_freq_init(n, rng)))

        if best_a is not None and len(best_a) != n:
            a_res = _resample(best_a, n)
            inits.append(("resampled", a_res))

        # Score all inits
        scored = []
        for label, a_init in inits:
            sc = _score(a_init)
            scored.append((sc, label, a_init))
        scored.sort(key=lambda x: x[0])
        print(f"  n={n}: best inits: {[(l, f'{s:.6f}') for s, l, _ in scored[:4]]}")

        # Optimize top 2 inits
        n_opt = 2
        remaining_for_phase = deadline - time.monotonic() - 100
        time_per = max(20.0, min(80.0, remaining_for_phase / (n_opt * len(n_list))))
        for i in range(min(n_opt, len(scored))):
            if time.monotonic() > deadline - 100:
                break
            sc_i, label, a_i = scored[i]
            init_dl = min(deadline - 100, time.monotonic() + time_per)
            if init_dl - time.monotonic() < 10:
                continue
            a_opt, sc_opt = _momentum_gd(a_i, init_dl, max_iter=3000, top_k=12, lr=0.03)
            # Quick L-BFGS on the result
            if time.monotonic() < deadline - 80:
                a_opt2 = _lbfgs_refine(a_opt, min(deadline - 60, time.monotonic() + 30), min_remaining=5.0)
                sc_opt2 = _score(a_opt2)
                if sc_opt2 < sc_opt:
                    a_opt, sc_opt = a_opt2, sc_opt2
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  BEST: {best_score:.10f} (n={n}, {label}) [diverse] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak shaving ===
    print(f"\n[P4] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_a is not None and time.monotonic() < deadline - 80:
        p4_dl = min(deadline - 80, t_start + 900)
        a4, sc4 = _peak_shave(best_a, p4_dl, n_trials=200)
        if sc4 < best_score:
            best_a = a4.copy()
            best_score = sc4
            print(f"  BEST: {best_score:.10f} [peak_shave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final L-BFGS polish ===
    print(f"\n[P5] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_a is not None and time.monotonic() < deadline - 20:
        a5 = _lbfgs_refine(best_a, deadline - 10, min_remaining=5.0)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_a = a5.copy()
            best_score = sc5
            print(f"  BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final momentum GD + peak shave ===
    if best_a is not None and time.monotonic() < deadline - 5:
        a6, sc6 = _momentum_gd(best_a, deadline - 3, max_iter=2000, top_k=8, lr=0.015)
        if sc6 < best_score:
            best_a = a6.copy()
            best_score = sc6
            print(f"  BEST: {best_score:.10f} [final_gd] t={time.monotonic()-t_start:.1f}s")
        if time.monotonic() < deadline - 2:
            a7, sc7 = _peak_shave(best_a, deadline - 1, n_trials=30)
            if sc7 < best_score:
                best_a = a7.copy()
                best_score = sc7
                print(f"  BEST: {best_score:.10f} [final_shave] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
