2. A refined active-set Newton that precisely solves the KKT conditions at the identified active set
3. Direct peak-targeting descent for the single worst convolution peak
4. Better exploitation of the Toeplitz/banded structure of the gradient
5. Multiple restart strategies with fine polishing

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Extended L-BFGS with ultra-fine beta continuation
2. Refined active-set Newton with precise KKT solving
3. Direct peak-targeting projected descent
4. Multi-n exploration and perturbed restarts
5. Iterative active-set refinement loop
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_conv_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    for i in range(i_lo, i_hi + 1):
        j = k - i
        g[i] += a[j]
        if j != i:
            g[j] += a[i]
    return g


def _grad_conv_k_fast(a, k, n):
    """FFT-based gradient of conv(a,a)[k]."""
    # grad[i] = a[k-i] + a[i] if 0 <= k-i < n, else 0
    # This is: shift(a, k) [reversed] + a[k] (with masking)
    # Actually: g[i] = a[k-i] + a[i] for i in [max(0,k-n+1), min(k,n-1)]
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    g = np.zeros(n)
    indices = np.arange(i_lo, i_hi + 1)
    g[indices] = a[indices] + a[k - indices]
    return g


def _corr_a_A(a, A_vec, n):
    """Compute sum_k A_vec[k] * a[k-i] + A_vec_sum * a[i] (the combined gradient)."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _active_set_newton_v2(a_init, deadline, max_iter=500):
    """Refined active-set Newton with precise KKT solving."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_list = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_dir_score = np.inf
        best_t = 1.0
        best_A_vec = None

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 6 * n:
                if m > 6 * n:
                    top_idx = np.argsort(c)[::-1][:6 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                else:
                    continue

            A_vec = active.astype(np.float64)

            # Gradient of the smoothed max (uniform weights on active set)
            g = _corr_a_A(a, A_vec, n)
            g = g / m
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            # Hessian approximation: (2/m) * T(A_vec) + mu * I
            mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

            def hess_vp(v):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            u = _cg_solve(hess_vp, g_t, max_cg=300)
            w = _cg_solve(hess_vp, np.ones(n), max_cg=300)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = dx - np.mean(dx)

            # Line search
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            t = min(1.0, t_max)
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

            if sc_try < best_dir_score:
                best_dir_score = sc_try
                best_dir = dx
                best_t = t
                best_A_vec = A_vec.copy()

        if best_dir is None:
            # Fallback: steepest descent on top peak
            k_max = int(np.argmax(c))
            g = _grad_conv_k_fast(a, k_max, n)
            g_t = g - np.mean(g)
            if np.linalg.norm(g_t) < 1e-14:
                break
            t = 0.01
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    break
                t *= 0.5
            else:
                break
            a = a_try
        else:
            # Backtracking line search
            f_old = c_max
            accepted = False
            t = best_t
            for ls in range(60):
                a_try = a + t * best_dir
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= f_old - 1e-13 * t or ls >= 50:
                    if f_try <= f_old + 1e-11:
                        a = a_try
                        accepted = True
                        break
                t *= 0.5

            if not accepted:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _peak_descent(a_init, deadline, n_steps=5000):
    """Direct projected descent targeting the single worst peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        c_max = float(c[k_max])

        # Gradient of conv[k_max]
        g = _grad_conv_k_fast(a, k_max, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        t = 0.005 / (1.0 + 0.001 * step)
        for ls in range(40):
            a_try = a - t * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= c_max - 1e-15:
                a = a_try
                break
            t *= 0.5
        else:
            if t < 1e-10:
                break

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def _run_lbfgs_extended(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with extended ultra-fine beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Extended beta schedule: go much smaller
    betas = max_c * np.array([
        0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
        0.000001, 0.0000005, 0.0000002, 0.0000001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.005 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 4))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    L-BFGS beta_idx={i}, beta={beta:.2e}, sc={current_sc:.12f}, "
                  f"iters={result.nit}, t={time.monotonic():.1f}")
        except Exception as e:
            print(f"    L-BFGS error at beta {beta:.2e}: {e}")
            continue

    return current_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_new = _softmax(result.x)
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        continue
                    a_new = a_new / s_new
                    sc_new = _score(a_new)
                    if sc_new < current_sc - 1e-14:
                        current_a = a_new
                        current_sc = sc_new
                        z = np.log(np.maximum(a_new, 1e-30))
                        z = z - z.mean()
                except Exception:
                    pass
            update_best(current_a, _score(current_a), f"restart_s{sigma}")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        # Very small beta L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        for beta_frac in [0.00005, 0.00002, 0.00001, 0.000005, 0.000002]:
            if time.monotonic() > deadline - 2:
                break
            beta_final = mb * beta_frac
            remaining = deadline - 3 - time.monotonic()
            if remaining < 2:
                break
            max_iter = min(5000, max(50, int(remaining / 0.04)))
            try:
                result = minimize(
                    lambda z_local: _lsobj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 80}
                )
                a_fin = _softmax(result.x)
                a_fin = np.maximum(a_fin, 0)
                s_fin = a_fin.sum()
                if s_fin < 1e-15:
                    continue
                a_fin = a_fin / s_fin
                update_best(a_fin, _score(a_fin), f"final_beta{beta_frac}")
                z = np.log(np.maximum(a_fin, 1e-30))
                z = z - z.mean()
            except Exception:
                pass

        # Final active-set Newton
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _active_set_newton_v2(best_seq, deadline - 0.5, max_iter=100)
            update_best(a_fin2, sc_fin2, "final_as")

        # Final peak descent
        if time.monotonic() < deadline - 1:
            a_fin3 = _peak_descent(best_seq, deadline - 0.5, n_steps=2000)
            update_best(a_fin3, _score(a_fin3), "final_pd")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
