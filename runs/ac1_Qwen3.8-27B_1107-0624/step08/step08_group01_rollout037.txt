def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

        return np.inf
    c = _full_convolve(a, a)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _structured_perturb(a, rng, kind="cosine"):
    """Apply structured perturbation."""
    n = len(a)
    t = np.arange(n)
    if kind == "cosine_low":
        # Low frequency cosine
        f = rng.choice([2, 3, 4, 5, 7])
        amp = rng.uniform(0.01, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1 + amp * np.cos(2 * np.pi * f * t / n + phase))
    elif kind == "cosine_high":
        # High frequency cosine
        f = rng.choice([n//4, n//3, n//5, n//6])
        amp = rng.uniform(0.005, 0.02)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1 + amp * np.cos(2 * np.pi * f * t / n + phase))
    elif kind == "taper":
        # Edge tapering
        edge = rng.uniform(0.02, 0.1)
        taper = 1.0
        for i in range(n):
            d = min(i, n - 1 - i)
            if d < edge * n:
                taper *= (d / (edge * n)) ** 0.5
        a_new = a * taper
    elif kind == "random":
        sigma = rng.uniform(0.01, 0.05)
        z = np.log(np.maximum(a, 1e-30)) + sigma * rng.standard_normal(n)
        a_new = _softmax(z)
    else:
        a_new = a.copy()
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _peak_shave_targeted(a, deadline, n_trials=100):
    """Targeted peak shaving: reduce specific elements contributing to the peak."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(42)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Find top contributors to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i]
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue
        
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        
        if total_contrib < 1e-15:
            continue
        
        # Reduce top contributors, redistribute
        n_reduce = max(1, int(len(contrib) * 0.1))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]
        
        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * rng.uniform(0.05, 0.15)
            a_try[idx] -= red
            reduction += red
        
        # Redistribute to elements NOT contributing to the peak
        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False
        
        if low_mask.sum() < n * 0.1:
            low_mask[:] = True
            for idx in top_idx:
                low_mask[idx] = False
        
        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try
            a = a_try
    
    return best_a, best_sc

**Yet another idea: Use the "Birkhoff-von Neumann" decomposition.**

Any doubly stochastic matrix is a convex combination of permutation matrices. Similarly, any point in the simplex is a convex combination of basis vectors. The Frank-Wolfe method exploits this.

For our problem, the Frank-Wolfe step would be:
1. Compute gradient g of the objective
2. Find the vertex v = e_k that minimizes <g, v> = g[k] (the smallest component of g)
3. Move toward v: a_new = (1-t) * a + t * e_k

This is equivalent to: reduce the component a[k] that has the smallest gradient (i.e., increasing it the most helps reduce the objective).

Actually, for minimization, we want to move in the direction that decreases the objective. The linear oracle for the simplex gives us the vertex where the gradient is most negative. So we move toward the vertex with the most negative gradient component.

    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()

        if time.monotonic() > deadline - 1:

            best_a = a.copy()

        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        h = 2.0 * wc[n-1:2*n-1]  # gradient of c_max wrt a

        h_proj = h - np.mean(h)
        
        # Linear oracle: find vertex with most negative <h_proj, e_k> = h_proj[k]
        k_min = int(np.argmin(h_proj))
        
        # Frank-Wolfe step size: backtracking or closed-form
        # For the simplex, the optimal step is:
        # t = argmin <h_proj, (1-t)a + t e_k> = argmin t * (h_proj[k] - h_proj @ a)
        # This is t = 1 if h_proj[k] < h_proj @ a, t = 0 otherwise
        # But we use a smaller step for stability
        gap = h_proj @ a - h_proj[k_min]
        if gap < 1e-15:

        t = min(2.0 / (it + 2), 0.5)  # diminishing step
        
        a_new = (1 - t) * a + t * np.zeros(n)
        a_new[k_min] += t
        
        a = a_new
        
        if it % 200 == 0:
            sc = _score(a)
            print(f"    FW it={it}, score={sc:.10f}, best={best_sc:.10f}")
    
    return best_a, best_sc

"""
Step-function optimization using multi-method approach: LSE L-BFGS with careful
annealing, power-sum refinement, Polyak subgradient averaging, targeted peak
shaving, and multi-length exploration. Focuses on escaping local minima through
structured perturbations and diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def gradient(self, z):
        return _lse_grad(z, self.beta, self.n)


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-300:
        return 0.0
    ratios = c / c_max
    ps = float(np.sum(ratios ** p))
    return c_max * ps ** (1.0 / p)


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-300:
        return np.zeros_like(z)
    ratios = c / c_max
    c_pw = np.power(ratios, p - 1) * c_max ** (p - 1)
    a_rev = a[::-1]
    wc = _full_convolve(c_pw, a_rev)
    h = wc[n - 1:2 * n - 1]
    h = p * h
    W = float(np.dot(h, a))
    grad_z = a * (h - W)
    return grad_z


class _PowerSumProblem:

    def __call__(self, z):

                z = z - z.mean()
                print(f"      p={p}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _subgradient_polyak(a_init, deadline, max_iter=20000):
    """
    Subgradient method with Polyak step size and moving average.
    Uses adaptive active set and tracks both the current best and the average.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.sum()) / len(c)

    a_avg = a.copy()
    count = 0
    tol_frac = 0.02

    for it in range(max_iter):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        c_mean = float(c.sum()) / len(c)

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        gap = c_max - c_mean
        if gap < 1e-16:
            break

        t_p = gap / (c_max + 1e-30)
        t_p = min(t_p, 0.15)
        t_p = max(t_p, 0.001)

        tol_val = tol_frac * c_max
        active = c >= (c_max - tol_val)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = 2.0 * wc[n - 1:2 * n - 1]

        h_proj = h - np.mean(h)

        a_new = a - t_p * h_proj
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        a_avg = ((count * a_avg + a_new) / (count + 1))
        a = a_new
        count += 1

        # Adaptively shrink tolerance
        tol_frac = max(0.001, tol_frac * 0.999)

        if it % 500 == 0:
            sc = _score(a)
            print(f"    subgrad it={it}, cur={sc:.10f}, best={best_sc:.10f}, avg_sc={_score(a_avg):.10f}")

    # Check moving average
    sc_avg = _score(a_avg)
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_a = a_avg.copy()

    return best_a, best_sc


def _peak_shave_targeted(a, deadline, n_trials=300):
    """Targeted peak shaving with proper redistribution."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        n_reduce = max(1, int(len(contrib) * rng.uniform(0.05, 0.15)))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * rng.uniform(0.03, 0.1)
            a_try[idx] -= red
            reduction += red

        low_mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            low_mask[idx] = False
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False

        if low_mask.sum() < n * 0.05:
            low_mask[:] = True
            for idx in top_idx:
                low_mask[idx] = False
            if low_mask.sum() == 0:
                continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try
            a = a_try

    return best_a, best_sc


def _active_set_refine(a_init, deadline, max_iter=5000):
    """Active-set gradient with adaptive tolerance and backtracking."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.05
    tol = 0.01
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            tol = max(0.0005, tol * 0.8)
            lr *= 0.7
            no_improve = 0
            if tol < 0.001 and lr < 0.005:
                break

        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())
        if n_active > 50:
            top_idx = np.argsort(c)[::-1][:50]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 50

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = 2.0 * wc[n - 1:2 * n - 1]
        h_proj = h - np.mean(h)

        gnorm = np.linalg.norm(h_proj)
        if gnorm < 1e-18:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False

        for _ in range(60):
            a_try = a - alpha * h_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve > 300:
                break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _structured_perturb(a, rng, kind):
    n = len(a)
    t = np.arange(n, dtype=float)
    if kind == "cos_low":
        f = int(rng.choice([2, 3, 4, 5, 7, 11]))
        amp = rng.uniform(0.01, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1.0 + amp * np.cos(2.0 * np.pi * f * t / n + phase))
    elif kind == "cos_high":
        f = int(n // int(rng.choice([5, 7, 11, 13])))
        amp = rng.uniform(0.003, 0.015)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a * (1.0 + amp * np.cos(2.0 * np.pi * f * t / n + phase))
    elif kind == "taper":
        edge = rng.uniform(0.02, 0.08)
        d = np.minimum(t, n - 1 - t)
        edge_n = edge * n
        a_new = a.copy()
        mask = d < edge_n
        a_new[mask] = a[mask] * (1.0 - 0.3 * (1.0 - d[mask] / edge_n) ** 2)
    elif kind == "mix_uniform":
        frac = rng.uniform(0.02, 0.08)
        uniform = np.ones(n) / n
        a_new = (1.0 - frac) * a + frac * uniform
    else:
        sigma = rng.uniform(0.01, 0.03)
        z = np.log(np.maximum(a, 1e-30)) + sigma * rng.standard_normal(n)
        a_new = _softmax(z)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining LSE L-BFGS, power-sum refinement,
    Polyak subgradient, active-set polishing, peak shaving, multi-n exploration,
    and structured restarts to minimize the step-function evaluation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, budget_s - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = np.ones(n_parent) / n_parent
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: LSE L-BFGS on parent ===
    print(f"\n[P1] LSE L-BFGS parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1050.0, t_start + 350.0)
    a1 = _lbfgs_lse(best_a, p1_deadline)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1_lse] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc1:.10f} vs {best_score:.10f}")

    # === Phase 2: Power-sum L-BFGS ===
    print(f"\n[P2] Power-sum L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800.0, t_start + 550.0)
    a2 = _lbfgs_power_sum(best_a, p2_deadline)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [P2_powersum] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Subgradient with Polyak ===
    print(f"\n[P3] Subgradient+Polyak (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 600.0, t_start + 800.0)
    a3, sc3 = _subgradient_polyak(best_a, p3_deadline, max_iter=30000)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [P3_subgrad] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Active-set refine ===
    print(f"\n[P4] Active-set refine (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 450.0, t_start + 1000.0)
    a4, sc4 = _active_set_refine(best_a, p4_deadline, max_iter=10000)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [P4_activeset] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak shaving ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 350.0, t_start + 1150.0)
    a5, sc5 = _peak_shave_targeted(best_a, p5_deadline, n_trials=300)
    if sc5 < best_score:
        best_a = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} [P5_peakshave] t={time.monotonic()-t_start:.1f}s")

    # Re-optimize after peak shaving
    if sc5 < best_score + 0.002 and time.monotonic() < deadline - 250:
        p5b_deadline = min(deadline - 220.0, t_start + 1300.0)
        a5b = _lbfgs_lse(best_a, p5b_deadline)
        sc5b = _score(a5b)
        if sc5b < best_score:
            best_a = a5b.copy()
            best_score = sc5b
            print(f"  NEW BEST: {best_score:.10f} [P5b_postshave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 20480]
    if len(best_a) not in n_targets:
        for n_try in n_targets:
            if time.monotonic() > deadline - 150.0:
                break
            if remaining := (deadline - time.monotonic() - 100.0) < 40:
                continue
            a_init = _resample(best_a, n_try)
            sc_init = _score(a_init)
            print(f"  n={n_try}: init score={sc_init:.10f}")
            p6_deadline = time.monotonic() + min(150.0, deadline - time.monotonic() - 80.0)
            a_opt = _lbfgs_lse(a_init, p6_deadline)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P6] t={time.monotonic()-t_start:.1f}s")
            # Quick subgradient polish
            if time.monotonic() < deadline - 80.0:
                p6b = time.monotonic() + min(50.0, deadline - time.monotonic() - 60.0)
                a_pol, sc_pol = _subgradient_polyak(a_opt, p6b, max_iter=5000)
                if sc_pol < best_score:
                    best_a = a_pol.copy()
                    best_score = sc_pol
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P6_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Structured restarts ===
    print(f"\n[P7] Structured restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        kinds = ["cos_low", "cos_high", "taper", "mix_uniform"]
        for kind in kinds:
            if time.monotonic() > deadline - 50.0:
                break
            a_r = _structured_perturb(best_a, rng, kind)
            remaining = min(60.0, deadline - time.monotonic() - 30.0)
            if remaining < 15.0:
                break
            a_opt = _lbfgs_lse(a_r, time.monotonic() + remaining - 5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [P7_{kind}] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  {kind}: {sc_opt:.10f} (no improve)")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        p8_deadline = deadline - 3.0
        a_fin, sc_fin = _active_set_refine(best_a, p8_deadline, max_iter=5000)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P8_activeset] t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 2.0:
            a_sg, sc_sg = _subgradient_polyak(best_a, deadline - 2.0, max_iter=10000)
            if sc_sg < best_score:
                best_a = a_sg.copy()
                best_score = sc_sg
                print(f"  NEW BEST: {best_score:.10f} [P8_subgrad] t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < deadline - 1.0:
            a_ps, sc_ps = _peak_shave_targeted(best_a, deadline - 1.0, n_trials=50)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} [P8_shave] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
