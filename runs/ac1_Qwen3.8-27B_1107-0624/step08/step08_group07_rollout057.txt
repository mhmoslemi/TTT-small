```python
def newton_equalize(a, active_set, max_iter=100):
    n = len(a)
    a = a.copy() / a.sum()
    t = max(_fft_convolve(a, a)[k] for k in active_set)
    
    for it in range(max_iter):
        c = _fft_convolve(a, a)
        
        # Check if any inactive peak exceeds t
        inactive_max = max(c[k] for k in range(2*n-1) if k not in active_set)
        if inactive_max > t * (1 + 1e-6):
            # Add to active set
            k_new = argmax(c[k] for k not in active_set)
            active_set.add(k_new)
            t = max(c[k] for k in active_set)
            continue
        
        # Build the system: J @ delta = -residual
        # where residual = [c[k_i] - t for k_i in active_set] + [sum(a) - 1]
        m = len(active_set)
        J = np.zeros((m+1, n))
        res = np.zeros(m+1)
        
        for i, k in enumerate(active_set):
            J[i, :] = subgradient(c[k])  # 2 * a shifted
            res[i] = c[k] - t
        
        J[m, :] = np.ones(n)  # sum constraint
        res[m] = a.sum() - 1
        
        # Solve J @ delta = -res (minimum norm)
        # delta = -J^T (J J^T)^{-1} res
        JJT = J @ J.T
        try:
            delta = -J.T @ np.linalg.solve(JJT + 1e-12*np.eye(m+1), res)
        except:
            break
        
        # Line search
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_new = a + alpha * delta
            a_new = np.maximum(0, a_new)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            c_new = _fft_convolve(a_new, a_new)
            t_new = max(c_new[k] for k in active_set)
            if t_new < t:
                a = a_new
                t = t_new
                break
        else:
            break
    
    return a, t
```

At the optimum, suppose the active set is S = {k_1, ..., k_m} (the peaks where c[k] = t).

The KKT conditions (interior, a > 0):
1. c[k_i] = t for i = 1, ..., m
2. sum(a) = 1
3. There exist w_1, ..., w_m >= 0 (sum w_i = 1) and lambda such that:
   sum_i w_i * grad_{k_i}(a) = lambda * ones (n-vector)
   (This is the condition that the weighted gradient is constant, i.e., orthogonal to the tangent of the simplex.)

The unknowns are: a (n variables), t (1 variable), w (m variables), lambda (1 variable).
The equations are: m (peaks equal) + 1 (sum) + n (gradient condition) = m + n + 1.
But we also have sum(w) = 1, so total unknowns = n + 1 + m + 1 = n + m + 2, and total equations = m + n + 1 + 1 = m + n + 2. ✓

So the system is square (in the interior case).

For the Newton method, we can:
1. Given current (a, t, w, lambda), compute the residual.
2. Compute the Jacobian of the residual w.r.t. (a, t, w, lambda).
3. Take a Newton step.

The Jacobian is:
- ∂(c[k_i] - t)/∂a = grad_{k_i}
- ∂(c[k_i] - t)/∂t = -1
- ∂(c[k_i] - t)/∂w_j = 0
- ∂(c[k_i] - t)/∂lambda = 0

- ∂(sum(a) - 1)/∂a = ones
- ∂(sum(a) - 1)/∂t = 0
- etc.

- ∂(sum_i w_i * grad_{k_i} - lambda * ones)/∂a = sum_i w_i * Hessian_{k_i} (this is complicated!)
- ∂(...)/∂t = 0
- ∂(...)/∂w_j = grad_{k_j}
- ∂(...)/∂lambda = -ones

The Hessian of c[k] with respect to a is:
H_k[j, j'] = ∂²c[k]/∂a[j]∂a[j']

c[k] = sum_j a[j] * a[k-j]
∂c[k]/∂a[j] = 2*a[k-j] (for valid j)
∂²c[k]/∂a[j]∂a[j'] = 2 * δ_{j, k-j'} (for valid indices)

So H_k is a "shifted" identity matrix: H_k[j, j'] = 2 if j' = k-j, 0 otherwise.

This means the Jacobian of the gradient condition w.r.t. a is:
sum_i w_i * H_{k_i}

which is a sum of "shifted" identity matrices, weighted by w_i. This is a banded matrix (with at most m non-zero diagonals), so it can be applied to a vector in O(m*n).

```python
"""
Optimized step function search using active-set Newton method and
smart gradient descent for minimizing the normalized autocorrelation peak.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _get_subgradient(c, a, active_peaks):
    """Compute the subgradient of max(c[k] for k in active_peaks) at a.
    Returns the average subgradient over the active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in active_peaks:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(active_peaks)
    return g

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _smart_grad_descent(a, deadline, max_iter=200000, verbose=True):
    """Gradient descent with exact subgradient and adaptive step size."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_cmax = float(_fft_convolve(a, a).max())
    lr = 0.01
    
    t = 0
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks (within 1% of max)
        tol = c_max * 0.99
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 10:
            top_idx = np.argsort(c[active])[::-1][:10]
            active = active[top_idx]
        
        # Subgradient
        g = _get_subgradient(c, a, active)
        
        # Project to tangent space of simplex (subtract mean weighted by a)
        # For the simplicity constraint sum(a)=1: g_tan = g - g.mean()
        # For the non-negativity: use a "mirror" projection
        g_tan = g - g.mean()
        
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break
        
        # Step
        a_new = a - lr * g_tan / g_norm
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5

        a_new /= s

            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
    
    return best_a, _score(best_a)
```

adaptive active set growth and diverse initialization strategies.

def _newton_active_set(a, active_set, deadline, max_iter=5000):
    """Newton's method to equalize active set peaks and minimize the common value."""
    n = len(a)
    a = a.astype(np.float64).copy() / a.sum()
    active_set = sorted(set(active_set))
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        t = max(c[k] for k in active_set)
        
        # Check for newly violated peaks
        c_max = c.max()
        if c_max > t * 1.001:
            k_new = int(c.argmax())
            if k_new not in active_set:
                active_set.append(k_new)
                active_set.sort()
                t = c_max
                continue
        
        m = len(active_set)
        
        # Build the linearized system
        # Variables: da (n), dt (1)
        # Equations: Grad_k @ da - dt = -(c[k] - t) for k in active_set (m eqs)
        #            ones @ da = -(sum(a) - 1) (1 eq)
        # Total: (m+1) x (n+1) system, solve for minimum norm solution
        
        Jp = np.zeros((m + 1, n + 1))
        rp = np.zeros(m + 1)
        for i, k in enumerate(active_set):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                Jp[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
            Jp[i, n] = -1.0
            rp[i] = c[k] - t
        Jp[m, :n] = 1.0
        rp[m] = a.sum() - 1.0
        
        # Minimum norm solution
        JJT = Jp @ Jp.T  # (m+1) x (m+1)
        try:
            sol = np.linalg.solve(JJT + 1e-10 * np.eye(m + 1), rp)
            x = -Jp.T @ sol  # (n+1) vector
        except np.linalg.LinAlgError:
            x = -Jp.T @ np.linalg.lstsq(JJT, rp, rcond=None)[0]
        
        da = x[:n]
        dt = x[n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            a_new = a + alpha * da
            a_new = np.maximum(0, a_new)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            c_new = _fft_convolve(a_new, a_new)
            new_t = max(c_new[k] for k in active_set)
            new_max = c_new.max()
            if new_max <= t + 1e-12 and new_t < t:

                t = new_t
                improved = True
                break
            elif new_max <= t + 1e-8:
                # Accept even if t didn't decrease (progress on equalization)
                a = a_new
                improved = True
                break
        
        if not improved:
            break
    
    return a, active_set
