"""
Adam + log-sum-exp optimizer with aggressive beta annealing,
targeted peak shaving, and multi-n exploration up to 65536.
Key improvements: Adam optimizer for faster convergence,
no cvxpy dependency, larger n exploration, efficient FFT-based gradients.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    if a.sum() < 1e-15:
        return np.ones(n) / n
    # Already non-negative and sum to ~1?
    if a.min() >= -1e-12 and abs(a.sum() - 1.0) < 1e-10:
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            return a / s
    # General projection
    a = a.astype(np.float64)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(a - theta, 0)


def _smoobj_and_grad(z, beta, n):
    """Compute smoothed objective and gradient in log-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    # Gradient: df/dz[j] where a = softmax(z)
    # df/da[l] = 2n * sum_k w[k] * (da[l]/da[l]) ... 
    # Actually: c[k] = sum_i a[i]*a[k-i], dc[k]/da[l] = 2*a[k-l] (approx)
    # df/da[l] = 2n * sum_k w[k] * 2*a[k-l] ... no wait
    # f = 2n * (c_max + beta * log(sum exp((c-c_max)/beta)))
    # df/dc[k] = 2n * w[k]
    # df/da[l] = 2n * sum_k w[k] * dc[k]/da[l] = 2n * 2 * sum_k w[k] * a[k-l]
    #         = 4n * conv(w, a)[l] ... let me be careful
    # c[k] = sum_i a[i]*a[k-i], so dc[k]/da[l] = a[k-l] + a[k-l] = 2*a[k-l] (when valid)
    # df/da[l] = sum_k (df/dc[k]) * (dc[k]/da[l]) = sum_k 2n*w[k] * 2*a[k-l] = 4n * sum_k w[k]*a[k-l]
    # The sum over k of w[k]*a[k-l] is (conv(w, a))[l + n - 1] ... 
    # Actually conv(w,a)[m] = sum_k w[k]*a[m-k]. We want sum_k w[k]*a[k-l] = sum_k w[k]*a[-(l-k)]
    # Let j = k-l, then sum_j w[j+l]*a[j] = (correlate(w, a))[l] = conv(w, rev(a))[l + n - 1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]  # h[l] = sum_k w[k]*a[k-l]
    grad_a = 4.0 * n * h
    # Now df/dz[j] = sum_l (df/da[l]) * da[l]/dz[j]
    # da[l]/dz[j] = a[l]*(delta[l,j] - a[j])
    # df/dz[j] = a[j] * (grad_a[j] - sum_l grad_a[l]*a[l])
    W = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W)
    return obj, grad_z


def _adam_optimize(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """Adam optimizer in log-space with beta annealing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Beta schedule: start large, decrease aggressively
    c0 = _full_convolve(a, a)
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = float(c_sorted[0]) * 0.01
    beta_start = max(gap, float(c_sorted[0]) * 0.05)
    beta_end = float(c_sorted[0]) * 0.0002
    
    # Use geometric decay for beta
    n_steps_est = 5000
    beta_ratio = (beta_end / beta_start) ** (1.0 / n_steps_est)
    
    current_beta = beta_start
    patience = 0
    max_patience = 500
    
    if verbose:
        print(f"    Adam: n={n}, beta_start={beta_start:.2e}, best_sc={best_sc:.10f}")
    
    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        current_beta = max(beta_end, beta_start * (beta_ratio ** t))
        
        obj, grad = _smoobj_and_grad(z, current_beta, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = _softmax(z_new)
        sc_new = _score(a_new)
        
        # Line search: if it got worse, try smaller step
        if sc_new > best_sc * 1.001:
            # Backtrack: try half step
            for _ in range(3):
                z_new = z - 0.5 * lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = _softmax(z_new)
                sc_new = _score(a_new)
                if sc_new <= best_sc * 1.001:
                    break
        
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            patience = 0
            z = z_new
            a = a_new
        else:
            patience += 1
            z = z_new
            a = a_new
            if patience >= max_patience:
                # Reset to best and reduce learning rate
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                a = best_a.copy()
                lr *= 0.7
                m = np.zeros(n)
                v = np.zeros(n)
                patience = 0
                if lr < 0.001:
                    # Try fresh beta
                    current_beta = max(beta_end, current_beta * 0.5)
                    beta_start = current_beta
                    t = 0
                if verbose and patience == 0 and lr < 0.01:
                    print(f"    Adam iter {t}: lr={lr:.4f}, beta={current_beta:.2e}, best={best_sc:.10f}")
        
        # Periodic logging
        if verbose and t % 2000 == 0:
            el = time.monotonic()
            print(f"    Adam iter {t}: beta={current_beta:.2e}, lr={lr:.4f}, "
                  f"best={best_sc:.10f}, elapsed={el - (deadline - min_time_remaining - (deadline - time.monotonic() + min_time_remaining))::.0f}s")
    
    return best_a, best_sc


def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Iteratively shave the top peaks by adjusting contributing elements."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.02
    top_frac = 0.01
    stall = 0
    max_stall = 200
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                a = best_a.copy()
                lr *= 0.7
                top_frac = max(0.001, top_frac * 0.7)
                stall = 0
                if lr < 0.001:
                    break
        
        # Find active peaks
        n_active = max(3, int(n * top_frac))
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:n_active]
        
        # Compute weighted gradient (average over active peaks)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad -= np.mean(grad)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Line search
        alpha = lr
        improved = False
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < 2.0 * n * c_max - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try reducing specific peak contributors
            k_peak = int(np.argmax(c))
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contributions = a[i_lo:i_hi + 1].copy()
            contributions = contributions + a[k_peak - i_hi:k_peak - i_lo + 1][::-1]
            contributions[i_lo - i_lo] -= a[i_lo]  # avoid double counting center
            n_reduce = max(1, n // 200)
            top_idx = np.argsort(contributions)[::-1][:n_reduce]
            
            reduction = c_max * 0.005
            for idx in top_idx:
                a[idx] -= reduction / (contributions[idx] + 1e-30) * 0.01
            a = _project_simplex(a)
        
        if stall % 50 == 0 and stall > 0:
            top_frac = max(0.002, top_frac * 1.2)
    
    return best_a, best_sc


def _lbfgs_refine(a_init, deadline, min_time_remaining=30.0):

             mb * 0.0005, mb * 0.0002, mb * 0.0001]
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.005 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 10))))
        
        def obj(z_in):

            return o
        
        def grad(z_in):
            _, g = _smoobj_and_grad(z_in, beta, n)
            return g
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,

                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: improved to {best_sc:.10f}")
        except Exception:

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    1. Load parent, normalize
    2. Adam optimization with beta annealing (main workhorse)
    3. L-BFGS refinement with tight beta
    4. Peak shaving
    5. Multi-n exploration (resample to larger n, re-optimize)
    6. Final combined polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Adam optimization ===
    print(f"\n[P1] Adam optimization (t={time.monotonic()-t_start:.1f}s)")
    t1_deadline = min(deadline - 600, t_start + 500)
    a1, sc1 = _adam_optimize(parent, t1_deadline, min_time_remaining=150.0, verbose=True)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [adam] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Adam: no improvement (best={best_score:.10f})")
    
    # === Phase 2: L-BFGS refinement ===
    print(f"\n[P2] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    t2_deadline = min(deadline - 400, t_start + 800)
    a2, sc2 = _lbfgs_refine(best_seq, t2_deadline, min_time_remaining=100.0)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Peak shaving ===
    print(f"\n[P3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    t3_deadline = min(deadline - 300, t_start + 1000)
    a3, sc3 = _peak_shave(best_seq, t3_deadline, min_time_remaining=80.0)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [shave] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [32768, 65536]
    if n_parent not in n_candidates and n_parent > 4096:
        n_candidates.insert(0, n_parent)
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        # For 65536, check if we have enough time
        if n_try == 65536 and (deadline - time.monotonic()) < 300:
            continue
        
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        print(f"  n={n_try}: resampled score={sc_res:.10f}")
        
        # Adam on resampled
        rem = deadline - 150
        t_n_deadline = min(rem, time.monotonic() + 150)
        a_n, sc_n = _adam_optimize(a_res, t_n_deadline, min_time_remaining=50.0, verbose=False)
        if sc_n < best_score:
            best_seq = a_n.copy()
            best_score = sc_n
            print(f"  NEW BEST: {best_score:.10f} [adam_n{n_try}] t={time.monotonic()-t_start:.1f}s")
        
        # L-BFGS on resampled
        if time.monotonic() < deadline - 60:
            t_n2 = min(deadline - 50, time.monotonic() + 60)
            a_n2, sc_n2 = _lbfgs_refine(a_n, t_n2, min_time_remaining=20.0)
            if sc_n2 < best_score:
                best_seq = a_n2.copy()
                best_score = sc_n2
                print(f"  NEW BEST: {best_score:.10f} [lbfgs_n{n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Try different initializations ===
    print(f"\n[P5] Alternative initializations (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        for kind in ["tapered", "cosine"]:
            if time.monotonic() > deadline - 60:
                break
            n_try = len(best_seq)
            a_alt = _make_profile(n_try, kind)
            # Blend with current best
            alpha_blend = 0.5
            a_mixed = (1 - alpha_blend) * best_seq + alpha_blend * a_alt
            a_mixed = _project_simplex(a_mixed)
            
            t_alt_deadline = min(deadline - 50, time.monotonic() + 60)
            a_alt_opt, sc_alt = _adam_optimize(a_mixed, t_alt_deadline,
                                               min_time_remaining=20.0, verbose=False)
            if sc_alt < best_score:
                best_seq = a_alt_opt.copy()
                best_score = sc_alt
                print(f"  NEW BEST: {best_score:.10f} [alt_{kind}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Final combined polish ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        # Peak shave
        t_fin1 = min(deadline - 5, time.monotonic() + 30)
        a_f1, sc_f1 = _peak_shave(best_seq, t_fin1, min_time_remaining=5.0)
        if sc_f1 < best_score:
            best_seq = a_f1.copy()
            best_score = sc_f1
            print(f"  NEW BEST: {best_score:.10f} [fin_shave] t={time.monotonic()-t_start:.1f}s")
        
        # L-BFGS
        if time.monotonic() < deadline - 3:
            t_fin2 = deadline - 2
            a_f2, sc_f2 = _lbfgs_refine(best_seq, t_fin2, min_time_remaining=2.0)
            if sc_f2 < best_score:
                best_seq = a_f2.copy()
                best_score = sc_f2
                print(f"  NEW BEST: {best_score:.10f} [fin_lbfgs] t={time.monotonic()-t_start:.1f}s")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Double check with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            if ev_score != sc_final:
                print(f"  WARNING: _score={sc_final:.10f} but evaluate={ev_score:.10f}")
                if ev_score < best_score:
                    best_score = ev_score
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
