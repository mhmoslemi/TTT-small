Optimized step-function search using Nesterov-accelerated gradient with

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind="uniform"):

        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2

        return a / a.sum()
    elif kind == "tapered":

    old_n = len(a)
    if old_n == new_n:

    x_new = np.linspace(0, 1, new_n)

    a_new = np.maximum(a_new, 0.0)
