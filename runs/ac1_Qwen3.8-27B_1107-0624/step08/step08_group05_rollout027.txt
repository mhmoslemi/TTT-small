```python
"""
Improved optimizer for minimizing step function autocorrelation peak.
Uses novel multiplicative subgradient method, aggressive LSE smoothing,
adaptive Newton active-set, analytical shape exploration, and multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE smoothing objectives and gradients
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Power-sum objectives
# ============================================================

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Active set gradient and Hess
# ============================================================

def _active_gradient(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

# Newton active-set solver
# ============================================================

def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.01):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=16):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.001)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.7, 0.45, 0.3, 0.18, 0.1,
                                   0.06, 0.035, 0.02, 0.01,
                                   0.005, 0.0025, 0.001, 0.0005,
                                   0.0002, 0.0001])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


# ============================================================
# Subgradient polish
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_mix(a, rng):
    a1 = _perturb_log(a, 0.02, rng)
    a2 = _perturb_fourier_phase(a1, rng, n_modes=5, max_amp=0.2)
    return a2


def _perturb_asymmetric(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    shift = int(rng.integers(0, n // 4))
    z = np.roll(z, shift)
    return _softmax(z)

# Profile generators with parameter sweep

def _make_profile(n, kind="uniform", param=1.0):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** param
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        a = 1.0 + param * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        a = np.exp(-((t - 0.5) ** 2) / (2 * param ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        a = np.exp(-((t - 0.25) ** 2) / (param * 0.02)) + np.exp(-((t - 0.75) ** 2) / (param * 0.02))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "lorentz":
        a = 1.0 / (1.0 + param * ((t - 0.5) / (n * 0.1)) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "poly":
        x = 2.0 * (t - 0.5)
        a = np.maximum(1.0 - x ** 2, 0.0) ** param
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = 1.0 + param * t
        return a / a.sum()
    elif kind == "asym_gauss":
        a = np.exp(-((t - param) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _sweep_profiles(n, deadline):
    best_a = None
    best_sc = np.inf
    kinds_params = [
        ("uniform", 1.0), ("tapered", 0.5), ("tapered", 1.0),
        ("tapered", 1.5), ("tapered", 2.0), ("tapered", 3.0),
        ("cos2", 0.2), ("cos2", 0.4), ("cos2", 0.6),
        ("gauss", 0.1), ("gauss", 0.15), ("gauss", 0.2),
        ("gauss", 0.25), ("gauss", 0.3),
        ("lorentz", 0.5), ("lorentz", 1.0), ("lorentz", 2.0),
        ("poly", 0.5), ("poly", 1.0), ("poly", 2.0), ("poly", 3.0),
        ("ramp", 0.5), ("ramp", 1.0), ("ramp", 2.0),
        ("asym_gauss", 0.3), ("asym_gauss", 0.4), ("asym_gauss", 0.7),
        ("bimodal", 0.5), ("bimodal", 1.0),
    ]
    for kind, param in kinds_params:
        if time.monotonic() > deadline:
            break
        a = _make_profile(n, kind, param)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining multiplicative subgradient, aggressive
    LSE smoothing, adaptive Newton active-set, analytical shape sweep,
    multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    _best_seq = [0.0]
    _best_score = [np.inf]

    def upd(p, sc=None, label=""):
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score[0]:
            _best_seq[0] = p.copy()
            _best_score[0] = float(sc)
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    target_n = 16384

    # === Phase 1: Profile sweep at small n (40s) ===
    print(f"\n[Phase 1] Profile sweep")
    p1_deadline = min(deadline, t_start + 40)
    n_small = 512
    best_small, sc_small = _sweep_profiles(n_small, p1_deadline)
    if best_small is not None:
        print(f"  Best small profile: score={sc_small:.8f}")
        # Upscale to target_n
        best_small_up = _resize_sequence(best_small, target_n)
        upd(best_small_up, _score(best_small_up), "profile_sweep_up")

    # Load parent
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
        if n_parent != target_n:
            parent_resized = _resize_sequence(parent, target_n)
            upd(parent_resized, _score(parent_resized), "parent_resized")

    if _best_score[0] == np.inf:
        a_init = _make_profile(target_n, "tapered", 1.0)
        upd(a_init, _score(a_init), "init_tapered")

    # === Phase 2: Multiplicative subgradient (150s) ===
    print(f"\n[Phase 2] Multiplicative subgradient")
    if time.monotonic() < deadline - 100:
        p2_deadline = min(deadline - 100, time.monotonic() + 150)
        a_ms, sc_ms = _mult_subgrad(_best_seq[0].copy(), p2_deadline,
                                     max_iter=10000, tol_frac=0.01, eta_init=0.03)
        upd(a_ms, sc_ms, "mult_subgrad")

        # Second pass with different params
        if time.monotonic() < p2_deadline - 30:
            a_ms2, sc_ms2 = _mult_subgrad(a_ms, p2_deadline,
                                           max_iter=5000, tol_frac=0.005, eta_init=0.01)
            upd(a_ms2, sc_ms2, "mult_subgrad_2")

    # === Phase 3: L-BFGS with LSE - aggressive (350s) ===
    print(f"\n[Phase 3] L-BFGS LSE aggressive")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, time.monotonic() + 350)
        a_lse, sc_lse = _lbfgs_lse(_best_seq[0].copy(), p3_deadline,
                                     min_remaining=50, n_stages=16)
        upd(a_lse, sc_lse, "lbfgs_lse")

    # === Phase 4: Power-sum L-BFGS (150s) ===
    print(f"\n[Phase 4] Power-sum L-BFGS")
    if time.monotonic() < deadline - 120:
        p4_deadline = min(deadline - 120, time.monotonic() + 150)
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq[0].copy(), p4_deadline,
                                         min_remaining=30)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 5: Newton active-set (200s) ===
    print(f"\n[Phase 5] Newton active-set")
    if time.monotonic() < deadline - 180:
        p5_deadline = min(deadline - 180, time.monotonic() + 200)
        # Try multiple tol_frac values
        for tf in [0.02, 0.01, 0.005, 0.002]:
            if time.monotonic() > p5_deadline:
                break
            nw_dead = min(p5_deadline, time.monotonic() + 50)
            a_nw, sc_nw = _newton_active_set(_best_seq[0].copy(), nw_dead,
                                              max_iter=150, tol_frac=tf)
            upd(a_nw, sc_nw, f"newton_tf{tf}")

    # === Phase 6: Subgradient polish (100s) ===
    print(f"\n[Phase 6] Subgradient polish")
    if time.monotonic() < deadline - 80:
        p6_deadline = min(deadline - 80, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(_best_seq[0].copy(), p6_deadline,
                                       n_steps=5000, lr_init=0.006, tol_frac=0.005)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 7: Multi-n exploration (250s) ===
    print(f"\n[Phase 7] Multi-n exploration")
    n_candidates = [4096, 8192, 12288, 16384, 20480, 24576]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq[0]) == n_try:
            a_init_n = _best_seq[0].copy()
        else:
            a_init_n = _resize_sequence(_best_seq[0], n_try)

        # Quick LSE
        if time.monotonic() < deadline - 40:
            a_opt_n, sc_opt_n = _lbfgs_lse(a_init_n, deadline,
                                            min_remaining=30, n_stages=10)
            upd(a_opt_n, sc_opt_n, f"n{n_try}_lse")

            # Newton polish
            if time.monotonic() < deadline - 20:
                nw_d = min(deadline - 15, time.monotonic() + 40)
                a_nw_n, sc_nw_n = _newton_active_set(a_opt_n, nw_d,
                                                      max_iter=80, tol_frac=0.01)
                upd(a_nw_n, sc_nw_n, f"n{n_try}_newton")

    # === Phase 8: Diversified restarts (200s) ===
    print(f"\n[Phase 8] Diversified restarts")
    if time.monotonic() < deadline - 50:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq[0], 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq[0], rng, n_modes=10, max_amp=0.3)
            elif i == 2:
                p_start = _perturb_block_swap(_best_seq[0], rng, n_swaps=6)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq[0], rng, sigma=0.1)
            elif i == 4:
                p_start = _perturb_log(_best_seq[0], 0.05, rng)
            elif i == 5:
                p_start = _perturb_asymmetric(_best_seq[0], rng)
            elif i == 6:
                p_start = _perturb_mix(_best_seq[0], rng)
            else:
                p_start = _perturb_log(_best_seq[0], 0.01, rng)

            # LSE optimization from perturbed start
            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=20, n_stages=10)
            upd(a_r, sc_r, f"restart_{i}")

            # Newton polish if promising
            if sc_r < _best_score[0] * 1.005 and time.monotonic() < deadline - 10:
                nw_dead = min(deadline - 5, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.01)
                upd(a_r2, sc_r2, f"restart_{i}_nw")

    # === Phase 9: Final aggressive polish (80s) ===
    print(f"\n[Phase 9] Final polish")
    if _best_seq[0] is not None and time.monotonic() < deadline - 10:
        n_fin = len(_best_seq[0])

        # Very low beta LSE
        c_fin = _full_convolve(_best_seq[0], _best_seq[0])
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq[0], 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.00003
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 80}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton with very wide active set
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq[0], deadline - 3,
                                                  max_iter=100, tol_frac=0.03)
            upd(a_fin2, sc_fin2, "final_newton_wide")

        # Final Newton with narrow active set
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _newton_active_set(_best_seq[0], deadline - 2,
                                                  max_iter=80, tol_frac=0.002)
            upd(a_fin3, sc_fin3, "final_newton_narrow")

        # Final multiplicative subgradient
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _mult_subgrad(_best_seq[0], deadline - 1,
                                             max_iter=3000, tol_frac=0.002, eta_init=0.01)
            upd(a_fin4, sc_fin4, "final_mult_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq[0] is not None and isinstance(_best_seq[0], np.ndarray):
        final = np.clip(_best_seq[0], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            ev_sc = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal_score={sc_final:.10f}, eval_score={ev_sc:.10f}, "
                  f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        except Exception:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
                  f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(4096, "tapered", 1.0)
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
