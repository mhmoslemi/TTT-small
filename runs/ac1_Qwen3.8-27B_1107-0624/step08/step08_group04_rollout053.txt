        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        if not np.isfinite(r_new).all():
            break
        rs_new = float(r_new @ r_new)
        if rs_new > rs * 1.2:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    """L-BFGS with adaptive beta continuation on log-softmax parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                           0.00005, 0.000025, 0.00001, 0.000005, 0.0000025,
                           0.000001, 0.0000005, 0.00000025, 0.0000001,
                           0.00000005, 0.00000002])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002])

    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 15))))

        try:
            res = minimize(
                _make_smoobj_fn(beta, n), z,
                jac=_make_smograd_fn(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_smoobj_fn(beta, n):
    def fn(z):
        return _smoobj(z, beta, n)
    return fn


def _make_smograd_fn(beta, n):
    def fn(z):
        return _smograd(z, beta, n)
    return fn


def _active_set_qp(W, n, a_init, max_outer=80, cg_max_iter=5000):
    """Solve min a^T (2M) a s.t. sum(a)=1, a>=0 using active-set + PGCG."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n))

    for outer in range(max_outer):
        # Solve equality-constrained QP on full space: M*a = lam*1, 1^T*a=1
        y = _pgcg_solve(W, n, np.ones(n), max_iter=cg_max_iter, tol=1e-15)
        if not np.isfinite(y).any():
            y = _cg_solve_basic(W, n, np.ones(n), max_iter=cg_max_iter, tol=1e-15)
        oney = float(np.sum(y))
        if abs(oney) < 1e-20 or not np.isfinite(oney):
            # Fallback: projected gradient
            grad = _hankel_matvec(a, W, n)
            lam = grad.mean()
            step = 0.1 / (np.max(np.abs(grad)) + 1e-15)
            a_new = a - step * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            a_new = y / oney

        # Project onto a >= 0
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        val_new = float(a_new @ _hankel_matvec(a_new, W, n))
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

        # Check KKT: gradient - lam*1 should be >= 0 where a=0
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-12
        if support.any():
            lam = grad[support].mean()
            resid = np.max(np.abs(grad[support] - lam))
            off_s = ~support
            if off_s.any():
                resid = max(resid, float(np.max(lam - grad[off_s])))
            total_scale = np.max(np.abs(grad)) + 1e-15
            if resid < 1e-12 * total_scale:
                break

    return best_a


def _equiripple_polish(a_init, deadline, max_iters=200, top_frac=0.005):
    """Equiripple enforcement: drive top convolution values to equal the max."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Top M values
        M = max(2, int(len(c) * top_frac))
        top_idx = np.argsort(c)[::-1][:M]
        top_vals = c[top_idx]
        T = float(top_vals.mean())

        # Excess above T for top values, plus any values above T
        excess = np.maximum(c - T, 0)
        excess_max = float(excess.max())
        if excess_max < 1e-14 * c_max:
            break

        # Gradient of sum(excess^2) w.r.t. a
        # d(sum_k excess_k^2)/da_i = 2 * sum_k excess_k * d(c_k)/da_i
        # = 2 * sum_k excess_k * 2*a[k-i] = 4 * (excess * a_reversed)_i
        a_rev = a[::-1]
        grad_conv = _full_convolve(excess, a_rev)
        grad = 4.0 * grad_conv[n - 1:2 * n - 1]

        # Also add a small regularization to keep sum=1 and a>=0
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Damped step
        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad / (n * c_max + 1e-30)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_half = a - 0.5 * step * grad / (n * c_max + 1e-30)
            a_half = np.maximum(a_half, 0)
            s_half = a_half.sum()
            if s_half > 1e-15:
                a_half = a_half / s_half
                sc_half = _score(a_half)
                if sc_half < best_sc - 1e-14:
                    best_sc = sc_half
                    best_a = a_half.copy()
                    a = a_half
                else:
                    break
            else:
                break

    return best_a, best_sc


def _newton_active_polish(a_init, deadline, max_rounds=100, tol_schedule=None):
    """Progressive active-set QP with tightening tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001,
                        0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                        0.000001, 0.0000005, 0.0000002, 0.0000001]

    tol_idx = 0
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)] * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        a_qp = _active_set_qp(W, n, a, max_outer=40,
                               cg_max_iter=max(1000, 200 * n_active))
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped Newton fallback
            grad = _hankel_matvec(a, W, n)
            g_t = grad - np.mean(grad)
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-15:
                pass
            else:
                Hg = _pgcg_solve(W, n, grad, max_iter=800, tol=1e-13)
                H1 = _pgcg_solve(W, n, np.ones(n), max_iter=800, tol=1e-13)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)

                alpha = 1.0
                for ls in range(30):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max:
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        break
                    alpha *= 0.5
                    if alpha < 1e-14:
                        break

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, max_steps=2000):
    """Nesterov-accelerated subgradient on soft-max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02
    momentum = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = z_prev + momentum * (z_new - z) + (1 - momentum) * (-step * grad)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.35) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS beta continuation, progressive active-set QP
    with preconditioned CG, equiripple enforcement, n=32768 exploration,
    and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=500.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Active-set QP with progressive tightening
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _newton_active_polish(best_seq,
                                         min(deadline - 400, time.monotonic() + 250),
                                         max_rounds=80)
        update_best(a2, sc2, "active_qp_1")

    # Phase 3: Equiripple polish
    print(f"\n[Phase 3] Equiripple polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3, sc3 = _equiripple_polish(best_seq,
                                      min(deadline - 350, time.monotonic() + 80),
                                      max_iters=150, top_frac=0.003)
        update_best(a3, sc3, "equiripple_1")

    # Phase 4: Another active-set QP round (tighter)
    print(f"\n[Phase 4] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _newton_active_polish(best_seq,
                                         min(deadline - 300, time.monotonic() + 150),
                                         max_rounds=60,
                                         tol_schedule=[0.0001, 0.00005, 0.00002,
                                                       0.00001, 0.000005,
                                                       0.000002, 0.000001,
                                                       0.0000005, 0.0000002,
                                                       0.0000001])
        update_best(a4, sc4, "active_qp_2")

    # Phase 5: Subgradient with momentum
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _subgradient_polish(best_seq,
                                       min(deadline - 250, time.monotonic() + 80))
        update_best(a5, sc5, "subgrad_1")

    # Phase 6: Equiripple round 2 (tighter)
    print(f"\n[Phase 6] Equiripple round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a6, sc6 = _equiripple_polish(best_seq,
                                      min(deadline - 200, time.monotonic() + 60),
                                      max_iters=100, top_frac=0.001)
        update_best(a6, sc6, "equiripple_2")

    # Phase 7: n=32768 exploration
    print(f"\n[Phase 7] n=32768 exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_big = 32768
        a_big = _resample(best_seq, n_big)
        a_big_opt = _run_lbfgs(a_big, deadline, min_time_remaining=80.0, aggressive=True)
        sc_big = _score(a_big_opt)
        update_best(a_big_opt, sc_big, "n32768_lbfgs")

        if sc_big < best_score + 0.003 and time.monotonic() < deadline - 60:
            a_big_qp, sc_big_qp = _newton_active_polish(
                a_big_opt, min(deadline - 50, time.monotonic() + 80), max_rounds=40)
            update_best(a_big_qp, sc_big_qp, "n32768_qp")

            a_big_eq, sc_big_eq = _equiripple_polish(
                a_big_qp, min(deadline - 40, time.monotonic() + 40),
                max_iters=80, top_frac=0.002)
            update_best(a_big_eq, sc_big_eq, "n32768_eq")

    # Phase 8: Diverse restarts on best n
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restarts = [
            ("perturb", lambda: _perturb(best_seq, 0.03, rng)),
            ("asym", lambda: _make_profile(len(best_seq), "asymmetric")),
            ("centered", lambda: _make_profile(len(best_seq), "centered")),
        ]
        for name, gen_fn in restarts[:2]:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{name}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 15:
                a_rq, sc_rq = _newton_active_polish(
                    a_r, min(deadline - 10, time.monotonic() + 20), max_rounds=30)
                update_best(a_rq, sc_rq, f"restart_{name}_qp")

    # Phase 9: Final ultra-tight polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(
            best_seq, deadline - 2, max_rounds=60,
            tol_schedule=[0.00001, 0.000005, 0.000002, 0.000001,
                          0.0000005, 0.0000002, 0.0000001, 0.00000005,
                          0.00000002, 0.00000001])
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 3:
            a_feq, sc_feq = _equiripple_polish(
                best_seq, deadline - 2, max_iters=80, top_frac=0.0005)
            update_best(a_feq, sc_feq, "final_eq")

        # Ultra-fine L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 3e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(1500, max(30, int(remaining / 0.01)))
                try:
                    result = minimize(
                        _make_smoobj_fn(beta_fin, n), z,
                        jac=_make_smograd_fn(beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-15, 'maxcor': 80}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
