    return float(2.0 * n * c.max())


def _compute_gradient(a, active, n):
    """Compute gradient of (1/m)*sum_{k in active} c[k] wrt a[i].
    Returns the centered gradient (for use in mirror descent)."""
    m = len(active)
    A_vec = np.zeros(2*n-1)
    A_vec[active] = 1.0 / m
    
    # Cross-correlation: g[i] = sum_k A_vec[k] * a[k-i] (with bounds checking via the corr function)
    flip_a = a[::-1]
    out_len = n + (2*n-1) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    g = result[n-1:2*n-1]
    
    # The gradient is approximately 2*g (accounting for the two terms in the derivative)
    # But the "centering" is done externally
    return 2.0 * g


def _mirror_descent(a_init, deadline, max_iter=50000):
    """Mirror descent with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = 0.1
    tol_fraction = 0.05  # Active set tolerance (fraction of max)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set
        threshold = c_max * (1.0 - tol_fraction)
        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c)])
        
        # Compute gradient
        g = _compute_gradient(a, active_idx, n)
        g_centered = g - np.mean(g)
        
        # Adaptive step size via backtracking
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-15:
            break
        
        # Try the step
        accepted = False
        for attempt in range(30):
            z_new = z - eta * g_centered
            a_new = np.exp(z_new - z_new.max())
            s_new = a_new.sum()
            if s_new < 1e-300:
                eta *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new <= best_sc + 1e-15:
                z = z_new
                a = a_new
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_a = a_new.copy()
                accepted = True
                break
            eta *= 0.5
        
        if not accepted:
            # Try smaller base step
            eta = max(eta, 1e-8)
            z_new = z - eta * g_centered
            a_new = np.exp(z_new - z_new.max())
            s_new = a_new.sum()
            a_new = a_new / s_new
            z = z_new
            a = a_new
            sc_new = _score(a)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
        
        # Gradually tighten the active set tolerance
        if it % 100 == 0:
            tol_fraction = max(0.001, tol_fraction * 0.95)
        
        # Periodic: check if we're making progress
        if it % 50 == 0:
            if gnorm / c_max < 1e-10:
                break
    
    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=300, tol_list=None):
    """Active-set Newton method for the minimax problem."""
    if tol_list is None:
        tol_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        best_dir = None
        best_dir_score = np.inf

        for tol_a in tol_list:
            active = c >= (c_max - tol_a * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2 or m > 6 * n:
                if m > 6 * n:
                    top_idx = np.argsort(c)[::-1][:6 * n]
                    active = np.zeros_like(c, dtype=bool)
                    active[top_idx] = True
                    m = int(active.sum())
                continue

            A_vec = active.astype(np.float64) / m
            g = _compute_gradient(a, np.where(active)[0], n)
            g = g / max(m, 1)  # Normalize
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            # Newton step: solve (H + mu*I) dx = -g_t where H is the Hessian
            mu = 1e-6 * (c_max / max(m, 1)) + 1e-9

            # Hessian-vector product: H*v = 2 * corr_of(v, A) (approximately)
            def hess_vp(v):
                flip_v = v[::-1]
                out_len = n + (2*n-1) - 1
                N = _next_pow2(out_len)
                fv = np.fft.rfft(flip_v, N)
                fA = np.fft.rfft(A_vec, N)
                result = np.fft.irfft(fv * fA, N)[:out_len]
                h = result[n-1:2*n-1]
                return 2.0 * h + mu * v

            # CG solve
            x = np.zeros(n)
            r = -g_t - hess_vp(x)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(200):
                Ap = hess_vp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r = r - alpha * Ap
                rs_new = np.dot(r, r)
                if np.sqrt(rs_new) < 1e-12:
                    break
                p = r + (rs_new / rs) * p
                rs = rs_new

            # Adjust for the sum constraint
            w = np.zeros(n)
            r_w = np.ones(n) - hess_vp(w)
            p_w = r_w.copy()
            rs_w = np.dot(r_w, r_w)
            for _ in range(200):
                Ap_w = hess_vp(p_w)
                pAp_w = np.dot(p_w, Ap_w)
                if pAp_w < 1e-30:
                    break
                alpha_w = rs_w / pAp_w
                w = w + alpha_w * p_w
                r_w = r_w - alpha_w * Ap_w
                rs_w_new = np.dot(r_w, r_w)
                if np.sqrt(rs_w_new) < 1e-12:
                    break
                p_w = r_w + (rs_w_new / rs_w) * p_w
                rs_w = rs_w_new

            w_sum = float(w.sum())
            if abs(w_sum) < 1e-15:
                dx = x
            else:
                lam = float(x.sum()) / w_sum
                dx = x - lam * w

            dx = dx - np.mean(dx)

            # Line search
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)
            t = min(1.0, t_max)
            
            for ls in range(40):
                a_try = a + t * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_dir_score:
                    best_dir_score = sc_try
                    best_dir = dx
                    best_t = t
                if sc_try <= best_sc - 1e-13:
                    best_dir = dx
                    best_t = t
                    best_dir_score = sc_try
                    break
                t *= 0.5

        if best_dir is None:
            break

        # Apply the best direction
        a_new = a + best_t * best_dir
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _lsobj(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))


def _lsgrad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    flip_a = a[::-1]
    wc = _full_convolve(w, flip_a)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 5))))

        try:
            result = minimize(
                _lsobj,
                z,
                jac=_lsgrad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 100,
                }
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _deflation_step(a, deadline, n_attempts=30):
    """Targeted deflation of top convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        top_k = np.argsort(c)[::-1]
        n_top = max(5, min(len(c), int(0.03 * len(c))))
        top_positions = top_k[:n_top]

        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 30)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.99
                else:
                    a_new[j] *= 0.99

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return np.exp(z - z.max()) / np.exp(z - z.max()).sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining mirror descent, active-set Newton,
    L-BFGS with log-sum-exp smoothing, deflation, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Mirror descent (primary method - more robust than Newton)
    print(f"\n[Phase 1] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    md_dl = min(deadline - 600, time.monotonic() + 300)
    a_md, sc_md = _mirror_descent(parent, md_dl, max_iter=20000)
    update_best(a_md, sc_md, "mirror_descent_1")

    # Phase 2: Active-set Newton from mirror descent result
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_dl = min(deadline - 400, time.monotonic() + 200)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=300,
                                         tol_list=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        update_best(a_as, sc_as, "as_newton_1")

        # Second Newton pass with tighter tolerances
        if time.monotonic() < deadline - 300:
            as_dl2 = min(deadline - 300, time.monotonic() + 120)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=200,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            update_best(a_as2, sc_as, "as_newton_2")
    
    # ... continuing with more phases

    # Phase 3: L-BFGS with log-sum-exp smoothing
    print(f"\n[Phase 3] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        lb_dl = min(deadline - 200, time.monotonic() + 250)
        a_lb, sc_lb = _run_lbfgs(best_seq, lb_dl, min_time_remaining=100.0)
        update_best(a_lb, sc_lb, "lbfgs")

    # Phase 4: Deflation
    print(f"\n[Phase 4] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        def_dl = min(deadline - 150, time.monotonic() + 60)
        a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=30)
        update_best(a_def, sc_def, "deflation")

    # Phase 5: Mirror descent again with tighter tolerance
    print(f"\n[Phase 5] Mirror descent tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        md_dl2 = min(deadline - 100, time.monotonic() + 80)
        a_md2, sc_md2 = _mirror_descent(best_seq, md_dl2, max_iter=15000)
        update_best(a_md2, sc_md2, "mirror_descent_2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent:
            continue
        if n_try > 32768:
            continue

        a_init = _interpolate_to_n(parent, n_try) if parent is not None else _make_profile(n_try, "uniform")
        
        # Quick L-BFGS
        if time.monotonic() < deadline - 40:
            a_opt, sc_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)
            update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick mirror descent
        if time.monotonic() < deadline - 30:
            a_md_n, sc_md_n = _mirror_descent(a_init, min(deadline - 20, time.monotonic() + 30), max_iter=5000)
            update_best(a_md_n, sc_md_n, f"n_{n_try}_md")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.01, 0.03, 0.05]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _mirror_descent(p_start, min(deadline - 10, time.monotonic() + 20), max_iter=5000)
            update_best(a_r, sc_r, f"restart_s{sigma}")

    # Phase 8: Final polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight Newton
        as_final = min(deadline - 2, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=200,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_newton")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0001
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj,
                        z,
                        jac=_lsgrad,
                        args=(beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 80}
                    )
                    a_fin3 = np.exp(result.x - result.x.max())
                    a_fin3 = a_fin3 / a_fin3.sum()
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _corr_a_A(a, A_vec, n):

    flip_a = a[::-1]

    result = np.fft.irfft(fa * fA, N)[:out_len]

def _lsobj(z, beta, n):
    a = np.exp(z - z.max())
    s = a.sum()
    if s < 1e-300:
        return 2.0 * n
    a = a / s
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    flip_a = a[::-1]
    wc = _full_convolve(w, flip_a)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation on log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.max()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5.0:
            break

        est_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(10000, max(50, int(remaining / (est_time * 4))))

        try:
            result = minimize(
                _lsobj,
                z,
                jac=_lsgrad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 120,
                }
            )
            a_new = np.exp(result.x - result.x.max())
            s_new = a_new.sum()
            if s_new < 1e-300:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.max()
        except Exception:
            pass

    return current_a, current_sc


def _deflation_step(a, deadline, n_attempts=40):
    """Targeted deflation: reduce the largest contributing pairs at peak lags."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        top_k = np.argsort(c)[::-1]
        n_top = max(5, min(len(c), int(0.04 * len(c))))
        top_positions = top_k[:n_top]

        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 40)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a_new[i] * a_new[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                j = k_pos - max_i
                if a_new[max_i] >= a_new[j]:
                    a_new[max_i] *= 0.99
                else:
                    a_new[j] *= 0.99

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try gentler reduction
            a_gentle = a.copy()
            for k_pos in top_positions[:10]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0.0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_gentle[i] * a_gentle[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    if a_gentle[max_i] >= a_gentle[j]:
                        a_gentle[max_i] *= 0.998
                    else:
                        a_gentle[j] *= 0.998
            a_gentle = np.maximum(a_gentle, 0)
            s_g = a_gentle.sum()
            if s_g > 1e-15:
                a_gentle = a_gentle / s_g
                sc_g = _score(a_gentle)
                if sc_g < best_sc - 1e-13:
                    best_sc = sc_g
                    best_a = a_gentle.copy()
                else:
                    break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    return np.exp(z) / np.exp(z).sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "bottleneck":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: mirror descent (primary), active-set Newton,
    L-BFGS log-sum-exp, deflation, multi-n exploration, diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Mirror descent (primary - robust and effective)
    print(f"\n[Phase 1] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    md_dl = min(deadline - 700, time.monotonic() + 350)
    a_md, sc_md = _mirror_descent(parent, md_dl, max_iter=25000)
    update_best(a_md, sc_md, "mirror_descent_1")

    # Phase 2: Active-set Newton from MD result
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 250)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=350,
                                         tol_list=[0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003])
        update_best(a_as, sc_as, "as_newton_1")

        # Second Newton with tighter tolerances
        if time.monotonic() < deadline - 350:
            as_dl2 = min(deadline - 350, time.monotonic() + 150)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=250,
                                               tol_list=[0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005])
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: L-BFGS log-sum-exp
    print(f"\n[Phase 3] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        lb_dl = min(deadline - 250, time.monotonic() + 200)
        a_lb, sc_lb = _run_lbfgs(best_seq, lb_dl, min_time_remaining=100.0)
        update_best(a_lb, sc_lb, "lbfgs")

    # Phase 4: Deflation
    print(f"\n[Phase 4] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        def_dl = min(deadline - 180, time.monotonic() + 60)
        a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation")

        if sc_def < best_score and time.monotonic() < deadline - 120:
            # Newton after deflation
            a_def_n, sc_def_n = _active_set_newton(best_seq, min(deadline - 100, time.monotonic() + 50),
                                                    max_iter=150,
                                                    tol_list=[0.002, 0.001, 0.0005, 0.0003, 0.0001])
            update_best(a_def_n, sc_def_n, "deflation_newton")

    # Phase 5: Mirror descent again (tighter)
    print(f"\n[Phase 5] Mirror descent round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        md_dl2 = min(deadline - 120, time.monotonic() + 80)
        a_md2, sc_md2 = _mirror_descent(best_seq, md_dl2, max_iter=20000)
        update_best(a_md2, sc_md2, "mirror_descent_2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        if n_try > 32768:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        # Quick L-BFGS
        if time.monotonic() < deadline - 50:
            a_opt, sc_opt = _run_lbfgs(a_init, min(deadline - 40, time.monotonic() + 60), min_time_remaining=30.0)
            update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        # Quick mirror descent
        if time.monotonic() < deadline - 35:
            a_md_n, sc_md_n = _mirror_descent(a_init, min(deadline - 25, time.monotonic() + 30), max_iter=8000)
            update_best(a_md_n, sc_md_n, f"n_{n_try}_md")

        # Quick Newton if promising
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 25:
            a_n_n, sc_n_n = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 20),
                                                max_iter=80,
                                                tol_list=[0.003, 0.002, 0.001, 0.0005, 0.0002])
            update_best(a_n_n, sc_n_n, f"n_{n_try}_newton")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for sigma in [0.01, 0.03, 0.06]:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _mirror_descent(p_start, min(deadline - 15, time.monotonic() + 20), max_iter=8000)
            update_best(a_r, sc_r, f"restart_s{sigma}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 15),
                                                  max_iter=60,
                                                  tol_list=[0.002, 0.001, 0.0005, 0.0002])
                update_best(a_r2, sc_r2, f"restart_s{sigma}_newton")

    # Phase 8: Final polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight Newton
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            a_fin, sc_fin = _active_set_newton(best_seq, time.monotonic() + remaining * 0.5,
                                               max_iter=300,
                                               tol_list=[0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00001])
            update_best(a_fin, sc_fin, "final_newton")

        # Final L-BFGS with tiny beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.max()
            beta_final = mb * 0.00005
            max_iter = min(4000, max(50, int(remaining * 0.7 / 0.02)))
            try:
                result = minimize(
                    _lsobj,
                    z,
                    jac=_lsgrad,
                    args=(beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100}
                )
                a_fin3 = np.exp(result.x - result.x.max())
                s_fin = a_fin3.sum()
                if s_fin > 1e-300:
                    a_fin3 = a_fin3 / s_fin
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
            except Exception:
                pass

        # Final mirror descent micro-step
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            a_fin4, sc_fin4 = _mirror_descent(best_seq, time.monotonic() + remaining * 0.6, max_iter=5000)
            update_best(a_fin4, sc_fin4, "final_md")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
