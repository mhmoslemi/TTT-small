def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    h = wc[n - 1:2 * n - 1]

        g_half += grad_full[half:][-2::-1]
    # Account for softmax Jacobian on the half
    g_half = g_half - a_half * np.sum(g_half * a_half)
    return g_half


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2.0 * t - 1.0) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fourier":
        # Based on random phases for flat DFT
        rng = np.random.default_rng(12345)
        phases = rng.uniform(0, 2 * np.pi, _next_pow2(n))
        a = np.fft.irfft(np.exp(1j * phases), _next_pow2(n))[:n]
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.1) ** 2) + np.exp(-50.0 * (t - 0.9) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_symmetric(a, sigma, rng):
    """Perturb a sequence while maintaining symmetry."""
    n = len(a)
    half = (n + 1) // 2
    z = np.log(np.maximum(a[:half], 1e-30))
    z = z + sigma * rng.standard_normal(half)
    z = z - z.mean()
    a_half = _softmax(z)
    a_new = np.zeros(n)
    a_new[:half] = a_half
    if n % 2 == 0:
        a_new[half:] = a_half[::-1]
    else:
        a_new[half:] = a_half[-2::-1]
    return a_new / a_new.sum()


def _run_lbfgs_refined(a_init, deadline, min_time_remaining=30.0, beta_factor=1.0):
    """L-BFGS with adaptive beta continuation - refined version."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    # More gradual beta schedule
    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                                  0.08, 0.05, 0.035, 0.025, 0.018,
                                  0.012, 0.008, 0.005, 0.003, 0.002, 0.001])
    betas = betas * beta_factor

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                _make_lsobj_wrapper(beta, n),
                z,
                jac=_make_lsgrad_wrapper(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_lsobj_wrapper(beta, n):
    def wrapper(z):
        return _lsobj(z, beta, n)
    return wrapper


def _make_lsgrad_wrapper(beta, n):
    def wrapper(z):
        return _lsgrad(z, beta, n)
    return wrapper


def _run_lbfgs_sym(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)

    c_cur = _full_convolve(a_init, a_init)

    z_half = np.log(np.maximum(a_init[:half], 1e-30))

    current_a = a_init.copy()

            if sc_new < current_sc - 1e-15:

                z_half = np.log(np.maximum(a_half_new, 1e-30))

    return current_a


def _active_set_newton_v2(a_init, deadline, max_iter=500):
    """Improved active-set Newton with adaptive tolerance and exact KKT handling."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Adaptive tolerance: start loose, tighten over iterations
    base_tols = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Select tolerances based on iteration (progressively tighter)
        tol_idx = min(it // 20, len(base_tols) - 1)
        tol_a = base_tols[tol_idx]

        # Try a few tolerance levels around the current one
        tol_candidates = [base_tols[max(0, tol_idx - 1)], tol_a,
                          base_tols[min(len(base_tols) - 1, tol_idx + 1)]]

        best_dir = None
        best_t = 0.0
        best_obj_after = best_sc

        for tol_try in tol_candidates:
            active = c >= (c_max - tol_try * c_max)
            if not active.any():
                active[np.argmax(c)] = True
            m = int(active.sum())
            if m < 2:
                continue
            if m > 6 * n:
                top_idx = np.argsort(c)[::-1][:6 * n]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                m = int(active.sum())

            A_vec = active.astype(np.float64)

            # Compute gradient on active set
            g = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            # Regularize Hessian
            mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

            def hess_vp(v, A_vec=A_vec, m=m, mu=mu, n=n):
                return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            u = _cg_solve(hess_vp, g_t, max_cg=250)
            w = _cg_solve(hess_vp, np.ones(n), max_cg=250)
            w_sum = float(w.sum())

            if abs(w_sum) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / w_sum
                dx = -u + lam * w

            dx = dx - np.mean(dx)

            # Check if this direction actually reduces the max
            # Quick evaluation
            neg = dx < -1e-12
            if neg.any():
                t_max = np.min(a[neg] / (-dx[neg])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            # Try full step first
            t = min(1.0, t_max)
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())

            if f_try < c_max - 1e-14:
                sc_try = _score(a_try)
                if sc_try < best_obj_after:
                    best_obj_after = sc_try
                    best_dir = dx
                    best_t = t

        if best_dir is not None:
            a = a + best_t * best_dir
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

            c_new = _full_convolve(a, a)
            if abs(float(c_new.max()) - c_max) < 5e-14:
                break
        else:
            # Try a small gradient step as fallback
            g_all = _corr_a_A(a, np.ones(2 * n - 1), n)
            g_t = g_all - np.mean(g_all)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            # Backtracking line search
            t = 0.1
            found = False
            for _ in range(30):
                a_try = a - t * g_t / gnorm
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_full_convolve(a_try, a_try).max())
                if f_try < c_max - 1e-14:
                    a = a_try
                    found = True
                    break
                t *= 0.5

            if not found:
                break

    return best_a, best_sc


def _water_filling(a_init, deadline, n_iters=100):
    """Water-filling: iteratively reduce peaks and equalize convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_sorted = np.sort(c)

        # Identify peaks (top 10%) and valleys (bottom 10%)
        n_proc = max(3, n // 10)
        peak_threshold = c_sorted[-n_proc]
        valley_threshold = c_sorted[n_proc]

        # Compute the "pressure" at each position:
        # positive where c is above average, negative where below
        avg_c = c.mean()
        pressure = c - avg_c

        # For each position i in a, compute the weighted contribution to
        # peaks vs valleys
        # Gradient of peak weight: sum_{k: c[k]>thresh} a[k-i]
        # Gradient of valley weight: sum_{k: c[k]<thresh} a[k-i]

        peak_w = np.zeros(2 * n - 1)
        peak_w[c >= peak_threshold] = 1.0
        valley_w = np.zeros(2 * n - 1)
        valley_w[c <= valley_threshold] = 1.0

        # Normalize
        if peak_w.sum() > 0:
            peak_w /= peak_w.sum()
        if valley_w.sum() > 0:
            valley_w /= valley_w.sum()

        grad_peak = _corr_a_A(a, peak_w, n)
        grad_valley = _corr_a_A(a, valley_w, n)

        # Direction: reduce contribution to peaks, increase contribution to valleys
        direction = grad_valley - grad_peak
        direction = direction - np.mean(direction)  # maintain sum

        # Scale step
        step = 0.05 / (1 + 0.01 * it)
        # Backtracking
        for ls in range(20):
            a_try = a + step * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            f_try = float(_full_convolve(a_try, a_try).max())
            if f_try < c_max - 1e-14:
                a = a_try
                break
            step *= 0.5
        else:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _deflation_v2(a_init, deadline, n_attempts=30):
    """Improved deflation: target top peaks with proportional reduction."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top k peaks
        n_top = max(5, min(len(c), int(0.02 * len(c))))
        top_idx = np.argsort(c)[::-1][:n_top]

        # For each top peak, compute the gradient (which a[i] contributes most)
        # and reduce proportionally
        a_new = a.copy()
        total_reduction = 0.0

        for k_pos in top_idx:
            # Gradient of c[k_pos] w.r.t. a[i] is 2*a[k_pos - i] for valid i
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            if i_hi < i_lo:
                continue

            # The contribution from a[i] to c[k_pos] is a[i] * a[k_pos - i]
            # We want to reduce a[i] where this product is largest
            contributions = np.array([a[i] * a[k_pos - i] for i in range(i_lo, i_hi + 1)])
            total_contrib = contributions.sum()
            if total_contrib < 1e-30:
                continue

            # Reduce each a[i] proportionally to its contribution
            reduction_frac = 0.002
            for i in range(i_lo, i_hi + 1):
                if contributions[i - i_lo] > 0:
                    a_new[i] -= reduction_frac * contributions[i - i_lo] / total_contrib * c[k_pos]
                    total_reduction += reduction_frac * contributions[i - i_lo] / total_contrib * c[k_pos]

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try smaller reduction
            reduction_frac = 0.0005
            a_small = a.copy()
            for k_pos in top_idx[:10]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                if i_hi < i_lo:
                    continue
                contributions = np.array([a_small[i] * a_small[k_pos - i] for i in range(i_lo, i_hi + 1)])
                total_contrib = contributions.sum()
                if total_contrib < 1e-30:
                    continue
                for i in range(i_lo, i_hi + 1):
                    if contributions[i - i_lo] > 0:
                        a_small[i] -= reduction_frac * contributions[i - i_lo] / total_contrib * c[k_pos]

            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-13:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break

    return best_a, best_sc


def _fista_v2(a_init, deadline, n_steps=15000):
    """FISTA with adaptive step size and active set averaging."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = _score(a)
    c_max_initial = float(_full_convolve(a, a).max())

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Active set with tolerance that shrinks over time
        tol = max(0.001, 0.05 * (1 - step / n_steps)) * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad = 2.0 * _corr_a_A(y, w, n)
        g_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step_size = 0.5 / (gnorm * (1 + 0.005 * step))
        step_size = min(step_size, 0.1)

        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t = t_new

        if step % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_symmetric(a):
    """Make a sequence symmetric."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    a_sym[:half] = a[:half]
    if n % 2 == 0:
        a_sym[half:] = a[:half][::-1]
    else:
        a_sym[half:] = a[:half-1][::-1]
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s


def _equalize_active(a_init, deadline, max_iter=200):
    """Try to equalize all active (near-max) convolution values exactly."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: top values within tolerance
        active = c >= (c_max - 0.001 * c_max)
        if not active.any():
            active[np.argmax(c)] = True
        m = int(active.sum())

        # At optimum: for all active k, c[k] should be as equal as possible
        # and the gradient condition should hold
        # Use a simple relaxation: nudge a to equalize active c values

        # Compute the "imbalance" vector
        c_active = c[active]
        c_mean_active = c_active.mean()
        c_dev = c - c_mean_active  # positive where above mean, negative below

        # Gradient of the "variance" of active c values w.r.t. a
        # d/dc[k] for a[i] is a[k-i]
        # We want to reduce variance, so move in direction that reduces
        # the spread of c[k] for active k

        # Simple approach: weighted gradient
        w_dev = np.zeros(2 * n - 1)
        w_dev[active] = c_dev[active] / (c_mean_active * m) if c_mean_active > 0 else 0

        grad = _corr_a_A(a, w_dev, n)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step to reduce the spread
        step = 0.02 / (1 + 0.01 * it)
        for ls in range(20):
            a_try = a + step * grad / gnorm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-14:
                a = a_try
                break
            step *= 0.5
        else:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining refined L-BFGS, active-set Newton,
    water-filling, deflation, and diverse restarts with structured initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Refined L-BFGS on parent (long run with gradual beta)
    print(f"\n[Phase 1] Refined L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 600, t_start + 400)
    a_main = _run_lbfgs_refined(parent, lbfgs_deadline, min_time_remaining=600.0, beta_factor=1.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton (multi-pass with tightening tolerance)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 200)
        a_as, sc_as = _active_set_newton_v2(best_seq, as_dl, max_iter=400)
        update_best(a_as, sc_as, "as_newton_1")

    # Phase 3: Water filling
    print(f"\n[Phase 3] Water filling (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        wf_dl = min(deadline - 400, time.monotonic() + 80)
        a_wf, sc_wf = _water_filling(best_seq, wf_dl, n_iters=200)
        update_best(a_wf, sc_wf, "water_fill")

    # Phase 4: Equalize active set
    print(f"\n[Phase 4] Equalize active (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        eq_dl = min(deadline - 350, time.monotonic() + 60)
        a_eq, sc_eq = _equalize_active(best_seq, eq_dl, max_iter=200)
        update_best(a_eq, sc_eq, "equalize")

    # Phase 5: Deflation v2
    print(f"\n[Phase 5] Deflation v2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        def_dl = min(deadline - 300, time.monotonic() + 60)
        a_def, sc_def = _deflation_v2(best_seq, def_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation_v2")

    # Phase 6: Second L-BFGS pass (with smaller beta factor)
    print(f"\n[Phase 6] L-BFGS pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        lbfgs2_dl = min(deadline - 250, time.monotonic() + 200)
        a_main2 = _run_lbfgs_refined(best_seq, lbfgs2_dl, min_time_remaining=250.0, beta_factor=0.5)
        sc_main2 = _score(a_main2)
        update_best(a_main2, sc_main2, "lbfgs_pass2")

    # Phase 7: Symmetric optimization
    print(f"\n[Phase 7] Symmetric L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        sym_a = _make_symmetric(best_seq)
        sym_sc = _score(sym_a)
        update_best(sym_a, sym_sc, "sym_init")

        sym_dl = min(deadline - 200, time.monotonic() + 120)
        a_sym_opt = _run_lbfgs_sym(sym_a, sym_dl, min_time_remaining=200.0)
        sc_sym = _score(a_sym_opt)
        update_best(a_sym_opt, sc_sym, "lbfgs_sym")

        # Also try symmetric optimization from a symmetric initialization
        if time.monotonic() < deadline - 200:
            n_try = len(best_seq)
            a_sym_init = _make_profile(n_try, "edge_heavy")
            a_sym_init = _make_symmetric(a_sym_init)
            sym_dl2 = min(deadline - 200, time.monotonic() + 80)
            a_sym_opt2 = _run_lbfgs_sym(a_sym_init, sym_dl2, min_time_remaining=150.0)
            sc_sym2 = _score(a_sym_opt2)
            update_best(a_sym_opt2, sc_sym2, "lbfgs_sym_init")

    # Phase 8: Diverse restarts with structured initializations
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    n_main = len(best_seq) if best_seq is not None else 16384

    restart_profiles = [
        ("edge_heavy", "edge_heavy"),
        ("quadratic", "quadratic"),
        ("gaussian", "gaussian"),
        ("bimodal", "bimodal"),
        ("cos", "cos"),
        ("fourier", "fourier"),
    ]

    for label, kind in restart_profiles:
        if time.monotonic() > deadline - 60:
            break

        a_init_r = _make_profile(n_main, kind)
        # If we have a parent, blend with it
        if parent is not None:
            a_parent_interp = _interpolate_to_n(parent, n_main)
            a_init_r = 0.5 * a_init_r + 0.5 * a_parent_interp
            a_init_r = a_init_r / a_init_r.sum()

        sc_init_r = _score(a_init_r)
        if sc_init_r < best_score + 0.1:
            r_dl = min(deadline - 40, time.monotonic() + 80)
            a_r = _run_lbfgs_refined(a_init_r, r_dl, min_time_remaining=40.0, beta_factor=0.8)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 9: Perturbation restarts around best
    print(f"\n[Phase 9] Perturbation restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for sigma in [0.01, 0.03, 0.05, 0.08]:
            if time.monotonic() > deadline - 30:
                break
            a_p = _perturb(best_seq, sigma, rng)
            p_dl = min(deadline - 20, time.monotonic() + 50)
            a_p_opt = _run_lbfgs_refined(a_p, p_dl, min_time_remaining=20.0, beta_factor=0.5)
            sc_p = _score(a_p_opt)
            update_best(a_p_opt, sc_p, f"perturb_{sigma}")

        # Symmetric perturbation
        if time.monotonic() < deadline - 30:
            a_sp = _perturb_symmetric(best_seq, 0.03, rng)
            sp_dl = min(deadline - 20, time.monotonic() + 40)
            a_sp_opt = _run_lbfgs_sym(a_sp, sp_dl, min_time_remaining=20.0)
            sc_sp = _score(a_sp_opt)
            update_best(a_sp_opt, sc_sp, "perturb_sym")

    # Phase 10: Multi-n exploration
    print(f"\n[Phase 10] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_main:
            continue

        if parent is not None:
            a_init_n = _interpolate_to_n(parent, n_try)
            sc_init_n = _score(a_init_n)
        else:
            a_init_n = _make_profile(n_try, "uniform")
            sc_init_n = _score(a_init_n)

        n_dl = min(deadline - 40, time.monotonic() + 100)
        a_n_opt = _run_lbfgs_refined(a_init_n, n_dl, min_time_remaining=40.0, beta_factor=1.0)
        sc_n = _score(a_n_opt)
        update_best(a_n_opt, sc_n, f"n_{n_try}")

        # Quick Newton polish
        if sc_n < best_score + 0.01 and time.monotonic() < deadline - 30:
            as_polish = min(deadline - 20, time.monotonic() + 40)
            a_n_as, sc_n_as = _active_set_newton_v2(a_n_opt, as_polish, max_iter=100)
            update_best(a_n_as, sc_n_as, f"n_{n_try}_as")

    # Phase 11: Final intensive polish on best
    print(f"\n[Phase 11] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Active-set Newton with very tight tolerance
        as_final = min(deadline - 3, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_newton_v2(best_seq, as_final, max_iter=300)
        update_best(a_fin, sc_fin, "final_as")

        # Water filling
        if time.monotonic() < deadline - 3:
            wf_final = min(deadline - 2, time.monotonic() + 30)
            a_wf2, sc_wf2 = _water_filling(best_seq, wf_final, n_iters=100)
            update_best(a_wf2, sc_wf2, "final_wf")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        _make_lsobj_wrapper(beta_final, n),
                        z,
                        jac=_make_lsgrad_wrapper(beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # One more active-set Newton after L-BFGS
        if time.monotonic() < deadline - 2:
            as_ult = min(deadline - 1, time.monotonic() + 20)
            a_ult, sc_ult = _active_set_newton_v2(best_seq, as_ult, max_iter=100)
            update_best(a_ult, sc_ult, "final_ult_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
