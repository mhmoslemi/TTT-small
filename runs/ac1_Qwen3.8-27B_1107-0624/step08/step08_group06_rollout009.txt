def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _hess_vp(v, active, n, mu=0.0):
    """Hessian-vector product: (2/m) * (H_S * v) + mu * v
    where H_S = (1/m) * sum_{k in active} H_k
    (H_S * v)[p] = (2/m) * sum_{k in active} v[k-p] (if 0 <= k-p < n)
    """
    m = float(active.sum())
    if m < 1e-15:
        return mu * v
    # (H_S * v)[p] = sum_{q=0}^{n-1} [p+q in active] * v[q] / m
    # This is cross-correlation of active with v
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    f_active = np.fft.rfft(active, N)
    f_v = np.fft.rfft(v, N)
    # cross-correlation: ifft(fft(active) * conj(fft(v)))
    corr = np.fft.irfft(f_active * np.conj(f_v), N)[:out_len]
    # corr[p] = sum_q active[p+q] * v[q] for p = 0, ..., 2n-2
    # We want result[p] = (2/m) * corr[p] + mu * v[p] for p = 0, ..., n-1
    result = (2.0 / m) * corr[:n] + mu * v
    return result


def _grad_active(a, active, n):

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:

    x_old = np.linspace(0, 1, n_old)

    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    f_v = np.fft.rfft(v, N)
    corr = np.fft.irfft(f_a * np.conj(f_v), N)[:out_len]
    return (2.0 / m) * corr[:n] + mu * v


def _grad_active(a, active, n):
    """Gradient of (1/m) * sum_{k in S} (a*a)[k], shape (n,)."""
    m = float(active.sum())
    if m < 1e-15:
        return np.zeros(n)
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    f_a = np.fft.rfft(a, N)
    f_act = np.fft.rfft(active, N)
    corr = np.fft.irfft(f_act * np.conj(f_a), N)[:out_len]
    return (2.0 / m) * corr[:n]


def _cg_solve(hess_fn, b, max_iter=300, tol=1e-13):
    """Solve hess_fn(x) = b using conjugate gradient on tangent space."""
    n = len(b)
    x = np.zeros(n)
    r = b - np.mean(b)
    p = r.copy()
    rs = np.dot(r, r)
    b_norm = np.linalg.norm(b)
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        if rs < tol * tol * (1 + b_norm * b_norm):
            break
        Hp = hess_fn(p)
        Hp_t = Hp - np.mean(Hp)
        pHp = np.dot(p, Hp_t)
        if pHp < 1e-28:
            break
        alpha = rs / pHp
        x = x + alpha * p
        x = x - np.mean(x)
        r = b - hess_fn(x)
        r = r - np.mean(r)
        rs_new = np.dot(r, r)
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        p = p - np.mean(p)
        rs = rs_new
    return x


def _newton_step(a, active, n, mu):
    """One Newton step for the active set QP. Returns direction d (tangent)."""
    g = _grad_active(a, active, n)
    g_t = g - np.mean(g)
    gnorm = np.linalg.norm(g_t)
    if gnorm < 1e-15:
        return None, gnorm

    def hess_fn(v):
        return _hess_vec(v, active, n, mu)

    b = -g_t
    d = _cg_solve(hess_fn, b, max_iter=250)
    return d, gnorm


def _newton_optimize(a_init, deadline, max_iter=300, tol_frac=0.002, n_max_active_frac=4.0):
    """Newton method with adaptive active set and trust region."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    best_cmax = _self_convolve(a).max()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _self_convolve(a)
        c_max = float(c.max())

        # Hysteresis-based active set
        threshold = c_max * (1.0 - tol_frac)
        active_bool = c >= threshold
        m = int(active_bool.sum())

        if m < 3:
            top_k = np.argsort(c)[::-1][:max(3, int(0.01 * n))]
            active_bool = np.zeros(len(c), dtype=bool)
            active_bool[top_k] = True
            m = int(active_bool.sum())

        max_active = int(n_max_active_frac * n)
        if m > max_active:
            top_idx = np.argsort(c)[::-1][:max_active]
            active_bool = np.zeros(len(c), dtype=bool)
            active_bool[top_idx] = True
            m = max_active

        active_f = active_bool.astype(np.float64)

        # Regularization
        mu = max(1e-12, 1e-8 * c_max / m)

        d, gnorm = _newton_step(a, active_f, n, mu)
        if d is None:
            break

        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break

        # Trust region step size
        neg = d < -1e-12
        if neg.any():
            t_boundary = float(np.min(a[neg] / (-d[neg]))) * 0.99
        else:
            t_boundary = 1.0

        # Scale direction
        t_scale = min(1.0, 2.0 * gnorm / (dnorm * dnorm + 1e-30))
        t = min(t_boundary, t_scale)
        t = max(t, 1e-10)

        # Backtracking line search
        accepted = False
        for ls in range(50):
            a_try = a + t * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _self_convolve(a_try)
            cmax_try = float(c_try.max())

            if cmax_try < c_max - 1e-14 * c_max:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Try gradient descent step as fallback
            g_all = _grad_active(a, active_f, n)
            g_t = g_all - np.mean(g_all)
            gnorm_all = np.linalg.norm(g_t)
            if gnorm_all > 1e-15:
                t_g = 0.1 / gnorm_all
                a_try = a - t_g * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    c_try = _self_convolve(a_try)
                    if float(c_try.max()) < c_max:
                        a = a_try
                        accepted = True

        if not accepted:
            # Try with smaller tol (more active constraints)
            tol_frac_new = tol_frac * 0.5
            if tol_frac_new < 1e-5:
                break
            threshold2 = c_max * (1.0 - tol_frac_new)
            active_bool2 = c >= threshold2
            m2 = int(active_bool2.sum())
            if m2 > min(max_active, 6 * n):
                top_idx2 = np.argsort(c)[::-1][:min(max_active, 6 * n)]
                active_bool2 = np.zeros(len(c), dtype=bool)
                active_bool2[top_idx2] = True
                m2 = int(active_bool2.sum())
            active_f2 = active_bool2.astype(np.float64)
            d2, _ = _newton_step(a, active_f2, n, mu)
            if d2 is not None:
                dnorm2 = np.linalg.norm(d2)
                neg2 = d2 < -1e-12
                if neg2.any():
                    t2 = float(np.min(a[neg2] / (-d2[neg2]))) * 0.99
                else:
                    t2 = min(1.0, 2.0 * gnorm / (dnorm2 * dnorm2 + 1e-30))
                t2 = max(t2, 1e-10)
                for ls in range(30):
                    a_try = a + t2 * d2
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        t2 *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _self_convolve(a_try)
                    if float(c_try.max()) < c_max - 1e-14:
                        a = a_try
                        accepted = True
                        break
                    t2 *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_cmax = c_max
            best_a = a.copy()

        # Convergence check
        c_new = _self_convolve(a)
        if abs(float(c_new.max()) - c_max) < 1e-13 * c_max:
            break

    return best_a, best_sc


def _spectral_flatten(a, n_iters=30):
    """Frequency-domain iterative flattening of autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_mag_sq = A_mag ** 2

        # Reduce spectral peaks (makes autocorrelation flatter)
        max_mag_sq = float(np.max(A_mag_sq))
        if max_mag_sq < 1e-30:
            break

        # Proportional peak reduction
        alpha = 0.3
        A_mag_sq_new = A_mag_sq - alpha * A_mag_sq ** 2 / max_mag_sq
        A_mag_sq_new = np.maximum(A_mag_sq_new, 0)

        # Preserve DC (sum constraint)
        A_mag_sq_new[0] = 1.0

        A_new = np.sqrt(A_mag_sq_new) * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _logsumexp_smooth(a, beta, n):
    """Log-sum-exp smoothing of max convolution."""
    c = _self_convolve(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_smooth(a_init, deadline, beta_start=None, n_betas=8):
    """L-BFGS with log-sum-exp smoothing and adaptive beta."""
    from scipy.optimize import minimize

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    c = _self_convolve(a)
    max_c = float(c.max())
    if beta_start is None:
        beta_start = max_c * 0.01

    betas = beta_start * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.007, 0.003, 0.001])
    betas = betas[:n_betas]

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    current_a = a.copy()
    current_sc = _score(a)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        def obj_func(z_local):
            a_local = _softmax_helper(z_local)
            return _logsumexp_smooth(a_local, beta, n)

        def grad_func(z_local):
            a_local = _softmax_helper(z_local)
            c_local = _self_convolve(a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -500, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s_e
            # Gradient of logsumexp w.r.t. a: 4*n*a*(corr(w, a_rev)[n-1:2n-1] - dot(w,c))
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_local))
            grad_a = 4.0 * n * a_local * (h - W)
            # Chain rule: d/da * da/dz = grad_a * (a - outer(a,a))
            grad_z = a_local * (grad_a - np.dot(a_local, grad_a))
            return grad_z

        max_iter = max(50, min(3000, int(remaining / 0.05)))

        try:
            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxcor': 50}
            )
            a_new = _softmax_helper(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _softmax_helper(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_initial(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1,)
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        bs = n // 4
        a[:bs] = 1.0
        a[3 * bs:] = 1.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Spectral flattening preprocessing
    2. Multi-resolution Newton (n/4 -> n/2 -> n)
    3. L-BFGS with adaptive log-sum-exp smoothing
    4. Final tight Newton polish
    5. Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_initial(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_main = n_parent if n_parent > 0 else 8192

    # Phase 1: Spectral flattening on parent
    print(f"\n[Phase 1] Spectral flattening (t={time.monotonic() - t_start:.1f}s)")
    spec_dl = min(deadline - 1400, time.monotonic() + 30)
    a_spec = _spectral_flatten(parent, n_iters=25)
    sc_spec = _score(a_spec)
    update_best(a_spec, sc_spec, "spectral")

    # Use the best so far as main working point
    a_main = best_seq.copy() if best_seq is not None else parent.copy()
    n_main = len(a_main)

    # Phase 2: Multi-resolution Newton
    print(f"\n[Phase 2] Multi-resolution Newton (t={time.monotonic() - t_start:.1f}s)")
    # Start at n/4
    n_crude = max(64, n_main // 4)
    if n_crude < n_main:
        a_crude = _interpolate_to_n(a_main, n_crude)
        # Optimize at coarse level
        crude_dl = min(deadline - 1200, time.monotonic() + 60)
        a_crude, sc_crude = _newton_optimize(a_crude, crude_dl, max_iter=200, tol_frac=0.005)
        print(f"  n={n_crude}: score={sc_crude:.10f}")
        # Upscale
        a_main = _interpolate_to_n(a_crude, n_main)
        update_best(a_main, _score(a_main), "upscaled_4x")

    # Newton at full resolution - multiple rounds with decreasing tol
    tol_rounds = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for i, tol in enumerate(tol_rounds):
        if time.monotonic() > deadline - 400:
            break
        round_dl = min(deadline - 400, time.monotonic() + 150)
        a_main, sc = _newton_optimize(a_main, round_dl, max_iter=200, tol_frac=tol)
        update_best(a_main, sc, f"newton_tol_{tol}")
        if sc >= best_score - 1e-13:
            # No improvement, try a different starting perturbation
            if i < len(tol_rounds) - 1:
                rng = np.random.default_rng(12345 + i)
                z = np.log(np.maximum(a_main, 1e-30))
                z = z + 0.01 * rng.standard_normal(len(z))
                a_main = _softmax_helper(z - z.mean())
                update_best(a_main, _score(a_main), f"perturb_tol_{tol}")

    # Phase 3: L-BFGS with log-sum-exp smoothing
    print(f"\n[Phase 3] L-BFGS smoothing (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        lbfgs_dl = min(deadline - 200, time.monotonic() + 200)
        a_lbfgs = _lbfgs_smooth(a_main, lbfgs_dl, beta_start=None, n_betas=6)
        sc_lbfgs = _score(a_lbfgs)
        update_best(a_lbfgs, sc_lbfgs, "lbfgs_smooth")

        # Newton after L-BFGS
        if time.monotonic() < deadline - 150:
            nt_dl = min(deadline - 150, time.monotonic() + 120)
            a_post, sc_post = _newton_optimize(a_lbfgs, nt_dl, max_iter=150, tol_frac=0.001)
            update_best(a_post, sc_post, "newton_post_lbfgs")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_main // 2, n_main, n_main * 2]
    # Add some nearby values
    n_candidates.append(max(64, n_main - 1))
    n_candidates.append(n_main + 1)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_main:
            continue
        if n_try > 65536:
            continue

        a_init_n = _interpolate_to_n(best_seq, n_try)
        # Quick spectral pass
        a_init_n = _spectral_flatten(a_init_n, n_iters=5)

        # Newton
        if time.monotonic() < deadline - 30:
            nt_dl = min(deadline - 30, time.monotonic() + 60)
            a_n, sc_n = _newton_optimize(a_init_n, nt_dl, max_iter=100, tol_frac=0.002)
            update_best(a_n, sc_n, f"n_{n_try}")

    # Phase 5: Final tight polish
    print(f"\n[Phase 5] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        fin_dl = deadline - 3
        a_fin, sc_fin = _newton_optimize(best_seq, fin_dl, max_iter=200, tol_frac=0.0001)
        update_best(a_fin, sc_fin, "final_newton_tight")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            n_fin = len(best_seq)
            c_fin = _self_convolve(best_seq)
            mb = float(c_fin.max())
            beta_tiny = mb * 0.0001
            z_fin = np.log(np.maximum(best_seq, 1e-30))
            z_fin = z_fin - z_fin.mean()

            def tiny_obj(z_local):
                a_local = _softmax_helper(z_local)
                return _logsumexp_smooth(a_local, beta_tiny, n_fin)

            def tiny_grad(z_local):
                a_local = _softmax_helper(z_local)
                c_local = _self_convolve(a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta_tiny
                e = np.exp(np.clip(shifted, -500, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                a_rev = a_local[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n_fin - 1:2 * n_fin - 1]
                W = float(np.dot(w, c_local))
                grad_a = 4.0 * n_fin * a_local * (h - W)
                grad_z = a_local * (grad_a - np.dot(a_local, grad_a))
                return grad_z

            from scipy.optimize import minimize
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                try:
                    result = minimize(
                        tiny_obj, z_fin,
                        jac=tiny_grad,
                        method='L-BFGS-B',
                        options={'maxiter': min(1000, max(30, int(remaining / 0.03))),
                                 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 40}
                    )
                    a_tiny = _softmax_helper(result.x)
                    update_best(a_tiny, _score(a_tiny), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
