# Helper functions used in the optimization
def _next_power_of_two(n):
    return 1 << (n - 1).bit_length()

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_len = n + m - 1
    full_len_power = _next_power_of_two(out_len)
    fa = np.fft.rfft(a, full_len_power)
    fb = np.fft.rfft(b, full_len_power)
    conv = np.fft.irfft(fa * np.conj(fb), out_len)[:out_len]
    return conv

def _clip_and_normalize(z, a):

    z_max = np.max(z)
    return np.exp(z / z_max) if z_max != 0 else np.ones(len(z))

def score(a):
    n = len(a)
    conv = _full_convolve(a, a)
    conv_max = np.max(conv)
    a_sum = np.sum(a)
    return 2.0 * n * conv_max / (a_sum ** 2)

def full_convolve(a, b=None, mode='full'):
    if b is None:
        b = a
    conv = _full_convolve(a, b)
    return conv[::-1] if mode == 'full' else conv[:]
