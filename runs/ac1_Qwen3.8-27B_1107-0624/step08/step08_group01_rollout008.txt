    # Initial random non-negative sequence
    a = np.random.default_rng(rng).random(n) + 0.1
    a = a / a.sum()
    
    target_mag = np.ones(n) / np.sqrt(n)
    
    for it in range(n_iter):
        # Forward DFT
        A = np.fft.fft(a)
        # Replace magnitude with target, keep phase
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Inverse DFT
        a = np.real(np.fft.ifft(A_new))
        # Project to non-negative
        a = np.maximum(a, 0)
        # Normalize
        s = a.sum()
        if s > 0:
            a = a / s
    
    return a

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

This is interesting! It says: increase `z[i]` (increase `a[i]`) where `a[k*-i]` is small compared to the peak, and decrease where `a[k*-i]` is large.

But wait, this is for a single peak. If there are multiple peaks at the same height, we should average over them.

    """One step of subgradient descent on the peak of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    
    if k_lags is None:
        k_lags = [int(np.argmax(c))]
    
    # Average gradient over the selected lags
    # z-gradient for lag k: 2*a[i]*(c[k] - a[k-i]) for valid i
    grad_z = np.zeros(n)
    for k in k_lags:
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                grad_z[i] += 2 * a[i] * (c[k] - a[j])
    grad_z /= len(k_lags)

    z = np.log(np.maximum(a, 1e-30))
    z -= lr * grad_z
    z -= z.mean()  # Center (scale invariance)
    a_new = np.exp(z)
    a_new = a_new / a_new.sum()
    return a_new

V2 optimizer: Combines Adam-based LSE optimization, Gerchberg-Saxton phase
retrieval restarts, direct peak-attack subgradient steps, and multi-n search
to improve upon the parent's score of ~1.5057. Key innovations:

2. Gerchberg-Saxton flat-spectrum initialization for diverse restarts
3. Aggressive peak-attack with smart mass redistribution
4. Convex polishing via projected gradient in a LOCAL (fixed-active-set) regime
5. Multi-n exploration with conditional resampling

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_grad(z, beta, n):
    """Compute LSE-smoothed objective and gradient in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _adam_lbfgs(z_init, n, deadline, beta_start, beta_end, min_remaining=15.0,
                lr_init=0.5, lr_decay=0.9995):
    """Adam optimizer on LSE-smoothed objective with beta annealing."""
    z = z_init.copy()
    a_cur = _softmax(z)
    best_a = a_cur.copy()
    best_sc = _score(best_a)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = lr_init

    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)

    while time.monotonic() < deadline - min_remaining:
        progress = min(1.0, (time.monotonic() - t_start) / total_time)
        # Exponential beta annealing
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        beta = max(beta, 1e-10)

        obj, grad = _lse_obj_grad(z, beta, n)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _softmax(z_new)
        sc = _score(a_new)

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new.copy()
            z = z_new.copy()
        else:
            # Still update z but decay lr
            z = z_new.copy()
            lr *= lr_decay
            if lr < 1e-6:
                lr = 1e-6

        if t % 5000 == 0:
            print(f"    adam: t={t}, sc={best_sc:.10f}, beta={beta:.6f}, lr={lr:.6f}")

    return best_a, best_sc


def _lbfgs_lbfgs(z_init, n, deadline, beta_start, beta_end, min_remaining=15.0):
    """L-BFGS on LSE with aggressive annealing."""
    n_betas = 25
    betas = beta_start * (beta_end / beta_start) ** (np.arange(n_betas) / (n_betas - 1))
    z = z_init.copy()
    a_cur = _softmax(z)
    best_a = a_cur.copy()
    best_sc = _score(best_a)

    class LseProb:
        def __init__(self, beta_val, n_val):
            self._beta = beta_val
            self._n = n_val

        def __call__(self, z):
            obj, _ = _lse_obj_grad(z, self._beta, self._n)
            return obj

        def grad(self, z):
            _, g = _lse_obj_grad(z, self._beta, self._n)
            return g

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - min_remaining:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break
        scale = max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (0.003 * scale * min(1.0, (i+1)/n_betas)))))
        prob = LseProb(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-17,
                                       'gtol': 1e-13, 'maxcor': 60, 'maxls': 40})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = result.x - result.x.mean()
        except Exception:
            pass
        if (i + 1) % 5 == 0:
            print(f"    lbfgs: beta_idx={i+1}/{n_betas}, sc={best_sc:.10f}")

    return best_a, best_sc


def _peak_attack(a, deadline, n_iters=5000, min_remaining=10.0):
    """Direct peak-attack: iteratively reduce the maximum convolution value."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(best_a)
    lr = 0.01
    window = 5  # number of top lags to consider

    for it in range(n_iters):
        if time.monotonic() > deadline - min_remaining:
            break
        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc

        # Select top lags
        top_lags = np.argsort(c)[::-1][:window]
        w = np.zeros(len(c))
        w[top_lags] = 1.0 / window

        # Gradient in a-space: for each active lag k, grad_a[i] += a[k-i]
        grad_a = np.zeros(n)
        for k in top_lags:
            k_lo = max(0, k - n + 1)
            k_hi = min(n - 1, k)
            for i in range(k_lo, k_hi + 1):
                j = k - i
                if 0 <= j < n:
                    grad_a[i] += a_reverse_at(k, i, best_a)

        grad_a /= window
        # Center the gradient (preserve sum)
        grad_a -= grad_a.mean()

        # Line search
        alpha = lr
        improved = False
        for _ in range(30):
            a_try = best_a - alpha * grad_a
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc * (1 - 1e-10):
                best_a = a_try
                best_sc = sc_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.7
            if lr < 1e-8:
                break
        else:
            lr = min(0.05, lr * 1.02)

    return best_a, best_sc


def a_reverse_at(k, i, a):
    """Get a[k-i] with bounds check."""
    j = k - i
    if 0 <= j < len(a):
        return a[j]
    return 0.0


def _peak_attack_v2(a, deadline, n_iters=3000, min_remaining=10.0):
    """Faster peak attack using FFT-based gradient computation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(best_a)
    lr = 0.015
    window = 3

    for it in range(n_iters):
        if time.monotonic() > deadline - min_remaining:
            break
        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())

        top_lags = np.argsort(c)[::-1][:window]
        w = np.zeros(len(c))
        w[top_lags] = 1.0 / window

        # grad_a[i] = sum_k w[k] * a[k-i] (for valid k-i)
        # This is a correlation: grad_a = cross_corr(w, best_a) restricted to [0,n)
        a_rev = best_a[::-1]
        wc = _full_convolve(w, a_rev)
        grad_a = wc[n - 1:2 * n - 1].copy()

        # Center
        grad_a -= grad_a.mean()
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-16:

        # Normalized gradient
        grad_a = grad_a / gnorm

        # Adaptive step size
        alpha = lr
        improved = False
        for _ in range(25):
            a_try = best_a - alpha * grad_a
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            a_try = a_try / s_try

                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.6
            if lr < 1e-9:
                break
        else:
            lr = min(0.05, lr * 1.03)

    return best_a, best_sc


def _gerchberg_saxton_init(n, rng, n_iter=200, target_type='flat'):
    """Gerchberg-Saxton style initialization for flat magnitude spectrum."""
    a = rng.random(n) + 0.05
    a = a / a.sum()

    if target_type == 'flat':
        target_mag = np.ones(n) / np.sqrt(n)
    elif target_type == 'tapered':
        # Slightly tapered target to help with edges
        t = np.linspace(0, 1, n)
        target_mag = 0.7 + 0.3 * np.sin(np.pi * t)
        target_mag = target_mag / np.sqrt(np.sum(target_mag ** 2))
    else:
        target_mag = np.ones(n) / np.sqrt(n)

    for it in range(n_iter):
        A = np.fft.fft(a)
        phase = np.angle(A)
        # Avoid zero-magnitude issues
        mag = np.abs(A)
        safe_phase = phase.copy()
        safe_phase[mag < 1e-15] = rng.standard_normal(np.sum(mag < 1e-15))
        A_new = target_mag * np.exp(1j * safe_phase)
        a = np.real(np.fft.ifft(A_new))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    return a

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_power":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-0.3 * t ** 2)
        return a / a.sum()
    elif kind == "cos2":
        t = np.arange(n) / n
        a = 1.0 + 0.4 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    print(f"\n[P4] GS restarts (t={time.monotonic()-t_start:.1f}s)")
    p4_end = min(deadline - 150.0, t_start + 1800.0 * 0.78)
    if time.monotonic() < p4_end:
        gs_times = max(10.0, (p4_end - time.monotonic()) / 4.0)
        for gs_trial in range(4):
            if time.monotonic() > p4_end - 60.0:
                break
            a_gs = _gerchberg_saxton_init(n_best, rng, n_iter=100,
                                          target_type='flat' if gs_trial % 2 == 0 else 'tapered')
            sc_gs = _score(a_gs)
            print(f"    GS trial {gs_trial+1}: init score={sc_gs:.10f}")
            if sc_gs > best_score + 0.05:
                # Only optimize if reasonably close
                continue
            remaining_gs = p4_end - time.monotonic() - 30.0
            if remaining_gs < 30:
                break
            z = np.log(np.maximum(a_gs, 1e-30))
            z = z - z.mean()
            c_gs = _full_convolve(a_gs, a_gs)
            cm_gs = float(c_gs.max())
            bs_gs = cm_gs * 0.003
            a_opt, sc_opt = _adam_lbfgs(z, n_best, time.monotonic() + remaining_gs,
                                         bs_gs, cm_gs * 1e-5, min_remaining=20.0, lr_init=0.3)
            if sc_opt < best_score:
                best_a = a_opt
                best_score = sc_opt
                print(f"  P4 GS trial {gs_trial+1} improved: {best_score:.10f}")

    # ============================================================
    # Phase 5: Multi-n exploration (~15 min)
    # ============================================================
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    p5_end = min(deadline - 60.0, t_start + 1800.0 * 0.92)
    n_targets = [16384, 8192, 20480]
    if n_best not in n_targets:
        n_targets.append(n_best)

    for n_try in sorted(n_targets):
        if time.monotonic() > p5_end - 60.0:
            break
        if n_try == len(best_a):
            continue
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"    n={n_try}: init score={sc_init:.10f}")
        if sc_init > best_score + 0.02:
            continue
        remaining_n = p5_end - time.monotonic() - 30.0
        if remaining_n < 30:
            break
        z = np.log(np.maximum(a_init, 1e-30))
        z = z - z.mean()
        c_t = _full_convolve(a_init, a_init)
        cm_t = float(c_t.max())
        bs_t = cm_t * 0.003
        a_opt, sc_opt = _adam_lbfgs(z, n_try, time.monotonic() + remaining_n,
                                     bs_t, cm_t * 1e-5, min_remaining=20.0, lr_init=0.3)
        if sc_opt < best_score:
            best_a = a_opt
            best_score = sc_opt
            n_best = n_try
            print(f"  P5 n={n_try} improved: {best_score:.10f}")
        # Quick peak attack
        if time.monotonic() < p5_end - 30:
            a_pa, sc_pa = _peak_attack_v2(best_a, min(p5_end - 30, time.monotonic() + 60.0),
                                          n_iters=1000, min_remaining=10.0)
            if sc_pa < best_score:
                best_a = a_pa
                best_score = sc_pa
                print(f"  P5 n={n_try} peak_attack improved: {best_score:.10f}")

    # ============================================================
    # Phase 6: Final intense polish (~remaining time)
    # ============================================================
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 30.0:
        # Alternating peak attack + LSE
        n_iter_phases = max(1, int(remaining / 40.0))
        for phase in range(n_iter_phases):
            if time.monotonic() > deadline - 10.0:
                break
            half_time = remaining / (n_iter_phases - phase) if n_iter_phases > phase + 1 else remaining
            pa_end = min(deadline - 10.0, time.monotonic() + half_time * 0.5)
            if time.monotonic() < pa_end:
                a_pa, sc_pa = _peak_attack_v2(best_a, pa_end, n_iters=2000, min_remaining=5.0)
                if sc_pa < best_score:
                    best_a = a_pa
                    best_score = sc_pa
            lse_end = min(deadline - 5.0, time.monotonic() + half_time * 0.4)
            if time.monotonic() < lse_end:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                c_f = _full_convolve(best_a, best_a)
                cm_f = float(c_f.max())
                bs_f = cm_f * 0.002
                a_lse, sc_lse = _adam_lbfgs(z, len(best_a), lse_end, bs_f, cm_f * 5e-6,
                                             min_remaining=5.0, lr_init=0.1)
                if sc_lse < best_score:
                    best_a = a_lse
                    best_score = sc_lse
                    print(f"  P6 phase {phase+1}: {best_score:.10f}")

    # Final tiny LSE
    if time.monotonic() < deadline - 8.0:
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()
        c_f = _full_convolve(best_a, best_a)
        cm_f = float(c_f.max())
        try:
            class FinalLse:
                def __init__(self, bv, nv):
                    self._b = bv
                    self._n = nv
                def __call__(self, zz):
                    o, _ = _lse_obj_grad(zz, self._b, self._n)
                    return o
                def grad(self, zz):
                    _, g = _lse_obj_grad(zz, self._b, self._n)
                    return g
            fl = FinalLse(cm_f * 5e-6, len(best_a))
            result = minimize(fl, z, jac=fl.grad, method='L-BFGS-B',
                              options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 40})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_a = a_fin
                best_score = sc_fin
                print(f"  Final L-BFGS: {best_score:.10f}")
        except Exception:
            pass

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    final_score = _score(final)
    print(f"\n{'='*60}")
    print(f"FINAL: score={final_score:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
