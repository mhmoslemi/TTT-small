"""
Multi-phase optimizer for minimum peak autocorrelation sequences.
Uses direct simplex projection, Nesterov momentum, LP-based direction finding,
and adaptive peak-focusing subgradient methods.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        theta = (v.sum() - target_sum) / n
        return np.maximum(v - theta, 0.0)
    rho = idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _smoothed_grad(a, beta, n):
    """Compute gradient of -beta*log(sum exp(c[k]/beta)) w.r.t. a."""
    c = _full_convolve(a, a)
    shifted = (c - c.max()) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad_{a[j]} of sum_k w[k]*c[k] = sum_k w[k] * a[k-j] (for valid k-j)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return h  # This is the gradient of sum_k w[k]*c[k] w.r.t. a

def _subgradient_max(a, top_frac=0.05):
    """Compute subgradient of max_k c[k] using top_frac of peaks."""
    c = _full_convolve(a, a)
    n = len(a)
    n_active = max(1, int(len(c) * top_frac))
    c_sorted_idx = np.argsort(c)[::-1]
    active = c_sorted_idx[:n_active]
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return h, c, c_sorted_idx

def _nesterov_smoothed_opt(a_init, deadline, min_time_remaining=60.0):
    """Optimize using Nesterov-accelerated smoothed gradient on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    
    # Initialize momentum
    v = np.zeros(n)
    
    # Beta schedule: start large, decrease
    c0 = _full_convolve(a, a)
    max_c0 = c0.max()
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < 1e-15:
        gap = max_c0 * 0.01
    beta_start = max(gap * 2, max_c0 * 0.05)
    
    # Run with multiple beta levels
    n_levels = 12
    for level in range(n_levels):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        beta = beta_start * (0.5 ** level)
        # Number of iterations for this level
        n_iters = min(500, max(20, int(remaining / (0.01 * max(1, n/4096)) / n_levels)))
        
        # Step size: use 1/L where L is Lipschitz constant of the gradient
        # For the smoothed problem, L ≈ 4*n*max(a) / beta (rough estimate)
        lr = 1.0 / (4.0 * max(a.max() * n, 1.0))

def _lp_direction(a, active_indices, deadline):
    """Solve LP to find best direction d: min t s.t. B@d <= t, d>=-a, sum(d)=0.
    Returns the direction d and the predicted improvement."""
    n = len(a)
    c = _full_convolve(a, a)
    m = len(active_indices)
    
    # Build constraint matrix B where B[k, j] gives the coefficient of d[j] in conv(a+d, a)[k] - conv(a,a)[k]
    # conv(a+d, a)[k] = conv(a,a)[k] + conv(d,a)[k]
    # conv(d,a)[k] = sum_i d[i]*a[k-i]
    # So conv(d,a)[k] is linear in d with coefficients a[k-i]
    
    # We want: conv(d,a)[k] <= t - c[k] + c[k] = ... 
    # Actually: conv(a+d, a)[k] = c[k] + conv(d,a)[k] <= c_max_target
    # So: conv(d,a)[k] <= c_max_target - c[k]
    
    c_max_target = c.max() * 0.99  # try to reduce peak by 1%
    
    # Variables: d[0:n], t -> total n+1 variables
    # Constraints:
    # For each active k: sum_j B[k,j] * d[j] - t <= c_max_target - c[k]
    # d[j] >= -a[j] for all j (i.e., -d[j] <= a[j])
    # sum(d) = 0
    # Minimize: t (or 0, and just check feasibility with different targets)
    
    # Actually, let's minimize t:
    # min t
    # s.t. sum_j B[k,j]*d[j] - t <= c_max_target - c[k] for k in active
    #      -d[j] <= a[j] for all j
    #      sum(d) = 0
    
    # This is a standard LP. For n=16384, it has 16385 variables and ~16384+m constraints.
    # This might be too large for most LP solvers. Let's restrict to a subset of d.
    
    # Alternative: only allow d to be non-zero at positions that contribute to the active peaks.
    # For peak at position k, d[j] contributes if a[k-j] > 0, i.e., j is in [k-n+1, k] (clipped to [0,n-1]).
    
    # For simplicity, let's use a subset of positions.
    
    # Actually, for large n, LP might be too slow. Let's compute the direction analytically.
    # The optimal d minimizes t subject to the constraints. By duality, the optimal d
    # is in the direction of the "tightest" constraint.
    
    # Simpler approach: d = -alpha * h where h is the subgradient, with alpha chosen
    # to be the maximum step before hitting a constraint.
    
    # This is basically what the subgradient method does. So LP doesn't add much here.
    # Let's return the subgradient direction.
    
    w = np.zeros_like(c)
    for k in active_indices:
        w[k] = 1.0
    w /= len(active_indices)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    
    return -h  # direction to move

    a = a / s
    best_a = a.copy()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        # h is the gradient, we want to choose alpha so that a - alpha*h projected is good
        # Simple: alpha = 1 / ||h||_inf
        alpha = lr / (np.max(np.abs(h)) + 1e-15)

        for _ in range(30):

        c_new = _full_convolve(a, a)
        if c_new.max() < c_max * 0.999:
            lr = min(0.5, lr * 1.1)
        else:
            lr *= 0.9
    
    return best_a, best_score

"""
Optimized minimum-peak autocorrelation sequence finder.
Uses direct simplex projection, weighted subgradient with momentum,
and multi-phase strategy for progressive refinement.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        # All elements would be zeroed out; distribute evenly
        return np.full(n, target_sum / n)
    rho = idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _weighted_subgrad(a, c, weights, n):
    """Compute sum_k weights[k] * a[k-i] for valid i, via FFT."""
    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    return wc[n - 1:2 * n - 1]

                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
        
        beta *= 0.5
        momentum = min(0.99, momentum * 1.01)
        print(f"  Smoothed: beta={beta:.8f}, best={best_sc:.10f}, iters={total_iters}")
    
    return best_a, best_sc

def _phase_subgrad(a_init, deadline, min_remaining=30.0):
    """Phase 2: Weighted subgradient of exact max, with momentum and adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    v = np.zeros(n)
    momentum = 0.85
    lr = 0.1
    tol_frac = 0.1  # active set: peaks within 10% of max
    no_improve = 0
    
    for it in range(200000):
        if time.monotonic() + 1 > deadline - min_remaining:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive active set
        if no_improve > 50:
            tol_frac = max(0.01, tol_frac * 0.8)
            lr = max(0.001, lr * 0.7)
            v *= 0.5
            no_improve = 0
        elif no_improve > 20:
            tol_frac = max(0.01, tol_frac * 0.95)
        
        # Weighted subgradient: weight peaks by how much they exceed the mean
        threshold = c_max * (1 - tol_frac)
        active = c >= threshold
        n_active = max(1, int(active.sum()))
        
        # Weight proportional to excess
        excess = np.maximum(c - threshold, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            w = np.zeros_like(c)
            w[active] = 1.0 / n_active
        else:
            w = excess / total_excess
        
        g = _weighted_subgrad(a, c, w, n)
        
        # Nesterov step
        v = momentum * v - lr * g
        a_new = _project_simplex(a + v)
        
        # Verify improvement
        c_new = _full_convolve(a_new, a_new)
        if c_new.max() > c_max * 1.01:
            # Step was too large, reduce
            v = v * 0.5
            a_new = _project_simplex(a + 0.5 * v)
        
        a = a_new
    
    return best_a, best_sc

def _phase_qp_refine(a_init, deadline, min_remaining=30.0):
    """Phase 3: QP-based refinement using scipy's minimize for small active sets."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    from scipy.optimize import minimize
    
    for round in range(20):
        if time.monotonic() + 5 > deadline - min_remaining:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Select top peaks (small active set for QP)
        n_active = max(3, int(2 * (2*n-1) * 0.001))  # top 0.1%
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        # Build linearized constraint: for each active k,
        # conv(a+d, a)[k] <= c_max * 0.99
        # conv(a+d, a)[k] = c[k] + sum_i d[i]*a[k-i]
        # So: sum_i d[i]*a[k-i] <= c_max*0.99 - c[k]
        
        # This is a QP: min ||d||^2 s.t. B@d <= rhs, d >= -a, sum(d) = 0
        # Use scipy minimize with SLSQP
        
        # Only solve if we have time
        if time.monotonic() + 30 > deadline - min_remaining:
            break
        
        # Build B matrix (sparse-ish, but use dense for small active sets)
        B = np.zeros((n_active, n))
        rhs = np.zeros(n_active)
        for idx, k in enumerate(active_idx):
            i_lo = max(0, k - (n-1))
            i_hi = min(n-1, k)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                B[idx, j] = a[i]  # coefficient of d[j] in conv(d,a)[k]
            rhs[idx] = c_max * 0.99 - c[k]
        
        # Objective: 0.5 * d@d
        # Constraints: B@d <= rhs, -d <= a (i.e., d >= -a), sum(d) = 0
        
        # This is too large for SLSQP with n=16384. Let's use a simpler approach.
        # Instead, compute the direction that most reduces the active peaks.
        
        # Direction: d = -pinv(B_active) @ 1 (minimum norm solution)
        # Or simply: d = -B.T @ (B @ B.T)^{-1} @ rhs (if B@B.T is invertible)
        # For simplicity, just use the weighted subgradient direction.
        
        w = np.zeros(len(c))
        w[active_idx] = 1.0 / n_active
        g = _weighted_subgrad(a, c, w, n)
        
        # Step with backtracking
        alpha = 0.1
        for _ in range(50):
            a_try = _project_simplex(a - alpha * g)
            c_try = _full_convolve(a_try, a_try)
            if c_try.max() < c_max - 1e-15:
                a = a_try
                sc = 2.0 * n * c_try.max()
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            alpha *= 0.7
    
    return best_a, best_sc
