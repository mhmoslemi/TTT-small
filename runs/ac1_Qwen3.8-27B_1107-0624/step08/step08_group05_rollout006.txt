def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _eval_exact(a):
    return evaluate_sequence([float(x) for x in a])

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val

def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _newton_active_set(a_init, deadline, max_iter=300, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break
        mu = 1e-8
        active_f = active.astype(np.float64)
        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v
        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if alpha < 1e-12:
                break
            a_new = a
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

def _peak_shaving(a_init, deadline, n_cycles=50):
    """Direct peak shaving: iteratively reduce the maximum convolution entry."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.005
    for cyc in range(n_cycles):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Use wider active set for peak shaving
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        # Compute gradient from active peaks
        a_rev = a[::-1]
        w = np.zeros_like(c)
        # Weight by how close to max (exponential weighting)
        c_active = c[active]
        w_active = np.exp(-(c_max - c_active) / (0.005 * c_max))
        w_active = w_active / w_active.sum()
        w[active] = w_active
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        # Project to simplex direction
        dirn = -(grad_a - np.dot(a, grad_a))
        dirn = dirn - a * np.dot(dirn, a) / a.sum() if a.sum() > 0 else dirn
        # Simpler: projected gradient in z-space
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        gz = a * (grad_a - np.dot(a, grad_a))
        z = z - lr * gz
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.9
            if lr < 1e-6:
                a = best_a
                lr = 0.002
    a_final = _softmax(np.log(np.maximum(a, 1e-30)))
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_spectrum":
        # Random phase flat spectrum, then threshold
        A = np.ones(n // 2 + 1, dtype=complex)
        phases = np.random.default_rng(12345).uniform(0, 2 * np.pi, n // 2 + 1)
        A = A * np.exp(1j * phases)
        if n % 2 == 0:
            A[-1] = 1.0
        # Make it Hermitian for real output
        A_full = np.zeros(n, dtype=complex)
        A_full[:n // 2 + 1] = A
        for k in range(2, min(n, n // 2)):
            A_full[n - k] = np.conj(A[k])
        a = np.fft.irfft(A_full[:n // 2 + 1], n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def _update_best(p, sc=None, label="", t_start=0):
    global _best_seq, _best_score
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = _score(p)
    if np.isfinite(sc) and sc < _best_score:
        _best_seq = p.copy()
        _best_score = float(sc)
        print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
              f"t={time.monotonic() - t_start:.1f}s")

def propose_candidate(seed=42, budget_s=1800.0):

    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "parent", t_start)
    else:
        parent = _make_profile(8192, "uniform")
        _update_best(parent, _score(parent), "init_uniform", t_start)

    # === Phase 1: Quick diverse exploration (40s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 40)
    n_expl = 4096
    for kind in ["uniform", "tapered", "cos2", "wheatstone", "gauss", "flat_spectrum"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        _update_best(a_try, sc_try, f"expl_{kind}", t_start)

    # Quick LSE on best exploration result
    if _best_seq is not None and time.monotonic() < deadline - 600:
        a_up = _best_seq.copy()
        if n_parent > 0 and len(a_up) != n_parent:
            a_up = _resize_sequence(a_up, n_parent)
        a_opt, sc_opt = _lbfgs_lse_adaptive(a_up, deadline, min_remaining=500)
        _update_best(a_opt, sc_opt, "expl_refine", t_start)

    # === Phase 2: Main LSE L-BFGS (500s) ===
    print(f"\n[Phase 2] Main LSE L-BFGS (adaptive beta)")
    if time.monotonic() < deadline - 450:
        a_main, sc_main = _lbfgs_lse_adaptive(_best_seq.copy(), deadline, min_remaining=400)
        _update_best(a_main, sc_main, "lbfgs_lse_main", t_start)

        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=130)
        _update_best(a_ps, sc_ps, "lbfgs_power_sum", t_start)

    # === Phase 4: Newton active-set (150s) ===
    print(f"\n[Phase 4] Newton active-set")
    if time.monotonic() < deadline - 100:
        n_dead = min(deadline - 80, time.monotonic() + 150)
        a_nw, sc_nw = _newton_active_set(_best_seq, n_dead, max_iter=400, tol_frac=0.001)
        _update_best(a_nw, sc_nw, "newton_main", t_start)

    # === Phase 5: Peak shaving (150s) ===
    print(f"\n[Phase 5] Peak shaving")
    if time.monotonic() < deadline - 100:
        ps_dead = min(deadline - 80, time.monotonic() + 150)
        a_ps2, sc_ps2 = _peak_shaving(_best_seq, ps_dead, n_cycles=200)
        _update_best(a_ps2, sc_ps2, "peak_shave", t_start)

    # === Phase 6: Subgradient polish (150s) ===
    print(f"\n[Phase 6] Subgradient polish")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 80, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=10000, lr_init=0.006, tol_frac=0.001)

            nw_dead = min(deadline - 100, time.monotonic() + 50)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=150, tol_frac=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            _update_best(a_cur, _score(a_cur), f"alt_nw_{round_i}", t_start)
            ps_dead2 = min(deadline - 100, time.monotonic() + 40)
            a_ps3, sc_ps3 = _peak_shaving(a_cur, ps_dead2, n_cycles=50)
            if sc_ps3 < _score(a_cur):
                a_cur = a_ps3
            _update_best(a_cur, _score(a_cur), f"alt_ps_{round_i}", t_start)
            sg_dead3 = min(deadline - 100, time.monotonic() + 30)
            a_sg3, sc_sg3 = _subgrad_polish(a_cur, sg_dead3, n_steps=3000, lr_init=0.004, tol_frac=0.001)
            if sc_sg3 < _score(a_cur):
                a_cur = a_sg3
            _update_best(a_cur, _score(a_cur), f"alt_sg_{round_i}", t_start)

    print(f"\n[Phase 8] Multi-n exploration")

        if time.monotonic() < deadline - 30:
            nw_dead = min(deadline - 20, time.monotonic() + 50)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=100, tol_frac=0.002)
            _update_best(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton", t_start)

    # === Phase 9: Diverse restarts (200s) ===
    print(f"\n[Phase 9] Diverse restarts")
    if time.monotonic() < deadline - 60:
        n_restarts = 8
        for i in range(n_restarts):

            elif i == 3:
                p_start = _perturb_log(_best_seq, 0.04, rng)
            elif i == 4:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=5, max_amp=0.4)
            elif i == 5:
                p_start = _perturb_log(_best_seq, 0.01, rng)
            elif i == 6:

            else:
                p_start = _perturb_log(_best_seq, 0.03, rng)

            a_r, sc_r = _lbfgs_lse_adaptive(p_start, deadline, min_remaining=20)
            _update_best(a_r, sc_r, f"restart_{i}", t_start)

            if time.monotonic() < deadline - 15:

                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)

    if _best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-fine LSE
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.00002

        # Final subgradient

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
