   z_{t+1} = z_t - eta * v_{t+1}

   def accelerated_subgradient(a_init, deadline):
       n = len(a_init)
       a = a_init.copy()
       a = a / a.sum()
       z = np.log(a + 1e-30)
       z = z - z.mean()  # center (invariance)
       
       v = np.zeros(n)  # momentum
       mu = 0.9  # momentum coefficient
       eta = 0.1  # initial step size
       beta = 0.01  # initial temperature (relative to max c)
       
       best_z = z.copy()
       best_score = _score(a)
       no_improve = 0
       
       while time.monotonic() < deadline:
           a = _softmax(z)
           c = _full_convolve(a, a)
           c_max = c.max()
           score = 2 * n * c_max
           
           if score < best_score:
               best_score = score
               best_z = z.copy()
               no_improve = 0
           else:
               no_improve += 1
               
           if no_improve > 200:
               # Restart with higher temperature
               beta *= 2
               v = np.zeros(n)
               eta *= 0.5
               no_improve = 0
               if beta > c_max:
                   break
           
           # Compute weights
           w = _softmax(c / beta)  # shape (2n-1,)
           
           # Compute gradient w.r.t. a
           grad_a = _compute_grad(a, w, n)  # shape (n,)
           
           # Compute gradient w.r.t. z
           grad_z = a * (grad_a - np.dot(a, grad_a))
           
           # Nesterov update
           v = mu * v + grad_z
           z = z - eta * v
           z = z - z.mean()  # maintain centering
           
           # Adapt step size
           if no_improve > 50:
               eta *= 0.8
           elif no_improve == 1:
               eta = min(eta * 1.1, 1.0)

"""
Accelerated projected subgradient with Nesterov momentum, adaptive softmax peak weights,
and multi-start strategy for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: Nesterov acceleration, adaptive temperature annealing,
simplex-free softmax parameterization, and diverse initialization.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weighted_grad(a, w, n):
    """Compute g[j] = 2 * sum_k w[k] * a[k-j] for valid k (0<=k-j<n)."""
    N = _next_pow2(2 * (2 * n - 1))
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    cc = np.fft.irfft(A * np.conj(W), N)
    result = np.empty(n)
    result[0] = cc[0]
    if n > 1:
        result[1:] = cc[N - n + 1:N][::-1]
    return 2.0 * result

def _lbfgs_smoothed(a_init, deadline, min_time=20.0, verbose=False):
    """L-BFGS on log-sum-exp smoothed objective in softmax space."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)

    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.005)

    betas = [base_beta * f for f in [1.0, 0.5, 0.25, 0.125, 0.0625,
                                     0.03, 0.015, 0.0075, 0.004, 0.002,
                                     0.001, 0.0005, 0.00025, 0.000125]]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        def obj(z):
            a_local = _softmax(z)
            c_local = _full_convolve(a_local, a_local)
            cm = float(c_local.max())
            shifted = (c_local - cm) / beta
            e = np.exp(np.clip(shifted, -500, 0))
            s_e = e.sum()
            if s_e < 1e-300 or not np.isfinite(s_e):
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s_e))

        def grad(z):
            a_local = _softmax(z)
            c_local = _full_convolve(a_local, a_local)
            cm = float(c_local.max())
            shifted = (c_local - cm) / beta

            else:
                w = e / s_e

            grad_z_local = a_local * (grad_a - np.dot(a_local, grad_a))
            return grad_z_local

        est_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_iter * 15))))

        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',

            a_new = _softmax(result.x)
            sc_new = _score(a_new)

                best_score = sc_new
                best_a = a_new.copy()
                best_z = result.x.copy()
                best_z = best_z - best_z.mean()
                z = best_z
            elif verbose:
                print(f"    L-BFGS beta={beta:.6f}: no improvement (best={best_score:.10f})")
        except Exception as e:

                print(f"    L-BFGS failed: {e}")

    return best_a, best_score
