```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 with non-negative sequences.
Key improvements: asymmetric-focused optimization, direct subgradient polishing,
multi-n exploration, and adaptive L-BFGS with tighter beta schedules.
Exploits the fact that self-convolution (not autocorrelation) allows asymmetric
sequences to beat the symmetric lower bound of 2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H @ v where H is the Hankel matrix with weight vector W."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Conjugate gradient solve for H @ x = rhs."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _smoobj_adapter(z, args):
    beta, n = args
    return _smoobj(z, beta, n)


def _smograd_adapter(z, args):
    beta, n = args
    return _smograd(z, beta, n)


def _newton_polish(a_init, deadline, max_rounds=80):
    """Newton's method with active set for final refinement."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-30)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Tight active set
        tol = 1e-6 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
            tol = 0.0

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=500)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=500)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.01 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _subgrad_direct(a_init, deadline, max_steps=5000):
    """Direct subgradient method on the true (non-smoothed) objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = _softmax(z).copy()
    best_sc = _score(best_a)
    patience = 0
    lr = 0.02
    step_acc = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Active set with soft weighting
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        # Weight by proximity to peak
        w = np.zeros_like(c)
        w[active] = (1.0 - (c_max - c[active]) / max(tol, 1e-30))
        w = np.maximum(w, 0.01)
        w = w / w.sum()

        # Subgradient of max_k c[k] projected to sum-zero
        a_rev = a[::-1]
        wc = _fft_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Polyak step size
        step_acc += 1
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, 0.05)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """Adaptive L-BFGS with decreasing beta (smoothing parameter)."""
    n = len(a_init)
    c_cur = _fft_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.3, max_c * 0.005)
    min_beta = max_c * 5e-10

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.004 * scale
        max_iter = min(15000, max(50, int(remaining / (est_time_per_iter * 30))))

        try:
            args = (beta, n)
            result = minimize(
                _smoobj_adapter, z, args=args, jac=_smograd_adapter,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _fft_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(30, len(c_sorted_new) - 1)]
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4
        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_profile(n, kind="asym_ramp"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t) / (np.exp(2.0) - 1.0)
        return a
    elif kind == "asym_ramp2":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t) / (np.exp(3.0) - 1.0)
        return a
    elif kind == "asym_ramp_neg":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) / (1.0 - np.exp(-2.0))
        return a
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.35) ** 2) + 0.01
        return a / a.sum()
    elif kind == "asym_gauss2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.65) ** 2) + 0.01
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "step":
        a = np.ones(n)
        half = n // 2
        a[:half] = 1.0
        a[half:] = 0.3
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0.001, 1, n)
        a = t ** 1.5
        return a / a.sum()
    elif kind == "power_law_neg":
        t = np.linspace(0.001, 1, n)
        a = t ** (-0.5)
        return a / a.sum()
    elif kind == "bimodal_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + 0.5 * np.exp(-15.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "plateau_ramp":
        t = np.linspace(0, 1, n)
        a = np.minimum(4.0 * t, np.minimum(1.0, 4.0 * (1.0 - t)))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _coordinate_descent(a_init, deadline, max_rounds=200):
    """Coordinate descent: update one coordinate at a time to minimize score."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Pick a batch of coordinates to update
        batch_size = min(n, 200)
        idx = rng.choice(n, size=batch_size, replace=False)
        improved = False

        for i in idx:
            if time.monotonic() > deadline:
                break
            # Try several values for a[i]
            old_val = a[i]
            # The optimal a[i] should reduce the max convolution
            # Try geometric spread around current value
            for factor in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0, 0.8, 1.2]:
                a_try = a.copy()
                a_try[i] = old_val * factor
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                c_max_try = c_try.max()
                if c_max_try < c.max() - 1e-18:
                    a = a_try
                    improved = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Phase 0: Load parent sequence
    print("=" * 70)
    print("Phase 0: Loading parent")
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"  Parent loaded: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception as e:
        print(f"  Could not load parent: {e}")

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "asym_ramp")
        print(f"  Using default profile, n={n_parent}")
        update_best(parent, _score(parent), "default_init")

    # Phase 1: Profile exploration
    print(f"\n{'=' * 70}")
    print(f"Phase 1: Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["asym_ramp", "asym_ramp2", "asym_ramp_neg", "asym_gauss",
                "asym_gauss2", "power_law", "power_law_neg", "bimodal_asym",
                "plateau_ramp", "tapered", "step"]
    p1_deadline = t_start + 60
    best_prof_a = best_seq.copy() if best_seq is not None else parent.copy()
    best_prof_sc = best_score

    for pk in profiles:
        if time.monotonic() > p1_deadline:
            break
        a_p = _make_profile(n_parent, pk)
        sc_p = _score(a_p)
        if sc_p < best_prof_sc:
            best_prof_sc = sc_p
            best_prof_a = a_p
            print(f"  Profile '{pk}': score={sc_p:.10f} (new best profile)")
        else:
            print(f"  Profile '{pk}': score={sc_p:.10f}")

    # Quick L-BFGS on best profile
    if time.monotonic() < p1_deadline - 10:
        a_q = _run_adaptive_lbfgs(best_prof_a, p1_deadline, min_time_remaining=300)
        sc_q = _score(a_q)
        update_best(a_q, sc_q, "profile_lbfgs")
        if sc_q < best_prof_sc:
            best_prof_a = a_q
            best_prof_sc = sc_q

    # Phase 2: Main L-BFGS optimization from best candidate
    print(f"\n{'=' * 70}")
    print(f"Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = t_start + int(0.45 * budget_s)
    start_a = best_seq if best_seq is not None else best_prof_a
    a_main = _run_adaptive_lbfgs(start_a, p2_deadline, min_time_remaining=600)
    update_best(a_main, _score(a_main), "main_lbfgs")

    # Phase 3: Subgradient polishing on true objective
    print(f"\n{'=' * 70}")
    print(f"Phase 3: Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = t_start + int(0.60 * budget_s)
    a_sg = best_seq if best_seq is not None else a_main
    a_sg_opt, sc_sg = _subgrad_direct(a_sg, p3_deadline, max_steps=8000)
    update_best(a_sg_opt, sc_sg, "subgrad")

    # Phase 4: Newton polish
    print(f"\n{'=' * 70}")
    print(f"Phase 4: Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        p4_deadline = t_start + int(0.70 * budget_s)
        a_np, sc_np = _newton_polish(best_seq, p4_deadline, max_rounds=100)

        sc_r = _score(a_r)
        update_best(a_r, sc_r, f"restart_{r}_s{sigma:.3f}")

        if sc_r < best_score and time.monotonic() < p5_deadline - 10:
            a_rp, sc_rp = _newton_polish(a_r, time.monotonic() + 10, max_rounds=30)
            update_best(a_rp, sc_rp, f"restart_{r}_np")

    for n_try in n_candidates:
        if time.monotonic() > p6_deadline:
            break
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt_n = _run_adaptive_lbfgs(a_init_n, min(p6_deadline, time.monotonic() + 120),
                                       min_time_remaining=30.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")

        if sc_opt_n < best_score + 0.005 and time.monotonic() < p6_deadline - 10:
            a_np_n, sc_np_n = _newton_polish(a_opt_n, min(p6_deadline, time.monotonic() + 20),
                                              max_rounds=40)
            update_best(a_np_n, sc_np_n, f"n_{n_try}_np")

    # Phase 7: Final ultra-fine polish
    print(f"\n{'=' * 70}")
    print(f"Phase 7: Final polish (t={time.monotonic() - t_start:.1f}s)")
    final_deadline = deadline - 5
    if best_seq is not None and time.monotonic() < final_deadline:
        a_fin, sc_fin = _newton_polish(best_seq, final_deadline, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # One more subgradient round
        if time.monotonic() < final_deadline - 10:
            a_fg, sc_fg = _subgrad_direct(best_seq, final_deadline - 5, max_steps=5000)
            update_best(a_fg, sc_fg, "final_subgrad")

            if sc_fg < best_score:
                a_fn, sc_fn = _newton_polish(a_fg, final_deadline - 1, max_rounds=50)
                update_best(a_fn, sc_fn, "final_newton2")

    # Final validation and return
    print(f"\n{'=' * 70}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(16384, "asym_ramp")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
