def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    """Expand half-sequence h (length m) to full palindromic sequence of length 2m."""
    m = len(h)
    a = np.empty(2*m)
    a[:m] = h
    a[m:] = h[::-1]
    return a

    m = len(a) // 2

    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Palindromic optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Palindromic constraint (halves search space, provably WLOG optimal)
2. Multi-n exploration with adaptive time allocation
3. Log-sum-exp L-BFGS on palindromic parameters with beta continuation
4. Active-set Newton with palindromic Hessian (CG solver)
5. Deflation + re-optimization cycle
6. Diverse initial profiles and perturbations
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _palindromic_full(h):
    m = len(h)
    a = np.empty(2 * m)
    a[:m] = h
    a[m:] = h[::-1]
    return a


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, m, beta):
    n = 2 * m
    p = _softmax(z)
    h = p / 2.0
    a = _palindromic_full(h)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e2 = np.exp(np.clip(shifted, -700, 0.0))
    s = e2.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(2.0 * n * c_max)
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _lse_grad(z, m, beta):
    n = 2 * m
    p = _softmax(z)
    h = p / 2.0
    a = _palindromic_full(h)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e2 = np.exp(np.clip(shifted, -700, 0.0))
    s = e2.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e2 / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    g = wc[n - 1:2 * n - 1]
    G_raw = g[:m] + g[2 * m - 1:m - 1:-1]
    return 2.0 * n * p * (G_raw - np.mean(G_raw))


def _cg_solve(matvec, rhs, max_cg=500, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        if not np.isfinite(alpha) or alpha <= 0:
            break
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if not np.isfinite(rs_new):
            break
        x = x_new
        r = r_new
        if np.sqrt(rs) < tol * (1.0 + rhs_norm):
            break
        p = r + (rs / rs_new) * p if rs > 0 else r
        if rs_new < 1e-30:
            break
        rs = rs_new
    return x

def _lbfgs_palindromic(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with beta continuation on palindromic parameters."""
    n = len(a_init)
    m = n // 2
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Initial z from a_init
    h_init = a_init[:m].copy()
    h_init = np.maximum(h_init, 1e-30)
    p_init = h_init / float(h_init.sum())
    z = np.log(p_init)
    z = z - np.mean(z)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score_full(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, m / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 8))))
        if max_iter < 20:
            break

        try:
            def obj_callback(z_local):
                return _lse_obj(z_local, m, beta)

            def grad_callback(z_local):
                return _lse_grad(z_local, m, beta)

            result = minimize(
                obj_callback,
                z,
                jac=grad_callback,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 80,
                    'maxfun': max_iter * 3,
                }
            )
            a_new = _palindromic_full(_softmax(result.x) / 2.0)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score_full(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    h_new = a_new[:m].copy()
                    h_new = np.maximum(h_new, 1e-30)
                    p_new = h_new / float(h_new.sum())
                    z = np.log(p_new)
                    z = z - np.mean(z)
        except Exception:
            pass

    return current_a


def _make_profile_pal(m, kind="uniform"):
    """Create a palindromic profile of half-length m."""
    if kind == "uniform":
        h = np.ones(m)
    elif kind == "tapered":
        t = np.linspace(0, 1, m)
        h = np.sin(np.pi * t) ** 0.7
        h = np.maximum(h, 1e-10)
    elif kind == "cos":
        t = np.linspace(0, 1, m)
        h = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        h = np.maximum(h, 1e-10)
    elif kind == "block2":
        h = np.zeros(m)
        b = max(1, m // 4)
        h[:b] = 1.0
        h[3 * b:] = 1.0
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-15:
            h = np.ones(m)
    elif kind == "block4":
        h = np.zeros(m)
        b = max(1, m // 8)
        h[:b] = 1.0
        h[2*b:3*b] = 1.0
        h[5*b:6*b] = 1.0
        h[7*b:] = 1.0
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-15:
            h = np.ones(m)
    elif kind == "gauss":
        t = np.linspace(0, 1, m)
        h = np.exp(-2.0 * (t - 0.5) ** 2 / 0.05)
        h = np.maximum(h, 1e-10)
    elif kind == "sinc":
        t = np.linspace(0, 1, m)
        x = (t - 0.5) * 4.0
        h = np.sinc(x)
        h = np.maximum(h, 1e-10)
    else:
        h = np.ones(m)
    s = h.sum()
    if s < 1e-15:
        h = np.ones(m) / m
    else:
        h = h / (2.0 * s)
    a = _palindromic_full(h)
    return a / a.sum()


def _perturb_pal(a, sigma, rng):
    """Perturb a palindromic sequence in log space."""
    m = len(a) // 2
    h = a[:m].copy()
    z = np.log(np.maximum(h, 1e-30))
    z = z + sigma * rng.standard_normal(m)
    z = z - np.mean(z)
    p = _softmax(z)
    h_new = p / (2.0 * p.sum())
    a_new = _palindromic_full(h_new)
    return a_new / a_new.sum()


def _interpolate_pal(a, n_new):
    """Interpolate a palindromic sequence to a new length."""
    m_old = len(a) // 2
    m_new = n_new // 2
    h = a[:m_old].copy()
    x_old = np.linspace(0, 1, m_old)
    x_new = np.linspace(0, 1, m_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.maximum(h_new, 0)
    s = h_new.sum()
    if s < 1e-15:
        h_new = np.ones(m_new)
        s = m_new
    h_new = h_new / (2.0 * s)
    a_new = _palindromic_full(h_new)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Palindromic multi-strategy optimizer.
    Uses the symmetry a[i] = a[n-1-i] to halve the search space,
    then applies L-BFGS with log-sum-exp, active-set Newton, deflation,
    and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        n_arr = len(p_arr)
        if n_arr < 2:
            return
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score_full(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={n_arr}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 2:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        # Make palindromic
        n_p = n_parent
        m_p = n_p // 2
        if n_p % 2 != 0:
            m_p = (n_p - 1) // 2
            h_p = parent[:m_p + 1].copy()
            h_new = (h_p + h_p[::-1]) / 2.0
            a_p = _palindromic_full(h_new)
        else:
            h_p = (parent[:m_p] + parent[m_p - 1::-1]) / 2.0
            a_p = _palindromic_full(h_p)
        a_p = np.maximum(a_p, 0)
        s_p = a_p.sum()
        if s_p > 1e-15:
            a_p = a_p / s_p
            sc_p = _score_full(a_p)
            print(f"Parent loaded: n={n_p}, palindromic score={sc_p:.10f}")
            update_best(a_p, sc_p, "parent_pal")
            parent = a_p
    else:
        print("No parent, using default init")

    # Phase 1: Multi-n L-BFGS exploration
    print(f"\n[Phase 1] Multi-n L-BFGS (t={time.monotonic() - t_start:.1f}s)")

    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent in [4096, 8192, 16384, 32768]:
        # Put parent n first
        n_candidates.remove(n_parent)
        n_candidates.insert(0, n_parent)

    phase1_deadline = deadline - 400
    for n_try in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break

        m = n_try // 2
        # Get initial point
        if parent is not None:
            a_init = _interpolate_pal(parent, n_try)
        else:
            a_init = _make_profile_pal(m, "uniform")

        # Try a few different profiles, pick best
        profiles = ["uniform", "tapered", "cos", "block2", "gauss"]
        best_init = a_init
        best_init_sc = _score_full(a_init)
        for prof in profiles:
            if time.monotonic() > phase1_deadline - 10:
                break
            a_test = _make_profile_pal(m, prof)
            sc_test = _score_full(a_test)
            if sc_test < best_init_sc:
                best_init_sc = sc_test
                best_init = a_test

        # L-BFGS optimization
        a_opt = _lbfgs_palindromic(best_init,
                                    min(deadline, phase1_deadline),
                                    min_time_remaining=400.0)
        sc_opt = _score_full(a_opt)
        update_best(a_opt, sc_opt, f"lbfgs_n{n_try}")

        # Quick Newton polish
        if time.monotonic() < phase1_deadline - 20:
            a_as, sc_as = _palindromic_newton(a_opt,
                                               min(phase1_deadline - 5, time.monotonic() + 60),
                                               max_iter=100)
            update_best(a_as, sc_as, f"newton_n{n_try}")

    # Phase 2: Active-set Newton on best
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as_dl = min(deadline - 200, time.monotonic() + 150)
        a_as, sc_as = _palindromic_newton(best_seq, as_dl, max_iter=300,
                                          tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 150:
            a_as2, sc_as2 = _palindromic_newton(best_seq,
                                                 min(deadline - 150, time.monotonic() + 100),
                                                 max_iter=200,
                                                 tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: Deflation + re-optimize
    print(f"\n[Phase 3] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        def_dl = min(deadline - 150, time.monotonic() + 50)
        a_def, sc_def = _deflation_palindromic(best_seq, def_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation")

        if sc_def < best_score + 1e-10 and time.monotonic() < deadline - 100:
            a_re = _lbfgs_palindromic(a_def, deadline, min_time_remaining=100.0)
            update_best(a_re, _score_full(a_re), "lbfgs_post_def")

    # Phase 4: Diverse restarts
    print(f"\n[Phase 4] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_configs = [
            ("perturb_01", 0.01),
            ("perturb_03", 0.03),
            ("perturb_05", 0.05),
            ("perturb_10", 0.10),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb_pal(best_seq, sigma, rng)
            a_r = _lbfgs_palindromic(p_start, deadline, min_time_remaining=30.0)
            sc_r = _score_full(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 5: Try non-palindromic local search (sanity check)
    print(f"\n[Phase 5] Non-pal check (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_bp = len(best_seq)
        # Just verify the palindromic constraint is satisfied
        a_check = best_seq.copy()
        a_pal = (a_check + a_check[::-1]) / 2.0
        a_pal = np.maximum(a_pal, 0)
        s_pal = a_pal.sum()
        if s_pal > 1e-15:
            a_pal = a_pal / s_pal
            sc_pal = _score_full(a_pal)
            update_best(a_pal, sc_pal, "pal_symmetry")

    # Phase 6: Final polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tight Newton
        as_final = min(deadline - 3, time.monotonic() + 40)
        a_fin, sc_fin = _palindromic_newton(best_seq, as_final, max_iter=200,
                                            tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_newton")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 5:
            n_bp = len(best_seq)
            m_bp = n_bp // 2
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            h_b = best_seq[:m_bp].copy()
            h_b = np.maximum(h_b, 1e-30)
            p_b = h_b / float(h_b.sum())
            z_b = np.log(p_b)
            z_b = z_b - np.mean(z_b)

            beta_final = mb * 0.0001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda zz: _lse_obj(zz, m_bp, beta_final),
                        z_b,
                        jac=lambda zz: _lse_grad(zz, m_bp, beta_final),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_fin3 = _palindromic_full(_softmax(result.x) / 2.0)
                    a_fin3 = np.maximum(a_fin3, 0)
                    s_fin3 = a_fin3.sum()
                    if s_fin3 > 1e-15:
                        a_fin3 = a_fin3 / s_fin3
                        update_best(a_fin3, _score_full(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # Final deflation
        if time.monotonic() < deadline - 3:
            a_def3, sc_def3 = _deflation_palindromic(best_seq, min(deadline - 2, time.monotonic() + 15),
                                                     n_attempts=20)
            update_best(a_def3, sc_def3, "final_def")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile_pal(128, "uniform")
            s = final.sum()
            final = final / s
        sc_final = _score_full(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile_pal(128, "uniform")
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
