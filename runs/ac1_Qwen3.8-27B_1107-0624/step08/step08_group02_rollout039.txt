def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _autocorr(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc
    
    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.002 * scale
        
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 20))))
        
        try:
            def obj(z_val):
                a_val = _softmax(z_val)
                c_val = _autocorr(a_val)
                c_max_val = float(c_val.max())
                shifted = (c_val - c_max_val) / beta
                e_val = np.exp(np.clip(shifted, -700, 0))
                s_val = e_val.sum()
                if s_val < 1e-300 or not np.isfinite(s_val):
                    return 2.0 * n * c_max_val
                return 2.0 * n * (c_max_val + beta * np.log(s_val))
            
            def grad(z_val):
                a_val = _softmax(z_val)
                c_val = _autocorr(a_val)
                c_max_val = float(c_val.max())
                shifted = (c_val - c_max_val) / beta
                e_val = np.exp(np.clip(shifted, -700, 0))
                s_val = e_val.sum()
                if s_val < 1e-300 or not np.isfinite(s_val):
                    w_val = np.zeros_like(c_val)
                    w_val[int(np.argmax(c_val))] = 1.0
                else:
                    w_val = e_val / s_val
                W_val = _hankel_grad(a_val, w_val, n)
                W_dot_c = float(np.dot(w_val, c_val))
                return 4.0 * n * a_val * (W_val - W_dot_c)
            
            result = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        
        # Adaptive beta reduction
        c_new = _autocorr(current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc
        
        if new_gap < 1e-8 * c_max_new and sc_improved > 1e-14:
            beta *= 0.15
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4
        
        beta = min(beta, c_max_new * 0.001)
    
    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- L-BFGS with adaptive log-sum-exp smoothing for bulk optimization
- Preconditioned KKT residual gradient descent with Nesterov momentum for polishing
- Adaptive active set management with multiple tolerance levels
- Deflation strategy for support reduction
- Multi-n exploration with warm starts
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _autocorr(a):
    return _full_convolve(a, a)


def _hankel_grad(a, W, n):
    """Compute 2 * conv(W, a_rev)[n-1:2n-1]."""
    a_rev = a[::-1]
    out_len = len(a_rev) + len(W) - 1
    N = _next_pow2(out_len)
    F_ar = np.fft.rfft(a_rev, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_ar * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _kkt_polish(a_init, deadline, rng, max_iters=500000):
    """
    Preconditioned KKT gradient descent with Nesterov momentum.
    At optimum: h_j = lambda for all j in support(a).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)  # momentum velocity
    beta_mom = 0.9
    eta = 0.05
    no_improve = 0
    patience = 200
    tol_levels = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000001]
    tol_idx = 0
    best_c_max = float(_autocorr(a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            pass  # throttle prints

        c = _autocorr(a)
        c_max = float(c.max())

        # Adaptive active set
        tol = tol_levels[min(tol_idx, len(tol_levels) - 1)]
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        elif active.sum() < 3:
            # Ensure minimum active set size
            sorted_idx = np.argsort(c)[::-1]
            for idx in sorted_idx[:5]:
                active[idx] = True
        W = np.zeros_like(c)
        W[active] = 1.0 / active.sum()

        # Compute KKT gradient
        h = _hankel_grad(a, W, n)

        # Preconditioner: diagonal of H is 2*W[2j]
        diag = 2.0 * W[2 * np.arange(n)]
        diag = np.maximum(diag, 1e-15)

        # KKT residual on support
        support = a > 1e-12
        if support.sum() == 0:
            break
        lam = float(np.mean(h[support]))
        residual = (h - lam) / diag  # preconditioned

        # Nesterov step
        v = beta_mom * v - eta * residual
        a_new = a + v

        # Project
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
            v *= 0.5
        else:
            a_new = a_new / s_new

        c_new = _autocorr(a_new)
        c_max_new = float(c_new.max())

        if c_max_new < c_max - 1e-18:
            a = a_new
            no_improve = 0
            if no_improve == 0 and it % 1000 == 999:

            # Slowly increase eta
            eta = min(eta * 1.002, 0.3)
            # Slowly increase momentum
            beta_mom = min(beta_mom + 0.0001, 0.95)
        else:
            no_improve += 1
            eta *= 0.8
            beta_mom = max(beta_mom - 0.01, 0.7)
            if no_improve > patience:
                # Try to tighten active set
                if tol_idx < len(tol_levels) - 1:
                    tol_idx += 1
                elif tol_idx >= 5:
                    # Perturbation to escape
                    a = a + 0.005 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    v *= 0.1
                no_improve = 0
                patience = int(patience * 1.3)

        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    # Final check
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _deflation_polish(a_init, deadline, rng, max_rounds=20):
    """Try removing small elements from support and re-optimizing."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Find smallest elements in the support
        sorted_idx = np.argsort(a)
        # Try removing the smallest element
        if a[sorted_idx[0]] < 1e-8 * a.max():
            a_try = a.copy()
            a_try[sorted_idx[0]] = 0.0
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_a = a_try.copy()
                best_sc = sc_try
                a = a_try
            else:
                break
        else:
            break

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _autocorr(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_a and current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 20))))

        try:
            def obj(z_val):
                a_val = _softmax(z_val)
                c_val = _autocorr(a_val)
                c_max_val = float(c_val.max())
                shifted = (c_val - c_max_val) / beta
                e_val = np.exp(np.clip(shifted, -700, 0))
                s_val = e_val.sum()
                if s_val < 1e-300 or not np.isfinite(s_val):
                    return 2.0 * n * c_max_val
                return 2.0 * n * (c_max_val + beta * np.log(s_val))

            def grad(z_val):
                a_val = _softmax(z_val)
                c_val = _autocorr(a_val)
                c_max_val = float(c_val.max())
                shifted = (c_val - c_max_val) / beta
                e_val = np.exp(np.clip(shifted, -700, 0))
                s_val = e_val.sum()
                if s_val < 1e-300 or not np.isfinite(s_val):
                    w_val = np.zeros_like(c_val)
                    w_val[int(np.argmax(c_val))] = 1.0
                else:
                    w_val = e_val / s_val
                W_val = _hankel_grad(a_val, w_val, n)
                W_dot_c = float(np.dot(w_val, c_val))
                return 4.0 * n * a_val * (W_val - W_dot_c)

            result = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _autocorr(current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-8 * c_max_new and sc_improved > 1e-14:
            beta *= 0.12
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.35

        beta = min(beta, c_max_new * 0.0005)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 1.5 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
