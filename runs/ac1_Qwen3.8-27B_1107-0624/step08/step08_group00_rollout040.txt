Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines variance minimization, L-BFGS with log-sum-exp smoothing,
frequency-domain initialization, and direct peak manipulation.
Key innovations: (1) variance-based pre-conditioning to flatten convolution,
(2) frequency-domain alternating projections for initialization,
(3) adaptive multi-scale temperature schedule.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _variance_obj_and_grad(a):
    """Compute sum(c^2) and its gradient for c = conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    obj = float(np.sum(c * c))
    # grad[m] = 4 * sum_j a[j] * c[m+j] for m = 0,...,n-1
    # = 4 * (a_rev * c)[n-1 : 2n-1]
    a_rev = a[::-1]
    g_full = _full_convolve(a_rev, c)
    grad = 4.0 * g_full[n-1:2*n-1]
    return obj, grad

    """Minimize sum(conv(a,a)^2) via gradient descent to flatten the convolution."""
    n = len(a)
    a = a.copy().astype(float)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

        # Project gradient to maintain sum(a) = 1
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)

                a = a_try
                improved = True
                break

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Variance minimization pre-conditioning to flatten convolution before max-optimization
2. Frequency-domain alternating projections for diverse initialization
3. Improved multi-scale log-sum-exp temperature schedule with more steps at low beta
4. Direct top-k peak equalization using subgradient steps
5. Aggressive multi-n exploration with resampling
6. Exponential-tilt and asymmetric initializations to escape symmetric local minima
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_lse(a, beta, n):
    """Compute gradient of the log-sum-exp smoothed objective."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lse_objective_val(a, beta, n):
    """Value of the smoothed objective."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        return _lse_objective_val(a, self.beta, self.n)

    def grad(self, z):
        a = _softmax(z)
        base_grad = _compute_grad_lse(a, self.beta, self.n)
        return base_grad * a  # chain rule through softmax: grad_z = a * (grad_a - sum(a*grad_a))
        # Actually for softmax parameterization with constraint sum(a)=1,
        # the gradient with respect to z is: a * (grad_a - sum_j a[j]*grad_a[j])
        # But since we'll project anyway, just return a * grad_a
        # The mean centering is handled by the optimizer's constraint handling


class _LseProblem2:
    """Better softmax gradient: grad_z = a * (grad_a - dot(a, grad_a))"""
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        return _lse_objective_val(a, self.beta, self.n)

    def grad(self, z):
        a = _softmax(z)
        base_grad = _compute_grad_lse(a, self.beta, self.n)
        return a * (base_grad - float(np.dot(a, base_grad)))


def _run_lbfgs_multiscale(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with improved multi-scale temperature schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # More aggressive schedule: more steps at low temperature
    ratios = [1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.008,
              0.004, 0.002, 0.001, 0.0006, 0.0004, 0.0003, 0.0002,
              0.00015, 0.0001, 0.00007, 0.00005, 0.00003, 0.00002]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 4096.0)
        # More iterations at lower beta
        max_iter = min(3000, max(50, int(remaining / (est_time * 15))))
        if i > 10:
            max_iter = min(2000, max(100, int(remaining / (est_time * 10))))

        prob = _LseProblem2(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 40,
                    'maxls': 30,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 5 == 0:
                    print(f"    L-BFGS beta_idx={i}, beta={beta:.2e}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _variance_descent(a_init, deadline, min_time_remaining=30.0):
    """Minimize sum(conv(a,a)^2) to flatten the convolution."""
    n = len(a_init)
    a = a_init.copy().astype(float)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 10.0 / n
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        obj = float(np.sum(c * c))
        # Gradient: grad[m] = 4 * sum_j a[j] * c[m+j]
        a_rev = a[::-1]
        g_full = _full_convolve(a_rev, c)
        grad = 4.0 * g_full[n - 1:2 * n - 1]
        # Project to maintain sum=1
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            obj_try = float(np.sum(c_try * c_try))
            if obj_try < obj - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
            no_improve = 0
            lr = min(50.0 / n, lr * 1.005)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve >= max_no_improve:
                break

    return best_a, best_sc


def _topk_peak_equalization(a_init, deadline, min_time_remaining=10.0):
    """Directly reduce the top-k peaks using subgradient steps."""
    n = len(a_init)
    a = a_init.copy().astype(float)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    k = max(3, n // 2000)  # number of top peaks to target
    lr = 2.0 / n
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Find top-k peaks
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        # Weight by how much above minimum
        c_min = float(c.min())
        weights = np.maximum(top_vals - c_min, 1e-15)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        # For peak at position k, subgrad[m] = 2*a[k-m] (if valid)
        grad = np.zeros(n)
        for w_k, k_pos in zip(weights, top_idx):
            i_lo = max(0, k_pos - (n - 1))
            i_hi = min(n - 1, k_pos)
            # grad[m] += 2 * w_k * a[k_pos - m] for m in [0, n-1]
            # a[k_pos - m] is valid when 0 <= k_pos - m < n, i.e., k_pos - (n-1) <= m <= k_pos
            m_lo = max(0, k_pos - (n - 1))
            m_hi = min(n - 1, k_pos)
            # grad[m] = 2 * w_k * a[k_pos - m] for m in [m_lo, m_hi]
            # a[k_pos - m] for m from m_lo to m_hi: indices from k_pos-m_hi to k_pos-m_lo
            a_slice = a[k_pos - m_hi : k_pos - m_lo + 1][::-1]
            grad[m_lo:m_hi + 1] += 2.0 * w_k * a_slice

        # Project
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.9
            k = max(2, int(k * 0.9))
            if no_improve > 500:
                break
        else:
            lr = min(5.0 / n, lr * 1.01)

    return best_a, best_sc


def _freq_init(n, n_iters=300, rng=None):
    """Frequency-domain alternating projection initialization."""
    if rng is None:
        rng = np.random.default_rng(123)
    N_fft = _next_pow2(2 * n - 1)

    # Initialize: random phase, unit magnitude
    A = np.exp(1j * rng.uniform(0, 2 * np.pi, N_fft // 2 + 1))

    a_full = np.ones(n) / n
    for it in range(n_iters):
        a_full = np.fft.irfft(A, N_fft)
        a_full = np.maximum(a_full[:n], 0.0)
        s = a_full.sum()
        if s < 1e-10:
            a_full = np.ones(n) / n
            s = 1.0
        a_full = a_full / s

        A_new = np.fft.rfft(a_full, N_fft)
        mags = np.abs(A_new)
        mean_mag = np.mean(mags)
        # Relax toward equal magnitude
        relax = 0.5 + 0.5 * min(1.0, it / 50.0)
        A = A_new / np.maximum(mags, 1e-12) * ((1 - relax) * mags + relax * mean_mag)

    a = np.maximum(a_full[:n], 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _exp_tilt_init(n, alpha_val=1.0):
    """Exponential tilt initialization (asymmetric)."""
    t = np.arange(n, dtype=float)
    a = np.exp(alpha_val * t / n)
    a = a / a.sum()
    return a


def _poly_tilt_init(n, power=2.0):
    """Polynomial tilt initialization (asymmetric)."""
    t = np.arange(n, dtype=float) / n
    a = t ** power + 0.01
    a = a / a.sum()
    return a


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_reoptimize(a, sigma, deadline, min_time_remaining=30.0, rng=None):
    """Perturb and re-optimize to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    a_pert = _softmax(z)
    a_opt = _run_lbfgs_multiscale(a_pert, deadline, min_time_remaining=min_time_remaining)
    return a_opt


def propose_candidate(seed=42, budget_s=1800):
    """Multi-strategy optimizer combining variance minimization, L-BFGS,
    peak equalization, and multi-n exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Variance minimization on parent ===
    print(f"\n[P1] Variance minimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        v_deadline = min(deadline - 600, time.monotonic() + 200)
        a_var, sc_var = _variance_descent(best_seq, v_deadline, min_time_remaining=100.0)
        if sc_var < best_score:
            best_seq = a_var.copy()
            best_score = sc_var
            print(f"  NEW BEST: {best_score:.10f} [variance] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Variance: no score improvement (score={sc_var:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS multi-scale on best ===
    print(f"\n[P2] L-BFGS multi-scale (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        lb_deadline = min(deadline - 400, time.monotonic() + 300)
        a_lb = _run_lbfgs_multiscale(best_seq, lb_deadline, min_time_remaining=60.0)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  L-BFGS: no improvement (score={sc_lb:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Top-k peak equalization ===
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pk_deadline = min(deadline - 300, time.monotonic() + 150)
        a_pk, sc_pk = _topk_peak_equalization(best_seq, pk_deadline, min_time_remaining=50.0)
        if sc_pk < best_score:
            best_seq = a_pk.copy()
            best_score = sc_pk
            print(f"  NEW BEST: {best_score:.10f} [peak_eq] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS again after peak equalization ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        lb2_deadline = min(deadline - 250, time.monotonic() + 200)
        a_lb2 = _run_lbfgs_multiscale(best_seq, lb2_deadline, min_time_remaining=40.0)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < best_score:
            best_seq = a_lb2.copy()
            best_score = sc_lb2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 8192, 32768, 12288]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    # Remove duplicates
    seen = set()
    n_candidates = [x for x in n_candidates if not (x in seen or seen.add(x))]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        # Check feasibility (time cost)
        est_conv_time = 0.02 * (n_try / 16384.0)
        if est_conv_time > 1.0:
            print(f"  n={n_try}: skipping (too slow)")
            continue

        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        if time.monotonic() < deadline - 80:
            a_opt = _run_lbfgs_multiscale(a_init, deadline, min_time_remaining=70.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

            if time.monotonic() < deadline - 40:
                a_pk2, sc_pk2 = _topk_peak_equalization(a_opt,
                    min(deadline - 35, time.monotonic() + 30), min_time_remaining=5.0)
                if sc_pk2 < best_score:
                    best_seq = a_pk2.copy()
                    best_score = sc_pk2
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_pk] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: perturbation-based restart ===
    print(f"\n[P6] Perturbation restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        for sigma in [0.01, 0.005, 0.002]:
            if time.monotonic() > deadline - 50:
                break
            a_pert = _perturb_reoptimize(best_seq, sigma, deadline,
                                         min_time_remaining=40.0, rng=rng)
            sc_pert = _score(a_pert)
            if sc_pert < best_score:
                best_seq = a_pert.copy()
                best_score = sc_pert
                print(f"  NEW BEST: {best_score:.10f} [perturb sigma={sigma}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Alternative initializations ===
    print(f"\n[P7] Alternative inits (t={time.monotonic()-t_start:.1f}s)")
    n_main = len(best_seq)
    if time.monotonic() < deadline - 60:
        # Try exponential tilt
        a_tilt = _exp_tilt_init(n_main, alpha_val=0.8)
        sc_tilt = _score(a_tilt)
        print(f"  Exp tilt init: score={sc_tilt:.10f}")
        if sc_tilt < best_score + 0.3:  # only refine if competitive
            a_tilt_opt = _run_lbfgs_multiscale(a_tilt, deadline, min_time_remaining=40.0)
            sc_tilt_opt = _score(a_tilt_opt)
            if sc_tilt_opt < best_score:
                best_seq = a_tilt_opt.copy()
                best_score = sc_tilt_opt
                print(f"  NEW BEST: {best_score:.10f} [tilt] t={time.monotonic()-t_start:.1f}s")

    if time.monotonic() < deadline - 40:
        # Try frequency-domain init
        a_freq = _freq_init(n_main, n_iters=200, rng=rng)
        sc_freq = _score(a_freq)
        print(f"  Freq init: score={sc_freq:.10f}")
        if sc_freq < best_score + 0.3:
            a_freq_opt = _run_lbfgs_multiscale(a_freq, deadline, min_time_remaining=30.0)
            sc_freq_opt = _score(a_freq_opt)
            if sc_freq_opt < best_score:
                best_seq = a_freq_opt.copy()
                best_score = sc_freq_opt
                print(f"  NEW BEST: {best_score:.10f} [freq] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fin, sc_fin = _topk_peak_equalization(best_seq, deadline - 3,
                                                min_time_remaining=3.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_pk] t={time.monotonic()-t_start:.1f}s")

    if remaining > 15:
        # Tiny L-BFGS with very low beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.00003
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _LseProblem2(beta_fin, n)
        max_iter = min(3000, max(50, int((deadline - time.monotonic() - 3) / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 30})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
