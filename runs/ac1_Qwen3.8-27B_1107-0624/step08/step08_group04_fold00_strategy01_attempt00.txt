            for i in range(3):
                deadline_remaining -= 60
                if deadline_remaining <= 0: break
                candidate_i = multi_n_exploration(len(refined), deadline)
                update_best(candidate_i)
            
            doubling = double_sequence(refined)
            halving = halve_sequence(refined)
            candidate_2d = 2 * len(refined)
            candidates = [doubling, halving, candidate_2d]
            
            for cand in candidates:
                cand_score = evaluate_sequence(cand.tolist())
                if np.isfinite(cand_score) and cand_score < best_score:
                    best_score = cand_score
                    best_sequence = cand
                    
                cand_score_val = cand_score if cand_score < best_score else best_score
                candidate_final = final_adjustment(best_sequence.tolist(), deadline_remaining + start_time)
            
            crossover_result = crossover(refined, candidate_final[0] if best_sequence else None)
            if np.isfinite(crossover_result) and crossover_result < best_score:
                best_sequence = crossover_result
            
    if best_sequence is None: return initializer()
    return best_sequence

def crossover(a, b):
    n = len(a)
    crossover_point = np.random.randint(1, n-1)
    offspring = np.concatenate((a[:crossover_point], b[crossover_point:]))
    offspring = aggressive_optimization(offspring, False, 30, 0.6)
    return normalize(offspring, n)

def normalize(seq, n):
    total = float(np.sum(seq))
    return np.array(seq) / total if total >= 0.01 else np.ones(n) / n

def initial_lower_bound(n_val):
    n_param = min(1024, n_val)
    x_val = 5e10 ** n_val
    x_val = x_val / (1.5 ** n_param)
    x_val = np.clip(x_val, 0, 5000)

class SMOptimizer:
    def __init__(self, beta, n):
        self.beta = np.clip(beta, 0.0001, 10.0)  # Value is a float, but beta is passed as parameter
        self.n = len(n)
        self.gamma = self.beta * 0.3
        
    def fit(self, candidate):
        delta_x = 10e-5
        autocorr_vals = []
        for i in range(3):
            candidate_new = candidate + delta_x * np.random.random(self.n)
            autocorr_val = self.get_autocorrelation(candidate_new)
            autocorr_vals.append(autocorr_val)
            
        return np.mean(autocorr_vals)
        
    def get_autocorrelation(self, x):
        conv = np.convolve(x, x, mode='full')
        return float(conv.max())
        
    def objective(self, sequence, deadline):
        score = evaluate_sequence(sequence.tolist())
        self.larm = np.convolve(sequence, np.ones(15), mode='valid')
        if score >= 1e10: return np.inf
        return score
        
    def validate(self, sequence):
        valid = np.isfinite(evaluate_sequence(sequence.tolist()))
        if not valid: self.run_bound_adjustment(sequence)
        return np.sum((sequence - np.mean(sequence)) ** 2)

def evaluate_sequence(seq_list):
    n = len(seq_list)
    valid_flag = True
    try:
        n_list = [float(x) for x in seq_list]
    except Exception:
        return np.inf
        
    n_arr = np.array(n_list, dtype='float64')
    if any(n_arr < 0): return np.inf
    sum_val = float(np.sum(n_arr))
    if sum_val < 0.01: return np.inf
    convolve_result = np.convolve(n_arr, n_arr, mode='full')
    max_conv = convolve_result.max()
    return auto_conv_obj(n, max_conv, convolve_result) 

def check_convergence(original_score, candidate_score, delta_len):
    delta = (original_score - candidate_score) / max(original_score, candidate_score)
    if delta < 1e-5: return True
    if delta_len <= 3: return True
    if delta < 50 * (delta_len ** (-1)): return True
    return False

def add_multi_n_exploration(original_score, new_score, delta_n):
    improvement_factor = (new_score - original_score) / original_score
    if improvement_factor <= 15e-3 * pow(0.9, delta_n): return True
    return False

def adjust_dimension(candidate, target_n):
    n_old = len(candidate)
    n_ratio = target_n / n_old
    new_candidate = []
    
    while len(new_candidate) < target_n:
        if n_ratio >= 1.0:
            # Append elements until we reach the target length
            append_count = max(1, int((target_n - len(new_candidate)) * n_ratio))
        else:
            # Interpolate until we reach the target length
            append_count = max(1, int((target_n - len(new_candidate)) * n_ratio))
            
        for i in range(append_count):
            if i < target_n:
                interpolation = float(candidate[i] + candidate[i] * np.random.rand())
                new_candidate.append(interpolation)
                
    return normalize(new_candidate, target_n)
