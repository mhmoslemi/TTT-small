"""
Multi-strategy optimizer for minimizing the step function linear autocorrelation peak.
Key improvements over previous: asymmetric profile exploitation, coordinate descent,
better multi-n exploration, adaptive LSE with more aggressive beta scheduling,
and improved time management.

import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


# ============================================================
# LSE objective and gradient
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

# Power-sum objective and gradient
# ============================================================

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):

    z = np.log(np.maximum(a_init, 1e-30))

        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:

                    z = z - z.mean()
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)

def _prox_subgrad(a_init, deadline, n_steps=3000, lr_init=0.005, tol_frac=0.001):
    """Proximal subgradient with Nesterov acceleration."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()  # Nesterov momentum variable
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        z_prev = z.copy()
        a = _softmax(y)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr_init / (1.0 + 0.01 * i)
        y = z - step * grad_z
        y = np.clip(y, -80.0, 80.0)
        y = y - y.mean()

        # Nesterov update
        t_k = 2.0 / (np.sqrt(1.0 + 4.0 * i * i) + 1.0)
        z = t_k * y + (1.0 - t_k) * z_prev
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _coordinate_descent(a_init, deadline, n_sweeps=50, restart_prob=0.1):
    """Greedy coordinate descent on the log-domain parameters."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order for stochastic effect
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() > deadline:
                break
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            S = float(a.sum())

            # Try small perturbations of z[idx]
            base = z[idx]
            for delta in [0.01, -0.01, 0.03, -0.03, 0.1, -0.1, 0.3, -0.3]:
                z_old = z[idx]
                z[idx] = base + delta
                a_try = _softmax(z)
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(2.0 * n * c_try.max() / (a_try.sum() ** 2))
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    improved = True
                    break
                z[idx] = z_old

        if not improved and rng.random() < restart_prob:
            # Small random reset
            z = z + 0.01 * rng.standard_normal(n)
            z = z - z.mean()

    return best_a, best_sc

def _lbfgs_lse(a_init, deadline, min_remaining=30.0, n_stages=None):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    if n_stages is None:
        n_stages = 16

    betas = base_beta * np.array([1.0, 0.7, 0.45, 0.3, 0.2, 0.13,
                                   0.085, 0.055, 0.035, 0.022,
                                   0.014, 0.009, 0.0055, 0.003,
                                   0.0015, 0.0008, 0.0004, 0.0002])
    betas = betas[:n_stages]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lbfgs_power_sum(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [3, 5, 10, 20, 40, 80, 160, 320, 640]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_asym_shift(a, rng, max_shift_frac=0.1):
    """Shift the mass distribution to break symmetry."""
    n = len(a)
    shift = int(rng.integers(-n // 10, n // 10 + 1))
    a_shifted = np.roll(a, shift)
    # Re-normalize and clip
    a_shifted = np.maximum(a_shifted, 0)
    s = a_shifted.sum()
    if s < 1e-10:
        return a.copy()
    return a_shifted / s


def _perturb_mirror_mix(a, rng, alpha_range=(0.3, 0.7)):
    """Mix with its reverse to adjust asymmetry."""
    alpha = rng.uniform(*alpha_range)
    a_rev = a[::-1].copy()
    a_mix = alpha * a + (1 - alpha) * a_rev
    s = a_mix.sum()
    if s < 1e-10:
        return a.copy()
    return a_mix / s


def _perturb_random_projection(a, rng, n_components=20):
    """Project onto random Fourier subspace and back."""
    n = len(a)
    A = np.fft.rfft(a).astype(complex)
    mask = np.zeros(len(A), dtype=bool)
    indices = rng.choice(len(A), size=min(n_components, len(A)), replace=False)
    mask[indices] = True
    A[~mask] *= 0.1  # attenuate unselected modes
    a_new = np.fft.irfft(A, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_rise":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bilinear":
        t = np.linspace(0, 1, n)
        a = (1 - t) * 0.5 + t * 1.5
        a = 0.1 + a
        return a / a.sum()
    elif kind == "cubic_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t - 0.3 * t**2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_ramp":
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.5, 2.0 * t, 2.0 * (1 - t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Optimizer that minimizes 2*n*max(convolve(a,a)) / (sum(a))^2.
    Uses multi-phase strategy: asymmetric exploration, L-BFGS with LSE/power-sum,
    Newton active-set, subgradient with Nesterov, coordinate descent,
    diverse restarts, and multi-n refinement.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "ramp")
        upd(parent, _score(parent), "init_ramp")

    # === Phase 1: Asymmetric profile exploration (80s) ===
    print(f"\n[Phase 1] Asymmetric profile exploration")
    expl_deadline = min(deadline, t_start + 80)
    n_expl = max(4096, n_parent // 4) if n_parent > 8192 else 4096
    profiles = ["ramp", "exp_decay", "exp_rise", "sawtooth", "asym_gauss",
                "bilinear", "cubic_asym", "wheatstone", "double_ramp"]
    for kind in profiles:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # Also try the reversed parent
    if parent is not None:
        a_rev = parent[::-1].copy()
        upd(a_rev, _score(a_rev), "parent_reversed")

    # Quick LSE on best found so far
    if _best_seq is not None and time.monotonic() < deadline - 600:
        a_q, sc_q = _lbfgs_lse(_best_seq.copy(), min(deadline - 500, t_start + 150),
                                min_remaining=50, n_stages=8)
        upd(a_q, sc_q, "quick_lse")

    # === Phase 2: Main L-BFGS with LSE (500s) ===
    print(f"\n[Phase 2] Main L-BFGS LSE")
    if time.monotonic() < deadline - 450:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline,
                                      min_remaining=400, n_stages=18)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 3: Newton active-set (200s) ===
    print(f"\n[Phase 3] Newton active-set")
    if time.monotonic() < deadline - 350:
        nw_dead = min(deadline - 300, time.monotonic() + 200)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 4: Power-sum L-BFGS (200s) ===
    print(f"\n[Phase 4] Power-sum L-BFGS")
    if time.monotonic() < deadline - 300:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=280)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 5: Nesterov subgradient (150s) ===
    print(f"\n[Phase 5] Nesterov subgradient")
    if time.monotonic() < deadline - 250:
        sg_dead = min(deadline - 230, time.monotonic() + 150)
        a_ng, sc_ng = _prox_subgrad(_best_seq, sg_dead, n_steps=4000,
                                     lr_init=0.006, tol_frac=0.001)
        upd(a_ng, sc_ng, "nesterov_sg")

    # === Phase 6: Alternating Newton + Subgradient (200s) ===
    print(f"\n[Phase 6] Newton + SG alternating")
    if time.monotonic() < deadline - 200:
        a_cur = _best_seq.copy()
        for round_i in range(4):
            if time.monotonic() > deadline - 180:
                break
            nw_dead = min(deadline - 180, time.monotonic() + 35)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=80, tol_frac=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")

            sg_dead2 = min(deadline - 180, time.monotonic() + 25)
            a_sg2, sc_sg2 = _subgrad_polish(a_cur, sg_dead2, n_steps=2000,
                                             lr_init=0.004, tol_frac=0.001)
            if sc_sg2 < _score(a_cur):
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 7: Coordinate descent (150s) ===
    print(f"\n[Phase 7] Coordinate descent")
    if time.monotonic() < deadline - 150:
        cd_dead = min(deadline - 130, time.monotonic() + 150)
        a_cd, sc_cd = _coordinate_descent(_best_seq, cd_dead, n_sweeps=30)
        upd(a_cd, sc_cd, "coord_desc")

    # === Phase 8: Diverse restarts (300s) ===
    print(f"\n[Phase 8] Diverse restarts")
    if time.monotonic() < deadline - 80:
        n_restarts = 8
        perturb_types = [
            ("log_003", lambda: _perturb_log(_best_seq, 0.03, rng)),
            ("log_005", lambda: _perturb_log(_best_seq, 0.05, rng)),
            ("fourier", lambda: _perturb_fourier_phase(_best_seq, rng, 8, 0.3)),
            ("amp", lambda: _perturb_amplitude(_best_seq, rng, 0.08)),
            ("block", lambda: _perturb_block_swap(_best_seq, rng, 4)),
            ("mirror", lambda: _perturb_mirror_mix(_best_seq, rng)),
            ("proj", lambda: _perturb_random_projection(_best_seq, rng, 30)),
            ("shift", lambda: _perturb_asym_shift(_best_seq, rng)),
        ]
        for i, (name, gen_func) in enumerate(perturb_types):
            if time.monotonic() > deadline - 60:
                break
            p_start = gen_func()
            sc_start = _score(p_start)
            upd(p_start, sc_start, f"restart_{name}_init")

            # Quick LSE
            a_r, sc_r = _lbfgs_lse(p_start, min(deadline - 50, time.monotonic() + 30),
                                    min_remaining=20, n_stages=6)
            upd(a_r, sc_r, f"restart_{name}")

            # Newton polish if promising
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 40:
                nw_dead = min(deadline - 35, time.monotonic() + 20)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{name}_newton")

    # === Phase 9: Multi-n exploration (200s) ===
    print(f"\n[Phase 9] Multi-n exploration")
    n_base = len(_best_seq)
    n_candidates = sorted(set([
        max(1024, n_base // 2),
        n_base,
        n_base * 2,
        max(1024, int(n_base * 0.75)),
        int(n_base * 1.25),
    ]))
    # Filter to reasonable range
    n_candidates = [x for x in n_candidates if 2048 <= x <= 65536]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        sc_init = _score(a_init)
        upd(a_init, sc_init, f"n_{n_try}_interp")

        if time.monotonic() < deadline - 50:
            a_opt_n, sc_opt_n = _lbfgs_lse(a_init, min(deadline - 45, time.monotonic() + 40),
                                            min_remaining=30, n_stages=10)
            upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

            if sc_opt_n < _best_score * 1.003 and time.monotonic() < deadline - 35:
                nw_dead = min(deadline - 30, time.monotonic() + 20)
                a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead,
                                                          max_iter=60, tol_frac=0.002)
                upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 10: Final aggressive polish (120s) ===
    print(f"\n[Phase 10] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())

        # Ultra-low beta LSE
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 100}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 3,
                                                  max_iter=80, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final Nesterov
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _prox_subgrad(_best_seq, deadline - 1,
                                             n_steps=2000, lr_init=0.003,
                                             tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_nesterov")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "ramp")
            print(f"FALLBACK ramp n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "ramp")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
