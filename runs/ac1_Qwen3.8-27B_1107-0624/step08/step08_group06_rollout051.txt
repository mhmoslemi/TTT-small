def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):

    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x

        pAp = np.dot(p, Ap)

        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break

        rs = rs_new
    return x
