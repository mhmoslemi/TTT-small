    """CG with preconditioner."""
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    z = precond_func(r)
    p = z.copy()
    rz = float(np.dot(r, z))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rz / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        z_new = precond_func(r)
        rz_new = float(np.dot(r, z_new))
        if abs(rz) < 1e-30:
            beta = 0.0
        else:
            beta = rz_new / rz
        p = z_new + beta * p
        z = z_new
        rz = rz_new
    return x

def _simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Binary search for the threshold
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, len(v)+1) > (cv - target_sum))[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

"""
Multi-stage optimizer for minimizing step function autocorrelation peak.
Uses improved Newton active-set with adaptive regularization, proximal gradient
methods, flat-spectrum initialization, and aggressive multi-n exploration.
Key improvements: adaptive Hessian regularization, better time management,
diverse restarts with structure-aware perturbations.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x, 0
    for it in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if pH <= 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x, it


def _newton_active_set(a_init, deadline, max_iter=500, tol_frac=0.002, mu_init=1e-6):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    mu = mu_init
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-15)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_f = active.astype(np.float64)

        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx, cg_iters = _cg_solve(_hv, -grad_t, tol=1e-11, max_iter=300)
        dx = dx - np.mean(dx)

        # If CG didn't converge well, increase mu
        if cg_iters >= 299:
            mu *= 2.0
            if mu > 1.0:
                mu = mu_init * 0.1
                continue

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            stall_count += 1
            if stall_count > 15:
                break
            mu *= 0.5
            continue

        stall_count = 0
        mu = max(mu * 0.9, mu_init * 0.01)
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _lbfgs_lse(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.astype(np.float64).copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    betas = base_beta * betas

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

def _proximal_gradient(a_init, deadline, max_iter=50000):
    """Proximal gradient with simplex projection for the normalized problem."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_c_max = float(_full_convolve(a, a).max())

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-15)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Subgradient of sum of active c_k
        active_f = active.astype(np.float64)
        subgrad = _hess_matvec(a, active_f, n)

        # Polyak-style step size
        step = 2.0 / ((t + 2) * max(active.sum(), 1.0))

        a_try = a - step * subgrad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try

        c_try_max = float(_full_convolve(a_try, a_try).max())
        if c_try_max < best_c_max:
            a = a_try
            best_c_max = c_try_max
            best_a = a_try.copy()
            best_sc = _score(a)

    return best_a, best_sc

        a_opt, sc_opt = _lbfgs_lse(a_init, min(deadline, time.monotonic() + 60), min_remaining=0)
        quick_results[n_try] = (a_opt, sc_opt)
        update_candidate(a_opt, sc_opt, f"quick_n{n_try}")

    # Pick the best n from quick exploration
    if quick_results:
        best_n = min(quick_results, key=lambda k: quick_results[k][1])
        print(f"  Best n from exploration: {best_n} with score {quick_results[best_n][1]:.10f}", flush=True)
    else:
        best_n = 16384

    # === Phase 2: Main optimization at best n (600s) ===
    print(f"\n[Phase 2] Main optimization at n={best_n}", flush=True)
    main_deadline_end = t_start + 700
    if best_n in quick_results:
        a_main = quick_results[best_n][0].copy()
    else:
        a_main = _resize_sequence(best_seq, best_n)

    a_opt, sc_opt = _optimize_at_n(a_main, min(deadline, main_deadline_end), min_remaining=300)
    update_candidate(a_opt, sc_opt, f"main_n{best_n}")

    # === Phase 3: Optimize at other n values (300s) ===
    print(f"\n[Phase 3] Optimize at other n values", flush=True)
    for n_try in sorted(quick_results.keys()):
        if n_try == best_n:
            continue
        if time.monotonic() > deadline - 300:
            break
        if n_try in quick_results:
            a_other = quick_results[n_try][0].copy()
        else:
            a_other = _resize_sequence(best_seq, n_try)
        # Resize best to this n for better init
        a_resized = _resize_sequence(best_seq, n_try)
        a_other = np.maximum(a_other, a_resized * 0.5)  # blend
        a_other = a_other / a_other.sum()

        a_opt2, sc_opt2 = _optimize_at_n(a_other, min(deadline, time.monotonic() + 200), min_remaining=100)
        update_candidate(a_opt2, sc_opt2, f"other_n{n_try}")

    # === Phase 4: Diverse restarts (300s) ===
    print(f"\n[Phase 4] Diverse restarts", flush=True)
    n_restarts = 8
    for i in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break
        if i == 0:
            p_start = _perturb_log(best_seq, 0.02, rng)
        elif i == 1:
            p_start = _perturb_fourier(best_seq, rng, max_amp=0.2)
        elif i == 2:
            p_start = _perturb_amplitude(best_seq, rng, sigma=0.04)
        elif i == 3:
            p_start = _perturb_local(best_seq, rng, n_blocks=12, block_frac=0.03)
        elif i == 4:
            p_start = _flat_spectrum_init(len(best_seq), rng)
            # Blend with current best
            p_start = 0.7 * best_seq + 0.3 * p_start
            p_start = np.maximum(p_start, 0)
            p_start = p_start / p_start.sum()
        elif i == 5:
            p_start = _perturb_log(best_seq, 0.05, rng)
        elif i == 6:
            p_start = _perturb_fourier(best_seq, rng, max_amp=0.5)
        else:
            p_start = _perturb_local(best_seq, rng, n_blocks=20, block_frac=0.02)

        # Quick optimization
        a_r, sc_r = _lbfgs_lse(p_start, min(deadline, time.monotonic() + 120), min_remaining=30)
        update_candidate(a_r, sc_r, f"restart_{i}_lse")

        # Newton polish if promising
        if sc_r < best_score * 1.01 and time.monotonic() < deadline - 40:
            a_r2, sc_r2 = _newton_active_set(a_r, min(deadline, time.monotonic() + 60),
                                             max_iter=150, tol_frac=0.002)
            update_candidate(a_r2, sc_r2, f"restart_{i}_newton")

            # Subgradient
            if time.monotonic() < deadline - 20:
                a_r3, sc_r3 = _subgrad_polish(a_r2, min(deadline, time.monotonic() + 40),
                                              n_steps=3000, lr_init=0.01, tol_frac=0.001)
                update_candidate(a_r3, sc_r3, f"restart_{i}_sg")

    # === Phase 5: Proximal gradient at best n (150s) ===
    print(f"\n[Phase 5] Proximal gradient", flush=True)
    if time.monotonic() < deadline - 30:
        a_pg, sc_pg = _proximal_gradient(best_seq, min(deadline, time.monotonic() + 150), max_iter=50000)
        update_candidate(a_pg, sc_pg, "proximal_grad")

    # === Phase 6: Final aggressive polish (120s) ===
    print(f"\n[Phase 6] Final aggressive polish", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Ultra-low beta LSE
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                update_candidate(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton with very tight tolerance
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_active_set(best_seq, deadline - 3,
                                                 max_iter=300, tol_frac=0.0003, mu_init=1e-8)
            update_candidate(a_fin2, sc_fin2, "final_newton_tight")

        # Final subgradient with very tight active set
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 2,
                                              n_steps=5000, lr_init=0.008, tol_frac=0.0003)
            update_candidate(a_fin3, sc_fin3, "final_sg_tight")

    # === Final validation ===
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256", flush=True)
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
