```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over nonnegative a.

Strategy:
- Start from the provided parent if available.
- Use FFT-based LSE smoothing of the max-convolution objective in log-space.
- Multi-scale search: LP-style coarse refinement, 1024/4096/16384/32768,
  optional 65536 for shape guidance, then final polish at 16384/32768.
- Custom Adam in log-space plus SciPy L-BFGS-B with beta-continuation.
- Linear-program best-response iterations on medium sizes as an alternative basin.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize, linprog
except Exception:
    from scipy.optimize import minimize
    linprog = None


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _uniform(n):
    return np.ones(int(n), dtype=np.float64) / float(int(n))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s


def _self_conv(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:out_n]


def _h_conv(a, w):
    """
    h_i = sum_l a_l w_{i+l}, i=0..n-1.
    Computed with one FFT of size ~2n by exploiting no wrap-around in the
    needed middle block of conv(a[::-1], w).
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    c = np.fft.irfft(fa * fw, N)
    return c[n - 1:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * len(a) * float(np.max(c)) / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = len(z)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) / float(n)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _a_to_z(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    z = np.log(np.maximum(a, 1e-30))
    z = z - float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def _initial_beta(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    c = _self_conv(a)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return 1e-12
    cs = np.sort(c)[::-1]
    if cs.size > 50:
        gap = float(cs[0] - cs[50])
    elif cs.size > 10:
        gap = float(cs[0] - cs[10])
    else:
        gap = float(cs[0])
    if gap < 0.0 or not np.isfinite(gap):
        gap = 0.1 * cmax
    base = max(0.25 * gap, 0.01 * cmax, 1e-6 * cmax, 1e-15)
    return float(base)


def _lse_obj(z, beta, n):
    if not np.all(np.isfinite(z)):
        return np.inf
    a = _softmax(z)
    c = _self_conv(a)
    cmax = float(np.max(c))
    if beta <= 1e-18:
        return float(2.0 * n * cmax)
    shift = np.clip((c - cmax) / beta, -700.0, 0.0)
    e = np.exp(shift)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return float(2.0 * n * cmax)
    val = 2.0 * n * (cmax + beta * np.log(s))
    if not np.isfinite(val):
        return np.inf
    return float(val)


def _lse_grad(z, beta, n):
    if not np.all(np.isfinite(z)):
        return np.zeros(int(n), dtype=np.float64)
    a = _softmax(z)
    c = _self_conv(a)
    cmax = float(np.max(c))
    if beta <= 1e-18:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        shift = np.clip((c - cmax) / beta, -700.0, 0.0)
        e = np.exp(shift)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    h = _h_conv(a, w)
    W = float(np.dot(a, h))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(grad)
    return grad


def _lse_evaluate(z, beta, n):
    a = _softmax(z)
    c = _self_conv(a)
    cmax = float(np.max(c))
    if beta <= 1e-18:
        obj = 2.0 * n * cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        shift = np.clip((c - cmax) / beta, -700.0, 0.0)
        e = np.exp(shift)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            obj = 2.0 * n * cmax
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
            obj = 2.0 * n * (cmax + beta * np.log(s))
    h = _h_conv(a, w)
    W = float(np.dot(a, h))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(grad)
    return a, float(obj), cmax, w, h, grad


def _run_adam(a_init, deadline, base_beta=None, lr=0.15, beta_factors=None, label=""):
    a = _normalize(a_init)
    if a is None:
        return np.zeros(0, dtype=np.float64), np.inf
    n = len(a)
    z = _a_to_z(a)
    best_a = a.copy()
    best_sc = _score(a)

    if beta_factors is None:
        beta_factors = [
            1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
            0.0075, 0.00375, 0.001875, 0.0009, 0.00045, 0.000225
        ]
    if base_beta is None:
        base_beta = _initial_beta(a)
    if base_beta <= 0.0 or not np.isfinite(base_beta):
        base_beta = 1e-8

    print(f"[adam] {label} n={n} start={best_sc:.10f}")

    now0 = time.monotonic()
    if deadline <= now0 + 1.0:
        return best_a, best_sc

    total_time = max(1.0, float(deadline - now0))
    seg = total_time / max(1, len(beta_factors))

    m1 = np.zeros(n, dtype=np.float64)
    m2 = np.zeros(n, dtype=np.float64)
    beta_idx = 0
    step_lr = lr
    no_improve = 0
    phase_best = best_sc
    t = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - now0
        target_idx = int(elapsed / seg)
        if target_idx >= len(beta_factors):
            target_idx = len(beta_factors) - 1
        if target_idx != beta_idx:
            beta_idx = target_idx
            m1.fill(0.0)
            m2.fill(0.0)

        beta = base_beta * float(beta_factors[beta_idx])
        step_lr = lr * (0.35 + 0.65 * float(beta_factors[beta_idx]))

        a_obj, obj, cmax, w, h, grad = _lse_evaluate(z, beta, n)
        s = float(a_obj.sum())
        sc = 2.0 * n * cmax / (s * s) if s > 0 else np.inf
        if np.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = float(sc)
            best_a = a_obj.copy()

        t += 1
        m1 = 0.9 * m1 + 0.1 * grad
        m2 = 0.999 * m2 + 0.001 * (grad * grad)
        z_new = z - step_lr * (m1 / (np.sqrt(m2) + 1e-8))
        if not np.all(np.isfinite(z_new)):
            z_new = z
        else:
            z_new = z_new - float(np.mean(z_new))
            z = np.clip(z_new, -50.0, 50.0)

        if t % 250 == 0:
            if best_sc < phase_best - 1e-12:
                phase_best = best_sc
                no_improve = 0
            else:
                no_improve += 250
            if no_improve > 5000:
                step_lr *= 0.7
                no_improve = 0
                m1.fill(0.0)
                m2.fill(0.0)
                z = _a_to_z(best_a)
                if step_lr < 1e-4:
                    break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = float(sc_final)
        best_a = a_final.copy()

    print(f"[adam] {label} done={best_sc:.10f} n={n}")
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, base_beta=None, aggressive=True, label=""):
    a = _normalize(a_init)
    if a is None:
        return np.zeros(0, dtype=np.float64), np.inf
    n = len(a)
    z = _a_to_z(a)
    best_a = a.copy()
    best_sc = _score(a)

    if aggressive:
        factors = [
            1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
            0.0075, 0.0035, 0.0018, 0.0009, 0.00045,
            0.00022, 0.00011, 0.000055
        ]
    else:
        factors = [
            1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015,
            0.0075, 0.003, 0.001, 0.0005
        ]

    if base_beta is None:
        base_beta = _initial_beta(a)
    if base_beta <= 0.0 or not np.isfinite(base_beta):
        base_beta = 1e-8

    print(f"[lbfgs] {label} n={n} start={best_sc:.10f}")

    if n <= 2048:
        cap, tpi, maxcor = 2000, 0.001, 80
    elif n <= 4096:
        cap, tpi, maxcor = 1500, 0.002, 80
    elif n <= 8192:
        cap, tpi, maxcor = 1000, 0.005, 80
    elif n <= 16384:
        cap, tpi, maxcor = 800, 0.012, 100
    elif n <= 32768:
        cap, tpi, maxcor = 600, 0.03, 60
    else:
        cap, tpi, maxcor = 400, 0.06, 40

    m = len(factors)
    for i, fac in enumerate(factors):
        remaining = deadline - time.monotonic()
        if remaining < 8.0:
            break
        factors_left = m - i
        max_iter = max(20, min(cap, int(remaining / (max(1, factors_left) * tpi * 5.0))))
        beta = base_beta * float(fac)
        try:
            res = minimize(
                _lse_obj,
                z,
                jac=_lse_grad,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': int(max_iter),
                    'ftol': 1e-14,
                    'gtol': 1e-9,
                    'maxcor': int(maxcor),
                    'maxls': 100,
                    'disp': False
                }
            )
            if getattr(res, "x", None) is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax(res.x)
                sc_new = _score(a_new)
                if sc_new < best_sc:
                    best_sc = float(sc_new)
                    best_a = a_new.copy()
                    z = _a_to_z(a_new)
                    print(f"[lbfgs] {label} beta_fac={fac:.6g} new={best_sc:.10f}")
                else:
                    # Continue from the improved area only if not too far.
                    if sc_new < best_sc + 0.01:
                        z = _a_to_z(a_new.copy())
                    else:
                        z = _a_to_z(best_a)
            else:
                break
        except Exception:
            break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = float(sc_final)
        best_a = a_final.copy()

    print(f"[lbfgs] {label} done={best_sc:.10f} n={n}")
    return best_a, best_sc


def _resample(a, n_new):
    a = _normalize(a)
    if a is None:
        return _uniform(int(n_new))
    n_new = int(n_new)
    n_old = len(a)
    if n_new <= 0:
        return None
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    loga = np.log(np.maximum(a, 1e-30))
    lognew = np.interp(x_new, x_old, loga)
    b = np.exp(lognew - np.max(lognew))
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _uniform(n_new)
    return b / s


def _lp_best_response(f, M, timeout=2.0, max_n=1024):
    if linprog is None:
        return None
    n = len(f)
    if n > int(max_n) or n <= 0:
        return None
    if M <= 0.0 or not np.isfinite(M):
        return None

    fpos = np.maximum(f, 1e-10)
    sp = float(fpos.sum())
    if sp <= 0.0:
        return None
    fpos = fpos / sp

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = fpos[i]

    b = np.full(2 * n - 1, float(M), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method='highs',
            options={'presolve': True, 'time_limit': max(0.5, float(timeout))}
        )
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            return None

    x = getattr(res, "x", None)
    if x is None:
        return None
    g = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        return None
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def _lp_iterate(a_init, deadline, lp_timeout=2.0, max_iter=100, label=""):
    a = _normalize(a_init)
    if a is None:
        return np.zeros(0, dtype=np.float64), np.inf
    f = a.copy()
    best_f = f.copy()
    best_sc = _score(f)
    print(f"[lp] {label} n={len(f)} start={best_sc:.10f}")

    for it in range(int(max_iter)):
        if time.monotonic() >= deadline:
            break
        c = _self_conv(f)
        cmax = float(np.max(c))
        M = max(1e-15, cmax * 1.01)
        g = _lp_best_response(f, M, timeout=lp_timeout, max_n=len(f))
        if g is None:
            break

        cand_seq = [0.5, 0.25, 0.10, 0.05, 0.02, 0.01]
        best_t = None
        best_mix = None
        best_mix_sc = best_sc
        for t in cand_seq:
            mix = (1.0 - t) * f + t * g
            mix = _normalize(mix)
            if mix is None:
                continue
            sc = _score(mix)
            if sc < best_mix_sc:
                best_mix_sc = sc
                best_mix = mix
                best_t = t

        if best_mix is not None and best_mix_sc < best_sc - 1e-12:
            f = best_mix
            best_f = f.copy()
            best_sc = float(best_mix_sc)
            print(f"[lp] {label} it={it} t={best_t} score={best_sc:.10f}")
        else:
            # One very small step before giving up.
            mix = (1.0 - 0.002) * f + 0.002 * g
            mix = _normalize(mix)
            sc = _score(mix) if mix is not None else np.inf
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                f = mix
                best_f = f.copy()
                best_sc = float(sc)
                print(f"[lp] {label} it={it} tiny score={best_sc:.10f}")
            else:
                break

    print(f"[lp] {label} done={best_sc:.10f} n={len(f)}")
    return best_f, best_sc


def _random_smooth_a(n, seed, m=256):
    rng = np.random.default_rng(int(seed))
    freqs = np.arange(m, dtype=np.float64)
    amp = np.exp(-freqs / max(1.0, float(m) / 4.0))
    coefs = rng.standard_normal(m) * amp
    sig = np.fft.irfft(coefs, m)
    sig = sig - float(np.mean(sig))
    sig = sig + rng.uniform(-1.0, 1.0) * np.linspace(-1.0, 1.0, m, dtype=np.float64)
    sig = np.clip(sig, -3.0, 3.0)
    b = np.exp(sig)
    s = float(b.sum())
    if s <= 0.0 or not np.isfinite(s):
        b = _uniform(m)
    else:
        b = b / s
    return _resample(b, int(n))


def _load_parent():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        a = _normalize(seq)
        return a
    except Exception:
        return None


def _update_best(best, seq, sc, label):
    if seq is None:
        return
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = _score(a)
    if not np.isfinite(sc):
        return
    if sc < best["score"]:
        best["score"] = float(sc)
        best["seq"] = a / s
        best["label"] = str(label)
        print(f"NEW BEST score={best['score']:.10f} n={len(a)} {label}")


def propose_candidate(seed=42, budget_s=1800.0):
    t0 = time.monotonic()
    total = max(60.0, float(budget_s))
    master_dl = t0 + max(10.0, total - 50.0)

    best = {"score": np.inf, "seq": None, "label": "init"}

    parent = _load_parent()
    if parent is not None:
        _update_best(best, parent, None, "parent")

    if best["seq"] is None:
        init = _uniform(4096)
        _update_best(best, init, None, "init_uniform")

    if best["seq"] is None:
        fallback = _uniform(16384)
        return [float(x) for x in (_normalize(fallback) if fallback.sum() > 0.01 else fallback)]

    print(f"initial best={best['score']:.10f} n={len(best['seq'])} t={time.monotonic()-t0:.1f}s")

    # Coarse LP-style search.
    if time.monotonic() < master_dl - 700:
        n = 1024
        a0 = _resample(best["seq"], n)
        if a0 is None:
            a0 = _uniform(n)
        dl = min(master_dl - 650, time.monotonic() + 80.0)
        a1, sc1 = _lp_iterate(a0, dl, lp_timeout=2.0, max_iter=80, label="lp1024")
        _update_best(best, a1, sc1, "lp1024")

    # Mid-size 4096.
    if time.monotonic() < master_dl - 600:
        n = 4096
        a0 = _resample(best["seq"], n)
        if a0 is None:
            a0 = _uniform(n)
        adl = min(master_dl - 550, time.monotonic() + 50.0)
        a1, sc1 = _run_adam(a0, adl, lr=0.25, label="adam4096")
        _update_best(best, a1, sc1, "adam4096")
        ldl = min(master_dl - 500, time.monotonic() + 70.0)
        a2, sc2 = _run_lbfgs(a1, ldl, aggressive=True, label="lbfgs4096")
        _update_best(best, a2, sc2, "lbfgs4096")

    # Primary 16384 refinement.
    if time.monotonic() < master_dl - 500:
        n = 16384
        a0 = _resample(best["seq"], n)
        if a0 is None:
            a0 = _uniform(n)
        adl = min(master_dl - 450, time.monotonic() + 120.0)
        a1, sc1 = _run_adam(a0, adl, lr=0.18, label="adam16k")
        _update_best(best, a1, sc1, "adam16k")
        ldl = min(master_dl - 300, time.monotonic() + 300.0)
        a2, sc2 = _run_lbfgs(a1, ldl, aggressive=True, label="lbfgs16k")
        _update_best(best, a2, sc2, "lbfgs16k")

    # Larger 32768 refinement.
    if time.monotonic() < master_dl - 350:
        n = 32768
        a0 = _resample(best["seq"], n)
        if a0 is None:
            a0 = _uniform(n)
        adl = min(master_dl - 300, time.monotonic() + 120.0)
        a1, sc1 = _run_adam(a0, adl, lr=0.14, label="adam32k")
        _update_best(best, a1, sc1, "adam32k")
        ldl = min(master_dl - 200, time.monotonic() + 300.0)
        a2, sc2 = _run_lbfgs(a1, ldl, aggressive=True, label="lbfgs32k")
        _update_best(best, a2, sc2, "lbfgs32k")

    # Optional very-long shape search at 65536; only resampled back to 32768.
    if time.monotonic() < master_dl - 450:
        n = 65536
        a0 = _resample(best["seq"], n)
        if a0 is not None:
            adl = min(master_dl - 400, time.monotonic() + 80.0)
            a1, sc1 = _run_adam(a0, adl, lr=0.12, label="adam65k")
            a32 = _resample(a1, 32768)
            if a32 is not None and time.monotonic() < master_dl - 200:
                adl2 = min(master_dl - 180, time.monotonic() + 60.0)
                a2, sc2 = _run_adam(a32, adl2, lr=0.10, label="adam32k_from65")
                _update_best(best, a2, sc2, "adam32k_from65")
                ldl2 = min(master_dl - 100, time.monotonic() + 120.0)
                a3, sc3 = _run_lbfgs(a2, ldl2, aggressive=True, label="lbfgs32k_from65")
                _update_best(best, a3, sc3, "lbfgs32k_from65")

    # Diversifying random smooth starts if time remains.
    if time.monotonic() < master_dl - 250:
        for rs, n_rs in ((int(seed) + 101, 4096), (int(seed) + 202, 16384)):
            if time.monotonic() >= master_dl - 200:
                break
            a0 = _random_smooth_a(n_rs, rs)
            adl = min(master_dl - 190, time.monotonic() + 50.0)
            a1, sc1 = _run_adam(a0, adl, lr=0.20, label=f"randadam{n_rs}_{rs}")
            _update_best(best, a1, sc1, f"randadam{n_rs}_{rs}")
            ldl = min(master_dl - 170, time.monotonic() + 60.0)
            a2, sc2 = _run_lbfgs(a1, ldl, aggressive=False, label=f"randlbfgs{n_rs}_{rs}")
            _update_best(best, a2, sc2, f"randlbfgs{n_rs}_{rs}")

    # Final polish at 32768 and 16384.
    if best["seq"] is not None and time.monotonic() < master_dl - 15:
        rem = max(0.0, master_dl - 15 - time.monotonic())
        if rem > 30:
            a0 = _resample(best["seq"], 32768)
            adl = min(master_dl - 40, time.monotonic() + max(10.0, rem * 0.30))
            a1, sc1 = _run_adam(a0, adl, lr=0.08, label="final32adam")
            _update_best(best, a1, sc1, "final32adam")
            ldl = min(master_dl - 25, time.monotonic() + max(10.0, rem * 0.45))
            a2, sc2 = _run_lbfgs(a1, ldl, aggressive=False, label="final32lbfgs")
            _update_best(best, a2, sc2, "final32lbfgs")

        if time.monotonic() < master_dl - 10:
            rem2 = max(0.0, master_dl - 10 - time.monotonic())
            if rem2 > 15:
                a0 = _resample(best["seq"], 16384)
                adl = min(master_dl - 10, time.monotonic() + max(5.0, rem2 * 0.40))
                a1, sc1 = _run_adam(a0, adl, lr=0.08, label="final16adam")
                _update_best(best, a1, sc1, "final16adam")
                ldl = min(master_dl - 5, time.monotonic() + max(5.0, rem2 * 0.40))
                a2, sc2 = _run_lbfgs(a1, ldl, aggressive=False, label="final16lbfgs")
                _update_best(best, a2, sc2, "final16lbfgs")

    final_seq = best["seq"]
    if final_seq is None:
        final_seq = _uniform(16384)

    # Prefer a final length <= 32768 for direct-convolution evaluation safety.
    if len(final_seq) > 32768:
        final_seq = _resample(final_seq, 32768)

    final_seq = np.asarray(final_seq, dtype=np.float64).ravel()
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _uniform(16384)
    else:
        final_seq = final_seq / s

    final_seq = np.nan_to_num(final_seq, nan=0.0, posinf=0.0, neginf=0.0)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    if final_seq.sum() < 0.01:
        final_seq = _uniform(16384)

    print(f"FINAL returned n={len(final_seq)} est_score={_score(final_seq):.10f} "
          f"time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final_seq]
```
