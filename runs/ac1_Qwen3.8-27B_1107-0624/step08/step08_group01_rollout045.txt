def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj_and_grad(z, beta, n):
    """Compute LSE objective and gradient jointly."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        obj, _ = _lse_obj_and_grad(z, self.beta, self.n)
        return obj

    def grad(self, z):
        _, g = _lse_obj_and_grad(z, self.beta, self.n)
        return g


def _run_fine_lbfgs(a_init, deadline, n_beta=300):
    """L-BFGS with ultra-fine geometric beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Determine beta range
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    beta_start = max(max_c * 0.5, gap * 10)
    beta_end = max(max_c * 0.00001, 1e-15)

    # Geometric schedule
    if beta_start <= beta_end:
        betas = np.array([beta_start])
    else:
        ratio = (beta_end / beta_start) ** (1.0 / (n_beta - 1))
        betas = beta_start * (ratio ** np.arange(n_beta))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Time budget per beta step
    total_time_available = deadline - time.monotonic()
    time_per_beta = max(0.1, total_time_available / (n_beta + 50))

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 10.0:
            break

        # Adaptive iterations: more at large beta, fewer at small beta
        frac = i / n_beta
        if frac < 0.3:
            max_iter = min(200, max(20, int(time_per_beta / 0.005)))
        elif frac < 0.7:
            max_iter = min(100, max(10, int(time_per_beta / 0.005)))
        else:
            max_iter = min(50, max(5, int(time_per_beta / 0.005)))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 30}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 50 == 0:
                    print(f"    LSE step {i}/{n_beta}, beta={beta:.6g}, score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _active_set_gd(a_init, deadline, tol_start=0.003, tol_end=0.00001,
                   lr_start=0.02, top_k_start=3, top_k_end=15):
    """Active-set GD with adaptive top-k and step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    no_improve = 0
    max_stall = 500
    stall_count = 0

    total_steps = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline - 2.0:
        progress = min(1.0, (time.monotonic() - t0) / max(1.0, deadline - t0))
        tol = tol_start * (tol_end / tol_start) ** progress
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        lr = lr_start * (1.0 - 0.5 * progress)

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Select active peaks
        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 5:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = min(lr, 1.0 / (gnorm * 2 + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= max_stall:
                a = best_a.copy()
                no_improve = 0
                stall_count += 1
                if stall_count > 20:
                    break
        else:
            total_steps += 1

    return best_a, _score(best_a)


def _peak_balance(a_init, deadline, n_trials=500):
    """Targeted peak balancing: find pairs of peaks and redistribute mass."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find elements contributing to the peak
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)

        if k_lo > k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * j_range[j_range >= 0].size * 0  # placeholder
        # Actually: contrib[i] = a[i] * a[k_star - i]
        valid_j = (j_range >= 0) & (j_range < n)
        if valid_j.sum() == 0:
            continue

        contrib_vals = np.zeros(len(i_range))
        contrib_vals[valid_j] = a[i_range[valid_j]] * a[j_range[valid_j]]
        total = contrib_vals.sum()
        if total < 1e-15:
            continue

        # Find the top contributor and a low contributor
        top_idx_local = int(np.argmax(contrib_vals))
        low_indices = np.where(contrib_vals < 0.1 * contrib_vals.max())[0]
        if len(low_indices) == 0:
            low_indices = np.array([top_idx_local])

        low_idx_local = low_indices[rng.integers(len(low_indices))]

        top_i = int(i_range[top_idx_local])
        top_j = int(j_range[top_idx_local])
        low_i = int(i_range[low_idx_local])
        low_j = int(j_range[low_idx_local])

        # Redistribute: reduce a[top_i] (and a[top_j] if different) and increase a[low_i] (and a[low_j])
        eps = rng.uniform(0.001, 0.01) * max(a[top_i], a[top_j if top_j != top_i else top_i])

        a_try = a.copy()
        a_try[top_i] -= eps
        if top_j != top_i:
            a_try[top_j] -= eps / 2
        a_try[low_i] += eps
        if low_j != low_i and low_j != low_i:
            a_try[low_j] += eps / 2

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(777)
        a = rng.exponential(1.0, n)
        return a / a.sum()
    return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
