Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed-max L-BFGS, active-set QP with preconditioned CG,
coordinate descent polish, and multi-n exploration.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_precond(W, n, rhs, diag, max_iter=1000, tol=1e-13):
    """Preconditioned CG to solve H*x = rhs."""
    x = np.zeros(n)
    r = rhs.copy()
    z = r / np.maximum(diag, 1e-30)
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:

            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        z = r / np.maximum(diag, 1e-30)

        if not np.isfinite(rz_new) or abs(rz_new) < 1e-30:
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        beta = rz_new / rz
        p = z + beta * p
        rz = rz_new
    return x

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    h = wc[n - 1:2 * n - 1]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_precond(W, n, rhs, max_iter=1000, tol=1e-13):
    x = np.zeros(n)
    r = rhs.copy()
    z = r.copy()
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        z = r.copy()
        rz_new = float(r @ z)
        if not np.isfinite(rz_new) or abs(rz_new) < tol * tol * abs(rz0):
            break
        beta_cg = rz_new / rz
        p = z + beta_cg * p
        rz = rz_new
    return x

            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_iter_time = 0.004 * (n / 4096.0 + 0.5)
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 8))))

        obj = _SmoothMaxObj(float(beta), n)
        try:
            result = minimize(
                obj.eval_obj, z,
                jac=obj.eval_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                prev_sc = current_sc
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
                if i < 20 or (prev_sc - sc_new) > 1e-8:
                    print(f"    LBFGS b={beta:.2e} sc={sc_new:.10f} d={prev_sc-sc_new:.2e} t={time.monotonic()-t_start_global:.1f}s")
            else:
                stall_count += 1
                if stall_count >= 4:
                    break
        except Exception:
            pass

    return current_a


def _run_power_mean_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4.0, 6.0, 8.0, 12.0, 16.0]
    for p in p_values:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(3000, max(100, int(remaining / 0.05)))
        obj = _PowerMeanObj(p, n)
        try:
            result = minimize(
                obj.eval_obj, z,
                jac=obj.eval_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_qp(a_init, deadline, max_rounds=100, tol_schedule=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                        0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                        0.0000005, 0.0000002, 0.0000001, 0.00000005,
                        0.00000002, 0.00000001]

    tol_idx = 0
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute diagonal of H for preconditioning
        # H_{j,m} = 2*W[j+m], so H_{j,j} = 2*W[j+j] = 2*W[2j]
        # But W has length 2n-1, so 2j can be up to 2(n-1) = 2n-2 which is valid
        diag_H = 2.0 * W[0:2 * n - 1:2]  # W[0], W[2], W[4], ... W[2n-2]

        # Solve KKT system:
        # H a = lambda * 1
        # 1^T a = 1
        # Use CG for H*y = 1
        y = _cg_solve_diag_precond(W, n, np.ones(n), diag_H, max_iter=800, tol=1e-13)
        oney = float(np.sum(y))
        if abs(oney) < 1e-20 or not np.isfinite(oney):
            # Fallback: use unpreconditioned
            y = _cg_solve_precond(W, n, np.ones(n), max_iter=800, tol=1e-13)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
                continue

        a_new = y / oney
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
            continue
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Line search: blend
            for alpha in [0.5, 0.3, 0.1, 0.05]:
                a_blend = (1 - alpha) * a + alpha * a_new
                a_blend = np.maximum(a_blend, 0)
                s_blend = a_blend.sum()
                if s_blend < 1e-15:
                    continue
                a_blend = a_blend / s_blend
                sc_blend = _score(a_blend)
                if sc_blend < best_sc - 1e-15:
                    best_sc = sc_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    break

        # Check convergence
        c_check = _full_convolve(a, a)
        c_max_check = float(c_check.max())
        active_check = c_check >= (c_max_check - tol * c_max_check)
        if active_check.sum() > 0:
            W_check = np.zeros(2 * n - 1)
            W_check[active_check] = 1.0 / active_check.sum()
            grad = _hankel_matvec(a, W_check, n)
            lam = float(np.mean(grad[a > 1e-10])) if (a > 1e-10).any() else float(np.mean(grad))
            resid = float(np.max(np.abs(grad[a > 1e-10] - lam))) if (a > 1e-10).sum() > 1 else 0.0
            off = a <= 1e-10
            if off.any():
                resid = max(resid, float(np.max(np.maximum(0, lam - grad[off]))))
            if resid < 1e-10 * (float(np.max(np.abs(grad))) + 1e-15):
                if (rnd + 1) % rounds_per_tol >= rounds_per_tol - 1:
                    tol_idx += 1
        else:
            tol_idx += 1

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _block_descent(a_init, deadline, block_size=64, max_sweeps=50):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    n_blocks = (n + block_size - 1) // block_size
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Shuffle block order
        block_order = rng.permutation(n_blocks)
        improved = False
        for bi in block_order:
            if time.monotonic() > deadline:
                break
            start_idx = bi * block_size
            end_idx = min(start_idx + block_size, n)
            if end_idx - start_idx < 2:
                continue

            # Optimize the block while keeping others fixed
            a_block = a[start_idx:end_idx].copy()
            a_rest = np.concatenate([a[:start_idx], a[end_idx:]])
            s_rest = float(a_rest.sum())

            for inner in range(20):
                # Compute gradient w.r.t. block
                a_full = np.concatenate([a[:start_idx], a_block, a[end_idx:]])
                c = _full_convolve(a_full, a_full)
                c_max = float(c.max())
                tol = 0.001 * c_max
                active = c >= (c_max - tol)
                if active.sum() == 0:
                    active[int(np.argmax(c))] = True
                w = np.zeros_like(c)
                w[active] = 1.0 / active.sum()

                # Gradient of a^T H_w a w.r.t. a_block
                a_full_rev = a_full[::-1]
                wc = _full_convolve(w, a_full_rev)
                h_full = wc[n - 1:2 * n - 1]
                W_val = float(np.dot(w, c))
                grad_full = 4.0 * n * a_full * (h_full - W_val)
                grad_block = grad_full[start_idx:end_idx]

                # Remove mean to handle the sum constraint
                grad_block_t = grad_block - np.mean(grad_block)

                # Newton-like step
                step_size = 0.1 / (1.0 + 0.05 * inner)
                a_block_new = a_block - step_size * grad_block_t
                a_block_new = np.maximum(a_block_new, 0)
                s_block = a_block_new.sum()
                if s_block < 1e-15:
                    break
                # Renormalize to maintain total sum = 1
                target_s = 1.0 - s_rest
                a_block_new = a_block_new * (target_s / s_block)

                a_try = np.concatenate([a[:start_idx], a_block_new, a[end_idx:]])
                sc_try = _score(a_try)
                if sc_try < _score(a_full) - 1e-16:
                    a_block = a_block_new
                    a = a_try
                    improved = True
                else:
                    break

            a = np.concatenate([a[:start_idx], a_block, a[end_idx:]])

        sc_cur = _score(a)
        if sc_cur < best_sc - 1e-14:
            best_sc = sc_cur
            best_a = a.copy()
        if not improved:
            break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "smooth_taper":
        t = np.linspace(0, 1, n)
        # Very smooth taper with wide plateau
        a = 1.0 / (1.0 + np.exp(-20 * (t - 0.1)) * np.exp(-20 * (0.9 - t)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, seed=42):
    rng = np.random.default_rng(seed)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def propose_candidate(seed=42, budget_s=1800):
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = float('inf')

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start_global:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_par = float(parent_raw.sum())
            if s_par > 0.01:
                parent = parent_raw / s_par
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent found, using tapered profile n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS from parent ===
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start_global:.1f}s)")
    a1 = _run_lbfgs_adaptive(parent, deadline, min_time_remaining=600.0, max_betas=35)
    _update_best(a1, _score(a1), "lbfgs_p1")

    # === Phase 2: Power-mean L-BFGS ===
    print(f"\n[Phase 2] Power-mean L-BFGS (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2 = _run_power_mean_lbfgs(best_seq, deadline, min_time_remaining=100.0)
        _update_best(a2, _score(a2), "power_mean")

    # === Phase 3: Active-set QP polish ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        qp_deadline = min(deadline - 400, time.monotonic() + 250)
        a3, sc3 = _active_set_qp(best_seq, qp_deadline, max_rounds=80)
        _update_best(a3, sc3, "qp_p3")

    # === Phase 4: L-BFGS from QP result ===
    print(f"\n[Phase 4] L-BFGS from QP (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4 = _run_lbfgs_adaptive(best_seq, deadline, min_time_remaining=300.0, max_betas=25)
        _update_best(a4, _score(a4), "lbfgs_p4")

    # === Phase 5: Second QP round (tighter) ===
    print(f"\n[Phase 5] QP round 2 (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        qp_deadline2 = min(deadline - 250, time.monotonic() + 150)
        a5, sc5 = _active_set_qp(best_seq, qp_deadline2, max_rounds=100,
                                  tol_schedule=[0.0001, 0.00005, 0.00002, 0.00001,
                                                0.000005, 0.000002, 0.000001,
                                                0.0000005, 0.0000002, 0.0000001,
                                                0.00000005, 0.00000002, 0.00000001])
        _update_best(a5, sc5, "qp_p5")

    # === Phase 6: Block descent ===
    print(f"\n[Phase 6] Block descent (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a6, sc6 = _block_descent(best_seq, min(deadline - 200, time.monotonic() + 100),
                                  block_size=128, max_sweeps=30)
        _update_best(a6, sc6, "block")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start_global:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=80.0, max_betas=20)
        sc_opt = _score(a_opt)
        _update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 80:
            qp_dl = min(deadline - 80, time.monotonic() + 60)
            a_np, sc_np = _active_set_qp(a_opt, qp_dl, max_rounds=40)
            _update_best(a_np, sc_np, f"n_{n_try}_qp")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_r = len(best_seq)

        # Restart 1: Perturb best
        p1 = _perturb(best_seq, 0.03, seed=777)
        a_r1 = _run_lbfgs_adaptive(p1, deadline, min_time_remaining=30.0, max_betas=15)
        _update_best(a_r1, _score(a_r1), "restart_perturb")

        # Restart 2: Smooth taper profile
        if time.monotonic() < deadline - 40:
            p2 = _make_profile(n_r, "smooth_taper")
            a_r2 = _run_lbfgs_adaptive(p2, deadline, min_time_remaining=30.0, max_betas=15)
            _update_best(a_r2, _score(a_r2), "restart_smooth")

        # Restart 3: Shrink + reoptimize
        if time.monotonic() < deadline - 30:
            p3 = best_seq.copy()
            idx3 = np.argsort(p3)
            p3[idx3[:len(idx3) // 8]] = 0.0
            p3 = p3 / p3.sum()
            a_r3 = _run_lbfgs_adaptive(p3, deadline, min_time_remaining=25.0, max_betas=10)
            _update_best(a_r3, _score(a_r3), "restart_shrink")

    # === Phase 9: Final ultra-tight QP ===
    print(f"\n[Phase 9] Final QP (t={time.monotonic() - t_start_global:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_qp(
            best_seq, deadline - 3, max_rounds=80,
            tol_schedule=[0.000005, 0.000002, 0.000001, 0.0000005,
                          0.0000002, 0.0000001, 0.00000005, 0.00000002,
                          0.00000001, 0.000000005, 0.000000002, 0.000000001])
        _update_best(a_fin, sc_fin, "final_qp")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start_global:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
