Using FFT: This is equivalent to the convolution of `a_reversed` with `w`, taking the appropriate slice.

`conv(a_reversed, w)[k] = sum_i a_reversed[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]`

Let j = n-1-i, so i = n-1-j:
`= sum_j a[j] * w[k-n+1+j]`

I want `sum_j a[j] * w[i+j]`, so set k-n+1 = i, i.e., k = i+n-1.

So: `result[i] = conv(a_reversed, w)[i+n-1]` for i = 0, ..., n-1.

That means: `result = conv(a[::-1], w)[(n-1):(2n-1)]`

And this can be computed with one FFT-based convolution! 

So the W*a product costs O(n log n) using FFT.

This is a system of equations! If I know the active set of a (where a[i] > 0) and the weights w, I can solve for a and lambda.

But this is a(n)s a system of n equations in n+1 unknowns (a_0, ..., a_{n-1}, lambda), with the additional constraint sum(a) = 1.

For a dense active set (all a[i] > 0), the system is:
`sum_j w_{i+j} * a[j] = lambda` for i = 0, ..., n-1
`sum_i a[i] = 1`

The first n equations can be written as `W*a = lambda * 1`, where W[i,j] = w_{i+j}.

From this: `a = W^{-1} * lambda * 1 = lambda * W^{-1} * 1`.

And `sum(a) = 1` gives: `lambda * 1^T W^{-1} * 1 = 1`, so `lambda = 1 / (1^T W^{-1} * 1)`.

So `a = W^{-1} * 1 / (1^T W^{-1} * 1)`.

This is a closed-form solution! If I can compute W^{-1} * 1 efficiently, I have the optimum (assuming all a[i] > 0).

"""
Minimax-based optimization for step function score minimization.
Uses alternating minimax ( multiplicative weights) + LSE polish.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    cond = u * np.arange(1, n + 1) > cssv
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _wa_product(a, w):
    """Compute W*a where W[i,j] = w[i+j]. Uses FFT."""
    n = len(a)
    # result[i] = sum_j a[j] * w[i+j] for i=0..n-1
    # = conv(a[::-1], w)[n-1 : 2*n-1]
    prod = _full_convolve(a[::-1], w)
    return prod[(n-1):(2*n-1)]

def _minimax_phase(a_init, deadline, print_every=500):

    n = len(a_init)
    s = a_init.sum()
    a = a_init / s
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Initialize temperature
    tau = c_max * 0.05
    
    # Initialize w from current autocorrelation
    exp_c = np.exp(np.clip(c / (tau * 10) - c.max() / (tau * 10), -700, 0))
    w = exp_c / exp_c.sum()
    
    # Log-domain variables
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = 0.5
    
    best_a = a.copy()
    best_score = 2.0 * n * c_max
    best_c = c.copy()
    
    iter_count = 0
    no_improve_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Inner loop: multiple Adam steps
        n_inner = 20 if tau > c_max * 0.01 else 10
        for _ in range(n_inner):
            a_cur = _softmax(z)
            wa = _wa_product(a_cur, w)
            obj = float(np.dot(a_cur, wa))
            grad_z = 2.0 * a_cur * wa - 2.0 * obj
            
            t_adam += 1
            m_adam = 0.8 * m_adam + 0.2 * grad_z
            v_adam = 0.999 * v_adam + 0.001 * grad_z ** 2
            m_hat = m_adam / (1.0 - 0.8 ** t_adam)
            v_hat = v_adam / (1.0 - 0.999 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            z = z - z.mean()
        
        # Evaluate

        c_new = _full_convolve(a_new, a_new)
        c_new_max = float(c_new.max())
        sc = 2.0 * n * c_new_max
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            best_c = c_new.copy()
            c = c_new
            c_max = c_new_max

"""
Minimax (multiplicative weights) + LSE L-BFGS hybrid optimizer for step function
score minimization. Uses the key insight that the objective is a minimax problem
between the sequence (minimizer) and a distribution over lags (maximizer),
solved via alternating optimization with Adam in log-space, followed by
aggressive LSE polishing and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    cond = u * np.arange(1, n + 1) > cssv
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _wa_product(a, w):
    n = len(a)
    prod = _full_convolve(a[::-1], w)
    return prod[(n - 1):(2 * n - 1)]


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lse_run(a_init, deadline, beta_multiplier):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    top_k = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[top_k]
    if gap < max_c * 0.0001:
        gap = max_c * 0.0001
    base_beta = max(gap, max_c * 0.001) * beta_multiplier

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
         0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break

        def obj_func(zv):
            return _lse_obj(zv, float(beta), n)

        def grad_func(zv):
            return _lse_grad(zv, float(beta), n)

        scale = max(1.0, n / 4096.0)
        est_iter = 0.004 * scale
        max_iter = min(4000, max(30, int(remaining / (est_iter * 12))))

        try:
            result = minimize(obj_func, z, jac=grad_func, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _minimax_phase(a_init, deadline):
    n = len(a_init)
    s = a_init.sum()
    a = a_init / s
    c = _full_convolve(a, a)
    c_max = float(c.max())

    tau = c_max * 0.05
    tau_floor = c_max * 1e-9

    exp_c = np.exp(np.clip(c / (tau * 5) - c.max() / (tau * 5), -700, 0))
    w = exp_c / exp_c.sum()

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = 0.4

    best_a = a.copy()
    best_score = 2.0 * n * c_max
    no_improve = 0
    iter_count = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        iter_count += 1
        elapsed = time.monotonic() - t0

        n_inner = 30 if tau > c_max * 0.005 else (15 if tau > c_max * 0.0005 else 8)

        for _ in range(n_inner):
            a_cur = _softmax(z)
            wa = _wa_product(a_cur, w)
            obj = float(np.dot(a_cur, wa))
            grad_z = 2.0 * a_cur * wa - 2.0 * obj

            t_adam += 1
            m_adam = 0.8 * m_adam + 0.2 * grad_z
            v_adam = 0.999 * v_adam + 0.001 * grad_z ** 2
            m_hat = m_adam / (1.0 - 0.8 ** t_adam)
            v_hat = v_adam / (1.0 - 0.999 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            z = z - z.mean()

        a_new = _softmax(z)
        c_new = _full_convolve(a_new, a_new)
        c_new_max = float(c_new.max())
        sc = 2.0 * n * c_new_max

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            c = c_new
            c_max = c_new_max
            no_improve = 0
        else:
            no_improve += 1

        c_for_w = c
        max_cw = float(c_for_w.max())
        if tau < tau_floor:
            thresh = max_cw - max_cw * 0.0003
            w = (c_for_w >= thresh).astype(np.float64)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros_like(c_for_w)
                w[int(np.argmax(c_for_w))] = 1.0
            else:
                w = w / ws
        else:
            exp_w = np.exp(np.clip(c_for_w / tau - max_cw / tau, -700, 0))
            wes = exp_w.sum()
            if wes < 1e-300:
                w = np.zeros_like(c_for_w)
                w[int(np.argmax(c_for_w))] = 1.0
            else:
                w = exp_w / wes

        tau *= 0.997
        if tau < tau_floor:
            tau = tau_floor

        lr *= 0.99997
        if lr < 0.005:
            lr = 0.005

        if no_improve > 300:
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            lr = 0.25
            tau = max(c_max * 0.01, tau * 3.0)
            c = _full_convolve(best_a, best_a)
            no_improve = 0

        if iter_count % 2000 == 0:
            print(f"    Minimax iter {iter_count}: score={sc:.10f}, best={best_score:.10f}, "
                  f"tau={tau:.2e}, lr={lr:.5f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def _cg_solve_wa(a_init, w, max_iter=500, tol=1e-10):
    """Solve W*x = 1 using conjugate gradient, where W[i,j]=w[i+j].
    Returns the solution x (unnormalized)."""
    n = len(a_init)
    x = np.ones(n) / n
    r = _wa_product(x, w).copy()
    r = r - r.mean()  # We want W*x = lambda*1, so solve (W - mean(W)*I)*x = 0... 
    # Actually let's solve W*x = 1 directly
    r = _wa_product(x, w) - 1.0
    p = -r.copy()
    rs = float(np.dot(r, r))
    if rs < tol:
        return x

    for _ in range(max_iter):
        Ap = _wa_product(p, w)
        pAp = float(np.dot(p, Ap))
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r_new = r + alpha * Ap
        rs_new = float(np.dot(r_new, r_new))
        if rs_new < tol:
            r = r_new
            break
        beta_cg = rs_new / rs
        p = -r_new + beta_cg * p
        r = r_new
        rs = rs_new

    s = x.sum()
    if s > 1e-15:
        x = x / s
    return x


def _kkt_solve(a_init, deadline):
    """Try to solve KKT conditions directly using CG."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    for trial in range(3):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tau_kkt = c_max * 0.01
        exp_c = np.exp(np.clip(c / tau_kkt - c_max / tau_kkt, -700, 0))
        w = exp_c / exp_c.sum()

        x = _cg_solve_wa(a, w, max_iter=300, tol=1e-12)
        x_pos = np.maximum(x, 0.0)
        s_pos = x_pos.sum()
        if s_pos > 1e-10:
            x_pos = x_pos / s_pos
            sc = _score(x_pos)
            if sc < best_score:
                best_score = sc
                best_a = x_pos.copy()
                a = x_pos

        # Also try: take the CG solution, zero negatives, and renormalize
        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
