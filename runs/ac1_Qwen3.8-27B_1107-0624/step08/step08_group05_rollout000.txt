def solve_kkt(a_init, S, n):
    """Solve M_S * x = 1 using CG with FFT matvec."""
    b = np.zeros(2*n - 1)
    for k in S:
        b[k] = 1.0
    
    def matvec(v):

        N = next_pow2(4*n)

        fb = np.fft.rfft(b, N)
        r = np.fft.irfft(fv * fb, N)
