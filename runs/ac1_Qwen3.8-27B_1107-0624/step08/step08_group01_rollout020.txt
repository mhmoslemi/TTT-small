Optimized step-function search using:
1. FFT-based convolution and correlation for efficient gradient/Hessian computation
2. L-BFGS with adaptive log-sum-exp smoothing
3. Hankel-structured conjugate gradient polishing
4. Active-set gradient descent with adaptive tolerance
5. Coordinate transfer local search
6. Multi-n exploration and diverse restarts
Targets below 1.5027 by exploiting structured optimization and careful local refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_weighted(a, w):
    """Compute gradient of sum_k w[k]*c[k] where c = convolve(a,a).
    Returns grad[i] = 2 * sum_k w[k] * a[k-i] (with a out of range = 0).
    Equivalently: grad = 2 * correlate(w, a, 'valid')
    """
    n = len(a)
    # w has length 2n-1
    # We want g[i] = 2 * sum_{k} w[k] * a[k-i] for i=0..n-1
    # = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is correlate(w, a, 'valid') which gives output of length len(w)-n+1 = n
    # np.correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j]
    g = np.correlate(w, a, mode='valid')
    return 2.0 * g


def _hankel_mv(w, v):
    """Compute H * v where H[i,j] = w[i+j] (Hankel matrix, 0 if i+j >= len(w)).
    H*v = correlate(w, v, 'valid') (up to factor).
    """
    n = len(v)
    # (Hv)[i] = sum_j w[i+j] * v[j] = correlate(w, v, 'valid')[i]
    return np.correlate(w, v, mode='valid')

    """Compute log-sum-exp objective and gradient in z-space (softmax parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    # Gradient in a-space
    grad_a = _grad_weighted(a, w)
    # Gradient in z-space: grad_z[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
    mean_g = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_g)
    return obj, grad_z


def _run_lbfgs(a_init, deadline, min_remaining=15.0, verbose=False):
    """L-BFGS with aggressive beta annealing on softmax-parameterized problem."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3.0:
            break

        scale = max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (0.003 * scale * 10))))

        def fun(zv):
            obj, _ = _lse_obj_and_grad(zv, beta, n)
            return obj

        def jac(zv):
            _, g = _lse_obj_and_grad(zv, beta, n)
            return g

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.8f}, iter={result.nit}")
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_start=0.004, tol_end=0.00001,
                       lr_start=0.02, top_k=5, verbose=False):
    """Active-set gradient descent with adaptive tolerance and backtracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall_count = 0
    max_stall = 400

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Select active set
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > top_k * 15:
            top_indices = np.argsort(c)[::-1][:top_k]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = top_k

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        grad_a = _grad_weighted(a, w)
        grad_t = grad_a - np.mean(grad_a)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = min(lr, 1.0 / (gnorm * gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(100):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 25 == 0:
                tol = max(tol_end, tol * 0.65)
                lr *= 0.55
            if no_improve >= max_stall:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 15:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.98)
            lr = min(lr_start, lr * 1.02)

    return best_a, _score(best_a)


def _cg_polish(a_init, deadline, n_iter=50, verbose=False):
    """Conjugate gradient polish using the Hankel Hessian structure.
    Solves: minimize sum_k w[k]*c[k] subject to a>=0, sum(a)=1
    using CG with the Hessian H[i,j] = 2*w[i+j].
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Use softmax weights for a smooth objective
    beta = c_max * 0.01
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e

    # CG iterations
    # f(a) = sum_k w[k] * c[k]
    # grad = _grad_weighted(a, w)
    # Hessian H[i,j] = 2*w[i+j], H*v = 2*correlate(w, v, 'valid')

    r = -_grad_weighted(a, w)  # negative gradient (desired direction)
    r = r - np.mean(r)  # project onto sum=0 (sum constraint)
    p = r.copy()
    rp = float(np.dot(r, p))

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        if rp < 1e-30:
            break
        Hp = 2.0 * _hankel_mv(w, p)
        Hp = Hp - np.mean(Hp)  # project
        pHp = float(np.dot(p, Hp))
        if pHp < 1e-30:
            break
        alpha = rp / pHp

        # Line search
        base_f = float(np.dot(w, _full_convolve(a, a)))
        improved = False
        for ls in range(30):
            a_try = a + alpha * p
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(w, c_try))
            if f_try < base_f - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            break

        sc_try = _score(a)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a.copy()

        r_new = -_grad_weighted(a, w)
        r_new = r_new - np.mean(r_new)
        rp_new = float(np.dot(r_new, r_new))
        if rp_new < 1e-30:
            break
        p = r_new + (rp_new / max(rp, 1e-30)) * p
        p = p - np.mean(p)
        r = r_new
        rp = float(np.dot(r, p))

    return best_a, best_sc


def _coordinate_transfer(a, deadline, n_rounds=100, verbose=False):
    """Local search: transfer small amounts of mass between coordinates."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Try several random transfers
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Transfer a fraction of a[i] to a[j]
            frac = rng.uniform(0.001, 0.05)
            delta = a[i] * frac
            if delta < 1e-15:
                continue

            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0:
                continue
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            # Quick check: compute new max convolution
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break  # Start new round with improved a

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t * 0.7) ** 0.5 + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_decay":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 0.5 + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: L-BFGS + CG polish + active-set GD + coordinate transfer.
    Explores multiple n values and initialization strategies.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 8192
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS on parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 900.0, t_start + 400.0)
    a1 = _run_lbfgs(best_a, p1_deadline, min_remaining=30.0, verbose=True)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_p1] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: CG polish ===
    print(f"\n[P2] CG polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600.0:
        p2_deadline = min(deadline - 600.0, t_start + 500.0)
        a2, sc2 = _cg_polish(best_a, p2_deadline, n_iter=100)
        if sc2 < best_score:
            best_a = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [cg_p2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set polish ===
    print(f"\n[P3] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500.0:
        p3_deadline = min(deadline - 500.0, t_start + 650.0)
        a3, sc3 = _active_set_polish(best_a, p3_deadline, tol_start=0.003,
                                     tol_end=0.00001, lr_start=0.02, top_k=5)
        if sc3 < best_score:
            best_a = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [active_p3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS again with finer betas ===
    print(f"\n[P4] L-BFGS fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400.0:
        p4_deadline = min(deadline - 400.0, t_start + 800.0)
        a4 = _run_lbfgs(best_a, p4_deadline, min_remaining=20.0, verbose=True)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_a = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_fine_p4] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Coordinate transfer ===
    print(f"\n[P5] Coordinate transfer (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250.0:
        p5_deadline = min(deadline - 250.0, t_start + 950.0)
        a5, sc5 = _coordinate_transfer(best_a, p5_deadline, n_rounds=200)
        if sc5 < best_score:
            best_a = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [coord_p5] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_a)
    n_targets = sorted(set([n_best, n_best * 2, n_best // 2 if n_best > 4096 else n_best,
                            12288, 16384, 24576]))
    n_targets = [x for x in n_targets if 2048 <= x <= 32768]

    for n_try in n_targets:
        if time.monotonic() > deadline - 120.0:
            break
        if n_try == len(best_a):
            continue
        print(f"  Trying n={n_try} (remaining={deadline-time.monotonic():.0f}s)")
        remaining_for_n = deadline - time.monotonic() - 80.0
        if remaining_for_n < 40:
            continue
        a_init = _resample(best_a, n_try)
        # L-BFGS
        sub_deadline = time.monotonic() + remaining_for_n * 0.5
        a_opt = _run_lbfgs(a_init, sub_deadline, min_remaining=10.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        # Active set
        if time.monotonic() < deadline - 60.0:
            sub_dl2 = min(deadline - 50.0, time.monotonic() + remaining_for_n * 0.3)
            a_pol, sc_pol = _active_set_polish(a_opt, sub_dl2, tol_start=0.002,
                                               tol_end=0.00005, lr_start=0.015, top_k=3)
            if sc_pol < best_score:
                best_a = a_pol.copy()
                best_score = sc_pol
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60.0:
        n_best = len(best_a)
        restarts = [
            ("pert_small", _perturb(best_a, 0.01, rng)),
            ("pert_med", _perturb(best_a, 0.04, rng)),
            ("asym_taper", _make_profile(n_best, "asymmetric_taper")),
            ("power_decay", _make_profile(n_best, "power_decay")),
            ("cos_mod", _make_profile(n_best, "cosine_mod")),
        ]
        # Also try resampled parent
        if parent is not None and len(parent) != n_best:
            restarts.append(("parent_rs", _resample(parent, n_best)))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 15.0:
                break
            a_opt = _run_lbfgs(a_r, deadline - 20.0, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        # Active set with very tight tolerance
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 2.0,
                                           tol_start=0.0003, tol_end=0.000001,
                                           lr_start=0.006, top_k=3)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_active] t={time.monotonic()-t_start:.1f}s")

        # CG polish
        if time.monotonic() < deadline - 3.0:
            a_cg, sc_cg = _cg_polish(best_a, deadline - 2.0, n_iter=50)
            if sc_cg < best_score:
                best_a = a_cg.copy()
                best_score = sc_cg
                print(f"  NEW BEST: {best_score:.10f} [final_cg] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 2.0 - time.monotonic()
        if remaining > 5.0:
            a_t = _run_lbfgs(best_a, deadline - 1.5, min_remaining=2.0)
            sc_t = _score(a_t)
            if sc_t < best_score:
                best_a = a_t.copy()
                best_score = sc_t
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
