Optimized step function search using mirror descent with correct FFT-based gradients,
LP-based active-set improvement, and multiple structural initializations.
Targets n=8192 with focus on correct gradient computation and efficient optimization.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max_conv."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _compute_gradient_fft(a, p):
    """
    Compute g[j] = sum_k p[k] * a[k-j] for j=0,...,n-1
    using FFT-based cross-correlation.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(P) * A, N)
    return corr[:n]  # We need corr[0], ..., corr[n-1]

    """
    Mirror descent (exponentiated gradient) step on the simplex.
    a_new[j] = a[j] * exp(-eta * g[j]) / Z

    g_centered = g - g.mean()
    log_update = -eta * g_centered

    a_new = a * np.exp(log_update)
    s = a_new.sum()
    if s < 1e-300:

    a_new /= s

    """

    find b that maximizes sum(b) subject to (a*b)[k] <= M*(1-reduction) for active k, b >= 0.
    """
    from scipy.optimize import linprog
    
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        
        # Find active lags
        threshold = M * (1 - 0.01)
        active = np.where(c >= threshold)[0]
        if len(active) > n_active:
            # Take the top n_active
            active = active[np.argsort(c[active])[::-1][:n_active]]
        
        # Build LP: maximize sum(b) subject to (a*b)[k] <= M_target for k in active, b >= 0
        M_target = M * (1 - reduction)
        
        # (a*b)[k] = sum_j a[j] * b[k-j] for valid k-j
        # For each active k, the constraint is: sum_j A_k[j] * b[j] <= M_target
        # where A_k[j] = a[k-j] if 0 <= k-j < n, else 0
        
        K = len(active)
        A_ub = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A_ub[i, lo:hi+1] = a[k-hi:k-lo+1][::-1]
        
        # Maximize sum(b) => minimize -sum(b)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(K, M_target),
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        b = res.x
        s_b = float(b.sum())
        if s_b < 1e-15:
            break
        
        b_norm = b / s_b
        sc = _score_norm(b_norm)
        
        if sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = b_norm.copy()
            a = b_norm
        else:
            # No improvement, reduce the reduction factor
            reduction *= 0.5
            if reduction < 1e-6:
                break
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, T_start=0.01, T_end=0.0001,
                   lr_start=0.05, lr_end=0.001, max_iter=100000,
                   eval_interval=50, active_k=5):
    """Adam optimizer in log-space with correct FFT gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    T_scale = max_c
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_scale * T_start * (T_end / T_start) ** (frac ** 0.5)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.6) * lr_mult
        
        a = np.exp(z - z.max())
        a /= a.sum()
        
        # Recompute c if needed (do it every iteration for correctness)
        c = _fft_convolve(a, a)
        
        # Softmax weights
        if T < 1e-15:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s_e = e.sum()
            if s_e < 1e-300 or not np.isfinite(s_e):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s_e
        
        # FFT gradient
        g = _compute_gradient_fft(a, p)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        # Center gradient for log-space parameterization
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update / update_norm
        
        z_new = z - update
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur /= a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > 20:
                lr_mult *= 0.8
                patience = 0
                if lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.01 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.3
    
    a_final = np.exp(z - z.max())
    a_final /= a_final.sum()
    sc = _score_norm(a_final)
    if sc < best_score:
        best_a = a_final.copy()
        best_score = sc
    return best_a, best_score

def _subgradient_polish(a, deadline, max_iter=20000, top_k=5, initial_lr=0.005):
    """Subgradient method with top-k peak averaging."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        # Average gradient over active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space
        g_tangent = g - np.dot(g, a) * a
        
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break
        
        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_norm(a_new)
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9998
        else:
            lr *= 0.97
            if lr < 1e-10:
                break
    
    return best_a, _score_norm(best_a)

def _coordinate_descent(a, deadline, max_rounds=100):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        improved = False
        # Try modifying elements that contribute to the peak
        step = max(1, (hi - lo + 1) // 50)
        for j in range(lo, hi + 1, step):
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try broader search
            any_imp = False
            for j in range(0, n, max(1, n // 100)):
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.7, 1.3]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    a_cand /= a_cand.sum()
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        any_imp = True
                        improved = True
                        break
                if improved:
                    break
            if not any_imp:
                break
    
    return best_a, best_score

"""
Optimized step function search using mirror descent with correct FFT-based gradients,
LP-based active-set improvement, Adam in log-space, and subgradient polish.
Key innovation: correct FFT cross-correlation for gradient computation, enabling
efficient and accurate optimization on the simplex constraint.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_gradient_fft(a, p):
    """
    Compute g[j] = sum_k p[k] * a[k-j] for j=0,...,n-1 via FFT cross-correlation.
    a has length n, p has length 2n-1.
    Returns gradient/2 (without the factor of 2).
    """
    n = len(a)
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(P) * A, N)
    return 2.0 * corr[:n]


def _softmax_lags(c, T):
    """Compute softmax weights over lags with temperature T."""
    if T < 1e-15:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return p
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return p
    return e / s_e


def _mirror_descent(a_init, deadline, T_start=0.02, T_end=0.0005,
                    eta_start=0.2, eta_end=0.005, max_iter=500000,
                    eval_interval=20, patience_limit=50):
    """Mirror descent (exponentiated gradient) on the simplex with adaptive T and eta."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    T_scale = float(c.max())
    t = 0
    patience = 0
    eta_mult = 1.0
    log_a = np.log(np.maximum(a, 1e-300))

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_scale * T_start * (T_end / T_start) ** (frac ** 0.4)
        eta = eta_start * (eta_end / eta_start) ** (frac ** 0.5) * eta_mult

        p = _softmax_lags(c, T)
        g = _compute_gradient_fft(a, p)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Mirror descent: a_new[j] = a[j] * exp(-eta * g[j]) / Z
        log_update = -eta * g
        log_update = np.clip(log_update, -60, 60)
        log_a_new = log_a + log_update
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new /= s
        a = a_new
        log_a = np.log(np.maximum(a, 1e-300))

        if t % eval_interval == 0:
            c = _fft_convolve(a, a)
            sc = float(c.max())
            if np.isfinite(sc) and sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if patience > patience_limit:
                eta_mult *= 0.8
                patience = 0
                if eta_mult < 0.05:
                    a = best_a.copy()
                    log_a = np.log(np.maximum(a, 1e-300))
                    c = _fft_convolve(a, a)
                    T_scale = float(c.max())
                    eta_mult = 0.3

    c = _fft_convolve(a, a)
    sc = float(c.max())
    if sc < best_score:
        best_a = a.copy()
        best_score = sc
    return best_a, best_score


def _adam_logspace(a_init, deadline, T_start=0.01, T_end=0.0001,
                   lr_start=0.04, lr_end=0.001, max_iter=500000,
                   eval_interval=40, patience_limit=30):
    """Adam optimizer in log-space with correct FFT gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c = _fft_convolve(a_init, a_init)
    T_scale = float(c.max())
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_scale * T_start * (T_end / T_start) ** (frac ** 0.4)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.5) * lr_mult

        a = np.exp(z - z.max())
        a /= a.sum()
        c = _fft_convolve(a, a)
        p = _softmax_lags(c, T)
        g = _compute_gradient_fft(a, p)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Center gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update / update_norm

        z = z - update
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur /= a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            if patience > patience_limit:
                lr_mult *= 0.8
                patience = 0
                if lr_mult < 0.05:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    lr_mult = 0.3
                    T_scale = float(_fft_convolve(best_a, best_a).max())

    a_final = np.exp(z - z.max())
    a_final /= a_final.sum()
    sc = _score_norm(a_final)
    if sc < best_score:
        best_a = a_final.copy()
        best_score = sc
    return best_a, best_score


def _lp_improve(a_init, deadline, max_iters=100, n_active=25, reduction=0.003):
    """
    LP-based improvement: maximize sum(b) s.t. (a*b)[k] <= M*(1-reduction) for active k, b >= 0.
    Iteratively updates a.
    """
    from scipy.optimize import linprog

    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        M = float(c.max())

        threshold = M * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > n_active:
            active = active[np.argsort(c[active])[::-1][:n_active]]

        M_target = M * (1.0 - reduction)
        K = len(active)
        A_ub = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A_ub[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(K, M_target),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 5})
        except Exception:
            break

        if res.status != 0:
            reduction *= 0.5
            if reduction < 1e-6:
                break
            continue

        b = res.x
        s_b = float(b.sum())
        if s_b < 1e-15:
            break

        b_norm = b / s_b
        sc = _score_norm(b_norm)

        if sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = b_norm.copy()
            a = b_norm
        else:
            reduction *= 0.7
            if reduction < 1e-6:
                break

    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_iter=50000, top_k=5, initial_lr=0.004):
    """Subgradient method with top-k peak averaging and adaptive step."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        g_tangent = g - np.dot(g, a) * a
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break

        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s

        sc = _score_norm(a_new)
        if sc < best_score * (1 - 1e-12):
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9999
        else:
            lr *= 0.98
            if lr < 1e-10:
                break

    return best_a, _score_norm(best_a)
