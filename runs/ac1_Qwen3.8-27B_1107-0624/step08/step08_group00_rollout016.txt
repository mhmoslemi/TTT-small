"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Symmetry-breaking perturbation to escape symmetric local minima
2. Peak-directed descent with backtracking (fast custom method)

import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _solve_qp(a, active_indices, target, reg_strength, deadline):
    """Solve regularized QP: min ||d||^2 + reg*||d||^2 s.t. B@d <= rhs, sum(d)=0."""
    n = len(a)
    c = _full_convolve(a, a)
    m = len(active_indices)

    B_rows = []
    rhs = []
    for k in active_indices:
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        j_lo = k - i_hi
        j_hi = k - i_lo
        row = np.zeros(n)
        a_vals = a[j_lo:k - j_lo + 1][::-1]
        row[j_lo:j_hi + 1] = a_vals
        B_rows.append(row)
        rhs.append((target - c[k]) / 2.0)

    B = np.array(B_rows)
    rhs = np.array(rhs)

    if not HAS_CVXPY:
        return None

    d_var = cp.Variable(n)
    constraints = [cp.sum(d_var) == 0, B @ d_var <= rhs]
    objective = cp.Minimize(cp.sum_squares(d_var) + reg_strength * cp.sum_squares(d_var))
    # Actually just use sum_squares with effective regularization
    objective = cp.Minimize((1.0 + reg_strength) * cp.sum_squares(d_var))
    prob = cp.Problem(objective, constraints)

    for solver in ['ECOS', 'SCS']:
        if time.monotonic() > deadline:
            return None
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return d_var.value.copy()
        except Exception:
            continue
    return None

def _peak_directed_descent(a_init, deadline, min_time_remaining=20.0):
    """Fast peak-directed descent: directly reduce top peaks via
    multiplicative/subgradient updates with backtracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(best_a, best_a).max())
    t_start = time.monotonic()

    lr = 0.1
    top_k = max(5, int(n * 0.01))
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_time_remaining < deadline:
        if no_improve >= max_no_improve:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.5, lr * 1.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.7

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]

        # Build pressure vector: weighted sum of "influence" of peaks
        # For peak at k, the gradient contribution to a[j] is 2*a[k-j] (if valid)
        # Weight by peak value
        total_weight = float(top_vals.sum())
        if total_weight < 1e-30:
            break

        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            w = top_vals[idx] / total_weight
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                if 0 <= j < n:
                    grad[j] += w * 2.0 * float(a[i])

        # Center the gradient to preserve sum
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search
        improved = False
        alpha = lr / gnorm
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 5
            # Try multiplicative update as fallback
            a_try = a * np.exp(-0.01 * (grad - np.mean(grad)))
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / a_try.sum()
            c_try = _full_convolve(a_try, a_try)
            if 2.0 * n * float(c_try.max()) < current_sc - 1e-14:
                a = a_try
                no_improve = 0

    return best_a, best_sc


def _symmetry_break(a, epsilon, rng):
    """Add a small symmetry-breaking perturbation."""
    n = len(a)
    a_new = a.copy()
    # Add first and second harmonic perturbation
    t = np.arange(n)
    pert = epsilon * (np.sin(2.0 * np.pi * t / n) + 0.5 * np.sin(2.0 * np.pi * 2.0 * t / n))
    # Also add small random perturbation
    pert += epsilon * 0.1 * rng.standard_normal(n)
    a_new = a_new * (1.0 + pert)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new


def _check_symmetry(a):

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "strong_u":
        t = np.linspace(0, 1, n)
        a = 0.3 + np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "w_shape":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(12345)
        a = rng.standard_normal(n)
        # Smooth with a wide kernel
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    """Main search function combining multiple optimization strategies."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
                best_seq = parent.copy()
                best_score = parent_sc
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "u_shape")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, u_shape n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS from parent ===
    print(f"\n[P1] L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200, time.monotonic() + 300)
    a_p1 = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=300.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_parent] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc_p1:.10f} vs {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Symmetry breaking ===
    print(f"\n[P2] Symmetry analysis and breaking (t={time.monotonic()-t_start:.1f}s)")
    if _check_symmetry(best_seq):
        print(f"  Sequence is symmetric! Breaking symmetry...")
        a_sym_break = _symmetry_break(best_seq, 0.01, rng)
        sc_sym = _score(a_sym_break)
        print(f"  After break: score={sc_sym:.10f}")
        # Optimize the broken sequence
        p2_deadline = min(deadline - 900, time.monotonic() + 200)
        a_p2 = _run_lbfgs(a_sym_break, p2_deadline, min_time_remaining=200.0)
        sc_p2 = _score(a_p2)
        if sc_p2 < best_score:
            best_seq = a_p2.copy()
            best_score = sc_p2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [sym_break] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Sequence is asymmetric, skipping symmetry break.")

    # === Phase 3: Diverse initializations ===
    print(f"\n[P3] Diverse initializations (t={time.monotonic()-t_start:.1f}s)")
    n_try = len(best_seq)
    profiles = ["u_shape", "strong_u", "w_shape", "bimodal", "random_smooth"]
    for prof_name in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_init = _make_profile(n_try, prof_name)
        sc_init = _score(a_init)
        if sc_init >= best_score * 1.05:
            continue  # Skip if too far from best
        p3_deadline = min(deadline - 700, time.monotonic() + 80)
        a_opt = _run_lbfgs(a_init, p3_deadline, min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [{prof_name}] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  {prof_name}: {sc_opt:.10f} (no improvement)")

    # === Phase 4: QP peak reduction ===
    print(f"\n[P4] QP peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p4_deadline = min(deadline - 400, time.monotonic() + 300)
        a_qp, sc_qp = _qp_phase(best_seq, p4_deadline, min_time_remaining=100.0)
        if sc_qp < best_score:
            best_seq = a_qp.copy()
            best_score = sc_qp
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [qp] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  QP: no improvement (best={best_score:.10f})")

    # === Phase 5: Peak-directed descent ===
    print(f"\n[P5] Peak-directed descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p5_deadline = min(deadline - 200, time.monotonic() + 200)
        a_pd, sc_pd = _peak_directed_descent(best_seq, p5_deadline, min_time_remaining=50.0)
        if sc_pd < best_score:
            best_seq = a_pd.copy()
            best_score = sc_pd
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_directed] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Peak-directed: no improvement (best={best_score:.10f})")

    # === Phase 6: Active set polish ===
    print(f"\n[P6] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p6_deadline = min(deadline - 100, time.monotonic() + 150)
        a_pol, sc_pol = _active_set_polish(best_seq, p6_deadline,
                                           tol_frac_start=0.003, tol_frac_end=0.00005,
                                           lr_init=0.03)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Second round L-BFGS + QP ===
    print(f"\n[P7] Second optimization round (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        p7_deadline = min(deadline - 80, time.monotonic() + 80)
        a_p7 = _run_lbfgs(best_seq, p7_deadline, min_time_remaining=60.0)
        sc_p7 = _score(a_p7)
        if sc_p7 < best_score:
            best_seq = a_p7.copy()
            best_score = sc_p7
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_r2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 32768, 8192, 12288]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    # Remove duplicates, keep order
    seen = set()
    n_candidates_unique = []
    for nc in n_candidates:
        if nc not in seen:
            seen.add(nc)
            n_candidates_unique.append(nc)
    n_candidates = n_candidates_unique

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) and n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if time.monotonic() < deadline - 60:
            p8_deadline = min(deadline - 55, time.monotonic() + 60)
            a_opt = _run_lbfgs(a_init, p8_deadline, min_time_remaining=40.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 30:
                a_pol2, sc_pol2 = _active_set_polish(a_opt, min(deadline - 28, time.monotonic() + 25),
                                                     tol_frac_start=0.002, tol_frac_end=0.00005,
                                                     lr_init=0.015)
                if sc_pol2 < best_score:
                    best_seq = a_pol2.copy()
                    best_score = sc_pol2
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Shrinkage ===
    print(f"\n[P9] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 50)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize
            if time.monotonic() < deadline - 30:
                a_shr2 = _run_lbfgs(best_seq, min(deadline - 25, time.monotonic() + 30), min_time_remaining=20.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 5,
                                           tol_frac_start=0.001, tol_frac_end=0.00002,
                                           lr_init=0.005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.0001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(30, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # Peak-directed final
        if time.monotonic() < deadline - 8:
            a_pd2, sc_pd2 = _peak_directed_descent(best_seq, deadline - 6, min_time_remaining=5.0)
            if sc_pd2 < best_score:
                best_seq = a_pd2.copy()
                best_score = sc_pd2
                print(f"  NEW BEST: {best_score:.10f} [final_peak_directed] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
