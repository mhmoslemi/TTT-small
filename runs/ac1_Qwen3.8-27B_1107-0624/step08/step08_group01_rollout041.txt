    while k < n:

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:

    e = np.exp(np.clip(z, -700, 700))

    if s < 1e-300:

"""
Optimized step-function search using asymmetric monotone ramp initializations,
LSE-smoothed L-BFGS optimization, stochastic mirror descent, and multi-scale
refinement. Key insight: asymmetric (monotone) ramps give much lower initial
scores than symmetric profiles, enabling faster convergence below 1.4.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _linear_ramp(n, power=1.0):
    """Create a monotone increasing ramp: a[i] proportional to (i+1)^power."""
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _quadratic_ramp(n):
    """Quadratic ramp: a[i] proportional to (i+1)^2."""
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t * t
    return a / a.sum()


def _log_ramp(n):
    """Logarithmic ramp: a[i] proportional to log(i+2)."""
    t = np.arange(2, n + 2, dtype=np.float64)
    a = np.log(t)
    return a / a.sum()


def _exp_ramp(n, rate=3.0):
    """Exponential ramp: a[i] proportional to exp(rate * i / n)."""
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _sine_ramp(n):
    """Sine-weighted ramp: a[i] proportional to sin(pi * (i+1) / (n+1))^2."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _two_cluster(n, frac=0.5):
    """Two-cluster: mass at left and right ends."""
    a = np.zeros(n)
    m = max(1, int(n * frac))
    a[:m] = 1.0
    a[-m:] = 1.0
    return a / a.sum()


def _cosine_mod_ramp(n):
    """Ramp with cosine modulation for texture."""
    t = np.arange(n, dtype=np.float64)
    base = t / n
    mod = 1.0 + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
    a = np.maximum(base * mod, 1e-10)
    return a / a.sum()


def _make_init(n, kind="linear"):
    if kind == "linear":
        return _linear_ramp(n)
    elif kind == "quad":
        return _quadratic_ramp(n)
    elif kind == "log":
        return _log_ramp(n)
    elif kind == "exp":
        return _exp_ramp(n)
    elif kind == "sine":
        return _sine_ramp(n)
    elif kind == "two_cluster":
        return _two_cluster(n)
    elif kind == "cos_mod":
        return _cosine_mod_ramp(n)
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_remaining=15.0):
    """L-BFGS with aggressive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0002,
         0.0001, 0.00005, 0.00002, 0.00001]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.003 * scale
        max_iter = min(8000, max(50, int(remaining / (est_iter_time * 12))))

        prob = LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _stochastic_mirror_descent(a_init, deadline, min_remaining=10.0):
    """Stochastic mirror descent with multiplicative updates and peak cycling."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    eta = 0.05
    ko = 0

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Select peak to attack
        r = rng.random()
        if r < 0.4:
            # Global max
            k_sel = int(np.argmax(c))
        elif r < 0.7:
            # Random from top 10%
            threshold = np.percentile(c, 90)
            candidates = np.where(c >= threshold)[0]
            if len(candidates) == 0:
                k_sel = int(np.argmax(c))
            else:
                k_sel = int(candidates[rng.integers(len(candidates))])
        else:
            # Random valid lag
            k_sel = int(rng.integers(0, 2 * n - 1))

        # Compute gradient: grad[i] = 2 * a[k_sel - i] for valid i
        k_lo = max(0, k_sel - n + 1)
        k_hi = min(n - 1, k_sel)
        if k_lo > k_hi:
            # No valid terms (shouldn't happen for 0 <= k_sel <= 2n-2)
            break

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_sel - i_range
        contrib = a[j_range]  # a[k_sel - i] for each i in range

        # Multiplicative update: a[i] *= exp(-eta * contrib_i / a[i])
        # This is equivalent to log-space gradient step
        # To avoid numerical issues, use: a_new[i] = a[i] * exp(-eta * contrib_i / max(a[i], eps))
        eps = 1e-15
        log_update = np.zeros(n)
        log_update[i_range] = -eta * contrib / np.maximum(a[i_range], eps)
        log_update = np.clip(log_update, -5.0, 5.0)  # Clip to avoid overflow
        log_update -= log_update.mean()  # Center to preserve sum approximately

        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new < current_sc - 1e-14:
            a = a_new
            eta = min(eta * 1.05, 0.5)
        else:
            eta *= 0.6

        ko += 1
        if ko % 500 == 0 and time.monotonic() > deadline - min_remaining - 5:
            break

    return best_a


def _direct_peak_shaving(a_init, deadline, n_trials=200):
    """Direct peak shaving: identify top contributors to the peak and redistribute."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a = best_a.copy()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo > k_hi:
            continue

        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total = contrib.sum()
        if total < 1e-15:
            continue

        # Reduce top contributors
        frac = rng.uniform(0.02, 0.1)
        n_reduce = max(1, int(len(contrib) * frac))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * 0.15
            a_try[idx] -= red
            reduction += red

        # Distribute to low-contributing elements
        low_mask = np.ones(n, dtype=bool)
        low_mask[top_idx] = False
        # Also exclude elements that pair with top_idx at the peak
        for idx in top_idx:
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False

        if low_mask.sum() == 0:
            low_mask = np.ones(n, dtype=bool)
            low_mask[top_idx] = False
            if low_mask.sum() == 0:
                continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _active_set_polish(a_init, deadline, min_remaining=10.0,
                       tol_start=0.005, tol_end=0.00001, lr_start=0.03, top_k=5):
    """Active-set GD with top-k peaks and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    max_stall = 200

    while time.monotonic() < deadline - min_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 10:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.4
                no_improve = 0
                if tol < tol_end * 1.1 and lr < lr_start * 0.1:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.98)
            lr = min(lr_start, lr * 1.02)

    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using asymmetric ramp initializations,
    LSE+L-BFGS, stochastic mirror descent, active-set polishing,
    and peak shaving. Targets score below 1.4.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    best_a = None
    best_score = np.inf

    # === Evaluate initial candidates quickly ===
    print(f"=== Initial candidate evaluation ===", flush=True)

    # Try linear ramps at various n
    init_configs = [
        ("ramp_12288", 12288, "linear"),
        ("ramp_16384", 16384, "linear"),
        ("ramp_8192", 8192, "linear"),
        ("exp_12288", 12288, "exp"),
        ("quad_12288", 12288, "quad"),
        ("log_12288", 12288, "log"),
    ]

    if parent is not None:
        init_configs.append(("parent", len(parent), "parent"))

    candidates = []
    for name, n_try, kind in init_configs:
        if name == "parent" and parent is not None:
            a_cand = parent.copy()
        else:
            a_cand = _make_init(n_try, kind)
        sc = _score(a_cand)
        candidates.append((name, a_cand, sc))
        print(f"  {name}: n={len(a_cand)}, score={sc:.10f}", flush=True)

    # Sort by score
    candidates.sort(key=lambda x: x[2])
    best_a = candidates[0][1].copy()
    best_score = candidates[0][2]
    print(f"\nBest initial: {candidates[0][0]} with score {best_score:.10f}", flush=True)

    # === Phase 1: L-BFGS on best candidate ===
    print(f"\n=== [P1] L-BFGS optimization (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
    p1_deadline = min(deadline - 600.0, t_start + 400.0)
    a1 = _run_lbfgs(best_a, p1_deadline, min_remaining=30.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_p1]", flush=True)

    # === Phase 2: Stochastic mirror descent ===
    print(f"\n=== [P2] Stochastic mirror descent (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
    p2_deadline = min(deadline - 400.0, t_start + 600.0)
    a2 = _stochastic_mirror_descent(best_a, p2_deadline, min_remaining=20.0)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [mirror_p2]", flush=True)

    # === Phase 3: L-BFGS again (finer) ===
    print(f"\n=== [P3] L-BFGS refine (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
    p3_deadline = min(deadline - 300.0, t_start + 800.0)
    a3 = _run_lbfgs(best_a, p3_deadline, min_remaining=20.0)
    sc3 = _score(a3)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_p3]", flush=True)

    # === Phase 4: Active-set polish ===
    print(f"\n=== [P4] Active-set polish (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
    p4_deadline = min(deadline - 200.0, t_start + 1000.0)
    a4, sc4 = _active_set_polish(best_a, p4_deadline, min_remaining=20.0,
                                  tol_start=0.003, tol_end=0.00001, lr_start=0.02)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [polish_p4]", flush=True)

    # === Phase 5: Peak shaving ===
    print(f"\n=== [P5] Peak shaving (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
    p5_deadline = min(deadline - 150.0, t_start + 1200.0)
    a5, sc5 = _direct_peak_shaving(best_a, p5_deadline, n_trials=300)
    if sc5 < best_score:
        best_a = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} [shave_p5]", flush=True)

    # === Phase 6: Try second-best initial with full pipeline ===
    # If the second-best initial is close, try optimizing it too
    if len(candidates) > 1 and candidates[1][2] < best_score + 0.1:
        if time.monotonic() < deadline - 200.0:
            print(f"\n=== [P6] Optimizing second-best init: {candidates[1][0]} (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
            a6 = _run_lbfgs(candidates[1][1], min(deadline - 100.0, time.monotonic() + 150.0), min_remaining=20.0)
            sc6 = _score(a6)
            if sc6 < best_score:
                best_a = a6.copy()
                best_score = sc6
                print(f"  NEW BEST: {best_score:.10f} [second_init]", flush=True)

            if time.monotonic() < deadline - 60.0:
                a6b, sc6b = _active_set_polish(a6, min(deadline - 30.0, time.monotonic() + 60.0),
                                                min_remaining=10.0, tol_start=0.002,
                                                tol_end=0.00001, lr_start=0.015)
                if sc6b < best_score:
                    best_a = a6b.copy()
                    best_score = sc6b
                    print(f"  NEW BEST: {best_score:.10f} [second_init_polish]", flush=True)

    # === Phase 7: Diverse restarts ===
    print(f"\n=== [P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s) ===", flush=True)
    if time.monotonic() < deadline - 100.0:
        n_best = len(best_a)
        restarts = []

        # Perturbed versions of best
        restarts.append(("pert_small", _perturb_log(best_a, 0.02, rng)))
        restarts.append(("pert_med", _perturb_log(best_a, 0.05, rng)))

        # Different ramp types at same n
        restarts.append(("exp_restart", _make_init(n_best, "exp")))
        restarts.append(("quad_restart", _make_init(n_best, "quad")))
        restarts.append(("cos_mod_restart", _make_init(n_best, "cos_mod")))

        # Try a different n
        for n_alt in [n_best * 2, n_best // 2]:
            if 1024 <= n_alt <= 32768:
                restarts.append((f"ramp_{n_alt}", _make_init(n_alt, "linear")))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 50.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 20.0:
                break
            a_opt = _run_lbfgs(a_r, deadline - 40.0, min_remaining=15.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [restart_{label}]", flush=True)

    # === Phase 8: Final ultra-fine polishing ===
    print(f"\n=== [P8] Final polish (t={time.monotonic()-t_start:.1f}s) ===", flush=True)

        a_fin, sc_fin = _active_set_polish(best_a, deadline - 3.0,
                                            min_remaining=3.0, tol_start=0.001,
                                            tol_end=0.000005, lr_start=0.005, top_k=3)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]", flush=True)
