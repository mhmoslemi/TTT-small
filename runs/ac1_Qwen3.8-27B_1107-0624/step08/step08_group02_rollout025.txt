Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) Adaptive log-sum-exp smoothed L-BFGS, (2) Active-set Newton with CG,

Key improvements: KKT-residual-based active set verification, more aggressive beta

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Restart
            p = r.copy()
            continue
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            p = r.copy()
            continue
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

        # Adaptive active set: start tight, relax if needed
        active = None
        for tol_frac in [1e-6, 5e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 2:
                break
        if active is None or active.sum() < 1:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        n_active = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            if verbose:
                print(f"    Newton: converged at round {rnd}, grad_norm={gn:.2e}")
            break

        # Solve H dx = -g_t with constraint 1^T dx = 0
        Hg = _cg_solve(W, n, g, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with Armijo condition
        alpha = 1.0
        step_done = False
        c_max_start = c_max
        for ls in range(100):
            a_try = a + alpha * dx
            if np.any(a_try < 0):
                # Find max alpha that keeps a >= 0
                neg_mask = dx < 0
                if np.any(neg_mask):
                    max_alpha = np.min(a[neg_mask] / (-dx[neg_mask]))
                    alpha = max(0, max_alpha) * 0.95
                else:
                    alpha *= 0.5
                continue
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo: c_max_try < c_max_start - c*alpha*gn
            if c_max_try < c_max_start - 1e-4 * alpha * gn / n:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.15, max_c * 0.0005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_a and current_sc
    prev_sc = current_sc
    iterations_without_improvement = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(15000, max(20, int(remaining / (est_time_per_iter * 20))))

        try:
            obj_fn = _ObjAD(beta, n)
            grad_fn = _GradAD(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                iterations_without_improvement = 0
            else:
                iterations_without_improvement += 1
        except Exception:
            pass

        # Adaptive beta reduction - more aggressive
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-14:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        # Additional reduction if no improvement
        if iterations_without_improvement >= 2:
            beta *= 0.5

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _peak_shaving(a_init, deadline, max_rounds=5000):
    """
    Targeted subgradient descent focusing on the peak convolution value.
    Uses adaptive step sizes and periodic re-normalization.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    lr = 0.005
    patience = 0

    for i in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max_pos = int(np.argmax(c))
        c_max = float(c[c_max_pos])

        # Subgradient at the peak
        # d(c[p])/da[k] = a[p-k] (for valid p-k) + a[k-p] (for valid k-p)
        # = 2*a[|k - something|] pattern... actually:
        # c[p] = sum_k a[k]*a[p-k], so dc[p]/da[k] = a[p-k] (if 0<=p-k<n)
        # The gradient of max c w.r.t. a is just the gradient at the argmax
        k_arr = np.arange(n)
        # dc[p]/da[k] = a[p-k] if 0 <= p-k < n, else 0
        idx = c_max_pos - k_arr
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = a[idx[valid]]
        # Also: c[p] includes a[k]*a[p-k], so derivative w.r.t. a[k] is a[p-k]
        # But we also need to account for the symmetric term if p-k != k
        # Actually for c = a*a (full conv), c[p] = sum_{k=max(0,p-n+1)}^{min(p,n-1)} a[k]*a[p-k]
        # dc[p]/da[k] = a[p-k] for k in valid range (by product rule, both terms give a[p-k])
        # Wait no: c[p] = sum_k a[k]*a[p-k]. dc[p]/da[j] = a[p-j] (from the first term where k=j)
        # + a[j-p] (from the second term where p-k=j, i.e., k=p-j) ... 
        # Actually c[p] = sum_k a[k]*a[p-k]. Taking derivative w.r.t. a[j]:
        # = a[p-j] (when k=j) ... but also when p-k=j, i.e., k=p-j, we get a[p-j]
        # So dc[p]/da[j] = a[p-j] if 0<=p-j<n (from k=j term) + a[p-j] if 0<=p-j<n (from k=p-j term)
        # Wait, these are the same condition. If p-j is valid, both terms contribute a[p-j].
        # But if j = p-j (i.e., j = p/2), it's just one term.
        # Actually no: c[p] = sum_k a[k] * a[p-k]. The partial w.r.t. a[j]:
        # Term 1: when k=j, contribution is a[p-j]
        # Term 2: when p-k=j, i.e., k=p-j, contribution is a[p-j] (since a[k]=a[p-j])
        # But these are the same term when j = p-j. So:
        # dc[p]/da[j] = a[p-j] + a[p-j] = 2*a[p-j] if j != p-j
        #              = a[p-j] if j == p-j (i.e., 2j = p)
        # Hmm wait, that's not right either. Let me be more careful.
        # c[p] = sum_{k=0}^{n-1} a[k] * a[p-k] (where a[.] = 0 outside [0,n-1])
        # dc[p]/da[j] = a[p-j] (from the k=j term) + a[j] * 0 ... 
        # No wait: c[p] = sum_k a[k]*a[p-k]. This is a bilinear form.
        # dc[p]/da[j] = sum_k [d(a[k])/da[j] * a[p-k] + a[k] * d(a[p-k])/da[j]]
        # = a[p-j] + a[p-j]  (the first term is nonzero when k=j, the second when p-k=j)
        # = 2*a[p-j] if j != p-j, or a[p-j] if j = p-j
        # But actually for the case j = p-j: the term a[j]*a[j] has derivative 2*a[j] = 2*a[p-j]
        # So in all cases: dc[p]/da[j] = 2*a[p-j] (with a[x]=0 for x outside [0,n-1])
        # Wait no, that's for the case where we treat a[k] and a[p-k] as independent.
        # But they're the SAME array! So dc[p]/da[j] = a[p-j] (from k=j) + a[p-j] (from k=p-j)
        # If j = p-j: it's just a[j]*a[j], derivative is 2*a[j] = 2*a[p-j]. ✓
        # If j != p-j: it's a[j]*a[p-j] + a[p-j]*a[j] = 2*a[j]*a[p-j], derivative w.r.t. a[j] is 2*a[p-j]. ✓
        # So dc[p]/da[j] = 2*a[p-j] for all j (with a[x]=0 outside range).
        
        # I was right the first time but forgot the factor of 2 in some cases.
        # Let me just use: grad[j] = a[p-j] if valid + a[j-p] if valid
        # which equals 2*a[|p-j|] adjusted... no let me just do it directly.
        
        # Redo: dc[p]/da[j] = a[p-j] + a[j-p] where we define a[x]=0 for x<0 or x>=n
        idx2 = k_arr - c_max_pos  # j - p
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad = np.zeros(n)
        grad[valid] += a[idx[valid]]
        grad[valid2] += a[idx2[valid2]]
        # This gives dc[p]/da[j] = a[p-j] (if valid) + a[j-p] (if valid)

        grad_t = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_t)
        if grad_norm < 1e-14:
            break

        z = z - lr * grad_t
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Periodic evaluation
        if i % 50 == 49:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-13:
                best_sc = sc_check
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.02)
            else:
                patience += 1
                lr *= 0.8
                if patience > 100:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.003

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.0 + 0.01
        return a / a.sum()
    elif kind == "flat_center":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = max(1, n // 10)
        for i in range(edge):
            a[i] = 0.5 + 0.5 * (i / edge)
            a[n-1-i] = 0.5 + 0.5 * (i / edge)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick profile search (120s) ===
    print(f"\n[Phase 1] Quick profile search (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + 120
    profiles = ["tapered", "flat_center", "asym_taper", "cos", "left_heavy"]
    best_init = parent.copy()
    best_init_sc = _score(parent)

    for pk in profiles:
        if time.monotonic() > p1_deadline - 30:
            break
        ap = _make_profile(n_parent, pk)
        sp = _score(ap)
        print(f"  Profile '{pk}': {sp:.10f}")
        if sp < best_init_sc:
            best_init_sc = sp
            best_init = ap

    # Quick L-BFGS on best init
    if time.monotonic() < p1_deadline - 10:
        a_q = _run_lbfgs(best_init, p1_deadline, min_time_remaining=5)
        sq = _score(a_q)
        update_best(a_q, sq, "quick_lbfgs")
        if sq < best_init_sc:
            best_init = a_q
            best_init_sc = sq

    # === Phase 2: Main L-BFGS (500s) ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    p2_deadline = min(t_start + 620, deadline - 500)
    start_a = best_seq if best_seq is not None else best_init
    a_main = _run_lbfgs(start_a, p2_deadline, min_time_remaining=20)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: L-BFGS + Newton cycling (400s) ===
    print(f"\n[Phase 3] L-BFGS + Newton cycling (t={time.monotonic() - t_start:.1f}s)")
    p3_deadline = min(t_start + 1020, deadline - 400)
    current_a = best_seq if best_seq is not None else a_main
    current_sc = _score(current_a)
    cycle_time = 60.0
    n_cycles = 0

    while time.monotonic() < p3_deadline and n_cycles < 8:
        t_end = min(time.monotonic() + cycle_time, p3_deadline)
        if t_end - time.monotonic() < 10:
            break

        # Short L-BFGS burst
        if time.monotonic() < t_end - 20:
            a_lb = _run_lbfgs(current_a, min(t_end - 20, p3_deadline), min_time_remaining=15)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc:
                current_a = a_lb
                current_sc = sc_lb
                update_best(current_a, current_sc, f"cycle_{n_cycles}_lbfgs")

        # Newton polish
        if time.monotonic() < p3_deadline - 5:
            a_np, sc_np = _newton_polish(current_a, p3_deadline, max_rounds=50)
            if sc_np < current_sc:
                current_a = a_np
                current_sc = sc_np
                update_best(current_a, current_sc, f"cycle_{n_cycles}_newton")

        cycle_time = min(cycle_time * 1.3, 120)
        n_cycles += 1

    # === Phase 4: Peak shaving + Newton (300s) ===
    print(f"\n[Phase 4] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = min(t_start + 1320, deadline - 300)
    if time.monotonic() < p4_deadline:
        a_ps, sc_ps = _peak_shaving(current_a, min(p4_deadline, deadline - 200), max_rounds=3000)
        if sc_ps < current_sc:
            current_a = a_ps
            current_sc = sc_ps
            update_best(current_a, current_sc, "peak_shave")

    # Newton after peak shaving
    if time.monotonic() < p4_deadline:
        a_np, sc_np = _newton_polish(current_a, min(p4_deadline, deadline - 100), max_rounds=80)
        if sc_np < current_sc:
            current_a = a_np
            current_sc = sc_np
            update_best(current_a, current_sc, "post_ps_newton")

    # === Phase 5: Multi-n exploration (150s) ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = min(t_start + 1470, deadline - 200)
    n_tries = [32768, 8192]
    for n_t in n_tries:
        if time.monotonic() > p5_deadline - 30:
            break
        a_init_n = _resample(current_a, n_t)
        a_opt_n = _run_lbfgs(a_init_n, min(p5_deadline, deadline - 100), min_time_remaining=30)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_t}_lbfgs")

        if sc_opt_n < best_score + 0.005 and time.monotonic() < p5_deadline - 10:
            a_np_n, sc_np_n = _newton_polish(a_opt_n, min(p5_deadline, deadline - 50), max_rounds=40)
            update_best(a_np_n, sc_np_n, f"n_{n_t}_newton")

    # === Phase 6: Diverse restarts (100s) ===
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = min(t_start + 1570, deadline - 150)
    if best_seq is not None:
        for i in range(3):
            if time.monotonic() > p6_deadline - 20:
                break
            sigma = 0.03 + 0.03 * i
            a_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(a_pert, min(p6_deadline, deadline - 50), min_time_remaining=20)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")
            if sc_r < best_score + 0.003 and time.monotonic() < p6_deadline - 10:
                a_rp, sc_rp = _newton_polish(a_r, p6_deadline, max_rounds=30)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    # === Phase 7: Final ultra-fine Newton (remaining time) ===
    print(f"\n[Phase 7] Final ultra-fine Newton (t={time.monotonic() - t_start:.1f}s)")
    final_deadline = deadline - 2
    if best_seq is not None:
        # First, a final L-BFGS with very small beta
        if time.monotonic() < final_deadline - 60:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-10
            remaining = final_deadline - 60 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(20, int(remaining / 0.01)))
                try:
                    obj_fn = _ObjAD(beta_fin, n)
                    grad_fn = _GradAD(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100, 'maxls': 400}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_small_beta")
                except Exception:
                    pass

        # Extended Newton until deadline
        if best_seq is not None and time.monotonic() < final_deadline - 1:
            a_fin, sc_fin = _newton_polish(best_seq, final_deadline, max_rounds=300, verbose=True)
            update_best(a_fin, sc_fin, "final_newton")

            # One more round if still improving
            if time.monotonic() < final_deadline - 0.5:
                a_veryfin, sc_veryfin = _newton_polish(best_seq, final_deadline, max_rounds=100)
                update_best(a_veryfin, sc_veryfin, "ultra_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
