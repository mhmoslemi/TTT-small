import time
from scipy.optimize import minimize

# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    N = next_pow2(N)  # Ensure N is power of 2 for FFT
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        # Handle sum constraint: adjust dx to have zero mean
        lam = np.mean(dx)
        dx = dx - lam

        # Line search with non-negativity
        f_old = c_max
        accepted = False
        t = 1.0
        for ls in range(50):
            a_try = a + t * dx
            if (a_try < -1e-12).any():
                neg = a_try < 0
                if neg.any():
                    t = max(1e-10, 0.95 * np.min(a[neg] / (-dx[neg])))
                else:
                    t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-13:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Gradient fallback
            t = 0.05
            for ls in range(30):
                a_try = np.maximum(a - t * g_t, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc

def _run_lbfgs(a_init, deadline, min_remaining=60.0, aggressive=False):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        multipliers = [1.0, 0.7, 0.5, 0.35, 0.25, 0.17, 0.12, 0.08, 0.05, 0.035,
                      0.025, 0.017, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001,
                      0.0006, 0.0003]
    else:
        multipliers = [1.0, 0.6, 0.4, 0.25, 0.15, 0.09, 0.055, 0.035, 0.02, 0.012,
                      0.007, 0.004, 0.002, 0.001, 0.0005]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, mult in enumerate(multipliers):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * mult

        # Estimate iterations based on time
        if n <= 8192:
            est_time = 0.004
        elif n <= 16384:
            est_time = 0.008
        elif n <= 32768:
            est_time = 0.02
        else:
            est_time = 0.04

        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        try:
            t0 = time.monotonic()
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                elapsed = time.monotonic() - t0
                print(f"    beta={beta:.2e}: score={sc_new:.10f} ({elapsed:.1f}s)")
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.2e}: {e}")
            break

    return current_a

def _peak_refine(a_init, deadline, max_rounds=80):
    """Target top peaks with weighted subgradient steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive number of peaks
        n_peaks = min(20, max(3, n // 200))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        w = np.zeros_like(c)
        # Weight peaks by how close they are to max
        for k in top_idx:
            w[k] = max(0.01, (c[k] - c_max * 0.9) / max(c_max * 0.1, 1e-15))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w[top_idx[0]] = 1.0
            w_sum = 1.0
        w /= w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Backtracking line search
        f_old = c_max
        step = 0.1 / (1.0 + 0.02 * rnd)
        accepted = False
        for ls in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-15:
                a = a_try
                accepted = True
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        if not accepted:
            break

    return best_a, best_sc

def _subgrad_adaptive(a_init, deadline, max_steps=5000, lr_init=0.04):
    """Adaptive subgradient with momentum in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    momentum = np.zeros(n)
    beta_m = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top-k peaks with smooth weighting
        k = min(30, max(5, n // 500))
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        peak_vals = c[top_k]
        weights = np.exp(-(c_max - peak_vals) / (0.01 * c_max + 1e-15))
        w[top_k] = weights / weights.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Momentum update
        momentum = beta_m * momentum - lr * grad
        z = z + momentum
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.6

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.arange(n) / (n - 1)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp_up":
        t = np.linspace(0.1, 1.0, n)
        return t / t.sum()
    elif kind == "ramp_down":
        t = np.linspace(1.0, 0.1, n)
        return t / t.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.03) + np.exp(-((t - 0.7) ** 2) / 0.03) + 0.1
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(123)
        a = rng.standard_normal(n)
        k = max(3, n // 80)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp continuation,
    active-set Newton, peak-targeted refinement, and diverse restarts.
    Explores multiple sequence lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "hann")
        print(f"No parent. Hann init n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "hann_init")

    # --- Phase 1: Aggressive L-BFGS from parent ---
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    t_phase1_end = deadline - 600
    if time.monotonic() < t_phase1_end:
        a1 = _run_lbfgs(parent, t_phase1_end, min_remaining=300, aggressive=True)
        update_best(a1, _score(a1), "lbfgs_p1")

        # Symmetrize and re-optimize
        a1s = _symmetrize(a1)
        sc1s = _score(a1s)
        print(f"  Symmetrized: score={sc1s:.10f}")
        if sc1s < best_score + 0.001:
            a1s_opt = _run_lbfgs(a1s, t_phase1_end, min_remaining=60, aggressive=True)
            update_best(a1s_opt, _score(a1s_opt), "lbfgs_sym_p1")

    # --- Phase 2: Active-set Newton on best ---
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_as = None, np.inf
        t_as = min(deadline - 200, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, t_as, max_iter=300, tol_frac=0.001)
        update_best(a_as, sc_as, "as_newton")

        # Tighter
        if time.monotonic() < deadline - 150:
            a_as2, sc_as2 = _active_set_newton(a_as, min(deadline - 100, time.monotonic() + 80),
                                                max_iter=200, tol_frac=0.0003)
            update_best(a_as2, sc_as2, "as_tight")

    # --- Phase 3: Peak-targeted refinement ---
    print(f"\n[Phase 3] Peak refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_pf, sc_pf = _peak_refine(best_seq, min(deadline - 150, time.monotonic() + 100), max_rounds=100)
        update_best(a_pf, sc_pf, "peak_refine")

        # Subgradient with momentum
        if time.monotonic() < deadline - 100:
            a_sg, sc_sg = _subgrad_adaptive(best_seq, min(deadline - 50, time.monotonic() + 60),
                                            max_steps=3000, lr_init=0.03)
            update_best(a_sg, sc_sg, "subgrad_mom")

    # --- Phase 4: Diverse initializations ---
    print(f"\n[Phase 4] Diverse inits (t={time.monotonic()-t_start:.1f}s)")
    n_div = n_parent
    profiles = ["hann", "blackman", "exp_ramp", "bimodal", "sine_mod", "random_smooth"]
    for kind in profiles:
        if time.monotonic() > deadline - 150:
            break
        a_init = _make_profile(n_div, kind, rng)
        sc_init = _score(a_init)
        print(f"  {kind}: init={sc_init:.10f}")

        # Only optimize if promising
        if sc_init < best_score + 0.15:
            a_opt = _run_lbfgs(a_init, deadline - 100, min_remaining=30, aggressive=True)
            update_best(a_opt, _score(a_opt), f"div_{kind}")

            if time.monotonic() < deadline - 40:
                a_as, sc_as = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 40),
                                                  max_iter=100, tol_frac=0.001)
                update_best(a_as, sc_as, f"div_{kind}_as")

    # --- Phase 5: Try larger n ---
    print(f"\n[Phase 5] Larger n (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_large = 32768
        a_large_init = _interpolate_to_n(best_seq, n_large)
        sc_large = _score(a_large_init)
        print(f"  n={n_large} interpolated: score={sc_large:.10f}")
        update_best(a_large_init, sc_large, f"interp_n{n_large}")

        if time.monotonic() < deadline - 80:
            a_large_opt = _run_lbfgs(a_large_init, deadline - 60, min_remaining=30, aggressive=True)
            update_best(a_large_opt, _score(a_large_opt), f"lbfgs_n{n_large}")

            if time.monotonic() < deadline - 30:
                a_large_as, sc_lar_as = _active_set_newton(a_large_opt,
                                                            min(deadline - 10, time.monotonic() + 30),
                                                            max_iter=80, tol_frac=0.001)
                update_best(a_large_as, sc_lar_as, f"as_n{n_large}")

    # --- Phase 6: Try smaller n (might find better per-n ratio) ---
    print(f"\n[Phase 6] Smaller n (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_small = 8192
        a_small = _interpolate_to_n(best_seq, n_small)
        sc_small = _score(a_small)
        print(f"  n={n_small}: score={sc_small:.10f}")
        update_best(a_small, sc_small, f"interp_n{n_small}")

        if time.monotonic() < deadline - 40:
            a_small_opt = _run_lbfgs(a_small, deadline - 30, min_remaining=10, aggressive=True)
            update_best(a_small_opt, _score(a_small_opt), f"lbfgs_n{n_small}")

    # --- Phase 7: Perturbation restarts on best ---
    print(f"\n[Phase 7] Perturbations (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        for sigma in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline - 15:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax(z)
            sc_pert = _score(a_pert)

            if sc_pert < best_score + 0.01:
                a_pert_opt = _run_lbfgs(a_pert, deadline - 10, min_remaining=5, aggressive=False)
                update_best(a_pert_opt, _score(a_pert_opt), f"perturb_{sigma}")

    # --- Phase 8: Final active-set polish ---
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 3, max_iter=100, tol_frac=0.0001)
        update_best(a_fin, sc_fin, "final_as")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
