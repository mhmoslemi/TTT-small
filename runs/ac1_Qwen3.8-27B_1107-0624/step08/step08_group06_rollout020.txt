def _spectral_perturb(a, rng, strength=0.1):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Perturb the magnitude in frequency domain
    phase = np.angle(A)
    mag = np.abs(A)
    noise = 1.0 + strength * rng.standard_normal(len(mag))
    noise[0] = 1.0  # Keep DC
    new_mag = mag * np.abs(noise)
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Aggressive L-BFGS with fine geometric beta sweep
2. Improved active-set Newton with warm starts
3. Spectral-domain perturbations for diversity
4. Coordinate descent refinement
5. Multi-n exploration with interpolation
6. Dynamic time management

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _cg_solve(matvec, rhs, max_cg=400, tol=1e-13):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _spectral_perturb(a, rng, strength=0.08):

    mag = np.abs(A)
    noise = 1.0 + strength * rng.standard_normal(len(mag))

    new_mag = mag * np.abs(noise)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try smaller reduction
            a_small = a.copy()
            for k_pos in top_positions[:10]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                if i_lo > i_hi:
                    continue
                best_prod = -1
                best_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > best_prod:
                        best_prod = prod
                        best_i = i
                if best_i >= 0:
                    j = k_pos - best_i
                    if a_small[best_i] >= a_small[j]:
                        a_small[best_i] *= 0.998
                    else:
                        a_small[j] *= 0.998
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-12:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: aggressive L-BFGS, active-set Newton,
    spectral perturbations, targeted deflation, multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    lbfgs_dl = deadline - 500.0
    a_main = _run_lbfgs_aggressive(parent, lbfgs_dl, min_time_remaining=500.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        as_dl = min(deadline - 350, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=400,
                                         tol_list=[0.01, 0.005, 0.002, 0.001,
                                                   0.0005, 0.0002, 0.0001, 0.00005])
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 250 and best_seq is not None:
            as_dl2 = min(deadline - 250, time.monotonic() + 100)
            a_as2, sc_as2 = _active_set_newton(a_as, as_dl2, max_iter=250,
                                               tol_list=[0.001, 0.0005, 0.0002,
                                                         0.0001, 0.00005, 0.00002])
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: Targeted deflation
    print(f"\n[Phase 3] Targeted deflation (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        def_dl = min(deadline - 200, time.monotonic() + 60)
        a_def, sc_def = _deflation_targeted(best_seq, def_dl, n_attempts=40)
        update_best(a_def, sc_def, "deflation")
        if sc_def < best_score + 1e-10 and time.monotonic() < deadline - 100:
            a_def2 = _run_lbfgs_aggressive(a_def, min(deadline - 50, time.monotonic() + 80),
                                           min_time_remaining=50.0)
            update_best(a_def2, _score(a_def2), "lbfgs_post_def")

    # Phase 4: Spectral perturbations + re-optimize
    print(f"\n[Phase 4] Spectral restarts (t={time.monotonic()-t_start:.1f}s)")
    for restart_i in range(5):
        if time.monotonic() > deadline - 50:
            break
        strength = 0.03 + 0.02 * restart_i
        a_pert = _spectral_perturb(best_seq, rng, strength)
        sc_pert = _score(a_pert)
        # Quick L-BFGS on perturbed
        if time.monotonic() < deadline - 30:
            a_opt = _run_lbfgs_aggressive(a_pert, min(deadline - 25, time.monotonic() + 40),
                                          min_time_remaining=25.0)
            update_best(a_opt, _score(a_opt), f"spectral_r{restart_i}")
        else:
            update_best(a_pert, sc_pert, f"spectral_r{restart_i}_raw")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = []
    if n_parent != 8192:
        n_candidates.append(8192)
    if n_parent != 16384:
        n_candidates.append(16384)
    if n_parent != 4096:
        n_candidates.append(4096)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 40:
            break
        a_init = _interpolate_to_n(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _run_lbfgs_aggressive(a_init, min(deadline - 30, time.monotonic() + 50),
                                      min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 6: Log-perturb restarts
    print(f"\n[Phase 6] Log-perturb restarts (t={time.monotonic()-t_start:.1f}s)")
    for sig in [0.01, 0.03, 0.05]:
        if time.monotonic() > deadline - 20:
            break
        a_p = _perturb_log(best_seq, sig, rng)
        a_opt = _run_lbfgs_aggressive(a_p, min(deadline - 15, time.monotonic() + 30),
                                      min_time_remaining=15.0)
        update_best(a_opt, _score(a_opt), f"perturb_{sig}")

    # Phase 7: Final polish - tight active-set + tiny beta L-BFGS
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        as_final = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=200,
                                           tol_list=[0.0005, 0.0002, 0.0001,
                                                     0.00005, 0.00002, 0.00001])
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(100, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-15, 'maxcor': 80, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
