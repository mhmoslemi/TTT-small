import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

    s = float(a.sum())

    c = _full_convolve(a, a)

    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    obj_val = 2.0 * n * _soft_max_value(c, beta)
    grad_a = _grad_soft(a, beta, n)
    W = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W)
    return obj_val, grad_z

        if no_improve > max_no_improve:
            break
        
        # Find active set (peaks within tol of max)
        tol = 0.001 * c_max
        active_mask = c >= (c_max - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True
            n_active = 1
        
        # Average the gradients over the active set
        # Grad for peak at position k: 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        # For the average: grad_avg[j] = 2/n_active * sum_{k in active} a[k-j] * 1[valid]
        # This is: 2/n_active * conv(w_avg, a_rev)[n-1+j]
        # where w_avg is uniform on active set
        
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active_mask] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        # Project negative gradient onto {a >= 0, sum a = 1}
        # Simple approach: a_new = a - step * (grad - mean(grad)), then clip and renormalize
        step = step0 / (1.0 + 0.01 * t)
        grad_centered = grad - np.mean(grad)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t += 1
        step *= decay
    
    return best_a, best_sc

def _peak_shave(a_init, deadline, min_time_remaining=30.0):
    """Iteratively identify the peak and solve a local QP to reduce it."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    iteration = 0
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
        
        # Find the peak position
        k_star = int(np.argmax(c))
        
        # The peak value is c[k_star] = sum_i a[i]*a[k_star-i]
        # We want to reduce this by modifying a.
        # The gradient at the peak is: 2*a[k_star - j] for valid j
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        
        # Contributions to the peak: a[i] * a[k_star-i] for i in [i_lo, i_hi]
        contribs = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = k_star - i
            contribs[i] = a[j]
            if j != i:
                contribs[j] += a[i]
        # Actually the gradient is 2*a[k_star - j] for j in [i_lo, i_hi]
        grad = np.zeros(n)
        for j in range(i_lo, i_hi + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        # We want to move in the direction that reduces the peak the most
        # subject to sum(a) = 1 and a >= 0
        # Simple approach: reduce elements with high gradient, increase others
        # to maintain the sum
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Try several step sizes
        best_step = 0
        best_trial_sc = current_sc
        for frac in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            step_frac = frac * c_max / (g_norm + 1e-30)
            # Center the gradient to maintain sum
            grad_c = grad - np.mean(grad)
            a_try = a - step_frac * grad_c
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_trial_sc:
                best_trial_sc = sc_try
                best_step = step_frac
                a_best_try = a_try.copy()
        
        if best_step > 0:
            a = a_best_try
        else:
            # Try a different approach: directly reduce the top contributing elements
            n_reduce = max(1, int(n * 0.002))
            top_idx = np.argsort(grad)[::-1][:n_reduce]
            total_reduce = 0.01 * c_max
            for idx in top_idx:
                reduce_amt = total_reduce * grad[idx] / (grad[top_idx].sum() + 1e-30)
                a[idx] = max(0, a[idx] - reduce_amt)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            if _score(a) >= best_sc:
                a = best_a.copy()
                break
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 7.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.01
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * (2*t - 1)**2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _random_restart_optimize(a_init, deadline, min_time_remaining=30.0, n_restarts=3):
    """Try multiple random perturbations and pick the best."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(12345)

        if time.monotonic() + min_time_remaining > deadline:
            break
        # Perturb in log space
        sigma = 0.1 * (1.0 - r / n_restarts)
        z = np.log(np.maximum(a_init, 1e-30)) + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax(z)

        if sc_opt < best_sc:

            best_a = a_opt.copy()
    
    return best_a, best_sc

"""
Focused multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous:
- Extended adaptive L-BFGS with many fine-grain beta levels (main improvement)
- KKT-inspired active-set refinement replacing heavy QP

- Multi-profile search with diverse initializations
- Careful time management to maximize iterations at small beta

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_conv_at_k(a, k, n):
    """Gradient of conv(a,a)[k] w.r.t. a: equals 2*a[k-j] for valid j."""
    grad = np.zeros(n)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    for j in range(j_lo, j_hi + 1):
        grad[j] = 2.0 * a[k - j]
    return grad


def _grad_soft_structured(a, beta, n):
    """Efficient gradient of 2*n*soft_max(conv(a,a)) using structure.
    
    grad[j] = 2 * conv(w, a_rev)[n-1+j] where w = soft_max_weights
    """
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with aggressive adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    # Many fine-grain beta levels
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001
    ])
    betas = gap * multipliers

    t_level_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max iterations by remaining time and problem size
        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))

        # For very small beta, cap iterations to avoid getting stuck
        if beta < gap * 0.001:
            max_iter = min(max_iter, 2000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start_global
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 10:
                    print(f"    beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement.
    
    At the optimum, for active peaks S and support T of a:
    mean_{k in S} a[k-j] = const for all j in T
    
    We iteratively:
    1. Identify active set S (peaks near max)
    2. Compute the "target" gradient from S
    3. Adjust a to equalize the gradient on the support
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 300
    tol_frac = 0.005
    min_tol = 0.0001

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute average gradient from active set
        # grad[j] = 2/n_active * sum_{k in active} a[k-j] (for valid k-j)
        # = 2/n_active * conv(w_active, a_rev)[n-1+j]
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Identify support of a
        support = a > 1e-8 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT condition: grad_avg should be constant on support
        # Target: make grad_avg[j] = lambda for j in support
        # lambda = mean of grad_avg over support
        lam = float(np.mean(grad_avg[support]))
        # Residual: how much each support element deviates
        residual = grad_avg[support] - lam

        # Update: move a in direction that reduces residual
        # da[j] = -eta * (grad_avg[j] - lam) for j in support
        # This is like a projected gradient step toward KKT satisfaction
        eta = 0.05
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Line search
        sc_new = _score(a_new)
        if sc_new >= current_sc:
            # Try smaller step
            found = False
            for trial_eta in [0.02, 0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-12:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                # No improvement, tighten tolerance
                tol_frac = max(min_tol, tol_frac * 0.8)
                continue

        a = a_new

        # Gradually tighten
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _subgradient_method(a_init, deadline, min_time_remaining=30.0):
    """Pure subgradient method with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Lower bound: 2n/(2n-1)
    lb = 2.0 * n / (2 * n - 1)
    sc_best = best_sc

    t = 0
    step_scale = 0.5

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            sc_best = current_sc

        # Active set
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Gradient (average over active set)
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Polyak step size
        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_norm_sq = float(np.dot(grad - np.mean(grad), grad - np.mean(grad)))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.1)

        grad_c = grad - np.mean(grad)
        a_new = a - step * grad_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t += 1

        # Decay step scale
        if t % 100 == 0:
            step_scale *= 0.95

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _coordinate_descent_pass(a, n_coord=50):
    """One pass of coordinate descent: optimize individual elements.
    
    For a[j], the relevant convolution values are c[k] for k in [j, j+n-1].
    The dependency is: c[k] includes a[j]*a[k-j] and a[k-j]*a[j] (if distinct).
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s

    # Random order
    order = np.random.default_rng(999).permutation(n)
    order = order[:n]  # all elements

    for idx in order:
        if idx is None:
            continue
        j = int(idx)
        # Find which c[k] values depend on a[j]
        # c[k] depends on a[j] when k-j is in [0, n-1], i.e., k in [j, j+n-1]
        # Also when k-j < 0 but j is in valid range... actually for full conv:
        # c[k] = sum_i a[i]*a[k-i], a[j] appears when i=j (needs k-j in [0,n-1])
        # or when k-i=j, i.e., i=k-j (needs k-j in [0,n-1], same condition)
        # So k ranges from j to j+n-1, clipped to [0, 2n-2]
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)

        if k_lo > k_hi:
            continue

        # For each k in [k_lo, k_hi], the contribution of a[j] to c[k] is:
        # a[j]*a[k-j] if k-j != j (i.e., k != 2j)
        # a[j]^2 if k == 2j
        # Plus a[k-j]*a[j] if k-j != j (same as above, double counted)
        # Total: 2*a[j]*a[k-j] if k != 2j, or a[j]^2 if k == 2j

        # Let's compute the base (without a[j]) and the coefficient of a[j]
        aj = a[j]
        # For the objective max_k c[k], we want to reduce the max
        # Simple approach: try setting a[j] to 0 and see if max decreases
        a[j] = 0.0
        s_temp = a.sum()
        if s_temp < 1e-10:
            a[j] = aj
            continue
        a_norm = a / s_temp
        c_temp = _full_convolve(a_norm, a_norm)
        max_temp = float(c_temp.max())
        score_temp = 2.0 * n * max_temp

        a[j] = aj
        s = a.sum()
        a_norm_orig = a / s
        c_orig = _full_convolve(a_norm_orig, a_norm_orig)
        max_orig = float(c_orig.max())
        score_orig = 2.0 * n * max_orig

        if score_temp < score_orig - 1e-12:
            # Setting to zero helps
            a[j] = 0.0
        else:
            # Try a small reduction
            a[j] = aj * 0.9
            s = a.sum()
            a_try = a / s
            c_try = _full_convolve(a_try, a_try)
            score_try = 2.0 * n * float(c_try.max())
            if score_try < score_orig - 1e-12:
                a = a_try
            else:
                a[j] = aj

    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 300)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=100, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Subgradient ===
    print(f"\n[P4] Subgradient (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 100, time.monotonic() + 150)
        a4, sc4 = _subgradient_method(best_seq, p4_end, min_time_remaining=50)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")
        # L-BFGS after subgradient
        if time.monotonic() + 20 < deadline:
            p4b_end = min(deadline - 60, time.monotonic() + 100)
            a4b, sc4b = _run_lbfgs(best_seq, p4b_end, min_time_remaining=30, verbose=False)
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  P4b result: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [8192, 12288, 16384, 20480]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p5_end = min(deadline - 50, time.monotonic() + 100)
        a5, sc5 = _run_lbfgs(a_r, p5_end, min_time_remaining=30, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 6: Diverse profiles ===
    print(f"\n[P6] Diverse profiles (t={time.monotonic()-t_start_global:.0f}s)")
    profiles = ["periodic", "bimodal", "cubic_taper", "gaussian", "raised_cosine"]
    n_for_profiles = len(best_seq)
    for prof in profiles:
        if time.monotonic() + 50 > deadline:
            break
        a_p = _make_profile(n_for_profiles, prof)
        sc_p = _score(a_p)
        # Only optimize if promising
        if sc_p < best_score + 0.5:
            p6_end = min(deadline - 30, time.monotonic() + 80)
            a_p_opt, sc_p_opt = _run_lbfgs(a_p, p6_end, min_time_remaining=20, verbose=False)
            if sc_p_opt < best_score:
                best_score = sc_p_opt
                best_seq = a_p_opt.copy()
                print(f"  {prof}: {best_score:.10f}")

    # === Phase 7: Final L-BFGS + active set ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 120)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=5, verbose=False)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 10 < deadline:
            p7b_end = min(deadline - 3, time.monotonic() + 40)
            a7b, sc7b = _active_set_refine(best_seq, p7b_end, min_time_remaining=3)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
