def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):

    c = _fft_convolve(a, a)

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cross_corr_fft(p, a):
    """Compute cross-correlation: result[j] = sum_k p[k] * a[k-j] for j=0..n-1"""
    n = len(a)
    N = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # ifft(conj(P) * A) gives: sum_k p[k] * a[(j-k) mod N]
    # What we want: sum_k p[k] * a[k-j]
    # Note: a[k-j] = a[(k-j) mod N] with zero padding
    # ifft(conj(P)*A)[j] = sum_k p[k] * a[j-k] (circular)
    # We want sum_k p[k] * a[k-j] = sum_k p[k] * a[-(j-k)] 
    # = ifft(P * conj(A))[j] ... let me verify
    # Actually: ifft(P * conj(A))[j] = sum_k p[k] * conj(a[k-j]) 
    # For real a: = sum_k p[k] * a[k-j] (circular correlation)
    # With zero padding N >= len(p)+n-1, this gives the linear result
    cross = np.fft.irfft(P * np.conj(A), N)
    return cross[:len(p)]

            # Gradient in z-space: a * (g_a - dot(a, g_a))
            center = float(np.dot(aa, g_a))
            g_z = aa * (g_a - center)
            return float(val), g_z
        
        def obj_only(z_flat):
            aa = _softmax_stable(z_flat)
            cc = _fft_convolve(aa, aa)
            c_shifted = (cc - cc.max()) / T
            e = np.exp(c_shifted)
            s = e.sum()
            if s < 1e-300:
                return float(cc.max()) * 2.0 * n
            return 2.0 * n * T * float(np.log(s) + cc.max() / T)
        
        try:
            res = scipy_minimize(
                obj_only, z, 
                jac=lambda z_flat: obj_and_grad(z_flat)[1],
                method='L-BFGS-B',
                options={'maxiter': 2000, 'maxfun': 5000,
                         'ftol': 1e-12, 'gtol': 1e-8,
                         'iprint': 0}
            )
            a_new = _softmax_stable(res.x)
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            if verbose:
                print(f"  LBFSGS T={T:.6f}: score={_score(a):.8f} "
                      f"iters={res.nit} t={time.monotonic()-time_of_start:.0f}s")
        except Exception as ex:
            if verbose:
                print(f"  LBFSGS T={T:.6f}: error {ex}")
    
    return best_a, best_score


def _upsample(a, n_new):
    """Upsample sequence to new length using linear interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def _coordinate_descent(a, deadline, max_rounds=100, n_try=300):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        any_improve = False
        
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        k_max = int(np.argmax(c))
        
        # Find elements contributing to the peak
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        # Try random subset of contributing elements
        n_cantry = min(n_try, hi - lo + 1)
        indices = np.random.choice(hi - lo + 1, size=min(n_cantry, hi-lo+1), 
                                   replace=False) + lo
        
        for j in indices:
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max())
                if sc < c_max * (1 - 1e-12):
                    best_a = a_cand
                    c = c_cand
                    c_max = sc
                    any_improve = True
                    k_max = int(np.argmax(c))
                    break
            if any_improve and np.random.rand() < 0.3:
                break  # Sometimes continue, sometimes break early
        
        if not any_improve:
            # Expanding search: try all elements with fewer scales
            any_improve2 = False
            for j in range(0, n, max(1, n // 500)):
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.9, 1.1]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max())
                    if sc < c_max * (1 - 1e-12):
                        best_a = a_cand
                        c_max = sc
                        any_improve2 = True
                        break
                if any_improve2:
                    break
            if not any_improve2:
                break
    
    return best_a, _score_norm(best_a)


def _smoothed_gradient_descent(a, deadline, T_start=None, lr_start=0.01,
                                max_iter=50000, verbose=True):
    """Polynomial-homotopy style: decrease T while doing gradient steps."""
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    if T_start is None:
        T_start = c_max * 0.02
    
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    T = T_start
    lr = lr_start
    n_steps_per_T = 200
    n_cool = 10
    time_per_cool = max(2.0, (deadline - time.monotonic()) / n_cool)
    
    for cool_idx in range(n_cool):
        if time.monotonic() > deadline:
            break
        T = T_start * (0.5 ** cool_idx)
        sub_dl = min(deadline, time.monotonic() + time_per_cool)
        
        for step_idx in range(n_steps_per_T):
            if time.monotonic() > sub_dl:
                break
            
            c = _fft_convolve(a, a)
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s = e.sum()
            if s < 1e-300:
                pc = np.zeros_like(c)
                pc[np.argmax(c)] = 1.0
            else:
                pc = e / s
            
            # Gradient
            g_a = 2.0 * _cross_corr_fft(pc, a)
            # Project to tangent space
            g_a = g_a - float(np.dot(a, g_a)) * a
            g_norm = np.linalg.norm(g_a)
            if g_norm < 1e-15:
                break
            
            a_new = a - lr * g_a / g_norm
            a_new = np.maximum(0, a_new)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new /= s_new
            
            sc_new = float(_fft_convolve(a_new, a_new).max())
            if sc_new < float(c.max()) * (1 - 1e-12):
                a = a_new
                if sc_new * 2.0 * n < best_score:
                    best_score = sc_new * 2.0 * n
                    best_a = a.copy()
                lr = min(lr * 1.01, 0.05)
            else:
                lr *= 0.8
        
        if verbose and cool_idx % 2 == 0:
            print(f"  SGD cool[{cool_idx}] T={T:.6f}: score={_score(a):.8f} "
                  f"t={time.monotonic()-time_of_start:.0f}s")
    
    return best_a, best_score


# Global for timing (set in propose_candidate)
time_of_start = 0.0


def propose_candidate(seed=42, budget_s=1800):
    global time_of_start
    time_of_start = time.monotonic()
    deadline = time_of_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
        n = len(a)
        print(f"Loaded parent: n={n}, score={_score(a):.10f}")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")
    
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial score: {best_score:.10f}")
    
    # Phase 1: Aggressive active-set (0 - 400s)
    p1_end = min(deadline, time_of_start + 400.0)
    print(f"Phase 1: Active-set (deadline {p1_end - time_of_start:.0f}s)")
    a_as, sc_as = _active_set_phase(best_a, p1_end, K=50, tol_frac=0.003,
                                     max_iter=5000, verbose=True)
    if sc_as < best_score:
        best_score = sc_as
        best_a = a_as.copy()
    print(f"Phase 1 done: score={_score(best_a):.10f}, t={time.monotonic()-time_of_start:.0f}s")
    
    # Phase 2: L-BFGS (400 - 700s)
    p2_end = min(deadline, time_of_start + 700.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: L-BFGS (deadline {p2_end - time_of_start:.0f}s)")
        a_lb, sc_lb = _lbfsgs_phase(best_a, p2_end, verbose=True)
        if sc_lb < best_score:
            best_score = sc_lb
            best_a = a_lb.copy()
        print(f"Phase 2 done: score={_score(best_a):.10f}, t={time.monotonic()-time_of_start:.0f}s")
    
    # Phase 3: Smoothed gradient descent (700 - 900s)
    p3_end = min(deadline, time_of_start + 900.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Smoothed GD (deadline {p3_end - time_of_start:.0f}s)")
        a_sg, sc_sg = _smoothed_gradient_descent(best_a, p3_end, 
                                                  T_start=float(_fft_convolve(best_a, best_a).max()) * 0.01,
                                                  lr_start=0.005, verbose=True)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
        print(f"Phase 3 done: score={_score(best_a):.10f}, t={time.monotonic()-time_of_start:.0f}s")
    
    # Phase 4: Try n=16384 (900 - 1200s)
    p4_end = min(deadline, time_of_start + 1200.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: n=16384 attempt (deadline {p4_end - time_of_start:.0f}s)")
        n_new = 16384
        a_up = _upsample(best_a, n_new)
        sc_up = _score(a_up)
        print(f"  Upsampled score: {sc_up:.10f}")
        
        if sc_up <= best_score * 1.01:  # Only continue if competitive
            a_up_opt, sc_up_opt = _active_set_phase(a_up, p4_end, K=50, 
                                                      tol_frac=0.003, max_iter=3000,
                                                      verbose=True)
            sc_up_final = _score(a_up_opt)
            print(f"  n=16384 optimized: {sc_up_final:.10f}")
            if sc_up_final < best_score:
                best_score = sc_up_final
                best_a = a_up_opt.copy()
                n = n_new
        # Also try n=4096
        n_small = 4096
        a_small = _upsample(best_a, n_small)
        sc_small = _score(a_small)
        print(f"  n=4096 resampled: {sc_small:.10f}")
        if sc_small < best_score:
            a_small_opt, _ = _active_set_phase(a_small, 
                                                min(deadline, time.monotonic() + 60),
                                                K=30, tol_frac=0.003, max_iter=1000,
                                                verbose=False)
            sc_small_opt = _score(a_small_opt)
            if sc_small_opt < best_score:
                best_score = sc_small_opt
                best_a = a_small_opt.copy()
    
    # Phase 5: Multi-restart (1200 - 1500s)
    p5_end = min(deadline, time_of_start + 1500.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: Multi-restart (deadline {p5_end - time_of_start:.0f}s)")
        n = len(best_a)
        n_restarts = 6
        time_per_restarts = max(15.0, (p5_end - time.monotonic()) / n_restarts)
        
        for r in range(n_restarts):
            if time.monotonic() > p5_end - time_per_restarts * 0.5:
                break
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(best_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            
            sub_dl = min(p5_end - 5, time.monotonic() + time_per_restarts * 0.8)
            a_pert_opt, sc_pert = _active_set_phase(a_pert, sub_dl, K=40,
                                                     tol_frac=0.003, max_iter=2000,
                                                     verbose=False)
            sc_pert_final = _score(a_pert_opt)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={sc_pert_final:.10f} "
                  f"t={time.monotonic()-time_of_start:.0f}s")
            if sc_pert_final < best_score:
                best_score = sc_pert_final
                best_a = a_pert_opt.copy()
    
    # Phase 6: Final coordinate descent polish (1500 - 1750s)
    p6_end = min(deadline, time_of_start + 1750.0)
    if time.monotonic() < p6_end:
        print(f"Phase 6: Coordinate descent (deadline {p6_end - time_of_start:.0f}s)")
        a_cd, sc_cd = _coordinate_descent(best_a, p6_end, max_rounds=200, n_try=400)
        sc_cd_final = _score(a_cd)
        print(f"Phase 6: score={sc_cd_final:.10f}, t={time.monotonic()-time_of_start:.0f}s")
        if sc_cd_final < best_score:
            best_score = sc_cd_final
            best_a = a_cd.copy()
    
    # Phase 7: Ultra-fine active-set (1750 - deadline)
    if time.monotonic() < deadline - 5:
        print("Phase 7: Ultra-fine active-set")
        a_fine, sc_fine = _active_set_phase(best_a, deadline - 3, K=20,
                                             tol_frac=0.001, max_iter=500,
                                             verbose=False)
        sc_fine_final = _score(a_fine)
        if sc_fine_final < best_score:
            best_score = sc_fine_final
            best_a = a_fine.copy()
        print(f"Phase 7: score={sc_fine_final:.10f}")
    
    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    else:
        final = final / s * s  # keep as-is, evaluator handles normalization
    
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-time_of_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
