"""
LP-based active set optimization for minimum peak autocorrelation.
Uses conditional gradient (Frank-Wolfe) with LP oracle, combined with
subgradient descent for initial convergence and coordinate descent for polishing.
"""
import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1, so score = 2n * max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _get_active_lags(c, tol_frac=0.005, max_k=10):
    """Get lags where convolution value is within tol_frac of max."""
    c_max = float(c.max())
    threshold = c_max * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        return np.array([int(np.argmax(c))])
    if len(active) > max_k:
        # Keep only the top max_k
        sorted_idx = active[np.argsort(c[active])[::-1][:max_k]]
        return sorted_idx
    return active


def _solve_lp_direction(a, M, active_lags):
    """
    Solve the LP: maximize sum(b) subject to conv(a,b)[k] <= M for k in active_lags, b >= 0.
    Returns the optimal b (normalized to sum 1) or None if failed.
    """
    n = len(a)
    K = len(active_lags)

    # Build constraint matrix C: C[i, j] = a[k_i - j] if 0 <= k_i - j < n
    C = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            C[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    # Solve dual: minimize M * sum(y) subject to C^T y >= 1, y >= 0
    # Standard form: min c^T y s.t. A_ub y <= b_ub, bounds
    c_obj = np.full(K, M)
    A_ub = -C.T  # shape (n, K)
    b_ub = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * K,
                      method='highs',
                      options={"time_limit": 10, "presolve": True})
    except Exception:
        return None

    if res.status != 0:
        # Fallback: solve primal directly
        try:
            c_obj_p = -np.ones(n)
            A_ub_p = C / M
            b_ub_p = np.ones(K)
            res_p = linprog(c_obj_p, A_ub=A_ub_p, b_ub=b_ub_p,
                            bounds=[(0, None)] * n,
                            method='highs',
                            options={"time_limit": 15, "presolve": True})
            if res_p.status == 0 and res_p.x is not None:
                b = np.maximum(0, res_p.x)
                s = b.sum()
                if s > 1e-15:
                    return b / s
        except Exception:
            pass
        return None

    y = np.maximum(0, res.x)

    # Extract primal solution from complementary slackness
    # b[j] > 0 only if (C^T y)[j] = 1
    Cy = C.T @ y  # shape (n,)
    tol = 1e-6
    support = np.where(np.abs(Cy - 1.0) < tol)[0]

    if len(support) == 0:
        # Fallback: use the largest n indices close to 1
        idx = np.argsort(-Cy)[:K + 2]
        support = idx

    if len(support) > 2 * K + 5:
        # Too many, trim to top
        support = support[np.argsort(-Cy[support])[:2 * K + 5]]

    # Solve: for each i in 0..K-1, sum_{j in support} C[i,j] * b[j] = M
    # This is an underdetermined or square system
    Cs = C[:, support]  # shape (K, len(support))
    rhs = np.full(K, M)

    if len(support) <= K:
        # Square or underdetermined - use lstsq
        b_s, _, _, _ = np.linalg.lstsq(Cs, rhs, rcond=None)
    else:
        # Overdetermined from support perspective, use minimum norm
        # b_s = Cs^T (Cs Cs^T)^{-1} rhs
        G = Cs @ Cs.T
        try:
            G_inv = np.linalg.inv(G + 1e-12 * np.eye(K))
            b_s = Cs.T @ (G_inv @ rhs)
        except np.linalg.LinAlgError:
            b_s, _, _, _ = np.linalg.lstsq(Cs, rhs, rcond=None)

    b_s = np.maximum(0, b_s)
    s = b_s.sum()
    if s < 1e-15:
        return None

    b = np.zeros(n)
    b[support] = b_s
    return b / s


def _lp_frank_wolfe(a_init, deadline, max_iter=5000, tol_frac=0.003, max_k=8,
                    initial_step=0.5, step_decay=0.999, restart_interval=500):
    """
    Frank-Wolfe with LP oracle for active set optimization.
    a_init should be normalized to sum=1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a /= s

    best_a = a.copy()
    best_score = _score_norm(a)
    step = initial_step
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        M = float(c.max())

        # Get active set
        active = _get_active_lags(c, tol_frac=tol_frac, max_k=max_k)

        # Solve LP for direction
        b = _solve_lp_direction(a, M, active)
        if b is None:
            b = np.ones(n) / n

        # Frank-Wolfe step
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new

        sc_new = _score_norm(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                a_try = (1.0 - step * shrink) * a + step * shrink * b
                a_try = np.maximum(0, a_try)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try /= s_try
                sc_try = _score_norm(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    break

        # Decay step size
        step *= step_decay

        # Restart logic
        if no_improve > 50 and it % restart_interval < 100:
            # Perturb and restart
            step = max(step, 0.3)
            noise = 0.001 * np.random.randn(n)
            a = best_a + noise * best_a
            a = np.maximum(0, a)
            s = a.sum()
            if s > 1e-15:
                a /= s
            no_improve = 0

        # If convergence is very close, tighten tolerance
        if it > 1000 and no_improve > 200:
            break

    return best_a, best_score


def _subgrad_logspace(a_init, deadline, max_iter=200000, T_start=None, T_end=None,
                      lr_start=0.03, lr_end=0.003, active_k=5, eval_interval=50):
    """Subgradient descent in log-space (similar to previous but refined)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    last_improve_t = 0
    current_lr_mult = 1.0

    if T_start is None:
        c = _fft_convolve(a_init, a_init)
        T_start = float(c.max()) * 0.01
    if T_end is None:
        T_end = T_start * 0.01

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Compute gradient using top-k active lags
        c_sorted = np.argsort(c)[::-1]
        num_active = min(active_k, len(c_sorted))
        g = np.zeros(n)
        for idx in range(num_active):
            k = int(c_sorted[idx])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= num_active

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Center the gradient for log-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve_t = t
            else:
                patience += 1

            if patience > 10 and t - last_improve_t > 500:
                current_lr_mult *= 0.7
                patience = 0
                if current_lr_mult < 0.02:
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.005 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    current_lr_mult = 0.2

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

    Multi-phase optimizer:
    Phase 1: Subgradient log-space descent (fast convergence)
    Phase 2: LP-based Frank-Wolfe (active set polishing)
    Phase 3: Second subgradient pass (escape local minima)
    Phase 4: LP polishing with tighter active set
    Phase 5: Coordinate descent final polish
    Phase 6: Random restarts with quick LP
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Subgradient log-space (0 - 400s)
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: Subgradient log-space, time_budget={phase1_end - t_start:.0f}s")
    a_opt, sc = _subgrad_logspace(
        a, phase1_end, max_iter=300000,
        T_start=None, T_end=None,
        lr_start=0.04, lr_end=0.003,
        active_k=7, eval_interval=40
    )
    full_sc = _score(a_opt)
    print(f"Phase 1 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 2: LP-based Frank-Wolfe (400 - 800s)
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: LP Frank-Wolfe, time_budget={phase2_end - time.monotonic():.0f}s")
        a_fw, sc_fw = _lp_frank_wolfe(
            best_a, phase2_end, max_iter=3000,
            tol_frac=0.003, max_k=8,
            initial_step=0.5, step_decay=0.9995
        )
        full_sc_fw = _score(a_fw)
        print(f"Phase 2 done: score={full_sc_fw:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_fw < best_score:
            best_score = full_sc_fw
            best_a = a_fw.copy()

    # Phase 3: Second subgradient pass with lower temperature (800 - 1000s)
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Refined subgradient, time_budget={phase3_end - time.monotonic():.0f}s")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt2, sc2 = _subgrad_logspace(
            best_a, phase3_end, max_iter=200000,
            T_start=max_cb * 0.003, T_end=max_cb * 0.0001,
            lr_start=0.015, lr_end=0.001,
            active_k=6, eval_interval=30
        )
        full_sc2 = _score(a_opt2)
        print(f"Phase 3 done: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 4: LP with tighter active set (1000 - 1300s)
    phase4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Tight LP polishing, time_budget={phase4_end - time.monotonic():.0f}s")
        a_fw2, sc_fw2 = _lp_frank_wolfe(
            best_a, phase4_end, max_iter=5000,
            tol_frac=0.001, max_k=5,
            initial_step=0.3, step_decay=0.9998
        )
        full_sc_fw2 = _score(a_fw2)
        print(f"Phase 4 done: score={full_sc_fw2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_fw2 < best_score:
            best_score = full_sc_fw2
            best_a = a_fw2.copy()

    # Phase 5: Coordinate descent (1300 - 1500s)
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Coordinate descent, time_budget={phase5_end - time.monotonic():.0f}s")
        a_cd, sc_cd = _coord_descent_polish(best_a, phase5_end, max_rounds=800)
        full_sc_cd = _score(a_cd)
        print(f"Phase 5 done: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 6: Random restarts with quick LP (1500 - 1750s)
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end:
        print(f"Phase 6: Random restarts, time_budget={phase6_end - time.monotonic():.0f}s")
        n_restarts = 6
        time_per = max(15.0, (phase6_end - time.monotonic()) / n_restarts)
        for r in range(n_restarts):
            if time.monotonic() > phase6_end - 20:
                break
            sigma = 0.003 + 0.002 * r
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax_stable(z)

            sub_dl = min(phase6_end - 10, time.monotonic() + time_per * 0.7)
            a_pert_opt, sc_pert = _lp_frank_wolfe(
                a_pert, sub_dl, max_iter=1000,
                tol_frac=0.002, max_k=6,
                initial_step=0.4, step_decay=0.999
            )
            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_pert:.10f}")
            if full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

    # Phase 7: Final ultra-fine polish (1750 - deadline)
    if time.monotonic() < deadline - 10:
        print(f"Phase 7: Final polish, time_budget={deadline - time.monotonic() - 10:.0f}s")
        # Quick LP with very tight tolerance
        a_final, sc_final = _lp_frank_wolfe(
            best_a, deadline - 8, max_iter=2000,
            tol_frac=0.0005, max_k=4,
            initial_step=0.2, step_decay=0.9999
        )
        full_sc_final = _score(a_final)
        print(f"Phase 7a: score={full_sc_final:.10f}")
        if full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()

        # Quick coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd_f, sc_cd_f = _coord_descent_polish(best_a, deadline - 3, max_rounds=200)
            full_sc_cd_f = _score(a_cd_f)
            if full_sc_cd_f < best_score:
                best_score = full_sc_cd_f
                best_a = a_cd_f.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
