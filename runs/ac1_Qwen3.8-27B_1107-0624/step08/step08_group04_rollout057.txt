(2) IRLS-style iterative quadratic minimization with FFT Hankel matvec,
(3) active-set Newton with progressive tolerance tightening,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=1000, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            # Restart CG
            r = r_new
            p = r.copy()
            rs = float(r @ r)
            continue
        x = x_new
        r = r_new
        if rs_new < tol * tol * rs0:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _make_spectral_init(n, asymmetry=0.3):
    """Create initial sequence with asymmetric spectral shaping.
    Uses a skewed Gaussian in log-domain to break symmetry."""
    t = np.linspace(0, 1, n)
    # Asymmetric profile: heavier on one side
    center = 0.5 + asymmetry * 0.3
    width = 0.4
    a = np.exp(-((t - center) / width) ** 2)
    # Add a small secondary lobe on the opposite side
    a += 0.15 * np.exp(-((t - (1 - center)) / (width * 0.7)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.7):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _run_lbfgs_phased(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with geometric beta continuation schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # Geometric schedule: many more steps at the end
    num_steps = 30
    ratios = np.geomspace(1.0, 1e-8, num_steps)
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.003 * n / 4096.0
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 5))))

        try:
            result = minimize(
                _smoobj,
                z,
                args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _irls_optimize(a_init, deadline, max_rounds=200):
    """IRLS-style optimization: iteratively reweight and solve QP."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Temperature schedule for soft weighting
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Soft weights: exponential of c/beta, beta decreases over iterations
        beta_irls = c_max * max(0.001, 0.1 * (1.0 - rnd / max_rounds))
        shifted = (c - c_max) / beta_irls
        w = np.exp(np.clip(shifted, -700, 0))
        w = w / w.sum()

        # Build Hankel matrix implicitly via W
        W = w.copy()

        # Solve QP: min a^T H a s.t. sum(a) = 1, a >= 0
        # Use projected gradient + CG for indefinite H
        grad = _hankel_matvec(a, W, n)
        lam = float(np.mean(grad))
        g_centered = grad - lam

        # CG step (might not converge if H is indefinite, but helps locally)
        Hg = _cg_solve(W, n, g_centered, max_iter=200, tol=1e-12)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=200, tol=1e-12)

        H1_sum = float(np.sum(H1))
        if abs(H1_sum) > 1e-20 and np.isfinite(H1_sum):
            lam2 = float(np.sum(Hg)) / H1_sum
            dx = -Hg + lam2 * H1
        else:
            dx = -g_centered

        dx = dx - np.mean(dx)
        dx_norm = np.linalg.norm(dx)
        if dx_norm < 1e-18:
            break

        # Line search
        alpha = 1.0
        c_max_cur = c_max
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_cur - 1e-16:
                a = a_try
                c_max_cur = c_max_try
                sc_try = _score(a)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if not improved:
            # Try smaller step
            alpha = 0.1 / (1.0 + 0.01 * rnd)
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a, best_sc


def _active_set_qp(a_init, deadline, tol_schedule=None, max_rounds=150):
    """Active-set QP with progressive tightening and CG inner solver."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                        0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
                        0.000002, 0.000001, 0.0000005, 0.0000002, 0.0000001,
                        0.00000005, 0.00000002, 0.00000001]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(rnd // 3, len(tol_schedule) - 1)]

        # Active set: positions where c is within tol*c_max of the max
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP: min a^T H_W a s.t. sum(a) = 1, a >= 0
        # First try: solve H*y = 1, then a = y / sum(y), project to non-negative
        y = _cg_solve(W, n, np.ones(n), max_iter=800, tol=1e-14)
        oney = float(np.sum(y))

        if abs(oney) > 1e-20 and np.isfinite(oney):
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-15:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                else:
                    # Damped step toward the QP solution
                    a_new2 = 0.5 * a + 0.5 * a_new
                    a_new2 = np.maximum(a_new2, 0)
                    s_new2 = a_new2.sum()
                    if s_new2 > 1e-15:
                        a_new2 = a_new2 / s_new2
                        sc_new2 = _score(a_new2)
                        if sc_new2 < best_sc - 1e-15:
                            best_sc = sc_new2
                            best_a = a_new2.copy()
                            a = a_new2.copy()
        else:
            # Fallback: projected gradient
            grad = _hankel_matvec(a, W, n)
            lam = float(np.mean(grad))
            step = 0.1 / (np.max(np.abs(grad)) + 1e-15)
            a_new = a - step * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-15:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new.copy()

        # Convergence check
        c_check = _full_convolve(a, a)
        c_max_check = float(c_check.max())
        if c_max_check < c_max * (1 - 1e-12):
            pass  # Still improving
        else:
            # Try tighter tol
            if tol < 1e-8:
                break

    return best_a, best_sc


def _subgradient_polyak(a_init, deadline, max_steps=2000):
    """Exact subgradient descent with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Lower bound: 2n/(2n-1) from averaging
    lower_bound = 2.0 * n / (2 * n - 1)

    # Track best c_max for Polyak step
    best_cmax = float(_full_convolve(a, a).max())
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        if c_max < best_cmax:
            best_cmax = c_max
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > 200:
            a = best_a.copy()
            patience = 0
            # Reduce step size by restarting from best
            c = _full_convolve(a, a)
            c_max = float(c.max())
            best_cmax = c_max

        # Polyak step size
        gap = c_max - best_cmax / (2.0 * n)  # approximate gap
        if gap < 1e-18:
            break
        # Use Polyak: alpha = (f(x) - f*) / ||g||^2
        # f(x) = c_max, f* >= best_cmax (lower bound on optimal c_max)
        polyak_num = (c_max - best_cmax)
        if polyak_num < 1e-18:
            alpha = 0.01 / (1.0 + i * 0.001)
        else:
            # Subgradient of c_max w.r.t. a
            k_star = int(np.argmax(c))
            # dc[k_star]/da[i] = a[k_star - i] + a[i - (k_star - n + 1)]
            # More precisely: dc[k]/da[i] = sum of a[j] where j+i=k or i+j=k
            # = a[k-i] (if valid) + a[i-k+n-1]... let me use the Hankel matvec
            e_k = np.zeros(2 * n - 1)
            e_k[k_star] = 1.0
            grad = _hankel_matvec(e_k, np.ones(1) * 0.5, n)
            # Actually, dc[k]/da = the i-th row of 2*B_k applied to a
            # Which is: for each i, dc[k]/da[i] = a[k-i] + a[k-i] (if k-i != i)
            # Let me just compute it directly
            grad = np.zeros(n)
            for i in range(n):
                val = 0.0
                j1 = k_star - i
                if 0 <= j1 < n:
                    val += a[j1]
                j2 = k_star - i  # same as j1 for convolution
                # Actually conv(a,a)[k] = sum_j a[j]*a[k-j]
                # d/d a[i] = a[k-i] (from the first a) + a[k-i] (from the second a)
                # if k-i is valid and i != k-i, otherwise just a[k-i]
                # More carefully: sum_j a[j]*a[k-j], d/da[i]:
                # term j=i: a[k-i]
                # term k-j=i => j=k-i: a[k-i]
                # so if i != k-i: 2*a[k-i], else a[k-i] (but only if valid)
                if 0 <= k_star - i < n:
                    if i == k_star - i:
                        val = a[k_star - i]
                    else:
                        val = 2.0 * a[k_star - i]
                grad[i] = val

            grad_norm_sq = float(np.dot(grad, grad))
            if grad_norm_sq > 1e-30:
                alpha = polyak_num / grad_norm_sq
                alpha = min(alpha, 1.0)
            else:
                alpha = 0.01

        # Projected gradient step
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check if it improves
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        if c_max_new <= c_max + 1e-15:
            a = a_new
        else:
            # Halve step
            alpha *= 0.25
            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                a = a_new

    return best_a, best_sc


def _resample_sequence(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_asymmetric(a, sigma, rng):
    """Perturb with asymmetric noise to break symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.standard_normal(n)
    # Make noise asymmetric: heavier on one side
    t = np.linspace(0, 1, n)
    asym = 1.0 + 0.5 * np.sin(np.pi * t)
    z = z + sigma * asym * noise
    z = z - z.mean()
    return _softmax(z)


def _flatten_direct(a_init, deadline, max_iters=300):
    """Direct flattening: minimize sum of (c[k] - target)^2 for c[k] > target."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_mean = float(c.mean())
        c_max = float(c.max())

        # Excess above mean
        excess = np.maximum(c - c_mean, 0)
        excess_max = float(excess.max())
        if excess_max < 1e-16:
            break

        # Gradient of sum(excess^2) w.r.t. a
        a_rev = a[::-1]
        ec = _full_convolve(excess, a_rev)
        grad = ec[n - 1:2 * n - 1]

        # Also penalize the peak specifically
        peak_idx = int(np.argmax(c))
        e_peak = np.zeros_like(c)
        e_peak[peak_idx] = 1.0
        pc = _full_convolve(e_peak, a_rev)
        grad_peak = pc[n - 1:2 * n - 1]

        combined_grad = 0.5 * grad + 0.5 * grad_peak
        step = 0.5 / (1.0 + 0.02 * it)
        a_new = a - step * combined_grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            a_new2 = 0.7 * a + 0.3 * a_new
            a_new2 = np.maximum(a_new2, 0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                sc_new2 = _score(a_new2)
                if sc_new2 < best_sc - 1e-15:
                    best_sc = sc_new2
                    best_a = a_new2.copy()
                    a = a_new2

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, IRLS, active-set QP,
    subgradient, and direct flattening with multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_spectral_init(n_parent, asymmetry=0.3)
        print(f"No parent, using spectral init n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_work = n_parent

    # Phase 1: L-BFGS from asymmetric perturbation of parent
    print(f"\n[Phase 1] L-BFGS from asymmetric perturbation (t={time.monotonic() - t_start:.1f}s)")
    a_pert = _perturb_asymmetric(parent, 0.03, rng)
    a1 = _run_lbfgs_phased(a_pert, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: L-BFGS from parent directly
    print(f"\n[Phase 2] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2 = _run_lbfgs_phased(parent, deadline, min_time_remaining=400.0)
        update_best(a2, _score(a2), "lbfgs_2")

    # Phase 3: IRLS optimization
    print(f"\n[Phase 3] IRLS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3, sc3 = _irls_optimize(best_seq, min(deadline - 350, time.monotonic() + 200),
                                  max_rounds=100)
        update_best(a3, sc3, "irls_1")

    # Phase 4: Active-set QP
    print(f"\n[Phase 4] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _active_set_qp(best_seq, min(deadline - 250, time.monotonic() + 150),
                                  max_rounds=100)
        update_best(a4, sc4, "active_qp_1")

    # Phase 5: Direct flattening
    print(f"\n[Phase 5] Direct flattening (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _flatten_direct(best_seq, min(deadline - 200, time.monotonic() + 100),
                                   max_iters=200)
        update_best(a5, sc5, "flatten_1")

    # Phase 6: Another IRLS round (tighter)
    print(f"\n[Phase 6] IRLS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a6, sc6 = _irls_optimize(best_seq, min(deadline - 150, time.monotonic() + 100),
                                  max_rounds=80)
        update_best(a6, sc6, "irls_2")

    # Phase 7: Active-set QP (tighter)
    print(f"\n[Phase 7] Active-set QP tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        tol_tight = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
                     0.00001, 0.000005, 0.000002, 0.000001]
        a7, sc7 = _active_set_qp(best_seq, min(deadline - 100, time.monotonic() + 80),
                                  tol_schedule=tol_tight, max_rounds=80)
        update_best(a7, sc7, "active_qp_tight")

    # Phase 8: Subgradient with Polyak
    print(f"\n[Phase 8] Subgradient Polyak (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a8, sc8 = _subgradient_polyak(best_seq, min(deadline - 60, time.monotonic() + 50),
                                       max_steps=1000)
        update_best(a8, sc8, "subgrad")

    # Phase 9: Multi-n exploration
    print(f"\n[Phase 9] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == n_parent:
            continue
        # Resample from best
        if best_seq is not None:
            a_init_n = _resample_sequence(best_seq, n_try)
        else:
            a_init_n = _make_spectral_init(n_try, 0.3)
        # Quick L-BFGS
        a_n = _run_lbfgs_phased(a_init_n, deadline, min_time_remaining=40.0)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

        # Quick active-set QP
        if sc_n < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_np, sc_np = _active_set_qp(a_n, min(deadline - 30, time.monotonic() + 30),
                                          max_rounds=30)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 10: Diverse restarts
    print(f"\n[Phase 10] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Restart from different initializations
        inits = [
            _make_spectral_init(n_work, 0.5),
            _make_spectral_init(n_work, 0.1),
            _make_tapered(n_work, power=1.2),
        ]
        for i, init_i in enumerate(inits):
            if time.monotonic() > deadline - 15:
                break
            a_r = _run_lbfgs_phased(init_i, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _active_set_qp(a_r, min(deadline - 10, time.monotonic() + 10),
                                              max_rounds=20)
                update_best(a_rp, sc_rp, f"restart_{i}_qp")

    # Phase 11: Final ultra-tight polish
    print(f"\n[Phase 11] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        tol_ult = [0.000001, 0.0000005, 0.0000002, 0.0000001,
                   0.00000005, 0.00000002, 0.00000001]
        a_fin, sc_fin = _active_set_qp(best_seq, deadline - 1,
                                        tol_schedule=tol_ult, max_rounds=50)
        update_best(a_fin, sc_fin, "final_qp")

        # Final flatten
        if time.monotonic() < deadline - 2:
            a_fl, sc_fl = _flatten_direct(best_seq, deadline - 1, max_iters=50)
            update_best(a_fl, sc_fl, "final_flatten")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_spectral_init(16384, 0.3)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
