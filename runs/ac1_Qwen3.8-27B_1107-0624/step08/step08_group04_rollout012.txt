from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=800, tol=1e-13):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        rs = rs_new
        p = r + (rs / min(rs0, 1e10)) * p
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0
    else:

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n))

    for outer in range(max_outer):
        support = a > 1e-12
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n

        if n_sup == n:
            # Solve H*y = 1 on full support
            y = _cg_solve(W, n, np.ones(n), max_iter=cg_iter)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            # Use projected gradient for partial support
            grad = _hankel_matvec(a, W, n)
            if support.any():
                lam = np.mean(grad[support])
            else:
                lam = 0
            step = 1.0 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.05)
            a_new = a.copy()
            a_new[support] -= step * (grad[support] - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

        val_new = float(a_new @ _hankel_matvec(a_new, W, n))
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

        # Check convergence via KKT residuals
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-12
        if support.any():
            lam = np.mean(grad[support])
            resid = np.max(np.abs(grad[support] - lam))
            off_s = ~support
            if off_s.any():
                resid = max(resid, float(np.max(lam - grad[off_s])))
            if resid < 1e-10 * (np.max(np.abs(grad)) + 1e-15):
                break
        else:
            break

    return best_a


def _newton_active_polish(a_init, deadline, max_rounds=120):
    """CG-Newton polish with progressive active set tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003,
                    0.0002, 0.0001, 0.00005, 0.00003, 0.00002,
                    0.00001, 0.000005, 0.000003, 0.000002,
                    0.000001, 0.0000005, 0.0000003, 0.0000002, 0.0000001]

    tol_idx = 0
    rounds_per_tol = 4
    stagnation = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP
        a_qp = _active_set_qp(W, n, a, max_outer=60, cg_iter=400)
        sc_qp = _score(a_qp)

        improved = False
        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
            improved = True
        else:
            # Try damped Newton step
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-15:
                Hg = _cg_solve(W, n, g, max_iter=300)
                H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)

                alpha = 1.0
                for ls in range(50):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    c_max_try = float(c_try.max())
                    if c_max_try < c_max * (1 - 1e-10):
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        improved = True
                        break
                    alpha *= 0.5
                    if alpha < 1e-12:
                        break

        if not improved:
            stagnation += 1
            if stagnation > 6:
                tol_idx += 1
                stagnation = 0
        else:
            if (rnd + 1) % rounds_per_tol == 0:
                tol_idx += 1

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _peak_peel(a_init, deadline, max_iters=200):
    """Directly peel the peak: iteratively reduce the maximum convolution entry."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Gradient of c[peak_idx] w.r.t. a
        # d c[k] / d a[j] = a[k-j] + a[j-k] (with bounds)
        k = peak_idx
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            grad[j] = val

        # Remove mean to maintain sum constraint
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search
        alpha = 0.001
        improved = False
        for ls in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1 - 1e-8):
                a = a_try
                sc_try = _score(a)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break

        if not improved:
            break

    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=100):
    """Convolution flattening via excess gradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_mean = float(np.sum(c)) / len(c)
        c_max = float(c.max())

        excess = np.maximum(c - c_mean, 0)
        if excess.max() < 1e-15:
            break

        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]

        step = 0.05 / (1.0 + 0.02 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Partial step
            a = a + 0.3 * (a_new - a)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc


def _flat_spectrum_init(n, n_iter=20):
    """Generate a non-negative sequence with approximately flat DFT magnitude."""
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        if mag.max() > 0:
            target_mag = np.full_like(mag, np.mean(mag[mag > mag.max() * 0.5]))
            # Keep phase
            phase = np.angle(A)
            A_new = target_mag * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, n)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a = a_new / s
    return a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "arcsine":
        t = np.linspace(1e-6, 1 - 1e-6, n)
        a = 1.0 / (np.pi * np.sqrt(t * (1 - t)))
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _multi_stage_optimize(a_init, deadline, min_time_remaining=60.0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(best_a)

    # Stage 1: L-BFGS
    t_rem = deadline - min_time_remaining - time.monotonic()
    if t_rem > 20:
        a1 = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
        sc1 = _score(a1)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1

    # Stage 2: Active-set QP
    t_rem = deadline - min_time_remaining - time.monotonic()
    if t_rem > 30:
        a2, sc2 = _newton_active_polish(best_a, deadline - min_time_remaining,
                                         max_rounds=80)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2

    # Stage 3: Peak peel
    t_rem = deadline - min_time_remaining - time.monotonic()
    if t_rem > 15:
        a3, sc3 = _peak_peel(best_a, deadline - min_time_remaining)
        if sc3 < best_sc:
            best_sc = sc3
            best_a = a3

    # Stage 4: Flatten polish
    t_rem = deadline - min_time_remaining - time.monotonic()
    if t_rem > 10:
        a4, sc4 = _flatten_polish(best_a, deadline - min_time_remaining)
        if sc4 < best_sc:
            best_sc = sc4
            best_a = a4

    # Stage 5: Final active-set with tight tolerance
    t_rem = deadline - min_time_remaining - time.monotonic()
    if t_rem > 10:
        a5, sc5 = _newton_active_polish(best_a, deadline - min_time_remaining + 5,
                                         max_rounds=60)
        if sc5 < best_sc:
            best_sc = sc5
            best_a = a5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with active-set QP, peak peeling,
    flattening polish, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # Phase 1: Primary optimization on parent or best available n
    print(f"\n[Phase 1] Primary optimization (t={time.monotonic() - t_start:.1f}s)")
    if parent is not None:
        n_primary = n_parent
    else:
        n_primary = 16384
        parent = _make_profile(n_primary, "tapered")
        print(f"No parent, using tapered n={n_primary}")
        update_best(parent, _score(parent), "init")

    a_primary = _multi_stage_optimize(parent, deadline, min_time_remaining=300.0)
    sc_primary = _score(a_primary)
    update_best(a_primary, sc_primary, "primary_opt")

    # Phase 2: Try different initializations at same n
    print(f"\n[Phase 2] Alternative inits (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        alts = []
        alts.append(("tapered", _make_profile(n_primary, "tapered")))
        alts.append(("arcsine", _make_profile(n_primary, "arcsine")))
        alts.append(("flat_spec", _flat_spectrum_init(n_primary, n_iter=30)))

        for name, a_alt in alts:
            if time.monotonic() > deadline - 100:
                break
            sc_alt = _score(a_alt)
            # Only optimize if it's close to best
            if sc_alt < best_score + 0.05:
                a_opt = _run_lbfgs(a_alt, deadline, min_time_remaining=120.0)
                sc_opt = _score(a_opt)
                update_best(a_opt, sc_opt, f"alt_{name}_lbfgs")

                if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 80:
                    a_pol, sc_pol = _newton_active_polish(
                        a_opt, deadline - 80, max_rounds=50)
                    update_best(a_pol, sc_pol, f"alt_{name}_qp")

    # Phase 3: Shrinkage (zero out small elements, re-optimize)
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        a_shrunk = best_seq.copy()
        a_sorted_idx = np.argsort(a_shrunk)
        for frac in [0.01, 0.03, 0.05, 0.10]:
            n_zero = max(1, int(n * frac))
            a_try = best_seq.copy()
            a_try[a_sorted_idx[:n_zero]] = 0.0
            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_score - 1e-12:
                update_best(a_try, sc_try, f"shrink_{frac}")
                if time.monotonic() < deadline - 60:
                    a_reopt = _run_lbfgs(a_try, deadline, min_time_remaining=80.0)
                    sc_reopt = _score(a_reopt)
                    update_best(a_reopt, sc_reopt, f"shrink_{frac}_reopt")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = []
    if n_primary != 8192:
        n_candidates.append(8192)
    if n_primary != 32768:
        n_candidates.append(32768)
    if n_primary != 2048:
        n_candidates.append(2048)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.02 and time.monotonic() < deadline - 40:
            a_np, sc_np = _newton_active_polish(a_opt, deadline - 40, max_rounds=40)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 5: Perturbation restarts
    print(f"\n[Phase 5] Perturbation restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 15:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
            sc_pert = _score(a_pert)
            if sc_pert < best_score + 0.03:
                a_opt = _run_lbfgs(a_pert, deadline, min_time_remaining=15.0)
                sc_opt = _score(a_opt)
                update_best(a_opt, sc_opt, f"perturb_{sigma}")

    # Phase 6: Final ultra-tight polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=80)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 2:
            a_fl, sc_fl = _flatten_polish(best_seq, deadline - 1, max_iters=50)
            update_best(a_fl, sc_fl, "final_flatten")

        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-7
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(1000, max(30, int(remaining / 0.01)))
                try:
                    obj_fn = _make_lsmoobj(beta_fin, n)
                    grad_fn = _make_lsgrad(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-14, 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        if time.monotonic() < deadline - 2:
            a_pp, sc_pp = _peak_peel(best_seq, deadline - 1, max_iters=50)
            update_best(a_pp, sc_pp, "final_peel")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
