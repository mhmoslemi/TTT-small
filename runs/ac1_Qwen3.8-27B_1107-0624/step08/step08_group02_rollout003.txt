    out_a = _hankel_row(a, w, n)

    coeff = (4.0 * n / p) * sum_cp_pos ** (1.0 / p - 1.0)
    grad = coeff * a * (out_a + out_b)
    
    return obj, grad


def _hankel_row(a, w, n):
    """Compute h[j] = sum_k w[k] * a[j + (n-1-k)] for j=0..n-1.
    This is the middle part of convolve(w, a) shifted appropriately."""
    # w has length 2n-1, a has length n
    # conv(w, a) has length 3n-2
    # We want indices n-1 to 2n-2 of conv(w, a)
    out_n = len(w) + len(a) - 1
    N = _next_pow2(out_n)
    fw = rfft(w, N)
    fa = rfft(a, N)
    conv = irfft(fw * fa, N)[:out_n]
    return conv[n - 1:2 * n - 1]


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i][j] = W[i-j+n-1] (Hankel matrix)."""
    v_rev = v[::-1]
    out_n = len(v_rev) + len(W) - 1
    N = _next_pow2(out_n)
    f_v = rfft(v_rev, N)
    f_w = rfft(W, N)
    conv = irfft(f_v * f_w, N)[:out_n]
    return 2.0 * conv[n - 1:2 * n - 1]


def _pcg_solve(W, n, rhs, max_iter=600, tol=1e-12):
    """Preconditioned CG for Hankel system Hx = rhs, where H[i][j] = W[|i-j|+n-1] roughly."""
    x = np.zeros(n)
    r = rhs.copy()
    # Diagonal preconditioner: M[i] = H[i][i] = W[n-1] (constant for this Hankel structure)
    # Actually for a general W, H[i][i] = W[n-1] for all i, so M = W[n-1]*I
    # That's not a useful preconditioner. Let's use a different one.
    # Try: preconditioner based on the diagonal of H, which is W[n-1] for all i.
    # So M^{-1} = I / W[n-1]. This just scales, not very helpful.
    # Better: use the first few CG iterations as a mini-preconditioner (incomplete Cholesky-like)
    # For simplicity, just do regular CG with more iterations.
    
    z = r.copy()
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rz_new = float(r_new @ r_new)
        if not np.isfinite(rz_new) or rz_new > rz * 1.5:
            break
        x = x_new
        r = r_new
        beta_cg = rz_new / max(rz, 1e-30)
        p = (beta_cg) * p + r
        rz = rz_new
        if rz < tol * tol * rz0:
            break
    
    return x


def _newton_polish_v2(a_init, deadline, max_rounds=200, active_tols=None):
    """Improved Newton active-set method with hysteresis and multiple tolerance tries."""
    if active_tols is None:
        active_tols = [1e-5, 5e-5, 1e-4, 3e-4, 1e-3, 3e-3]
    
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    current_tol_idx = 0
    tol_stuck = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set with hysteresis
        tol = active_tols[min(current_tol_idx, len(active_tols) - 1)] * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        
        if n_active < 1:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        # If stuck at current tolerance, try next
        if tol_stuck > 8 and current_tol_idx < len(active_tols) - 1:
            current_tol_idx += 1
            tol_stuck = 0
            continue
        tol_stuck += 1 if no_improve > 3 else 0
        
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break
        
        # Solve Hx = g using CG
        Hg = _pcg_solve(W, n, g, max_iter=600)
        H1 = _pcg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)
        
        # Line search with relaxed Armijo condition
        alpha = 1.0
        step_done = False
        c_max_before = c_max
        for ls in range(100):
            a_try = a + alpha * dx
            n_neg = int((a_try < 0).sum())
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Relaxed Armijo: allow small increases
            if c_max_try < c_max_before * (1.0 - 0.002 * alpha) + 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break
        
        if not step_done:
            # Try subgradient directions with various magnitudes
            best_sub_sc = np.inf
            best_sub_a = None
            for trial in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sub_sc:
                    best_sub_sc = sc_try
                    best_sub_a = a_try.copy()
            
            if best_sub_a is not None and best_sub_sc < _score(a) - 1e-14:
                a = best_sub_a
                step_done = True
        
        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            # If improving well, try tighter tolerance
            if current_tol_idx > 0 and no_improve == 0 and rnd % 10 == 0:
                tol_stuck = 0
            else:
                tol_stuck = 0
        else:
            no_improve += 1
        
        if not step_done and no_improve > 15:
            break
        if no_improve > 40:
            break
    
    return best_a, best_sc


def _pnorm_optimize(z_init, n, deadline, p_start=20, p_end=500, min_time_remaining=30.0):
    """Optimize using p-norm smoothing with adaptive p."""
    current_z = z_init.copy()
    current_a = _softmax(current_z)
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    p = p_start
    p_mult = 2.0
    
    while p < p_end:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.004 * scale
        max_iter = min(5000, max(20, int(remaining / (est_per_iter * 30))))
        
        # Define objective and gradient for this p
        def pobj(z):
            a_tmp = _softmax(z)
            c = _full_convolve(a_tmp, a_tmp)
            c_safe = np.maximum(c, 1e-30)
            cp = c_safe ** p
            sum_cp = max(cp.sum(), 1e-300)
            return float(2.0 * n * sum_cp ** (1.0 / p))
        
        def pgrad(z):
            a_tmp = _softmax(z)
            c = _full_convolve(a_tmp, a_tmp)
            c_safe = np.maximum(c, 1e-30)
            w = c_safe ** (p - 1)
            sum_cp = max((c_safe ** p).sum(), 1e-300)
            coeff = (4.0 * n / p) * sum_cp ** (1.0 / p - 1.0)
            h = _hankel_row(a_tmp, w, n)
            grad = coeff * a_tmp * 2.0 * h
            return grad
        
        try:
            result = minimize(
                pobj, current_z, jac=pgrad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                current_z = result.x.copy()
                current_z = current_z - np.mean(current_z)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    print(f"    p-norm p={p}: score={sc_new:.10f}")
        except Exception as e:
            pass
        
        # Increase p
        p = int(p * p_mult)
        p_mult = max(1.3, p_mult * 0.95)
    
    return best_a


def _deflation_step(a, deadline, max_iters=50):
    """Equalize active peaks by iterative deflation."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_lags = np.where(active)[0]
        
        # For each active lag k, the contribution to c[k] from a[j] is:
        # c[k] = sum_i a[i]*a[i+k-n+1] (with the full conv indexing)
        # Actually for full conv of length n with itself: c[m] = sum_i a[i]*a[i-(m-n+1)]
        # where valid indices are those where both are in [0,n-1]
        
        # The "gradient" of c[k] w.r.t. a is: g_k[i] = a[i + (k-(n-1))] + a[i - (k-(n-1))]
        # (with out-of-range = 0)
        
        # Deflation: reduce a where it contributes most to active peaks
        # Compute aggregate gradient
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()
        g = _hankel_matvec(a, W, n)
        g_centered = g - np.mean(g)
        
        # Small step
        step = 0.01 * np.exp(-0.02 * it)
        a_new = a - step * g_centered / max(np.linalg.norm(g_centered), 1e-30)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            step2 = step * 0.1
            a_try = a - step2 * g_centered / max(np.linalg.norm(g_centered), 1e-30)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
    
    return best_a


def _multi_scale_optimize(a_init, n_target, deadline, min_time_remaining=60.0):
    """Multi-scale: downsample, optimize, upsample, optimize."""
    n_init = len(a_init)
    scales = []
    # Generate scales from small to n_target
    s = max(64, n_init // 16)
    while s < n_target:
        scales.append(s)
        s = min(s * 2, n_target)
    scales.append(n_target)
    
    current_a = a_init.copy()
    
    for target_n in scales:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Resample to target_n
        if len(current_a) != target_n:
            x_old = np.linspace(0, 1, len(current_a))
            x_new = np.linspace(0, 1, target_n)
            current_a = np.interp(x_new, x_old, current_a)
            current_a = np.maximum(current_a, 0)
            s = current_a.sum()
            if s > 1e-15:
                current_a = current_a / s
        
        # Brief optimization at this scale
        time_alloc = min(remaining * 0.3, 60.0)
        a_opt, sc = _newton_polish_v2(current_a, time.monotonic() + time_alloc, max_rounds=30)
        if sc < _score(current_a):
            current_a = a_opt
    
    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "water":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_to(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - np.mean(z)
    return _softmax(z)


def _lbfgs_logsumexp(a_init, deadline, min_time_remaining=30.0):
    """Standard log-sum-exp smoothed optimization (backup method)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    beta = base_beta
    min_beta = max_c * 1e-9
    prev_sc = current_sc
    
    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        def lse_obj(zv):
            av = _softmax(zv)
            cv = _full_convolve(av, av)
            cm = float(cv.max())
            shifted = (cv - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            sv = e.sum()
            if sv < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(sv))
        
        def lse_grad(zv):
            av = _softmax(zv)
            cv = _full_convolve(av, av)
            cm = float(cv.max())
            shifted = (cv - cm) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            sv = e.sum()
            if sv < 1e-300:
                w = np.zeros_like(cv)
                w[int(np.argmax(cv))] = 1.0
            else:
                w = e / sv
            a_rev = av[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_val = float(np.dot(w, cv))
            return 4.0 * n * av * (h - W_val)
        
        scale = max(1.0, n / 8192.0)
        max_iter = min(5000, max(30, int(remaining / (0.005 * scale * 20))))
        
        try:
            result = minimize(
                lse_obj, z, jac=lse_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = result.x - np.mean(result.x)
        except Exception:
            pass
        
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]
        
        sc_improved = prev_sc - current_sc
        prev_sc = current_sc
        
        if new_gap < 1e-7 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.2
        else:
            beta *= 0.4
        beta = min(beta, c_max_new * 0.001)
    
    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """Main search function combining multiple optimization strategies."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass
    
    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent found, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    start_a = parent.copy()
    n_work = len(start_a)
    
    # Phase 1: Quick Newton polish on parent (up to 200s)
    print(f"\n[Phase 1] Newton polish on parent (n={n_work})")
    phase1_deadline = min(t_start + 200.0, deadline - 300.0)
    if time.monotonic() < phase1_deadline:
        a_p1, sc_p1 = _newton_polish_v2(start_a, phase1_deadline, max_rounds=150)
        update_best(a_p1, sc_p1, "newton_p1")
        if sc_p1 < _score(start_a):
            start_a = a_p1
    
    # Phase 2: P-norm optimization (different landscape)
    print(f"\n[Phase 2] P-norm optimization")
    phase2_deadline = min(t_start + 500.0, deadline - 200.0)
    if time.monotonic() < phase2_deadline:
        z_init = np.log(np.maximum(start_a, 1e-30))
        z_init = z_init - np.mean(z_init)
        a_p2 = _pnorm_optimize(z_init, n_work, phase2_deadline, p_start=10, p_end=200,
                               min_time_remaining=100.0)
        update_best(a_p2, _score(a_p2), "pnorm")
        if _score(a_p2) < _score(start_a):
            start_a = a_p2
    
    # Phase 3: Log-sum-exp L-BFGS (classic approach, different from p-norm)
    print(f"\n[Phase 3] Log-sum-exp L-BFGS")
    phase3_deadline = min(t_start + 750.0, deadline - 150.0)
    if time.monotonic() < phase3_deadline:
        a_p3 = _lbfgs_logsumexp(start_a, phase3_deadline, min_time_remaining=100.0)
        update_best(a_p3, _score(a_p3), "lse_lbfgs")
        if _score(a_p3) < _score(start_a):
            start_a = a_p3
    
    # Phase 4: Deflation step
    print(f"\n[Phase 4] Deflation")
    if time.monotonic() < deadline - 150:
        a_p4 = _deflation_step(start_a, deadline - 150, max_iters=100)
        update_best(a_p4, _score(a_p4), "deflation")
        if _score(a_p4) < _score(start_a):
            start_a = a_p4
    
    # Phase 5: Multi-scale optimization
    print(f"\n[Phase 5] Multi-scale")
    if time.monotonic() < deadline - 150:
        a_p5 = _multi_scale_optimize(start_a, n_work, deadline, min_time_remaining=120.0)
        update_best(a_p5, _score(a_p5), "multi_scale")
        if _score(a_p5) < _score(start_a):
            start_a = a_p5
    
    # Phase 6: Newton polish with different active set tolerances
    print(f"\n[Phase 6] Newton with varied tolerances")
    if time.monotonic() < deadline - 100:
        for tol_set, label in [
            ([5e-6, 1e-5, 5e-5, 1e-4, 5e-4], "tight"),
            ([1e-4, 5e-4, 1e-3, 5e-3, 1e-2], "loose"),
        ]:
            if time.monotonic() < deadline - 80:
                a_p6, sc_p6 = _newton_polish_v2(start_a, deadline - 80, max_rounds=80,
                                                  active_tols=tol_set)
                update_best(a_p6, sc_p6, f"newton_{label}")
                if sc_p6 < _score(start_a):
                    start_a = a_p6
    
    # Phase 7: Diversified restarts from best
    print(f"\n[Phase 7] Diversified restarts")
    n_restarts = 4
    for i in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break
        sigma = 0.02 + 0.03 * i
        a_pert = _perturb(best_seq if best_seq is not None else start_a, sigma, rng)
        
        # Quick Newton on perturbed
        if time.monotonic() < deadline - 50:
            a_np, sc_np = _newton_polish_v2(a_pert, deadline - 50, max_rounds=50)
            update_best(a_np, sc_np, f"restart_{i}")
            
            # Also try p-norm on perturbed
            if sc_np < best_score + 0.005 and time.monotonic() < deadline - 40:
                z_r = np.log(np.maximum(a_np, 1e-30))
                z_r = z_r - np.mean(z_r)
                a_pr = _pnorm_optimize(z_r, len(a_np), deadline - 30, p_start=20, p_end=100,
                                       min_time_remaining=20.0)
                update_best(a_pr, _score(a_pr), f"restart_{i}_pnorm")
    
    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n exploration")
    n_candidates = [n_work * 2, n_work // 2, 32768]
    n_candidates = [x for x in n_candidates if 256 <= x <= 65536 and x != n_work]
    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 50:
            break
        a_resampled = _resample_to(best_seq if best_seq is not None else start_a, n_try)
        
        # Quick optimization
        if time.monotonic() < deadline - 40:
            a_opt, sc_opt = _newton_polish_v2(a_resampled, deadline - 40, max_rounds=40)
            update_best(a_opt, sc_opt, f"n_{n_try}")
    
    # Phase 9: Final intensive Newton polish
    print(f"\n[Phase 9] Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final, sc_final = _newton_polish_v2(best_seq, deadline - 2, max_rounds=100)
        update_best(a_final, sc_final, "final_newton")
        
        # One more deflation
        if time.monotonic() < deadline - 1:
            a_def = _deflation_step(a_final, deadline - 0.5, max_iters=20)
            update_best(a_def, _score(a_def), "final_deflation")
    
    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
