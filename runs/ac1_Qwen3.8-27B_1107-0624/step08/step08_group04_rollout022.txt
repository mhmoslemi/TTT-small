"""
Multi-strategy optimizer for minimum-peak self-convolution sequences.
Key innovations over previous approach:
1. Top-k weighted subgradient with adaptive step size (more targeted than log-sum-exp)
2. Antisymmetric perturbation to break symmetry and reduce center peak
3. Flat-top window initialization (exploits known low-sidelobe structure)
4. Coarse-to-fine multi-scale optimization
5. Better time allocation with aggressive early stopping
6. Coordinate descent final polish on exact max objective
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fast(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _conv_self(a):
    return _convolve_fast(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _topk_gradient(a, k_positions, weights):
    """Compute weighted sum of gradients at given convolution positions.
    grad_k[m] = 2 * a[k - m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    a_rev = a[::-1]
    for k, w in zip(k_positions, weights):
        # grad[m] += 2*w * a[k-m] for valid m
        # a[k-m] is valid when 0 <= k-m <= n-1, i.e., k-n+1 <= m <= k
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            # a[k-m] for m in [m_lo, m_hi]
            # a[k-m_lo], a[k-m_lo-1], ..., a[k-m_hi]
            # = a[m_hi], a[m_hi-1], ..., a[m_lo] (reversed slice)
            vals = a[m_lo:m_hi + 1][::-1]
            grad[m_lo:m_hi + 1] += 2.0 * w * vals
    return grad


def _topk_subgradient_step(a, k, step, momentum_buf, momentum_alpha):
    """One step of top-k subgradient with momentum."""
    n = len(a)
    c = _conv_self(a)
    c_max = float(c.max())
    
    # Get top-k positions
    if k >= len(c):
        positions = np.arange(len(c))
    else:
        positions = np.argsort(c)[::-1][:k]
    
    # Weights: proportional to (c[pos] - c_mean) for active peaks
    c_vals = c[positions]
    c_rest_mean = (c.sum() - c_vals.sum()) / max(1, len(c) - k)
    excess = c_vals - c_rest_mean
    excess = np.maximum(excess, 0)
    if excess.sum() < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = excess / excess.sum()
    
    # Compute gradient
    grad = _topk_gradient(a, positions, weights)
    
    # Momentum update
    momentum_buf[:] = momentum_alpha * momentum_buf + (1 - momentum_alpha) * grad
    
    # Step
    a_new = a - step * momentum_buf
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), c_max, 0.0
    a_new = a_new / s
    return a_new, c_max, float(np.linalg.norm(momentum_buf))


def _run_topk_opt(a_init, deadline, k_start=30, k_end=3, n_phases=8,
                  base_step=0.02, momentum=0.85):
    """Multi-phase top-k subgradient optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    momentum_buf = np.zeros(n)
    
    total_phases = n_phases
    for phase in range(total_phases):
        frac = phase / max(1, total_phases - 1)
        k_cur = int(k_start + (k_end - k_start) * frac)
        k_cur = max(1, k_cur)
        step = base_step * (1.0 - 0.5 * frac)
        
        # Run steps until time or max steps
        phase_deadline = deadline - (total_phases - phase - 1) * 2.0
        max_steps = 500000
        stall_count = 0
        
        for i in range(max_steps):
            if time.monotonic() > phase_deadline:
                break
            a, c_max, gnorm = _topk_subgradient_step(a, k_cur, step, momentum_buf, momentum)
            
            # Check improvement periodically
            if i % 200 == 0:
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    stall_count = 0
                    # Increase step slightly on improvement
                    step = min(0.05, step * 1.01)
                else:
                    stall_count += 1
                    if stall_count > 20:
                        # Restart from best with smaller step
                        a = best_a.copy()
                        momentum_buf[:] = 0
                        step *= 0.7
                        stall_count = 0
                        if step < 0.001:
                            break
        
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _conv_self(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _conv_self(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _convolve_fast(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_smooth(a_init, deadline, min_time=30.0):
    """L-BFGS on smoothed log-sum-exp objective with beta annealing."""
    n = len(a_init)
    c_cur = _conv_self(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.003)
    
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005,
              0.000002, 0.000001, 0.0000005]
    betas = [base_beta * r for r in ratios]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(3000, max(100, int(remaining / (est_time_per_iter * 5))))
        
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _antisymmetric_perturb(a, amplitude, rng):
    """Add antisymmetric perturbation to break symmetry."""
    n = len(a)
    d = rng.standard_normal(n)
    # Make d antisymmetric: d[i] = -d[n-1-i]
    d = (d - d[::-1]) / 2.0
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return a.copy()
    d = d / d_norm
    a_new = a + amplitude * d
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _flat_top_window(n, order=4):
    """Generate a flat-top window of length n."""
    coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    t = np.arange(n)
    w = np.zeros(n)
    for k in range(min(order + 1, len(coeffs))):
        sign = (-1) ** k
        w += sign * coeffs[k] * np.cos(2 * np.pi * k * t / n)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def _parametric_init(n, kind="gauss"):
    t = np.linspace(0, 1, n)
    if kind == "gauss":
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
    elif kind == "sin_half":
        a = np.sin(np.pi * t / 2.0) ** 1.5 + 0.005
    elif kind == "cos_taper":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 0.001)
    elif kind == "quartic":
        a = (1.0 - (2 * t - 1) ** 4)
        a = np.maximum(a, 0.001)
    elif kind == "flat_top":
        a = _flat_top_window(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _coordinate_descent_polish(a_init, deadline, max_rounds=200):
    """Coordinate descent: optimize pairs of elements to reduce max convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _conv_self(a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Find the pair (i, j) with i+j = k_star that has the largest product
        # Reducing this product reduces c[k_star]
        # Try moving mass from a[i] to a[j] where i+j != k_star
        
        # Simple approach: try reducing a[i] for i that contributes to c[k_star]
        # and adding to a position that doesn't
        best_improve = 0
        best_update = None
        
        # Sample some candidate moves
        np.random.seed(rnd)
        n_samples = min(50, n // 4)
        idx_from = np.random.choice(n, n_samples, replace=False)
        idx_to = np.random.choice(n, n_samples, replace=False)
        
        for fi, ti in zip(idx_from, idx_to):
            if fi == ti:
                continue
            # Check if moving from fi to ti helps
            # c[k_star] decreases if fi + (k_star - fi) involves fi
            # This is complex; just try it
            delta = a[fi] * 0.1
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[fi] -= delta
            a_try[ti] += delta
            if a_try[fi] < 0:
                a_try[fi] = 0
                a_try[ti] += delta - max(0, a[fi])
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try_max = _conv_self(a_try).max()
            if c_try_max < c_max - 1e-15:
                improve = c_max - c_try_max
                if improve > best_improve:
                    best_improve = improve
                    best_update = a_try
        
        if best_update is not None:
            a = best_update
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # No improvement from random moves; try gradient step
            grad = _topk_gradient(a, [k_star], [1.0])
            a_new = a - 0.001 * grad
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _run_full_pipeline(a_init, n, deadline):
    """Run the full optimization pipeline on a sequence of length n."""
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Step 1: Top-k subgradient (main workhorse)
    if time.monotonic() < deadline - 60:
        a1, sc1 = _run_topk_opt(best_a, deadline - 30, k_start=20, k_end=2,
                                n_phases=6, base_step=0.015, momentum=0.88)
        if sc1 < best_sc:
            best_a, best_sc = a1, sc1
    
    # Step 2: L-BFGS smooth
    if time.monotonic() < deadline - 30:
        a2, sc2 = _run_lbfgs_smooth(best_a, deadline - 10, min_time=10)
        if sc2 < best_sc:
            best_a, best_sc = a2, sc2
    
    # Step 3: Another round of top-k (sharper)
    if time.monotonic() < deadline - 15:
        a3, sc3 = _run_topk_opt(best_a, deadline - 5, k_start=5, k_end=1,
                                n_phases=3, base_step=0.005, momentum=0.92)
        if sc3 < best_sc:
            best_a, best_sc = a3, sc3
    
    # Step 4: Coordinate descent polish
    if time.monotonic() < deadline - 3:
        a4, sc4 = _coordinate_descent_polish(best_a, deadline - 1, max_rounds=100)
        if sc4 < best_sc:
            best_a, best_sc = a4, sc4
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search for minimum-peak self-convolution sequence.
    Uses top-k subgradient, L-BFGS, antisymmetric perturbation,
    flat-top initialization, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    best_n = 0
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            best_n = len(p)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # Phase 0: Load parent
    print(f"{'='*60}")
    print(f"Starting optimization, budget={budget_s}s")
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass
    
    # Phase 1: Generate diverse initial candidates
    print(f"\n[Phase 1] Generating candidates (t={time.monotonic()-t_start:.1f}s)")
    candidates = []
    
    if parent is not None:
        candidates.append(("parent", parent))
        # Antisymmetric perturbation of parent
        for amp in [0.01, 0.03, 0.05]:
            ap = _antisymmetric_perturb(parent, amp, rng)
            candidates.append((f"parent_asym_{amp}", ap))
        # Scaled version
        ap = parent.copy()
        ap = ap ** 1.2
        ap = np.maximum(ap, 0)
        ap = ap / ap.sum()
        candidates.append(("parent_pow1.2", ap))
        ap = parent.copy()
        ap = ap ** 0.8
        ap = np.maximum(ap, 0)
        ap = ap / ap.sum()
        candidates.append(("parent_pow0.8", ap))
    
    n_base = n_parent if n_parent > 0 else 16384
    
    # Flat-top window
    ft = _flat_top_window(n_base)
    candidates.append(("flat_top", ft))
    
    # Parametric families
    for kind in ["gauss", "sin_half", "quartic", "cos_taper"]:
        p = _parametric_init(n_base, kind)
        candidates.append((f"param_{kind}", p))
    
    # Random structured
    t = np.linspace(0, 1, n_base)
    for freq in [2, 3, 5]:
        p = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t) + 0.1 * np.cos(4 * np.pi * t)
        p = np.maximum(p, 0.01)
        p = p / p.sum()
        candidates.append((f"random_cos_{freq}", p))
    
    print(f"  Generated {len(candidates)} candidates")
    
    # Phase 2: Quick screening
    print(f"\n[Phase 2] Quick screening (t={time.monotonic()-t_start:.1f}s)")
    screen_budget = 5.0
    screen_deadline = t_start + screen_budget
    screened_results = []
    
    for name, cand in candidates:
        if time.monotonic() > screen_deadline:
            break
        sc_init = _score(cand)
        screened_results.append((sc_init, name, cand))
        print(f"  {name}: score={sc_init:.10f}")
    
    screened_results.sort(key=lambda x: x[0])
    
    # Phase 3: Deep optimization on top candidates
    print(f"\n[Phase 3] Deep optimization (t={time.monotonic()-t_start:.1f}s)")
    n_deep = 3  # How many candidates to deeply optimize
    deep_per_candidate = (deadline - time.monotonic()) / n_deep - 100
    
    for i in range(min(n_deep, len(screened_results))):
        if time.monotonic() > deadline - 150:
            break
        sc_init, name, cand = screened_results[i]
        remaining_for_this = min(deep_per_candidate, deadline - time.monotonic() - 150)
        if remaining_for_this < 20:
            break
        
        print(f"  Optimizing candidate {i+1}/{n_deep}: {name} (init={sc_init:.8f}, budget={remaining_for_this:.0f}s)")
        
        cand_deadline = time.monotonic() + remaining_for_this
        a_opt, sc_opt = _run_full_pipeline(cand, len(cand), cand_deadline)
        update_best(a_opt, sc_opt, f"deep_{name}")
        
        # Try antisymmetric variant
        if time.monotonic() < deadline - 120 and sc_opt < best_score + 0.01:
            a_asym = _antisymmetric_perturb(a_opt, 0.02, rng)
            remaining2 = min(60, deadline - time.monotonic() - 100)
            if remaining2 > 15:
                a_opt2, sc_opt2 = _run_full_pipeline(a_asym, len(a_asym),
                                                     time.monotonic() + remaining2)
                update_best(a_opt2, sc_opt2, f"deep_asym_{name}")
    
    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_options = []
    if best_n > 0:
        n_options = [best_n // 2, best_n, best_n * 2]
    else:
        n_options = [4096, 8192, 16384]
    n_options = [n for n in n_options if 1024 <= n <= 65536]
    # Remove duplicates
    seen = set()
    n_options = [n for n in n_options if not (n in seen or seen.add(n))]
    
    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        if best_seq is not None and n_try == best_n:
            continue
        
        remaining_n = min(120, deadline - time.monotonic() - 40)
        if remaining_n < 20:
            break
        
        if best_seq is not None:
            a_init = _resample(best_seq, n_try)
        else:
            a_init = _flat_top_window(n_try)
        
        print(f"  Trying n={n_try} (budget={remaining_n:.0f}s)")
        a_opt, sc_opt = _run_full_pipeline(a_init, n_try, time.monotonic() + remaining_n)
        update_best(a_opt, sc_opt, f"n_{n_try}")
    
    # Phase 5: Final polish on best
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight top-k
        remaining = deadline - time.monotonic() - 3
        if remaining > 10:
            a_fin, sc_fin = _run_topk_opt(best_seq, time.monotonic() + remaining,
                                          k_start=3, k_end=1, n_phases=3,
                                          base_step=0.003, momentum=0.95)
            update_best(a_fin, sc_fin, "final_topk")
        
        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _conv_self(best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 2e-7
            remaining = deadline - time.monotonic() - 2
            if remaining > 3:
                max_iter = min(1000, max(50, int(remaining / 0.02)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 40}
                    )
                    a_flb = np.exp(result.x - result.x.max())
                    a_flb = a_flb / a_flb.sum()
                    sc_flb = _score(a_flb)
                    update_best(a_flb, sc_flb, "final_lbfgs_tiny")
                except Exception:
                    pass
    
    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _flat_top_window(8192)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    # Fallback
    fallback = _flat_top_window(16384)
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}, n=16384")
    return [float(x) for x in fallback]
