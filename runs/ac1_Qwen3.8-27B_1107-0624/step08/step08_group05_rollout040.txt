def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

    n = len(a)
    s = float(a.sum())

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Multi-stage optimizer for minimizing step function autocorrelation peak.
Uses LSE smoothing with adaptive beta schedule, Newton active-set refinement,
subgradient polishing, and perturbation-based global exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    return 2.0 * n * float(np.sum(cp)) ** (1.0 / p)


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p - 1)
    sum_cp = float(np.sum(cp))
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = cp / sum_cp
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * p
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)
        N = _next_pow2(4 * n)
        a_rev = a[::-1]
        fb = np.fft.rfft(active_f, N)

        def _matvec(v):
            v_rev = v[::-1]
            fv = np.fft.rfft(v_rev, N)
            fr = np.fft.irfft(fv * fb, N)
            return 2.0 * fr[n - 1:2 * n - 1]

        grad = _matvec(a)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-6

        def _reg_matvec(v):
            return _matvec(v) + mu * v

        x = np.zeros(n)
        r_vec = grad_t.copy()
        p_vec = r_vec.copy()
        rs = float(np.dot(r_vec, r_vec))
        b_norm = gnorm
        for _ in range(150):
            Hp = _reg_matvec(p_vec)
            pH = float(np.dot(p_vec, Hp))
            if abs(pH) < 1e-30:
                break
            alpha = rs / pH
            x = x + alpha * p_vec
            r_vec = r_vec - alpha * Hp
            r_new = float(np.dot(r_vec, r_vec))
            if np.sqrt(r_new) < 1e-10 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p_vec = r_vec + beta_cg * p_vec
            rs = r_new

        dx = x - np.mean(x)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(30):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.008, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block(a, rng, n_swaps=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(8, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_mixture(a, rng, n_mix=3, alpha=0.3):
    n = len(a)
    result = a.copy()
    for _ in range(n_mix):
        idx1 = int(rng.integers(0, n - 1))
        idx2 = int(rng.integers(0, n - 1))
        if idx1 == idx2:
            continue
        tmp = result[idx1]
        result[idx1] = alpha * result[idx1] + (1 - alpha) * result[idx2]
        result[idx2] = alpha * result[idx2] + (1 - alpha) * tmp
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return a.copy()
    return result / s


def _optimize_lbfgs_lse(a_init, deadline, min_remaining=30.0):

    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = [base_beta * f for f in
             [1.0, 0.7, 0.5, 0.3, 0.2, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]]

    for beta in betas:

        est_time = 0.002 * max(1.0, n / 4096.0)

        try:
