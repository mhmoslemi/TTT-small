"""
Aggressive multi-scale optimizer for minimum peak convolution.
Uses progressive n refinement, Nesterov momentum with adaptive LR,
aggressive active-set gradient descent, and periodic L-BFGS restarts.
Key innovations: momentum-based gradient flow, adaptive peak tracking,
symmetry-constrained optimization, and multi-resolution search.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _log_softmax(z):
    z_max = np.max(z)
    e = np.exp(np.clip(z - z_max, -700, 700))
    return z_max + np.log(e.sum() + 1e-300)

def _grad_smo(z, beta, n):
    """Gradient of smoothed objective in z-space (log-space)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad_a[i] = sum_j w[i+j] * a[j] for j=0..n-1 (where i+j is valid index in c)
    # This is cross-correlation of w and a
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # Gradient of 2n * (c_max + beta * log(sum exp)) w.r.t. a[i]
    # = 2n * sum_k w[k] * a[k-i] (for valid k-i)
    # In z-space: grad_z[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j] / sum(a))
    # Since sum(a) = 1 (softmax), this simplifies
    grad_a = 2.0 * n * h
    # Project to sum-zero (since softmax constraint)
    grad_a_mean = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - grad_a_mean)
    return grad_z

def _run_momentum(z_init, beta_schedule, n, deadline, min_time_remaining=30.0):
    """Nesterov momentum optimizer in z-space."""
    z = z_init.copy().astype(np.float64)
    v = np.zeros_like(z)
    momentum = 0.9
    best_z = z.copy()
    a_best = _softmax(z)
    sc_best = _score(a_best)
    
    t_start = time.monotonic()
    lr_base = 0.1
    lr_decay = 0.999
    
    for beta in beta_schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations we can do
        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(10, int(remaining / (est_time_per_iter * 10))))
        
        lr = lr_base
        for it in range(max_iter):
            if time.monotonic() + min_time_remaining > deadline:

            # Nesterov look-ahead

            g = _grad_smo(z_look, beta, n)

            gnorm = np.linalg.norm(g)

        momentum = max(0.8, momentum * 0.99)
        lr_base *= 0.7

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "periodic5":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "center_dip":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.35 * np.exp(-40.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "center_dip2":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.25 * np.exp(-20.0 * (t - 0.5) ** 2) - 0.1 * np.exp(-10.0 * (t - 0.3) ** 2) - 0.1 * np.exp(-10.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "multi_tone":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.sin(2.0 * np.pi * 3.0 * t / n) + 0.15 * np.sin(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _stochastic_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_lbfgs_deep(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very deep beta annealing for final polish."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    threshold = max_c * 0.98
    n_above = int(np.sum(c_sorted >= threshold))
    if n_above < 3:
        n_above = 3
    gap = c_sorted[0] - c_sorted[min(n_above, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.008, 0.004, 0.002, 0.001,
                                  0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_per_iter = 0.0025 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(30, int(remaining / (est_per_iter * 8))))

        def _obj(z, _beta=beta, _n=n):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / _beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * _n * c_max
            return 2.0 * _n * (c_max + _beta * np.log(s))

        def _grad(z, _beta=beta, _n=n):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / _beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[_n - 1:2 * _n - 1]
            W = float(np.dot(w, c))
            return 4.0 * _n * a * (h - W)

        try:
            result = minimize(
                _obj, z, jac=_grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _momentum_gradient(a_init, deadline, min_time_remaining=30.0):
    """Nesterov momentum gradient descent in log-space with adaptive LR."""
    n = len(a_init)
    a = a_init.copy()
    a = _norm_a(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    momentum = 0.92
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    base_beta = max_c * 0.02

    beta_schedule = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05,
                                          0.025, 0.01, 0.005, 0.002, 0.001])
    lr_starts = [0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.04, 0.03, 0.02, 0.01]

    for bi, (beta, lr_start) in enumerate(zip(beta_schedule, lr_starts)):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_per_iter = 0.0015 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(20, int(remaining / (est_per_iter * 8))))

        lr = lr_start
        decay = 0.9995
        for it in range(max_iter):
            if time.monotonic() + min_time_remaining > deadline:
                break

            z_look = z + momentum * v
            a_look = _softmax(z_look)
            c_look = _full_convolve(a_look, a_look)
            c_max = float(c_look.max())
            shifted = (c_look - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c_look)
                w[int(np.argmax(c_look))] = 1.0
            else:
                w = e / s_e

            a_rev = a_look[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W_dot = float(np.dot(w, c_look))
            grad = 4.0 * n * a_look * (h - W_dot)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break

            lr_eff = lr
            v = momentum * v - lr_eff * grad
            z = z + v

            if it % 100 == 0 or it == max_iter - 1:
                a_cur = _softmax(z)
                sc_cur = _score(a_cur)
                if sc_cur < best_sc:
                    best_sc = sc_cur
                    best_a = a_cur.copy()
            lr *= decay

        momentum = max(0.7, momentum * 0.95)
        v *= 0.3

    return best_a, best_sc


def _active_set_v2(a_init, deadline, min_time_remaining=30.0):
    """Aggressive active-set with adaptive weights and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = _norm_a(a)
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    tol_frac = 0.03
    tol_frac_min = 0.000005
    lr = 0.05
    lr_max = 0.15
    no_improve = 0
    max_stall = 150
    stall_count = 0
    mom_v = np.zeros(n)
    mom_coeff = 0.4

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > n:
            top_idx = np.argsort(c)[::-1][:n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = n

        # Weight by inverse peak height (heavier weight on higher peaks)
        w = np.zeros_like(c)
        active_vals = c[active]
        # Use power weighting: higher peaks get more weight
        w[active] = active_vals ** 2
        w_sum = w[active].sum()
        if w_sum > 1e-30:
            w[active] /= w_sum
        else:
            w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        step = mom_coeff * mom_v - alpha * grad_t
        mom_v = step

        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a + step
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-15:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 15 == 0:
                tol_frac = max(tol_frac_min, tol_frac * 0.6)
                lr *= 0.6
                mom_coeff = max(0.05, mom_coeff * 0.8)
            if no_improve >= max_stall:
                a = best_a.copy()
                a = _norm_a(a)
                tol_frac = max(tol_frac_min, tol_frac * 0.4)
                lr = max(0.001, lr * 0.3)
                mom_v = np.zeros(n)
                mom_coeff = 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 12:
                    break
        else:
            if tol_frac > tol_frac_min and no_improve < 5:
                tol_frac = max(tol_frac_min, tol_frac * 0.93)
            lr = min(lr_max, lr * 1.02)

    return best_a, best_sc


def _peak_shaving(a_init, deadline, min_time_remaining=30.0):
    """Direct peak shaving: identify top contributing elements for each peak
    and redistribute mass to elements that don't contribute to peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = _norm_a(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    n_top = max(3, int(n * 0.005))
    reduce_frac = 0.005
    target_frac = 0.995

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_top]

        d = np.zeros(n)
        for k in top_idx:
            excess = c[k] - c_max * target_frac
            if excess < 1e-15:
                continue

            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            contribs = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                contribs[i] += a[j]
                if i != j:
                    contribs[j] += a[i]

            n_reduce = max(1, int(n * 0.002))
            top_contrib = np.argsort(contribs)[::-1][:n_reduce]
            total_red = 0.0
            for idx in top_contrib:
                if contribs[idx] > 1e-15:
                    reduction = min(excess / (2.0 * contribs[idx]),
                                    a[idx] * 0.03, total_red + excess / (2.0 * contribs[idx] + 1e-30) * 0.01)
                    d[idx] -= reduction
                    total_red += reduction
                    excess -= 2.0 * contribs[idx] * reduction
                    if excess <= 0:
                        break

        if np.abs(d).sum() < 1e-15:
            n_top = min(n // 2, n_top + max(1, int(n * 0.001)))
            reduce_frac *= 0.8
            continue

        # Find elements to add mass to (those with low peak contribution)
        peak_contrib_total = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                peak_contrib_total[i] += a[k - i]

        n_add = max(1, int(np.abs(d).sum() * 0))
        n_add = max(1, int(n * 0.002))
        add_idx = np.argsort(peak_contrib_total)[:n_add]
        total_red = -float(d[d < 0].sum())
        if total_red > 1e-15:
            for idx in add_idx:
                d[idx] += total_red / n_add

        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            target_frac = max(0.999, target_frac + 0.001)
            n_top = min(n // 2, n_top + 1)

    return best_a, best_sc


def _kkt_solve(a_init, active_indices, n, max_iter=100):
    """Try to solve KKT conditions for given active set.
    Returns improved a or None."""
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    m = len(active_indices)

    # Build matrix A_write[i, k] = a_init[k - i] if valid, else 0
    # KKT: sum_k lambda_k * A_write[i, k] = mu for all i in support
    A_write = np.zeros((n, m))
    for idx, k in enumerate(active_indices):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_write[i, idx] = a_init[k - i]

    # Solve: A_write @ lambda = mu * 1 (for free variables)
    # Use pseudo-inverse approach
    # Actually, let's set up: find lambda, mu such that A_write @ lambda + nu = 0
    # with sum(lambda) = 1
    # This is: A_write @ lambda = -nu * 1, sum(lambda) = 1
    # Stack: [A_write; 1^T] [lambda; nu] = [0; 1]
    # But this is underdetermined (m+1 unknowns, n+1 equations, usually m < n)
    # Least squares:
    M = np.vstack([A_write, np.ones((1, m))])
    rhs = np.zeros(n + 1)
    rhs[-1] = 1.0

    try:
        sol, residuals, rank, sv = np.linalg.lstsq(M, rhs, rcond=None)
    except Exception:
        return None

    lam = sol[:m]
    # Ensure lambda >= 0
    lam = np.maximum(lam, 0)
    lam_sum = lam.sum()
    if lam_sum < 1e-10:
        return None
    lam = lam / lam_sum

    # Verify: does this reduce the objective?
    # The improvement comes from the fact that at KKT, the gradient is "flat"
    # on the active set. We can use this as a direction.
    # Compute the "imbalance": for each i, imbalance[i] = sum_k lam[k]*a[k-i] - mean
    imbalance = A_write @ lam
    imb_mean = imbalance.mean()
    direction = -(imbalance - imb_mean)

    # Normalize
    dir_norm = np.linalg.norm(direction)
    if dir_norm < 1e-15:
        return None
    direction = direction / dir_norm

    return direction


def _final_ultra_polish(a_init, deadline):
    """Very fine-grained polish using tiny steps on the single largest peak."""
    n = len(a_init)
    a = a_init.copy()
    a = _norm_a(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    while time.monotonic() < deadline - 2:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()

        # Focus on the single top peak
        k_peak = int(np.argmax(c))

        # Computational gradient for this single peak
        a_rev = a[::-1]
        e_peak = np.zeros(2 * n - 1)
        e_peak[k_peak] = 1.0
        wc = _full_convolve(e_peak, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Very small step
        alpha = min(0.01, 0.1 / (gnorm + 1e-15))
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                break
            alpha *= 0.5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale aggressive optimizer with progressive n refinement,
    momentum-based gradient flow, deep L-BFGS annealing, and
    active-set peak reduction.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        sc = _score(a)
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # Determine n values for progressive approach
    if n_parent >= 16384:
        n_stages = [4096, 8192, 16384, 32768]
    elif n_parent >= 8192:
        n_stages = [4096, 8192, 16384]
    else:
        n_stages = [2048, 4096, 8192, 16384]

    # === Phase 1: Progressive n refinement ===
    print(f"\n[P1] Progressive n refinement (t={time.monotonic()-t_start:.1f}s)")
    time_per_stage = (deadline - t_start - 300) / max(1, len(n_stages))

    current_a = parent.copy()
    for si, n_try in enumerate(n_stages):
        stage_end = min(t_start + time_per_stage * (si + 1) + 60, deadline - 200)
        if time.monotonic() > stage_end - 10:
            continue

        a_stage = _resample(current_a, n_try)
        sc_stage = _score(a_stage)
        print(f"  n={n_try}: init={sc_stage:.10f}, budget={stage_end - time.monotonic():.0f}s")

        # L-BFGS on this n
        lb_end = min(stage_end, time.monotonic() + 0.45 * max(1, stage_end - time.monotonic()))
        a_lb = _run_lbfgs_deep(a_stage, lb_end, min_time_remaining=20.0)
        _try_update(a_lb, f"lbfgs_n{n_try}")

        # Active set
        if time.monotonic() < stage_end - 30:
            act_end = min(stage_end, time.monotonic() + 0.3 * max(1, stage_end - time.monotonic()))
            a_act, sc_act = _active_set_v2(a_lb, act_end, min_time_remaining=20.0)
            _try_update(a_act, f"active_n{n_try}")
            a_stage_best = a_act if sc_act < _score(a_lb) else a_lb
        else:
            a_stage_best = a_lb

        # Peak shaving
        if time.monotonic() < stage_end - 20:
            ps_end = min(stage_end - 5, time.monotonic() + 0.2 * max(1, stage_end - time.monotonic()))
            a_ps, sc_ps = _peak_shaving(a_stage_best, ps_end, min_time_remaining=15.0)
            _try_update(a_ps, f"peaks_n{n_try}")

        current_a = _resample(best_seq, n_stages[min(si + 1, len(n_stages) - 1)]) if si + 1 < len(n_stages) else best_seq

    # === Phase 2: Momentum gradient on best ===
    print(f"\n[P2] Momentum gradient (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    if time.monotonic() < deadline - 250:
        mom_end = min(deadline - 250, time.monotonic() + 200)
        a_mom, sc_mom = _momentum_gradient(best_seq, mom_end, min_time_remaining=30.0)
        _try_update(a_mom, "momentum")

    # === Phase 3: Deep L-BFGS ===
    print(f"\n[P3] Deep L-BFGS (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    if time.monotonic() < deadline - 200:
        lb_end = min(deadline - 200, time.monotonic() + 200)
        a_lb2 = _run_lbfgs_deep(best_seq, lb_end, min_time_remaining=30.0)
        _try_update(a_lb2, "lbfgs_deep")

    # === Phase 4: Active set aggressive ===
    print(f"\n[P4] Active set (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    if time.monotonic() < deadline - 150:
        act_end = min(deadline - 150, time.monotonic() + 150)
        a_act2, sc_act2 = _active_set_v2(best_seq, act_end, min_time_remaining=20.0)
        _try_update(a_act2, "active_v2")

    # === Phase 5: Peak shaving on best ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    if time.monotonic() < deadline - 100:
        ps_end = min(deadline - 100, time.monotonic() + 100)
        a_ps2, sc_ps2 = _peak_shaving(best_seq, ps_end, min_time_remaining=20.0)
        _try_update(a_ps2, "peak_shave")

    # === Phase 6: Try different profiles blended with best ===
    print(f"\n[P6] Profile blending (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    n_final = len(best_seq)
    profiles = ["center_dip", "center_dip2", "multi_tone", "cosine_mod"]
    for prof in profiles:
        if time.monotonic() > deadline - 60:
            break
        a_prof = _make_profile(n_final, prof)
        for blend in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 50:
                break
            a_blend = (1 - blend) * best_seq + blend * a_prof
            a_blend = np.maximum(a_blend, 0)
            a_blend = _norm_a(a_blend)
            sc_blend = _score(a_blend)
            if sc_blend < best_score * 1.002:
                bl_end = min(deadline - 40, time.monotonic() + 30)
                a_bl_opt = _active_set_v2(a_blend, bl_end, min_time_remaining=10.0)
                _try_update(a_bl_opt, f"blend_{prof}_{blend}")

    # === Phase 7: Try different n ===
    print(f"\n[P7] Alt n (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    n_alts = [n_final * 2, n_final // 2, n_final + 2048]
    n_alts = [x for x in n_alts if 2048 <= x <= 65536 and x != n_final]
    for n_alt in n_alts[:3]:
        if time.monotonic() > deadline - 40:
            break
        a_alt = _resample(best_seq, n_alt)
        sc_alt = _score(a_alt)
        print(f"  n={n_alt}: resampled={sc_alt:.10f}")
        alt_end = min(deadline - 20, time.monotonic() + 35)
        a_alt_opt = _run_lbfgs_deep(a_alt, alt_end, min_time_remaining=10.0)
        _try_update(a_alt_opt, f"alt_n_{n_alt}")

    # === Phase 8: Stochastic restart ===
    print(f"\n[P8] Stochastic (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    if time.monotonic() < deadline - 40:
        for sigma in [0.005, 0.002]:
            if time.monotonic() > deadline - 25:
                break
            a_pert = _stochastic_perturb(best_seq, sigma, rng)
            sc_pert = _score(a_pert)
            if sc_pert < best_score * 1.003:
                per_end = min(deadline - 15, time.monotonic() + 20)
                a_pert_opt, sc_pert_opt = _active_set_v2(a_pert, per_end, min_time_remaining=10.0)
                _try_update(a_pert_opt, f"stoch_{sigma}")

    # === Phase 9: Ultra-fine polish ===
    print(f"\n[P9] Ultra-fine (t={time.monotonic()-t_start:.1f}s, score={best_score:.10f})")
    if time.monotonic() < deadline - 5:
        uf_end = deadline - 3
        a_uf, sc_uf = _final_ultra_polish(best_seq, uf_end)
        _try_update(a_uf, "ultra_fine")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
