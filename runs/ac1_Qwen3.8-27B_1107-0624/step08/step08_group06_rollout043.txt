```python
"""
Step-function peak-convolution optimizer.

This searches for non-negative heights a_k minimizing
    2*n*max(conv(a,a)) / sum(a)^2
using a combination of:
  - warm-starting from the provided previous construction,
  - a multiplicative / sinkhorn-like smoother for global basin improvement,
  - L-BFGS on a log-sum-exp smoothed maximum (with the exact softmax gradient),
  - active weighted-gradient polish,
  - local peak deflation,
  - modest multi-length exploration (mostly 32768/65536 warm starts).

All helpers are top-level and no closures/lambdas are used.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None


def _next_pow2(x):
    m = int(max(1, x))
    k = 1
    while k < m:
        k <<= 1
    return max(2, k)


def _self_conv(a):
    """Linear convolution a*a via FFT, clipped to nonnegative."""
    n = a.size
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:out_n]
    out[out < 0.0] = 0.0
    return out


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score_norm(a):
    """Score for normalized/nonnegative input; scale invariant."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return float("inf")
    a = a / s
    c = _self_conv(a)
    if c.size == 0:
        return float("inf")
    cm = float(c.max())
    if not np.isfinite(cm):
        return float("inf")
    return float(2.0 * a.size * cm)


def _gap_base(a):
    """Return cmax and a beta scale based on top quantile of convolution."""
    c = _self_conv(a)
    if c.size == 0:
        return 0.0, 1e-12
    cmax = float(c.max())
    m = c.size
    top_count = max(10, min(1000, m // 200))
    if m > top_count:
        part = np.partition(c, m - top_count)
        gap = cmax - float(part[-top_count:].min())
    else:
        gap = cmax
    if not np.isfinite(gap) or gap <= 0.0:
        gap = 0.0
    base = max(gap, cmax * 1e-3)
    if not np.isfinite(base) or base <= 0.0:
        base = cmax * 1e-3 if cmax > 0 else 1e-12
    return cmax, float(base)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = z - float(np.max(z))
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _exp_weights(c, beta):
    """Normalize exp(c/beta) stably."""
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return np.ones(1, dtype=np.float64)
    if beta is None or beta <= 1e-18:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    shift = float(c.max())
    if not np.isfinite(shift):
        w = np.zeros(c.size, dtype=np.float64)
        w[0] = 1.0
        return w
    z = (c - shift) / beta
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _prepare_weights(w, min_frac=1e-12):
    """Trim weights to a compact support. Returns (w_trim, offset, wsum)."""
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.zeros(0, dtype=np.float64), 0, 0.0

    if min_frac <= 0.0:
        wsum = float(w.sum())
        if wsum <= 0.0 or not np.isfinite(wsum):
            return np.zeros(0, dtype=np.float64), 0, 0.0
        return w, 0, wsum

    maxw = float(np.max(w))
    if maxw <= 0.0 or not np.isfinite(maxw):
        return np.zeros(0, dtype=np.float64), 0, 0.0

    thr = maxw * min_frac
    if thr <= 0.0:
        idx = np.nonzero(w > 0.0)[0]
    else:
        idx = np.nonzero(w > thr)[0]

    if idx.size == 0:
        i = int(np.argmax(w))
        return np.array([w[i]], dtype=np.float64), i, float(w[i])

    s = int(idx[0])
    e = int(idx[-1])
    wt = w[s:e + 1].astype(np.float64, copy=True)
    if thr > 0.0:
        wt[wt <= thr] = 0.0
    wsum = float(wt.sum())
    if wsum <= 0.0 or not np.isfinite(wsum):
        i = int(np.argmax(wt))
        wt = np.array([wt[i]], dtype=np.float64)
        s = s + i
        wsum = float(wt[0])
    return wt, s, wsum


def _corr_from_trim(a, w_trim, offset, n):
    """
    For h_i = sum_j a_j w_{i+j}, where w is non-zero on [offset, offset+L-1]
    represented by w_trim, compute h for i=0..n-1 using one FFT.
    """
    n = int(n)
    L = int(w_trim.size)
    if L == 0 or n == 0:
        return np.zeros(n, dtype=np.float64)

    out_len = n + L - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_trim, N)
    conv = np.fft.irfft(fa * fw, N)[:out_len]

    h = np.zeros(n, dtype=np.float64)
    start = n - 1 - offset

    if start >= 0:
        take = min(n, max(0, out_len - start))
        if take > 0:
            h[:take] = conv[start:start + take]
    else:
        off = -start
        if off < n:
            take = min(n - off, max(0, out_len))
            if take > 0:
                h[off:off + take] = conv[:take]
    return h


def _corr_a_w(a, w, n, min_frac=1e-12):
    """Weighted correlation h_i = sum_j a_j w_{i+j}, with weights normalized."""
    n = int(n)
    w_trim, s, wsum = _prepare_weights(w, min_frac)
    if wsum <= 0.0 or not np.isfinite(wsum):
        return np.zeros(n, dtype=np.float64)
    h = _corr_from_trim(a, w_trim, s, n)
    if abs(wsum - 1.0) > 1e-14:
        h = h / wsum
    return h


def _logsum_fg(z, n, beta):
    """
    Smooth score for a=softmax(z):
        2*n * beta * log(sum(exp(c/beta)))
    Returns (value, gradient wrt z). This uses the exact softmax-chain rule:
        grad_z = 4*n * a * (h - inner(a,h)),  h_i=sum_j a_j w_{i+j}.
    """
    n = int(n)
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        if z.size < n:
            z = np.pad(z, (0, n - z.size))
        else:
            z = z[:n]

    a = _softmax(z)
    c = _self_conv(a)
    shift = float(c.max()) if c.size else 0.0

    if beta is None or beta <= 1e-18:
        w = np.zeros(c.size, dtype=np.float64)
        if c.size:
            w[int(np.argmax(c))] = 1.0
        f_score = 2.0 * n * shift
    else:
        zc = (c - shift) / beta
        zc = np.clip(zc, -700.0, 0.0)
        e = np.exp(zc)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.zeros(c.size, dtype=np.float64)
            if c.size:
                w[int(np.argmax(c))] = 1.0
            f_score = 2.0 * n * shift
        else:
            w = e / s
            f_score = 2.0 * n * (shift + beta * np.log(s))

    h = _corr_a_w(a, w, n, 1e-12)
    ag = float(np.dot(a, h))
    grad = 4.0 * float(n) * a * (h - ag)

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    if not np.isfinite(f_score):
        f_score = 1e300
    return float(f_score), grad


def _logsum_f(z, n, beta):
    val, _ = _logsum_fg(z, n, beta)
    return val


def _logsum_g(z, n, beta):
    _, g = _logsum_fg(z, n, beta)
    return g


def _lbfgs_run(a, deadline, min_time_remaining=30.0):
    """Beta-continuation L-BFGS on softmax-parameterized log-sum-exp objective."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_sc = _score_norm(a)

    if _scipy_minimize is None:
        return best_a, best_sc

    if time.monotonic() >= deadline:
        return best_a, best_sc

    coeffs = [
        0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6
    ]

    no_improve = 0
    for coeff in coeffs:
        if time.monotonic() >= deadline:
            break

        cmax, base = _gap_base(best_a)
        beta = base * coeff
        min_beta = max(cmax * 1e-7, 1e-15)
        if beta < min_beta or beta <= 0.0:
            break

        available = deadline - min_time_remaining - time.monotonic()
        if available < 5.0:
            break

        # Conservative per-function-evaluation time estimate to avoid overrunning.
        est = 0.01 + 0.0000025 * n
        if n <= 8192:
            cap = 3000
        elif n <= 16384:
            cap = 1800
        elif n <= 32768:
            cap = 900
        elif n <= 65536:
            cap = 500
        else:
            cap = 250

        if available < 50.0 * est:
            break

        maxfun = int(min(float(cap), max(50.0, available / est)))
        maxfun = max(50, min(maxfun, cap))

        # Re-enable tiny coordinates to avoid永久 zero support.
        cur = _normalize(np.maximum(best_a, 1e-14 / max(1, n)))
        z = np.log(np.maximum(cur, 1e-30))
        z -= float(np.mean(z))

        opts = {
            "maxiter": maxfun,
            "maxfun": maxfun,
            "maxls": 20,
            "ftol": 1e-14,
            "gtol": 1e-11,
            "maxcor": 30
        }

        result = None
        try:
            result = _scipy_minimize(
                _logsum_fg, z, args=(n, beta),
                method="L-BFGS-B", jac=True, options=opts
            )
        except Exception:
            try:
                result = _scipy_minimize(
                    _logsum_f, z, args=(n, beta),
                    method="L-BFGS-B", jac=_logsum_g, options=opts
                )
            except Exception:
                break

        if result is None or not hasattr(result, "x") or result.x is None:
            break

        x = np.asarray(result.x, dtype=np.float64).ravel()
        if x.size != n:
            if x.size < n:
                x = np.pad(x, (0, n - x.size))
            else:
                x = x[:n]

        a_new = _softmax(x)
        a_new = _normalize(a_new)
        sc_new = _score_norm(a_new)

        if np.isfinite(sc_new) and sc_new < best_sc - 1e-14:
            best_a = a_new
            best_sc = float(sc_new)
            no_improve = 0
            print(f"    [lbfgs] n={n} beta={beta:.3e} score={best_sc:.10f}")
        else:
            no_improve += 1
            if coeff < 0.01 and no_improve >= 3:
                break

    return best_a, float(best_sc)


def _multiplicative_smooth(a, deadline, tag=""):
    """
    Multiplicative / mirror-descent smoother.
    For weights w from flattened peaks, gradient h_i=sum_j a_j w_{i+j};
    update a_i proportional to a_i * (mean(h)/h_i)^tau.
    """
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_sc = _score_norm(a)

    if time.monotonic() >= deadline:
        return best_a, best_sc

    phases = [
        (0.50, 1.00, 0.50),
        (0.25, 1.00, 0.50),
        (0.10, 0.80, 0.45),
        (0.05, 0.80, 0.40),
        (0.02, 0.60, 0.35),
        (0.01, 0.60, 0.30),
        (0.005, 0.50, 0.25),
        (0.002, 0.50, 0.25),
    ]

    t0 = time.monotonic()
    rem = max(0.0, deadline - t0)
    if rem < 5.0:
        return best_a, best_sc

    phase_time = rem / float(len(phases))

    for beta_frac, tau, damp in phases:
        end = time.monotonic() + phase_time
        cur_a = best_a.copy()
        cur_sc = best_sc
        stagnant = 0
        it = 0

        while time.monotonic() < end:
            # Tiny floor keeps full support possible.
            cur_a = np.maximum(cur_a, 1e-12 / max(1, n))
            cur_a = _normalize(cur_a)

            c = _self_conv(cur_a)
            cmax = float(c.max()) if c.size else 0.0
            if cmax <= 0.0:
                break

            beta = max(cmax * beta_frac, 1e-14)
            w = _exp_weights(c, beta)
            h = _corr_a_w(cur_a, w, n, 1e-10)
            mean_h = float(np.mean(h))
            if mean_h <= 0.0 or not np.isfinite(mean_h):
                break

            h_floor = np.maximum(h, mean_h * 1e-8)
            ratio = mean_h / h_floor
            ratio = np.clip(ratio, 1e-2, 1e2)
            ratio_power = np.power(ratio, tau)

            a_new = cur_a * ratio_power
            a_new = _normalize(a_new)
            a_new = (1.0 - damp) * cur_a + damp * a_new
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)

            sc_new = _score_norm(a_new)
            it += 1

            if np.isfinite(sc_new) and sc_new < best_sc:
                best_a = a_new.copy()
                best_sc = float(sc_new)
                stagnant = 0

            if np.isfinite(sc_new) and sc_new < cur_sc + 1e-4:
                cur_a = a_new
                cur_sc = float(sc_new)
            else:
                stagnant += 1
                if stagnant > 12:
                    break

            if tag and it % 300 == 0:
                print(f"    [multi] {tag} phase beta_frac={beta_frac:.3f} it={it} best={best_sc:.10f}")

        if time.monotonic() >= deadline:
            break

    return best_a, float(best_sc)


def _active_gradient_polish(a, deadline):
    """
    Weighted-gradient polish.  Uses exponentiated weights to choose a subgradient
    and takes backtracked Euclidean steps on the simplex.
    """
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_sc = _score_norm(a)

    if time.monotonic() >= deadline:
        return best_a, best_sc

    t0 = time.monotonic()
    rem = max(0.0, deadline - t0)
    if rem < 5.0:
        return best_a, best_sc

    _, base = _gap_base(a)
    coeffs = [
        0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002
    ]
    phase_time = rem / float(len(coeffs))

    for coeff in coeffs:
        end = time.monotonic() + phase_time
        beta = base * coeff
        if beta <= 0.0:
            break

        cur_a = best_a.copy()
        cur_sc = best_sc
        stagnant = 0

        while time.monotonic() < end:
            c = _self_conv(cur_a)
            cmax = float(c.max()) if c.size else 0.0
            if cmax <= 0.0:
                break

            w = _exp_weights(c, beta)
            h = _corr_a_w(cur_a, w, n, 1e-12)
            g = 2.0 * (h - float(np.mean(h)))
            gnorm = float(np.linalg.norm(g))
            if gnorm < 1e-13:
                break

            maxg = float(np.max(np.abs(g)))
            if maxg <= 0.0 or not np.isfinite(maxg):
                break

            tstep = 0.05 / max(maxg, 1e-12)
            accepted = False

            for _ in range(18):
                trial = cur_a - tstep * g
                trial = np.maximum(trial, 0.0)
                trial = _normalize(trial)
                sc = _score_norm(trial)
                if np.isfinite(sc) and sc < cur_sc - 1e-14:
                    cur_a = trial
                    cur_sc = float(sc)
                    if sc < best_sc:
                        best_a = trial.copy()
                        best_sc = float(sc)
                    accepted = True
                    break
                tstep *= 0.5

            if not accepted:
                stagnant += 1
                if stagnant > 4:
                    break
            else:
                stagnant = 0

    return best_a, float(best_sc)


def _deflation_step(a, deadline):
    """Reducing the largest product contributing to top convolution peaks."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_sc = _score_norm(a)

    if time.monotonic() >= deadline:
        return best_a, best_sc

    n_top = 8
    attempt = 0
    while time.monotonic() < deadline:
        c = _self_conv(a)
        if c.size == 0:
            break
        if c.size <= n_top:
            idx = np.arange(c.size, dtype=int)[::-1]
        else:
            idx = np.argsort(c, kind="stable")[::-1][:n_top]

        a_new = a.copy()
        changed = False

        for kk in idx:
            k = int(kk)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo > i_hi:
                continue

            ii = np.arange(i_lo, i_hi + 1, dtype=int)
            jj = k - ii
            prods = a_new[ii] * a_new[jj]
            pidx = int(np.argmax(prods))
            mi = int(ii[pidx])
            mj = int(jj[pidx])

            if mi == mj:
                if a_new[mi] > 0.0:
                    a_new[mi] *= 0.999
                    changed = True
            else:
                if a_new[mi] >= a_new[mj]:
                    a_new[mi] *= 0.995
                else:
                    a_new[mj] *= 0.995
                changed = True

        if not changed:
            break

        a_new = _normalize(a_new)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best_a = a_new.copy()
            best_sc = float(sc)
            a = a_new.copy()
        else:
            break

        attempt += 1
        if attempt > 30:
            break

    return best_a, float(best_sc)


def _make_profile(n, kind="uniform"):
    n = int(max(2, n))
    if kind == "uniform":
        return np.ones(n, dtype=np.float64) / float(n)

    if kind == "tapered":
        t = np.linspace(0.0, 1.0, n)
        a = np.power(np.sin(np.pi * t), 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()

    if kind == "cos":
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()

    if kind == "block":
        a = np.zeros(n, dtype=np.float64)
        b = max(1, n // 4)
        a[:b] = 1.0
        a[3 * b:] = 1.0
        s = a.sum()
        if s <= 0.0:
            return np.ones(n, dtype=np.float64) / float(n)
        return a / s

    if kind == "biased":
        t = np.linspace(0.0, 1.0, n)
        a = 0.15 + 0.85 * np.power(np.sin(np.pi * t), 0.7)
        a = np.maximum(a, 1e-10)
        return a / a.sum()

    return np.ones(n, dtype=np.float64) / float(n)


def _interpolate_to_n(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == 0:
        return _make_profile(int(n_new), "uniform")
    if n_old == int(n_new):
        return _normalize(a)

    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _optimize_sequence(a, deadline, tag=""):
    """Run a pipeline on a sequence until approximately deadline."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_norm(a)

    t = time.monotonic()
    rem = max(0.0, deadline - t)
    if tag:
        print(f"  [{tag}] start n={a.size} score={best_sc:.10f} rem={rem:.1f}s")

    if rem < 10.0:
        return best_a, float(best_sc)

    # 1) Multiplicative smoother: cheap, often improves basin before L-BFGS.
    mult_end = t + 0.25 * rem
    if mult_end > deadline - 30.0:
        mult_end = deadline - 30.0
    if mult_end > t + 10.0:
        am, scm = _multiplicative_smooth(a, mult_end, tag + "_multi")
        if scm < best_sc - 1e-15:
            best_a = am.copy()
            best_sc = float(scm)
        a = best_a.copy()

    # 2) L-BFGS smoothed-max polish.
    lb_end = min(deadline - 20.0, t + 0.70 * rem)
    if time.monotonic() < lb_end:
        al, scl = _lbfgs_run(a, lb_end, min_time_remaining=20.0)
        if scl < best_sc - 1e-15:
            best_a = al.copy()
            best_sc = float(scl)
        a = best_a.copy()

    # 3) Subgradient / active weighted-gradient polish.
    ag_end = min(deadline - 10.0, time.monotonic() + min(60.0, 0.10 * rem + 20.0))
    if time.monotonic() < ag_end:
        aa, sca = _active_gradient_polish(a, ag_end)
        if sca < best_sc - 1e-15:
            best_a = aa.copy()
            best_sc = float(sca)
        a = best_a.copy()

    # 4) Deflation of largest product peaks.
    def_end = min(deadline - 5.0, time.monotonic() + min(30.0, 0.05 * rem + 10.0))
    if time.monotonic() < def_end:
        ad, scd = _deflation_step(a, def_end)
        if scd < best_sc - 1e-15:
            best_a = ad.copy()
            best_sc = float(scd)
        a = best_a.copy()

    # 5) Final short L-BFGS squeeze if time remains.
    if time.monotonic() < deadline - 5.0 and _scipy_minimize is not None:
        fin_end = min(deadline - 2.0, time.monotonic() + min(80.0, 0.12 * rem + 30.0))
        af, scf = _lbfgs_run(best_a, fin_end, min_time_remaining=2.0)
        if scf < best_sc - 1e-15:
            best_a = af.copy()
            best_sc = float(scf)

    if tag:
        print(f"  [{tag}] done n={best_a.size} score={best_sc:.10f} t={time.monotonic() - t:.1f}s")

    return best_a, float(best_sc)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point.  Returns a Python list of nonnegative heights.
    Uses up to budget_s seconds (default 1800), with a safety reservation.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s) if budget_s is not None else 1800.0
    total_budget = max(60.0, min(3600.0, total_budget))
    # Reserve enough time to return safely from long optimizer calls.
    deadline = t_start + total_budget - 120.0
    if deadline <= t_start:
        deadline = t_start + 30.0

    rng = np.random.default_rng(int(seed) % (2**63 - 1))  # noqa: F841 (kept for possible extensions)

    # Load parent construction if available.
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent_arr = np.clip(parent_arr, 0.0, None)
        if parent_arr.size > 0:
            s = float(parent_arr.sum())
            if s > 1e-12:
                parent = parent_arr / s
    except Exception:
        parent = None

    if parent is None:
        parent = _make_profile(8192, "uniform")

    parent_n = int(parent.size)
    best_a = parent.copy()
    best_sc = _score_norm(best_a)
    best_small_a = best_a.copy() if best_a.size <= 32768 else None
    best_small_sc = best_sc if best_a.size <= 32768 else float("inf")

    print(f"Initial n={best_a.size} score={best_sc:.10f}")

    def_set = set()

    def _consider(a_candidate, sc_candidate, label=""):
        nonlocal best_a, best_sc, best_small_a, best_small_sc
        if a_candidate is None or not np.isfinite(sc_candidate):
            return
        a_candidate = np.asarray(a_candidate, dtype=np.float64).ravel()
        a_candidate = np.clip(a_candidate, 0.0, None)
        s = float(a_candidate.sum())
        if s < 1e-12:
            return
        a_candidate = a_candidate / s
        sc_candidate = float(_score_norm(a_candidate))
        if np.isfinite(sc_candidate) and sc_candidate < best_sc - 1e-15:
            best_a = a_candidate.copy()
            best_sc = sc_candidate
            if best_a.size <= 32768:
                best_small_a = best_a.copy()
                best_small_sc = best_sc
            print(f"  NEW BEST: {best_sc:.10f} n={best_a.size} [{label}]")

    # Phase 1: polish the parent at its native length.
    t = time.monotonic()
    remaining = max(0.0, deadline - t)
    if remaining > 250.0:
        if parent_n <= 20000:
            par_alloc = min(650.0, 0.42 * remaining)
        elif parent_n <= 40000:
            par_alloc = min(850.0, 0.45 * remaining)
        else:
            par_alloc = min(1050.0, 0.50 * remaining)
        par_alloc = min(par_alloc, max(90.0, remaining - 350.0))
        if par_alloc > 25.0:
            par_dl = min(deadline - 150.0, t + par_alloc)
            pa, ps = _optimize_sequence(best_a, par_dl, f"parent_n{parent_n}")
            _consider(pa, ps, "parent")

    # Phase 2: explore modestly larger lengths by interpolation/warm-start.
    candidates = []
    if best_a.size != 32768 and time.monotonic() < deadline - 350.0:
        candidates.append(32768)
    if best_a.size != 65536 and time.monotonic() < deadline - 250.0:
        candidates.append(65536)

    candidates = [n for n in candidates if n > best_a.size]

    for n_c in sorted(candidates):
        if time.monotonic() >= deadline - 90.0:
            break
        remaining = max(0.0, deadline - time.monotonic())
        if n_c == 32768:
            alloc = min(500.0, 0.38 * remaining)
            alloc = max(0.0, min(alloc, remaining - 350.0))
        else:
            alloc = min(remaining - 90.0, max(300.0, 0.58 * remaining))

        if alloc < 80.0:
            continue

        dl = min(deadline - 40.0, time.monotonic() + alloc)
        init_a = _interpolate_to_n(best_a, n_c)
        # If parent is much smaller, also consider a fresh biased profile take if
        # it looks better before optimization.
        profile = _make_profile(n_c, "biased")
        sp = _score_norm(profile)
        si = _score_norm(init_a)
        if sp < si - 0.01:
            init_a = profile

        a_c, s_c = _optimize_sequence(init_a, dl, f"n{n_c}")
        _consider(a_c, s_c, f"n{n_c}")

    # Phase 3: final polish on the current best using remaining time.
    if time.monotonic() < deadline - 35.0:
        rem = max(0.0, deadline - time.monotonic())
        last_alloc = min(rem - 10.0, max(60.0, 0.55 * rem))
        if last_alloc > 25.0:
            last_dl = min(deadline - 5.0, time.monotonic() + last_alloc)
            af, sf = _optimize_sequence(best_a, last_dl, "final")
            _consider(af, sf, "final")

    # Optional safety: if a very long sequence is only marginally better than a
    # shorter one, prefer the shorter sequence to reduce evaluation risk.
    if best_small_a is not None and best_a.size > 32768:
        if best_small_sc <= best_sc + 0.0008:
            best_a = best_small_a.copy()
            best_sc = best_small_sc

    # Final validation and safe scaling.
    best_a = _normalize(best_a)
    best_a = np.clip(best_a, 0.0, None)
    if best_a.size == 0:
        best_a = _make_profile(256, "uniform")

    # Scale to sum ~1000, while ensuring every entry is below 1000.
    max_val = float(np.max(best_a)) if best_a.size else 0.0
    if max_val > 0.0:
        scale = min(1000.0, 999.0 / max_val)
        out = best_a * scale
    else:
        out = _make_profile(256, "uniform") * 1000.0

    out = np.clip(out, 0.0, 1000.0)
    out_sum = float(out.sum())
    if out_sum < 0.01:
        out = out * 1000.0
        out = np.clip(out, 0.0, 1000.0)

    out = np.asarray(out, dtype=np.float64)
    final_score = _score_norm(out / out.sum()) if out.sum() > 0 else float("inf")
    print(f"FINAL returned: n={out.size}, local_score={final_score:.10f}, "
          f"time={time.monotonic() - t_start:.1f}s")

    return [float(x) for x in out]
```
