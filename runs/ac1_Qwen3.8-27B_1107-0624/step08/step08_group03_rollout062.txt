The idea: We want to allocate power across frequencies to minimize the peak of the time-domain autocorrelation. This is analogous to the "spectrum shaping" problem in communications.

    n = len(a_init)
    N = next_pow2(2*n - 1)  # FFT size for full convolution
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2  # Power spectrum
        
        # Current autocorrelation
        c = np.fft.irfft(P, N)[:2*n-1]
        c_max = c.max()
        c_mean = c.mean()
        
        # We want to flatten c. The contribution of frequency k to c_j 
        # is P_k * cos(2*pi*k*j/N) (simplified)
        # To reduce the peak at position j_peak, we want to reduce P_k 
        # where cos(2*pi*k*j_peak/N) is large (positive contribution)
        
        j_peak = np.argmax(c)
        
        # Compute "influence" of each frequency on the peak
        # c_j = (1/N) sum_k P_k * cos(2*pi*k*j/N) * 2 (for k > 0, k < N/2)
        # Actually c_j = (1/N) * P_0 + (2/N) * sum_{k=1}^{N/2-1} P_k * cos(2*pi*k*j/N) + (1/N) * P_{N/2} * cos(pi*j)

        k_vals = np.arange(N//2 + 1)
        influence = np.cos(2 * np.pi * k_vals * j_peak / N)
        
        # Water-filling: reduce P where influence is positive (contributes to peak)
        # and increase where influence is negative
        # But we need to maintain sum(a) = 1, which means A_0 = N (fixed)
        # And sum(P_k) = sum(a^2) * N (by Parseval)
        
        # Actually, let's be more careful.

        # Wait, numpy's FFT: A_k = sum_j a_j * exp(-2*pi*i*j*k/N)
        # So A_0 = sum(a) = 1
        # And a_j = (1/N) sum_k A_k exp(2*pi*i*j*k/N)
        
        # By Parseval: sum(a^2) = (1/N) sum(|A_k|^2) = (1/N) sum(P_k)
        
        # The autocorrelation: c_j = (1/N) sum_{k=0}^{N-1} |A_k|^2 exp(2*pi*i*k*j/N)
        # = (1/N) [P_0 + sum_{k=1}^{N/2-1} P_k * 2*cos(2*pi*k*j/N) + P_{N/2} * (-1)^j]

        # - A_0 = 1 (fixed, sum constraint)
        # - a = IFFT(A) >= 0
        # - A is Hermitian (real a)
        
        # The constraint a >= 0 is: (1/N) sum_k A_k exp(2*pi*i*k*j/N) >= 0
        # This couples all frequencies, making it hard to separate.

        # and project back to feasibility.

        # dc_jpeak / dP_k = (2/N) * cos(2*pi*k*jpeak/N) for 0 < k < N/2
        # dc_jpeak / dP_{N/2} = (1/N) * (-1)^jpeak

        P_new = P.copy()

            best_a = a_new.copy()
        
        a = a_new

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())

    s = e.sum()
    if s < 1e-300:
