    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=800, tol=1e-15):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _identify_active_set(c, c_max, tol_frac):
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    return active


def _newton_polish_dynamic(a_init, deadline, max_rounds=200, tol_fracs=None):
    """Newton polish with dynamic active set that adapts."""
    if tol_fracs is None:
        tol_fracs = [1e-4, 3e-4, 1e-3, 3e-3, 1e-2]
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Try multiple active set tolerances, use the tightest that gives >= 2 active
        current_active = None
        current_n_active = 0
        for tol_frac in tol_fracs:
            active = _identify_active_set(c, c_max, tol_frac)
            n_act = int(active.sum())
            if n_act >= 2:
                current_active = active
                current_n_active = n_act
                break
        if current_active is None:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            current_active = active
            current_n_active = 1

        W = np.zeros(len(c))
        W[current_active] = 1.0 / current_n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=800)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=800)
        ones = np.ones(n)
        denom = float(ones @ H1)
        if abs(denom) < 1e-25:
            dx = -g_t / max(gn, 1e-30)
        else:
            lam = float(ones @ Hg) / denom
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / max(gn, 1e-30)

        # Line search with aggressive backtracking
        alpha = 1.0
        step_done = False
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max * (1.0 - 0.001 * alpha) - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done:
            # Subgradient fallback
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _lbfgs_aggressive(a_init, deadline, min_time_remaining=60.0):
    """Aggressive log-sum-exp L-BFGS with geometric beta cooling."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Geometric beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    beta = max(gap * 0.5, max_c * 0.01)
    min_beta = max_c * 1e-18
    beta_ratio = 0.55  # aggressive geometric cooling
    prev_sc = current_sc
    iteration = 0

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_per_iter = 0.002 * scale
        max_iter = min(6000, max(20, int(remaining / (est_per_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Update beta
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-10 * c_max_new:
            beta *= beta_ratio * 0.5  # extra aggressive when very flat
        elif sc_improved < 1e-14:
            beta *= beta_ratio
        else:
            beta *= beta_ratio

        beta = min(beta, c_max_new * 0.001)
        iteration += 1

    return current_a, current_sc


def _mirror_descent(a_init, deadline, max_iter=50000):
    """Exponential weights / mirror descent for the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: top peaks within tolerance
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        W = np.zeros(len(c))
        W[active] = 1.0 / n_active

        # Gradient: (H_K a)_i = sum_{k in active} a[k-i] / |K|
        W_restricted = W.copy()
        a_rev = a[::-1]
        wc = _full_convolve(W_restricted, a_rev)
        h = wc[n - 1:2 * n - 1]

        # Adaptive step size
        eta = 0.1 / (1.0 + 0.01 * i)
        if i < 100:
            eta = 0.3 / (1.0 + 0.05 * i)

        z = np.log(np.maximum(a, 1e-30))
        z = z - eta * h
        z = z - z.mean()
        a_new = _softmax(z)

        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                z = np.log(np.maximum(best_a, 1e-30))
                patience = 0

        a = a_new

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 0.1 + 0.9 * t
        return a / a.sum()
    elif kind == "asym_ramp2":
        t = np.linspace(0, 1, n)
        a = 0.05 + 0.95 * t ** 1.5
        return a / a.sum()
    elif kind == "exp_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * (t - 0.5))
        return a / a.sum()
    elif kind == "bump_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.6) ** 2) + 0.05
        return a / a.sum()
    elif kind == "left_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.35) ** 2) + 0.05
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (2 * t - 0.3))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization: multi-strategy search for minimum score sequence."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            t_el = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={t_el:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "asym_ramp")
        print(f"No parent, using asym_ramp n={n_parent}")
        _update_best(parent, _score(parent), "default_init")

    # Phase 1: Profile exploration (find best starting shape)
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic()-t_start:.1f}s)")
    t_phase_end = time.monotonic() + 120
    profiles = ["asym_ramp", "asym_ramp2", "exp_asym", "bump_asym",
                "left_bump", "cos_asym", "tapered", "uniform"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > t_phase_end:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': {sc_p:.10f} (new best profile)")

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 1200:
        a_q, sc_q = _lbfgs_aggressive(best_profile_a, min(time.monotonic() + 90, deadline - 1200),
                                       min_time_remaining=800.0)
        _update_best(a_q, sc_q, "quick_profile")
        if sc_q < best_profile_sc:
            best_profile_a = a_q
            best_profile_sc = sc_q

    # Set the main starting point
    start_a = best_seq.copy() if best_seq is not None else best_profile_a
    print(f"  Starting point score: {best_score:.10f}")

    # Phase 2: Main aggressive L-BFGS (bulk of time)
    print(f"\n[Phase 2] Main aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    t2_start = time.monotonic()
    a_main, sc_main = _lbfgs_aggressive(start_a, deadline, min_time_remaining=600.0)
    _update_best(a_main, sc_main, "lbfgs_main")
    print(f"  After main L-BFGS: {best_score:.10f} (t={time.monotonic()-t2_start:.1f}s)")

    # Phase 3: Newton polish with dynamic active set
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_np, sc_np = _newton_polish_dynamic(best_seq, min(time.monotonic() + 200, deadline - 400),
                                              max_rounds=200)
        _update_best(a_np, sc_np, "newton_dynamic")

    # Phase 4: Mirror descent (alternative geometry)
    print(f"\n[Phase 4] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_md, sc_md = _mirror_descent(best_seq, min(time.monotonic() + 120, deadline - 300),
                                      max_iter=80000)
        _update_best(a_md, sc_md, "mirror")
        # Polish mirror result with Newton
        if sc_md < best_score + 0.01 and time.monotonic() < deadline - 200:
            a_md_np, sc_md_np = _newton_polish_dynamic(a_md, min(time.monotonic() + 80, deadline - 200),
                                                        max_rounds=100)
            _update_best(a_md_np, sc_md_np, "mirror_newton")

    # Phase 5: Second L-BFGS from best (different beta schedule)
    print(f"\n[Phase 5] Second L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_s2, sc_s2 = _lbfgs_aggressive(best_seq, deadline, min_time_remaining=150.0)
        _update_best(a_s2, sc_s2, "lbfgs_2nd")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 32768, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq if best_seq is not None else parent):
            continue
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt_n, sc_opt_n = _lbfgs_aggressive(a_init_n, deadline, min_time_remaining=60.0)
        _update_best(a_opt_n, sc_opt_n, f"n_{n_try}")

        if sc_opt_n < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_np_n, sc_np_n = _newton_polish_dynamic(a_opt_n, min(time.monotonic() + 30, deadline - 30),
                                                      max_rounds=50)
            _update_best(a_np_n, sc_np_n, f"n_{n_try}_np")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_configs = [
            (0.03, "perturb_small"),
            (0.1, "perturb_med"),
            (0.3, "perturb_large"),
        ]
        for sigma, label in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            a_p = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _lbfgs_aggressive(a_p, deadline, min_time_remaining=20.0)
            _update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 20:
                a_r_np, sc_r_np = _newton_polish_dynamic(a_r, min(time.monotonic() + 15, deadline - 10),
                                                          max_rounds=50)
                _update_best(a_r_np, sc_r_np, f"restart_{label}_np")

    # Phase 8: Final ultra-fine Newton polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        a_fin, sc_fin = _newton_polish_dynamic(best_seq, deadline - 1, max_rounds=300,
                                                tol_fracs=[1e-5, 3e-5, 1e-4, 3e-4, 1e-3])
        _update_best(a_fin, sc_fin, "final_newton_tight")

        # One more L-BFGS at tiny beta if time allows
        if time.monotonic() < deadline - 2:
            n_fin = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-12
            remaining = deadline - 2 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(30, int(remaining / 0.015)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n_fin)
                    grad_fn = _SmoGradAdapter(beta_fin, n_fin)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-17,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    _update_best(a_flb, _score(a_flb), "final_lbfgs_tiny_beta")
                except Exception:
                    pass

        # Very final Newton
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_vf, sc_vf = _newton_polish_dynamic(best_seq, deadline - 0.1, max_rounds=100,
                                                  tol_fracs=[1e-6, 1e-5, 3e-5, 1e-4])
            _update_best(a_vf, sc_vf, "very_final_newton")

    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "asym_ramp")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
