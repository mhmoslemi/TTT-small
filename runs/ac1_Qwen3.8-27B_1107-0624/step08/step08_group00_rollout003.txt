"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, active-set gradient descent on top peaks,
power-sum objectives, and multi-scale exploration with restarts.
Key: exploits problem structure via FFT convolution and targets asymmetry.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _power_sum_obj(z, power, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    ps = float(np.sum(c_pos ** power)) ** (1.0 / power)
    return 2.0 * n * ps


def _power_sum_grad(z, power, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    ps = float(np.sum(c_pos ** power)) ** (1.0 / power)
    w = (c_pos ** (power - 1)) / (power * ps * (len(c) ** (1.0 - 1.0 / power)))
    w = w / w.sum() if w.sum() > 0 else np.zeros_like(c)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * ps * a * (h - W) / ps if ps > 0 else np.zeros(n)


def _run_lbfgs_smoothed(a_init, deadline, min_remaining=60.0):
    """L-BFGS with geometrically decreasing beta for log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # Finer schedule with more steps
    n_steps = 18
    ratios = np.concatenate([np.linspace(1.0, 0.01, n_steps - 4),
                             np.linspace(0.005, 0.00005, 4)])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 15))))

        obj = lambda z: _logsumexp_obj(z, beta, n)
        grad = lambda z: _logsumexp_grad(z, beta, n)

        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter,
                                       'ftol': 1e-18,
                                       'gtol': 1e-14,
                                       'maxcor': 80,
                                       'maxls': 50,
                                       'iprint': -1})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 4 == 0:
                print(f"    L-BFGS beta={beta:.6g}: score={current_sc:.10f}")
        except Exception as ex:
            print(f"    L-BFGS beta={beta:.6g}: error {ex}")

    return current_a

    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

def _active_set_gradient(a_init, deadline, min_remaining=30.0,
                         top_frac_start=0.05, top_frac_end=0.002,
                         lr_start=0.02, lr_end=0.001):
    """Active-set gradient descent targeting top-K convolution peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()
    total_time = max(1.0, deadline - t_start)
    no_improve = 0
    max_stall = 200

    while time.monotonic() + min_remaining < deadline:
        elapsed_frac = min(1.0, (time.monotonic() - t_start) / total_time)
        top_frac = top_frac_start * (top_frac_end / top_frac_start) ** elapsed_frac
        lr = lr_start * (lr_end / lr_start) ** elapsed_frac

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_stall:
            a = best_a.copy()
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max
            no_improve = 0
            lr *= 0.5
            if lr < lr_end * 0.1:
                break

        # Select top peaks
        n_top = max(3, int(n * top_frac))
        top_idx = np.argpartition(c, -n_top)[-n_top:]
        top_vals = c[top_idx]
        # Weight by how close to max
        weights = np.maximum(top_vals - 0.5 * c_max, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(n_top)
        weights = weights / weights.sum()

        # Compute weighted gradient
        # grad of c[k] wrt a[j] = a[k-j] + a[j-k] (with a[m]=0 for m out of range)
        # For weighted sum: grad[j] = sum_p w_p * (a[k_p - j] + a[j - k_p])
        grad = np.zeros(n)
        for p_idx, k in enumerate(top_idx):
            if weights[p_idx] < 1e-15:
                continue
            w_p = weights[p_idx]
            # a[k-j] for j in [0,n-1]: k-j in [k-(n-1), k], valid when 0<=k-j<n
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                vals = a[k - j_hi:k - j_lo + 1][::-1]
                grad[j_lo:j_hi + 1] += w_p * vals
            # a[j-k] for j in [0,n-1]: j-k in [0-k, n-1-k], valid when 0<=j-k<n
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                vals2 = a[j_lo2 - k:j_hi2 - k + 1]
                grad[j_lo2:j_hi2 + 1] += w_p * vals2

        # Project gradient: remove mean (to maintain sum=1)
        grad_proj = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        a_old_sc = current_sc
        step = lr
        improved = False
        for _ in range(50):
            a_try = a - step * grad_proj
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc * (1.0 - 1e-12):
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try a direct peak-reduction step
            k_peak = int(np.argmax(c))
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
            top_c = np.argsort(contrib)[::-1][:max(1, n // 50)]
            reduction = c[k_peak] * 0.005
            for idx in top_c:
                if contrib[idx] > 1e-15:
                    a[idx] = max(0.0, a[idx] - reduction / (2.0 * contrib[idx]))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s

    return best_a, best_sc


def _coordinate_peak_reduction(a_init, deadline, min_remaining=30.0):
    """Iteratively reduce each top peak by adjusting contributing elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Find top 5 peaks
        top_k = np.argpartition(c, -5)[-5:]
        top_k = top_k[np.argsort(c[top_k])[::-1]]

        improvement = False
        for k_peak in top_k:
            if time.monotonic() + min_remaining > deadline:
                break
            # Compute contributions to this peak
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

            # Reduce top contributors proportionally
            n_reduce = max(1, min(10, n // 1000))
            top_contrib = np.argsort(contrib)[::-1][:n_reduce]
            total_contrib = sum(contrib[idx] for idx in top_contrib)
            if total_contrib < 1e-15:
                continue
            target_reduction = c[k_peak] * 0.01
            for idx in top_contrib:
                frac = contrib[idx] / total_contrib
                delta = target_reduction * frac / (2.0 * contrib[idx] + 1e-15)
                a[idx] = max(0.0, a[idx] - delta)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

    return best_a, best_sc


def _perturb_restart(a, sigma, rng):
    """Perturb a solution in log-space and renormalize."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_asymmetric_init(n, rng):

            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS power-sum ===
    print(f"\n[P3] L-BFGS power-sum (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        min_rem = 200
        a3 = _run_lbfgs_power(best_seq, deadline, min_remaining=min_rem)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [power_sum] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Another active-set pass (tighter) ===
    print(f"\n[P4] Active-set tight (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        min_rem = 150
        pol_deadline = min(deadline - min_rem, time.monotonic() + 120)
        a4, sc4 = _active_set_gradient(best_seq, pol_deadline, min_remaining=min_rem,
                                       top_frac_start=0.03, top_frac_end=0.001,
                                       lr_start=0.015, lr_end=0.001)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [active_set_tight] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Restart with perturbation ===
    print(f"\n[P5] Restart + re-optimize (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        for sigma in [0.005, 0.01, 0.02]:
            if time.monotonic() + 80 > deadline:
                break
            a_pert = _perturb_restart(best_seq, sigma, rng)
            # Quick optimize
            min_rem = 60
            a_opt = _run_lbfgs_smoothed(a_pert, deadline, min_remaining=min_rem)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_sigma={sigma}] t={time.monotonic()-t_start:.1f}s")
            # Follow with active set
            if time.monotonic() < deadline - 50:
                a_as, sc_as = _active_set_gradient(a_opt, deadline, min_remaining=40,
                                                   top_frac_start=0.05, top_frac_end=0.002,
                                                   lr_start=0.02, lr_end=0.001)
                if sc_as < best_score:
                    best_seq = a_as.copy()
                    best_score = sc_as
                    print(f"  NEW BEST: {best_score:.10f} [restart_as] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate peak reduction ===
    print(f"\n[P6] Coordinate peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a6, sc6 = _coordinate_peak_reduction(best_seq, deadline, min_remaining=80)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [coord] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 32768, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        if n_try > 32768:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        if time.monotonic() < deadline - 50:
            a_opt = _run_lbfgs_smoothed(a_init, deadline, min_remaining=40)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        # Very fine L-BFGS
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        mb = float(c_fin.max())
        beta_fin = mb * 0.0001
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        remaining = deadline - 8 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(20, int(remaining / 0.03)))
            try:
                obj = lambda z: _logsumexp_obj(z, beta_fin, n_fin)
                grad = lambda z: _logsumexp_grad(z, beta_fin, n_fin)
                result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter,
                                           'ftol': 1e-18,
                                           'gtol': 1e-15,
                                           'maxcor': 80,
                                           'iprint': -1})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # Active set final
        if time.monotonic() < deadline - 5:
            a_fs, sc_fs = _active_set_gradient(best_seq, deadline, min_remaining=3,
                                              top_frac_start=0.02, top_frac_end=0.001,
                                              lr_start=0.01, lr_end=0.001)
            if sc_fs < best_score:
                best_seq = a_fs.copy()
                best_score = sc_fs
                print(f"  NEW BEST: {best_score:.10f} [final_active] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        # Scale so sum is in a good range (not too close to 0 or 1000*n)
        # The evaluator clips to [0,1000] and checks sum >= 0.01
        # Best to have sum around 1 for numerical stability
        if s > 100.0:
            final = final / s
            s = 1.0
        elif s < 1.0 and s > 0.01:
            final = final / s
            s = 1.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
