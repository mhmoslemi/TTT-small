"""
Coordinate-descent and adaptive active-set optimizer for minimizing the
step-function autocorrelation peak score 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over prior approach:
  - Exact 1-D coordinate descent (no smoothing surrogate needed)
  - Adaptive active-set size with dynamic plateau detection
  - Palindromic constraint to halve搜索 space
  - Multi-n interpolation with progressive refinement

import numpy as np
import time
import math
from scipy.optimize import minimize_scalar, minimize

# ==============================================================
# Core utilities
# ==============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _coord_descent_step(a, j, c, S, n):
    """
    Find optimal a[j] given all other coordinates fixed.
    c = full convolution of current a, S = sum(a), n = len(a)
    Returns new value of a[j].
    """
    a_j = a[j]
    S_rest = S - a_j

    # e_k = a[k-j] + a[j-k] (out of range = 0)
    # c_k(x) = c_k - a_j*e_k + x*e_k (for k != 2j)
    #         c_k - a_j^2 + x^2     (for k = 2j, if 2j < 2n-1)
    #         c_k                    (if j out of range for lag k)

    # We need to find: min_{x>=0} 2n * max_k c_k(x) / (S_rest + x)^2
    # where c_k(x) = c_k_current + x*(e_k - a_j) + (1 if k==2j else 0)*(x^2 - a_j^2)

    # Define d_k = e_k - a_j (change in c_k per unit change in a[j], for k != 2j)
    # For k = 2j: c_k(x) = c_k - a_j^2 + x^2

    # Strategy: sample x values and find the minimum.
    # The function is a ratio of piecewise-linear (or quadratic) to quadratic.
    # Use a coarse-to-fine 1D search.

    k2j = 2 * j
    has_quad = k2j < 2 * n - 1

    # Precompute e[k] for all k where a[k-j] or a[j-k] is in range
    # e[k] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j-k != k-j)
    # Actually: e[k] = a[k-j] (if valid) + a[j-k] (if valid)
    # Note: if k-j == j-k (i.e., k=2j), then a[k-j]=a[j] and a[j-k]=a[-j]=0 (if j>0)
    #        so e[2j] = a[j] (only one term) -- but we handle quad separately

    # For efficiency, only consider the top K lags
    # Compute current top lags
    c_sorted_idx = np.argsort(c)[::-1]
    top_K = min(64, 2 * n - 1)
    active_lags = c_sorted_idx[:top_K]

    # If k=2j is not in active set, add it (since it's quadratic, could become max)
    if has_quad and k2j not in active_lags:
        active_lags = np.append(active_lags, k2j)

    def f_of_x(x):
        """Evaluate score with a[j] = x."""
        S_new = S_rest + x
        if S_new < 1e-15:
            return np.inf
        # Compute c_k(x) for active lags
        max_c = 0.0
        for k in active_lags:
            if k == k2j:
                # c_k(x) = c[k] - a_j^2 + x^2  (but only if the a_j*a_j term was in c[k])
                # Actually: c[k] includes a[j]*a[k-j] terms.
                # The terms involving a[j] in c[k]:
                #   a[j]*a[k-j] (from i=j): = a_j * a[k-j] if 0<=k-j<n
                #   a[k-j]*a[j] (from i=k-j): = a[k-j]*a_j if 0<=k-j<n and k-j!=j
                # For k=2j: only a[j]*a[j] = a_j^2 (one term)
                # So c[k](x) = c[k] - a_j^2 + x^2
                ck_x = c[k] - a_j * a_j + x * x
            else:
                # c[k](x) = c[k] + (x - a_j) * e_k
                # where e_k = a[k-j] (if valid) + a[j-k] (if valid, and j-k != k-j)
                ek = 0.0
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    ek += a[k_minus_j]
                j_minus_k = j - k
                if 0 <= j_minus_k < n:
                    ek += a[j_minus_k]
                ck_x = c[k] + (x - a_j) * ek
            if ck_x > max_c:
                max_c = ck_x
        # Also check that we're not missing a max outside active_lags
        # (conservative: add a small margin)
        return 2.0 * n * max_c / (S_new * S_new)

    # Bounded 1D search
    x_min = 0.0
    x_max = max(5.0 * a_j, S_rest * 2.0, 0.1) if a_j > 1e-10 else S_rest * 3.0
    x_max = min(x_max, 100.0 * S_rest + 1.0)

    result = minimize_scalar(f_of_x, bounds=(x_min, x_max), method='bounded',
                             options={'xatol': 1e-12, 'maxiter': 50})
    if result.success and result.fun < f_of_x(a_j):
        return float(result.x)
    return a_j


def _coord_descent(a, n_sweeps=3, deadline=None, top_K=32):
    """
    Cyclic coordinate descent. Optimizes each coordinate in turn.
    """
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = _score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        S = float(a.sum())

        # Only optimize coordinates that could improve the score
        # (all of them, but in random order for diversity)
        order = np.arange(n)
        np.random.shuffle(order)

        improved_any = False
        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            new_val = _coord_descent_step(a, int(j), c, S, n)
            diff = abs(new_val - a[int(j)])
            if diff > 1e-12:
                a[int(j)] = new_val
                S = S - a[int(j)] + new_val
                improved_any = True
                # Update c (approximate: recompute periodically)
                # For exactness, recompute c every few changes
        # Recompute c at end of sweep
        c = _full_convolve(a, a)
        a = _normalize(a)
        S = float(a.sum())
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved_any:
            break

    return best_a, best_sc

        # Second: sum_k active[k]*a[j-k]
        #   = sum_{k: 0<=j-k<n} active[k]*a[j-k]
        #   = sum_{m: 0<=m<n} active[j-m]*a[m]  (letting m=j-k)
        #   This is the cross-correlation at lag -j (or equivalently, correlation of reversed active with a)

        # For vectorized computation, use FFT:
        # h1[j] = sum_m active[m+j]*a[m] for j=0..n-1
        # h2[j] = sum_m active[j-m]*a[m] for j=0..n-1
        # h[j] = h1[j] + h2[j]

        # h1: cross-correlation of active and a
        # active has length 2n-1, a has length n
        # h1[j] = sum_m active[m+j]*a[m] for m=0..n-1 (where m+j < 2n-1)
        # This is like a "sliding dot product" -- can be done with FFT
        # h1 = ifft(fft(active_padded) * conj(fft(a_padded))) -- need correct indexing

        # Let me just use np.convolve for the correlation parts (slower but correct)
        # h1[j] = sum_{m=0}^{n-1} active[m+j] * a[m]
        # = convolve(active, a_flipped, 'valid' or 'full')[...] 

        # Actually, the simplest vectorized approach:
        # h[j] = sum_k active[k] * a[k-j] + active[k] * a[j-k]
        # For the first term, define b = a (length n)
        # h1[j] = sum_k active[k] * b[k-j] = (active * b)[j] in the "correlation" sense
        #       = np.correlate(active, b, mode='full')[n-1+j] (approximately)

        # Let me just compute h directly using the structure:
        # h[j] = sum_k active[k] * a[k-j] + sum_k active[k] * a[j-k]
        # Note: sum_k active[k]*a[j-k] = sum_k active[k]*a_rev[k-j] where a_rev[k]=a[n-1-k]
        #      = (active * a_rev)[j] ... not quite, let me think differently.

        # Simplest: just loop over the top K active lags and accumulate
        h = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # a[k-j] is valid for j in [max(0, k-n+1), min(n-1, k)]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            h[j_lo:j_hi + 1] += a[k - j_lo:k - j_hi + 1][::-1]
            # a[j-k] is valid for j in [max(0, k), min(n-1, k+n-1)] = [k, min(n-1, k+n-1)]
            # Since k can be up to 2n-2, j-k valid means 0 <= j-k < n, i.e., k <= j < k+n
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_hi2 >= j_lo2:
                h[j_lo2:j_hi2 + 1] += a[j_lo2 - k:j_hi2 - k + 1]

        W = float(np.dot(active, c))
        grad = 2.0 * n * (h - a * 2.0 * W / S) / (S * S)

        # Projected gradient step with line search
        # Minimize f(t) = 2n * max_k c_k(a - t*grad_projected) / (S - t*sum(grad_proj))^2
        # Simplified: just do a fixed step with backtracking
        a_proj = a - np.mean(a * grad) / np.sum(grad * grad) * grad if np.dot(a, grad) > 0 else a - grad * 0.01

        # Actually, let's do a simple gradient step in the "log" domain
        z = np.log(np.maximum(a, 1e-30))
        grad_z = a * (grad - np.mean(a * grad))
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            z_new = z - lr * grad_z
            a_new = _softmax(z_new)
            sc_new = _score(a_new)
            if sc_new < best_sc:
                a = a_new
                best_sc = sc_new
                best_a = a.copy()
                break

    return best_a, best_sc

# Adaptive active-set Newton method
# ==============================================================

def _hess_vec(v, active, n):
    """Compute H*v where H is the Hessian of sum_{k in active} c_k."""
    N = _next_pow2(4 * n)
    v_padded = np.zeros(N)
    v_padded[:len(v)] = v
    fa = np.fft.rfft(v_padded)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    # The result at positions n-1 to 2n-2 gives H*v
    return r[n - 1:2 * n - 1]


def _newton_active(a_init, deadline, max_iter=100, tol_frac=0.005, n_top=None):
    """
    Newton's method on the active set of near-maximum lags.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    if n_top is None:
        n_top = min(32, 2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        S = float(a.sum())

        # Determine active set
        if n_top == 1:
            top_idx = np.array([int(np.argmax(c))])
        else:
            top_idx = np.argsort(c)[::-1][:n_top]
            # Only include lags that are close to max
            threshold = c_max * (1.0 - tol_frac)
            top_idx = top_idx[c[top_idx] >= threshold]
            if len(top_idx) == 0:
                top_idx = np.array([int(np.argmax(c))])

        active = np.zeros(2 * n - 1)
        active[top_idx] = 1.0 / len(top_idx)

        # Gradient: grad[j] = sum_k active[k] * (a[k-j] + a[j-k])
        h = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                h[j_lo:j_hi + 1] += a[k - j_lo:k - j_hi + 1][::-1]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_hi2 >= j_lo2:
                h[j_lo2:j_hi2 + 1] += a[j_lo2 - k:j_hi2 - k + 1]

        W = float(np.dot(active, c))
        grad = 2.0 * n * (h - a * 2.0 * W / S) / (S * S)
        # Center the gradient (for normalization invariance)
        grad = grad - np.mean(grad)

        # Check convergence
        if np.linalg.norm(grad) < 1e-12:
            break

        # Newton direction: solve H*dx = -grad where H is Hessian of sum active[k]*c_k
        # H[j,m] = sum_k active[k] * (delta_{j,m+k} + delta_{j,k-m}) (approximately)
        # Use CG for the solve
        mu = 1e-6  # regularization
        active_s = active.astype(np.float64)

        def _hv(v):
            return _hess_vec(v, active_s, n) + mu * v

        # Conjugate gradient
        dx = np.zeros(n)
        r = -grad.copy()
        p = r.copy()
        rs = float(np.dot(r, r))
        gr_norm = float(np.linalg.norm(grad))
        if gr_norm < 1e-15:
            break
        max_cg = min(100, n)
        for _ in range(max_cg):
            Hp = _hv(p)
            pH = float(np.dot(p, Hp))
            if abs(pH) < 1e-30:
                break
            alpha_cg = rs / pH
            dx = dx + alpha_cg * p
            r = r - alpha_cg * Hp
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-10 * gr_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p = r + beta_cg * p
            rs = r_new

        dx = dx - np.mean(dx)  # center for normalization invariance

        # Line search
        f_old = float(np.dot(active, c))
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(active, c_try))
            if f_try < f_old - 1e-12 * alpha:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if not improved and alpha < 1e-10:
            break

    return best_a, best_sc

# Subgradient descent with exact evaluation

def _subgrad_descent(a_init, deadline, n_steps=5000, lr_init=0.005, n_active=16):
    """
    Projected subgradient descent with adaptive step size.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    no_improve = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        S = float(a.sum())

        # Active set: top n_active lags
        if n_active == 1:
            top_idx = np.array([int(np.argmax(c))])
        else:
            top_idx = np.argsort(c)[::-1][:n_active]
            threshold = c_max * 0.95
            top_idx = top_idx[c[top_idx] >= threshold]
            if len(top_idx) == 0:
                top_idx = np.array([int(np.argmax(c))])

        # Gradient
        h = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                h[j_lo:j_hi + 1] += a[k - j_lo:k - j_hi + 1][::-1]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_hi2 >= j_lo2:
                h[j_lo2:j_hi2 + 1] += a[j_lo2 - k:j_hi2 - k + 1]

        W = float(np.dot(c[top_idx])) / len(top_idx)
        grad = 2.0 * n * (h - a * 2.0 * W / S) / (S * S)
        grad_z = a * (grad - np.dot(a, grad) / np.dot(a, a) * a)
        grad_z = a * (grad - np.mean(a * grad))

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.9
                if no_improve > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    no_improve = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# LSE-smoothed L-BFGS (from previous approach, kept as backup)
# ==============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient computation
    h = np.zeros(n)
    for k in range(len(w)):
        if abs(w[k]) < 1e-15:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            h[j_lo:j_hi + 1] += w[k] * a[k - j_lo:k - j_hi + 1][::-1]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_hi2 >= j_lo2:
            h[j_lo2:j_hi2 + 1] += w[k] * a[j_lo2 - k:j_hi2 - k + 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - a * 2.0 * W)
    return a * (grad_a - np.dot(a, grad_a))


def _lbfgs_lse(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001])

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc

# Profile generators

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.03) + np.exp(-((t - 0.7) ** 2) / 0.03)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(42)
        a = np.abs(np.sin(np.pi * np.linspace(0, 1, n) * (1 + 0.3 * rng.standard_normal())))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fourier":
        rng = np.random.default_rng(42)
        n_harm = 5
        a = np.ones(n)
        for k in range(1, n_harm + 1):
            a += 0.3 / k * np.cos(2 * np.pi * k * np.linspace(0, 1, n) + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

# Perturbation functions
# ==============================================================

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_mixture(a, b, alpha, rng):
    z1 = np.log(np.maximum(a, 1e-30))
    z2 = np.log(np.maximum(b, 1e-30))
    z = z1 + np.log(alpha / (1 - alpha))
    return _softmax(z)

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining:
    1. Fast coordinate descent (exact 1D per coordinate)
    2. Adaptive active-set Newton
    3. LSE-smoothed L-BFGS
    4. Subgradient descent
    5. Multi-n exploration with progressive refinement
    6. Diverse restarts
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # === Load parent ===
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init")

    # === Phase 1: Quick exploration with different profiles (30s) ===
    print(f"\n[Phase 1] Quick exploration")
    expl_deadline = min(deadline, t_start + 30)
    n_expl = 2048
    for kind in ["tapered", "cos2", "gauss", "bimodal", "fourier", "random_smooth"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Coordinate descent on parent (200s) ===
    print(f"\n[Phase 2] Coordinate descent on parent")
    if time.monotonic() < deadline - 150:
        n_cd = min(8192, n_parent) if n_parent > 0 else 4096
        a_cd = _resize_sequence(_best_seq, n_cd) if len(_best_seq) != n_cd else _best_seq.copy()
        cd_deadline = min(deadline - 100, time.monotonic() + 200)
        a_cd_res, sc_cd = _coord_descent(a_cd, n_sweeps=50, deadline=cd_deadline, n_top=16)
        upd(a_cd_res, sc_cd, f"coord_descent_n{n_cd}")

    # === Phase 3: Active-set Newton (200s) ===
    print(f"\n[Phase 3] Active-set Newton")
    if time.monotonic() < deadline - 150:
        n_nw = min(8192, len(_best_seq))
        a_nw_init = _resize_sequence(_best_seq, n_nw) if len(_best_seq) != n_nw else _best_seq.copy()
        nw_deadline = min(deadline - 100, time.monotonic() + 200)
        a_nw_res, sc_nw = _newton_active(a_nw_init, nw_deadline, max_iter=200, tol_frac=0.01, n_top=16)
        upd(a_nw_res, sc_nw, f"newton_n{n_nw}")

    # === Phase 4: LSE L-BFGS (300s) ===
    print(f"\n[Phase 4] LSE L-BFGS")
    if time.monotonic() < deadline - 200:
        lse_deadline = min(deadline - 100, time.monotonic() + 300)
        a_lse, sc_lse = _lbfgs_lse(_best_seq.copy(), lse_deadline, min_remaining=60)
        upd(a_lse, sc_lse, "lse_lbfgs")

    # === Phase 5: Subgradient polish (150s) ===
    print(f"\n[Phase 5] Subgradient polish")
    if time.monotonic() < deadline - 100:
        sg_deadline = min(deadline - 60, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_descent(_best_seq, sg_deadline, n_steps=3000, lr_init=0.004, n_active=16)
        upd(a_sg, sc_sg, "subgrad")

    # === Phase 6: Multi-n exploration (400s) ===
    print(f"\n[Phase 6] Multi-n exploration")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates and 2048 <= n_parent <= 65536:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        a_init_n = _resize_sequence(_best_seq, n_try)
        tag = f"n{n_try}"

        # Coordinate descent
        if time.monotonic() < deadline - 60:
            cd_dl = min(deadline - 40, time.monotonic() + 80)
            a_cd2, sc_cd2 = _coord_descent(a_init_n, n_sweeps=30, deadline=cd_dl, n_top=16)
            upd(a_cd2, sc_cd2, f"cd_{tag}")

        # Newton
        if time.monotonic() < deadline - 40:
            nw_dl = min(deadline - 20, time.monotonic() + 80)
            a_nw2, sc_nw2 = _newton_active(_best_seq if len(_best_seq) == n_try else _resize_sequence(_best_seq, n_try),
                                           nw_dl, max_iter=100, tol_frac=0.005, n_top=32)
            upd(a_nw2, sc_nw2, f"newton_{tag}")

        # LSE
        if time.monotonic() < deadline - 30:
            lse_dl = min(deadline - 15, time.monotonic() + 100)
            a_lse2, sc_lse2 = _lbfgs_lse(_best_seq if len(_best_seq) == n_try else _resize_sequence(_best_seq, n_try),
                                         lse_dl, min_remaining=10)
            upd(a_lse2, sc_lse2, f"lse_{tag}")

    # === Phase 7: Diverse restarts (200s) ===
    print(f"\n[Phase 7] Diverse restarts")
    if time.monotonic() < deadline - 40:
        n_restarts = 8
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            sigma = 0.02 + 0.01 * i
            p_start = _perturb_log(_best_seq, sigma, rng)
            if i % 2 == 1:
                p_start = _perturb_fourier(p_start, rng, max_amp=0.15)

            # Quick Newton
            nw_dl = min(deadline - 15, time.monotonic() + 25)
            a_r, sc_r = _newton_active(p_start, nw_dl, max_iter=60, tol_frac=0.01, n_top=16)
            upd(a_r, sc_r, f"restart_{i}")

            # If good, do LSE
            if sc_r < _best_score * 1.002 and time.monotonic() < deadline - 20:
                lse_dl = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _lbfgs_lse(a_r, lse_dl, min_remaining=5)
                upd(a_r2, sc_r2, f"restart_{i}_lse")

    # === Phase 8: Final aggressive Newton + subgradient (100s) ===
    print(f"\n[Phase 8] Final polish")
    if time.monotonic() < deadline - 10:
        # Very tight Newton
        nw_dl = min(deadline - 5, time.monotonic() + 40)
        a_fin, sc_fin = _newton_active(_best_seq, nw_dl, max_iter=100, tol_frac=0.001, n_top=32)
        upd(a_fin, sc_fin, "final_newton")

        # Subgradient with very small steps
        if time.monotonic() < deadline - 5:
            sg_dl = min(deadline - 2, time.monotonic() + 30)
            a_fin2, sc_fin2 = _subgrad_descent(_best_seq, sg_dl, n_steps=2000, lr_init=0.002, n_active=32)
            upd(a_fin2, sc_fin2, "final_sg")

        # One more coordinate descent pass
        if time.monotonic() < deadline - 3:
            cd_dl = min(deadline - 1, time.monotonic() + 20)
            a_fin3, sc_fin3 = _coord_descent(_best_seq, n_sweeps=10, deadline=cd_dl, n_top=16)
            upd(a_fin3, sc_fin3, "final_cd")

    # === Final output ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform").tolist()
            print("FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
