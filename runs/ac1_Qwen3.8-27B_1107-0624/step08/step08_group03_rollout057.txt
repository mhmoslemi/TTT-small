"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
- Multi-n exploration (8192, 16384, 32768, 65536)
- Aggressive log-sum-exp L-BFGS with continuation

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()

    current_tol = tol_active

    for it in range(max_iter):
        if time.monotonic() > deadline:

        c = _full_convolve(a, a)

        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)

            a_try = a + t * dx

                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_beta_steps=20):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # Create beta schedule - more aggressive with more steps
    if n_beta_steps <= 10:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                      0.015, 0.0075, 0.0037, 0.00185])
    elif n_beta_steps <= 20:
        betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                                      0.08, 0.05, 0.035, 0.025, 0.018, 0.012,
                                      0.008, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002])
    else:
        betas = base_beta * np.logspace(0, np.log10(0.0001), n_beta_steps)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations based on n
        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.006
        elif n <= 16384:
            est_time_per_iter = 0.012
        elif n <= 32768:
            est_time_per_iter = 0.025
        else:
            est_time_per_iter = 0.05

        # Decrease max_iter as beta gets smaller (converges faster)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))
        if i > len(betas) // 2:
            max_iter = min(max_iter, 1000)

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _cyclical_peak_shave(a_init, deadline, max_rounds=100):
    """Cyclical peak shaving: iteratively target each of the top peaks in turn."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Get top peaks
        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[top_peaks]
        
        # Use exponential weighting favoring higher peaks
        weights = np.exp(-0.5 * (peak_vals - peak_vals.min()) / (peak_vals.max() - peak_vals.min() + 1e-15))
        weights = weights / weights.sum()

        w = np.zeros(2 * n - 1)
        w[top_peaks] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        grad = 2.0 * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Adaptive step size based on gradient magnitude
        step = 0.5 / (1.0 + 0.02 * rnd)
        
        for ls in range(30):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top-k with exponential weighting
        k = min(50, len(c) // 5)
        top_k = np.argsort(c)[::-1][:k]

        weights = np.exp(-0.3 * (peak_vals - peak_vals.min()) / (peak_vals.max() - peak_vals.min() + 1e-15))
        weights = weights / weights.sum()
        w = np.zeros_like(c)

def _interpolate_to_n(a, n_new):

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()

    return a_new / s

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-12 * (1.0 + rhs_norm):
            break
        if rs < 1e-30:
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001, adapt_tol=True):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    current_tol = tol_active

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = current_tol * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            n_active = 4 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        if adapt_tol:
            target_active = n
            if n_active < max(1, target_active // 4):
                current_tol = min(current_tol * 1.5, 0.01)
            elif n_active > target_active * 4:
                current_tol = max(current_tol * 0.5, 0.00005)

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        u = _cg_solve(g_t, A_vec, n, mu, m, max_cg=200)
        w = _cg_solve(np.ones(n), A_vec, n, mu, m, max_cg=200)
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if a_try.min() < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_beta_steps=20):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    if n_beta_steps <= 10:
        betas = base_beta * np.array([1.0, 0.6, 0.35, 0.2, 0.1, 0.05,
                                      0.025, 0.012, 0.006, 0.003])
    elif n_beta_steps <= 20:
        betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                                      0.08, 0.055, 0.038, 0.026, 0.018, 0.012,
                                      0.008, 0.0055, 0.0035, 0.0022, 0.0014,
                                      0.0008, 0.0004])
    else:
        betas = base_beta * np.logspace(0.0, np.log10(0.0003), n_beta_steps)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.006
        elif n <= 16384:
            est_time_per_iter = 0.012
        elif n <= 32768:
            est_time_per_iter = 0.025
        else:
            est_time_per_iter = 0.05

        # More iterations for larger beta (coarser), fewer for smaller
        frac = i / max(1, len(betas) - 1)
        if frac < 0.3:
            max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 5))))
        elif frac < 0.7:
            max_iter = min(2000, max(30, int(remaining / (est_time_per_iter * 8))))
        else:
            max_iter = min(1000, max(20, int(remaining / (est_time_per_iter * 12))))

        obj_func = functools.partial(_lsobj, beta=beta, n=n)
        grad_func = functools.partial(_lsgrad, beta=beta, n=n)

        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _cyclical_peak_shave(a_init, deadline, max_rounds=100):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_peaks = min(30, 2 * n - 1)
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[top_peaks]
        pv_range = peak_vals.max() - peak_vals.min()
        if pv_range < 1e-15:
            break
        weights = np.exp(-0.8 * (peak_vals - peak_vals.min()) / pv_range)
        weights = weights / weights.sum()

        w = np.zeros(2 * n - 1)
        w[top_peaks] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        grad = 2.0 * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = 0.4 / (1.0 + 0.015 * rnd)
        improved = False
        for ls in range(35):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-15:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            c_new = _full_convolve(a, a)
            if abs(float(c_new.max()) - c_max) < 1e-12:
                break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=5000, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        k = min(50, max(10, len(c) // 10))
        top_k = np.argsort(c)[::-1][:k]
        peak_vals = c[top_k]
        pv_range = peak_vals.max() - peak_vals.min()
        if pv_range < 1e-15:
            weights = np.ones(k) / k
        else:
            weights = np.exp(-0.5 * (peak_vals - peak_vals.min()) / pv_range)
            weights = weights / weights.sum()
        w = np.zeros_like(c)
        w[top_k] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_smooth(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    noise = rng.standard_normal(n)
    # Smooth the noise to avoid high-frequency perturbation
    k = max(2, n // 200)
    kernel = np.ones(k) / k
    noise = np.convolve(noise, kernel, mode='same')
    z = z + sigma * noise
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Combines L-BFGS continuation, active-set Newton, cyclical peak shaving,
    mirror descent, and multi-n exploration with cross-interpolation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, label):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        _try_update(parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _try_update(parent, "init")

    # Phase 1: L-BFGS on parent (allocate ~35% of time)
    print(f"\n[Phase 1] L-BFGS on n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + budget_s * 0.32
    p1_deadline = min(p1_deadline, deadline - budget_s * 0.3)
    if time.monotonic() < p1_deadline:
        a1 = _run_lbfgs(best_seq if best_seq is not None else parent,
                        p1_deadline, min_time_remaining=20.0, n_beta_steps=22)
        _try_update(a1, f"lbfgs_n{n_parent}")

    # Phase 2: L-BFGS on n=32768 (allocate ~25% of time)
    print(f"\n[Phase 2] L-BFGS on n=32768 (t={time.monotonic() - t_start:.1f}s)")
    n_large = 32768
    p2_deadline = t_start + budget_s * 0.55
    p2_deadline = min(p2_deadline, deadline - budget_s * 0.2)
    if time.monotonic() < p2_deadline:
        a2_init = _interpolate_to_n(best_seq, n_large)
        sc2_init = _score(a2_init)
        print(f"  n={n_large} interpolated init score: {sc2_init:.10f}")
        _try_update(a2_init, f"interp_n{n_large}")

        a2 = _run_lbfgs(a2_init, p2_deadline, min_time_remaining=20.0, n_beta_steps=15)
        _try_update(a2, f"lbfgs_n{n_large}")

        # Quick peak shave
        if time.monotonic() < p2_deadline + 120:
            p2b_deadline = min(p2_deadline + 120, deadline - budget_s * 0.15)
            a2b, sc2b = _cyclical_peak_shave(a2, p2b_deadline, max_rounds=60)
            _try_update(a2b, f"peakshave_n{n_large}")

    # Phase 3: L-BFGS on n=65536 (allocate ~20% of time)
    print(f"\n[Phase 3] L-BFGS on n=65536 (t={time.monotonic() - t_start:.1f}s)")
    n_xlarge = 65536
    p3_deadline = t_start + budget_s * 0.75
    p3_deadline = min(p3_deadline, deadline - budget_s * 0.1)
    if time.monotonic() < p3_deadline:
        a3_init = _interpolate_to_n(best_seq, n_xlarge)
        sc3_init = _score(a3_init)
        print(f"  n={n_xlarge} interpolated init score: {sc3_init:.10f}")
        _try_update(a3_init, f"interp_n{n_xlarge}")

        a3 = _run_lbfgs(a3_init, p3_deadline, min_time_remaining=20.0, n_beta_steps=10)
        _try_update(a3, f"lbfgs_n{n_xlarge}")

    # Phase 4: Active-set Newton on best (allocate ~10% of time)
    print(f"\n[Phase 4] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    p4_deadline = min(deadline - 30, time.monotonic() + budget_s * 0.10)
    if best_seq is not None and time.monotonic() < p4_deadline:
        a4, sc4 = _active_set_newton(best_seq, p4_deadline,
                                      max_iter=300, tol_active=0.002, adapt_tol=True)
        _try_update(a4, "active_set")

        # Tighter
        if time.monotonic() < deadline - 15:
            p4b_deadline = min(deadline - 8, time.monotonic() + 60)
            a4b, sc4b = _active_set_newton(best_seq, p4b_deadline,
                                            max_iter=150, tol_active=0.0003, adapt_tol=True)
            _try_update(a4b, "active_set_tight")

    # Phase 5: Cyclical peak shaving (allocate ~5%)
    print(f"\n[Phase 5] Cyclical peak shaving (t={time.monotonic() - t_start:.1f}s)")
    p5_deadline = min(deadline - 20, time.monotonic() + budget_s * 0.05)
    if best_seq is not None and time.monotonic() < p5_deadline:
        a5, sc5 = _cyclical_peak_shave(best_seq, p5_deadline, max_rounds=100)
        _try_update(a5, "cyclical_peak")

    # Phase 6: Mirror descent (allocate ~5%)
    print(f"\n[Phase 6] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    p6_deadline = min(deadline - 10, time.monotonic() + budget_s * 0.05)
    if best_seq is not None and time.monotonic() < p6_deadline:
        a6, sc6 = _mirror_descent(best_seq, p6_deadline, max_steps=5000, lr_init=0.04)
        _try_update(a6, "mirror_descent")

    # Phase 7: Random restarts + re-optimize (remaining time)
    print(f"\n[Phase 7] Random restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        restart_sigs = [0.02, 0.05, 0.10]
        for sig in restart_sigs:
            if time.monotonic() > deadline - 15:
                break
            p_re = _perturb_smooth(best_seq, sig, rng)
            sc_re = _score(p_re)
            # Quick L-BFGS on perturbed
            rl_deadline = min(deadline - 10, time.monotonic() + budget_s * 0.03)
            a_re = _run_lbfgs(p_re, rl_deadline, min_time_remaining=5.0, n_beta_steps=8)
            _try_update(a_re, f"restart_sig{sig}")
            # Quick Newton
            if time.monotonic() < deadline - 8:
                a_re2, sc_re2 = _active_set_newton(a_re, min(deadline - 5, time.monotonic() + 30),
                                                    max_iter=50, tol_active=0.001, adapt_tol=True)
                _try_update(a_re2, f"restart_sig{sig}_as")

    # Phase 8: Final very tight Newton
    print(f"\n[Phase 8] Final tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a8, sc8 = _active_set_newton(best_seq, deadline - 1,
                                      max_iter=100, tol_active=0.0001, adapt_tol=True)
        _try_update(a8, "final_tight")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
