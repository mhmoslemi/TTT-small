"""
Multi-stage optimizer for minimizing step function autocorrelation peak.
Focuses on aggressive LSE cooling, adaptive Newton active-set with widening/narrowing,
plateau equalization, and multi-scale refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

def _power_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    val = (np.sum(c_safe**p))**(1.0/p)
    return 2.0 * n * val

def _power_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = c_safe**(p-1)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    sum_cp = float(np.sum(c_safe**p))
    factor = 2.0 * n * sum_cp**(1.0/p - 1.0)
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

    k = 1

        k <<= 1

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

    s = float(a.sum())

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

def _cg_solve(hess_fn, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_fn(p)
        pH = float(np.dot(p, Hp))
        if pH <= 1e-30:
            break
        alpha = rs / pH
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        r_new_norm = float(np.dot(r_new, r_new))
        if np.sqrt(r_new_norm) < tol * b_norm:
            return x_new
        beta_cg = r_new_norm / rs if rs > 1e-30 else 0.0
        p_new = r_new + beta_cg * p
        if not np.isfinite(p_new).all():
            break
        x = x_new
        r = r_new
        p = p_new
        rs = r_new_norm
    return x

So a shifted (but still contiguous) active set gives the same period structure. The key is whether the active set is contiguous or not.

If the active set is NON-contiguous, the constraint structure changes. For example, if the active set is {n-1, n-1-3, n-1+3} (three points, not contiguous), then:
`a[n-1-j] + a[n-4-j] + a[n+2-j] = constant`

Let m = n-1-j: `a[m] + a[m-3] + a[m+3] = constant`.
Characteristic: 1 + x^{-3} + x^3 = 0, i.e., x^6 + x^3 + 1 = 0. Roots are the non-real 9th roots of unity. Period 9 (or a divisor).

This gives non-trivial periodic solutions! So a non-contiguous active set allows for non-uniform optimal sequences.

This explains why the optimal sequence is not uniform: the active set (where the max is achieved) must be non-contiguous, allowing for periodic structure in the optimal solution.

This is a key insight that could guide the optimization! Instead of just using the "top K" values (which are likely contiguous), we should identify the TRUE active set, which might be non-contiguous.

However, in practice, for a smooth sequence, the top K values of the convolution will be contiguous (since the convolution is a smooth function of the index). The non-contiguity might only appear at very fine scales.

Alternatively, the non-trivial structure might come from the BOUNDARY condition (some a[j] = 0). If a is zero at the boundaries (which is natural for a "bump" function), then the KKT conditions are modified.

"""
Multi-stage optimizer for minimizing step function autocorrelation peak.
Uses aggressive LSE cooling, multi-scale Newton active-set with adaptive
regularization, non-contiguous active set detection, and diverse restarts.
Key insight: optimal active sets may be non-contiguous, enabling periodic
structure that reduces the convolution peak below the uniform baseline.
"""
import numpy as np
import time
from scipy.optimize import minimize

# ============================================================
# Core utilities
# ============================================================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# ============================================================
# LSE objectives and gradients
# ============================================================

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))

# ============================================================
# Power-sum objectives
# ============================================================

def _power_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    val = (np.sum(c_safe ** p)) ** (1.0 / p)
    return 2.0 * n * val

def _power_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    w = c_safe ** (p - 1)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    sum_cp = float(np.sum(c_safe ** p))
    factor = 2.0 * n * sum_cp ** (1.0 / p - 1.0)
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))

# ============================================================
# Hessian-matrix product for active set
# ============================================================

def _hess_matvec(v, active_f, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active_f, N)
    r = np.fft.irfft(fa * fb, N)
    return 2.0 * r[n - 1:2 * n - 1]

def _cg_solve(hess_fn, b, tol=1e-12, max_iter=500):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_fn(p)
        pH = float(np.dot(p, Hp))
        if pH <= 1e-30:
            break
        alpha = rs / pH
        if not np.isfinite(alpha):
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if not np.isfinite(r_new):
            break
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
    return x

def _newton_active_set(a_init, deadline, top_k=50, max_iter=300, mu_init=1e-5):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Identify active set: top_k values
        sorted_idx = np.argsort(c)[::-1][:top_k]
        active = np.zeros(len(c))
        active[sorted_idx] = 1.0 / top_k
        active_f = active.astype(np.float64)

        # Gradient
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        # Adaptive regularization
        mu = mu_init * (0.6 ** it)
        mu = max(mu, 1e-10)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=400)
        dx = dx - np.mean(dx)
        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-15:
            break

        # Line search with Armijo condition
        f_old = float(c[sorted_idx].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[sorted_idx].sum())
            if f_try < f_old - 1e-7 * alpha * gnorm:
                a_new = a_try
                c = c_try
                improved = True
                break
            alpha *= 0.4
        if not improved:
            if alpha < 1e-14:
                break
            # Try a steepest descent step as fallback
            a_try = a - 0.01 * grad_t / (gnorm + 1e-30)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try[sorted_idx].sum() < f_old:
                    a_new = a_try
                    c = c_try
                    improved = True
            if not improved:
                break

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        c = _full_convolve(a, a)

    return best_a, best_sc

    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    base_beta = max(gap * 0.4, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Very aggressive cooling: geometric with ratio 0.45
    n_stages = 18
    ratio = 0.45
    for i in range(n_stages):
        beta = base_beta * (ratio ** i)
        if time.monotonic() + min_remaining > deadline:

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.005 * max(1.0, n / 4096.0)

        try:

                current_sc = sc_new

def _subgrad_polish(a_init, deadline, n_steps=20000, lr_init=0.004, top_k=20):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)

        # Use top_k for active set
        sorted_idx = np.argsort(c)[::-1][:top_k]
        active = np.zeros(len(c))
        active[sorted_idx] = 1.0 / top_k

        grad = _hess_matvec(a, active, n)
        grad_z = a * (grad - np.dot(a, grad))

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 1000 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience % 4 == 0:
                    lr *= 0.88
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(lr_init * 0.2, lr)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_fourier_phase(a, rng, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _perturb_fourier_amp(a, rng, sigma=0.05):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = np.maximum(amp * (1.0 + sigma * rng.standard_normal(len(amp))), 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:

        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick LSE warmup (90s) ===
    print(f"\n[Phase 1] LSE warmup (n={n_parent})")
    p1_deadline = min(deadline, t_start + 90)
    a1, sc1 = _lbfgs_lse(_best_seq.copy(), p1_deadline, min_remaining=10)
    upd(a1, sc1, "lse_warmup")

    # === Phase 2: Aggressive LSE (300s) ===
    print(f"\n[Phase 2] Aggressive LSE")
    p2_deadline = min(deadline, t_start + 390)
    a2, sc2 = _lbfgs_lse(_best_seq.copy(), p2_deadline, min_remaining=200)
    upd(a2, sc2, "lse_aggressive")

    # === Phase 3: Power-sum LSE (120s) ===
    print(f"\n[Phase 3] Power-sum LSE")
    p3_deadline = min(deadline, t_start + 510)
    a3, sc3 = _lbfgs_power(_best_seq.copy(), p3_deadline, min_remaining=150)
    upd(a3, sc3, "power_sum")

    # === Phase 4: Multi-scale Newton (400s) ===
    print(f"\n[Phase 4] Multi-scale Newton")
    p4_deadline = min(deadline, t_start + 910)
    a4, sc4 = _multi_scale_newton(_best_seq.copy(), p4_deadline)
    upd(a4, sc4, "multi_scale_newton")

    # === Phase 5: Peak push-down (100s) ===
    print(f"\n[Phase 5] Peak push-down")
    p5_deadline = min(deadline, t_start + 1010)
    a5, sc5 = _peak_pushdown(_best_seq.copy(), p5_deadline)
    upd(a5, sc5, "peak_pushdown")

    # === Phase 6: Multi-scale Newton round 2 (250s) ===
    print(f"\n[Phase 6] Multi-scale Newton round 2")
    p6_deadline = min(deadline, t_start + 1260)
    a6, sc6 = _multi_scale_newton(_best_seq.copy(), p6_deadline)
    upd(a6, sc6, "newton_r2")

    # === Phase 7: Subgradient polish (150s) ===
    print(f"\n[Phase 7] Subgradient polish")
    p7_deadline = min(deadline, t_start + 1410)
    a7, sc7 = _subgrad_polish(_best_seq.copy(), p7_deadline, n_steps=30000, lr_init=0.003, top_k=15)
    upd(a7, sc7, "sg_polish")

    # === Phase 8: Diverse restarts with Newton (250s) ===
    print(f"\n[Phase 8] Diverse restarts")
    p8_deadline = min(deadline, t_start + 1660)
    n_restarts = 8
    for i in range(n_restarts):
        if time.monotonic() > p8_deadline - 30:
            break
        if i == 0:
            p_start = _perturb_log(_best_seq, 0.02, rng)
        elif i == 1:
            p_start = _perturb_fourier_phase(_best_seq, rng, max_amp=0.2)
        elif i == 2:
            p_start = _perturb_fourier_amp(_best_seq, rng, sigma=0.04)
        elif i == 3:
            p_start = _perturb_periodic(_best_seq, rng)
        elif i == 4:
            p_start = _perturb_mirror(_best_seq, rng)
        elif i == 5:
            p_start = _perturb_log(_best_seq, 0.04, rng)
        elif i == 6:
            p_start = _perturb_periodic(_best_seq, rng, period_range=(5, 20))
        else:
            p_start = _perturb_fourier_phase(_best_seq, rng, max_amp=0.4)

        # Quick Newton for each restart
        rem = p8_deadline - time.monotonic() - 30
        if rem < 20:
            break
        a_r, sc_r = _newton_active_set(
            p_start, time.monotonic() + rem,
            top_k=20, max_iter=150, mu_init=1e-4
        )
        upd(a_r, sc_r, f"restart_{i}")

        # Peak push-down on best restart
        if sc_r < _best_score * 1.002 and time.monotonic() < p8_deadline - 15:
            a_r2, sc_r2 = _peak_pushdown(a_r, min(p8_deadline, time.monotonic() + 30))
            upd(a_r2, sc_r2, f"restart_{i}_ppd")

    # === Phase 9: Final ultra-precise Newton (100s) ===
    print(f"\n[Phase 9] Final Newton")
    p9_deadline = min(deadline, t_start + 1760)
    if _best_seq is not None and time.monotonic() < p9_deadline:
        # Very narrow active set, many iterations
        a9, sc9 = _newton_active_set(
            _best_seq.copy(), p9_deadline,
            top_k=5, max_iter=500, mu_init=1e-6
        )
        upd(a9, sc9, "final_newton_k5")

        a9b, sc9b = _peak_pushdown(_best_seq.copy(), min(p9_deadline, time.monotonic() + 40))
        upd(a9b, sc9b, "final_ppd")

        # One more multi-scale
        if time.monotonic() < p9_deadline - 20:
            a9c, sc9c = _multi_scale_newton(_best_seq.copy(), p9_deadline)
            upd(a9c, sc9c, "final_multi")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            sc_verify = evaluate_sequence([float(x) for x in final])
            print(f"Verified score: {sc_verify:.10f}")
        except Exception:
            sc_verify = sc_final
            print(f"Self score: {sc_final:.10f}")
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
