def _compute_subgrad(a, active_indices):
    n = len(a)
    grad = np.zeros(n)
    m = len(active_indices)
    for k in active_indices:
        # Contribution from a[k-j]: valid when 0 <= k-j < n => k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # For j in [j_lo, j_hi], a[k-j] has k-j in [k-j_hi, k-j_lo] = [max(0,k-n+1), min(n-1,k)]

            # So a[k-j] = a[k-j_lo], a[k-j_lo-1], ..., a[k-j_hi]
            # That's a[k-j_hi : k-j_lo+1][::-1] added to grad[j_lo : j_hi+1]

    # by the fact that adding a constant to z doesn't change a.

    # grad_z = grad_a * a - mean(grad_a * a) (since z_j = log(a_j) + const)
    # Hmm, actually: a_j = exp(z_j) / sum(exp(z)), so da_j/dz_k = a_j * (delta_jk - a_k)
    # d/dz_k [f(a)] = sum_j df/da_j * da_j/dz_k = sum_j g_j * a_j * (delta_jk - a_k)
    # = g_k * a_k - a_k * sum_j g_j * a_j
    # So grad_z_k = a_k * (g_k - sum_j g_j * a_j)
    g = 4.0 * n / beta * aw  # this is df/da_j / a_j (since df/da_j = 4n/beta * a_j * aw_j)
    # Wait, let me redo this.
    # f(a) = 2n * (c_max + beta * log(sum exp((c-c_max)/beta)))
    # df/da_j = 2n/beta * sum_k w_k * dc_k/da_j = 2n/beta * sum_k w_k * 2*a[k-j]*1[valid]
    # = 4n/beta * (a * w)[j] where (a*w)[j] = sum_i a[i]*w[i+j]... 
    # NO! dc_k/da_j = 2*a[k-j]*1[0<=k-j<n]. So sum_k w_k * dc_k/da_j = 2*sum_k w_k*a[k-j]*1[valid]
    # Let i = k-j: = 2*sum_i w_i*a[i-j+... ] hmm I'm confusing myself.
    # Let me just use: sum_k w_k * 2*a[k-j]*1[0<=k-j<n] = 2 * (w_convolved_with_a_at_offset_j)
    # = 2 * sum_i w[i+j] * a[i] = 2 * _cross_correlate(a, w)[j]
    # So df/da_j = 2n/beta * 2 * _cross_correlate(a, w)[j] = 4n/beta * aw[j]
    # where aw = _cross_correlate(a, w)
    
    # Now for the z-gradient:
    # grad_z_k = a_k * (df/da_k - sum_j a_j * df/da_j)
    # = a_k * (g_k - sum_j a_j * g_j) where g_j = df/da_j = 4n/beta * aw[j]
    g_full = 4.0 * n / beta * aw
    mean_g = float(np.dot(a, g_full))
    grad_z = a * (g_full - mean_g)
    return grad_z


def _lbfgs_polish(a_init, deadline, min_time=30.0):
    """L-BFGS polish with log-sum-exp smoothing, multiple beta values."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return best_a, best_sc

    base_beta = max_c * 0.005
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        max_iter = min(3000, max(50, int(remaining / (0.002 * max(1.0, n / 4096.0) * 10))))
        try:
            result = minimize(
                lambda zz: _smoobj(zz, beta, n),
                z,
                jac=lambda zz: _smograd(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 50}
            )
            a_new = np.exp(result.x - np.max(result.x))
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shaped":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "w_shaped":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 2.0 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return np.exp(z - np.max(z)) / np.exp(z - np.max(z)).sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Initialize from parent or diverse profiles
    2. Mirror descent (fast, exploits simple gradient structure)
    3. Hankel Richardson (exploits KKT structure for faster convergence)
    4. L-BFGS polish (fine-tuning)
    5. Multi-restart + n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_main = n_parent

    # === Phase 1: Mirror descent on parent ===
    print(f"\n[P1] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    md_deadline = min(deadline - 600, t_start + 120)
    a_md, sc_md = _mirror_descent_core(best_seq, md_deadline, min_time=30,
                                       n_peaks_init=1, n_peaks_max=n_main//200,
                                       eta_init=0.03)
    if sc_md < best_score:
        best_seq = a_md.copy()
        best_score = sc_md
        print(f"  Mirror descent: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Hankel Richardson ===
    print(f"\n[P2] Hankel Richardson (t={time.monotonic()-t_start:.1f}s)")
    hr_deadline = min(deadline - 400, t_start + 300)
    a_hr, sc_hr = _hankel_richardson(best_seq, hr_deadline, min_time=30,
                                     alpha_init=0.3)
    if sc_hr < best_score:
        best_seq = a_hr.copy()
        best_score = sc_hr
        print(f"  Hankel: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Alternate mirror + Hankel ===
    print(f"\n[P3] Alternating MD+HR (t={time.monotonic()-t_start:.1f}s)")
    alt_deadline = min(deadline - 200, t_start + 600)
    current = best_seq.copy()
    for round_idx in range(20):
        if time.monotonic() + 100 > alt_deadline:
            break
        sub_dl = min(alt_deadline - 50, time.monotonic() + 30)
        a1, sc1 = _mirror_descent_core(current, sub_dl, min_time=5,
                                       n_peaks_init=1, n_peaks_max=n_main//100,
                                       eta_init=0.02)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            current = a1.copy()
        sub_dl2 = min(alt_deadline - 30, time.monotonic() + 25)
        a2, sc2 = _hankel_richardson(current, sub_dl2, min_time=5, alpha_init=0.2)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            current = a2.copy()
        if round_idx % 5 == 4:
            print(f"  Round {round_idx}: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS polish ===
    print(f"\n[P4] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        lbfgs_dl = min(deadline - 140, t_start + 900)
        a_lp, sc_lp = _lbfgs_polish(best_seq, lbfgs_dl, min_time=60)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  L-BFGS: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Post-LBFGS mirror + Hankel ===
    print(f"\n[P5] Post-polish MD+HR (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        post_dl = min(deadline - 90, t_start + 1100)
        a_p1, sc_p1 = _mirror_descent_core(best_seq, post_dl, min_time=30,
                                           n_peaks_init=1, n_peaks_max=n_main//100,
                                           eta_init=0.01)
        if sc_p1 < best_score:
            best_seq = a_p1.copy()
            best_score = sc_p1
        a_p2, sc_p2 = _hankel_richardson(best_seq, min(deadline - 80, time.monotonic() + 50),
                                         min_time=20, alpha_init=0.1)
        if sc_p2 < best_score:
            best_seq = a_p2.copy()
            best_score = sc_p2
            print(f"  Post-polish: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        for restart in range(5):
            if time.monotonic() > deadline - 100:
                break
            sigma = 0.05 * (restart + 1) * 0.5
            a_pert = _perturb(best_seq, sigma, rng)
            sub_dl = min(deadline - 80, time.monotonic() + 40)
            a_r, sc_r = _mirror_descent_core(a_pert, sub_dl, min_time=10,
                                             n_peaks_init=1, n_peaks_max=n_main//200,
                                             eta_init=0.02)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  Restart {restart} (sigma={sigma:.3f}): {best_score:.10f}")
            # Quick polish
            if time.monotonic() < deadline - 60:
                a_r2, sc_r2 = _hankel_richardson(a_r, min(deadline - 60, time.monotonic() + 20),
                                                 min_time=10, alpha_init=0.1)
                if sc_r2 < best_score:
                    best_seq = a_r2.copy()
                    best_score = sc_r2

    # === Phase 7: Different n values ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_options = [8192, 16384, 32768]
    n_options = [x for x in n_options if x != n_main]
    n_options.insert(0, n_main)

    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_main:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick optimization
        sub_dl = min(deadline - 60, time.monotonic() + 60)
        a_opt, sc_opt = _mirror_descent_core(a_init, sub_dl, min_time=15,
                                             n_peaks_init=1, n_peaks_max=n_try//200,
                                             eta_init=0.03)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  n={n_try}: {best_score:.10f}")
        if time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _hankel_richardson(a_opt, min(deadline - 40, time.monotonic() + 30),
                                                 min_time=10, alpha_init=0.2)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  n={n_try} HR: {best_score:.10f}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        fin_dl = deadline - 5
        a_fin, sc_fin = _mirror_descent_core(best_seq, fin_dl, min_time=5,
                                             n_peaks_init=1, n_peaks_max=best_seq.size//50,
                                             eta_init=0.005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin

        # Tiny L-BFGS
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            max_iter = min(1000, max(50, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda zz: _smoobj(zz, beta_fin, n),
                    z,
                    jac=lambda zz: _smograd(zz, beta_fin, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 40}
                )
                a_fin2 = np.exp(result.x - np.max(result.x))
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
