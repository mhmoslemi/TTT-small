    n = len(a)
    N = _next_pow2(2*n - 1)
    a_cur = a.copy()
    a_cur = a_cur / a_cur.sum()
    best_score = _score(a_cur)
    best_a = a_cur.copy()
    
    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        # Zero-pad and compute FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a_cur
        A = np.fft.fft(a_pad)
        
        # Current magnitude
        mag = np.abs(A)
        
        # Target: flat magnitude for non-DC bins
        # Use a "soft" target - blend toward the geometric mean
        mag_non_dc = mag[1:N//2+1]  # For real signal, only need half
        # Actually for rfft we'd use rfft, but let's use full fft for flexibility
        target = np.median(mag[1:])  # or mean
        
        # Blend
        alpha_cur = alpha * (1.0 + 0.5 * np.cos(2*np.pi*it/n_iters))  # adaptive alpha
        mag_new = np.copy(mag)
        mag_new[1:] = (1 - alpha_cur) * mag[1:] + alpha_cur * target
        mag_new[0] = mag[0]  # Keep DC
        
        # Reconstruct with new magnitude, old phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]  # Take first n elements
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check score
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a_cur = a_new
    
    return best_a, best_score

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so just 2n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    a_rev = a[::-1]

    W = float(np.dot(w, c))

def _power_sum_grad(z, p, n):

def _active_gradient(a, c, n, tol_frac=0.002):

def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300, tol_frac=0.002, min_active_frac=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_sc = _score(a)

    for it in range(max_iter):

                    patience = 0

def _spectral_shape(a_init, deadline, alpha_init=0.05, n_max_iters=20000):
    """
    Gerchberg-Saxton style spectral shaping.
    Iteratively flatten the power spectrum while maintaining non-negativity.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a_cur = a_init.copy()
    a_cur = np.maximum(a_cur, 0)
    s = a_cur.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a_cur = a_cur / s
    best_a = a_cur.copy()
    best_sc = _score(a_cur)

    # Pre-compute DC value (should be sum(a) = 1)
    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break

        # Compute FFT with zero-padding
        a_pad = np.zeros(N)
        a_pad[:n] = a_cur
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)

        # For real signal, only need half: [0, N//2]
        half = N // 2 + 1
        mag_half = mag[:half]

        # Target: flat magnitude for non-DC bins
        # Use adaptive target based on current state
        mag_non_dc = mag_half[1:]
        if len(mag_non_dc) > 0:
            target = np.median(mag_non_dc)
        else:
            target = mag_half[0] / np.sqrt(n)

        # Adaptive alpha: start gentle, get more aggressive, then settle
        progress = it / max(1, min(n_max_iters, 5000))
        alpha = alpha_init * (1.0 + 2.0 * np.sin(np.pi * min(progress, 1.0)))
        alpha = np.clip(alpha, 0.005, 0.3)

        # Blend magnitude toward flat target
        mag_new = np.zeros(N, dtype=np.complex128)
        # Work in full spectrum for simplicity
        mag_full = np.abs(A)
        mag_full[1:] = (1 - alpha) * mag_full[1:] + alpha * target
        mag_full[0] = np.abs(A[0])  # Keep DC
        # For conjugate symmetry
        mag_full[N - np.arange(1, half)] = mag_full[np.arange(1, half)]

        A_new = mag_full * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check improvement
        if it % 50 == 0:
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

        a_cur = a_new

    # Final check
    sc_final = _score(a_cur)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_cur.copy()

    return best_a, best_sc

        return a_init, _score(a_init)
    a_cur = a_cur / s
    best_a = a_cur.copy()
    best_sc = _score(a_cur)

    for it in range(10000):
        if time.monotonic() > deadline:
            break

        a_pad = np.zeros(N)
        a_pad[:n] = a_cur
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)

        if it % 2 == 0:
            # Magnitude step: flatten
            target = np.median(mag[1:N // 2 + 1])
            mag_new = np.copy(mag)
            mag_new[1:] = (1 - alpha_mag) * mag[1:] + alpha_mag * target
            mag_new[0] = mag[0]
            A_new = mag_new * np.exp(1j * phase)
        else:
            # Phase step: adjust to improve non-negativity
            # Use a "projection" in phase space
            a_test = np.fft.ifft(A).real[:n]
            n_neg = np.sum(a_test < 0)
            if n_neg > 0:
                # Rotate phase slightly to reduce negative values
                neg_mask = a_test < 0
                if neg_mask.sum() > 0:
                    # Simple approach: add a small phase tilt
                    phase_tilt = np.linspace(0, 0.1 * alpha_phase, N)
                    phase_new = phase + phase_tilt
                    A_new = mag * np.exp(1j * phase_new)
                else:
                    A_new = A
            else:
                A_new = A

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_cur = a_new

        if it % 100 == 0:
            sc = _score(a_cur)
            if sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()

    sc_final = _score(a_cur)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_cur.copy()
    return best_a, best_sc


def _coordinate_descent_active(a_init, deadline, n_sweeps=20, tol_frac=0.002):
    """
    Coordinate descent on the active set. For each coordinate, find the optimal
    value that minimizes the active set cost while maintaining non-negativity.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_f = active.astype(np.float64)

        # Randomize coordinate order
        perm = np.random.permutation(n)

        for idx in perm:
            if time.monotonic() > deadline:
                break
            m = int(idx)
            # Current contribution to active cost from coordinate m
            # c[j] depends on a[m] through: a[m]*a[j-m] for valid j-m
            # For active j: c[j] = a[m]*a[j-m] + (other terms)
            # ∂(active_cost)/∂a[m] = sum_{j in active} (a[j-m] + a[m-j]) (with bounds)

            # Compute gradient at this coordinate
            j_range = np.where(active_f > 0)[0]
            # For each active j, the derivative w.r.t. a[m] is:
            # a[j-m] if 0 <= j-m <= n-1, plus a[m-j] if 0 <= m-j <= n-1 (and m != j-m)
            dm = j_range - m  # j - m for each active j
            valid1 = (dm >= 0) & (dm < n)
            contrib1 = np.where(valid1, a[dm[valid1]], 0.0)

            dm2 = m - j_range  # m - j
            valid2 = (dm2 >= 0) & (dm2 < n)
            contrib2 = np.where(valid2, a[dm2[valid2]], 0.0)

            # Special case: when j = 2m (dm = 0), we double count
            # Actually for linear conv, c[j] = sum_k a[k]*a[j-k]
            # dc[j]/da[m] = a[j-m] (if valid) + a[m-j] (if valid and m != j-m)
            # = a[j-m] + a[m-j] (if both valid), but if j-m == m (i.e., j=2m),
            #   then dc[j]/da[m] = 2*a[m]... no, it's a[m]*a[m] so derivative is 2*a[m]
            # Wait: c[j] contains the term a[m]*a[j-m]. If j-m = m, this is a[m]^2,
            #   derivative = 2*a[m]. If j-m != m, the term a[m]*a[j-m] has derivative a[j-m],
            #   AND the term a[j-m]*a[m] (which is the same term since multiplication is
            #   commutative) ... actually in the sum, each pair (k, j-k) appears once.
            #   So c[j] = sum_k a[k]*a[j-k]. The term with k=m is a[m]*a[j-m].
            #   dc[j]/da[m] = a[j-m] (from the k=m term)
            #   PLUS: if j-m is also a valid index in the sum (i.e., as k'=j-m),
            #   then there's a term a[j-m]*a[m], whose derivative w.r.t. a[m] is a[j-m].
            #   But wait, k'=j-m means the term is a[j-m]*a[j-(j-m)] = a[j-m]*a[m].
            #   So we have TWO terms: a[m]*a[j-m] (k=m) and a[j-m]*a[m] (k=j-m).
            #   If m != j-m, these are different terms, both contributing a[j-m] to the derivative.
            #   If m == j-m (i.e., j = 2m), it's the same term a[m]^2, derivative = 2*a[m].

            # Simplified: dc[j]/da[m] = a[j-m]*1_{valid(j-m)} + a[m-j]*1_{valid(m-j)}
            # where valid means in [0, n-1]
            # But if j-m = m (j=2m), then both terms give a[m], so total = 2*a[m].
            # If j-m != m but both valid, total = a[j-m] + a[m-j] = 2*a[j-m] (since a is the same array)
            # Wait no: a[j-m] and a[m-j] are different elements unless j-m = m-j, i.e., j = 2m.

            # Actually I realize my indexing might be off. Let me just compute numerically
            # for small n and verify. For now, let's use the convolution formula:
            # The gradient of (sum_j active[j] * c[j]) w.r.t. a is:
            # 2 * convolve(a, active_restricted) in the appropriate range
            # This is what _hess_matvec computes!

            # Let's use the full gradient approach instead for correctness
            pass

        # Actually, coordinate descent on 16384 dimensions is too slow with the
        # full gradient computation. Let's use a block approach instead.
        # Compute the full gradient once per sweep.
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)

        # Line search along the gradient direction (like gradient descent)
        c_cur = _full_convolve(a, a)
        f_cur = float(c_cur[active].sum())
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 1.0 / (gnorm + 1e-10)
        for _ in range(20):
            a_try = a - step * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_cur:
                a = a_try
                f_cur = f_try
                break
            step *= 0.5

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _lbfgs_lse_continuation(a_init, deadline, min_remaining=30.0):
    """
    L-BFGS with LSE smoothing and aggressive continuation schedule.
    Starts with large beta (very smooth) and rapidly decreases.
    """
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    # Estimate the "natural" beta scale from the gap structure
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 100:
        gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # More aggressive continuation: more stages, faster reduction
    n_stages = 20
    betas = base_beta * np.array([1.0, 0.7, 0.49, 0.34, 0.24, 0.17, 0.12,
                                   0.085, 0.06, 0.042, 0.03, 0.021, 0.015,
                                   0.01, 0.007, 0.005, 0.0035, 0.0025, 0.0017, 0.001])
    betas = betas[:n_stages]

    # Fix z initialization
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for stage_i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Allocate time proportional to remaining stages
        stages_left = n_stages - stage_i
        time_alloc = remaining / stages_left
        est_time_per_iter = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(time_alloc * 0.7 / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100,
                         'iprint': -1}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif stage_i > 5:
                # If not improving, keep old z
                pass
        except Exception as e:
            print(f"  LSE stage {stage_i} error: {e}")

    return current_a, current_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_cos":
        # Near-flat with tiny cosine perturbation
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.01 * np.cos(2.0 * np.pi * t * 7)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_flat":
        # Small random perturbation around uniform
        rng = np.random.default_rng(12345)
        a = 1.0 + 0.01 * rng.standard_normal(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.05):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s
