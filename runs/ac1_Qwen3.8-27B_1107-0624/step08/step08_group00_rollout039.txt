import time
from scipy.optimize import minimize

    import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """Run L-BFGS with adaptive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    current_sc = _score(current_a)

    z = z - z.mean()

    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Parameters
    top_k = 5  # number of peaks to target
    reduction_frac = 0.05  # target reduction of each peak
    max_iters = 200

    for it in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        target = c_max * (1.0 - reduction_frac)

        # Build the linearized constraint matrix
        # conv(a+d, a+d)[k] ≈ conv(a,a)[k] + 2*conv(a,d)[k]
        # We want: 2*conv(a,d)[k] <= target - c[k] for k in peak_indices
        
        m = len(peak_indices)
        # B[k, j] = 2 * a[k-j] for valid j (this is the gradient of conv(a+d,a+d) at d=0)
        B = np.zeros((m, n))
        rhs = np.zeros(m)
        for idx, k in enumerate(peak_indices):
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            j_lo = k - i_hi
            j_hi = k - i_lo
            for j in range(j_lo, j_hi + 1):
                B[idx, j] = 2.0 * a[k - j]
            rhs[idx] = target - c[k]

        # Solve: min ||d||^2 s.t. B@d <= rhs, sum(d) = 0, a+d >= 0
        if not HAS_CVXPY:
            # Fallback: simple gradient step
            # Compute gradient of the "top-k max"
            w = np.zeros_like(c)
            for k in peak_indices:
                w[k] = 1.0 / top_k
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad = 2.0 * h - 2.0 * np.mean(2.0 * h)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            alpha = 0.1 / gnorm
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc:
                a = a_try
            else:
                # Try smaller step
                alpha *= 0.1
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-10:
                    a = a_try / s_try
            reduction_frac = max(0.01, reduction_frac * 0.9)
            continue

        # QP solve
        d_var = cp.Variable(n)
        constraints = [cp.sum(d_var) == 0, B @ d_var <= rhs]
        # Non-negativity: only constrain where a is small
        small_mask = a < 0.01 * a.max()
        if small_mask.any():
            constraints.append(d_var[small_mask] >= -a[small_mask])

        objective = cp.Minimize(cp.sum_squares(d_var))
        prob = cp.Problem(objective, constraints)

        solved = False
        for solver in ['ECOS', 'SCS', 'CLARABEL']:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            try:
                prob.solve()
                solved = True
            except Exception:
                pass

        if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
            reduction_frac = max(0.01, reduction_frac * 0.7)
            continue

        d = d_var.value.copy()
        if d is None or not np.all(np.isfinite(d)):
            continue

        # Line search
        a_old_sc = current_sc
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break

        if not improved:
            reduction_frac = max(0.005, reduction_frac * 0.7)
            if reduction_frac < 0.005:
                break

        # Gradually reduce the target reduction
        if it > 50:
            reduction_frac = max(0.005, reduction_frac * 0.95)

    return best_a, best_sc


def _spectral_shaping(a_init, deadline, min_time_remaining=50.0):
    """
    Spectral shaping: work in the frequency domain to flatten the
    power spectrum of the self-convolution.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    N = _next_pow2(2 * n - 1)
    
    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Compute FFT of a
        A = np.fft.rfft(a, N)
        # Compute B = A^2 (this gives the spectrum of the convolution)
        B = A * A
        # Compute convolution
        c = np.fft.irfft(B, N)[:2 * n - 1]
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find peak positions in time domain
        peak_idx = int(np.argmax(c))
        
        # The peak in time domain at position peak_idx corresponds to
        # a specific pattern in frequency domain.
        # To reduce the peak, we need to modify B such that
        # IFFT(B)[peak_idx] is reduced.
        # IFFT(B)[peak_idx] = (1/N) * sum_f B[f] * exp(2j*pi*f*peak_idx/N)
        # To reduce this, we can reduce |B[f]| for frequencies where
        # exp(2j*pi*f*peak_idx/N) has the same phase as B[f].
        
        # Compute the "peak direction" in frequency domain
        # The gradient of IFFT(B)[peak_idx] with respect to B[f] is
        # (1/N) * exp(2j*pi*f*peak_idx/N)
        
        phase_factor = np.exp(2.0j * np.pi * np.arange(len(B)) * peak_idx / N)
        
        # We want to reduce the real part of B * conj(phase_factor)
        # This tells us which frequency components contribute to the peak
        
        # Simple approach: reduce the magnitude of A at frequencies
        # that most contribute to the peak
        contribution = np.real(B * np.conj(phase_factor))
        
        # Only reduce at the top contributing frequencies
        n_reduce = max(5, len(B) // 10)
        top_f = np.argsort(np.abs(contribution))[::-1][:n_reduce]
        
        # Apply a small reduction
        reduction = 0.01
        A_new = A.copy()
        for f in top_f:
            # Reduce the magnitude of A[f]
            mag = np.abs(A_new[f])
            if mag > 1e-15:
                A_new[f] *= (1.0 - reduction)
        
        # Inverse FFT to get new a
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new < current_sc - 1e-15:
            a = a_new
        else:
            # Try smaller reduction
            reduction *= 0.5
            A_new = A.copy()
            for f in top_f:
                mag = np.abs(A_new[f])
                if mag > 1e-15:
                    A_new[f] *= (1.0 - reduction)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
                if sc_new < current_sc - 1e-15:
                    a = a_new
                else:
                    break

    return best_a, best_sc


def _diverse_restarts(a_init, deadline, n_restarts=5, min_time_remaining=100.0):
    """Try multiple perturbations and optimize each."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(12345)

    time_per_restart = max(5.0, min(120.0, (deadline - min_time_remaining - time.monotonic()) / n_restarts))
    deadline = time.monotonic() + time_per_restart * n_restarts

    for r in range(n_restarts):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Create perturbation
        sigma = 0.1 / (1.0 + 0.3 * r)  # decreasing perturbation
        z = np.log(np.maximum(a_init, 1e-30)) + sigma * rng.standard_normal(n)
        a_pert = _softmax(z)
        
        # Optimize
        a_opt = _run_lbfgs(a_pert, min(deadline, time.monotonic() + time_per_restart), min_time_remaining=0)
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc - 1e-12:
            best_a = a_opt.copy()
            best_sc = sc_opt
            print(f"    Restart {r}: new best {best_sc:.10f} (sigma={sigma:.3f})")

    return best_a, best_sc


def _topk_direct_min(a_init, deadline, min_time_remaining=50.0):
    """
    Directly minimize the sum of top-k convolution values using
    projected subgradient method.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    k = 3  # top-k
    lr = 0.01
    decay = 0.999
    patience = 100
    no_improve = 0

    for it in range(2000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                lr *= decay
                no_improve = 0
                if lr < 1e-8:
                    break

        # Compute subgradient of top-k max
        # The top-k values
        sorted_idx = np.argsort(c)[::-1]
        top_k_idx = sorted_idx[:k]
        threshold = c[top_k_idx[-1]]

        # Subgradient: uniform weight on top-k
        w = np.zeros_like(c)
        w[top_k_idx] = 1.0 / k

        # Gradient of sum_k w[k] * conv(a,a)[k] = 2 * conv(w, a_rev) [in the right range]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Project gradient onto the simplex (sum = 0 for the update direction)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = lr / gnorm
        improved = False
        for _ in range(40):
            a_try = a - step * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            lr *= 0.9

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small entries to create sparser solutions."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, QP peak reduction,
    spectral shaping, diverse restarts, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warm-up ===
    print(f"\n[P1] L-BFGS warm-up (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=700.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc_main:.10f} vs {best_score:.10f}")

    # === Phase 2: Iterative peak reduction (QP) ===
    print(f"\n[P2] QP peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        qp_deadline = min(deadline - 500, time.monotonic() + 400)
        a_qp, sc_qp = _iterative_peak_reduction(best_seq, qp_deadline, min_time_remaining=100.0)
        if sc_qp < best_score:
            best_seq = a_qp.copy()
            best_score = sc_qp
            print(f"  NEW BEST: {best_score:.10f} [qp] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  QP: no improvement (best={best_score:.10f})")

    # === Phase 3: Active-set polish ===
    print(f"\n[P3] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 200)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.005, tol_frac_end=0.00005,
                                           lr_init=0.03)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Spectral shaping ===
    print(f"\n[P4] Spectral shaping (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        sp_deadline = min(deadline - 300, time.monotonic() + 100)
        a_sp, sc_sp = _spectral_shaping(best_seq, sp_deadline, min_time_remaining=50.0)
        if sc_sp < best_score:
            best_seq = a_sp.copy()
            best_score = sc_sp
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Top-k direct minimization ===
    print(f"\n[P5] Top-k direct min (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        tk_deadline = min(deadline - 200, time.monotonic() + 100)
        a_tk, sc_tk = _topk_direct_min(best_seq, tk_deadline, min_time_remaining=50.0)
        if sc_tk < best_score:
            best_seq = a_tk.copy()
            best_score = sc_tk
            print(f"  NEW BEST: {best_score:.10f} [topk] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        dr_deadline = min(deadline - 200, time.monotonic() + 150)
        a_dr, sc_dr = _diverse_restarts(best_seq, dr_deadline, n_restarts=4, min_time_remaining=100.0)
        if sc_dr < best_score:
            best_seq = a_dr.copy()
            best_score = sc_dr
            print(f"  NEW BEST: {best_score:.10f} [restarts] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 8192, 6144]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if abs(n_try - len(best_seq)) < 100:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt

        if time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 55, time.monotonic() + 50),
                                                 tol_frac_start=0.003, tol_frac_end=0.00005,
                                                 lr_init=0.015)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Shrinkage ===
    print(f"\n[P8] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 80)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 60:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=30.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final QP + polish ===
    print(f"\n[P9] Final optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        qp_deadline2 = min(deadline - 40, time.monotonic() + 50)
        a_qp2, sc_qp2 = _iterative_peak_reduction(best_seq, qp_deadline2, min_time_remaining=20.0)
        if sc_qp2 < best_score:
            best_seq = a_qp2.copy()
            best_score = sc_qp2
            print(f"  NEW BEST: {best_score:.10f} [qp_final] t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Ultra-fine final polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           tol_frac_start=0.001, tol_frac_end=0.00001,
                                           lr_init=0.003)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00003
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()

            def obj_fin(z):
                return _smoobj(z, beta_fin, n)

            def grad_fin(z):
                return _smograd(z, beta_fin, n)

            max_iter = min(3000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(obj_fin, z, jac=grad_fin, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 80, 'maxls': 50})

                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines aggressive L-BFGS with log-sum-exp smoothing, diverse random restarts,
QP-based peak reduction, spectral shaping, and ultra-fine active-set polishing.
Key innovations: adaptive beta annealing, top-k subgradient targeting,
asymmetric peak pairing in QP, and multi-scale n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Aggressive L-BFGS with extended beta annealing schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    # Extended beta schedule with finer granularity at the low end
    betas = base_beta * np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06,
                                  0.04, 0.025, 0.015, 0.01, 0.006, 0.004,
                                  0.002, 0.001, 0.0006, 0.0003, 0.00015, 0.00008])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations based on n and remaining time
        scale = max(1.0, n / 4096.0)
        est_time_per_iter = 0.002 * scale
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 8))))

        def obj(z_in):
            return _smoobj(z_in, beta, n)

        def grad(z_in):
            return _smograd(z_in, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00003, lr_init=0.05):
    """Active-set gradient polish with adaptive learning rate and stall detection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(100):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.65)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 12:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.95)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _iterative_qp_peak_reduction(a_init, deadline, min_time_remaining=50.0):
    """
    QP-based peak reduction: identify top peaks, build linearized constraints,
    solve QP to find minimum-norm correction that reduces peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    top_k = 5
    reduction_frac = 0.03
    consecutive_fail = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            consecutive_fail = 0

        if consecutive_fail > 30:
            break

        # Select active peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        target = c_max * (1.0 - reduction_frac)

        if not HAS_CVXPY:
            # Fallback: subgradient step on top-k
            w = np.zeros_like(c)
            w[peak_indices] = 1.0 / top_k
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad = 2.0 * h - 2.0 * np.mean(2.0 * h)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            alpha = 0.05 / gnorm
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
                if sc_try < current_sc:
                    a = a_try
                else:
                    consecutive_fail += 1
            reduction_frac = max(0.005, reduction_frac * 0.9)
            continue

        # Build constraint matrix B where B[k,j] = 2*a[k-j] (gradient of conv at d=0)
        m = len(peak_indices)
        B = np.zeros((m, n))
        rhs = np.zeros(m)
        for idx, k in enumerate(peak_indices):
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            j_lo = k - i_hi
            j_hi = k - i_lo
            for j in range(j_lo, j_hi + 1):
                B[idx, j] = 2.0 * a[k - j]
            rhs[idx] = target - c[k]

        d_var = cp.Variable(n)
        constraints = [cp.sum(d_var) == 0, B @ d_var <= rhs]
        small_mask = a < 0.005 * a.max()
        if small_mask.any():
            constraints.append(d_var[small_mask] >= -a[small_mask])

        objective = cp.Minimize(cp.sum_squares(d_var))
        prob = cp.Problem(objective, constraints)

        solved = False
        for solver in ['ECOS', 'SCS', 'CLARABEL']:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            try:
                prob.solve()
                solved = True
            except Exception:
                pass

        if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
            consecutive_fail += 1
            reduction_frac = max(0.005, reduction_frac * 0.7)
            continue

        d = d_var.value.copy()
        if d is None or not np.all(np.isfinite(d)):
            consecutive_fail += 1
            continue

        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break

        if not improved:
            consecutive_fail += 1
            reduction_frac = max(0.003, reduction_frac * 0.6)
        else:
            # Tighten over time
            reduction_frac = max(0.003, reduction_frac * 0.97)
            top_k = min(10, top_k + 1) if reduction_frac > 0.01 else top_k

    return best_a, best_sc


def _topk_subgradient(a_init, deadline, min_time_remaining=50.0):
    """
    Minimize top-k sum of convolution values using projected subgradient.
    More robust than single-peak targeting.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    k = 5
    lr = 0.02
    patience = 80
    no_improve = 0

    for it in range(3000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                lr *= 0.85
                no_improve = 0
                if lr < 1e-9:
                    break

        # Top-k subgradient
        sorted_idx = np.argsort(c)[::-1]
        top_k_idx = sorted_idx[:k]

        w = np.zeros_like(c)
        w[top_k_idx] = 1.0 / k

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        step = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(50):
            a_try = a - step * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            lr *= 0.95
            k = min(15, k + 1)

    return best_a, best_sc


def _spectral_flatening(a_init, deadline, min_time_remaining=50.0):
    """
    Spectral flattening: work in frequency domain to reduce peak-to-average
    ratio of the self-convolution by adjusting the amplitude spectrum.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    N = _next_pow2(2 * n - 1)
    target_peak_reduction = 0.02

    for it in range(800):
        if time.monotonic() + min_time_remaining > deadline:
            break

        A = np.fft.rfft(a, N)
        B = A * A
        c = np.fft.irfft(B, N)[:2 * n - 1]
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Identify the top 3 peaks
        peak_positions = np.argsort(c)[::-1][:3]

        # For each peak, compute which frequency bins contribute most
        # IFFT(B)[k] = (1/N) * sum_f B[f] * exp(2j*pi*f*k/N)
        # To reduce peak at position k, reduce |B[f]| where B[f]*exp(2j*pi*f*k/N)
        # has the same sign as the current peak contribution

        # Combined reduction direction
        combined_contribution = np.zeros(len(B), dtype=np.complex128)
        for pk in peak_positions:
            phase = np.exp(2.0j * np.pi * np.arange(len(B)) * pk / N)
            combined_contribution += B * phase

        # Reduce amplitude of A at bins with large combined contribution
        contrib_mag = np.abs(combined_contribution)
        n_reduce = max(10, len(B) // 8)
        top_f = np.argsort(contrib_mag)[::-1][:n_reduce]

        # Apply reduction
        reduction = target_peak_reduction
        A_new = A.copy()
        for f in top_f:
            if np.abs(A_new[f]) > 1e-15:
                A_new[f] *= (1.0 - reduction)

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        if sc_new < current_sc - 1e-15:
            a = a_new
            target_peak_reduction = max(0.005, target_peak_reduction * 0.98)
        else:
            # Try smaller reduction
            reduction *= 0.3
            A_new = A.copy()
            for f in top_f[:n_reduce // 2]:
                if np.abs(A_new[f]) > 1e-15:
                    A_new[f] *= (1.0 - reduction)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc_new = 2.0 * n * float(_full_convolve(a_new, a_new).max())
                if sc_new < current_sc - 1e-15:
                    a = a_new
                else:
                    target_peak_reduction *= 0.7
                    if target_peak_reduction < 0.002:
                        break
            else:
                break

    return best_a, best_sc


def _diverse_restarts(a_init, deadline, n_restarts=6, min_time_remaining=100.0):
    """Multiple perturbations followed by L-BFGS optimization."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(777)

    total_time = deadline - min_time_remaining - time.monotonic()
    if total_time < 10:
        return best_a, best_sc

    time_per = total_time / n_restarts
    eff_deadline = time.monotonic() + time_per * n_restarts

    # Use a range of perturbation strengths
    sigmas = [0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]

    for r in range(n_restarts):
        if time.monotonic() + min_time_remaining > eff_deadline:
            break

        sigma = sigmas[r % len(sigmas)]
        z = np.log(np.maximum(a_init, 1e-30)) + sigma * rng.standard_normal(n)
        a_pert = _softmax(z)

        restart_deadline = min(eff_deadline, time.monotonic() + time_per * 1.5)
        a_opt = _run_lbfgs(a_pert, restart_deadline, min_time_remaining=0)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-12:
            best_a = a_opt.copy()
            best_sc = sc_opt
            print(f"    Restart {r} (sigma={sigma:.3f}): {best_sc:.10f}")

    return best_a, best_sc


def _pair_peak_qp(a_init, deadline, min_time_remaining=30.0):
    """
    Target pairs of peaks simultaneously: if peaks at positions k1 and k2
    share contributing elements, find a correction that reduces both.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    if not HAS_CVXPY:
        return best_a, best_sc

    for it in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find the top 2 peaks
        peak_positions = np.argsort(c)[::-1][:2]
        k1, k2 = peak_positions
        if k1 == k2:
            break

        target = c_max * 0.97

        # Build 2x n constraint matrix
        B = np.zeros((2, n))
        rhs = np.zeros(2)
        for idx, k in enumerate([k1, k2]):
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            j_lo = k - i_hi
            j_hi = k - i_lo
            for j in range(j_lo, j_hi + 1):
                B[idx, j] = 2.0 * a[k - j]
            rhs[idx] = target - c[k]

        d_var = cp.Variable(n)
        constraints = [cp.sum(d_var) == 0, B @ d_var <= rhs]
        small_mask = a < 0.005 * a.max()
        if small_mask.any():
            constraints.append(d_var[small_mask] >= -a[small_mask])

        objective = cp.Minimize(cp.sum_squares(d_var))
        prob = cp.Problem(objective, constraints)

        solved = False
        for solver in ['ECOS', 'SCS', 'CLARABEL']:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            try:
                prob.solve()
                solved = True
            except Exception:
                pass

        if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
            continue

        d = d_var.value
        if d is None or not np.all(np.isfinite(d)):
            continue

        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break

        if not improved:
            break

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    """Zero out small entries to create sparser (potentially better) solutions."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: L-BFGS + QP peak reduction + spectral shaping
    + diverse restarts + active-set polish + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Aggressive L-BFGS ===
    print(f"\n[P1] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, time.monotonic() + 500)
    a_main = _run_lbfgs(parent, p1_deadline, min_time_remaining=0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement: {sc_main:.10f}")

    # === Phase 2: QP peak reduction ===
    print(f"\n[P2] QP peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        qp_deadline = min(deadline - 400, time.monotonic() + 300)
        a_qp, sc_qp = _iterative_qp_peak_reduction(best_seq, qp_deadline, min_time_remaining=50.0)
        if sc_qp < best_score:
            best_seq = a_qp.copy()
            best_score = sc_qp
            print(f"  NEW BEST: {best_score:.10f} [qp] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  QP: no improvement")

    # === Phase 3: Diverse restarts ===
    print(f"\n[P3] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        dr_deadline = min(deadline - 300, time.monotonic() + 250)
        a_dr, sc_dr = _diverse_restarts(best_seq, dr_deadline, n_restarts=6, min_time_remaining=80.0)
        if sc_dr < best_score:
            best_seq = a_dr.copy()
            best_score = sc_dr
            print(f"  NEW BEST: {best_score:.10f} [restarts] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Restarts: no improvement")

    # === Phase 4: Spectral flattening ===
    print(f"\n[P4] Spectral flattening (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        sp_deadline = min(deadline - 200, time.monotonic() + 100)
        a_sp, sc_sp = _spectral_flatening(best_seq, sp_deadline, min_time_remaining=50.0)
        if sc_sp < best_score:
            best_seq = a_sp.copy()
            best_score = sc_sp
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Spectral: no improvement")

    # === Phase 5: Top-k subgradient ===
    print(f"\n[P5] Top-k subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        tk_deadline = min(deadline - 150, time.monotonic() + 100)
        a_tk, sc_tk = _topk_subgradient(best_seq, tk_deadline, min_time_remaining=50.0)
        if sc_tk < best_score:
            best_seq = a_tk.copy()
            best_score = sc_tk
            print(f"  NEW BEST: {best_score:.10f} [topk] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Top-k: no improvement")

    # === Phase 6: Pair-peak QP ===
    print(f"\n[P6] Pair-peak QP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        pp_deadline = min(deadline - 100, time.monotonic() + 60)
        a_pp, sc_pp = _pair_peak_qp(best_seq, pp_deadline, min_time_remaining=30.0)
        if sc_pp < best_score:
            best_seq = a_pp.copy()
            best_score = sc_pp
            print(f"  NEW BEST: {best_score:.10f} [pair_qp] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Pair QP: no improvement")

    # === Phase 7: Active-set polish ===
    print(f"\n[P7] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        pol_deadline = min(deadline - 100, time.monotonic() + 80)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.005, tol_frac_end=0.00003,
                                           lr_init=0.03)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Polish: no improvement")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 8192, 24576]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if abs(n_try - len(best_seq)) < 200:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        if time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 35, time.monotonic() + 30),
                                                 tol_frac_start=0.003, tol_frac_end=0.00003,
                                                 lr_init=0.01)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Shrinkage ===
    print(f"\n[P9] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 40)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 30:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=15.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           tol_frac_start=0.0005, tol_frac_end=0.00001,
                                           lr_init=0.003)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [fin_polish] t={time.monotonic()-t_start:.1f}s")

        # Final L-BFGS with tiny beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()

            def obj_fin(z_in):
                return _smoobj(z_in, beta_fin, n)

            def grad_fin(z_in):
                return _smograd(z_in, beta_fin, n)

            max_iter = min(4000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(obj_fin, z, jac=grad_fin, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100, 'maxls': 60})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [fin_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
