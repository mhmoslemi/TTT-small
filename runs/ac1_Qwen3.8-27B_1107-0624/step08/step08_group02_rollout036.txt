    F_W = np.fft.rfft(W, N)
    conv_res = np.fft.irfft(F_v * F_W, N)[:out_len]
    hankel_part = 2.0 * conv_res[n - 1:2 * n - 1]

    # Toeplitz part: (H_t * v)[i] = sum_j W_t[i-j]*v[j] where W_t[k]=W[k] for k>=0
    # This is: conv(W, v)[i + (n-1)] ... actually it's a circular-ish thing
    # (H_t * v)[i] = sum_j W[i-j] * v[j] = (W * v)[i] in the linear conv sense
    # But we need W to be defined for all lags. W[k] for k in 0..2n-2.
    # (conv(W, v))[i] = sum_j W[i-j]*v[j] for valid j
    # We want this for i = 0..n-1, which corresponds to conv(W,v)[0:n]
    # But W has length 2n-1, v has length n, so conv(W,v) has length 3n-2
    # (conv(W,v))[i] for i=0..n-1 gives the Toeplitz part
    F_W2 = np.fft.rfft(W, N)
    F_v2 = np.fft.rfft(v, N)
    conv_res2 = np.fft.irfft(F_W2 * F_v2, N)
    toep_part = conv_res2[:n]

    return hankel_part + toep_part


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Conjugate gradient for Hessian system with FFT-based matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hessian_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_solve_direct(W, n, rhs):
    """Direct solve for small n using dense Hessian."""
    # Build Hessian explicitly
    H = np.zeros((n, n))
    active_indices = np.where(W != 0)[0]
    w_vals = W[active_indices]
    for k_idx in active_indices:
        k = k_idx
        wv = w_vals[k_idx]
        # Hankel part: H[i,j] += wv if i+j == k
        if k < 2 * n - 1:
            i_max = min(n - 1, k)
            for i in range(i_max + 1):
                j = k - i
                if 0 <= j < n:
                    H[i, j] += wv
        # Toeplitz part: H[i,j] += wv if |i-j| == k
        if k < n:
            for i in range(n - k):
                j = i + k
                H[i, j] += wv
                H[j, i] += wv
    # Add regularization for numerical stability
    H += 1e-12 * np.eye(n)
    try:
        x = np.linalg.solve(H, rhs)
        if np.all(np.isfinite(x)):
            return x
    except np.linalg.LinAlgError:
        pass
    return _cg_solve(W, n, rhs, max_iter=1000, tol=1e-15)


def _newton_active_polish(a_init, deadline, max_rounds=200, verbose=False):
    """Newton method with adaptive active set for the KKT conditions."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: start wide, tighten over iterations
        if rnd < 5:
            tol_frac = 0.01
        elif rnd < 15:
            tol_frac = 0.003
        elif rnd < 30:
            tol_frac = 0.001
        elif rnd < 60:
            tol_frac = 0.0003
        else:
            tol_frac = 0.00005

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Gradient of (1/m)*sum_{k in A} conv[k]
        g = _grad_of_weighted_conv(a, W, n)
        g_t = g - np.mean(g)  # project out the sum constraint direction
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            if verbose:
                print(f"    Newton: converged at round {rnd}, gn={gn:.2e}")
            break

        # Solve H * dx = -g_t (with constraint that sum(dx)=0 via projection)
        # Use direct solve for small n, CG for large n
        if n <= 2048:
            Hg = _newton_solve_direct(W, n, g_t)
            H1 = _newton_solve_direct(W, n, np.ones(n))
        else:
            Hg = _cg_solve(W, n, g_t, max_iter=600, tol=1e-14)
            H1 = _cg_solve(W, n, np.ones(n), max_iter=600, tol=1e-14)

        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / (float(np.ones(n) @ H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with strong descent
        alpha = 1.0
        step_done = False
        best_try_sc = c_max
        best_try_a = None
        for ls in range(60):
            a_try = a + alpha * dx
            n_neg = int((a_try < 0).sum())
            if n_neg > 0:
                alpha *= 0.5
                if alpha < 1e-16:
                    break
                continue
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < best_try_sc:
                best_try_sc = c_max_try
                best_try_a = a_try.copy()
            # Strong Wolfe-like condition
            if c_max_try < c_max * (1.0 - 0.1 * alpha) - 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if not step_done and best_try_a is not None and best_try_sc < c_max - 1e-14:
            a = best_try_a
            step_done = True

        if not step_done:
            # Try subgradient steps
            for trial in [0.1, 0.05, 0.02, 0.01, 0.003, 0.001]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            if verbose:
                print(f"    Newton: stalled at round {rnd}, score={best_sc:.10f}")
            break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = _grad_of_weighted_conv(a, w, n)
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (g - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Adaptive log-sum-exp L-BFGS optimization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Initial beta based on gap between top peaks
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.003 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 20))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-7 * c_max_new and sc_improved > 1e-12:
            beta *= 0.1
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.15
        else:
            beta *= 0.3

        beta = min(beta, c_max_new * 0.001)

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.7) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.5 * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.03
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 5 * t ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _freq_perturb(a, sigma, rng):
    """Perturb in frequency domain for structured diversity."""
    n = len(a)
    N = _next_pow2(n)
    F = np.fft.rfft(a, N)
    # Perturb magnitude (keep phase)
    phase = np.angle(F)
    mag = np.abs(F)
    mag = mag * (1.0 + sigma * rng.standard_normal(len(mag)))
    mag = np.maximum(mag, 0)
    F_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, N)[:n]
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=5000):
    """Subgradient method with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    lr_min = 0.0005

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance for active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        g = _grad_of_weighted_conv(a, w, n)
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (g - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr
        z = z - step * grad / max(grad_norm, 1e-30) * grad_norm
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.05)
            else:
                patience += 1
                lr = max(lr * 0.8, lr_min)
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.005

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _refined_lbfgs_newton_cycle(a_init, deadline, min_time_remaining=60.0):
    """Alternating L-BFGS and Newton with increasing time per cycle."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    base_cycle = min(30.0, max(10.0, (deadline - min_time_remaining - time.monotonic()) / 10.0))
    cycle_time = base_cycle

    for cycle in range(12):
        if time.monotonic() + min_time_remaining > deadline:
            break
        t_start_cycle = time.monotonic()
        t_end_cycle = min(t_start_cycle + cycle_time, deadline - min_time_remaining)

        # L-BFGS phase (60% of cycle time)
        if time.monotonic() < t_end_cycle - 3:
            t_lbfgs_end = t_start_cycle + (t_end_cycle - t_start_cycle) * 0.5
            a_lb = _run_adaptive_lbfgs(current_a, min(t_lbfgs_end, deadline - min_time_remaining),
                                        min_time_remaining=min_time_remaining)
            sc_lb = _score(a_lb)
            if sc_lb < current_sc - 1e-16:
                current_a = a_lb
                current_sc = sc_lb

        # Newton phase (40% of cycle time)
        if time.monotonic() + 5 < min(t_end_cycle, deadline - min_time_remaining):
            a_np, sc_np = _newton_active_polish(
                current_a,
                min(t_end_cycle, deadline - min_time_remaining),
                max_rounds=60
            )
            if sc_np < current_sc - 1e-16:
                current_a = a_np
                current_sc = sc_np

        cycle_time *= 1.15
        if cycle_time > 120:
            cycle_time = 120

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick exploration to find best starting point
    print(f"\n[Phase 1] Exploration (t={time.monotonic() - t_start:.1f}s)")
    n_base = n_parent
    profiles = ["tapered", "gaussian", "cos_mod", "chirp", "bimodal", "asym_taper"]
    best_start = parent.copy()
    best_start_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_base, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_start_sc:
            best_start_sc = sc_p
            best_start = a_p
            print(f"  Profile '{prof_kind}': {sc_p:.10f}")

    # Perturbed parent
    for sigma in [0.03, 0.08, 0.15]:
        if time.monotonic() > deadline - 500:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_start_sc:
            best_start_sc = sc_pert
            best_start = a_pert
            print(f"  Perturbed (sigma={sigma}): {sc_pert:.10f}")

    # Frequency perturbation
    a_fp = _freq_perturb(parent, 0.05, rng)
    sc_fp = _score(a_fp)
    if sc_fp < best_start_sc:
        best_start_sc = sc_fp
        best_start = a_fp
        print(f"  Freq perturb: {sc_fp:.10f}")

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 400:
        a_quick = _run_adaptive_lbfgs(best_start, min(deadline - 400, time.monotonic() + 90),
                                       min_time_remaining=200.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_explore")
        if sc_quick < best_start_sc:
            best_start = a_quick
            best_start_sc = sc_quick

    # Phase 2: Main optimization cycle
    print(f"\n[Phase 2] Main optimization (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_start
    a_main = _refined_lbfgs_newton_cycle(start_a, deadline, min_time_remaining=400.0)
    update_best(a_main, _score(a_main), "main_cycle")

    # Phase 3: Focused Newton polish
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 250, time.monotonic() + 200),
                                             max_rounds=150, verbose=True)
        update_best(a_np, sc_np, "newton_1")

    # Phase 4: Subgradient diversification + polish
    print(f"\n[Phase 4] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 80))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002 and time.monotonic() < deadline - 80:
            a_np2, sc_np2 = _newton_active_polish(a_sg, min(deadline - 80, time.monotonic() + 40),
                                                   max_rounds=50)
            update_best(a_np2, sc_np2, "subgrad_newton")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    base_n = len(best_seq) if best_seq is not None else n_base
    n_candidates = [base_n // 2, base_n * 2]
    n_candidates = [n for n in n_candidates if 1024 <= n <= 65536]
    # Avoid duplicate of current
    n_candidates = [n for n in n_candidates if n != base_n]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        a_init = _resample(best_seq, n_try)
        a_opt = _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 30:
            a_np3, sc_np3 = _newton_active_polish(a_opt, min(deadline - 30, time.monotonic() + 20),
                                                   max_rounds=30)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(3):
            if time.monotonic() > deadline - 15:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _freq_perturb(best_seq, 0.03, rng)
            else:
                p_start = _make_profile(len(best_seq), ["gaussian", "cos_mod", "chirp"][i % 3])

            t_rem = deadline - 15 - time.monotonic()
            if t_rem < 10:
                break
            a_r = _run_adaptive_lbfgs(p_start, min(time.monotonic() + t_rem * 0.5, deadline - 10),
                                       min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 8:
                a_rp, sc_rp = _newton_active_polish(a_r, deadline - 5, max_rounds=30)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 7: Final ultra-fine Newton
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS at tiny beta
        if best_seq is not None and time.monotonic() < deadline - 1.5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-11
            remaining = deadline - 1.5 - time.monotonic()
            if remaining > 1.5:
                max_iter = min(3000, max(30, int(remaining / 0.005)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Final Newton after last L-BFGS
        if best_seq is not None and time.monotonic() < deadline - 0.5:
            a_veryfin, sc_veryfin = _newton_active_polish(best_seq, deadline - 0.1, max_rounds=80)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev_sc = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {ev_sc:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
