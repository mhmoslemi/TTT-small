def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

    """Compute H @ v where H is the Hankel matrix defined by W and the autocorrelation structure."""
    V = v[::-1]

    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]

    p = r.copy()

        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):

    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):

        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

                    g_zero = g[~support]
                    # For our problem (minimizing), at a_i=0 we need subgrad >= 0
                    # But our constraint is a_i >= 0 and we're minimizing, so we need g_i <= 0 for inactive
                    # Actually in the normalized simplex, it's more subtle. Use magnitude.
                    residual += 0.1 * float(np.max(np.abs(g_zero)))
            else:
                residual = float(np.max(np.abs(g_proj)))
            if residual < best_kkt:
                best_kkt = residual
                best_active = active.copy()

        if best_active is None:
            best_active = c >= (c_max - 0.001 * c_max)
            if best_active.sum() == 0:
                best_active[int(np.argmax(c))] = True

        W = np.zeros(2 * n - 1)
        W[best_active] = 1.0 / best_active.sum()

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-12 * n:
            break

        # Newton step with CG solve
        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Armijo line search with curvature
        alpha = 1.0
        step_done = False
        c0 = c_max
        for ls in range(60):
            a_try = a + alpha * dx
            n_neg = int(np.sum(a_try < 0))
            if n_neg > 0:
                # Clip and note boundary
                a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo condition
            if c_max_try < c0 - 1e-4 * alpha * gn * np.max(np.abs(dx)) + 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Fallback: subgradient step
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 10:
            break

    return best_a, best_sc


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    """Aggressive L-BFGS with faster beta reduction."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.0005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-11
    prev_sc = current_sc
    stall_count = 0

    iter_count = 0
    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.002 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter_time * 30))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
        except Exception:
            stall_count += 1

        iter_count += 1

        # Adaptive beta reduction - more aggressive
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.15
        elif sc_improved > 1e-12:
            beta *= 0.25
        else:
            beta *= 0.4
            # If stalled, try a random restart in z-space
            if stall_count > 3:
                z = z + 0.01 * np.random.standard_normal(n)
                z = z - z.mean()
                stall_count = 0

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _subgrad_average(a_init, deadline, max_steps=20000):
    """Subgradient method with iterate averaging for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Polyak-like step size
    # Lower bound on optimal value (empirically ~ 0.75/n for normalized sequence)
    lb = 0.5 / n  # conservative lower bound for max(c) when sum(a)=1
    patience = 0
    lr = 0.01

    # Track average iterate
    avg_z = z.copy()
    count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Step size: use min(Polyak, fixed)
        gap = c_max - lb
        polyak_lr = max(gap, 0) / (grad_norm * grad_norm + 1e-30)
        step = min(lr, polyak_lr * 10)
        step = max(step, 1e-6)

        z = z - step * grad
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

        # Average iterate (running average)
        avg_z = (count * avg_z + z) / (count + 1)
        count += 1

        # Check progress every 100 steps
        if i % 100 == 99:
            a_check = _softmax(avg_z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-13:
                best_sc = sc_check
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.03)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    avg_z = z.copy()
                    count = 0
                    patience
                    patience = 0
                    lr = 0.008

    a_final = _softmax(avg_z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coord_descent_polish(a_init, deadline, max_sweeps=30):
    """Coordinate descent exploiting 1D subproblem structure."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    S = 1.0  # normalized

    rng = np.random.default_rng(12345)
    indices = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Randomize order for diversity
        order = rng.permutation(indices)
        improved = 0
        for i in order:
            if time.monotonic() > deadline:
                break
            a_i = a[i]
            if a_i < 1e-10:
                # Try small positive value
                delta = 0.01 * np.max(a)
            else:
                # Compute the 1D objective
                # c_new[k] = c[k] + 2*delta*a_ref[k-i] + delta^2*(k==2*(n-1-i))
                a_shifted = np.zeros(2 * n - 1)
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_shifted[lo:hi + 1] = a[hi - i:hi - i + (hi - lo + 1) - n + 1]
                # More carefully: a_ref[k-i] is non-zero when 0 <= k-i < n
                # i.e., i <= k <= i+n-1, and the value is a[k-i]
                a_shifted = np.zeros(2 * n - 1)
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                a_shifted[k_start:k_end + 1] = a[k_start - i:k_end - i + 1]

                corr_pos = 2 * (n - 1 - i)
                if 0 <= corr_pos < 2 * n - 1:
                    correction = np.zeros(2 * n - 1)
                    correction[corr_pos] = 1.0
                else:
                    correction = np.zeros(2 * n - 1)

                def obj_1d(delta):
                    c_new = c + 2.0 * delta * a_shifted + delta * delta * correction
                    s_new = S + delta
                    if s_new < 1e-10:
                        return 1e10
                    return c_new.max() / (s_new * s_new)

                # Bounded minimization
                lo_bound = -a_i * 0.99
                hi_bound = 2.0 * np.max(a)
                try:
                    res = minimize_scalar(obj_1d, bounds=(lo_bound, hi_bound),
                                          method='bounded',
                                          options={'xatol': 1e-12, 'maxiter': 30})
                    if res.fun < obj_1d(0.0) - 1e-15:
                        delta = res.x
                        a[i] = a_i + delta
                        S += delta
                        c = c + 2.0 * delta * a_shifted + delta * delta * correction
                        improved += 1
                except Exception:
                    pass

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        S = 1.0
        c = _full_convolve(a, a)

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        # Converged if no improvement in sweep
        if improved == 0:
            break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_use = n_parent
    start_a = parent.copy()

    # Phase 1: Quick L-BFGS warmup (~300s)
    print(f"\n[Phase 1] L-BFGS warmup (t={time.monotonic() - t_start:.1f}s)")
    t_phase1_end = min(deadline - 900, t_start + 300)
    a_lb1 = _run_lbfgs_aggressive(start_a, t_phase1_end, min_time_remaining=50.0)
    update_best(a_lb1, _score(a_lb1), "lbfgs_warmup")
    if best_seq is not None:
        start_a = best_seq.copy()

    # Phase 2: Heavy Newton polish (~600s)
    print(f"\n[Phase 2] Heavy Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        t_phase2_end = min(deadline - 400, t_start + 900)
        a_np, sc_np = _newton_polish(best_seq, t_phase2_end, max_rounds=300)
        update_best(a_np, sc_np, "newton_heavy")

    # Phase 3: Subgradient with averaging (~200s)
    print(f"\n[Phase 3] Subgradient averaging (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        t_phase3_end = min(deadline - 200, t_start + 1100)
        a_sg, sc_sg = _subgrad_average(best_seq, t_phase3_end, max_steps=50000)
        update_best(a_sg, sc_sg, "subgrad_avg")

    # Phase 4: Newton after subgrad
    print(f"\n[Phase 4] Newton after subgrad (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        t_phase4_end = min(deadline - 150, t_start + 1250)
        a_np2, sc_np2 = _newton_polish(best_seq, t_phase4_end, max_rounds=200)
        update_best(a_np2, sc_np2, "newton_2")

    # Phase 5: Coordinate descent polish (~150s)
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        t_phase5_end = min(deadline - 80, t_start + 1400)
        a_cd, sc_cd = _coord_descent_polish(best_seq, t_phase5_end, max_sweeps=20)
        update_best(a_cd, sc_cd, "coord_desc")

    # Phase 6: Final L-BFGS + Newton
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        remaining = deadline - 30 - time.monotonic()
        if remaining > 20:
            a_fin = _run_lbfgs_aggressive(best_seq, deadline - 30, min_time_remaining=10.0)
            update_best(a_fin, _score(a_fin), "final_lbfgs")

    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin2, sc_fin2 = _newton_polish(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fin2, sc_fin2, "final_newton")

    # Phase 7: Try different n if time permits
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for n_try in [2 * n_use, n_use // 2]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 1024 or n_try > 65536:
                continue
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs_aggressive(a_res, deadline - 20, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
