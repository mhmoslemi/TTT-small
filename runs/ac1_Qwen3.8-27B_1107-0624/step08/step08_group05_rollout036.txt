from scipy.optimize import minimize

_best_seq = None
_best_score = np.inf


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _autocorr_flatness(c):
    """Measure how flat the autocorrelation is. Returns ratio max/median of positive values."""
    c_pos = c[c > 1e-15]
    if len(c_pos) == 0:
        return 1.0
    med = np.median(c_pos)
    if med < 1e-15:
        return 1.0
    return float(c.max() / med)


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.002):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Weight active set by proximity to max (soft active set)
        active_weights = np.zeros_like(c)
        active_weights[active] = 1.0
        # Additional weighting: closer to max gets more weight
        range_c = c_max - (c_max - tol)
        if range_c > 1e-30:
            soft_w = (c[active] - (c_max - tol)) / range_c
            active_weights[active] = 0.5 + 0.5 * soft_w
        active_norm = active_weights.sum()
        if active_norm > 0:
            active_weights /= active_norm
        else:
            active_weights = np.zeros_like(c)
            active_weights[active] = 1.0 / active.sum()

        active_f = active.astype(np.float64)
        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8

        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(_hv, -grad_t, tol=1e-10, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(np.dot(active_weights, c))
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(active_weights, c_try))
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _adaptive_lse_schedule(a, n_stages=16):
    """Generate an adaptive LSE beta schedule based on current autocorrelation flatness."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    n_c = len(c_sorted)
    # Look at the gap between top and top-100
    top_k = min(100, n_c)
    gap_top = c_sorted[0] - c_sorted[top_k - 1]
    # Look at the overall spread
    gap_all = c_sorted[0] - c_sorted[n_c // 4]
    base = max(gap_top * 0.5, gap_all * 0.1, c_max * 0.005)
    # Generate schedule: start wide, end narrow
    betas = base * np.array([1.0, 0.7, 0.45, 0.28, 0.17, 0.10,
                              0.06, 0.035, 0.02, 0.012, 0.007,
                              0.004, 0.002, 0.001, 0.0005, 0.0002])
    return betas[:n_stages]


def _lbfgs_lse_adaptive(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive LSE smoothing schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    betas = _adaptive_lse_schedule(current_a, n_stages=16)

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                # Regenerate schedule if we made significant progress
                if betas.index(beta) % 3 == 2:
                    betas = _adaptive_lse_schedule(current_a, n_stages=16)
                    # Restart from current position in schedule
        except Exception:
            pass

    return current_a, current_sc


def _power_sum_continuation(a_init, deadline, min_remaining=30.0):
    """Power sum continuation with adaptive p values."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive p values based on flatness
    p_values = [4, 8, 16, 32, 64, 128, 256, 512, 1024]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(150, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 120}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _riemannian_cg(a_init, deadline, n_steps=3000, lr_init=0.05, tol_frac=0.001):
    """Riemannian conjugate gradient with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    # CG state
    prev_grad = None
    d = None
    prev_bnorm = None

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-14:
            break

        if prev_grad is not None and d is not None:
            # Polak-Ribiere beta
            num = float(np.dot(grad_z, grad_z - prev_grad))
            den = float(np.dot(prev_grad, prev_grad))
            beta = max(0.0, num / den) if den > 1e-30 else 0.0
            d = -grad_z + beta * d
        else:
            d = -grad_z

        # Adaptive step size
        step = lr / (1.0 + 0.01 * i)
        # Backtracking
        a_curr = _softmax(z)
        sc_curr = _score(a_curr)
        for bt in range(20):
            z_try = z + step * d
            z_try = np.clip(z_try, -100, 100)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < sc_curr - 1e-12 * step * gnorm or bt == 19:
                break
            step *= 0.5

        z = z + step * d
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        prev_grad = grad_z
        prev_bnorm = gnorm

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.3)
            else:
                lr *= 0.9
                if lr < lr_init * 0.01:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    d = None
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01, tol_frac=0.001):
    """Subgradient polishing with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad_a = _active_gradient(a, c, n, tol_frac=tol_frac)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init * 2)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_local_random(a, rng, frac=0.1):
    """Randomly replace a fraction of entries with new random values."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30)).copy()
    n_rand = max(1, int(frac * n))
    idx = rng.choice(n, size=n_rand, replace=False)
    z[idx] = z.mean() + rng.standard_normal(n_rand) * 0.5
    z = z - z.mean()
    return _softmax(z)


def _perturb_symmetric_mix(a, rng, alpha=0.5):
    """Mix with the reversed sequence to explore symmetric structures."""
    n = len(a)
    a_rev = a[::-1]
    a_mix = (1 - alpha) * a + alpha * a_rev
    a_mix = np.maximum(a_mix, 1e-10)
    noise = sigma = 0.01
    a_mix = a_mix + noise * rng.standard_normal(n)
    a_mix = np.maximum(a_mix, 0)
    s = a_mix.sum()
    if s < 1e-10:
        return a.copy()
    return a_mix / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "anti_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s


def _fista_lse(a_init, deadline, n_iter=2000, beta=None):
    """FISTA (accelerated projected gradient) for LSE objective."""
    n = len(a_init)
    if beta is None:
        c = _full_convolve(a_init, a_init)
        beta = float(c.max()) * 0.05

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    t_k = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Estimate Lipschitz constant
    L = 50.0 * n / beta  # rough estimate

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        grad = _lse_gradient(z_prev, beta, n)
        step = 1.0 / L
        z_new = z_prev - step * grad
        z_new = np.clip(z_new, -100, 100)
        z_new = z_new - z_new.mean()

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = z_new + ((t_k - 1.0) / t_new) * (z_new - z_prev)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        z_prev = z_new
        t_k = t_new

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            # Adaptive L
            if sc > best_sc * 1.01:
                L *= 1.5
            elif sc < best_sc * 0.99:
                L *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer: adaptive LSE, Riemannian CG, FISTA,
    improved Newton, diverse restarts, multi-n exploration.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (40s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 40)
    n_expl = max(1024, n_parent // 16) if n_parent > 2048 else 1024
    for kind in ["uniform", "tapered", "cos2", "gauss", "bimodal", "ramp", "anti_taper"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # === Phase 2: Main adaptive LSE at parent size (350s) ===
    print(f"\n[Phase 2] Adaptive LSE (main)")
    if time.monotonic() < deadline - 350:
        a_opt, sc_opt = _lbfgs_lse_adaptive(_best_seq.copy(), deadline, min_remaining=300)
        upd(a_opt, sc_opt, "lse_adaptive_main")

    # === Phase 3: FISTA supplement (100s) ===
    print(f"\n[Phase 3] FISTA")
    if time.monotonic() < deadline - 100:
        c_cur = _full_convolve(_best_seq, _best_seq)
        beta_f = float(c_cur.max()) * 0.1
        a_f, sc_f = _fista_lse(_best_seq, min(deadline - 90, time.monotonic() + 100), n_iter=3000, beta=beta_f)
        upd(a_f, sc_f, "fista")

    # === Phase 4: Riemannian CG (150s) ===
    print(f"\n[Phase 4] Riemannian CG")
    if time.monotonic() < deadline - 150:
        cg_dead = min(deadline - 140, time.monotonic() + 150)
        a_cg, sc_cg = _riemannian_cg(_best_seq, cg_dead, n_steps=5000, lr_init=0.03, tol_frac=0.001)
        upd(a_cg, sc_cg, "riemannian_cg")

    # === Phase 5: Newton active-set (200s) ===
    print(f"\n[Phase 5] Newton active-set")
    if time.monotonic() < deadline - 200:
        nw_dead = min(deadline - 180, time.monotonic() + 200)
        a_nw, sc_nw = _newton_active_set(_best_seq, nw_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 6: Power-sum continuation (150s) ===
    print(f"\n[Phase 6] Power-sum continuation")
    if time.monotonic() < deadline - 150:
        a_ps, sc_ps = _power_sum_continuation(_best_seq, deadline, min_remaining=130)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 7: Newton + subgradient alternating (150s) ===
    print(f"\n[Phase 7] Newton + SG alternating")
    if time.monotonic() < deadline - 150:
        a_cur = _best_seq.copy()
        for rnd in range(4):
            if time.monotonic() > deadline - 140:
                break
            nw_d = min(deadline - 140, time.monotonic() + 30)
            a_n2, sc_n2 = _newton_active_set(a_cur, nw_d, max_iter=80, tol_frac=0.002)
            if sc_n2 < _score(a_cur):
                a_cur = a_n2
            sg_d = min(deadline - 140, time.monotonic() + 30)
            a_s2, sc_s2 = _subgrad_polish(a_cur, sg_d, n_steps=3000, lr_init=0.005, tol_frac=0.001)
            if sc_s2 < _score(a_cur):
                a_cur = a_s2
            upd(a_cur, _score(a_cur), f"alt_{rnd}")

    # === Phase 8: Multi-n exploration (250s) ===
    print(f"\n[Phase 8] Multi-n exploration")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates and 1024 <= n_parent <= 65536:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    time_per_n = max(30.0, 250.0 / len(n_candidates))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(_best_seq) == n_try:
            a_init_n = _best_seq.copy()
        else:
            a_init_n = _resize_sequence(_best_seq, n_try)

        n_dead = min(deadline - 50, time.monotonic() + time_per_n)

        # LSE at this n
        a_opt_n, sc_opt_n = _lbfgs_lse_adaptive(a_init_n, n_dead, min_remaining=10)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        # Quick Newton polish
        if sc_opt_n <= _best_score * 1.01 and time.monotonic() < deadline - 30:
            nw_d2 = min(deadline - 30, time.monotonic() + min(30, time_per_n * 0.3))
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_d2, max_iter=60, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 9: Diverse restarts with simulated annealing (200s) ===
    print(f"\n[Phase 9] Diverse restarts (simulated annealing)")
    if time.monotonic() < deadline - 60:
        n_restarts = 8
        current_temp = 0.08
        for i in range(n_restarts):
            if time.monotonic() > deadline - 50:
                break
            # Choose perturbation based on temperature
            r = rng.random()
            if r < 0.2:
                p_start = _perturb_log(_best_seq, current_temp, rng)
            elif r < 0.4:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=current_temp * 4)
            elif r < 0.6:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=3)
            elif r < 0.75:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=current_temp)
            elif r < 0.85:
                p_start = _perturb_local_random(_best_seq, rng, frac=0.05)
            else:
                p_start = _perturb_symmetric_mix(_best_seq, rng, alpha=rng.uniform(0.1, 0.5))

            # Short LSE optimization after perturbation
            r_dead = min(deadline - 40, time.monotonic() + 25)
            a_r, sc_r = _lbfgs_lse_adaptive(p_start, r_dead, min_remaining=5)
            upd(a_r, sc_r, f"restart_sa_{i}_T={current_temp:.3f}")

            # Newton polish if close
            if sc_r < _best_score * 1.005 and time.monotonic() < deadline - 20:
                nw_d3 = min(deadline - 20, time.monotonic() + 15)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_d3, max_iter=40, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_sa_{i}_newton")

            # Cool down
            current_temp *= 0.75

    # === Phase 10: Final aggressive polish (60s) ===
    print(f"\n[Phase 10] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()

        # Ultra-low beta LSE
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            beta_final = mb * 0.0001
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-15, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=80, tol_frac=0.0005)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=4000, lr_init=0.003, tol_frac=0.0003)
            upd(a_fin3, sc_fin3, "final_sg_tight")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform").tolist()
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
