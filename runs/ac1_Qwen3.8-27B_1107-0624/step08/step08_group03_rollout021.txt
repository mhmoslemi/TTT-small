def _cg_solve_matvec(matvec_fn, rhs, max_iter=500, tol=1e-10):
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rs0 = rs
    for i in range(max_iter):
        if np.sqrt(rs) < tol * np.sqrt(rs0):
            break
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < tol * np.sqrt(rs0):
            r = r_new
            break
        beta = rs_new / rs
        p = r_new + beta * p
        r = r_new
        rs = rs_new
    return x
