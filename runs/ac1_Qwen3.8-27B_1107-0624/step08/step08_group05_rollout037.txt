```python
"""
Multi-strategy optimizer for minimizing step function convolution peak.
Uses spectral initialization, adaptive LSE annealing, LP-inspired gradient,
Newton active-set, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

_best_seq = None
_best_score = np.inf

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
```
