import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _newton_active_polish(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    tol_fracs = [0.000001, 0.000005, 0.00002, 0.0001, 0.0005, 0.002, 0.01]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        for tol_frac in tol_fracs:
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active >= 1:
                break
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        step_done = False
        alpha = 1.0
        best_alpha = 0
        best_cmax = c_max

        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < best_cmax:
                best_cmax = c_max_try
                best_alpha = alpha
            if c_max_try < c_max * (1.0 - 0.003 * alpha) - 1e-15:
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-16:
                break

        if step_done:
            a_try = a + best_alpha * dx
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            a = a_try
        elif best_alpha > 0:
            a_try = a + best_alpha * dx
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            a = a_try
            step_done = True

        if not step_done:
            for trial in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not step_done and no_improve > 8:
            break

    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.0005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter = 0.003 * scale
        max_iter = min(10000, max(50, int(remaining / (est_iter * 25))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 5e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-7 * c_max_new:
            beta *= 0.15
        elif new_gap < 1e-5 * c_max_new:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _subgrad_polish(a_init, deadline, max_steps=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.01
    n_steps_since_improve = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.0003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                n_steps_since_improve = 0
                lr = min(lr * 1.05, 0.05)
            else:
                patience += 1
                n_steps_since_improve += 1
                if patience > 4:
                    lr *= 0.7
                if n_steps_since_improve > 100:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    n_steps_since_improve = 0
                    lr = 0.008

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.4) ** 1.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay2":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.65) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2.5 + 0.01
        return a / a.sum()
    elif kind == "left_heavy2":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 2.5 + 0.01
        return a / a.sum()
    elif kind == "peak_left":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + 0.005
        return a / a.sum()
    elif kind == "peak_right":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.7) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _greedy_perturb(a_init, deadline, rng, max_steps=10000):
    """Try small random perturbations and keep improvements."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    step_size = 0.001

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Perturb a random subset
        n_perturb = max(1, n // 50)
        idx = rng.integers(0, n, size=n_perturb)
        a_try = current_a.copy()
        a_try[idx] *= np.exp(0.01 * rng.standard_normal(n_perturb))
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s

        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-15:
            current_a = a_try
            current_sc = sc_try
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = current_a.copy()
        else:
            # Occasionally reduce step size
            if i % 100 == 99:
                step_size *= 0.95

    return best_a, best_sc


def _multi_scale_optimize(a_init, n_target, deadline, min_time_remaining=60.0):
    """Optimize at increasing resolutions."""
    scales = []
    n_cur = n_target
    while n_cur > 512 and len(scales) < 4:
        scales.append(n_cur)
        n_cur = max(512, n_cur // 2)
    scales.reverse()
    scales.append(n_target)

    current_a = a_init
    for target_n in scales:
        if time.monotonic() + min_time_remaining > deadline:
            break
        if len(current_a) != target_n:
            current_a = _resample(current_a, target_n)

        t_phase = min(deadline - min_time_remaining, time.monotonic() + max(10, (deadline - min_time_remaining) / (len(scales) + 1)))

        a_opt = _run_adaptive_lbfgs(current_a, t_phase, min_time_remaining=min_time_remaining)
        sc_opt = _score(a_opt)

        if time.monotonic() < deadline - min_time_remaining - 5:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - min_time_remaining - 5, time.monotonic() + 20), max_rounds=30)
            if sc_np < sc_opt:
                a_opt = a_np
                sc_opt = sc_np

        current_a = a_opt
        print(f"    Multi-scale n={target_n}: score={sc_opt:.10f}")

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "asym_taper")
        print(f"No parent, using asym_taper n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_target = n_parent if n_parent > 0 else 16384

    # Phase 1: Quick profile exploration
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["asym_taper", "asym_taper2", "left_heavy", "left_heavy2",
                "right_heavy", "exp_decay", "exp_decay2", "peak_left",
                "peak_right", "bimodal"]
    best_profile_a = parent.copy()
    best_profile_sc = _score(parent)

    for prof_kind in profiles:
        if time.monotonic() > deadline - 800:
            break
        a_p = _make_profile(n_target, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile_a = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f} (new best profile)")

    # Perturb parent
    for sigma in [0.03, 0.08, 0.15]:
        if time.monotonic() > deadline - 700:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_profile_sc:
            best_profile_sc = sc_pert
            best_profile_a = a_pert

    # Quick optimization on best profile
    if time.monotonic() < deadline - 700:
        a_quick = _run_adaptive_lbfgs(best_profile_a, min(deadline - 700, time.monotonic() + 150),
                                       min_time_remaining=500.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_profile")
        if sc_quick < best_profile_sc:
            best_profile_a = a_quick
            best_profile_sc = sc_quick

        # Quick Newton
        if time.monotonic() < deadline - 650:
            a_qn, sc_qn = _newton_active_polish(a_quick, time.monotonic() + 30, max_rounds=50)
            update_best(a_qn, sc_qn, "quick_newton")

    # Phase 2: Multi-scale optimization from best candidate
    print(f"\n[Phase 2] Multi-scale optimization (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_profile_a
    a_ms = _multi_scale_optimize(start_a, n_target, deadline, min_time_remaining=400.0)
    update_best(a_ms, _score(a_ms), "multi_scale")

    # Phase 3: Main L-BFGS with aggressive beta reduction
    print(f"\n[Phase 3] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    current_best = best_seq if best_seq is not None else a_ms
    a_main = _run_adaptive_lbfgs(current_best, deadline, min_time_remaining=350.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 4: Newton polish (multiple rounds with tightening tolerance)
    print(f"\n[Phase 4] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_np1, sc_np1 = _newton_active_polish(best_seq, min(deadline - 250, time.monotonic() + 100),
                                               max_rounds=200)
        update_best(a_np1, sc_np1, "newton_1")

    # Phase 5: Subgradient for diversification
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 180, time.monotonic() + 60))
        update_best(a_sg, sc_sg, "subgrad")
        if sc_sg < best_score + 0.002:
            a_np2, sc_np2 = _newton_active_polish(a_sg, min(deadline - 150, time.monotonic() + 40),
                                                   max_rounds=100)
            update_best(a_np2, sc_np2, "subgrad_newton")

    # Phase 6: Try different n values
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_target * 2, n_target // 2, n_target + 1024]
    n_candidates = [n for n in n_candidates if 1024 <= n <= 65536]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_target:
            continue
        a_init = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt = _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 50:
            a_np3, sc_np3 = _newton_active_polish(a_opt, min(deadline - 40, time.monotonic() + 30),
                                                   max_rounds=80)
            update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_kinds = ["exp_decay2", "peak_left", "asym_taper2"]
        for i, rk in enumerate(restart_kinds):
            if time.monotonic() > deadline - 30:
                break
            if i < 2:
                p_start = _perturb(best_seq, 0.03 + 0.03 * i, rng)
            else:
                p_start = _make_profile(len(best_seq), rk)

            if time.monotonic() > deadline - 20:
                break
            a_r = _run_adaptive_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_active_polish(a_r, deadline - 5, max_rounds=50)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 8: Greedy perturbation
    print(f"\n[Phase 8] Greedy perturbation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_gp, sc_gp = _greedy_perturb(best_seq, min(deadline - 15, time.monotonic() + 30), rng,
                                       max_steps=5000)
        update_best(a_gp, sc_gp, "greedy")

    # Phase 9: Final intensive Newton polish
    print(f"\n[Phase 9] Final Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=300)
        update_best(a_fin, sc_fin, "final_newton")

    # Phase 10: Final L-BFGS at ultra-small beta + Newton
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 5e-11
        remaining = deadline - 3 - time.monotonic()
        if remaining > 2:
            max_iter = min(3000, max(30, int(remaining / 0.02)))
            try:
                obj_fn = _SmoObjAdapter(beta_fin, n)
                grad_fn = _SmoGradAdapter(beta_fin, n)
                result = minimize(
                    obj_fn, z, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-17,
                             'maxcor': 100, 'maxls': 300}
                )
                a_flb = _softmax(result.x)
                sc_flb = _score(a_flb)
                update_best(a_flb, sc_flb, "final_ultra_lbfgs")
            except Exception:
                pass

        if best_seq is not None and time.monotonic() < deadline - 1:
            a_veryfin, sc_veryfin = _newton_active_polish(best_seq, deadline - 0.5, max_rounds=100)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "asym_taper")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
