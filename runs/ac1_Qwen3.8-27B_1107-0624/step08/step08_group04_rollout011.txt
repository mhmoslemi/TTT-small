        z_new = r_new / diag
        beta_cg = float(r_new @ z_new) / rz
        p_new = z_new + beta_cg * p
        x = x_new
        r = r_new
        z = z_new
        p = p_new
        rz = float(r @ z)
        if rz < tol * tol * rz0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, aggressive=True):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.001)
    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001, 0.0000005, 0.0000002, 0.0000001,
                           0.00000005, 0.00000002])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001])
    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_iter_time * 15))))
        obj_fn = partial(_smoobj, beta=beta, n=n)
        grad_fn = partial(_smograd, beta=beta, n=n)
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _active_set_qp(a_init, W, n, max_outer=60, cg_max_iter=800):
    """Solve min 0.5*a^T H a s.t. sum(a)=1, a>=0 using active set."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n))
    for outer in range(max_outer):
        support = a > 1e-12
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            a = np.ones(n) / n
            continue
        if n_sup >= n:
            y = _pcg_solve(W, n, np.ones(n), max_iter=cg_max_iter)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            grad = _hankel_matvec(a, W, n)
            lam = float(np.mean(grad[support]))
            step = 0.5 / (float(np.max(np.abs(grad))) + 1e-15)
            step = min(step, 0.1)
            a_new = a - step * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        val_new = float(a_new @ _hankel_matvec(a_new, W, n))
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-12
        if support.any():
            lam = float(np.mean(grad[support]))
            resid = float(np.max(np.abs(grad[support] - lam)))
            off_s = ~support
            if off_s.any():
                resid = max(resid, float(np.max(lam - grad[off_s])))
            else:
                resid = max(resid, 0.0)
        else:
            resid = 1.0
        if resid < 1e-10 * (float(np.max(np.abs(grad))) + 1e-15):
            break
    return best_a


def _newton_active_polish(a_init, deadline, max_rounds=120, tol_schedule=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    if tol_schedule is None:
        tol_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                        0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                        0.000001, 0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 4
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        a_qp = _active_set_qp(a, W, n, max_outer=60, cg_max_iter=600)
        sc_qp = _score(a_qp)
        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-15:
                Hg = _pcg_solve(W, n, g, max_iter=400)
                H1 = _pcg_solve(W, n, np.ones(n), max_iter=400)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t
                alpha = 1.0
                for ls in range(50):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    c_max_try = float(c_try.max())
                    if c_max_try < c_max:
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        break
                    alpha *= 0.5
                    if alpha < 1e-14:
                        break
        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _peak_pushing(a_init, deadline, max_steps=2000, top_k=20):
    """Direct subgradient method targeting the top-k convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    patience = 0
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_index = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for k in k_index:
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    grad[j] += 2.0 * a[ki]
        grad /= top_k
        grad = grad - np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = lr / (1.0 + 0.001 * i)
        v = momentum * v + step * (grad / gn)
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 200:
                lr *= 0.5
                momentum *= 0.9
                v = 0.5 * v
                patience = 0
                if lr < 1e-6:
                    break
    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_mean = np.sum(c) / len(c)
        c_max = float(c.max())
        excess = np.maximum(c - c_mean, 0)
        excess_max = excess.max()
        if excess_max < 1e-15:
            break
        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]
        step = 0.05 / (1.0 + 0.005 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a + 0.3 * (a_new - a)
            a = np.maximum(a, 0)
            a = a / a.sum()
    return best_a, best_sc


def _subgradient_nesterov(a_init, deadline, max_steps=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02
    momentum = 0.85
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.0003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + 0.0003 * i)
        z_new = z - step * grad
        z_new = z_prev + momentum * (z_new - z)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.7
                    if lr < 1e-5:
                        break
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=800.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs1")

    # Phase 2: Active-set QP polish
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _newton_active_polish(best_seq,
                                         min(deadline - 500, time.monotonic() + 250),
                                         max_rounds=100)
        update_best(a2, sc2, "active_qp1")

    # Phase 3: Peak pushing (new method)
    print(f"\n[Phase 3] Peak pushing (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3, sc3 = _peak_pushing(best_seq,
                                 min(deadline - 350, time.monotonic() + 100),
                                 max_steps=2000, top_k=30)
        update_best(a3, sc3, "peak_push1")

    # Phase 4: Flatten polish
    print(f"\n[Phase 4] Flatten (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _flatten_polish(best_seq,
                                   min(deadline - 300, time.monotonic() + 80))
        update_best(a4, sc4, "flatten1")

    # Phase 5: Subgradient with Nesterov
    print(f"\n[Phase 5] Subgrad+Nesterov (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _subgradient_nesterov(best_seq,
                                         min(deadline - 250, time.monotonic() + 100))
        update_best(a5, sc5, "subgrad1")

    # Phase 6: Second round active-set QP (tighter)
    print(f"\n[Phase 6] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a6, sc6 = _newton_active_polish(best_seq,
                                         min(deadline - 200, time.monotonic() + 120),
                                         max_rounds=100,
                                         tol_schedule=[0.0001, 0.00005, 0.00002,
                                                       0.00001, 0.000005, 0.000002,
                                                       0.000001, 0.0000005,
                                                       0.0000002, 0.0000001])
        update_best(a6, sc6, "active_qp2")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [32768, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init_n = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs(a_init_n, deadline, min_time_remaining=60.0, aggressive=True)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 50:
            a_np, sc_np = _newton_active_polish(a_opt,
                                                 min(deadline - 40, time.monotonic() + 60),
                                                 max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 8: Peak pushing round 2
    print(f"\n[Phase 8] Peak pushing round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a8, sc8 = _peak_pushing(best_seq,
                                 min(deadline - 80, time.monotonic() + 60),
                                 max_steps=1500, top_k=50)
        update_best(a8, sc8, "peak_push2")

    # Phase 9: Restarts with different profiles
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_kinds = ["tapered", "centered", "cos"]
        for i, prof in enumerate(restart_kinds[:2]):
            if time.monotonic() > deadline - 25:
                break
            n_r = len(best_seq)
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            else:
                p_start = _make_profile(n_r, prof)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

    # Phase 10: Final ultra-tight polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _newton_active_polish(
            best_seq, deadline - 3,
            max_rounds=80,
            tol_schedule=[0.000001, 0.0000005, 0.0000002,
                          0.0000001, 0.00000005, 0.00000002,
                          0.00000001, 0.000000005, 0.000000002,
                          0.000000001])
        update_best(a_fin, sc_fin, "final_qp")
        if time.monotonic() < deadline - 5:
            a_fl, sc_fl = _flatten_polish(best_seq, deadline - 3, max_iters=100)
            update_best(a_fl, sc_fl, "final_flatten")
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 3e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(1500, max(30, int(remaining / 0.015)))
                obj_fn = partial(_smoobj, beta=beta_fin, n=n)
                grad_fn = partial(_smograd, beta=beta_fin, n=n)
                try:
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
