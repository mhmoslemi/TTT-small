Smooth multiplicative weights QP optimizer with FFT-accelerated
Hankel linear algebra, preconditioned CG, and Newton polish.

import numpy as np
import time

# --- Helper functions (all top-level, no closures) ---

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,m] = 2*W[i+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _diag_precond(W, n):
    """Compute diagonal of H = 2*W[i+m] for i,m in [0,n-1].
    diag[i] = 2*W[2i]... wait no.
    H[i,i] = 2*W[i+i] = 2*W[2i].
    """
    # Actually H[i,m] = 2*W[i+m], so H[i,i] = 2*W[2*i]
    # But 2*i can go up to 2(n-1) = 2n-2, which is within range of W (length 2n-1)
    idx = 2 * np.arange(n)
    d = 2.0 * W[idx]
    d = np.maximum(d, 1e-30)
    return d

def _cg_solve(W, n, rhs, max_iter=2000, tol=1e-12, precond=True):
    """Solve H*x = rhs using preconditioned CG."""
    x = np.zeros(n)
    if precond:
        d = _diag_precond(W, n)
        z = rhs / d
    else:
        z = rhs.copy()
    x = np.zeros(n)
    r = rhs.copy()
    if precond:
        z = r / d
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-60:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        if precond:
            z_new = r / d
        else:
            z_new = r
        rz_new = float(r @ z_new)
        if not np.isfinite(rz_new) or abs(rz_new) > rz * 2:
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        beta = rz_new / rz
        p = z_new + beta * p
        z = z_new
        rz = rz_new
    return x

"""
Smooth multiplicative weights QP optimizer with FFT-accelerated Hankel linear
algebra, preconditioned CG, active-set management, log-barrier Newton polish,
and multi-n exploration. Targets min 2n*max(autocorr)/(sum)^2 for non-negative
sequences. Improves on prior approach via: (1) preconditioned CG with diagonal
preconditioner, (2) smooth softmax weight updates instead of hard active sets,
(3) log-barrier Newton for tight QP solves, (4) asymmetric perturbations,
(5) multi-n exploration up to 32768, and (6) adaptive continuation schedule.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute (H v)[i] = 2 * sum_j W[i+j] * v[j] for i in [0,n-1]."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hankel_matvec_zero(v, W, n, fixed):
    """Hankel matvec that respects fixed (zero) components."""
    v_work = v.copy()
    v_work[fixed] = 0.0
    result = _hankel_matvec(v_work, W, n)
    result[fixed] = 0.0
    return result


def _diag_precond(W, n):
    """Diagonal of the Hankel matrix H[i,i] = 2*W[2*i]."""
    idx = 2 * np.arange(n)
    d = 2.0 * W[idx]
    d = np.maximum(d, 1e-30)
    return d


def _cg_hankel(W, n, rhs, fixed=None, max_iter=1000, tol=1e-13, use_precond=True):
    """Solve H*x = rhs using preconditioned CG, optionally with fixed components."""
    if fixed is None:
        fixed = np.zeros(n, dtype=bool)
    n_free = n - int(fixed.sum())
    if n_free == 0:
        return np.zeros(n)

    x = np.zeros(n)
    r = rhs.copy()
    r[fixed] = 0.0

    # Preconditioner: diagonal of H
    if use_precond:
        d_diag = _diag_precond(W, n)
        d_diag = np.where(fixed, 1.0, d_diag)
        z = r / d_diag
    else:
        z = r.copy()

    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-60 or not np.isfinite(rz0):
        return x

    for i in range(max_iter):
        if fixed.any():
            Hp = _hankel_matvec_zero(p, W, n, fixed)
        else:
            Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x += alpha * p
        r -= alpha * Hp
        r[fixed] = 0.0
        if use_precond:
            z_new = r / d_diag
        else:
            z_new = r.copy()
        rz_new = float(r @ z_new)
        if not np.isfinite(rz_new):
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            break
        if abs(rz_new) > 2.0 * abs(rz):
            # Restart
            z_new = r / d_diag if use_precond else r.copy()
            p = z_new.copy()
            rz = float(r @ z_new)
            if abs(rz) < 1e-60:
                break
            continue
        beta = rz_new / rz
        p = z_new + beta * p
        p[fixed] = 0.0
        z = z_new
        rz = rz_new
    return x


def _solve_qp_active(W, n, a_init, max_outer=30, cg_iter=800):
    """Solve min a^T H a / 2 s.t. sum(a)=1, a>=0 using active-set + CG."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fixed = np.zeros(n, dtype=bool)
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) * 0.5)

    for outer in range(max_outer):
        n_free = n - int(fixed.sum())
        if n_free == 0:
            break

        rhs = np.ones(n)
        rhs[fixed] = 0.0

        y = _cg_hankel(W, n, rhs, fixed=fixed, max_iter=cg_iter, tol=1e-14)
        y[fixed] = 0.0
        oney = float(np.sum(y))
        if abs(oney) < 1e-25 or not np.isfinite(oney):
            break
        a_new = y / oney
        a_new[fixed] = 0.0
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check KKT: gradient should be >= lambda on fixed set
        grad = _hankel_matvec(a_new, W, n)
        free_idx = ~fixed
        if free_idx.any():
            lam = grad[free_idx].mean()
            # Check if any fixed component should be freed
            should_free = fixed & (grad < lam - 1e-10 * (np.max(np.abs(grad)) + 1e-15))
            if should_free.any():
                # Free the most violated components
                n_free_candidates = min(n - int(fixed.sum()) - 1, int(should_free.sum()) + 1)
                fixed_grads = grad[fixed]
                order = np.argsort(fixed_grads)
                # Convert to full indices
                fixed_indices = np.where(fixed)[0]
                to_free = fixed_indices[order[:n_free_candidates]]
                fixed[to_free] = False
            # Check if any free component should be fixed
            should_fix = free_idx & (a_new < 1e-12)
            if should_fix.any() and int(should_fix.sum()) < n_free // 2:
                fixed[should_fix] = True

        val_new = float(a_new @ _hankel_matvec(a_new, W, n) * 0.5)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

        # Convergence check
        if free_idx.any():
            grad = _hankel_matvec(a, W, n)
            lam = grad[free_idx].mean()
            resid = np.max(np.abs(grad[free_idx] - lam))
            if fixed.any():
                resid = max(resid, np.max(lam - grad[fixed]))
            if resid < 1e-10 * (np.max(np.abs(grad)) + 1e-15):
                break
        else:
            break

    return best_a


def _smooth_weights(c, temperature):
    """Compute smooth weights from autocorrelation values."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.ones_like(c) / len(c)
    shifted = (c - c_max) / temperature
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _log_barrier_newton(W, n, a_init, max_iters=200, eps_start=0.05, eps_end=1e-8,

    log_eps_end = np.log(eps_end)

                    break
                beta = rz_new / rz
                p = z_new + beta * p
                z = z_new
                rz = rz_new

            # Line search
            step_size = 1.0
            for ls in range(30):
                a_try = a - step_size * d
                a_try = np.maximum(a_try, 1e-15)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    step_size *= 0.5
                    continue
                a_try = a_try / s_try
                val_try = float(a_try @ _hankel_matvec(a_try, W, n) * 0.5) - eps * np.log(a_try).sum()
                val_cur = float(a @ _hankel_matvec(a, W, n) * 0.5) - eps * np.log(a).sum()
                if val_try < val_cur:
                    a = a_try
                    break
                step_size *= 0.5
                if step_size < 1e-14:
                    break

        # Update best
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        val = float(a @ _hankel_matvec(a, W, n) * 0.5)
        if val < best_val - 1e-16:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


def _newton_polish_tight(a_init, deadline, max_rounds=50, min_time=20.0):
    """Tight active-set Newton polish: use very narrow active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                    0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                    0.0000005, 0.0000002, 0.0000001]
    tol_idx = 0
    rounds_per_tol = 3

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - min_time:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = max(1, int(active.sum()))

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP
        a_qp = _solve_qp_active(W, n, a, max_outer=30, cg_iter=1000)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()

        # Also try a damped step
        grad = _hankel_matvec(a, W, n)
        lam = float(np.mean(grad))
        g_t = grad - lam
        step = 0.5 / (np.max(np.abs(g_t)) + 1e-15)
        a_try = a - step * g_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try

        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

    return best_a, best_sc


def _flatten_direct(a_init, deadline, max_iters=200, min_time=20.0):
    """Direct flattening: reduce variance of autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - min_time:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Excess above mean
        excess = np.maximum(c - c_mean, 0)
        excess_max = excess.max()
        if excess_max < 1e-15:
            break

        # Gradient of sum(excess^2) w.r.t. a
        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]

        step = 0.05 / (1.0 + 0.02 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric_linear":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "asymmetric_skew":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.7) ** 2) + np.exp(-8.0 * (t - 0.3) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = t ** 0.3 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_reopt(a, deadline, min_time=60.0):
    """Try zeroing out smallest components and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.02, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() > deadline - min_time:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)

    return best_a, best_sc
