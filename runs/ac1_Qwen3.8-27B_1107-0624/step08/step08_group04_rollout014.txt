"""
Iterative reweighting + L-BFGS hybrid optimizer for minimum peak autocorrelation.
Uses a novel "spectral boosting" approach: iteratively solve weighted QP subproblems
where weights focus on autocorrelation peaks, interleaved with L-BFGS polish and
multi-length exploration. Core innovation is the Hankel-weighted CG solver that
exploits FFT structure for O(n log n) QP solves.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_w_matvec(v, w, n):
    """Compute (w*H) @ v where (w*H)_{i,j} = w_{i+j}, w has length 2n-1."""
    v_rev = v[::-1]
    out_len = len(v_rev) + len(w) - 1
    N = _next_pow2(out_len)
    F_v = np.fft.rfft(v_rev, N)
    F_w = np.fft.rfft(w, N)
    conv = np.fft.irfft(F_v * F_w, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(w, n, rhs, max_iter=3000, tol=1e-14, warm_start=None):
    """Solve B*x = rhs where B_{i,j} = w_{i+j} using CG with FFT matvec."""
    if warm_start is not None:
        x = warm_start.copy()
        r = rhs - _hankel_w_matvec(x, w, n)
    else:
        x = np.zeros(n)
        r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = max(rs, 1e-30)
    if rs < 1e-30:
        return x
    for i in range(max_iter):
        Bp = _hankel_w_matvec(p, w, n)
        pHp = float(p @ Bp)
        if pHp < 1e-30:
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Bp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * (1 + 1e-6):
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _solve_weighted_qp(w, n, a_init, max_cg_iter=2000):
    """Solve min a^T B a s.t. sum(a)=1, a>=0 where B_{i,j}=w_{i+j}."""
    rhs = np.ones(n)
    x = _cg_solve_hankel(w, n, rhs, max_iter=max_cg_iter, warm_start=a_init)
    s = float(x.sum())
    if abs(s) < 1e-20:
        return a_init.copy()
    a = x / s
    neg_count = int((a < 0).sum())
    if neg_count == 0:
        return a
    a = np.maximum(a, 0)
    s2 = a.sum()
    if s2 < 1e-15:
        return np.ones(n) / n
    return a / s2


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive beta continuation on log-softmax parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                            0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                            0.00002, 0.00001, 0.000005, 0.000002])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                            0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_iter_time * 5))))
        try:
            result = minimize(
                _zobj_wrapper, z, args=(beta, n),
                jac=_zgrad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _zobj_wrapper(z, beta, n):
    return _smoobj(z, beta, n)


def _zgrad_wrapper(z, beta, n):
    return _smograd(z, beta, n)


def _iterative_reweighting(a_init, deadline, max_iter=500):
    """Core algorithm: iterative reweighting with Hankel QP subproblems."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Initialize weights: uniform
    w = np.ones(2 * n - 1) / (2 * n - 1)

    # Adaptive beta for weight update
    beta = 0.02 * c_max
    beta_decay = 0.97
    min_beta = 1e-8 * c_max

    for it in range(max_iter):
        if time.monotonic() > deadline - 5:
            break

        # Solve weighted QP
        a_qp = _solve_weighted_qp(w, n, a, max_cg_iter=1500)

        # Evaluate
        c_new = _full_convolve(a_qp, a_qp)
        sc = _score(a_qp)

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_qp.copy()
            a = a_qp.copy()

        # Update weights: focus on peaks using softmax
        c_max_new = float(c_new.max())
        shifted = (c_new - c_max_new) / beta
        w_new = np.exp(np.clip(shifted, -500, 0))
        w_new_sum = w_new.sum()
        if w_new_sum < 1e-300:
            w_new = np.zeros(2 * n - 1)
            w_new[int(np.argmax(c_new))] = 1.0
        else:
            w_new = w_new / w_new_sum

        # Blend weights for stability
        blend = 0.5 + 0.3 * (it / max_iter)
        w = (1 - blend) * w + blend * w_new

        # Decay beta (concentrate weights more over time)
        beta = max(beta * beta_decay, min_beta)

        # Check convergence
        if it > 10 and abs(sc - best_sc) < 1e-14:
            if it > 50:
                break

    return best_a, best_sc


def _peak_targeting_polish(a_init, deadline):
    """Directly target the top few autocorrelation peaks with CG steps."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(200):
        if time.monotonic() > deadline - 5:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top-5 peaks
        top_k = min(5, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_k]
        active = np.zeros(2 * n - 1)
        active[top_indices] = 1.0 / top_k

        # Build weighted Hankel and solve QP
        w = active
        a_qp = _solve_weighted_qp(w, n, a, max_cg_iter=1000)

        # Also try: reduce the single max peak
        w_single = np.zeros(2 * n - 1)
        w_single[int(np.argmax(c))] = 1.0
        a_single = _solve_weighted_qp(w_single, n, a, max_cg_iter=800)

        # Pick the better
        sc_qp = _score(a_qp)
        sc_single = _score(a_single)

        if sc_single < sc_qp:
            a_candidate = a_single
            sc_candidate = sc_single
        else:
            a_candidate = a_qp
            sc_candidate = sc_qp

        if sc_candidate < best_sc - 1e-15:
            best_sc = sc_candidate
            best_a = a_candidate.copy()
            a = a_candidate.copy()
        else:
            # Try a damped step
            step = a_candidate - a
            for alpha in [0.5, 0.25, 0.1]:
                a_try = a + alpha * step
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=300):
    """Direct gradient descent on sum of squared excess above target."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.05
    for it in range(max_iters):
        if time.monotonic() > deadline - 3:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_target = c_max - 0.01 * c_max

        # Excess above target (only positive part)
        excess = np.maximum(c - c_target, 0)
        if excess.max() < 1e-15:
            break

        # Gradient of (1/2)*sum(excess^2) w.r.t. a
        # d(c_k)/da_i = a_{k-i} (for valid indices)
        # grad_i = sum_k excess_k * d(c_k)/da_i = (excess * a)_i (correlation)
        a_rev = a[::-1]
        ex_conv = _full_convolve(excess, a_rev)
        grad = ex_conv[n - 1:2 * n - 1]

        # Center the gradient (project off sum constraint)
        grad = grad - grad.mean()

        # Adaptive step size with backtracking
        step = lr / (1.0 + 0.001 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.8
            a = a + 0.5 * step * grad
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    if best_seq is not None and time.monotonic() < deadline - 60:
        a6, sc6 = _peak_targeting_polish(best_seq, min(deadline - 50, time.monotonic() + 50))
        _update_best(a6, sc6, "peak_target_2")

    # Phase 7: Multi-length exploration
    print(f"\n[Phase 7] Multi-length (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 32768, 24576]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 40:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        # Quick reweighting
        a_opt, sc_opt = _iterative_reweighting(a_init, min(deadline - 30, time.monotonic() + 40),
                                                max_iter=100)
        _update_best(a_opt, sc_opt, f"n_{n_try}_rw")

        if time.monotonic() < deadline - 20:
            a_fb = _run_lbfgs(a_opt, deadline, min_time_remaining=20, aggressive=False)
            _update_best(a_fb, _score(a_fb), f"n_{n_try}_lbfgs")

    # Phase 8: Randomized restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(3):
            if time.monotonic() > deadline - 10:
                break
            p_start = _perturb(best_seq, 0.03 + 0.02 * i, rng)
            a_r, sc_r = _iterative_reweighting(p_start, min(deadline - 8, time.monotonic() + 25),
                                                max_iter=80)
            _update_best(a_r, sc_r, f"restart_{i}")

    # Phase 9: Final ultra-precise polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _peak_targeting_polish(best_seq, deadline - 2)
        _update_best(a_fin, sc_fin, "final_peak")

        if time.monotonic() < deadline - 2:
            a_fl, sc_fl = _flatten_polish(best_seq, deadline - 1, max_iters=100)
            _update_best(a_fl, sc_fl, "final_flatten")

        if time.monotonic() < deadline - 2 and best_seq is not None:
            a_rw, sc_rw = _iterative_reweighting(best_seq, deadline - 1, max_iter=50)
            _update_best(a_rw, sc_rw, "final_rw")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
