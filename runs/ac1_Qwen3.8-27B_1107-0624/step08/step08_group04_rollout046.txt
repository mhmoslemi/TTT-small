def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)

"""
Optimized sequence generator using multi-phase approach:
1. L-BFGS with adaptive smooth-max (log-sum-exp) approximation

3. Spectral-domain refinement
4. Multi-n exploration with up/down-sampling

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    v_rev = v[::-1]
    out_len = len(v_rev) + len(W) - 1
    N = _next_pow2(out_len)
    F_v = np.fft.rfft(v_rev, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_v * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

def _cg_solve(W, n, rhs, max_iter=500, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.2, max_c * 0.001)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                           0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
                           0.00001, 0.000005, 0.000002, 0.000001, 0.0000005,
                           0.0000002, 0.0000001, 0.00000005, 0.00000002, 0.00000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                           0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
                           0.00001, 0.000005, 0.000002, 0.000001])

    betas = base_beta * ratios
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_iter_time * 15))))
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a

def _kkt_polish(a_init, deadline, max_rounds=200):
    """KKT-based active set refinement."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Tolerance schedule for active set
    tols = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
            0.00002, 0.00001, 0.000005, 0.000002, 0.000001, 0.0000005,
            0.0000002, 0.0000001]
    rounds_per_tol = max(1, max_rounds // len(tols))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_idx = min(rnd // rounds_per_tol, len(tols) - 1)
        tol = tols[tol_idx] * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        W = np.zeros(2 * n - 1)
        n_act = active.sum()
        W[active] = 1.0 / n_act

        # Compute gradient
        g = _hankel_matvec(a, W, n)
        
        # Project onto sum=0 subspace
        g_mean = g.mean()
        g_t = g - g_mean
        
        # Newton step: solve H*dx = -g_t, then adjust for sum constraint
        Hg = _cg_solve(W, n, g_t, max_iter=200)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=200)
        
        # dx = -Hg + lambda*H1 where lambda chosen so sum(dx) = 0
        sum_Hg = Hg.sum()
        sum_H1 = H1.sum()
        if abs(sum_H1) > 1e-20:
            lam = sum_Hg / sum_H1
            dx = -Hg + lam * H1
        else:
            dx = -g_t
        dx = dx - dx.mean()
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            sc_try = float(2.0 * n * c_max_try / s_try**2)  # wait, s_try is now 1 after normalization
            # Actually after normalization sum = 1, so score = 2n * max(c)
            if c_max_try < c_max * (1 - 1e-14):
                a = a_try
                sc = float(2.0 * n * c_max_try)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break
        
        if not improved:
            # Try smaller fixed step
            step = 0.01 / (1.0 + 0.01 * rnd)
            a_try = a - step * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max * (1 - 1e-14):
                    a = a_try
                    sc = float(2.0 * n * c_max_try)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()

        # Check KKT residual for convergence
        if rnd % 10 == 0:
            g = _hankel_matvec(a, W, n)
            support = a > 1e-12
            if support.sum() > 0:
                resid = np.max(np.abs(g[support] - g[support].mean()))
                if resid < 1e-10 * (np.max(np.abs(g)) + 1e-15):
                    break

    return best_a, best_sc

def _subgrad_accelerated(a_init, deadline, max_steps=5000):
    """Nesterov-accelerated subgradient for max autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    y = z.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    t_val = 1.0
    patience = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        c_max = float(c.max())
        
        # Adaptive active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient in z-space
        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad_a = 2.0 * n * a_y * (h - W_val)
        
        # Convert to z-space gradient (chain rule: da/dz = a*(I - a*a^T))
        # grad_z = grad_a / a (approximately, for the softmax parameterization)
        # Actually: d(f(a(z)))/dz_i = sum_j (df/da_j)(da_j/dz_i)
        # da_j/dz_i = a_j*(delta_ji - a_i)
        # So grad_z_i = sum_j grad_a_j * a_j*(delta_ji - a_i) = grad_a_i*a_i - a_i*sum_j grad_a_j*a_j
        # = a_i * (grad_a_i - a @ grad_a)
        ag = float(np.dot(a_y, grad_a))
        grad_z = a_y * (grad_a - ag)
        
        # Nesterov update
        t_new = 1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val) / 2.0
        beta_mom = (t_val - 1.0) / t_new
        
        step = 1.0 / (t_new * np.linalg.norm(grad_z) + 1e-30)
        z_new = z - step * grad_z
        z_new = np.clip(z_new, -50.0, 50.0)
        
        y = z_new + beta_mom * (z_new - z)
        y = y - y.mean()
        z = z_new
        z = z - z.mean()
        t_val = t_new
        
        # Periodic evaluation
        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    # Reset to best
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    y = z.copy()
                    patience = 0
                    t_val = 1.0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _flatten_polish(a_init, deadline, max_iters=300):
    """Iterative flattening of autocorrelation peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Excess above mean
        c_mean = c.sum() / len(c)
        excess = np.maximum(c - c_mean, 0)
        if excess.max() < 1e-15:
            break
        
        # Gradient of sum(excess^2) w.r.t. a
        a_rev = a[::-1]
        exc_conv = _full_convolve(excess, a_rev)
        grad = 2.0 * exc_conv[n - 1:2 * n - 1]
        
        # Step with decay
        step = 0.05 / (1.0 + 0.02 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _spectral_init(n, rng):
    """Create initialization with approximately flat power spectrum."""
    # Start with random phases in frequency domain
    N = _next_pow2(n)
    freq = np.fft.rfft(np.zeros(N))
    # Set DC component
    freq[0] = n  # sum = n (will normalize later)
    # Set other frequencies with random phases
    for j in range(1, N // 2 + 1):
        freq[j] = np.sqrt(n) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    a = np.fft.irfft(freq, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return _make_profile(n, "tapered")
    return a / s
