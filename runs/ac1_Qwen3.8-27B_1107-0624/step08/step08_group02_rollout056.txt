def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _pcg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Preconditioned conjugate gradient for Hankel system.
    Uses diagonal preconditioner derived from the weight vector."""
    x = np.zeros(n)
    r = rhs.copy()
    # Diagonal preconditioner: D = (H * e) where e ones, approximated
    Hp_ones = _hankel_matvec(np.ones(n) * 0.001, W, n)
    Hp_ones = Hp_ones / 0.001  # This is H * ones
    diag = np.abs(Hp_ones) + 1e-20
    # Simpler: use identity preconditioner but with better initial guess
    z = r.copy()
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if rz0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp) or pHp < 0:
            break
        alpha = rz / pHp
        if not np.isfinite(alpha) or abs(alpha) > 1e10:
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rz_new = float(r_new @ r_new)
        if not np.isfinite(rz_new) or rz_new > rz * 1.5:
            break
        if rz_new < tol * tol * rz0:
            x = x_new
            break
        beta_cg = rz_new / rz
        p = r_new + beta_cg * p
        z = r_new
        x = x_new
        r = r_new
        rz = rz_new
    return x


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp) or pHp < 0:
            break
        alpha = rs / pHp
        if not np.isfinite(alpha) or abs(alpha) > 1e10:
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        if rs_new < tol * tol * rs0:
            x = x_new
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r_new
        rs = rs_new
    return x
