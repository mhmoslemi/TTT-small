"""
Portfolio optimizer for sequence convolution objectives.

The program tries several families of sequences and several refinement
routines:
  - direct black-box derivative-free optimization of low-dimensional DCT
    parameters using the actual evaluate_sequence oracle,
  - high-dimensional log-sum-exp smoothing over the autocorrelation,
  - multiple surrogate lag weightings (full, balanced, off-diagonal-ish),
  - multi-length final selection using the actual evaluator when available.

All helper functions are top-level and lambda-free.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(x):
    x = max(2, int(x))
    k = 1
    while k < x:
        k <<= 1
    return k


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones_like(z, dtype=np.float64) / z.size
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / z.size
    return e / s


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s


def _fft_convolve(a, b=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.ones(1, dtype=np.float64) * 0.0
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    out = np.fft.irfft(fa * fb, N)[:out_n]
    return out


def _autocorr(a):
    return _fft_convolve(a, a)


def _logit(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.mean(z))
    return z


def _resample(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_new = max(1, int(n_new))
    if a.size == 0:
        return np.ones(n_new, dtype=np.float64) / n_new
    if a.size == n_new:
        return _normalize(a)
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    b = np.interp(x_new, x_old, a)
    b = np.clip(b, 0.0, None)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones(n_new, dtype=np.float64) / n_new
    return b / s


def _local_score_full(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * np.max(c))


def _local_score_off(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return np.inf
    c = _autocorr(a)
    if n == 1:
        return 0.0
    center = n - 1
    off = np.delete(c, center)
    return float(2.0 * n * np.max(off))


def _call_evaluate(a):
    a = _normalize(a)
    fn = globals().get("evaluate_sequence", None)
    if fn is not None:
        try:
            seq = [float(x) for x in a]
            val = fn(seq)
            val = float(val)
            if np.isfinite(val):
                return val
        except Exception:
            pass
    return _local_score_full(a)


def _state_update(state, a, sc, label=""):
    if a is None:
        return
    if not np.isfinite(sc):
        return
    a = _normalize(a)
    sc = float(sc)
    if sc < state["sc"] - 1e-15:
        state["a"] = a.copy()
        state["sc"] = sc
        print(f"  new best: {sc:.12f} (n={a.size}) [{label}]")


def _pool_key(item):
    return item[1]


def _pool_add(pool, a, sc):
    if a is None or not np.isfinite(sc):
        return
    a = _normalize(a)
    pool.append((a.copy(), float(sc), int(a.size)))
    pool.sort(key=_pool_key)
    if len(pool) > 40:
        del pool[40:]


def _make_profile(n, kind, rng=None, param=0.5):
    n = max(1, int(n))
    t = np.linspace(0.0, 1.0, n, endpoint=False)

    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "taper":
        p = max(0.05, float(param))
        a = np.sin(np.pi * t) ** p
        a = a + 1e-12
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
        a = a + 1e-12
    elif kind == "centered":
        w = max(0.05, float(param))
        a = np.exp(-14.0 * (w * (t - 0.5)) ** 2) + 1e-10
    elif kind == "bimodal":
        w = max(0.2, float(param))
        a = np.exp(-10.0 * w * (t - 0.3) ** 2) + np.exp(-10.0 * w * (t - 0.7) ** 2) + 1e-10
    elif kind == "alternating":
        alpha = float(param)
        alpha = min(0.99, max(0.0, alpha))
        s = np.where((np.arange(n) % 2) == 0, 1.0, -1.0)
        taper = np.sin(np.pi * t) ** 0.05 + 1e-9
        a = (1.0 + alpha * s) * taper
        a = np.maximum(a, 1e-12)
    elif kind == "spike":
        sigma = max(0.0008, min(0.25, abs(float(param)) * 0.15))
        if param < 0.03:
            a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 1e-12
        else:
            a = np.exp(-0.5 * ((t - 0.3) / sigma) ** 2) + np.exp(-0.5 * ((t - 0.7) / sigma) ** 2) + 1e-12
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(12345)
        scale = max(0.1, float(param))
        z = rng.normal(0.0, scale, n)
        a = _softmax(z)
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _perturb_logits(a, sigma, rng):
    z = _logit(a)
    z = z + sigma * rng.standard_normal(a.size)
    return _softmax(z)


def _make_dct_basis(n, K):
    n = max(1, int(n))
    K = max(1, int(K))
    K = min(K, max(1, n // 2))
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / float(n)
    m = np.arange(K, dtype=np.float64)[:, None]
    B = np.cos(np.pi * m * x[None, :])
    norms = np.linalg.norm(B, axis=1)
    B = B / (norms[:, None] + 1e-30)
    return B


def _project_to_basis(z, B):
    z = np.asarray(z, dtype=np.float64).ravel()
    return B @ z


def _basis_to_logits(coeff, B):
    coeff = np.asarray(coeff, dtype=np.float64).ravel()
    z = B.T @ coeff
    z = np.clip(z, -40.0, 40.0)
    z = z - float(np.mean(z))
    return z


def _blackbox_obj_coeff(coeff, B, n, deadline):
    if time.monotonic() > deadline:
        return 1e100
    coeff = np.asarray(coeff, dtype=np.float64).ravel()
    if not np.all(np.isfinite(coeff)):
        return 1e100
    z = B.T @ coeff
    if not np.all(np.isfinite(z)):
        return 1e100
    z = np.clip(z, -40.0, 40.0)
    z = z - float(np.mean(z))
    a = _softmax(z)
    return _call_evaluate(a)


def _smooth_lse_value_unnorm(a, beta, weights):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 1e100
    c = _autocorr(a)
    r = weights * c
    m = float(np.max(r))
    if not np.isfinite(m):
        return 1e100
    if beta <= 1e-30:
        return float(2.0 * n * m)
    shifted = (r - m) / beta
    e = np.exp(np.clip(shifted, -700.0, 0.0))
    S = float(e.sum())
    if S < 1e-300 or not np.isfinite(S):
        return float(2.0 * n * m)
    return float(2.0 * n * (m + beta * np.log(S)))


def _smooth_lse_obj(z, beta, n, weights, deadline):
    if time.monotonic() > deadline:
        return 1e100
    a = _softmax(z)
    return _smooth_lse_value_unnorm(a, beta, weights)


def _smooth_lse_jac(z, beta, n, weights, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z, dtype=np.float64)
    a = _softmax(z)
    c = _autocorr(a)
    r = weights * c
    m = float(np.max(r))
    if not np.isfinite(m):
        return np.zeros_like(z, dtype=np.float64)

    if m <= 1e-300 * max(1.0, float(np.max(weights))):
        return np.zeros_like(z, dtype=np.float64)

    if beta <= 1e-30:
        w = np.zeros_like(r, dtype=np.float64)
        w[int(np.argmax(r))] = 1.0
    else:
        shifted = (r - m) / beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        S = float(e.sum())
        if S < 1e-300 or not np.isfinite(S):
            w = np.zeros_like(r, dtype=np.float64)
            w[int(np.argmax(r))] = 1.0
        else:
            w = e / S

    eff = w * weights
    a_rev = a[::-1]
    wc = _fft_convolve(eff, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(eff, c))
    grad = 4.0 * n * a * (h - W)
    return grad


def _make_weights_lse(n, kind):
    m = 2 * n - 1
    w = np.ones(m, dtype=np.float64)
    center = n - 1
    if kind == "full":
        pass
    elif kind == "balanced":
        w[center] = 0.35
    elif kind == "off_reg":
        w[center] = 1e-8
    elif kind == "off_center2":

        if center + 1 < m:
            w[center + 1] = 0.5
        if center - 1 >= 0:
            w[center - 1] = 0.5
    else:

    return w


def _run_lbfgs_lse(a_init, weights, deadline, aggressive=False):
    a = _normalize(a_init)
    n = a.size
    if n == 0:
        return a
    z = _logit(a)
    current = a.copy()

    try:
        c0 = _autocorr(a)
        r0 = weights * c0
        maxr = float(np.max(r0))
        if not np.isfinite(maxr) or maxr <= 0.0:
            return a
        p95 = float(np.percentile(r0, 95))
        p05 = float(np.percentile(r0, 5))
        spread = max(0.0, p95 - p05)
        base = max(1e-6 * maxr, 0.25 * spread, 1e-12)
    except Exception:
        base = 1e-6

    if aggressive:
        ratios = [
            1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125,
            0.00390625, 0.001953125, 0.0009765625, 0.00048828125,
            0.000244140625, 0.0001220703125, 6.103515625e-05,
            3.0517578125e-05, 1.52587890625e-05, 7.62939453125e-06,
            3.814697265625e-06, 1.9073486328125e-06,
            9.5367431640625e-07, 4.76837158203125e-07,
            2.384185791015625e-07
        ]
    else:
        ratios = [
            1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125,
            0.00390625, 0.001953125, 0.0009765625, 0.00048828125,
            0.000244140625, 0.0001220703125, 6.103515625e-05,
            3.0517578125e-05, 1.52587890625e-05, 7.62939453125e-06,
            3.814697265625e-06, 1.9073486328125e-06
        ]

    betas = [base * q for q in ratios]

    for beta in betas:
        if time.monotonic() + 5.0 > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 8.0:
            break

        scale = max(1.0, float(n) / 4096.0)
        per_iter = 0.09 * scale
        maxiter = int(min(1500, max(25, remaining / max(per_iter, 1e-6))))

        val_before = _smooth_lse_value_unnorm(current, beta, weights)
        try:
            res = minimize(
                _smooth_lse_obj,
                z,
                jac=_smooth_lse_jac,
                args=(beta, n, weights, deadline),
                method="L-BFGS-B",
                options={
                    "maxcor": 50,
                    "maxiter": maxiter,
                    "ftol": 1e-18,
                    "gtol": 1e-12,
                    "maxls": 80,
                }
            )
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            if z_new.size != n:
                continue
            if not np.all(np.isfinite(z_new)):
                continue
            a_new = _softmax(z_new)
            val_after = _smooth_lse_value_unnorm(a_new, beta, weights)
            if np.isfinite(val_after) and val_after < val_before - 1e-14:
                current = a_new
                z = z_new - float(np.mean(z_new))
        except Exception:
            pass

    return current

    if raw_parent is not None:
        try:
            p = np.asarray(raw_parent, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            s = float(p.sum())
            if p.size > 10 and s > 0.01 and np.isfinite(s):
                parent = p / s
        except Exception:
            parent = None

    if parent is not None:
        eval_parent = parent
        if parent.size > 65536:
            eval_parent = _resample(parent, 16384)
        if time.monotonic() < deadline - 120.0:
            sc = _call_evaluate(eval_parent)
            _state_update(state, eval_parent, sc, "parent")
            _pool_add(pool, eval_parent, sc)

    # Small black-box optimization length.
    n_eval = 1024

    # Estimate evaluator cost at small size.
    test_a = _make_profile(n_eval, "random", rng, 1.0)
    te0 = time.monotonic()
    sc_test = _call_evaluate(test_a)
    per_eval_small = max(1e-4, time.monotonic() - te0)
    _pool_add(pool, test_a, sc_test)

    if per_eval_small > 0.025:
        n_eval = 512
        te0 = time.monotonic()
        sc_test = _call_evaluate(_make_profile(n_eval, "random", rng, 1.0))
        per_eval_small = max(1e-4, time.monotonic() - te0)

    # Initial fixed profiles at small size.
    specs = [
        ("uniform", None),
        ("taper", 1.0),
        ("taper", 2.0),
        ("cos", None),
        ("centered", 1.0),
        ("centered", 2.5),
        ("centered", 5.0),
        ("bimodal", 1.5),
        ("bimodal", 4.0),
        ("alternating", 0.35),
        ("alternating", 0.65),
        ("alternating", 0.92),
        ("spike", 0.01),
        ("spike", 0.05),
        ("spike", 0.20),
        ("random", 1.0),
        ("random", 2.0),
        ("random", 3.5),
    ]

    for kind, prm in specs:
        if time.monotonic() > deadline - 900.0:
            break
        a = _make_profile(n_eval, kind, rng, prm if prm is not None else 0.5)
        sc = _call_evaluate(a)
        _state_update(state, a, sc, f"init_{kind}_{prm}")
        _pool_add(pool, a, sc)

    # Low-dimensional DCT black-box optimization using actual evaluator.
    blackbox_end = min(deadline, t0 + min(600.0, 0.35 * total), deadline - 420.0)
    if time.monotonic() < blackbox_end - 60.0:
        K = 64
        B = _make_dct_basis(n_eval, K)

        restarts = []

        if parent is not None:
            a_par = _resample(parent, n_eval)
            restarts.append(_project_to_basis(_logit(a_par), B))

        if state["a"] is not None:
            restarts.append(_project_to_basis(_logit(_resample(state["a"], n_eval)), B))

        if len(pool) > 0:
            restarts.append(_project_to_basis(_logit(pool[0][0]), B))

        restarts.append(np.zeros(K, dtype=np.float64))
        restarts.append(rng.normal(0.0, 0.15, K))
        restarts.append(rng.normal(0.0, 0.35, K))

        per_eval_small = max(1e-4, per_eval_small)
        restarts = restarts[:5]

        for i, coeff0 in enumerate(restarts):
            remaining_to_phase = blackbox_end - time.monotonic()
            if remaining_to_phase < 60.0:
                break
            maxfev = int(max(300, min(8000, (remaining_to_phase - 30.0) / (per_eval_small * 3.0))))
            try:
                res = minimize(
                    _blackbox_obj_coeff,
                    coeff0,
                    args=(B, n_eval, blackbox_end),
                    method="Powell",
                    options={
                        "maxiter": 1200,
                        "maxfev": maxfev,
                        "xtol": 1e-10,
                        "ftol": 1e-10,
                        "disp": False,
                    }
                )
                coeff_new = np.asarray(res.x, dtype=np.float64).ravel()
                if coeff_new.size == K and np.all(np.isfinite(coeff_new)):
                    z = _basis_to_logits(coeff_new, B)
                    a = _softmax(z)
                    sc = _call_evaluate(a)
                    _state_update(state, a, sc, f"blackbox_{i}")
                    _pool_add(pool, a, sc)
                _pool_add(pool, _softmax(B.T @ coeff0), _call_evaluate(_softmax(B.T @ coeff0)))
            except Exception:
                pass

    # High-dimensional surrogate refinement.
    n_large = 8192
    if parent is not None and 4096 <= parent.size <= 32768:
        n_large = int(parent.size)
    n_large = max(4096, min(16384, n_large))

    large_end = deadline - max(120.0, 0.10 * total)
    if time.monotonic() < large_end - 120.0:
        candidates_large = []

        if state["a"] is not None:
            candidates_large.append(("best_small", _resample(state["a"], n_large)))

        if parent is not None:
            candidates_large.append(("parent", _resample(parent, n_large)))

        for label in ["uniform", "taper", "alternating", "spike"]:
            prm = 0.7 if label == "alternating" else (0.03 if label == "spike" else 1.0)
            candidates_large.append((label, _make_profile(n_large, label, rng, prm)))

        for item in pool[:5]:
            candidates_large.append((f"pool_{int(item[1] * 1000000)}", _resample(item[0], n_large)))

        # Evaluate and order a few large candidates.
        cand_scores = []
        for label, a in candidates_large:
            if time.monotonic() > large_end - 80.0:
                break
            a = _normalize(a)
            sc = _call_evaluate(a)
            cand_scores.append((a, sc, label))
            _pool_add(pool, a, sc)
            if len(cand_scores) >= 8:
                break

        cand_scores.sort(key=_pool_key)
        top_cands = [x[0] for x in cand_scores[:3]]
        if not top_cands and parent is not None:
            top_cands = [_resample(parent, n_large)]
        if not top_cands:
            top_cands = [_make_profile(n_large, "uniform")]

        weight_kinds = ["full", "balanced", "off_reg", "off_center2"]
        run_budget = max(70.0, min(220.0, 0.35 * (large_end - time.monotonic())))

        runs_done = 0
        for cand in top_cands:
            for wk in weight_kinds:
                if time.monotonic() > large_end - 80.0 or runs_done >= 6:
                    break
                run_end = min(large_end, time.monotonic() + run_budget)
                if run_end < time.monotonic() + 35.0:
                    break
                weights = _make_weights_lse(cand.size, wk)
                aggressive = (wk != "full")
                try:
                    a_out = _run_lbfgs_lse(cand, weights, run_end, aggressive=aggressive)
                    sc_out = _call_evaluate(a_out)
                    _state_update(state, a_out, sc_out, f"lse_{wk}")
                    _pool_add(pool, a_out, sc_out)
                    runs_done += 1
                except Exception:
                    pass
            if time.monotonic() > large_end - 120.0 or runs_done >= 6:
                break

    # Add a few perturbed restarts if time remains.
    if state["a"] is not None and time.monotonic() < deadline - 90.0:
        for sig in [0.02, 0.05, 0.10]:
            if time.monotonic() > deadline - 60.0:
                break
            a_p = _perturb_logits(state["a"], sig, rng)
            sc_p = _call_evaluate(a_p)
            _state_update(state, a_p, sc_p, f"perturb_{sig}")
            _pool_add(pool, a_p, sc_p)

    # Final multi-length selection.
    final_pool = pool.copy()
    final_pool.sort(key=_pool_key)

    final_lengths = [2048, 4096, 8192, 16384]
    if parent is not None and parent.size not in final_lengths:
        if 2048 <= parent.size <= 32768:
            final_lengths.append(int(parent.size))
    final_lengths = sorted(set(final_lengths))

    evaluated_pairs = set()
    for arr, sc, ln in final_pool[:8]:
        for L in final_lengths:
            if time.monotonic() > deadline - 3.0:
                break
            key = (int(sc * 1e6), ln, int(L))
            if key in evaluated_pairs:
                continue
            evaluated_pairs.add(key)
            a2 = _resample(arr, L)
            sc2 = _call_evaluate(a2)
            _state_update(state, a2, sc2, f"final_L{L}")
            _pool_add(pool, a2, sc2)
        if time.monotonic() > deadline - 10.0:
            break

    # Ensure at least one sane fallback.
    if state["a"] is None or not np.isfinite(state["sc"]):
        fallback = _make_profile(8192, "uniform")
        state["a"] = fallback
        state["sc"] = _local_score_full(fallback)

    final = np.asarray(state["a"], dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not np.isfinite(s) or s < 0.01:
        final = np.ones(8192, dtype=np.float64) / 8192.0
    else:
        final = final / s

    print(f"FINAL: n={final.size}, local_full={_local_score_full(final):.12f}, "
          f"eval={_call_evaluate(final):.12f}, time={time.monotonic() - t0:.1f}s")
    return [float(x) for x in final]
