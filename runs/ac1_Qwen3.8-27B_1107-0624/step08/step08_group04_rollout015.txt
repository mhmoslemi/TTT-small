def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_local(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(v, lam, n):
    """Compute H*v where H[i,j] = lam[i+j], lam has length 2n-1."""
    # H[i,j] = lam[i+j]
    # (Hv)[i] = sum_j lam[i+j] * v[j]
    # Let's compute this via FFT.
    # (Hv)[i] = sum_j lam[i+j] * v[j]
    # = convolution of lam with reversed v, evaluated at position i + (n-1)
    # Actually: let u = v[::-1], then (Hv)[i] = sum_j lam[i+j] * v[j]
    # = sum_j lam[i+j] * u[n-1-j]
    # Let k = n-1-j, j = n-1-k:
    # = sum_k lam[i+n-1-k] * u[k]
    # = (lam_rev * u)[i+n-1] where lam_rev[k] = lam[2n-2-k]
    # Wait, that's cross-correlation. Let me just do it differently.
    # (Hv)[i] = sum_j lam[i+j] * v[j] for j=0..n-1
    # This is the "anti-diagonal" sum. 
    # Alternative: (Hv)[i] = (lam * v_rev)[i + n - 1] where v_rev = v[::-1]
    # Let's verify: (lam * v_rev)[m] = sum_j lam[j] * v_rev[m-j] = sum_j lam[j] * v[n-1-(m-j)] = sum_j lam[j] * v[n-1-m+j]
    # We want sum_j lam[i+j] * v[j]. Set k = i+j, j = k-i:
    # = sum_k lam[k] * v[k-i] (for valid k, j)
    # = sum_k lam[k] * v[k-i] for k-i in [0, n-1], k in [0, 2n-2]
    # (lam * v_rev)[m] with m = i: sum_k lam[k] * v_rev[i-k] = sum_k lam[k] * v[n-1-i+k]
    # Let j = n-1-i+k, k = j-i+n-1: sum_j lam[j-i+n-1] * v[j]
    # So (lam * v_rev)[i] = sum_j lam[j-i+n-1] * v[j]
    # We want sum_j lam[i+j] * v[j].
    # These are different. Let me try (v_rev * lam)[m]:
    # (v_rev * lam)[m] = sum_j v_rev[j] * lam[m-j] = sum_j v[n-1-j] * lam[m-j]
    # Let k = n-1-j, j = n-1-k: sum_k v[k] * lam[m-n+1+k]
    # We want sum_j lam[i+j] * v[j] = sum_k v[k] * lam[i+k]
    # So (v_rev * lam)[m] with m-n+1+k = i+k, i.e., m = i+n-1.
    # So (Hv)[i] = (v_rev * lam / stuff)[i+n-1]
    # Actually, let me just verify the convolution direction.
    # np.convolve(a, b)[k] = sum_i a[i] * b[k-i]
    # So np.convolve(v_rev, lam)[k] = sum_i v_rev[i] * lam[k-i] = sum_i v[n-1-i] * lam[k-i]
    # Let j = n-1-i, i = n-1-j: sum_j v[j] * lam[k-n+1+j]
    # We want sum_j v[j] * lam[i+j]. So k-n+1 = i, k = i+n-1.
    # Therefore: (Hv)[i] = np.convolve(v_rev, lam)[i+n-1]
    
    v_rev = v[::-1]
    out_len = len(v_rev) + len(lam) - 1  # = n + (2n-1) - 1 = 3n-2
    N = _next_pow2(out_len)
    F_vr = np.fft.rfft(v_rev, N)
    F_lam = np.fft.rfft(lam, N)
    conv = np.fft.irfft(F_vr * F_lam, N)[:out_len]
    return conv[n-1:2*n-1]

def _cg_solve_hankel(lam, n, rhs, max_iter=1000, tol=1e-12):
    """Solve H*x = rhs where H[i,j] = lam[i+j] using CG."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, lam, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            # Restart CG
            p = r.copy()
            rs = rs_new
            continue
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _solve_qp_hankel(lam, n, a_init, max_outer=200, cg_max_iter=800):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 where H[i,j]=lam[i+j]."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, lam, n))
    
    support = np.ones(n, dtype=bool)
    
    for outer in range(max_outer):
        n_sup = int(support.sum())
        if n_sup < 2:
            support[:] = True
            n_sup = n
        
        if n_sup == n:
            # Solve full system: H*x = 1, then normalize
            ones = np.ones(n)
            y = _cg_solve_hankel(lam, n, ones, max_iter=cg_max_iter)
            y_sum = float(np.sum(y))
            if abs(y_sum) < 1e-20 or not np.isfinite(y_sum):
                break
            a_new = y / y_sum
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            # Solve on support
            # Extract sub-Hankel: H_S[i,j] = lam[supp[i] + supp[j]]
            # This is trickier. Use projected gradient instead.
            grad = _hankel_matvec(a, lam, n)
            # Projected gradient step on the constraint sum(a)=1, a>=0
            grad_proj = grad - np.mean(grad[support])
            # Only update on support
            grad_full = np.zeros(n)
            grad_full[support] = grad_proj[support]
            
            # Line search
            step = 1.0 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.5)
            a_new = a - step * grad_full
            a_new = np.maximum(a_new, 0)
            a_new[~support] = 0
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        
        val_new = float(a_new @ _hankel_matvec(a_new, lam, n))
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-15:
            best_val = val_new
            best_a = a_new.copy()
        
        a = a_new
        support = a > 1e-12
        
        # Check convergence
        grad = _hankel_matvec(a, lam, n)
        sup = a > 1e-12
        if sup.any():
            lam_val = np.mean(grad[sup])
            resid = np.max(np.abs(grad[sup] - lam_val))
            if (~sup).any():
                resid = max(resid, np.max(lam_val - grad[~sup]))
        else:
            resid = np.inf
        if resid < 1e-10:
            break
    
    return best_a

def _dual_ascent(a_init, deadline, min_remaining=60.0, n=None):
    """
    Dual ascent: iteratively update lambda (weights on convolution positions),
    solve the resulting QP, and update lambda based on current c values.
    """
    if n is None:
        n = len(a_init)
    
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score_local(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score_local(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Initialize lambda: exponential weighting based on c values
    temperature = 0.01 * c_max
    lam = np.exp(np.clip((c - c_max) / temperature, -50, 0))
    lam = lam / lam.sum()
    
    K = min(50, 2*n-1)  # Start with small active set
    
    for it in range(500):
        if time.monotonic() + min_remaining > deadline:
            break
        
        # Grow active set
        if it % 5 == 0:
            K = min(K + max(10, n // 100), 2*n-1)
            K = min(K, 2*n-1)
        
        # Build active lambda: keep top-K c positions, weighted exponentially
        c_sorted_idx = np.argsort(c)[::-1]
        active_mask = np.zeros(2*n-1, dtype=bool)
        active_mask[c_sorted_idx[:K]] = True
        
        # Temperature annealing
        temperature = max(0.001 * c_max, temperature * 0.95)
        
        lam_new = np.zeros(2*n-1)
        lam_new[active_mask] = np.exp(np.clip((c[active_mask] - c_max) / temperature, -50, 0))
        lam_sum = lam_new.sum()
        if lam_sum < 1e-30:
            lam_new[active_mask] = 1.0 / K
        else:
            lam_new = lam_new / lam_sum
        lam = lam_new
        
        # Solve QP
        a_qp = _solve_qp_hankel(lam, n, a, max_outer=100, cg_max_iter=500)
        sc_qp = _score_local(a_qp)
        
        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
        
        a = a_qp
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Check improvement plateau
        if it > 50 and it % 20 == 0:
            # Try increasing K more aggressively
            K = min(K * 2, 2*n-1)
    
    return best_a, best_sc

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, aggressive=True):
    """L-BFGS with adaptive beta continuation on softmax parameterization."""
    n = len(a_init)
    a_init = np.maximum(a_init, 0)
    s = a_init.sum()
    if s < 1e-15:
        return a_init
    a_init = a_init / s
    
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    if aggressive:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625,
                  0.03, 0.015, 0.0075, 0.00375, 0.002,
                  0.001, 0.0005, 0.00025, 0.00012, 0.00006,
                  0.00003, 0.000015, 0.000007, 0.000003, 0.000001]
    else:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625,
                  0.03, 0.015, 0.0075, 0.004, 0.002,
                  0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                  0.00002, 0.00001]
    
    betas = [base_beta * r for r in ratios]
    
    current_a = a_init.copy()
    current_sc = _score_local(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()

        est_time_per_iter = 0.001 * max(1.0, n / 8192.0)

                current_sc = sc_new

        except Exception:
            pass
    
    return current_a

def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "bimodal":
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.05
    elif kind == "centered":
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.02
    elif kind == "flat_top":
        center = 0.5
        a = np.where(np.abs(t - center) < 0.3, 1.0, 
                     np.exp(-50 * (np.abs(t - center) - 0.3) ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _flatten_polish(a_init, deadline, max_iters=200):
    """Iteratively reduce the peak of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_local(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top few peaks
        k = min(10, 2*n-1)
        top_idx = np.argsort(c)[::-1][:k]
        
        # Construct a "target" that's flat at the top
        target = c_max - 0.01 * c_max  # Slightly below max
        excess = np.zeros_like(c)
        for idx in top_idx:
            if c[idx] > target:
                excess[idx] = c[idx] - target
        
        if excess.max() < 1e-15:
            break
        
        # Gradient of sum(excess^2) w.r.t. a
        a_rev = a[::-1]
        exc_conv = _full_convolve(excess, a_rev)
        grad = exc_conv[n-1:2*n-1]
        
        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = 0.5 / (1.0 + 0.05 * it)
        a_new = a - step * grad / grad_norm * c_max
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score_local(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            a_new = a - 0.1 * step * grad / grad_norm * c_max
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
    
    return best_a, best_sc
