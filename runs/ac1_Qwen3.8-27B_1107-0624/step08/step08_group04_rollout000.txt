from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


class _SmoParams:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def obj(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W_val = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W_val)


def _run_lbfgs_fine(a_init, deadline, min_time_remaining=30.0, n_levels=50):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)
    # Very fine geometric schedule: 50 levels from base_beta to base_beta * 1e-7
    total_reduction = 1e-7
    ratio = total_reduction ** (1.0 / n_levels)
    betas = base_beta * (ratio ** np.arange(n_levels))
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        # Estimate time per iteration
        est_per_iter = 0.003 * max(1.0, n / 4096.0)
        # More iterations at finer beta levels
        if i < n_levels // 3:
            max_iter = min(3000, max(100, int(remaining / (est_per_iter * 15))))
        elif i < 2 * n_levels // 3:
            max_iter = min(5000, max(200, int(remaining / (est_per_iter * 10))))
        else:
            max_iter = min(8000, max(500, int(remaining / (est_per_iter * 5))))
        params = _SmoParams(beta, n)
        try:
            result = minimize(
                params.obj,
                z,
                jac=params.grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 80, 'maxls': 400}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _cg_solve_precond(W, n, rhs, max_iter=1000, tol=1e-14):
    """Preconditioned CG solve for H*x = rhs where H is Hankel with entries 2*W[i+j]."""
    x = np.zeros(n)
    r = rhs.copy()
    # Diagonal preconditioner: M[i] = H[i,i] = 2*W[2i]
    diag = 2.0 * W[2 * np.arange(n)]
    diag = np.maximum(diag, 1e-30)
    z = r / diag
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if abs(rz0) < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rz / pHp
        x = x + alpha * p
        r_new = r - alpha * Hp
        z_new = r_new / diag
        rz_new = float(r_new @ z_new)
        if not np.isfinite(rz_new) or abs(rz_new) > abs(rz) * 1.5:
            break
        if abs(rz_new) < tol * tol * abs(rz0):
            r = r_new
            break
        p = z_new + (rz_new / rz) * p
        r = r_new
        rz = rz_new
    return x


def _qp_solve(W, n, a_init, max_outer=80, cg_iter=800):
    """Solve min a'H a / 2 s.t. sum(a)=1, a>=0 using active-set with precond CG."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) / 2.0)
    for outer in range(max_outer):
        support = a > 1e-10
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n
        if n_sup == n:
            y = _cg_solve_precond(W, n, np.ones(n), max_iter=cg_iter)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            # Projected gradient on the support
            grad = _hankel_matvec(a, W, n)
            lam = grad[support].mean()
            step = 2.0 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.5)
            # Line search
            alpha = step
            for ls in range(20):
                a_try = a.copy()
                a_try[support] -= alpha * (grad[support] - lam)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                val_try = float(a_try @ _hankel_matvec(a_try, W, n) / 2.0)
                if val_try < best_val:
                    best_val = val_try
                    best_a = a_try.copy()
                a_new = a_try
                break
            else:
                a_new = a
        val_new = float(a_new @ _hankel_matvec(a_new, W, n) / 2.0)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new
        # Convergence check
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-10
        if support.any():
            lam = grad[support].mean()
            resid = np.max(np.abs(grad[support] - lam))
            off_s = ~support
            if off_s.any():
                viol = np.max(lam - grad[off_s])
                resid = max(resid, max(0, viol))
        else:
            break
        scale = np.max(np.abs(grad)) + 1e-15
        if resid < 1e-12 * scale:
            break
    return best_a


def _active_set_polish(a_init, deadline, max_rounds=120, top_fraction_start=0.01, top_fraction_end=0.5):
    """Active-set QP with adaptive active set by RANK (top fraction of lags)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    num_lags = 2 * n - 1
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        # Adaptive top fraction: start small, grow over time
        frac = top_fraction_start + (top_fraction_end - top_fraction_start) * (rnd / max(1, max_rounds - 1))
        frac = min(frac, 0.9)
        n_active = max(1, int(num_lags * frac))
        # Get top n_active lags by value
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        W = np.zeros(num_lags)
        W[top_idx] = 1.0 / n_active
        # Solve QP
        a_qp = _qp_solve(W, n, a, max_outer=40, cg_iter=600)
        sc_qp = _score(a_qp)
        if sc_qp < best_sc - 1e-15:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Damped Newton step
            grad = _hankel_matvec(a, W, n)
            g_t = grad - np.mean(grad)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-14:
                Hg = _cg_solve_precond(W, n, grad, max_iter=400)
                H1 = _cg_solve_precond(W, n, np.ones(n), max_iter=400)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)
                alpha = 1.0
                for ls in range(50):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    c_max_try = float(c_try.max())
                    if c_max_try < c_max * (1 + 1e-12):
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        break
                    alpha *= 0.5
                    if alpha < 1e-16:
                        break
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _peak_target_polish(a_init, deadline, max_iters=500):
    """Directly target the top-K autocorrelation peaks for reduction."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    K = 10  # number of peaks to target
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        # Get top K lags
        top_k = np.argpartition(c, -K)[-K:]
        c_top = c[top_k]
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        # Weighted sum of top peaks: sum of c[top_k]^2 / sum(c[top_k])
        # Gradient of sum_k in top_k: c[k]^2 w.r.t. a[i]
        # dc[k]/da[i] done via deconvolution
        weights = np.zeros_like(c)
        weights[top_k] = 2.0 * c[top_k] / c_top.sum()
        # Gradient = 2 * a * (convolve(weights, a_reversed))[n-1:2n-1]
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(weights, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        # Adaptive step size with backtracking
        step = 0.1 / (1.0 + 0.005 * it)
        for ls in range(30):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                break
            step *= 0.5
            if step < 1e-14:
                break
    return best_a, best_sc


def _fista_smooth(a_init, deadline, max_iters=2000):
    """FISTA (accelerated gradient) on the log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()
    t = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)
    # Adaptive beta: start moderate, decrease over time
    c0 = _full_convolve(a_init, a_init)
    beta0 = float(c0.max()) * 0.001
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        beta = beta0 / (1.0 + 0.01 * it)
        params = _SmoParams(beta, n)
        a = _softmax(y)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        # Step size: 1/L where L is Lipschitz constant (estimate)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = 4.0 / (n * (np.max(c) + 1e-15) + 1e-15)
        step = min(step, 0.5 / (grad_norm + 1e-15))
        z_new = y - step * grad
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = z_new + ((t - 1) / t_new) * (z_new - z)
        z = z_new
        t = t_new
        # Check improvement every 50 iterations
        if it % 50 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-14:
                best_sc = sc_check
                best_a = a_check.copy()
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** 2 + 0.01
        return a / a.sum()
    elif kind == "oscillatory":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(8 * np.pi * t) * np.sin(np.pi * t)
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _cosine_multiply(a, freq, phase, amount):
    """Multiply by (1 + amount * cos(2*pi*freq*t + phase)) and renormalize."""
    n = len(a)
    t = np.linspace(0, 1, n)
    mod = 1.0 + amount * np.cos(2 * np.pi * freq * t + phase)
    mod = np.maximum(mod, 0.01)
    a_new = a * mod
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with fine beta schedule,
    adaptive active-set QP, peak-targeting polish, FISTA,
    multi-n exploration, and diverse structured restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS with very fine beta schedule
    print(f"\n[Phase 1] Aggressive L-BFGS fine (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs_fine(parent, deadline, min_time_remaining=600.0, n_levels=50)
    update_best(a1, _score(a1), "lbfgs_fine")

    # Phase 2: Active-set QP with adaptive top-fraction
    print(f"\n[Phase 2] Active-set QP adaptive (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a2, sc2 = _active_set_polish(best_seq, min(deadline - 350, time.monotonic() + 200),
                                     max_rounds=60, top_fraction_start=0.005, top_fraction_end=0.3)
        update_best(a2, sc2, "active_qp_1")

    # Phase 3: Peak-targeting polish
    print(f"\n[Phase 3] Peak-target (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a3, sc3 = _peak_target_polish(best_seq, min(deadline - 250, time.monotonic() + 100), max_iters=300)
        update_best(a3, sc3, "peak_target_1")

    # Phase 4: FISTA smooth
    print(f"\n[Phase 4] FISTA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a4, sc4 = _fista_smooth(best_seq, min(deadline - 200, time.monotonic() + 100), max_iters=1500)
        update_best(a4, sc4, "fista_1")

    # Phase 5: Second round of active-set QP (tighter)
    print(f"\n[Phase 5] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a5, sc5 = _active_set_polish(best_seq, min(deadline - 180, time.monotonic() + 100),
                                     max_rounds=80, top_fraction_start=0.002, top_fraction_end=0.5)
        update_best(a5, sc5, "active_qp_2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 8192, 32768]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        n_try = min(n_try, 32768)
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        print(f"  Trying n={n_try}...")
        a_opt = _run_lbfgs_fine(a_init, deadline, min_time_remaining=80.0, n_levels=30)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 60:
            a_np, sc_np = _active_set_polish(a_opt,
                                             min(deadline - 50, time.monotonic() + 60),
                                             max_rounds=40, top_fraction_start=0.005, top_fraction_end=0.3)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 7: Diverse structured restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_r = len(best_seq)
        restart_configs = [
            ("osc_mult_1", lambda: _cosine_multiply(best_seq, 1, 0, 0.5)),
            ("osc_mult_3", lambda: _cosine_multiply(best_seq, 3, 1.0, 0.3)),
            ("osc_mult_5", lambda: _cosine_multiply(best_seq, 5, 0.5, 0.4)),
            ("osc_mult_7", lambda: _cosine_multiply(best_seq, 7, 2.0, 0.3)),
            ("perturb_small", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_med", lambda: _perturb(best_seq, 0.1, rng)),
            ("shrink_top", _make_shrink_top_func(best_seq, 0.2)),
            ("bimodal_mix", _make_mix_func(best_seq, "bimodal", 0.5)),
        ]
        for idx, (label, func) in enumerate(restart_configs):
            if time.monotonic() > deadline - 20:
                break
            try:
                p_start = func()
                if p_start is None:
                    continue
                a_r = _run_lbfgs_fine(p_start, deadline, min_time_remaining=20.0, n_levels=20)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{label}")
                if sc_r < best_score + 0.001 and time.monotonic() < deadline - 15:
                    a_rp, sc_rp = _peak_target_polish(a_r,
                                                      min(deadline - 10, time.monotonic() + 20),
                                                      max_iters=100)
                    update_best(a_rp, sc_rp, f"restart_{label}_pt")
            except Exception:
                pass

    # Phase 8: Final ultra-fine optimization
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 2,
                                           max_rounds=80, top_fraction_start=0.001, top_fraction_end=0.8)
        update_best(a_fin, sc_fin, "final_qp")

        if time.monotonic() < deadline - 5:
            a_fl, sc_fl = _peak_target_polish(best_seq, deadline - 2, max_iters=200)
            update_best(a_fl, sc_fl, "final_peak")

        # Final L-BFGS with ultra-small beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            remaining = deadline - 3 - time.monotonic()
            if remaining > 3:
                # Ultra-fine: 10 levels from mb*1e-5 to mb*1e-8
                base = mb * 1e-5
                ratio = (1e-3) ** (1.0 / 10)
                betas_ultra = base * (ratio ** np.arange(10))
                current_a = best_seq.copy()
                current_sc = best_score
                for beta_u in betas_ultra:
                    if time.monotonic() > deadline - 2:
                        break
                    est_per_iter = 0.003 * max(1.0, n / 4096.0)
                    max_iter = min(3000, max(100, int((deadline - 2 - time.monotonic()) / (est_per_iter * 5))))
                    params_u = _SmoParams(beta_u, n)
                    try:
                        result = minimize(
                            params_u.obj,
                            z,
                            jac=params_u.grad,
                            method='L-BFGS-B',
                            options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                                     'maxcor': 80, 'maxls': 400}
                        )
                        a_u = _softmax(result.x)
                        sc_u = _score(a_u)
                        if sc_u < current_sc - 1e-16:
                            current_a = a_u
                            current_sc = sc_u
                            z = np.log(np.maximum(a_u, 1e-30))
                            z = z - z.mean()
                    except Exception:
                        pass
                update_best(current_a, current_sc, "final_ultra_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]


def _make_shrink_top_func(base_a, fraction):
    """Create a function that shrinks top elements of base_a."""
    def func():
        a = base_a.copy()
        idx = np.argsort(a)[::-1]
        n_shrink = max(1, int(len(a) * fraction))
        a[idx[:n_shrink]] *= 0.5
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return base_a.copy()
        return a / s
    return func


def _make_mix_func(base_a, profile_kind, weight):
    """Create a function that mixes base_a with a profile."""
    def func():
        n = len(base_a)
        prof = _make_profile(n, profile_kind)
        a = (1 - weight) * base_a + weight * prof
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return base_a.copy()
        return a / s
    return func
