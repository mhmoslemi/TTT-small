    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_a = a_final.copy()
        best_score = sc
    return best_a, best_score

def _lp_direction(a, M, max_vars=8192):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Returns the normalized direction g/sum(g)."""
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[j] if 0 <= k-j < n
    # A is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if a[j] < 1e-15:
            continue
        # a[j] appears in row k for k = j, j+1, ..., j+n-1 (clipped to [0, 2n-2])
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    if len(rows) == 0:
        return a.copy()
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) → minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M, g >= 0
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                                         method='highs', options={'time_limit': 60})
        if result.success and result.x is not None:
            g = np.maximum(0, result.x)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception as e:
        print(f"  LP error: {e}")
    
    return a.copy()

def _equalize_peaks(a, deadline, max_iter=500):
    """Try to equalize the top peaks by solving a linear system."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks (top few close to max)
        tol = c_max * 0.9995
        active = np.where(c >= tol)[0]
        if len(active) < 2:
            break
        if len(active) > 12:
            sorted_a = active[np.argsort(c[active])[::-1][:12]]
            active = sorted_a
        K = len(active)
        
        # Build M matrix: M[i,j] = dc[active[i]]/da[j] = 2*a[active[i]-j] if valid
        M_mat = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M_mat[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # We want: M_mat @ d = -1 (reduce all active peaks by same amount)
        # subject to: sum(d * a) = 0 (preserve normalization to first order)
        # Minimum norm solution via KKT:
        # [M  0] [d]   [-1]
        # [w  0][d] = [ 0]  where w = ??? 
        # Actually: min ||d||^2 s.t. M@d = -1, a^T@d = 0
        # KKT: [2I  M^T  a][d]   [0]
        #      [M    0    0][λ] = [-1]
        #      [a^T  0    0][μ]   [0]
        # Simplify: d = -1/2 (M^T μ_vec + a * μ_scalar)... 
        # Actually: 2d + M^T λ + a * μ = 0, M d = -1, a^T d = 0
        # d = -1/2 (M^T λ + a μ)
        # M @ d = -1/2 (M M^T λ + M a μ) = -1
        # a^T d = -1/2 (a^T M^T λ + a^T a μ) = 0
        # Let G11 = M M^T (K x K), G12 = M a (K x 1), G22 = a^T a (scalar)
        # -1/2 (G11 λ + G12 μ) = -1  → G11 λ + G12 μ = 2
        # -1/2 (G12^T λ + G22 μ) = 0  → G12^T λ + G22 μ = 0
        # [G11  G12][λ]   [2]
        # [G12^T G22][μ] = [0]
        
        G11 = M_mat @ M_mat.T
        G12 = M_mat @ a  # K x 1
        G22 = float(a @ a)  # scalar
        
        sys_mat = np.zeros((K+1, K+1))
        sys_mat[:K, :K] = G11
        sys_mat[:K, K] = G12
        sys_mat[K, :K] = G12
        sys_mat[K, K] = G22
        rhs = np.concatenate([2*np.ones(K), [0.0]])
        
        try:
            sol = np.linalg.solve(sys_mat + 1e-10 * np.eye(K+1), rhs)
        except np.linalg.LinAlgError:
            break
        lam = sol[:K]
        mu = sol[K]
        d = -0.5 * (M_mat.T @ lam + a * mu)
        
        # Step
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score

def _subgradient_polish(a, deadline, max_iter=10000, top_k=5, init_lr=0.002):
    """Direct subgradient on the normalized sequence."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = init_lr
    best_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        # Include only those close to max
        active = [i for i in active if c[i] > c_max * 0.999]
        if not active:
            active = [sorted_idx[0]]
        
        # Subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space (preserve sum)
        g = g - np.dot(g, a) * a  # approximate projection
        g = g - g.mean()  # also subtract mean
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        step = lr * g / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, best_lr)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    
    return best_a, _score_norm(best_a)
