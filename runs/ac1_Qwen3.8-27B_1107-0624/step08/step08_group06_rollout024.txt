- Or use backtracking: start with alpha = 1 and halve until the objective decreases

"""
Optimizer using active-set linear system solving with FFT-based matvec,
combined with log-sum-exp polishing and multi-n exploration.

Key ideas:
1. Active-set "flatness" solver: iteratively make the correlation with the active set constant
2. Adaptive active set expansion
3. Log-sum-exp + L-BFGS for polishing
4. Multi-n exploration (16384, 32768)
5. Spectral/block initialization

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _matvec_M(v, S_indicator, n, scale):
    """Compute M*v where M[i,j] = scale * S_indicator[i+j]"""
    flip_v = v[::-1]
    out_len = n + len(S_indicator) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(flip_v, N)
    fS = np.fft.rfft(S_indicator, N)
    result = np.fft.irfft(fv * fS, N)[:out_len]
    return result[n - 1:2 * n - 1] * scale


def _flatness_solve(a_init, S_indicator, n, max_iter=500, tol=1e-10, deadline=None):
    """
    Solve for a such that M*a is constant (on the support of a)
    and sum(a) = 1, a >= 0.
    M[i,j] = (2/|S|) * S_indicator[i+j]
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, 0.0
    a = a / s
    
    m = int(S_indicator.sum())
    if m < 1:
        return a, 0.0
    scale = 2.0 / m
    
    best_a = a.copy()
    best_obj = np.inf
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute h = M*a
        h = _matvec_M(a, S_indicator, n, scale)
        
        # Compute deviation from constant
        d = h - np.mean(h)
        obj = float(np.sum(d * d))
        
        if obj < tol:
            break
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Gradient of ||d||^2/2 is M^T * d = M * d (M is symmetric)
        g = _matvec_M(d, S_indicator, n, scale)
        
        # Project gradient to maintain sum(a) = 1
        g_proj = g - np.mean(g)
        
        # Backtracking line search
        alpha = 1.0
        accepted = False
        for _ls in range(30):
            a_try = a - alpha * g_proj
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            h_try = _matvec_M(a_try, S_indicator, n, scale)
            d_try = h_try - np.mean(h_try)
            obj_try = float(np.sum(d_try * d_try))
            if obj_try < obj:
                a = a_try
                accepted = True
                break
            alpha *= 0.5
        
        if not accepted:
            break
    
    # Final check
    h = _matvec_M(a, S_indicator, n, scale)
    d = h - np.mean(h)
    return a, float(np.sum(d * d))


def _active_set_optimize(a_init, deadline, max_as_iter=50, tol_expand=1e-6):
    """
    Active-set optimization:
    1. Identify active set from convolution peaks
    2. Solve flatness problem
    3. Check for constraint violations
    4. Expand active set if needed
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for as_iter in range(max_as_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution and identify active set
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: positions where c[k] >= c_max * (1 - tol)
        # Start with a small tolerance, expand if needed
        tol = max(0.001, c_max * tol_expand)
        active = c >= (c_max - tol)
        
        # Ensure at least a few active positions
        if active.sum() < 3:
            top_idx = np.argsort(c)[::-1][:max(3, n // 100)]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
        
        # Limit active set size
        if active.sum() > n:
            top_idx = np.argsort(c)[::-1][:n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
        
        S_indicator = active.astype(np.float64)
        
        # Solve flatness problem
        a, flat_obj = _flatness_solve(a, S_indicator, n, max_iter=300,
                                       tol=1e-12, deadline=deadline)
        
        # Check result
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        
        # Check for constraint violations: is c[k] > c_max for k not in active?
        c_new = _full_convolve(a, a)
        c_new_max = float(c_new.max())
        
        # Find violations

        if new_active.sum() == 0:
            # No new violations, check if we've converged
            break
        
        # Expand active set
        active = active | new_active
        
        #更新 S_indicator for next iteration
        # (will be recomputed at the top of the loop)
    
    return best_a, best_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)

        return a_init

    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 6))))

        try:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

        except Exception:
            pass
