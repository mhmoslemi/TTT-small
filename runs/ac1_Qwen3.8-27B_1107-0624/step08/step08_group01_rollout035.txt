def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)

def _resample(a, new_n):
    old_n = len(a)

    x_new = np.linspace(0, 1, new_n)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _coordinate_descent(a_init, deadline, n_sweeps=50):
    """Exact coordinate descent: for each coordinate, find optimal value via bisection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 1.0:
            break

        improved = False
        # Process coordinates in order of contribution to the peak
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        # Sort by contribution to the peak
        contrib = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                contrib[j] = a[m]
            else:
                contrib[j] = 0.0

        order = np.argsort(contrib)[::-1]

        for j in order[:min(200, n)]:  # Only update top contributors
            if time.monotonic() > deadline - 1.0:
                break

            a_old = a[j]
            if a_old < 1e-15:
                continue

            # Compute base convolution (without a[j])
            # b[k] = base[k] + 2*a[j]*a[k-j] (for k != 2j)
            # b[2j] = base[2j] + a[j]^2
            # base[k] = c[k] - 2*a[j]*a[k-j] (adjusting for k=2j)

            # Find range of k affected by a[j]
            k_lo = max(0, j)  # k-j >= 0 means k >= j
            k_hi = min(2 * n - 2, j + n - 1)  # k-j < n means k < j+n

            affected = np.arange(k_lo, k_hi + 1)
            m_vals = affected - j  # = k - j, these index into a

            # base[k] for affected k
            base_affected = c[affected].copy()
            for idx, k_val in enumerate(affected):
                if k_val == 2 * j:
                    base_affected[idx] = c[k_val] - a_old * a_old
                else:
                    base_affected[idx] = c[k_val] - 2.0 * a_old * a[m_vals[idx]]

            # Unaffected k values
            unmask = np.ones(2 * n - 1, dtype=bool)
            unmask[affected] = False
            base_unaffected_max = float(c[unmask].max()) if unmask.any() else 0.0

            # Minimize max(base_affected + 2*x*a[m] or base_{2j} + x^2, base_unaffected_max)
            # over x >= 0
            # This is a convex function of x. Use bisection on the subgradient.

            # Find the optimal x by bisection
            # f(x) = max(base_unaffected_max, max_k (base_affected[k] + coeff_k * x))
            # where coeff_k = 2*a[m] for k != 2j, coeff_{2j} = x (quadratic)

            # Handle quadratic term at k=2j
            has_quad = (2 * j >= k_lo) and (2 * j <= k_hi)
            quad_idx = np.where(affected == 2 * j)[0]
            quad_base = base_affected[quad_idx[0]] if len(quad_idx) > 0 else 0.0

            # Linear coefficients
            lin_coeffs = 2.0 * a[m_vals].copy()
            if len(quad_idx) > 0:
                lin_coeffs[quad_idx] = 0.0  # handle separately

            # f(x) = max(base_unaffected, quad_base + x^2, base_lin + lin_coeff * x)
            # For small x: f(x) = max(base_unaffected, quad_base, max_lin)
            # We want the minimum of this convex function.

            # Simple approach: golden section search in [0, upper_bound]
            upper = max(a_old * 3.0, 1.0 / n * 5.0)

            def f_1d(x):
                if x < 0:
                    return np.inf
                val = base_unaffected_max
                if has_quad:
                    val = max(val, quad_base + x * x)
                lin_vals = base_affected + lin_coeffs * x
                val = max(val, float(lin_vals.max()))
                return val

            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            lo, hi = 0.0, upper
            a_g = hi - gr * (hi - lo)
            b_g = lo + gr * (hi - lo)
            fa = f_1d(a_g)
            fb = f_1d(b_g)

            for _ in range(40):
                if fa < fb:
                    hi = b_g
                    b_g = a_g
                    fb = fa
                    a_g = hi - gr * (hi - lo)
                    fa = f_1d(a_g)
                else:
                    lo = a_g
                    a_g = b_g
                    fa = fb
                    b_g = lo + gr * (hi - lo)
                    fb = f_1d(b_g)

            x_opt = (lo + hi) / 2.0
            a[j] = x_opt

            # Update c (only affected entries)
            for idx, k_val in enumerate(affected):
                m = m_vals[idx]
                if k_val == 2 * j:
                    c[k_val] = base_affected[idx] + x_opt * x_opt
                else:
                    c[k_val] = base_affected[idx] + 2.0 * x_opt * a[m]

            new_sc = 2.0 * n * float(c.max())
            if new_sc < best_sc - 1e-14:
                best_sc = new_sc
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_start=0.003, tol_end=0.00001,
                       lr_start=0.02, top_k=5):
    """Active-set projected gradient descent with backtracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0

    while time.monotonic() < deadline - 1.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        cur_sc = 2.0 * n * c_max

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0

        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 10:
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros(len(c), dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False

        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < cur_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= 400:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.5
                no_improve = 0
                stall += 1
                if stall > 15:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)
            lr = min(lr_start, lr * 1.005)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining Nesterov mirror descent, L-BFGS,
    coordinate descent, and multi-n exploration for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(best_a)
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={best_score:.10f}")
    else:
        n_parent = 12288
        best_a = _make_profile(n_parent, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS initial optimization ===
    print(f"\n[P1] L-BFGS initial (t={time.monotonic()-t_start:.1f}s)")
    p1_end = min(deadline - 800.0, t_start + 250.0)
    a1 = _lbfgs_full(best_a, p1_end, min_remaining=20.0)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_a = a1.copy()
        best_score = sc1
        print(f"  L-BFGS: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  L-BFGS no improvement: {sc1:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Nesterov Mirror Descent (main phase) ===
    print(f"\n[P2] Nesterov MD (t={time.monotonic()-t_start:.1f}s)")
    p2_end = min(deadline - 500.0, t_start + 750.0)
    a2, sc2 = _nesterov_mirror_descent(best_a, p2_end, K_start=100, K_end=1,
                                        eta_init=0.3, momentum_init=0.85,
                                        momentum_end=0.97, stall_limit=600)
    if sc2 < best_score:
        best_a = a2.copy()
        best_score = sc2
        print(f"  Nesterov MD: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Nesterov MD no improvement: {sc2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS polish after Nesterov ===
    print(f"\n[P3] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p3_end = min(deadline - 400.0, t_start + 1000.0)
    a3 = _lbfgs_full(best_a, p3_end, min_remaining=15.0)
    sc3 = _score(a3)
    if sc3 < best_score:
        best_a = a3.copy()
        best_score = sc3
        print(f"  L-BFGS polish: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Active-set polish ===
    print(f"\n[P4] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p4_end = min(deadline - 300.0, t_start + 1200.0)
    a4, sc4 = _active_set_polish(best_a, p4_end, tol_start=0.002, tol_end=0.00001,
                                  lr_start=0.015, top_k=3)
    if sc4 < best_score:
        best_a = a4.copy()
        best_score = sc4
        print(f"  Active-set: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Coordinate descent ===
    print(f"\n[P5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    p5_end = min(deadline - 200.0, t_start + 1400.0)
    a5, sc5 = _coordinate_descent(best_a, p5_end, n_sweeps=30)
    if sc5 < best_score:
        best_a = a5.copy()
        best_score = sc5
        print(f"  Coord descent: {best_score:.10f} (n={len(best_a)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 24576, 8192]
    if n_parent not in n_targets:
        n_targets.append(n_parent)

    for n_try in sorted(n_targets, key=lambda x: abs(x - len(best_a))):
        if time.monotonic() > deadline - 150.0:
            break
        if n_try == len(best_a):
            continue

        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        remaining = deadline - time.monotonic() - 50.0
        if remaining < 60:
            continue

        # Quick Nesterov + L-BFGS
        sub_deadline = time.monotonic() + min(remaining * 0.4, 200.0)
        a_opt, sc_opt = _nesterov_mirror_descent(a_init, sub_deadline,
                                                  K_start=50, K_end=1,
                                                  eta_init=0.3, stall_limit=300)
        if time.monotonic() < deadline - 40.0:
            sub_deadline2 = min(deadline - 30.0, time.monotonic() + min(remaining * 0.3, 150.0))
            a_opt = _lbfgs_full(a_opt, sub_deadline2, min_remaining=10.0)
            sc_opt = _score(a_opt)

        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100.0:
        n_best = len(best_a)
        restarts = [
            ("pert_small", _perturb_log(best_a, 0.02, rng)),
            ("pert_med", _perturb_log(best_a, 0.05, rng)),
            ("cos_mod", _make_profile(n_best, "fourier_rich")),
            ("tapered", _make_profile(n_best, "tapered")),
        ]

        for label, a_r in restarts:
            if time.monotonic() > deadline - 30.0:
                break
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 20.0:
                break

            # Quick Nesterov
            sub_dl = time.monotonic() + min(remaining * 0.5, 60.0)
            a_opt, sc_opt = _nesterov_mirror_descent(a_r, sub_dl,
                                                      K_start=30, K_end=1,
                                                      eta_init=0.3, stall_limit=200)
            # Quick L-BFGS
            if time.monotonic() < deadline - 15.0:
                sub_dl2 = min(deadline - 10.0, time.monotonic() + 30.0)
                a_opt = _lbfgs_quick(a_opt, sub_dl2, max_iter=100)
                sc_opt = _score(a_opt)

            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_polish(best_a, deadline - 2.0,
                                            tol_start=0.0003, tol_end=0.000002,
                                            lr_start=0.005, top_k=2)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  Final active-set: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

        # Last L-BFGS
        remaining = deadline - 2.0 - time.monotonic()
        if remaining > 5.0:
            a_fin2 = _lbfgs_quick(best_a, deadline - 1.0, max_iter=300)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_a = a_fin2.copy()
                best_score = sc_fin2
                print(f"  Final L-BFGS: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
