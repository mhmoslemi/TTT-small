Step-function optimization using multi-scale refinement, peak-directed

convergence below 1.5027.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    c = _full_convolve(a, a)
    c_topk = np.sort(c)[::-1][:k]

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

def _peak_grad_descent(a, deadline, top_k=5, lr_init=0.05, max_iter=100000):
    """Peak-directed gradient descent with top-k weighting."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    no_improve = 0
    max_no_improve = 5000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

        # Select top-k peaks
        k_sel = min(top_k, 2 * n - 1)
        # Use a threshold to include all peaks within 1% of max
        threshold = c_max * 0.99
        active_mask = c >= threshold
        if active_mask.sum() > 50:
            # Fall back to top-k
            top_indices = np.argsort(c)[::-1][:k_sel]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[top_indices] = True

        active_indices = np.where(active_mask)[0]
        n_active = len(active_indices)
        if n_active == 0:
            active_indices = np.array([int(np.argmax(c))])
            n_active = 1

        # Compute gradient: for each active peak k, gradient contribution to a[j] is
        # a[k-j] / n_active for each active k
        grad = np.zeros(n)
        for k in active_indices:
            j_vals = np.arange(max(0, n - k), min(n, k + 1))
            # c[k] = sum_{j} a[j]*a[k-j], dc[k]/da[j] = a[k-j] (for valid k-j)
            # Actually dc[k]/da[j] = a[k-j] if 0 <= k-j < n
            # j ranges over [max(0, k-n+1), min(n-1, k)]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += a[k - j_hi:k - j_lo + 1][::-1] / n_active

        # Center the gradient to preserve sum
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        alpha = lr / gnorm
        improved = False
        for _ in range(20):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.4

        if not improved:
            lr *= 0.7
            no_improve += 10
            if lr < 1e-10:
                break
        else:
            lr = min(lr_init, lr * 1.02)

    return best_a, best_sc

            a_opt, sc = _peak_grad_descent(a_opt, level_deadline, top_k=3,

            print(f"  Level n={n_level} after peakGrad: score={sc:.8f}")

        current_a = a_opt
        current_sc = sc

    return current_a, current_sc


def _adaptive_peak_shave(a, deadline, n_trials=2000):
    """Adaptive peak shaving with contribution analysis."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find top contributors to the peak
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo > j_hi:
            continue

        i_range = np.arange(j_lo, j_hi + 1)
        contrib = best_a[i_range] * best_a[k_star - i_range]
        total = contrib.sum()
        if total < 1e-15:
            continue

        # Reduce top contributors, distribute to low-contribution elements
        fraction = rng.uniform(0.05, 0.2)
        n_reduce = max(1, int(len(contrib) * fraction))
        top_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = best_a.copy()
        reduction = 0.0
        for idx in top_idx:
            red = a_try[idx] * rng.uniform(0.05, 0.3)
            a_try[idx] -= red
            reduction += red

        # Distribute to elements that DON'T contribute to this peak
        mask = np.ones(n, dtype=bool)
        for idx in top_idx:
            mask[idx] = False
            j_comp = k_star - idx
            if 0 <= j_comp < n:
                mask[j_comp] = False

        n_free = mask.sum()
        if n_free == 0:
            continue

        a_try[mask] += reduction / n_free
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try

    return best_a, best_sc


def _cosine_param_optimize(n, deadline, n_harmonics=20):
    """Optimize a cosine-parametrized sequence."""
    # a[i] = (1 + sum_k c_k * cos(2*pi*k*i/n)) normalized, with constraint a >= 0
    # Optimize c = (c_1, ..., c_K) to minimize score

    def _obj(params):
        c_coefs = params
        t = np.arange(n)
        a = np.ones(n)
        for k in range(1, len(c_coefs) + 1):
            a += c_coefs[k - 1] * np.cos(2 * np.pi * k * t / n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return 1e10
        a = a / s
        return _score(a)

    def _get_a(params):
        c_coefs = params
        t = np.arange(n)
        a = np.ones(n)
        for k in range(1, len(c_coefs) + 1):
            a += c_coefs[k - 1] * np.cos(2 * np.pi * k * t / n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s

    K = min(n_harmonics, n // 4)
    x0 = np.zeros(K)
    bounds = [(-0.9, 0.9)] * K

    est_time = 0.01 * max(1, n / 1024)
    max_iter = min(500, max(10, int((deadline - time.monotonic()) / (est_time * 10))))

    try:
        result = minimize(_obj, x0, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-15})
        a = _get_a(result.x)
        sc = _score(a)
        return a, sc
    except Exception:
        return _normalize(np.ones(n)), 2.0


def _diverse_init(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform
    inits.append((_normalize(np.ones(n)), "uniform"))
    # Tapered (raised cosine squared)
    t = np.linspace(0, 1, n)
    a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** 0.5
    inits.append((_normalize(a), "cos_taper"))
    # Inverse taper
    a = 1.0 / (0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append((_normalize(a), "inv_cos"))
    # Gaussian bump
    a = np.exp(-5 * (t - 0.5) ** 2)
    inits.append((_normalize(a), "gaussian"))
    # Bimodal
    a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2) + 0.01
    inits.append((_normalize(a), "bimodal"))
    # Random smooth
    z = rng.standard_normal(n)
    a = np.fft.irfft(np.abs(np.fft.rfft(z)) * 0.1, n)
    a = np.abs(a) + 0.1
    inits.append((_normalize(a), "rand_smooth"))
    # Perturbed uniform
    a = 1.0 + 0.3 * rng.standard_normal(n)
    inits.append((_normalize(np.maximum(a, 0.01)), "pert_uniform"))
    return inits


def _fine_polish(a, deadline):
    """Ultra-fine polishing with very small LSE beta and peak gradient."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    # Very small LSE
    c = _full_convolve(best_a, best_a)
    mb = float(c.max())
    beta = mb * 0.000001
    z = np.log(np.maximum(best_a, 1e-30))
    z = z - z.mean()

    remaining = deadline - time.monotonic()
    max_iter = min(5000, max(10, int(remaining / (0.003 * max(1, n / 4096)))))

    prob = _LseProblem(beta, n)
    try:
        result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxcor': 50})
        a_new = _softmax(result.x)
        a_new = _normalize(a_new)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_a = a_new
            best_sc = sc_new
    except Exception:
        pass

    # Peak gradient with very small lr
    if time.monotonic() < deadline - 1:
        best_a, best_sc = _peak_grad_descent(best_a, deadline - 0.5,
                                             top_k=3, lr_init=0.005, max_iter=20000)

    # Peak shave
    if time.monotonic() < deadline - 1:
        best_a, best_sc = _adaptive_peak_shave(best_a, deadline - 0.5, n_trials=200)

    return best_a, best_sc
