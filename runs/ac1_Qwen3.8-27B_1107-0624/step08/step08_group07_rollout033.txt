import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max_conv."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _subgradient(a, c, top_k=1):
    """Compute subgradient of max_k c[k] w.r.t. a.
    Average over top_k lags."""
    n = len(a)
    sorted_idx = np.argsort(c)[::-1]
    # Find lags that are "active" (close to max)
    c_max = c[sorted_idx[0]]
    active = [sorted_idx[0]]
    for idx in sorted_idx[1:top_k]:
        if c[idx] >= c_max * 0.9999:
            active.append(idx)
        else:
            break

        hi = min(k, n - 1)
        if lo <= hi:

"""
Convex optimization approach to minimize max autocorrelation peak of non-negative sequences.
Key innovations over previous approach:
1. Remez-exchange active-set method for fast convergence near optimum
2. Nesterov-accelerated Adam with better temperature scheduling
3. Multi-strategy phase allocation focused on final refinement
4. Exploits convexity for guaranteed global optimality
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_subgradient(a, c, top_k=5):
    """Compute averaged subgradient over top-k active lags."""
    n = len(a)
    sorted_idx = np.argsort(c)[::-1]
    c_max = c[sorted_idx[0]]
    active = [sorted_idx[0]]
    for idx in sorted_idx[1:top_k]:
        if c[idx] >= c_max * 0.999:
            active.append(idx)
        else:
            break
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    return g


def _softmax_from_z(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_optimize(a_init, deadline, T_init, T_final, lr_init, lr_final,
                   max_iter=200000, top_k=5, patience_max=60):
    """Adam in log-space with polynomial cooling and patience-based restart."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_scale = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_init * (T_final / T_init) ** (frac ** 0.6)
        lr = lr_init * lr_scale * (lr_final / lr_init) ** (frac ** 0.7)

        a = _softmax_from_z(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Softmax-weighted gradient
        c_shifted = (c - c_max) / T
        e = np.exp(c_shifted)
        w_sum = e.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / w_sum

        g = np.zeros(n)
        top_lags = np.argsort(c)[::-1][:min(top_k + 10, len(c))]
        for k in top_lags:
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += w[k] * 2.0 * a[k - hi:k - lo + 1][::-1]

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 1.0:
            update /= un
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if t % 50 == 0:
            a_cur = _softmax_from_z(z)
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_max:
                lr_scale *= 0.75
                patience = 0
                if lr_scale < 0.02:
                    break

    a_final = _softmax_from_z(z)
    if _score_norm(a_final) < best_score:
        best_a = a_final.copy()
        best_score = _score_norm(best_a)
    return best_a, best_score


def _remez_exchange(a_init, deadline, max_iter=5000, n_active=5):
    """
    Remez-exchange style active-set method.
    At each iteration: identify peak lags, solve linearized equalization,
    take a step, update the active set. Exploits convexity for fast convergence.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Identify active peaks
        active = np.where(c >= c_max * 0.998)[0]
        if len(active) < 2:
            active = np.argsort(c)[::-1][:n_active]
        if len(active) > n_active:
            active = active[np.argsort(c[active])[::-1][:n_active]]

        m = len(active)

        # Build the matrix M where M[i,j] = a[active[i] - j] if valid
        M = np.zeros((m, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        # Solve: find d minimizing ||d||^2 subject to:
        #   2 * M * d = -1 (all peaks decrease at same rate)
        #   1^T * d = 0 (preserve sum)
        A = np.vstack([2.0 * M, np.ones(n)])
        b_vec = np.concatenate([-np.ones(m), [0.0]])

        # Minimum norm solution: d = A^T (A A^T)^{-1} b
        G = A @ A.T  # (m+1) x (m+1)
        try:
            G_reg = G + 1e-10 * np.eye(G.shape[0])
            G_inv = np.linalg.inv(G_reg)
            d = A.T @ (G_inv @ b_vec)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(A, b_vec, rcond=None)[0]

        if not np.all(np.isfinite(d)):
            break

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break

        # Find max step before hitting zero
        neg_mask = d < -1e-15
        if np.any(neg_mask):
            alpha_max = float(np.min(a[neg_mask] / (-d[neg_mask])))
        else:
            alpha_max = 1.0

        alpha_max = min(alpha_max, 0.5)

        # Line search
        improved = False
        for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            alpha = alpha_max * frac
            a_cand = a + alpha * d
            a_cand = np.maximum(0.0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            # Try FW direction as fallback
            g = _compute_subgradient(a, c, top_k=m)
            j_min = int(np.argmin(g))
            for t_f in [0.05, 0.02, 0.01, 0.005, 0.001]:
                a_cand = a.copy()
                a_cand = (1 - t_f) * a_cand
                a_cand[j_min] += t_f
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                sc = _score_norm(a_cand)
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    improved = True
                    break

        if not improved:
            # Try gradient direction
            g = _compute_subgradient(a, c, top_k=m)
            g_tan = g - float(np.sum(a * g))
            gn = np.linalg.norm(g_tan)
            if gn < 1e-15:
                break
            g_tan /= gn
            for t_f in [0.005, 0.002, 0.001, 0.0005]:
                a_cand = a - t_f * g_tan
                a_cand = np.maximum(0.0, a_cand)
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                sc = _score_norm(a_cand)
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand
                    break

            # If still no improvement, break
            if _score_norm(a) >= best_score * (1 - 1e-12):
                c_check = _fft_convolve(best_a, best_a)
                if c_check.max() - c_check[c_check.argmax()] > 0:
                    # Check if we're stuck
                    pass
                break

    return best_a, best_score

def _multi_phase_adam(a_init, deadline, n_phases=4):
    """Run Adam at multiple temperature levels in sequence."""
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    time_per = max(30.0, (deadline - time.monotonic()) / n_phases)

    for i in range(n_phases):
        sub_dl = min(deadline, time.monotonic() + time_per)
        if time.monotonic() > sub_dl:
            break
        T_i = max_c * (0.05 * (0.3 ** i))
        lr_i = 0.04 * (0.6 ** i)
        a_opt, sc = _adam_optimize(
            best_a, sub_dl,
            T_init=T_i,
            T_final=T_i * 0.05,
            lr_init=lr_i,
            lr_final=lr_i * 0.2,
            max_iter=150000,
            top_k=5,
            patience_max=50
        )
        full_sc = _score_norm(a_opt)
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()
        print(f"    Adam phase {i}: T~{T_i:.6f}, score={full_sc:.10f}")

    return best_a, best_score

        a_opt2, sc2 = _adam_optimize(
            best_a, phase3_end,
            T_init=max_cb * 0.005,
            T_final=max_cb * 0.0001,
            lr_init=0.015,
            lr_final=0.001,
            max_iter=200000,
            top_k=4,
            patience_max=40
        )
        full_sc2 = _score(a_opt2)
        print(f"Phase 3 done: score={full_sc2:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # --- Phase 4: Second Remez exchange (next ~200s) ---
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Remez exchange round 2")
        a_rz2, sc_rz2 = _remez_exchange(best_a, phase4_end, max_iter=5000, n_active=4)
        full_sc_rz2 = _score(a_rz2)
        print(f"Phase 4 done: score={full_sc_rz2:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc_rz2 < best_score:
            best_score = full_sc_rz2
            best_a = a_rz2.copy()

    # --- Phase 5: Fine local search (next ~150s) ---
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Fine local search")
        a_ls, sc_ls = _fine_local_search(best_a, phase5_end, max_rounds=3000)
        full_sc_ls = _score(a_ls)
        print(f"Phase 5 done: score={full_sc_ls:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc_ls < best_score:
            best_score = full_sc_ls
            best_a = a_ls.copy()

    # --- Phase 6: Final Remez + local (remaining time) ---
    if time.monotonic() < deadline - 10:
        print(f"Phase 6: Final refinement")
        sub_dl = max(time.monotonic() + 5, min(deadline - 5, time.monotonic() + 80))
        a_fin, sc_fin = _remez_exchange(best_a, sub_dl, max_iter=2000, n_active=3)
        full_sc_fin = _score(a_fin)
        print(f"Phase 6a: score={full_sc_fin:.10f}")
        if full_sc_fin < best_score:
            best_score = full_sc_fin
            best_a = a_fin.copy()

        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _fine_local_search(best_a, deadline - 3, max_rounds=1000)
            full_sc_fin2 = _score(a_fin2)
            print(f"Phase 6b: score={full_sc_fin2:.10f}")
            if full_sc_fin2 < best_score:
                best_score = full_sc_fin2
                best_a = a_fin2.copy()

    # --- Final validation ---
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01

    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
