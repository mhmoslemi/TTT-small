"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses progressive active-set Newton with multi-level tolerance, L-BFGS with

Key innovations: adaptive CG with restarts, progressive active-set tightening,
Armijo line search, and final coordinate-descent polish on active peaks.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))

    if s < 1e-300:

    """Compute grad[i] = 2 * sum_k w[k] * a[k-i] using FFT."""
    out_len = len(w) + n - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)

def _lsobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = float(c.max())

    else:
        w = e / s
    h = _grad_from_active(a, w, n) / 2.0
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _cg_solve(hess_matvec, rhs, max_cg=300, tol=1e-10):
    """Conjugate gradient with restarts."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - hess_matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    target = tol * rhs_norm

    for it in range(max_cg):
        if np.sqrt(rs) < target:
            break
        Ap = hess_matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < target:
            break
        if rs < 1e-30:
            p = r_new
            rs = np.dot(r_new, r_new)
            continue
        p = r_new + (rs_new / rs) * p
        # CG restart every 50 iterations to avoid loss of orthogonality
        if (it + 1) % 50 == 0:
            p = r_new
        r = r_new
        rs = rs_new
    return x

        # Gradient of the "averaged active" objective
        g = (2.0 / m) * _grad_from_active(a, A_vec, n) / 2.0
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if verbose:
            print(f"    AS iter {it}: n_active={m}, gnorm={gnorm:.2e}, c_max={c_max:.6e}")
        if gnorm < 1e-14:
            break

        # Regularization
        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

        def hess_matvec(v):
            return (2.0 / m) * _grad_from_active(v, A_vec, n) / 2.0 + mu * v

        # Solve (H + mu*I) x = g_t using CG
        u = _cg_solve(hess_matvec, g_t, max_cg=300, tol=1e-12)

        # Handle the normalization constraint (sum a = 1)
        w_ones = np.ones(n)
        w = _cg_solve(hess_matvec, w_ones, max_cg=200, tol=1e-10)
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
        dx = dx - np.mean(dx)

        # Line search with Armijo condition
        f_old = c_max
        slope = float(np.dot(g_t, dx))
        if slope > 0:
            dx = -dx
            slope = -slope

        neg_mask = dx < -1e-14
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        accepted = False
        t = min(1.0, t_max)
        for ls in range(60):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            # Armijo: f_try <= f_old + c1 * t * slope
            if f_try <= f_old + 1e-4 * t * slope or (f_try < f_old - 1e-14):
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent step
            t_fallback = min(0.5, t_max)
            for ls in range(30):
                a_try = a - t_fallback * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t_fallback *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-15:
                    a = a_try
                    break
                t_fallback *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13 * c_max:
            break

    return best_a, best_sc


def _progressive_active_set(a_init, deadline):
    """Run active-set Newton with progressively tighter tolerance."""
    current = a_init.copy()
    current_sc = _score(current)
    print(f"  Progressive AS start: score={current_sc:.10f}")

    tolerances = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    for tol in tolerances:
        if time.monotonic() > deadline:
            break
        time_per_phase = max(10.0, (deadline - time.monotonic()) / (len(tolerances) + 1))
        a_new, sc_new = _active_set_newton(
            current,
            time.monotonic() + time_per_phase,
            max_iter=300,
            tol_active=tol
        )
        if sc_new < current_sc - 1e-15:
            current = a_new
            current_sc = sc_new
            print(f"  AS tol={tol:.6f}: score={current_sc:.10f} (improved)")
        else:
            print(f"  AS tol={tol:.6f}: no improvement (score={sc_new:.10f})")

    return current, current_sc


def _coord_descent_polish(a_init, deadline, max_rounds=200):
    """Coordinate descent targeting the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Focus on top peaks
        n_peaks = min(20, len(c) // 5)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        threshold = c[top_idx[-1]] * 0.9995
        active = c >= threshold

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        h = _grad_from_active(a, w, n) / 2.0
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)

        # Adaptive step size
        step = 0.05 / (1.0 + 0.01 * rnd)
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < c_max - 1e-16:
                a = a_try
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14 * c_max:
            break

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage(a, deadline):
    """Try zeroing out small entries to potentially improve score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.05, 0.10]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            print(f"  Shrinkage {frac:.3f}: score={best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. L-BFGS from parent for coarse optimization
    2. Progressive active-set Newton (loose to tight)
    3. L-BFGS from improved point
    4. More active-set Newton
    5. Coordinate descent polish
    6. Diverse restarts
    7. Final tightening
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.6e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    print(f"\n{'='*60}")
    print(f"[Phase 1] L-BFGS optimization (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    p1_deadline = t_start + 300
    a1 = _run_lbfgs(parent, min(p1_deadline, deadline - 300), min_time_remaining=200.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs_1")

    print(f"\n{'='*60}")
    print(f"[Phase 2] Progressive active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    p2_deadline = t_start + 700
    a2, sc2 = _progressive_active_set(a1, min(p2_deadline, deadline - 200))
    update_best(a2, sc2, "progressive_as_1")

    print(f"\n{'='*60}")
    print(f"[Phase 3] L-BFGS from improved (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    p3_deadline = t_start + 900
    a3 = _run_lbfgs(a2, min(p3_deadline, deadline - 200), min_time_remaining=100.0, aggressive=True)
    update_best(a3, _score(a3), "lbfgs_2")

    print(f"\n{'='*60}")
    print(f"[Phase 4] Active-set Newton round 2 (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    p4_deadline = t_start + 1200
    a4, sc4 = _progressive_active_set(a3, min(p4_deadline, deadline - 100))
    update_best(a4, sc4, "progressive_as_2")

    print(f"\n{'='*60}")
    print(f"[Phase 5] Coordinate descent polish (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    p5_deadline = t_start + 1400
    a5, sc5 = _coord_descent_polish(a4, min(p5_deadline, deadline - 100), max_rounds=300)
    update_best(a5, sc5, "coord_descent")

    print(f"\n{'='*60}")
    print(f"[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    p6_deadline = t_start + 1650
    if best_seq is not None:
        restarts = [
            ("pert_003", 0.003),
            ("pert_005", 0.005),
            ("pert_010", 0.01),
            ("pert_020", 0.02),
            ("pert_050", 0.05),
        ]
        time_per_restart = (min(p6_deadline, deadline - 30) - time.monotonic()) / len(restarts)
        for label, sigma in restarts:
            if time.monotonic() > min(p6_deadline, deadline - 30):
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            r_deadline = min(time.monotonic() + time_per_restart, deadline - 10)
            a_r = _run_lbfgs(p_start, r_deadline, min_time_remaining=5.0, aggressive=False)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            # Quick active-set on the restart result
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 30),
                                                 max_iter=100, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    print(f"\n{'='*60}")
    print(f"[Phase 7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sh, sc_sh = _shrinkage(best_seq, min(deadline - 10, time.monotonic() + 30))
        update_best(a_sh, sc_sh, "shrinkage")

    print(f"\n{'='*60}")
    print(f"[Phase 8] Final tightening (t={time.monotonic()-t_start:.1f}s)")
    print(f"{'='*60}")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active-set Newton
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 2, max_iter=200, tol_active=0.00001)
        update_best(a_fin, sc_fin, "final_as_tight")

        # One more coordinate descent
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _coord_descent_polish(best_seq, deadline - 1, max_rounds=100)
            update_best(a_fin2, sc_fin2, "final_cd")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
