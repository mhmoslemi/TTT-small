The key equation for the active-set Newton is:
[Hxx  Hxa  b_x] [dx]   [-g_x]

For our problem, the Lagrangian is:

The active constraints are those with lambda_k > 0 (i.e., conv(a,a)[k] = M).

The KKT conditions:
- grad_a L = 0: 2*sum_k lambda_k * (conv gradient at k) + nu*1 - mu = 0
- grad_M L = 0: 1 - sum_k lambda_k = 0
- Complementarity: lambda_k * (conv(a,a)[k] - M) = 0

This is a system of nonlinear equations in (a, M, lambda, nu, mu).

The Newton method for this system requires:
1. Computing the Hessian of the Lagrangian with respect to a
2. Solving the linear system

The Hessian of the Lagrangian w.r.t. a is:
H = 2 * sum_k lambda_k * H_k

where H_k is the Hessian of conv(a,a)[k] w.r.t. a.

= delta(i, k-j) + (0 or delta(j, k-i) if the term exists)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous:

import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (all top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def lsobj(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def lsgrad(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def peak_shave(a, deadline, n_rounds=20):
    """Reduce peaks by redistributing energy."""
    n = len(a)
    best_a = a.copy()
    best_sc = score(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve(best_a, best_a)
        c_max = float(c.max())
        
        # Find top peaks
        top_k = np.argsort(c)[::-1][:max(3, n // 100)]
        
        # For each top peak, find the responsible pairs and reduce them
        a_try = best_a.copy()
        reduction_total = 0.0
        reduction_positions = []
        
        for k_pos in top_k[:10]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            # Find the pair with largest product
            prods = np.array([a_try[i] * a_try[k_pos - i] for i in range(i_lo, i_hi + 1)])
            if len(prods) == 0:
                continue
            max_idx = i_lo + int(np.argmax(prods))
            partner = k_pos - max_idx
            if prods.max() > 0:
                # Reduce the larger of the pair
                if a_try[max_idx] >= a_try[partner]:
                    reduction = a_try[max_idx] * 0.001
                    a_try[max_idx] -= reduction
                    reduction_positions.append(max_idx)
                else:
                    reduction = a_try[partner] * 0.001
                    a_try[partner] -= reduction
                    reduction_positions.append(partner)
                reduction_total += reduction
        
        if reduction_total < 1e-15:
            break
        
        # Redistribute to positions with low convolution contribution
        # Find positions where the local convolution is below average
        avg_c = c.mean()
        low_positions = np.where(np.array([c[min(i, 2*n-2)] for i in range(n)]) < avg_c)[0]
        
        if len(low_positions) == 0:
            low_positions = np.arange(n)
        
        # Distribute proportionally to current values (or uniformly if all zero)
        weights = a_try[low_positions] + 1e-10
        weights /= weights.sum()
        
        for idx in range(len(reduction_positions)):
            a_try[low_positions[idx % len(low_positions)]] += reduction_total * weights[idx % len(low_positions)]
        
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try / s_try
            sc_try = score(a_try)

"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _lsobj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

                best_sc = sc_try
            else:
                if rnd > 5:
                    break

    return best_a, best_sc


def _fourier_perturb(a, deadline, n_trials=50):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        n_freqs = rng.integers(1, 4)
        z = np.log(np.maximum(a, 1e-30))
        z_pert = z.copy()

        for _ in range(n_freqs):
            freq = rng.integers(1, n // 8 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            amplitude = rng.uniform(0.005, 0.05)
            x = np.arange(n)
            z_pert += amplitude * np.cos(2 * np.pi * freq * x / n + phase)

        a_try = _softmax_fn(z_pert)
        sc_try = _score(a_try)

        if sc_try < best_sc + 0.005:
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            beta = max(c_max_try * 0.001, 1e-10)
            z_pert = np.log(np.maximum(a_try, 1e-30))
            z_pert = z_pert - z_pert.mean()
            obj_fn = _LsObj(beta, n)
            grad_fn = _LsGrad(beta, n)
            try:
                result = minimize(
                    obj_fn, z_pert, jac=grad_fn,
                    method='L-BFGS-B',
                    options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_pol = _softmax_fn(result.x)
                sc_pol = _score(a_pol)
                if sc_pol < best_sc - 1e-14:
                    best_a = a_pol.copy()
                    best_sc = sc_pol
                    print(f"    Fourier[{trial}]: score={best_sc:.12f}")
            except Exception:
                pass

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.1 * 0.1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        a = np.zeros(n)
        b = max(1, n // 4)
        a[:b] = 1.0
        a[3 * b:] = 1.0
        s = a.sum()
        if s > 0:
            return a / s
        return np.ones(n) / n
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with fine beta continuation,
    active-set Newton, peak shaving, Fourier perturbation, multi-n.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score_val = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p_arr.copy()
            best_score_val = float(sc)
            print(f"  NEW BEST: {best_score_val:.12f} (n={len(p_arr)}) "
                  f"improve={prev - best_score_val:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.12f}")
        update_best(parent, sc_parent)
    else:
        n_parent = 16384
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, uniform n={n_parent}")
        update_best(parent, _score(parent))

    n_main = n_parent if n_parent in (4096, 8192, 16384, 32768) else 16384
    if n_parent != n_main:
        parent = _interpolate_to_n(parent, n_main)
        sc_parent = _score(parent)
        update_best(parent, sc_parent)

    # Phase 1: Main L-BFGS
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=400.0)
    update_best(a1, _score(a1))

    # Phase 2: Active-set Newton (wide)
    print(f"\n[Phase 2] Active-set Newton wide (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        as_dl = min(deadline - 250, time.monotonic() + 200)
        a2, sc2 = _active_set_newton(best_seq, as_dl, max_iter=600,
                                      tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        update_best(a2, sc2)

    # Phase 3: Peak shaving + L-BFGS
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        ps_dl = min(deadline - 200, time.monotonic() + 60)
        a3, sc3 = _peak_shave(best_seq, ps_dl, n_rounds=40)
        update_best(a3, sc3)
        if sc3 < best_score_val:
            a3b = _run_lbfgs(a3, deadline, min_time_remaining=200.0)
            update_best(a3b, _score(a3b))

    # Phase 4: Fourier perturbation
    print(f"\n[Phase 4] Fourier perturbation (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        fp_dl = min(deadline - 150, time.monotonic() + 100)
        a4, sc4 = _fourier_perturb(best_seq, fp_dl, n_trials=50)
        update_best(a4, sc4)

    # Phase 5: Active-set Newton (tight)
    print(f"\n[Phase 5] Active-set Newton tight (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        as2_dl = min(deadline - 80, time.monotonic() + 120)
        a5, sc5 = _active_set_newton(best_seq, as2_dl, max_iter=500,
                                      tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        update_best(a5, sc5)

    # Phase 6: Try other n values
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [8192, 16384, 32768]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_main:
            continue
        a_init = _interpolate_to_n(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt)
        if sc_opt < best_score_val + 0.005 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 30),
                                                  max_iter=200,
                                                  tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_opt2, sc_opt2)

    # Phase 7: Final polish loop
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    polish_rounds = 5
    for pr in range(polish_rounds):
        if time.monotonic() > deadline - 5:
            break
        # Tight Newton
        if time.monotonic() < deadline - 10:
            ndl = min(deadline - 10, time.monotonic() + 30)
            a_p, sc_p = _active_set_newton(best_seq, ndl, max_iter=200,
                                            tol_list=[0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
            update_best(a_p, sc_p)
        # Tiny L-BFGS
        if time.monotonic() < deadline - 5:
            sc_cur = _score(best_seq)
            a_p2 = _run_lbfgs(best_seq, deadline, min_time_remaining=2.0)
            update_best(a_p2, _score(a_p2))
        # Peak shave
        if time.monotonic() < deadline - 3:
            a_p3, sc_p3 = _peak_shave(best_seq, min(deadline - 2, time.monotonic() + 10), n_rounds=20)
            update_best(a_p3, sc_p3)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"\n{'=' * 60}")
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
