"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using arcsine-distribution
initialization (which gives near-flat convolution in the continuous limit),
aggressive L-BFGS with log-sum-exp smoothing, progressive active-set Newton,
and multi-initialization exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-7 * (c_max / max(m, 1)) + 1e-10

        def _hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def _cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - _hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = _hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = _cg_solve(g_t)
        w = _cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    if aggressive:
        betas = base_beta * np.array(
            [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.035,
             0.025, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0005,
             0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        )
    else:
        betas = base_beta * np.array(
            [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01,
             0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.007
        elif n <= 16384:
            est_time_per_iter = 0.018
        else:
            est_time_per_iter = 0.04 * (n / 32768.0)

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            t0 = time.monotonic()
            result = minimize(
                _lsobj_wrapper,
                z,
                jac=_lsgrad_wrapper,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    beta={beta:.2e}: score={current_sc:.10f}, time={elapsed:.1f}s")
        except Exception as e:
            print(f"    L-BFGS error at beta={beta:.2e}: {e}")

    return current_a


def _lsobj_wrapper(z, beta, n):
    return _lsobj(z, beta, n)


def _lsgrad_wrapper(z, beta, n):
    return _lsgrad(z, beta, n)


def _make_arcsine(n):
    """Arcsine distribution: a_i proportional to 1/sqrt((i+1)(n-i))"""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_law(n, alpha=0.5):
    """a_i proportional to (i+1)^(-alpha) * (n-i)^(-alpha)"""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha) * (n + 1 - i) ** (-alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sine_window(n):
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.sin(np.pi * t) ** 0.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_arc(n):
    """Two-arc: high at both edges, low in middle"""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    a = 1.0 / np.sqrt(np.maximum(1.0, np.abs(i - center) + 1))
    return a / a.sum()


def _peak_targeted_refine(a_init, deadline, max_rounds=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.05

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)

        f_old = c_max
        improved = False
        step = step_size / (1.0 + 0.02 * rnd)
        for ls in range(40):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                improved = True
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                break
            step *= 0.5

        if not improved:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.6

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
