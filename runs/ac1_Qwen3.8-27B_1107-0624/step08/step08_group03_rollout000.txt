    """Gerchberg-Saxton algorithm to find a non-negative sequence with flat spectrum."""

def coord_descent_polish(a_init, deadline, max_rounds=100):
    """Coordinate descent: for each coordinate, optimize it while keeping others fixed."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, score_fn(a)
    a = a / s
    best_a = a.copy()
    best_sc = score_fn(a)
    
    # Precompute the "background" convolution (without each element)
    # This is expensive, so we'll use a simpler approach
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify the peak location
        peak_k = int(np.argmax(c))
        
        # The peak b[peak_k] = sum_j a[j] * a[peak_k - j] (for valid j)
        # To reduce this peak, we should reduce the a[j] that contribute most
        
        # Compute contribution of each a[j] to the peak
        # b[k] = sum_j a[j] * a[k-j]
        # For k = peak_k: contribution of a[j] is a[peak_k - j] (if 0 <= peak_k - j < n)
        
        contributions = np.zeros(n)
        for j in range(n):
            k2 = peak_k - j
            if 0 <= k2 < n:
                contributions[j] = a[j] * a[k2]
        
        # Find the top contributors
        # Reducing the largest contributors will most effectively reduce the peak
        
        # Simple approach: reduce the top few contributors
        top_k = max(5, n // 100)
        top_idx = np.argsort(contributions)[::-1][:top_k]
        
        # Reduce them slightly
        reduction = 0.001 * a[top_idx]
        a_new = a.copy()
        a_new[top_idx] -= reduction
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        c_new = full_convolve(a_new, a_new)
        if c_new.max() < c_max - 1e-14:
            a = a_new
            sc = score_fn(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        # Check convergence
        if rnd > 0 and abs(c_max - float(full_convolve(a, a).max())) < 1e-13:
            break
    
    return best_a, best_sc

def interpolate_to_n(a, n_new):
    n_old = len(a)

def subgrad_polish_v2(a_init, deadline, max_steps=5000, lr_init=0.02):

            break

        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if i % 50 == 0:
            a_check = softmax_fn(z)
            sc = score_fn(a_check)

                lr *= 0.92
