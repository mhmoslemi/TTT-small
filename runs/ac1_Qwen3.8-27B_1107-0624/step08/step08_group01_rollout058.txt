multi-n exploration, and targeted peak equalization. Focuses on very fine
annealing of the LSE temperature to push the convolution toward flatness.

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _estimate_beta(a, floor_frac=0.00001):

    c_max = float(c.max())
    top_k = min(200, len(c))

    gap = c_sorted[0] - c_sorted[min(top_k - 1, len(c_sorted) - 1)]
    if gap < c_max * 0.00005:
        gap = c_max * 0.00005
    beta = max(gap * 0.4, c_max * floor_frac)
    return beta, c_max

    p4_deadline = t_start + budget * 0.83
    a4, sc4 = _peak_equalize(best_a, p4_deadline, min_remaining=20)
    if sc4 < best_sc:
        best_a = a4.copy()
        best_sc = sc4
        print(f"  P4: score={best_sc:.10f}, t={time.monotonic()-t_start:.1f}s", flush=True)
    else:
        print(f"  P4: no improvement, t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 5: Diverse restarts (7% time) ===
    print(f"\n[P5] Diverse restarts", flush=True)
    p5_deadline = t_start + budget * 0.90
    n_best = len(best_a)
    restarts = []
    restarts.append(("pert_small", _perturb(best_a, 0.01, rng)))
    restarts.append(("pert_med", _perturb(best_a, 0.03, rng)))
    restarts.append(("tapered", _make_profile(n_best, "tapered")))
    restarts.append(("asym", _make_profile(n_best, "asymmetric")))

    for label, a_r in restarts:
        if time.monotonic() > p5_deadline - 20.0:
            break
        remaining = min(60.0, p5_deadline - time.monotonic() - 15.0)
        if remaining < 20.0:
            break
        a_opt, sc_opt = _run_adaptive_lbfgs(a_r, time.monotonic() + remaining, min_remaining=10)
        if sc_opt < best_sc:
            best_a = a_opt.copy()
            best_sc = sc_opt
            print(f"  Restart {label}: score={best_sc:.10f}, t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Final ultra-fine polishing (10% time) ===
    print(f"\n[P6] Final ultra-fine LSE", flush=True)
    p6_deadline = deadline
    a6, sc6 = _run_adaptive_lbfgs(best_a, p6_deadline, min_remaining=5)
    if sc6 < best_sc:
        best_a = a6.copy()
        best_sc = sc6
        print(f"  P6: score={best_sc:.10f}, t={time.monotonic()-t_start:.1f}s", flush=True)
    else:
        print(f"  P6: no improvement, t={time.monotonic()-t_start:.1f}s", flush=True)

    # Final peak equalization
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _peak_equalize(best_a, deadline - 2.0, min_remaining=1)
        if sc_fin < best_sc:
            best_a = a_fin.copy()
            best_sc = sc_fin
            print(f"  Final PE: score={best_sc:.10f}", flush=True)

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: local_score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # Verify with evaluate_sequence
    try:
        ev_score = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence score={ev_score:.10f}", flush=True)
    except Exception as e:
        print(f"evaluate_sequence error: {e}", flush=True)

    return [float(x) for x in final]
