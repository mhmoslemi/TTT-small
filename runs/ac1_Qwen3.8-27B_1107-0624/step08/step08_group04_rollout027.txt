"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) L-BFGS with adaptive beta continuation, (2) FISTA projected gradient,
(3) Frank-Wolfe with away-steps, (4) active-set QP with FFT-accelerated CG,
(5) peak-shaving iterations, and (6) multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


# ==================== Core utilities ====================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hankel_matvec(v, W, n):
    """Compute gradient of sum(W_k * c_k) w.r.t. a, where c = conv(a,a).
    Result[i] = 2 * sum_k W[k] * a[k-i] (with a[m]=0 for m out of [0,n))."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


# ==================== Smooth objective for L-BFGS ====================

# Global state for minimize callbacks
_g_beta = 0.01
_g_n = 1000


def _smoobj_func(z):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / _g_beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * _g_n * c_max
    return 2.0 * _g_n * (c_max + _g_beta * np.log(s))


def _smograd_func(z):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / _g_beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad = _hankel_matvec(a, w, _g_n)
    return grad  # Note: this is the gradient w.r.t. a, not w.r.t. z
    # For L-BFGS on z, we need a different formula (see _smograd_z below)


def _smograd_z(z):
    """Gradient of smooth objective w.r.t. z (log-space parameters)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / _g_beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad w.r.t. a:
    grad_a = _hankel_matvec(a, w, _g_n)
    # Convert to grad w.r.t. z: grad_z = a * (grad_a - sum(a * grad_a))
    # Actually, for softmax: d softmax(z)/dz_i = a_i * (delta_ij - a_j)
    # So grad_z_i = sum_j grad_a_j * d a_j / d z_i = sum_j grad_a_j * a_j * (delta_ij - a_i)
    # = a_i * grad_a_i - a_i * sum_j grad_a_j * a_j
    # = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    W_val = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W_val)
    return grad_z


# ==================== L-BFGS phase ====================

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    _g_n = n
    
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    # Beta schedule: start with plateau, then geometric decay
    ratios = [1.0, 1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05,
              0.035, 0.025, 0.018, 0.012, 0.008, 0.005, 0.003, 0.002,
              0.001, 0.0005, 0.0003, 0.00015, 0.00008, 0.00004, 0.00002,
              0.00001, 0.000005, 0.000002, 0.000001, 0.0000005, 0.0000002]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        _g_beta = beta
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(30, int(remaining / (est_iter_time * 5))))

        try:
            result = minimize(
                _smoobj_func,
                z,
                jac=_smograd_z,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


# ==================== FISTA / Projected Gradient ====================

def _fista_polish(a_init, deadline, max_iters=5000, beta_init=None):
    """FISTA (accelerated projected gradient) on the smoothed max objective."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    
    if beta_init is None:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
        beta_init = max(gap * 0.5, c_max * 0.01)

    beta = beta_init
    beta_decay = 0.95
    lr = 1.0 / (4.0 * n)  # initial step size
    t_k = 1.0
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e
        
        grad = _hankel_matvec(a, w, n)
        grad = grad - grad.mean()  # remove mean (since we're on the simplex)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
        a_new = _project_simplex(y - lr * grad)
        
        # Backtracking line search (simplified)
        sc_new = _score(a_new)
        if sc_new > best_sc * 1.01:
            # Step too big, reduce
            lr *= 0.5
            a_new = _project_simplex(y - lr * grad)
            sc_new = _score(a_new)
        
        a_prev = a
        a = a_new
        t_k = t_new
        
        # Adapt beta
        if it % 100 == 0 and beta > c_max * 1e-6:
            beta *= beta_decay
            lr = min(lr * 1.1, 1.0 / (2.0 * n))
        
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a.copy()

    return best_a, best_sc

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines: (1) L-BFGS with aggressive beta continuation, (2) FISTA projected gradient
with adaptive sharpening, (3) direct peak-shaving iterations, (4) active-set QP polish,
(5) multi-length exploration, and (6) randomized restarts.
Key improvement: sharper peak weighting + more aggressive low-beta L-BFGS phase.
"""
import numpy as np
import time
from scipy.optimize import minimize


# ==================== Core utilities ====================

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hankel_matvec(v, W, n):
    """Gradient of sum(W_k * c_k) w.r.t. a, where c=conv(a,a).
    Result[i] = 2 * sum_k W[k]*a[k-i] (a[m]=0 for m<0 or m>=n)."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_n)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


# ==================== Smooth objective (global params for minimize) ====================

_g_beta = 0.01
_g_n = 1000


def _smoobj_func(z):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / _g_beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * _g_n * c_max
    return 2.0 * _g_n * (c_max + _g_beta * np.log(s))


def _smograd_z_func(z):
    """Gradient of smooth objective w.r.t. z (log-space parameters)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / _g_beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_a = _hankel_matvec(a, w, _g_n)
    W_val = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W_val)
    return grad_z


# ==================== L-BFGS phase ====================

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation - aggressive low-beta push."""
    n = len(a_init)
    _g_n = n

    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.25, max_c * 0.002)

    ratios = np.array([1.0, 0.8, 0.6, 0.45, 0.32, 0.22, 0.15, 0.1, 0.07, 0.05,
                       0.035, 0.025, 0.018, 0.012, 0.008, 0.005, 0.0035, 0.002,
                       0.0012, 0.0007, 0.0004, 0.0002, 0.0001, 0.00005,
                       0.00003, 0.000015, 0.000008, 0.000004, 0.000002,
                       0.000001, 0.0000005, 0.0000002, 0.0000001,
                       0.00000005, 0.00000002, 0.00000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        _g_beta = beta
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 3))))

        try:
            result = minimize(
                _smoobj_func,
                z,
                jac=_smograd_z_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


# ==================== FISTA / Projected Gradient ====================

def _fista_polish(a_init, deadline, max_iters=3000, sharpness=3.0):
    """FISTA with adaptive sharpening on the peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = c_max * 0.01
    lr = 0.5 / n
    t_k = 1.0
    a_prev = a.copy()
    stale = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Sharper weighting: emphasize top values more
        threshold = c_max - 0.1 * c_max
        excess = np.maximum(c - threshold, 0)
        if excess.max() < 1e-15:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = excess ** sharpness
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / w_sum

        grad = _hankel_matvec(a, w, n)
        grad = grad - grad.mean()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
        a_new = _project_simplex(y - lr * grad)

        sc_new = _score(a_new)
        if not np.isfinite(sc_new):
            a_new = a
            sc_new = best_sc

        if sc_new < best_sc * 1.001:
            a_prev = a
            a = a_new
            t_k = t_new
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a.copy()
                stale = 0
            else:
                stale += 1
        else:
            stale += 1
            lr *= 0.7

        if stale > 50:
            a = best_a.copy()
            a_prev = a.copy()
            t_k = 1.0
            lr = 0.5 / n
            stale = 0

        if it % 80 == 0 and beta > c_max * 1e-7:
            beta *= 0.85

    return best_a, best_sc


# ==================== Direct peak shaving ====================

def _peak_shave(a_init, deadline, max_iters=2000, top_frac=0.02):
    """Directly shave the top convolution values via iterative gradient steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    lr = 2.0 / n
    momentum_v = np.zeros(n)
    momentum = 0.85

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Weight: strongly emphasize top values
        k_top = max(5, int(len(c) * top_frac))
        top_idx = np.argsort(c)[::-1][:k_top]
        w = np.zeros_like(c)
        # Exponential decay from top
        sorted_top = c[top_idx]
        decayed = np.exp(-3.0 * np.arange(k_top) / max(1, k_top // 3))
        w[top_idx] = decayed / decayed.sum()

        grad = _hankel_matvec(a, w, n)
        grad = grad - grad.mean()

        # Momentum update
        momentum_v = momentum * momentum_v + (1 - momentum) * grad
        a_new = _project_simplex(a - lr * momentum_v)

        sc_new = _score(a_new)
        if not np.isfinite(sc_new) or sc_new > best_sc * 1.005:
            lr *= 0.5
            momentum_v *= 0.5
            a_new = _project_simplex(a - lr * momentum_v)

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Adaptive lr
        if it % 50 == 0:
            lr = min(lr * 1.05, 5.0 / n)

        # Sharpness increases over time
        if it % 100 == 0:
            top_frac = max(0.001, top_frac * 0.8)

    return best_a, best_sc


# ==================== Active-set QP ====================

def _cg_solve(W, n, rhs, max_iter=1000, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _active_qp_polish(a_init, deadline, max_rounds=120):
    """Active-set QP with progressive tolerance tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_schedule = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                    0.000001, 0.0000005, 0.0000002, 0.0000001,
                    0.00000005, 0.00000002, 0.00000001]
    rounds_per_tol = 4

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(rnd // rounds_per_tol, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP: min (1/2) a^T H a s.t. sum(a)=1, a>=0
        # KKT: H a = lambda * 1 (on support), a_i > 0
        y = _cg_solve(W, n, np.ones(n), max_iter=500)
        oney = float(np.sum(y))
        if abs(oney) < 1e-20 or not np.isfinite(oney):
            # Fallback: damped gradient step
            grad = _hankel_matvec(a, W, n)
            grad = grad - grad.mean()
            step = 1.0 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.3)
            a_new = _project_simplex(a - step * grad)
        else:
            a_new = np.maximum(y, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a
            else:
                a_new = a_new / s_new

        sc_new = _score(a_new)
        if np.isfinite(sc_new) and sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try damped step towards a_new
            for alpha in [0.5, 0.3, 0.1]:
                a_try = (1 - alpha) * a + alpha * a_new
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

        # Check convergence
        if rnd > 0 and (rnd % rounds_per_tol == 0):
            c_check = _full_convolve(a, a)
            range_c = float(c_check.max() - c_check.min())
            if range_c < 1e-10 * c_check.max():
                break

    return best_a, best_sc


# ==================== Subgradient with Nesterov ====================

def _subgrad_nesterov(a_init, deadline, max_steps=5000):
    """Subgradient method with Nesterov acceleration in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.02
    momentum = 0.9
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        tol_frac = max(0.0001, 0.001 * (1 - i / max_steps))
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad_a = _hankel_matvec(a, w, n)
        W_val = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W_val)

        step = lr / (1.0 + 0.001 * i)
        z_new = z_prev + momentum * (z_prev - z) + (1 - momentum) * (-step * grad_z)
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.8
                    momentum = min(momentum + 0.02, 0.95)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


# ==================== Profile generators ====================

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = int(n * 0.15)
        a[:edge] = np.linspace(0.3, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.3, edge)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


# ==================== Main search function ====================

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS beta continuation, FISTA with sharpening,
    peak shaving, active-set QP, multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS with aggressive beta continuation
    print(f"\n[Phase 1] L-BFGS aggressive (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=500.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: FISTA with sharpening
    print(f"\n[Phase 2] FISTA sharpen (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _fista_polish(best_seq, min(deadline - 400, time.monotonic() + 200),
                                 max_iters=3000, sharpness=4.0)
        update_best(a2, sc2, "fista_1")

    # Phase 3: Peak shaving
    print(f"\n[Phase 3] Peak shave (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3, sc3 = _peak_shave(best_seq, min(deadline - 350, time.monotonic() + 150),
                               max_iters=2000, top_frac=0.03)
        update_best(a3, sc3, "peak_shave_1")

    # Phase 4: Active-set QP
    print(f"\n[Phase 4] Active-set QP (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _active_qp_polish(best_seq, min(deadline - 300, time.monotonic() + 200),
                                     max_rounds=100)
        update_best(a4, sc4, "active_qp_1")

    # Phase 5: Another L-BFGS round (from improved point)
    print(f"\n[Phase 5] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5 = _run_lbfgs(best_seq, deadline, min_time_remaining=200.0)
        update_best(a5, _score(a5), "lbfgs_2")

    # Phase 6: FISTA round 2 (sharper)
    print(f"\n[Phase 6] FISTA round 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a6, sc6 = _fista_polish(best_seq, min(deadline - 200, time.monotonic() + 100),
                                 max_iters=2000, sharpness=6.0)
        update_best(a6, sc6, "fista_2")

    # Phase 7: Peak shave round 2 (very sharp)
    print(f"\n[Phase 7] Peak shave 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a7, sc7 = _peak_shave(best_seq, min(deadline - 180, time.monotonic() + 100),
                               max_iters=1500, top_frac=0.005)
        update_best(a7, sc7, "peak_shave_2")

    # Phase 8: Active-set QP round 2 (tighter)
    print(f"\n[Phase 8] Active-set QP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a8, sc8 = _active_qp_polish(best_seq, min(deadline - 150, time.monotonic() + 120),
                                     max_rounds=120)
        update_best(a8, sc8, "active_qp_2")

    # Phase 9: Multi-n exploration
    print(f"\n[Phase 9] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 8192, 32768]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent and len(best_seq) == n_try:
            continue
        if n_try > 32768:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_np, sc_np = _active_qp_polish(a_opt, min(deadline - 30, time.monotonic() + 60),
                                             max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 10: Diverse restarts
    print(f"\n[Phase 10] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_kinds = ["flat_top", "bimodal", "centered"]
        for i, prof in enumerate(restart_kinds):
            if time.monotonic() > deadline - 20:
                break
            n_r = len(best_seq)
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            else:
                p_start = _make_profile(n_r, prof)
                # Blend with best
                p_start = 0.5 * p_start + 0.5 * best_seq
                p_start = p_start / p_start.sum()

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _active_qp_polish(a_r, min(deadline - 10, time.monotonic() + 20),
                                                 max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{i}_qp")

    # Phase 11: Final ultra-tight polish
    print(f"\n[Phase 11] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight QP
        a_fin, sc_fin = _active_qp_polish(best_seq, deadline - 3, max_rounds=80)
        update_best(a_fin, sc_fin, "final_qp")

        # Final peak shave (very sharp)
        if time.monotonic() < deadline - 3:
            a_ps, sc_ps = _peak_shave(best_seq, deadline - 2, max_iters=500, top_frac=0.001)
            update_best(a_ps, sc_ps, "final_peak")

        # Final FISTA
        if time.monotonic() < deadline - 3:
            a_fi, sc_fi = _fista_polish(best_seq, deadline - 2, max_iters=1000, sharpness=8.0)
            update_best(a_fi, sc_fi, "final_fista")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
