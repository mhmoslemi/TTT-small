```python
"""
Primal-dual + L-BFGS hybrid optimizer for step function peak minimization.
Key innovation: KKT-based primal-dual alternating iterations that exploit the
structural characterization of the optimum (weighted convolution is constant on
the support of a). Combined with adaptive log-sum-exp L-BFGS, active-set
projected gradient, and local pair-swap search for fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[min(50, len(c_sorted) - 1)] 
    top_val = c_sorted[0]
    gap = top_val - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap * 0.5, max_c * 0.002)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-13, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if (i + 1) % 4 == 0:
                    print(f"    L-BFGS beta_idx={i}, score={current_sc:.10f}")
        except Exception:
            pass
    return current_a


def _primal_dual(a_init, deadline, min_time_remaining=60.0):
    """
    KKT-based primal-dual alternating iteration.
    Primal: update a to make F[j] = 2*(w*a)[j] constant on support(a).
    Dual: update w to be concentrated on peaks of c = conv(a,a).
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Dual variable: probability on convolution positions
    c = _full_convolve(a, a)
    c_max = float(c.max())
    w_beta = max(c_max * 0.01, 1e-10)
    w = np.exp((c - c_max) / w_beta)
    w = w / w.sum()

    alpha = 0.5 / n
    no_improve = 0
    max_no_improve = 200
    iters = 0

    while time.monotonic() + min_time_remaining < deadline:
        # Compute F[j] = 2 * sum_m a[m] * w[m+j] using FFT
        # F[j] = 2 * conv(a, w_flip)[2n-2-j] for j=0..n-1
        w_flip = w[::-1]
        # conv(a, w_flip) has length n + (2n-1) - 1 = 3n-2
        N_conv = _next_pow2(3 * n - 2)
        A = np.fft.rfft(a, N_conv)
        WF = np.fft.rfft(w_flip, N_conv)
        conv_awf = np.fft.irfft(A * WF, N_conv)
        # F[j] = 2 * conv_awf[2n-2-j] for j=0..n-1
        # positions: 2n-2, 2n-3, ..., n-1
        F = 2.0 * conv_awf[n - 1:2 * n - 1][::-1]

        # Target value: median or mean of F on support
        support = a > 1e-10
        if support.sum() > 0:
            nu = np.median(F[support])
        else:
            nu = np.mean(F)

        # Update a: push towards making F uniform
        a_new = a + alpha * (nu - F)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            no_improve = 0
            # Slightly increase alpha
            alpha = min(2.0 / n, alpha * 1.02)
        else:
            # Try the update anyway for the dual, but keep old a
            # Actually, let's still update a partially (soft step)
            a_mix = 0.5 * a + 0.5 * a_new
            a_mix = np.maximum(a_mix, 0.0)
            s_mix = a_mix.sum()
            if s_mix > 1e-15:
                a_mix = a_mix / s_mix
                c_mix = _full_convolve(a_mix, a_mix)
                sc_mix = 2.0 * n * float(c_mix.max())
                if sc_mix < 2.0 * n * float(c.max()) + 1e-14:
                    a = a_mix
                    c = c_mix
            no_improve += 1
            alpha *= 0.85

        # Update dual: concentrate w on peaks
        w_beta *= 0.97
        w_beta = max(w_beta, c.max() * 0.001)
        w = np.exp((c - c.max()) / w_beta)
        w_sum = w.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        iters += 1
        if iters % 200 == 0:
            print(f"    PD iter {iters}: score={best_sc:.10f}, alpha={alpha:.2e}, w_beta={w_beta:.2e}")

        if no_improve >= max_no_improve:
            # Restart with perturbation
            a = best_a.copy()
            c = _full_convolve(a, a)
            alpha = 0.3 / n
            no_improve = 0
            # Perturb
            a_pert = a + 0.01 * np.random.default_rng(iters).standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                sc_pert = _score(a_pert)
                if sc_pert < best_sc * 1.01:
                    a = a_pert
            c = _full_convolve(a, a)
            w = np.exp((c - c.max()) / max(c.max() * 0.01, 1e-10))
            w = w / w.sum()
            if no_improve >= 600:
                break

    return best_a, best_sc


def _active_set_pg(a_init, deadline, min_time_remaining=30.0):
    """Active-set projected gradient on max(c)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    alpha = 0.05 / n
    active_tol = 0.02
    no_improve = 0
    max_stall = 300
    iters = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = active_tol * c_max
        active_mask = c >= (c_max - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            ai = int(np.argmax(c))
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[ai] = True
            n_active = 1

        # Compute gradient via FFT
        w = np.zeros(2 * n - 1)
        w[active_mask] = 1.0 / n_active
        w_flip = w[::-1]
        N_conv = _next_pow2(3 * n - 2)
        A = np.fft.rfft(a, N_conv)
        WF = np.fft.rfft(w_flip, N_conv)
        conv_awf = np.fft.irfft(A * WF, N_conv)
        F = 2.0 * conv_awf[n - 1:2 * n - 1][::-1]

        # Project onto simplex tangent
        F_t = F - F.mean()
        gnorm = np.linalg.norm(F_t)
        if gnorm < 1e-15:
            break

        # Line search
        improved = False
        a_old_sc = current_sc
        for _ in range(50):
            a_try = a - alpha * F_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            alpha *= 0.7
            active_tol = max(0.001, active_tol * 0.8)
            if no_improve > max_stall:
                a = best_a.copy()
                alpha = 0.02 / n
                active_tol = 0.01
                no_improve = 0
                if no_improve > 600:
                    break
        else:
            alpha = min(0.2 / n, alpha * 1.05)
            active_tol = max(0.0005, active_tol * 0.98)

        iters += 1
        if iters % 500 == 0:
            print(f"    APG iter {iters}: score={best_sc:.10f}, alpha={alpha:.2e}, tol={active_tol:.4f}")

    return best_a, best_sc


def _local_search(a_init, deadline, min_time_remaining=30.0):
    """Local search: random pair swaps and small perturbations."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    no_improve = 0
    max_stall = 500
    iters = 0

    while time.monotonic() + min_time_remaining < deadline:
        iters += 1
        mode = iters % 3

        if mode == 0:
            # Pair swap: redistribute mass between two random positions
            i, j = rng.integers(0, n, 2)
            if i == j:

            total = a[i] + a[j]
            if total < 1e-15:
                continue
            frac = rng.uniform(0.0, 1.0)
            a_try = a.copy()
            a_try[i] = frac * total
            a_try[j] = (1.0 - frac) * total
        elif mode == 1:
            # Small Gaussian perturbation on a few elements
            k = rng.integers(3, max(4, n // 1000))
            idx = rng.integers(0, n, k)
            a_try = a.copy()
            sigma = 0.01 / n
            a_try[idx] += sigma * rng.standard_normal(k)
            a_try = np.maximum(a_try, 0.0)
        else:
            # Block swap: swap mass between two random blocks
            block_size = max(1, n // 500)
            i = rng.integers(0, n - block_size)
            j = rng.integers(0, n - block_size)
            if abs(i - j) <= block_size:
                continue
            a_try = a.copy()
            block_i = a_try[i:i + block_size].copy()
            block_j = a_try[j:j + block_size].copy()
            a_try[i:i + block_size] = block_j
            a_try[j:j + block_size] = block_i
            a_try = np.maximum(a_try, 0.0)

        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_stall:
                break

        if iters % 1000 == 0:
            print(f"    LS iter {iters}: score={best_sc:.10f}")

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fractal":
        # Self-similar structure
        a = np.ones(n)
        for level in range(int(np.log2(n))):
            block = n >> level
            if block < 2:
                break
            a[:block] *= 1.2
            a[-block:] *= 1.2
            a[block:-block] *= 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    """Try zeroing out the smallest elements to reduce effective support."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid primal-dual + L-BFGS + active-set + local search optimizer.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warm-up ===
    print(f"\n[P1] L-BFGS warm-up (t={time.monotonic()-t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 800, t_start + 300)
    a_main = _run_lbfgs(parent, lbfgs_deadline, min_time_remaining=100.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  L-BFGS: score={sc_main:.10f} (no improvement)")

    # === Phase 2: Primal-dual iteration ===
    print(f"\n[P2] Primal-dual (t={time.monotonic()-t_start:.1f}s}")
    if time.monotonic() < deadline - 600:
        pd_deadline = min(deadline - 600, time.monotonic() + 400)
        a_pd, sc_pd = _primal_dual(best_seq, pd_deadline, min_time_remaining=100.0)
        if sc_pd < best_score:
            best_seq = a_pd.copy()
            best_score = sc_pd
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [primal_dual] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  PD: score={sc_pd:.10f} (no improvement)")

    # === Phase 3: Active-set projected gradient ===
    print(f"\n[P3] Active-set PG (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        apg_deadline = min(deadline - 400, time.monotonic() + 200)
        a_apg, sc_apg = _active_set_pg(best_seq, apg_deadline, min_time_remaining=60.0)
        if sc_apg < best_score:
            best_seq = a_apg.copy()
            best_score = sc_apg
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [apg] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  APG: score={sc_apg:.10f} (no improvement)")

    # === Phase 4: More primal-dual (with different init) ===
    print(f"\n[P4] Primal-dual round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        # Perturb and run PD again
        a_pert = best_seq + 0.005 * rng.standard_normal(len(best_seq))
        a_pert = np.maximum(a_pert, 0.0)
        sp = a_pert.sum()
        if sp > 1e-15:
            a_pert = a_pert / sp
        pd_deadline2 = min(deadline - 300, time.monotonic() + 150)
        a_pd2, sc_pd2 = _primal_dual(a_pert, pd_deadline2, min_time_remaining=60.0)
        if sc_pd2 < best_score:
            best_seq = a_pd2.copy()
            best_score = sc_pd2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [pd2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Local search ===
    print(f"\n[P5] Local search (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        ls_deadline = min(deadline - 200, time.monotonic() + 100)
        a_ls, sc_ls = _local_search(best_seq, ls_deadline, min_time_remaining=50.0)
        if sc_ls < best_score:
            best_seq = a_ls.copy()
            best_score = sc_ls
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [local] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [n_parent]
    for nn in [16384, 12288, 8192, 6144, 4096]:
        if nn not in n_candidates:
            n_candidates.append(nn)
    n_candidates.sort(reverse=True)

    for n_try in n_candidates[:4]:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # Short L-BFGS
        if time.monotonic() < deadline - 120:
            lb_deadline = min(deadline - 120, time.monotonic() + 80)
            a_opt = _run_lbfgs(a_init, lb_deadline, min_time_remaining=60.0)
            sc_opt = _score(a_opt)

        # Short PD
        if time.monotonic() < deadline - 90:
            pd_deadline = min(deadline - 90, time.monotonic() + 50)
            a_opt2, sc_opt2 = _primal_dual(a_opt, pd_deadline, min_time_remaining=50.0)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2

        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Shrinkage ===
    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 80)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize
            if time.monotonic() < deadline - 60:
                a_shr2, sc_shr2 = _primal_dual(best_seq, min(deadline - 50, time.monotonic() + 40),
                                                min_time_remaining=30.0)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink_pd] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        fin_deadline = min(deadline - 10, time.monotonic() + 60)
        a_fin = _run_lbfgs(best_seq, fin_deadline, min_time_remaining=5.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final active-set polish ===
    if time.monotonic() < deadline - 5:
        a_fin2, sc_fin2 = _active_set_pg(best_seq, deadline - 2, min_time_remaining=1.0)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_apg] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
