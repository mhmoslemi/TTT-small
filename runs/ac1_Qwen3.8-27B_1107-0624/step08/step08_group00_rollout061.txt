"""
Optimized autocorrelation peak minimization using multiplicative weight mirror descent,
smoothed L-BFGS with aggressive cooling, and active-set linearized LP refinement.
Key innovations: FFT-based subgradient computation, adaptive multiplicative updates,
and LP-based direction finding for the active peak set.
"""
import numpy as np
import time
from scipy.optimize import minimize
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_max_conv(a, peak_indices):
    """Compute weighted subgradient of max convolution at given peaks.
    Returns g where g_i = 2 * sum_l w_l * a_{k_l - i}.
    """
    n = len(a)
    K = len(peak_indices)
    if K == 0:
        return np.zeros(n)
    # Build q with weights at peak positions
    q = np.zeros(2 * n - 1)
    c_vals = np.array(peak_indices)
    # Weights proportional to excess
    c_full = _full_convolve(a, a)
    c_at_peaks = c_full[c_vals]
    c_mean = c_full.mean()
    w = np.maximum(0, c_at_peaks - c_mean)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    q[c_vals] = w
    a_rev = a[::-1]
    g_full = _full_convolve(q, a_rev)
    g = g_full[n - 1:2 * n - 1]
    return 2.0 * g


def _mirror_descent(a_init, deadline, min_remaining=60.0):
    """Multiplicative weight mirror descent for peak reduction."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()

    K = min(200, max(20, n // 100))  # number of peaks in subgradient
    eta = 2.0 / (np.max(a) * np.sqrt(n) + 1e-10)
    eta = min(eta, 5.0)
    stall = 0
    max_stall = 200
    iteration = 0

    while time.monotonic() + min_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Reset with perturbation
                a = best_a.copy()
                z = np.log(np.maximum(a, 1e-30))
                z += 0.01 * np.random.default_rng(iteration).standard_normal(n)
                z -= z.mean()
                a = _softmax(z)
                stall = 0
                max_stall = int(max_stall * 0.8)
                if max_stall < 50:
                    break

        # Compute subgradient
        top_idx = np.argsort(c)[::-1][:K]
        g = _grad_max_conv(a, top_idx)

        # Multiplicative update with line search
        improved = False
        for ls in range(30):
            exp_arg = -eta * g
            exp_arg = np.clip(exp_arg, -50, 50)
            a_new = a * np.exp(exp_arg)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-10:
                eta *= 0.5
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < sc - 1e-14:
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            eta *= 0.5
            if eta < 1e-8:
                break

        # Occasionally increase eta if making progress
        if improved and iteration % 20 == 0:
            eta = min(eta * 1.1, 5.0)

        iteration += 1
        if iteration % 200 == 0:
            print(f"    MD iter {iteration}: score={best_sc:.10f}, eta={eta:.6f}, stall={stall}")

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_remaining=60.0):
    """L-BFGS with log-sum-exp smoothing and multi-scale beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # Aggressive cooling schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025,
                                  0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate time per iteration (scale with n)
        est_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(20, int(remaining / (est_per_iter * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if (i + 1) % 4 == 0:
                    print(f"    L-BFGS beta={beta:.2e}: score={current_sc:.10f}")
        except Exception as e:
            pass

    return current_a


def _active_set_lp_direction(a, active_indices):
    """Solve LP: min t s.t. c_j(a) + 2*sum a_{j-i}*d_i <= t, sum(d)=0, d>=-a.
    Returns direction d."""
    n = len(a)
    c = _full_convolve(a, a)
    K = len(active_indices)

    if not HAS_CVXPY:
        # Fallback: use gradient direction
        g = _grad_max_conv(a, active_indices)
        d = -g
        d -= d.mean()
        # Ensure d >= -a
        for i in range(n):
            if d[i] < -a[i]:
                d[i] = -a[i]
        d -= d.mean()
        return d

    d_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        d_var >= -a,
        cp.sum(d_var) == 0,
        t_var >= 0
    ]

    # Linearized constraints: c_j + 2 * (a_ref[j-:] dot d) <= t
    a_rev = a[::-1]
    for j in active_indices:
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j)
        # conv(a,d)_j = sum_i a_i * d_{j-i} for valid i
        # = sum_{i in [i_lo, i_hi]} a[i] * d[j-i]
        # j-i ranges from j-i_hi to j-i_lo
        idx_in_d = j - np.arange(i_lo, i_hi + 1)
        # d[idx_in_d] dotted with a[i_lo:i_hi+1]
        lin_term = a[i_lo:i_hi + 1] @ d_var[idx_in_d]
        constraints.append(c[j] + 2.0 * lin_term <= t_var)

    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None

    d = d_var.value
    if d is None or not np.all(np.isfinite(d)):
        return None
    return d

def _coordinate_descent_polish(a_init, deadline, min_remaining=30.0):

            a_test[idx] = test_x
            a_test = a_test / a_test.sum()
            sc_test = _score(a_test)

            if sc_test >= best_sc + 1e-14 and a_old > 0:
                # Try increasing
                test_x2 = a_old * 1.5
                a_test2 = a.copy()
                a_test2[idx] = test_x2
                a_test2 = a_test2 / a_test2.sum()
                sc_test2 = _score(a_test2)
                if sc_test2 < best_sc - 1e-14:
                    # Golden section in [a_old, a_old*3]
                    for _ in range(10):
                        gr = 0.618
                        x1 = x_hi - gr * (x_hi - x_lo)
                        x2 = x_lo + gr * (x_hi - x_lo)
                        a1 = a.copy(); a1[idx] = x1; a1 /= a1.sum()
                        a2 = a.copy(); a2[idx] = x2; a2 /= a2.sum()
                        if _score(a1) < _score(a2):
                            x_hi = x2
                        else:
                            x_lo = x1
                        if x_hi - x_lo < 1e-10:
                            break
                    x_best = (x_lo + x_hi) / 2
                else:
                    continue
            else:
                # Decreasing helps, search in [0, a_old]
                x_lo = 0.0
                x_hi = a_old
                for _ in range(10):
                    gr = 0.618
                    x1 = x_hi - gr * (x_hi - x_lo)
                    x2 = x_lo + gr * (x_hi - x_lo)
                    a1 = a.copy(); a1[idx] = x1; a1 /= a1.sum()
                    a2 = a.copy(); a2[idx] = x2; a2 /= a2.sum()
                    if _score(a1) < _score(a2):
                        x_hi = x2
                    else:
                        x_lo = x1
                    if x_hi - x_lo < 1e-10:
                        break
                x_best = (x_lo + x_hi) / 2

            if x_best != a_old:
                a[idx] = max(0, x_best)
                s_new = a.sum()
                if s_new < 1e-10:
                    a[idx] = a_old
                    continue
                a = a / s_new
                sc_new = _score(a)
                if sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_a = a.copy()
                    improved_any = True

        if not improved_any:
            no_improve += 1
            # Perturb a few coordinates
            if no_improve % 50 == 0:
                rng = np.random.default_rng(no_improve)
                perturb_idx = rng.choice(n_focus, size=min(10, n_focus), replace=False)
                for pi in perturb_idx:
                    a[pi] *= (1.0 + 0.05 * rng.standard_normal())
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _try_symmetrize(a):
    """Try making a symmetric and check if score improves."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-10:
        return a, _score(a)
    a_sym = a_sym / s
    sc_orig = _score(a)
    sc_sym = _score(a_sym)
    if sc_sym < sc_orig:
        return a_sym, sc_sym
    return a, sc_orig


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: mirror descent -> L-BFGS -> active-set LP -> coord descent -> multi-n.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except NameError:
        pass
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
        # Try symmetrization
        best_seq, best_score = _try_symmetrize(best_seq)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Quick L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 1200:
        a1 = _run_lbfgs(best_seq, min(deadline - 1200, t_start + 200), min_remaining=60.0)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_seq, best_score = a1, sc1
            print(f"  P1: score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 2: Mirror Descent ===
    print(f"\n[P2] Mirror Descent (t={time.monotonic()-t_start:.1f}s)")
    md_deadline = min(deadline - 600, time.monotonic() + 500)
    a2, sc2 = _mirror_descent(best_seq, md_deadline, min_remaining=300.0)
    if sc2 < best_score:
        best_seq, best_score = a2, sc2
        print(f"  P2: score={best_score:.10f} (n={len(best_seq)})")
    else:
        print(f"  P2: no improvement (best={best_score:.10f})")

    # === Phase 3: L-BFGS round 2 ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        lbfgs_dl = min(deadline - 400, time.monotonic() + 300)
        a3 = _run_lbfgs(best_seq, lbfgs_dl, min_remaining=200.0)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq, best_score = a3, sc3
            print(f"  P3: score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 4: Active Set LP ===
    print(f"\n[P4] Active Set LP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        as_dl = min(deadline - 200, time.monotonic() + 250)
        a4, sc4 = _active_set_phase(best_seq, as_dl, min_remaining=100.0)
        if sc4 < best_score:
            best_seq, best_score = a4, sc4
            print(f"  P4: score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 5: Second L-BFGS pass ===
    print(f"\n[P5] L-BFGS round 3 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        lbfgs_dl2 = min(deadline - 150, time.monotonic() + 150)
        a5 = _run_lbfgs(best_seq, lbfgs_dl2, min_remaining=80.0)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_seq, best_score = a5, sc5
            print(f"  P5: score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 6: Mirror Descent round 2 (finer) ===
    print(f"\n[P6] MD round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        md_dl2 = min(deadline - 100, time.monotonic() + 100)
        a6, sc6 = _mirror_descent(best_seq, md_dl2, min_remaining=50.0)
        if sc6 < best_score:
            best_seq, best_score = a6, sc6
            print(f"  P6: score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 7: Active Set LP round 2 ===
    print(f"\n[P7] AS-LP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        as_dl2 = min(deadline - 80, time.monotonic() + 60)
        a7, sc7 = _active_set_phase(best_seq, as_dl2, min_remaining=30.0)
        if sc7 < best_score:
            best_seq, best_score = a7, sc7
            print(f"  P7: score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n_parent, 8192, 12288, 16384, 24576]))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        # Quick L-BFGS
        if time.monotonic() < deadline - 50:
            a_opt = _run_lbfgs(a_init, min(deadline - 50, time.monotonic() + 60), min_remaining=20.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Ultra-fine L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.00003
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _LbfgsProb(beta_fin, n)
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100, 'maxls': 50})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq, best_score = a_fin, sc_fin
                    print(f"  Final L-BFGS: {best_score:.10f}")
            except Exception:
                pass

        # Final active set
        if time.monotonic() < deadline - 3:
            a_as, sc_as = _active_set_phase(best_seq, deadline - 2, min_remaining=1.0)
            if sc_as < best_score:
                best_seq, best_score = a_as, sc_as
                print(f"  Final AS: {best_score:.10f}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"  evaluate_sequence confirms: {ev_score:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
