def lp_direction(f, M):
    n = len(f)
    b = cp.Variable(n)
    # Constraints: for each lag k, sum_j b[j] * f[k-j] <= M
    # f[k-j] is nonzero when 0 <= k-j < n, i.e., k-(n-1) <= j <= k
    # Build the constraint matrix A where A[k,j] = f[k-j] if valid, else 0
    # But this matrix is (2n-1) x n, which for n=8192 is 16383 x 8192 ≈ 134M entries
    # That's feasible but might be slow.
    
    constraints = [b >= 0]
    for k in range(2*n - 1):
        # sum_j b[j] * f[k-j] <= M
        # j range: max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:

            constraints.append(coeffs @ b[lo:hi+1] <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or CVXOPT, or ECOS

"""
Improved step function optimization using LP-based direction finding with
cutting planes, combined with accelerated gradient methods and multi-scale
approach. Focuses on n=8192 with careful active-set handling.
"""
import numpy as np
import time
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

Aims to minimize 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _lp_direction(a, M, max_cutting_iters=30, max_constraints=300):
    """Solve LP to find direction b that maximizes sum(b) s.t. conv(a,b) <= M, b>=0.
    Uses cutting planes for efficiency."""
    n = len(a)
    
    # Initial constraint set: top lags
    c = _fft_convolve(a, a)
    sorted_lags = np.argsort(c)[::-1][:min(max_constraints, 2*n-1)]
    active_lags = set(sorted_lags.tolist())
    
    b_sol = None
    for it in range(max_cutting_iters):
        # Build constraint matrix
        lag_list = sorted(active_lags)
        # A[k_idx, j] = a[k - j] if 0 <= k-j < n, else 0
        # For efficiency, build as list of (row, col, val)
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(lag_list):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            j_range = range(lo, hi + 1)
            # a[k-j] for j in j_range
            # when j=lo: k-lo, when j=hi: k-hi
            # So a[k-hi : k-lo+1] reversed
            coeffs = a[k-hi:k-lo+1][::-1]
            for jj, j in enumerate(j_range):
                if coeffs[jj] > 1e-15:
                    rows.append(idx)
                    cols.append(j)
                    vals.append(coeffs[jj])
        
        if not rows:
            return None
        
        # Use scipy.sparse for efficiency
        from scipy.sparse import csr_matrix
        A = csr_matrix((vals, (rows, cols)), shape=(len(lag_list), n))
        b_ub = np.full(len(lag_list), M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True})
        
        if result.status != 0:
            return None
        
        b_sol = np.maximum(0, result.x)
        
        # Check violations
        c_b = _fft_convolve(a, b_sol)
        violated = np.where(c_b > M * (1 + 1e-8))[0]
        violated = [v for v in violated if v not in active_lags]
        
        if not violated:
            return b_sol
        
        # Add most violated constraints
        violated_sorted = sorted(violated, key=lambda k: c_b[k], reverse=True)
        for v in violated_sorted[:30]:
            active_lags.add(int(v))
        
        # Cap total constraints
        if len(active_lags) > max_constraints:
            # Keep the most important ones
            c_active = np.array([c[k] for k in active_lags])
            # Actually, keep based on violation severity
            c_b_active = np.array([c_b[k] for k in active_lags])
            # Keep top max_constraints by c_b value
            keep = list(active_lags)
            keep_sorted = sorted(keep, key=lambda k: c_b[k], reverse=True)
            active_lags = set(keep_sorted[:max_constraints])
    
    return b_sol

def _lp_guided_iteration(a, deadline, max_iters=100, lr_decay=0.95):
    """Run LP-guided iterations."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_m = float(_fft_convolve(a, a).max())
    
    lr = 0.5
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = float(_fft_convolve(a, a).max())
        
        # Solve LP
        b = _lp_direction(a, M, max_cutting_iters=20, max_constraints=200)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum  # normalize to sum=1
        
        # Line search: try different step sizes
        improved = False
        for t in [lr, lr*0.7, lr*0.5, lr*0.3, lr*0.1]:
            a_cand = (1-t)*a + t*b
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            c_cand = _fft_convolve(a_cand, a_cand)
            m_cand = float(c_cand.max())
            if m_cand < best_m * (1 - 1e-12):
                best_m = m_cand
                best_a = a_cand.copy()
                a = a_cand.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            lr *= lr_decay
            if no_improve > 10 or lr < 0.01:
                break
    
    return best_a, best_m

def _projected_gradient(a, deadline, max_iters=5000, initial_lr=0.01):
    """Projected gradient method on the simplex."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_m = float(_fft_convolve(a, a).max())
    lr = initial_lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active lags (top k with c close to max)
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:min(10, 2*n-1)]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project gradient to tangent space (sum(d) = 0)
        g_mean = g.sum() / n
        g_tan = g - g_mean
        
        # Step
        a_new = a - lr * g_tan
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        m_new = float(_fft_convolve(a_new, a_new).max())
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr *= 0.999  # very slow decay
        else:
            lr *= 0.95
            if lr < 1e-8:
                break
    
    return best_a, best_m

def _adam_logspace(a_init, deadline, max_iters=100000, lr=0.01, T=0.001):
    """Adam optimizer in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_m = float(_fft_convolve(a_init, a_init).max())
    
    while t < max_iters:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Gradient: use top few lags
        sorted_idx = np.argsort(c)[::-1]
        num_active = min(5, 2*n-1)
        g = np.zeros(n)
        for idx in range(num_active):
            k = sorted_idx[idx]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= num_active
        
        # Center the gradient
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        m_buf = b1 * m_buf + (1-b1) * z_grad
        v_buf = b2 * v_buf + (1-b2) * z_grad**2
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        z_new = z - update
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % 100 == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            m_cur = float(_fft_convolve(a_cur, a_cur).max())
            if m_cur < best_m:
                best_m = m_cur
                best_a = a_cur.copy()
    
    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    m_final = float(_fft_convolve(a_final, a_final).max())
    if m_final < best_m:
        best_m = m_final
        best_a = a_final.copy()
    
    return best_a, best_m
