"""
Step-function optimization targeting score < 1.5027.
Uses multi-phase strategy: aggressive L-BFGS annealing, active-set 
gradient descent with momentum, spectral deflation, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _pad_len(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return k

def _conv(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    N = _pad_len(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n+m-1]

def _score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return 2.0 * n * c.max() / (s * s)

def _softmax(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    return e / s if s > 0 else np.ones_like(z) / len(z)

def _lse_obj_grad(z, beta, n):
    """Log-sum-exp objective and gradient for the score."""
    a = _softmax(z)
    c = _conv(a, a)
    cmax = c.max()
    # LSE approximation of max
    shifted = (c - cmax) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        obj = 2.0 * n * cmax
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        obj = 2.0 * n * (cmax + beta * np.log(s))
        w = e / s
    # Gradient
    a_rev = a[::-1]
    wc = _conv(w, a_rev)
    h = wc[n-1:2*n-1]
    W = np.dot(w, c)
    grad = 4.0 * n * a * (h - W)
    return obj, grad

class _LseFn:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _lse_obj_grad(z, self.beta, self.n)[0]
    def grad(self, z):
        return _lse_obj_grad(z, self.beta, self.n)[1]

def _pad_len(n):
    k = 1
    while k < n:
        k <<= 1

def _conv(a, b):
    n, m = len(a), len(b)
    N = _pad_len(n + m - 1)

def _score(a):

def _softmax(z):
    z = z - z.max()

    s = e.sum()

    if s < 1e-300 or not np.isfinite(s):

def _lse_grad(z, beta, n):

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
