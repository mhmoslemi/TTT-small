def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_active(c, k):

    if k >= len(c):
        return np.ones(len(c), dtype=bool)
    thresh = np.partition(c, len(c) - k)[len(c) - k]
    return c >= thresh

        if gnorm < 1e-15:
            break

        # Adaptive mu: start small, increase if CG struggles
        mu = 1e-10
        for mu_attempt in range(5):
            def _hv(v):
                return _hess_matvec(v, active_f, n) + mu * v
            dx = _cg_solve(_hv, -grad_t, 0.0, tol=1e-12, max_iter=200)
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            # Check if direction is descent
            if dx_norm > 1e-30:
                dir_grad = float(np.dot(dx_t, grad_t)) if hasattr(dx, 't') else float(np.dot(dx, grad_t))
                if dir_grad < -1e-20:
                    break
            mu *= 10.0
        else:
            # Fallback: steepest descent
            dx = -grad_t
            dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # Try a tiny step in negative gradient direction
            a_try = a - 1e-6 * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try[active].sum()) < f_old:
                    a_new = a_try
                else:
                    break
            else:
                break

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _newton_topk_only(a_init, deadline, max_iter=300, k=20):
    """Active-set Newton using exactly top-k peaks."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Use exactly top-k peaks
        active = _topk_active(c, k)
        active_f = active.astype(np.float64)

        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        def _hv(v):
            return _hess_matvec(v, active_f, n) + mu * v
        dx = _cg_solve(_hv, -grad_t, 0.0, tol=1e-12, max_iter=200)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

# Subgradient with adaptive step
# ============================================================

def _subgrad_polish(a_init, deadline, n_steps=10000, lr_init=0.005, tol_frac=0.001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                best_patience_sc = sc
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr *= 0.85
                    if lr < 1e-6:
                        break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# LSE-LBFGS with very aggressive beta schedule
# ============================================================

def _lbfgs_lse_aggressive(a_init, deadline, min_remaining=30.0, n_stages=20):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.01)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # More aggressive: exponentially decaying betas
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                   0.015, 0.007, 0.0035, 0.0018,
                                   0.0009, 0.00045, 0.0002, 0.0001,
                                   0.00005, 0.00002, 0.00001,
                                   0.000005, 0.000002, 0.000001])
    betas = betas[:n_stages]

    prev_sc = current_sc
    stall_count = 0

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate per-iteration time
        est_time = 0.005 * max(1.0, n / 8192.0)
        # For small beta, allow more iterations
        iter_factor = max(1.0, 1.0 / max(beta / base_beta, 0.0001))
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 3:
                    break
        except Exception:
            stall_count += 1
            if stall_count > 3:
                break

    return current_a, current_sc

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.05):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block_swap(a, rng, n_swaps=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)

# Profile generators

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Aggressive multi-phase optimizer: LSE-LBFGS with tiny betas,
    active-set Newton (adaptive + top-k), subgradient polish,
    diverse restarts, and final concentrated polishing.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        upd(parent, _score(parent), "init_uniform")
        n_parent = len(parent)

    # === Phase 1: Aggressive LSE with small betas (600s) ===
    print(f"\n[Phase 1] Aggressive LSE-LBFGS")
    phase1_deadline = min(deadline, t_start + 600)
    if time.monotonic() < phase1_deadline:
        a_p1, sc_p1 = _lbfgs_lse_aggressive(_best_seq.copy(), phase1_deadline, min_remaining=30, n_stages=18)
        upd(a_p1, sc_p1, "lse_aggressive")

    # === Phase 2: Active-set Newton (adaptive tolerance) (300s) ===
    print(f"\n[Phase 2] Active-set Newton (adaptive)")
    phase2_deadline = min(deadline, t_start + 900)
    if time.monotonic() < phase2_deadline:
        # Multiple passes with decreasing tolerance
        tol_levels = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        a_cur = _best_seq.copy()
        for li, tol in enumerate(tol_levels):
            if time.monotonic() > phase2_deadline - 10:
                break
            seg_deadline = min(phase2_deadline - 10, time.monotonic() + 50)
            a_nw, sc_nw = _newton_active_set(a_cur, seg_deadline, max_iter=200, tol_frac=tol, topk_frac=0.03)
            if sc_nw < _score(a_cur) - 1e-14:
                a_cur = a_nw
            upd(a_cur, _score(a_cur), f"newton_tol{tol}")

    # === Phase 3: Top-k Newton (200s) ===
    print(f"\n[Phase 3] Top-k Newton")
    phase3_deadline = min(deadline, t_start + 1100)
    if time.monotonic() < phase3_deadline:
        for k in [10, 20, 50, 100]:
            if time.monotonic() > phase3_deadline - 5:
                break
            seg_deadline = min(phase3_deadline - 5, time.monotonic() + 40)
            a_nw, sc_nw = _newton_topk_only(_best_seq, seg_deadline, max_iter=200, k=k)
            upd(a_nw, sc_nw, f"newton_topk_{k}")

    # === Phase 4: Subgradient polish (200s) ===
    print(f"\n[Phase 4] Subgradient polish")
    phase4_deadline = min(deadline, t_start + 1300)
    if time.monotonic() < phase4_deadline:
        a_sg, sc_sg = _subgrad_polish(_best_seq, phase4_deadline, n_steps=10000, lr_init=0.004, tol_frac=0.0005)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 5: Final aggressive LSE with tiny beta (150s) ===
    print(f"\n[Phase 5] Final tiny-beta LSE")
    phase5_deadline = min(deadline, t_start + 1450)
    if time.monotonic() < phase5_deadline:
        n = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        # Use extremely small beta
        for beta_final in [mb * 0.0001, mb * 0.00003, mb * 0.00001]:
            if time.monotonic() > phase5_deadline - 5:
                break
            remaining = phase5_deadline - 5 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(3000, max(100, int(remaining / 0.01)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18,
                             'gtol': 1e-16, 'maxcor': 100, 'maxls': 200}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < _best_score - 1e-15:
                    z_fin = np.log(np.maximum(a_fin, 1e-30))
                    z_fin = z_fin - z_fin.mean()
                upd(a_fin, sc_fin, f"lse_beta{beta_final:.2e}")
            except Exception:
                pass

    # === Phase 6: Diverse restarts (150s) ===
    print(f"\n[Phase 6] Diverse restarts")
    phase6_deadline = min(deadline, t_start + 1600)
    if time.monotonic() < phase6_deadline:
        restarts = [
            (lambda: _perturb_log(_best_seq, 0.01, rng), "pert_log_001"),
            (lambda: _perturb_log(_best_seq, 0.02, rng), "pert_log_002"),
            (lambda: _perturb_fourier_phase(_best_seq, rng, n_modes=5, max_amp=0.15), "pert_phase"),
            (lambda: _perturb_amplitude(_best_seq, rng, sigma=0.03), "pert_amp"),
            (lambda: _perturb_block_swap(_best_seq, rng, n_swaps=3), "pert_block"),
        ]
        for gen_func, label in restarts:
            if time.monotonic() > phase6_deadline - 20:
                break
            try:
                a_start = gen_func()
                # Quick LSE
                seg_dead = min(phase6_deadline - 15, time.monotonic() + 25)
                a_r, sc_r = _lbfgs_lse_aggressive(a_start, seg_dead, min_remaining=5, n_stages=8)
                upd(a_r, sc_r, f"restart_{label}")
                # Quick Newton
                if _score(a_r) < _best_score * 1.005:
                    nw_dead = min(phase6_deadline - 10, time.monotonic() + 15)
                    a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=100, tol_frac=0.001, topk_frac=0.02)
                    upd(a_r2, sc_r2, f"restart_{label}_nw")
            except Exception:
                pass

    # === Phase 7: Final concentrated Newton (100s) ===
    print(f"\n[Phase 7] Final concentrated Newton")
    if _best_seq is not None and time.monotonic() < deadline - 10:
        # Multiple Newton passes with very tight tolerance
        a_fin = _best_seq.copy()
        for pass_i in range(3):
            seg_dead = min(deadline - 5, time.monotonic() + 30)
            a_nw, sc_nw = _newton_active_set(a_fin, seg_dead, max_iter=300, tol_frac=0.00005, topk_frac=0.01)
            if sc_nw < _score(a_fin) - 1e-15:
                a_fin = a_nw
            upd(a_fin, _score(a_fin), f"final_newton_{pass_i}")

        # Top-k with very small k
        seg_dead = min(deadline - 3, time.monotonic() + 20)
        a_nw, sc_nw = _newton_topk_only(a_fin, seg_dead, max_iter=200, k=5)
        upd(a_nw, sc_nw, "final_topk5")

        # Final subgradient with very small tolerance
        seg_dead = min(deadline - 2, time.monotonic() + 15)
        a_sg, sc_sg = _subgrad_polish(a_fin, seg_dead, n_steps=5000, lr_init=0.002, tol_frac=0.0001)
        upd(a_sg, sc_sg, "final_sg_tight")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        # Verify with the actual evaluator
        actual_sc = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: internal={sc_final:.10f}, actual={actual_sc:.10f}, "
              f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        if actual_sc < sc_final:
            sc_final = actual_sc
        if not np.isfinite(sc_final):
            print("WARNING: non-finite score, using internal")
            sc_final = sc_final
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(2048, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
