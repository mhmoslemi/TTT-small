"""
Optimized minimax convolution optimizer.
Uses tilted parametric initialization, aggressive log-sum-exp smoothing,
tight active-set Newton refinement, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _tilted_profile(n, kind="cos_taper", tilt=0.0):
    t = np.linspace(0, 1, n)
    if kind == "cos_taper":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "sin_taper":
        a = np.sin(np.pi * t) ** 0.8
    elif kind == "gauss":
        a = np.exp(-4.0 * (t - 0.5) ** 2)
    elif kind == "uniform":
        a = np.ones(n)
    elif kind == "cube_taper":
        a = (1.0 - (2*t - 1)**2) ** 0.5  # semi-circle
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    # Apply tilt
    if tilt != 0.0:
        a = a * (1.0 + tilt * (2.0 * t - 1.0))
        a = np.maximum(a, 1e-10)
    return a / a.sum()

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

class _SmoObjAdapter:

def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

        else:
            tol_frac = max(tight_tol, 1e-5 * (1 - 0.5 * (progress - 0.5) / 0.5))

def _resample(a, n_new):

    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())

    return float(2.0 * n * c.max() / (s * s))

        a = np.sin(np.pi * t) ** 0.8
    elif kind == "gauss":
        a = np.exp(-4.0 * (t - 0.5) ** 2)
    elif kind == "uniform":
        a = np.ones(n)
    elif kind == "cube_taper":
        a = np.sqrt(np.maximum(1.0 - (2.0 * t - 1.0) ** 2, 0.0))
    elif kind == "pow_taper":
        a = np.sin(np.pi * t) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    if tilt != 0.0:
        a = a * (1.0 + tilt * (2.0 * t - 1.0))
        a = np.maximum(a, 1e-10)
    return a / a.sum()


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


class _SmoObjAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)


class _SmoGradAdapter:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smograd(z, self.beta, self.n)


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=600, tol=1e-15):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_polish(a_init, deadline, max_rounds=300, tight_tol=1e-9):
    """Newton method with progressive active set tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    stagnant = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Progressive tightening
        frac = rnd / max(1, max_rounds)
        if rnd < 5:
            tol_frac = 5e-3
        elif rnd < 20:
            tol_frac = 1e-3
        elif rnd < 50:
            tol_frac = 1e-4
        elif rnd < 100:
            tol_frac = 1e-5
        else:
            tol_frac = max(tight_tol, 1e-6 * (1.0 - 0.5 * frac))

        active = c >= (c_max - tol_frac * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)

        ones_H1 = float(np.ones(n) @ H1)
        if abs(ones_H1) > 1e-20:
            lam = float(np.ones(n) @ Hg) / ones_H1
            dx = -Hg + lam * H1
        else:
            dx = -g_t / max(gn, 1e-30)
        dx = dx - np.mean(dx)
        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            dx = -g_t / max(gn, 1e-30)
            dx_norm = max(gn, 1e-30)

        # Line search
        alpha = 1.0
        step_done = False
        c_max_old = c_max
        for ls in range(120):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_old - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-20:
                break

        if not step_done:
            for trial in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_old - 1e-16:
                    a = a_try
                    step_done = True
                    break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if not step_done and stagnant > 15:
            break

    return best_a, best_sc


def _adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Adaptive smoothed minimax with L-BFGS and aggressive beta schedule."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(100, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    beta = max(gap * 0.3, max_c * 0.002)
    min_beta = max_c * 1e-13

    iteration_count = 0
    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_time_per_iter = 0.0015 * scale
        max_iter = min(15000, max(30, int(remaining * 0.65 / (est_time_per_iter * 25))))

        try:
            obj_fn = _SmoObjAdapter(beta, n)
            grad_fn = _SmoGradAdapter(beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxcor': 120, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-18:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            iteration_count += 1
        except Exception:
            pass

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]

        top_k = min(30, len(c_sorted_new))
        top_spread = c_sorted_new[0] - c_sorted_new[top_k - 1]
        rel_spread = top_spread / max(c_max_new, 1e-30)

        if rel_spread < 1e-9:
            beta *= 0.03
        elif rel_spread < 1e-7:
            beta *= 0.08
        elif rel_spread < 1e-5:
            beta *= 0.15
        elif rel_spread < 1e-3:
            beta *= 0.25
        else:
            beta *= 0.4

        beta = min(beta, c_max_new * 0.0005)

        if time.monotonic() > deadline - min_time_remaining:
            break

    return current_a


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _subgradient_step(a, W, W_val, n, lr):
    """Single subgradient step in log-domain."""
    c = _full_convolve(a, a)
    w = np.zeros_like(c)
    w[active_mask = c >= (c.max() - 1e-4 * c.max())] = 1.0 / active_mask.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W_val)
    return grad


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # === Phase 1: Diverse initialization ===
    print(f"\n[Phase 1] Initialization (t={time.monotonic() - t_start:.1f}s)")

    if parent is None:
        n_parent = 16384
        parent = _tilted_profile(n_parent, "cos_taper", 0.05)
        update_best(parent, _score(parent), "default_init")

    best_init = parent.copy()
    best_init_sc = _score(parent)

    # Try various tilts and profiles at parent size
    for tilt in [-0.12, -0.08, -0.04, -0.02, 0.0, 0.02, 0.04, 0.06, 0.08, 0.12]:
        for kind in ["cos_taper", "sin_taper", "gauss", "cube_taper"]:
            if time.monotonic() > deadline - 1500:
                break
            a_try = _tilted_profile(n_parent, kind, tilt)
            sc_try = _score(a_try)
            if sc_try < best_init_sc:
                best_init_sc = sc_try
                best_init = a_try.copy()

    update_best(best_init, best_init_sc, "best_profile")

    # Perturbed parent
    for sigma in [0.03, 0.07, 0.12]:
        if time.monotonic() > deadline - 1400:
            break
        a_p = _perturb(parent, sigma, rng)
        sc_p = _score(a_p)
        if sc_p < best_init_sc:
            best_init_sc = sc_p
            best_init = a_p

    # Quick L-BFGS on best profile
    if time.monotonic() < deadline - 1200:
        a_quick = _adaptive_lbfgs(best_init, min(deadline - 1200, time.monotonic() + 200),
                                   min_time_remaining=1000.0)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "quick_lbfgs")
        if sc_quick < best_init_sc:
            best_init = a_quick
            best_init_sc = sc_quick

    # === Phase 2: Main L-BFGS optimization ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    start_a = best_seq if best_seq is not None else best_init
    a_main = _adaptive_lbfgs(start_a, deadline, min_time_remaining=500.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Newton polish (first pass) ===
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_np, sc_np = _newton_polish(best_seq, min(deadline - 300, time.monotonic() + 400),
                                       max_rounds=200, tight_tol=1e-8)
        update_best(a_np, sc_np, "newton_1")

    # === Phase 4: L-BFGS + Newton cycling ===
    print(f"\n[Phase 4] L-BFGS+Newton cycles (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        for cycle in range(5):
            if time.monotonic() + 80 > deadline:
                break
            # Short L-BFGS burst
            t_end_lb = min(time.monotonic() + 50, deadline - 80)
            if time.monotonic() < t_end_lb:
                a_lb = _adaptive_lbfgs(best_seq, t_end_lb, min_time_remaining=40.0)
                sc_lb = _score(a_lb)
                if sc_lb < best_score:
                    best_seq = a_lb.copy()
                    best_score = sc_lb
                    print(f"  cycle {cycle} L-BFGS: {sc_lb:.10f}")

            # Newton polish
            if time.monotonic() + 40 < deadline:
                t_end_np = min(time.monotonic() + 60, deadline - 20)
                a_np2, sc_np2 = _newton_polish(best_seq, t_end_np,
                                                 max_rounds=100, tight_tol=1e-9)
                update_best(a_np2, sc_np2, f"cycle_{cycle}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        base_n = len(best_seq)
        n_targets = sorted(set([base_n // 2, base_n, base_n * 2, 8192, 32768]))
        n_targets = [x for x in n_targets if 2048 <= x <= 65536 and x != base_n]

        for n_try in n_targets[:3]:
            if time.monotonic() > deadline - 50:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _adaptive_lbfgs(a_res, deadline, min_time_remaining=30.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")
            if sc_opt < best_score + 0.0005 and time.monotonic() < deadline - 15:
                a_np3, sc_np3 = _newton_polish(a_opt, deadline - 5, max_rounds=50, tight_tol=1e-9)
                update_best(a_np3, sc_np3, f"n_{n_try}_np")

    # === Phase 6: Diverse restarts ===
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 25:
        for i in range(4):
            if time.monotonic() > deadline - 12:
                break
            sigma = 0.02 + 0.02 * i
            a_pert = _perturb(best_seq, sigma, rng)
            a_opt = _adaptive_lbfgs(a_pert, deadline, min_time_remaining=8.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"restart_{i}")
            if sc_opt < best_score + 0.0005 and time.monotonic() < deadline - 6:
                a_np4, sc_np4 = _newton_polish(a_opt, deadline - 2, max_rounds=30, tight_tol=1e-10)
                update_best(a_np4, sc_np4, f"restart_{i}_np")

    # === Phase 7: Final ultra-fine Newton ===
    print(f"\n[Phase 7] Final ultra-fine Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 1:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 0.1, max_rounds=500, tight_tol=1e-11)
        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS at ultra-tiny beta
        if best_seq is not None and time.monotonic() < deadline - 1:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-12
            remaining = max(0.5, deadline - 1 - time.monotonic())
            if remaining > 1:
                max_iter = min(5000, max(50, int(remaining / 0.005)))
                try:
                    obj_fn = _SmoObjAdapter(beta_fin, n)
                    grad_fn = _SmoGradAdapter(beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-22, 'gtol': 1e-18,
                                 'maxcor': 120, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Final Newton after last L-BFGS
        if best_seq is not None and time.monotonic() < deadline - 0.1:
            a_veryfin, sc_veryfin = _newton_polish(best_seq, deadline - 0.01, max_rounds=200, tight_tol=1e-12)
            update_best(a_veryfin, sc_veryfin, "very_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _tilted_profile(8192, "cos_taper", 0.05)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _tilted_profile(8192, "cos_taper", 0.05)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
