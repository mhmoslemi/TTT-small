"""
Optimized step function search using L-BFGS with log-sum-exp smoothing,
FFT-based gradient computation, and multi-start strategy.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

# === Helper Functions ===

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve_full(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = rfft(a, N)
    return irfft(A * A, N)[:2 * n - 1]

def _cross_corr(p, a):
    """Compute (p cross_corr a)[m] = sum_l p[l+m] * a[l] for m = 0..n-1.
   .Here, p has length L, a has length n.
    """
    n = len(a)
    L = len(p)
    N = _next_pow2(L + n - 1)
    P = rfft(p, N)
    A = rfft(a, N)
    result = irfft(P * np.conj(A), N)
    return result[:n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_full(a)
    return float(2.0 * n * c.max() / (s * s))

def _obj_grad(a, T):
    """Objective and gradient for L-BFGS.
    Minimizes f(a) = 2n * log(sum exp(c/T)) / (sum a)^2 subject to a >= 0.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    c = _convolve_full(a)
    c_max = float(c.max())
    
    # Stable log-sum-exp
    c_shifted = (c - c_max) / T
    c_shifted = np.clip(c_shifted, -700.0, 700.0)
    e = np.exp(c_shifted)
    log_sum_exp = c_max / T + np.log(e.sum())
    
    obj = 2.0 * n * log_sum_exp / (s * s)
    
    # p[k] = exp((c[k] - c_max) / T) / sum exp((c[k'] - c_max) / T)
    p = e / e.sum()
    
    # Gradient of log_sum_exp w.r.t. a[m]:
    # = (1/T) * sum_k p[k] * dc[k]/da[m]
    # = (1/T) * sum_k p[k] * 2 * a[k-m] (for valid k-m)
    # = (2/T) * sum_k p[k] * a[k-m]
    # = (2/T) * (p cross_corr a)[m]
    
    xcorr = _cross_corr(p, a)  # xcorr[m] = sum_l p[l+m] * a[l] = sum_k p[k] * a[k-m]
    
    grad_lse = (2.0 / T) * xcorr
    
    # Full gradient of f = 2n * lse / s^2:
    # df/da[m] = (2n/s^2) * grad_lse[m] - (2n * lse * 2s) / s^4
    # = (2n/s^2) * (grad_lse[m] - 2*lse/s)
    
    grad = (2.0 * n / (s * s)) * (grad_lse - 2.0 * log_sum_exp / s)
    
    return float(obj), grad

# === Optimization Routine ===

def _l_bfgs_optimize(a_init, T, max_iter=5000, tol=1e-12, time_limit=60.0):
    """Run L-BFGS-B on the smoothed objective."""
    n = len(a_init)
    t0 = time.monotonic()
    
    def fun(x):
        return _obj_grad(x, T)[0]
    
    def jac(x):
        return _obj_grad(x, T)[1]
    
    # Actually, let's combine fun and jac for efficiency
    cache = {'val': None, 'grad': None}
    
    def fun_wrapper(x):
        cache['val'], cache['grad'] = _obj_grad(x, T)
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        return cache['val']
    
    def jac_wrapper(x):
        if cache['grad'] is None:
            cache['val'], cache['grad'] = _obj_grad(x, T)
        return cache['grad']
    
    try:
        result = minimize(
            fun_wrapper, a_init,
            method='L-BFGS-B',
            jac=jac_wrapper,
            bounds=[(0.0, None)] * n,
            options={'maxiter': max_iter, 'gtol': tol, 'maxfun': 100000,
                     ' Maksys': 20}
        )
        return result.x
    except (TimeoutError, Exception) as ex:
        print(f"  L-BFGS terminated: {ex}")
        return a_init

def _peak_shave(a, deadline, max_iter=10000, top_k=5):
    """Direct peak shaving: iteratively reduce the max autocorrelation peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_cmax = float(_convolve_full(a).max())
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_full(a)
        c_max = float(c.max())
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # Find active lags (close to max)
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        # Only include those very close to max
        active = [k for k in active if c[k] > c_max * 0.999]
        if not active:
            active = [sorted_idx[0]]
        
        # Compute subgradient (average over active lags)
        g = np.zeros(n)
        for k in active:
            # dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n
            # g[m] += 2 * a[k-m]
            # This is: for each k, add 2*a shifted by k
            # a[k-m] for m = 0..n-1 means: a[k], a[k-1], ..., a[k-(n-1)] (valid parts)
            # = a[max(0,k-n+1):k+1][::-1] placed at appropriate position
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        g /= len(active)
        
        # Project to tangent space (preserve sum)
        g -= g.mean()  # Actually, we should project properly
        # For simplicity, subtract the mean times a (approximate)
        # Better: g_tangent = g - (sum(g * a) / sum(a^2)) * a
        # But even simpler: just subtract mean of g
        g_tangent = g - np.dot(a, g) / np.dot(a, a) * a
        
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        lr = 0.01 * (0.999 ** it)
        a_new = a - lr * g_tangent / g_norm * c_max
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        # Check improvement
        c_new = _convolve_full(a_new)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - 0.001 * c_max * g_tangent / g_norm
            a_new2 = np.maximum(0, a_new2)
            s2 = a_new2.sum()
            if s2 < 1e-15:
                break
            a_new2 /= s2
            c_new2 = _convolve_full(a_new2)
            if c_new2.max() < c_max:
                a = a_new2
            else:
                if lr < 1e-10:
                    break
    
    return best_a, 2.0 * n * float(_convolve_full(best_a).max()) / float(best_a.sum())**2

def _random_perturb(a, sigma, rng):
    """Add random perturbation in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z += sigma * rng.standard_normal(n)
    z -= z.mean()
    # Softmax back
    z = z - z.max()
    a_new = np.exp(z)
    a_new /= a_new.sum()
    return a_new
