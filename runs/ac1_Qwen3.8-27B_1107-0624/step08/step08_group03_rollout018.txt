import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _convolve_with_weights(w, a, n):
    """Compute h[i] = sum_k w[k] * a[k-i+n-1] for i=0..n-1."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]

                continue
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                t = t * 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-14:
                a = a_try
                accepted = True
                break
            t = t * 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-16:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[50]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.005)

    # Adaptive schedule: start large, reduce based on progress
    n_betas = 30
    betas = base_beta * (0.5 ** np.arange(n_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    stall_count = 0

    for i in range(n_betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        if n <= 4096:
            est_time_per_iter = 0.003
        elif n <= 8192:
            est_time_per_iter = 0.005
        elif n <= 16384:
            est_time_per_iter = 0.012
        elif n <= 20480:
            est_time_per_iter = 0.02
        else:
            est_time_per_iter = 0.04

        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 12))))

        beta = betas[i]
        try:
            t0 = time.monotonic()
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-13,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    break
        except Exception:
            break

    return current_a


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.25) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.005):
    """Subgradient polishing in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use a narrow band of active constraints
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        cnt = int(active.sum())
        if cnt == 0:
            active[int(np.argmax(c))] = True
            cnt = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / cnt
        h = _convolve_with_weights(w, a, n)
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr = lr * 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _kkt_refine(a_init, deadline, max_iter=2000):
    """
    KKT-based refinement: directly minimize the max autocorrelation
    by ensuring the KKT conditions are satisfied.
    Uses a combination of active-set identification and Newton steps.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Adaptive tolerance for active set
    tol = 0.001

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify active set with adaptive tolerance
        active = c >= (c_max - tol * c_max)
        cnt = int(active.sum())
        if cnt == 0:
            active[int(np.argmax(c))] = True
            cnt = 1
        # Cap active set size
        if cnt > 10:
            top_idx = np.argsort(c)[::-1][:10]
            active = np.zeros(len(c), dtype=bool)
            active[top_idx] = True
            cnt = 10

        A_vec = active.astype(np.float64)
        m = cnt

        # Compute gradient
        g = (2.0 / m) * _convolve_with_weights(A_vec, a, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)

        if gnorm < 1e-12 * c_max:
            break

        # Newton step with regularization
        mu = max(1e-10 * c_max / m, 1e-14)

        def hm(v):
            return (2.0 / m) * _convolve_with_weights(A_vec, v, n) + mu * v

        # Solve H x = -g_t using CG
        x = np.zeros(n)
        r = -g_t - hm(x)
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(500):
            Ap = hm(p)
            pAp = np.dot(p, Ap)
            if abs(pAp) < 1e-30:
                break
            alpha = rs / pAp
            x = x + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if rs_new < 1e-28:
                break
            p = r + (rs_new / rs) * p
            rs = rs_new
        u = x

        # Solve H w = 1 for normalization
        w_sol = np.zeros(n)
        r = np.ones(n) - hm(np.zeros(n))
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(500):
            Ap = hm(p)
            pAp = np.dot(p, Ap)
            if abs(pAp) < 1e-30:
                break
            alpha = rs / pAp
            w_sol = w_sol + alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if rs_new < 1e-28:
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        sum_w = float(w_sol.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_sol
            dx = dx - np.mean(dx)

        # Line search with non-negativity
        neg_idx = dx < -1e-14
        if neg_idx.any():
            t_max = float(np.min(a[neg_idx] / (-dx[neg_idx]))) * 0.98
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-14)

        f_old = c_max
        t = min(1.0, t_max)
        accepted = False
        for ls in range(60):
            a_try = a + t * dx
            if np.min(a_try) < -1e-14:
                t = t * 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                t = t * 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-15:
                a = a_try
                accepted = True
                break
            t = t * 0.5

        if not accepted:
            # Try smaller fixed step
            for step in [0.1, 0.05, 0.01, 0.001]:
                a_try = a - step * g_t
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < f_old:
                    a = a_try
                    accepted = True
                    break
            if not accepted:
                # Gradually tighten tolerance
                tol = max(tol * 0.8, 1e-7)
                if tol <= 1e-7 and gnorm < 1e-10 * c_max:
                    break
                continue

        sc = _score(a)
        if sc < best_sc - 1e-16:
            best_sc = sc
            best_a = a.copy()

        # Adapt tolerance
        if it % 50 == 0 and it > 0:
            tol = max(tol * 0.85, 1e-8)

    return best_a, best_sc


def _multi_scale_optimize(a_small, n_target, deadline):
    """
    Multi-scale: upsample from small to large n in stages.
    """
    n_cur = len(a_small)
    current = a_small.copy()
    # Double n each step
    while n_cur < n_target:
        n_next = min(2 * n_cur, n_target)
        current = _interpolate_to_n(current, n_next)
        # Quick L-BFGS refinement at this scale
        if time.monotonic() < deadline - 30:
            current = _run_lbfgs_adaptive(current, deadline, min_time_remaining=30.0)
        n_cur = n_next
    return current


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    - Multi-n exploration
    - Adaptive L-BFGS with log-sum-exp continuation
    - KKT-based active-set Newton refinement
    - Subgradient polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS at parent n ===
    print(f"\n[Phase 1] L-BFGS at n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = min(deadline - 600, t_start + 400)
    if time.monotonic() < phase1_deadline:
        a1 = _run_lbfgs_adaptive(parent, phase1_deadline, min_time_remaining=30.0)
        update_best(a1, _score(a1), "lbfgs_parent_n")

    # === Phase 2: KKT refinement on current best ===
    print(f"\n[Phase 2] KKT refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase2_deadline = min(deadline - 400, time.monotonic() + 200)
        a2, sc2 = _kkt_refine(best_seq, phase2_deadline, max_iter=1500)
        update_best(a2, sc2, "kkt_refine")

    # === Phase 3: Try different n values ===
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = []
    if n_parent <= 8192:
        n_candidates = [8192, 12288, 16384, 20480]
    elif n_parent <= 16384:
        n_candidates = [n_parent, 12288, 20480, 24576]
    else:
        n_candidates = [n_parent, int(n_parent * 0.75), int(n_parent * 1.25)]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue  # Already tried this n

        # Initialize by interpolation from best
        if best_seq is not None:
            a_init_n = _interpolate_to_n(best_seq, n_try)
        else:
            a_init_n = _make_profile(n_try, "uniform")

        sc_init = _score(a_init_n)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # L-BFGS at this n
        n_deadline = min(deadline - 100, time.monotonic() + 120)
        a_n = _run_lbfgs_adaptive(a_init_n, n_deadline, min_time_remaining=20.0)
        update_best(a_n, _score(a_n), f"lbfgs_n{n_try}")

        # Quick KKT refinement
        if time.monotonic() < deadline - 50:
            a_n2, sc_n2 = _kkt_refine(a_n, min(deadline - 30, time.monotonic() + 60), max_iter=500)
            update_best(a_n2, sc_n2, f"kkt_n{n_try}")

    # === Phase 4: Active-set Newton (tight) ===
    print(f"\n[Phase 4] Tight active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase4_deadline = min(deadline - 60, time.monotonic() + 180)
        a4, sc4 = _active_set_newton_tight(best_seq, phase4_deadline, max_iter=2000)
        update_best(a4, sc4, "as_newton_tight")

    # === Phase 5: More KKT refinement ===
    print(f"\n[Phase 5] Second KKT pass (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase5_deadline = min(deadline - 30, time.monotonic() + 120)
        a5, sc5 = _kkt_refine(best_seq, phase5_deadline, max_iter=2000)
        update_best(a5, sc5, "kkt_second")

    # === Phase 6: Subgradient polish ===
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        phase6_deadline = min(deadline - 10, time.monotonic() + 60)
        a6, sc6 = _subgrad_polish(best_seq, phase6_deadline, n_max_steps=3000, lr_init=0.003)
        update_best(a6, sc6, "sg_polish")

    # === Phase 7: Final tight Newton ===
    print(f"\n[Phase 7] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a7, sc7 = _active_set_newton_improved(best_seq, deadline - 2, max_iter=500, tol_active=0.0002)
        update_best(a7, sc7, "final_newton")

    # === Phase 8: Tiny perturbations + quick refine ===
    print(f"\n[Phase 8] Micro-restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for sigma in [0.005, 0.01, 0.02]:
            if time.monotonic() > deadline - 3:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z + sigma * rng.standard_normal(len(best_seq))
            z = z - z.mean()
            a_pert = _softmax(z)
            a_pert_opt = _kkt_refine(a_pert, deadline - 2, max_iter=200)
            update_best(a_pert_opt, _score(a_pert_opt), f"micro_{sigma}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
