```python
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

    n = len(v)

    u = np.sort(v)[::-1]

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Optimized step function search using projected gradient methods with
FFT-based smoothed gradients, adaptive temperature annealing, and
multi-strategy restarts. Focuses on n=8192 with careful time management.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using water-filling."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_weights(c, T):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    shifted = (c - c_max) / T
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    return p


def _smoothed_grad_fft(a, p):
    """Compute g[j] = 2 * sum_k p[k] * a[k-j] using FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(P * np.conj(A), N)
    return 2.0 * cross[:n]


def _adam_projected(a_init, deadline, T_start=None, T_end=None,
                    lr_start=0.05, lr_end=0.001, max_iter=200000,
                    eval_interval=20, patience_limit=50):
    """Adam optimizer with simplex projection and adaptive temperature."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Set temperature based on initial peak spread
    c_init = _fft_convolve(a, a)
    if T_start is None:
        T_start = float(c_init.max()) * 0.01
    if T_end is None:
        T_end = T_start * 0.01
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        
        # Temperature: exponential annealing
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        c = _fft_convolve(a, a)
        p = _softmax_weights(c, T)
        g = _smoothed_grad_fft(a, p)
        
        # Project to tangent space of simplex (remove mean)
        g_tangent = g - g.mean()
        
        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * g_tangent
        v_buf = b2 * v_buf + (1 - b2) * (g_tangent ** 2)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = mh / (np.sqrt(vh) + eps)
        
        # Normalize update to prevent large steps
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update / update_norm
        
        a_new = a - lr * update
        a_new = _project_simplex(a_new)
        a = a_new
        
        # Evaluate
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > patience_limit:
                lr_mult *= 0.5
                patience = 0
                if lr_mult < 0.01:
                    break
    
    # Final check
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=10000, top_k=5, initial_lr=0.002):
    """Subgradient method on the max convolution peak with projection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Get active set (top peaks within tolerance)
        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
            else:
                break
        
        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            # d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n
            offset = k
            # a[k-j] for j=0..n-1 is a[k], a[k-1], ..., a[k-n+1]
            # Valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project to tangent space
        g_tangent = g - g.mean()
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break
        
        # Step
        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = _project_simplex(a_new)
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr *= 0.999
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-10 or no_improve > 200:
                break
    
    return best_a, _score_norm(best_a)


def _active_set_equalization(a, deadline, max_iter=2000, tol=0.002):
    """Equalize top peaks using least-squares direction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 20:
            sorted_a = active[np.argsort(c[active])[::-1][:20]]
            active = sorted_a
        
        # Build direction: want to decrease all active peaks
        # d(conv[k])/d(a[j]) = 2*a[k-j] for valid k-j
        # We want: sum_j M[k,j] * d[j] = -1 for active k
        #          sum_j d[j] = 0
        # Minimum norm: d = M^T (M M^T)^{-1} b
        
        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        M_full = np.vstack([M, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])
        
        G = M_full @ M_full.T
        try:
            G_reg = G + 1e-10 * np.eye(G.shape[0])
            d = M_full.T @ np.linalg.solve(G_reg, b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_cand = a + alpha * d
            a_cand = _project_simplex(a_cand)
            sc = _score_norm(a_cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _coordinate_polish(a, deadline, max_rounds=100):
    """Coordinate descent on elements affecting the peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        # Try modifying elements that contribute to the peak
        candidates = list(range(lo, hi + 1))
        # Sample if too many
        if len(candidates) > 50:
            idx = np.random.choice(len(candidates), 50, replace=False)
            candidates = [candidates[i] for i in idx]
        
        for j in candidates:
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.7, 0.85, 1.15, 1.3]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s_c = a_cand.sum()
                if s_c < 1e-15:
                    continue
                a_cand /= s_c
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-9):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score
```

Wait, I need to double-check my `_project_simplex` function. The standard algorithm for projecting onto the simplex {x >= 0, sum(x) = 1} is:

Given v, find the projection p such that p_i = max(v_i - tau, 0) and sum(p) = 1.

The algorithm:
1. Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find rho = max { j : v_(j) - (sum_{i=1}^{j} v_(i) - 1)/j > 0 }
3. tau = (sum_{i=1}^{rho} v_(i) - 1) / rho
4. p_i = max(v_i - tau, 0)

Let me verify my implementation:
```python
def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u)   # cssv[j] = sum_{i=0}^{j} u[i] = sum of top (j+1) elements
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    # cond[j] = u[j] * (j+1) > cssv[j] - 1
    # = (j+1) * u[j] > sum_{i=0}^{j} u[i] - 1
    # = (j+1) * u[j] + 1 > sum_{i=0}^{j} u[i]
    # = 1 > sum_{i=0}^{j} (u[i] - u[j])
    # This is the condition that u[j] > tau where tau = (cssv[j]-1)/(j+1)
    # i.e., u[j] > (cssv[j]-1)/(j+1)
    # which means (j+1)*u[j] > cssv[j]-1
    # which is our condition
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]  # largest j satisfying the condition
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Optimized step function search using FFT-based smoothed gradients with
simplex projection, adaptive temperature annealing, active-set equalization,
and multi-phase optimization. Targets n=8192 with careful time management.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via water-filling."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_weights(c, T):
    c_max = c.max()
    shifted = (c - c_max) / T
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    return p


def _smoothed_grad_fft(a, p):
    """Compute g[j] = 2 * sum_k p[k] * a[k-j] using FFT cross-correlation."""
    n = len(a)
    N = _next_pow2(3 * n)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(P * np.conj(A), N)
    return 2.0 * cross[:n]


def _subgrad_at_active(a, c, active_k):
    """Compute exact subgradient at the given active lags."""
    n = len(a)
    g = np.zeros(n)
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active_k)
    return g


def _adam_simplex(a_init, deadline, T_start, T_end,
                  lr_start=0.05, lr_end=0.001, max_iter=500000,
                  eval_interval=20, patience_limit=50,
                  use_exact_grad_after=0.6):
    """Adam optimizer with simplex projection and adaptive gradient mode."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    lr_mult = 1.0
    last_improve_t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)

        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        c = _fft_convolve(a, a)

        if frac > use_exact_grad_after:
            # Use exact subgradient at top active lags
            c_max = float(c.max())
            sorted_idx = np.argsort(c)[::-1]
            active = [sorted_idx[0]]
            for idx in sorted_idx[1:6]:
                if c[idx] > c_max * 0.999:
                    active.append(idx)
                else:
                    break
            g = _subgrad_at_active(a, c, active)
        else:
            p = _softmax_weights(c, T)
            g = _smoothed_grad_fft(a, p)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        g_tangent = g - g.mean()

        m_buf = b1 * m_buf + (1 - b1) * g_tangent
        v_buf = b2 * v_buf + (1 - b2) * (g_tangent ** 2)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update / update_norm

        a_new = a - lr * update
        a = _project_simplex(a_new)

        if t % eval_interval == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
                last_improve_t = t
            else:
                patience += 1

            if patience > patience_limit and t - last_improve_t > eval_interval * 5:
                lr_mult *= 0.5
                patience = 0
                if lr_mult < 0.02:
                    break

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=20000, top_k=5, initial_lr=0.002):
    """Subgradient descent with line search on the max peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.9995:
                active.append(idx)
            else:
                break

        g = _subgrad_at_active(a, c, active)
        g_tangent = g - g.mean()
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break

        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = _project_simplex(a_new)

        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr *= 0.9995
        else:
            no_improve += 1
            lr *= 0.97
            if lr < 1e-10 or no_improve > 300:
                break

    return best_a, _score_norm(best_a)


def _active_set_ls(a, deadline, max_iter=3000, tol=0.002):
    """Least-squares direction to equalize active peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 20:
            active = active[np.argsort(c[active])[::-1][:20]]

        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]

        M_full = np.vstack([M, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])

        G = M_full @ M_full.T
        try:
            G_reg = G + 1e-10 * np.eye(G.shape[0])
            d = M_full.T @ np.linalg.solve(G_reg, b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_cand = a + alpha * d
            a_cand = _project_simplex(a_cand)
            sc = _score_norm(a_cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _coordinate_polish(a, deadline, max_rounds=150, rng=None):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)

    if rng is None:
        rng = np.random.default_rng(123)

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)

        candidates = list(range(lo, hi + 1))
        if len(candidates) > 80:
            idx = rng.choice(len(candidates), 80, replace=False)
            candidates = [candidates[i] for i in idx]

        for j in candidates:
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.7, 0.85, 1.15, 1.3, 0.5]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s_c = a_cand.sum()
                if s_c < 1e-15:
                    continue
                a_cand /= s_c
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-9):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score

            break

        # Resample from the best sequence
        x_orig = np.linspace(0, 1, n_orig)
        x_new = np.linspace(0, 1, n_new)
        a_resampled = np.interp(x_new, x_orig, a_best)
        s = a_resampled.sum()
        if s < 1e-15:
            continue
        a_resampled /= s

        c_r = _fft_convolve(a_resampled, a_resampled)
        max_cr = float(c_r.max())

        a_opt, sc = _adam_simplex(
            a_resampled, min(deadline - 30, time.monotonic() + 40),
            T_start=max_cr * 0.002,
            T_end=max_cr * 0.0002,
            lr_start=0.02, lr_end=0.005,
            max_iter=50000, eval_interval=20, patience_limit=40
        )
        full_sc = _score(a_opt)
        print(f"  Try n={n_new}: score={full_sc:.10f}")
        if full_sc < score_best:
            return a_opt, full_sc

    return a_best, score_best


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Adam with simplex projection (smoothed gradient)
    Phase 2: Subgradient polish
    Phase 3: Active-set LS equalization
    Phase 4: Refined Adam (low temperature, exact gradient)
    Phase 5: Coordinate descent
    Phase 6: Random restarts + different n
    Phase 7: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Get initial sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic() - t_start:.1f}s")

    c_init = _fft_convolve(a, a)
    max_c_init = float(c_init.max())

    # Phase 1: Adam with smoothed gradient (0 - 500s)
    p1_end = min(deadline, t_start + 500.0)
    print(f"Phase 1: Adam simplex (smoothed), budget={p1_end - t_start:.0f}s")
    a_opt, sc = _adam_simplex(
        a, p1_end,
        T_start=max_c_init * 0.005,
        T_end=max_c_init * 0.00005,
        lr_start=0.04, lr_end=0.002,
        max_iter=400000, eval_interval=20,
        patience_limit=50, use_exact_grad_after=0.0
    )
    full_sc = _score(a_opt)
    print(f"Phase 1: score={full_sc:.10f}, t={time.monotonic() - t_start:.0f}s")
    if full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 2: Subgradient polish (500 - 700s)
    p2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Subgradient polish")
        a_sg, sc_sg = _subgradient_polish(best_a, p2_end,
                                          max_iter=30000, top_k=5, initial_lr=0.002)
        full_sc_sg = _score(a_sg)
        print(f"Phase 2: score={full_sc_sg:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc_sg < best_score:
            best_score = full_sc_sg
            best_a = a_sg.copy()

    # Phase 3: Active-set LS (700 - 900s)
    p3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Active-set LS equalization")
        a_as, sc_as = _active_set_ls(best_a, p3_end, max_iter=3000, tol=0.002)
        full_sc_as = _score(a_as)
        print(f"Phase 3: score={full_sc_as:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc_as < best_score:
            best_score = full_sc_as
            best_a = a_as.copy()

    # Phase 4: Refined Adam with exact gradient (900 - 1200s)
    p4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Refined Adam (exact grad)")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt2, sc2 = _adam_simplex(
            best_a, p4_end,
            T_start=max_cb * 0.001,
            T_end=max_cb * 0.00001,
            lr_start=0.015, lr_end=0.0005,
            max_iter=400000, eval_interval=15,
            patience_limit=40, use_exact_grad_after=0.3
        )
        full_sc2 = _score(a_opt2)
        print(f"Phase 4: score={full_sc2:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 5: Coordinate descent (1200 - 1400s)
    p5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < p5_end:
        print(f"Phase 5: Coordinate descent")
        a_cd, sc_cd = _coordinate_polish(best_a, p5_end, max_rounds=200, rng=rng)
        full_sc_cd = _score(a_cd)
        print(f"Phase 5: score={full_sc_cd:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 6: Random restarts + different n (1400 - 1650s)
    p6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < p6_end:
        print(f"Phase 6: Random restarts")
        n_restarts = 5
        for r in range(n_restarts):
            if time.monotonic() > p6_end - 40:
                break
            sigma = 0.003 + 0.002 * r
            a_pert = best_a + sigma * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p < 1e-15:
                continue
            a_pert /= s_p

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(p6_end - 20, time.monotonic() + 35)
            a_pert_opt, sc_pert = _adam_simplex(
                a_pert, sub_dl,
                T_start=max_cp * 0.001,
                T_end=max_cp * 0.0001,
                lr_start=0.02, lr_end=0.005,
                max_iter=60000, eval_interval=20,
                patience_limit=30, use_exact_grad_after=0.5
            )
            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{r}] sigma={sigma:.4f}: score={full_sc_pert:.10f} "
                  f"t={time.monotonic() - t_start:.0f}s")
            if full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

        # Try different n
        if time.monotonic() < p6_end - 30:
            print(f"Phase 6b: Try different n")
            a_try, sc_try = _try_different_n(best_a, best_score, p6_end, rng)
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try

    # Phase 7: Final polish (1650 - deadline)
    if time.monotonic() < deadline - 20:
        print(f"Phase 7: Final polish")
        a_final, sc_final = _subgradient_polish(best_a, deadline - 10,
                                                 max_iter=20000, top_k=3,
                                                 initial_lr=0.001)
        full_sc_final = _score(a_final)
        print(f"Phase 7a: score={full_sc_final:.10f}, t={time.monotonic() - t_start:.0f}s")
        if full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()

        if time.monotonic() < deadline - 15:
            a_cd_f, sc_cd_f = _coordinate_polish(best_a, deadline - 5,
                                                  max_rounds=80, rng=rng)
            full_sc_cd_f = _score(a_cd_f)
            if full_sc_cd_f < best_score:
                best_score = full_sc_cd_f
                best_a = a_cd_f.copy()

        # One more subgradient pass
        if time.monotonic() < deadline - 5:
            a_sg2, sc_sg2 = _subgradient_polish(best_a, deadline - 2,
                                                 max_iter=10000, top_k=3,
                                                 initial_lr=0.0005)
            full_sc_sg2 = _score(a_sg2)
            if full_sc_sg2 < best_score:
                best_score = full_sc_sg2
                best_a = a_sg2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, "
          f"time={time.monotonic() - t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
