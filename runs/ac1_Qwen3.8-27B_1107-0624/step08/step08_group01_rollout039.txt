```python
Multi-phase optimizer for step-function sequence design.
Combines L-BFGS with LSE annealing, Nesterov subgradient, 
active-set polishing, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)
    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)

def _lbfgs_phase(a_init, deadline):
    """L-BFGS with aggressive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]

    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    n_betas = 25
    ratios = np.linspace(np.log10(1.0), np.log10(1e-8), n_betas)

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

        if time.monotonic() > deadline - 5.0:
            break

        remaining = deadline - 5.0 - time.monotonic()

            if sc_new < current_sc - 1e-16:

def _subgradient_phase(a_init, deadline, eta0=0.03):
    """Nesterov-accelerated subgradient descent."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = eta0
    momentum = 0.5  # start with moderate momentum
    no_improve = 0
    rng = np.random.default_rng(99)
    
    while time.monotonic() < deadline - 3.0:
        # Nesterov step
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a.copy()
            s_y = 1.0
        else:
            y = y / s_y
        
        # Compute subgradient (use top-20 peaks for stability)

        # Adaptive active set: top-k where k grows as we tighten

        m = np.zeros_like(c)
        m[top_idx] = 1.0 / k
        g = _full_convolve(m, y[::-1])[n-1:2*n-1]
        g = g - g.mean()
        
        # Update
        a_prev = a.copy()
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:

            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive step size and momentum
        eta = max(0.0005, eta * 0.9995)

            a_prev = a.copy()
            momentum = 0.5
```

```python
"""
Multi-phase step-function sequence optimizer combining:
1. Aggressive L-BFGS with log-sum-exp annealing (25 stages down to 1e-8 relative beta)
2. Nesterov-accelerated subgradient descent with adaptive active set
3. Fine active-set polishing with backtracking line search
4. Multi-length exploration (16K, 20K, 24K, 32K)
5. Final combined ultra-fine polishing
Targets score below 1.5027.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _lbfgs_phase(a_init, deadline):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    n_stages = 25
    log_ratios = np.linspace(0.0, -8.0, n_stages)
    betas = base_beta * (10.0 ** log_ratios)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5.0:
            break
        remaining = deadline - 5.0 - time.monotonic()
        scale = max(1.0, n / 4096.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 20))))
        if max_iter < 10:
            break
        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxcor': 100, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgradient_phase(a_init, deadline, eta0=1.0):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta0
    momentum = 0.3
    no_improve = 0
    rng = np.random.default_rng(777)
    while time.monotonic() < deadline - 3.0:
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y
        c = _full_convolve(y, y)
        k = min(30, 2 * n - 1)
        top_idx = np.argpartition(c, -k)[-k:]
        m = np.zeros_like(c)
        m[top_idx] = 1.0 / k
        g_vec = _full_convolve(m, y[::-1])[n - 1:2 * n - 1]
        g_vec = 4.0 * n * g_vec
        g_vec = g_vec - g_vec.mean()
        a_prev = a.copy()
        a_new = y - eta * g_vec
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = a_prev.copy()
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        eta = max(0.01, eta * 0.9998)
        momentum = min(0.95, momentum + 0.0002)
        if no_improve > 2000:
            a = best_a + 0.003 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_prev = a.copy()
            momentum = 0.3
            eta = 0.5
            no_improve = 0
    return best_a, best_sc


def _active_set_polish(a_init, deadline):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.03
    tol_ratio = 0.005
    no_improve = 0
    stall = 0
    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        thresh = c_max * (1.0 - tol_ratio)
        n_active = int((c >= thresh).sum())
        k = min(max(n_active, 3), 40)
        top_idx = np.argpartition(c, -k)[-k:]
        m = np.zeros_like(c)
        m[top_idx] = 1.0 / k
        g_vec = _full_convolve(m, a[::-1])[n - 1:2 * n - 1]
        g_vec = 4.0 * n * g_vec
        g_vec = g_vec - g_vec.mean()
        gnorm = np.linalg.norm(g_vec)
        if gnorm < 1e-18:
            break
        alpha = min(eta, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * g_vec
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_ratio = max(0.00001, tol_ratio * 0.7)
                eta *= 0.7
            if no_improve > 400:
                a = best_a.copy()
                tol_ratio = max(0.00001, tol_ratio * 0.4)
                eta = 0.02
                no_improve = 0
                stall += 1
                if stall > 20:
                    break
        else:
            tol_ratio = max(0.00001, tol_ratio * 0.995)
            eta = min(0.1, eta * 1.003)
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
```
