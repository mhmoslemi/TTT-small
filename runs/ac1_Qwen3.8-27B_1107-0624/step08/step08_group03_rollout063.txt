```python
"""
Search for a non-negative coefficient sequence minimizing
    2*n*max(conv(a,a)) / (sum(a)**2)
using the imported evaluate_sequence() as the final ground-truth scorer.

The optimizer uses:
  * fast FFT convolution,
  * a corrected exact gradient for the log-sum-exp smoothing of the
    maximum convolution coefficient,
  * L-BFGS continuation in log-space with softmax parameterization,
  * active-set Newton polishing of the exact max-coefficient surrogate,
  * optional mirror-descent polishing,
  * multi-scale / multi-profile restarts,
  * a short final validation phase against the actual evaluate_sequence().

All helper functions are top-level, no closures or lambdas are used.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _next_pow2(n):
    n = int(max(2, n))
    k = 1
    while k < n:
        k <<= 1
    return k


def _conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    la = a.size
    lb = b.size
    if la == 0 or lb == 0:
        return np.empty(0, dtype=np.float64)
    out_n = la + lb - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.empty(0, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(z) / float(z.size)
    return e / s


def _normalize_nonneg(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return None
    return a / s


def _score_full_norm(a):
    a = _normalize_nonneg(a)
    if a is None:
        return float("inf")
    n = int(a.size)
    if n <= 0:
        return float("inf")
    c = _conv(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def _grad_from_weights(a, w, n):
    """
    Gradient of Q_w(a) = sum_l w_l (a*a)_l with respect to a.

    For direct self-convolution c = a*a, the exact derivative is:
        dQ/d a_j = 2 * sum_r a_r w_{j+r}, 0 <= r,j < n.
    """
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * float(w[0]) * float(a[0])], dtype=np.float64)

    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    g = _conv(a, w[::-1])
    return 2.0 * g[2 * n - 2:n - 2:-1]


def _base_beta(c):
    m = float(np.max(c))
    if not np.isfinite(m) or m <= 0.0:
        return 1e-12
    cs = np.sort(c)[::-1]
    L = cs.size
    if L > 200:
        gap = float(cs[0] - cs[199])
    elif L > 20:
        gap = float(cs[0] - cs[L // 10])
    else:
        gap = m
    if not np.isfinite(gap) or gap <= 0.0:
        gap = m
    return float(max(gap * 0.3, m * 0.05, 1e-15))


def _lse_obj_full(z, beta, n):
    try:
        a = _softmax(z)
        c = _conv(a, a)
        m = float(np.max(c))
        if not np.isfinite(m) or m <= 0.0:
            return 0.0
        if beta <= 0.0 or beta < 1e-15 * m:
            return m
        shifted = (c - m) / float(beta)
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s = float(e.sum())
        if not np.isfinite(s) or s <= 1e-300:
            return m
        return float(m + float(beta) * float(np.log(s)))
    except Exception:
        return 1e100


def _lse_grad_full(z, beta, n):
    n = int(n)
    zero = np.zeros(n, dtype=np.float64)
    try:
        a = _softmax(z)
        c = _conv(a, a)
        m = float(np.max(c))
        if not np.isfinite(m) or m <= 0.0:
            return zero

        w = np.zeros(c.size, dtype=np.float64)
        if beta <= 0.0 or beta < 1e-15 * m:
            thr = m - max(1e-12 * m, 1e-300)
            active = c >= thr
            if not np.any(active):
                active[int(np.argmax(c))] = True
            cnt = int(np.count_nonzero(active))
            w = np.zeros_like(c)
            w[active] = 1.0 / cnt
        else:
            shifted = (c - m) / float(beta)
            e = np.exp(np.clip(shifted, -700.0, 0.0))
            s = float(e.sum())
            if not np.isfinite(s) or s <= 1e-300:
                active = c >= (m - max(1e-12 * m, 1e-300))
                if not np.any(active):
                    active[int(np.argmax(c))] = True
                cnt = int(np.count_nonzero(active))
                w[active] = 1.0 / cnt
            else:
                w = e / s

        g = _grad_from_weights(a, w, n)
        if not np.all(np.isfinite(g)):
            return zero
        g = np.asarray(g, dtype=np.float64).ravel()
        center = g - float(np.dot(a, g))
        grad = a * center
        if not np.all(np.isfinite(grad)):
            return zero
        return grad
    except Exception:
        return zero


def _estimate_lse_dt(n):
    try:
        z = np.zeros(int(n), dtype=np.float64)
        beta = 1e-6
        t0 = time.monotonic()
        _lse_obj_full(z, beta, n)
        _lse_grad_full(z, beta, n)
        dt = time.monotonic() - t0
        return float(max(dt * 1.7, 1e-4))
    except Exception:
        if n <= 4096:
            return 1e-3
        if n <= 8192:
            return 1e-2
        return 0.04


def _smooth_lbfgs_full(a_init, deadline, min_rem=60.0, time_est=None, iters_phase=None):
    a = _normalize_nonneg(a_init)
    if a is None:
        return np.empty(0), float("inf")
    n = int(a.size)
    if n <= 0:
        return np.empty(0), float("inf")

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.mean(z))

    best_a = a.copy()
    best_sc = _score_full_norm(a)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    c0 = _conv(a, a)
    beta0 = _base_beta(c0)

    if time_est is None:
        time_est = _estimate_lse_dt(n)

    if iters_phase is None:
        if n <= 4096:
            iters_phase = 2500
        elif n <= 8192:
            iters_phase = 1200
        else:
            iters_phase = 700

    factors = [
        4.0, 2.0, 1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.001953125,
        0.0009765625, 0.00048828125
    ]

    no_improve = 0
    for i, f in enumerate(factors):
        now = time.monotonic()
        if now + min_rem >= deadline:
            break
        remaining = max(0.0, deadline - min_rem - now)
        if remaining <= 0.0:
            break

        beta = beta0 * f
        # Conservative cap: L-BFGS may evaluate the objective several times per iteration.
        max_iter = int(min(iters_phase, max(25, remaining / (max(time_est, 1e-4) * 8.0))))
        max_iter = int(max(max_iter, 25))

        options = {
            "maxiter": max_iter,
            "ftol": 1e-16,
            "gtol": 1e-12,
            "maxcor": 40,
            "maxls": 20,
            "iprint": 0,
        }

        try:
            res = minimize(
                _lse_obj_full,
                z,
                jac=_lse_grad_full,
                args=(beta, n),
                method="L-BFGS-B",
                options=options,
            )
        except Exception:
            break

        x = getattr(res, "x", None)
        if x is not None and np.all(np.isfinite(np.asarray(x))):
            z_new = np.asarray(x, dtype=np.float64).ravel()
            z_new = np.clip(z_new, -40.0, 40.0)
            z_new = z_new - float(np.mean(z_new))

            a_new = _softmax(z_new)
            sc_new = _score_full_norm(a_new)
            if np.isfinite(sc_new) and sc_new < best_sc - 1e-14:
                best_a = a_new
                best_sc = sc_new
                no_improve = 0
            else:
                no_improve += 1
            z = z_new
        else:
            no_improve += 1

        if no_improve > 5:
            break

    return best_a, best_sc


def _line_search_full(a, dx, current_max, gnorm, n, max_ls=30):
    a = np.asarray(a, dtype=np.float64).ravel()
    dx = np.asarray(dx, dtype=np.float64).ravel()
    n = int(n)
    if dx.size != n or a.size != n:
        return None, current_max
    if not np.all(np.isfinite(dx)):
        return None, current_max

    neg = dx < -1e-15
    if np.any(neg):
        ratios = a[neg] / (-dx[neg])
        finite = ratios[np.isfinite(ratios) & (ratios > 0.0)]
        if finite.size == 0:
            return None, current_max
        t_max = float(np.min(finite)) * 0.99
    else:
        t_max = 1.0

    t = min(1.0, max(t_max, 1e-12))
    armijo = 1e-14 * max(1e-14, float(gnorm))
    best_t = None
    best_a = None
    best_max = current_max

    for ls in range(max_ls):
        a_try = a + t * dx
        if np.min(a_try) < -1e-10:
            t *= 0.5
            continue
        a_try = np.maximum(a_try, 0.0)
        s = float(a_try.sum())
        if not np.isfinite(s) or s <= 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s
        c = _conv(a_try, a_try)
        m = float(np.max(c))
        if not np.isfinite(m):
            t *= 0.5
            continue

        ok_decrease = (m < best_max - armijo * t)
        ok_last = (ls == max_ls - 1) and (m <= best_max + 1e-12)
        if (ok_decrease or ok_last) and m <= best_max + 1e-11:
            best_t = t
            best_a = a_try
            best_max = m
            break

        t *= 0.5
        if t < 1e-13:
            break

    return best_a, best_max


def _hess_matvec_full(v, fw_rev, N, n, mu):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    fv = np.fft.rfft(v, N)
    g = np.fft.irfft(fv * fw_rev, N)
    if n == 1:
        return 2.0 * g[0:1] + mu * v
    return 2.0 * g[2 * n - 2:n - 2:-1] + float(mu) * v


def _cg_solve_full(rhs, fw_rev, N, n, mu, max_cg=80, tol=1e-10):
    rhs = np.asarray(rhs, dtype=np.float64).ravel()
    n = int(n)
    x = np.zeros(n, dtype=np.float64)
    rhs_norm = float(np.linalg.norm(rhs))
    if not np.isfinite(rhs_norm) or rhs_norm <= 0.0:
        return x

    r = rhs.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    if not np.isfinite(rs) or rs <= 0.0:
        return x

    for _ in range(int(max_cg)):
        Ap = _hess_matvec_full(p, fw_rev, N, n, mu)
        if not np.all(np.isfinite(Ap)):
            break
        pAp = float(np.dot(p, Ap))
        if not np.isfinite(pAp) or pAp <= 1e-300:
            break

        alpha = rs / pAp
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = float(np.dot(r_new, r_new))
        if not np.isfinite(rs_new):
            break

        x = x_new
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        if rs <= 1e-300:
            break

        rho = rs_new / rs
        p = r_new + rho * p
        r = r_new
        rs = rs_new

    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=np.float64)
    return x


def _active_set_newton_full(a_init, deadline, max_iter=80, tol_active=0.002, max_cg=60):
    a = _normalize_nonneg(a_init)
    if a is None:
        return np.empty(0), float("inf")
    n = int(a.size)
    best_a = a.copy()
    best_sc = _score_full_norm(a)
    if n < 2 or not np.isfinite(best_sc):
        return best_a, best_sc

    for _ in range(int(max_iter)):
        now = time.monotonic()
        if now > deadline:
            break

        c = _conv(a, a)
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 1e-300:
            break

        tol = float(tol_active) * cmax + 1e-15
        active = c >= cmax - tol
        if not np.any(active):
            active[int(np.argmax(c))] = True

        cnt = int(np.count_nonzero(active))
        cap = 3 * n
        if cnt > cap:
            active = np.zeros(c.size, dtype=bool)
            idx = np.argsort(c)[::-1][:cap]
            active[idx] = True
            cnt = int(cap)

        w = np.zeros(c.size, dtype=np.float64)
        w[active] = 1.0 / float(cnt)

        g = _grad_from_weights(a, w, n)
        if not np.all(np.isfinite(g)):
            break
        g = np.asarray(g, dtype=np.float64).ravel()
        g_t = g - float(np.mean(g))
        gnorm = float(np.linalg.norm(g_t))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        mu = 1e-8 * max(1.0, cmax) + 1e-12
        w_rev = w[::-1]
        N = _next_pow2(3 * n - 2)
        try:
            fw_rev = np.fft.rfft(w_rev, N)
        except Exception:
            break

        try:
            u = _cg_solve_full(g_t, fw_rev, N, n, mu, max_cg=int(max_cg))
            wv = _cg_solve_full(np.ones(n, dtype=np.float64), fw_rev, N, n, mu,
                                max_cg=int(max_cg))
        except Exception:
            u = np.zeros(n, dtype=np.float64)
            wv = np.zeros(n, dtype=np.float64)

        sw = float(np.sum(wv))
        if np.isfinite(sw) and abs(sw) > 1e-15:
            lam = float(np.sum(u)) / sw
            dx = -u + lam * wv
        else:
            dx = -g_t
        dx = dx - float(np.mean(dx))
        if not np.all(np.isfinite(dx)):
            break

        a_new, cmax_new = _line_search_full(a, dx, cmax, gnorm, n, max_ls=30)
        if a_new is None:
            a_new, cmax_new = _line_search_full(a, -g_t, cmax, gnorm, n, max_ls=20)
        if a_new is None:
            break

        a = a_new
        sc = float(2.0 * n * cmax_new)
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if abs(cmax_new - cmax) < 1e-13 * max(1.0, cmax):
            break

    return best_a, best_sc


def _active_set_polish_full(a_init, deadline):
    a = _normalize_nonneg(a_init)
    if a is None:
        return np.empty(0)
    n = int(a.size)
    best_a = a.copy()
    best_sc = _score_full_norm(a)

    if n <= 4096:
        base = 220
    elif n <= 8192:
        base = 120
    else:
        base = 70

    levels = [
        (0.012, 0.40),
        (0.004, 0.37),
        (0.0012, 0.23),
    ]

    for tol, frac in levels:
        now = time.monotonic()
        if now + 20.0 > deadline:
            break
        remaining = max(0.0, deadline - now - 10.0)
        if remaining <= 0.0:
            break
        allow = remaining * frac
        if allow < 12.0:
            continue
        phase_end = min(deadline - 8.0, now + allow)
        max_iter = int(max(8, base * frac * 2.0))
        a_new, sc_new = _active_set_newton_full(
            a,
            phase_end,
            max_iter=max_iter,
            tol_active=tol,
            max_cg=45,
        )
        if a_new is not None and a_new.size == a.size:
            a = a_new
            if np.isfinite(sc_new) and sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()

    return best_a


def _mirror_full(a_init, deadline, max_steps=700, eta_init=0.1):
    a = _normalize_nonneg(a_init)
    if a is None:
        return np.empty(0), float("inf")
    n = int(a.size)
    if n < 2:
        return a, _score_full_norm(a)

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.mean(z))

    best_a = a.copy()
    best_sc = _score_full_norm(a)
    c0 = _conv(a, a)
    beta = _base_beta(c0)
    step = float(eta_init)
    patience = 0

    for i in range(int(max_steps)):
        now = time.monotonic()
        if now > deadline:
            break

        a = _softmax(z)
        c = _conv(a, a)
        cmax = float(np.max(c))
        sc = float(2.0 * n * cmax)
        if not np.isfinite(sc):
            break

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        beta = max(beta * 0.999, cmax * 1e-4, 1e-300)

        shifted = (c - cmax) / beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        s = float(e.sum())
        if not np.isfinite(s) or s <= 1e-300:
            w = np.zeros(c.size, dtype=np.float64)
            thr = cmax - max(1e-12 * cmax, 1e-300)
            active = c >= thr
            if not np.any(active):
                active[int(np.argmax(c))] = True
            w[active] = 1.0 / float(np.count_nonzero(active))
        else:
            w = e / s

        g = _grad_from_weights(a, w, n)
        if not np.all(np.isfinite(g)):
            break
        g = np.asarray(g, dtype=np.float64).ravel()
        grad_z = a * (g - float(np.dot(a, g)))
        gnorm = float(np.linalg.norm(grad_z))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        accepted = False
        mult = step
        for _ in range(5):
            zn = z - mult * grad_z
            zn = np.clip(zn, -50.0, 50.0)
            zn = zn - float(np.mean(zn))
            an = _softmax(zn)
            cn = _conv(an, an)
            mn = float(np.max(cn))
            sn = float(2.0 * n * mn)
            if np.isfinite(sn) and sn <= max(sc, best_sc) + 1e-12:
                if sn < sc - 1e-15 or sn < best_sc - 1e-12:
                    z = zn
                    a = an
                    sc = sn
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    step = min(float(eta_init), step * 1.02)
                    patience = 0
                    accepted = True
                    break
            mult *= 0.5

        if not accepted:
            patience += 1
            step *= 0.65
            if patience > 18:
                z = np.log(np.maximum(best_a, 1e-300))
                z = z - float(np.mean(z))
                a = best_a.copy()
                step = max(1e-5, float(eta_init) * 0.25)
                patience = 0
            if step < 1e-6:
                break

    a = _softmax(z)
    sc = _score_full_norm(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a
    return best_a, best_sc


def _make_profile(n, kind, rng=None):
    n = int(max(1, n))
    t = np.linspace(0.0, 1.0, n)

    if kind == "uniform" or kind is None:
        a = np.ones(n, dtype=np.float64)
    elif kind == "ramp":
        a = 1.0 + 1.5 * t
    elif kind == "exp_ramp":
        a = 1.0 + np.exp(2.5 * t)
    elif kind == "inv_ramp":
        a = 1.0 + np.exp(2.5 * (1.0 - t))
    elif kind == "cos1":
        a = 1.0 + 0.95 * np.cos(2.0 * np.pi * t)
    elif kind == "cos2":
        a = 1.0 + 0.95 * np.cos(4.0 * np.pi * t)
    elif kind == "sin1":
        a = 1.0 + 0.95 * np.sin(2.0 * np.pi * t + 0.7)
    elif kind == "sin2":
        a = 1.0 + 0.95 * np.sin(4.0 * np.pi * t + 1.1)
    elif kind == "chirp1":
        a = 1.0 + 0.95 * np.cos(2.0 * np.pi * (2.0 * t + t * t))
    elif kind == "chirp2":
        a = 1.0 + 0.95 * np.cos(2.0 * np.pi * (3.0 * t + 0.7 * t * t))
    elif kind == "gauss_center":
        a = np.exp(-8.0 * (t - 0.5) ** 2)
    elif kind == "gauss_edge":
        a = np.exp(-8.0 * t * t) + np.exp(-8.0 * (1.0 - t) ** 2)
    elif kind == "bump_asym":
        a = np.exp(-30.0 * (t - 0.32) ** 2) + 0.35 * np.exp(-30.0 * (t - 0.76) ** 2)
    elif kind == "pow05":
        a = np.power(np.maximum(t, 1e-8), 0.5)
    elif kind == "pow02":
        a = np.power(np.maximum(t, 1e-8), 0.2)
    elif kind == "pow2":
        a = 1.0e-6 + np.power(t, 2.0)
    elif kind == "pow05_rev":
        a = np.power(np.maximum(1.0 - t, 1e-8), 0.5)
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(12345)
        a = rng.standard_normal(n).astype(np.float64)
        k = max(2, int(n // 80))
        kernel = np.ones(k, dtype=np.float64) / float(k)
        a = np.convolve(a, kernel, mode="same")
        a = np.maximum(a, 1e-6)
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.maximum(a, 1e-9)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def _interpolate_to_n(a, n_new):
    a = _normalize_nonneg(a)
    n_new = int(max(1, n_new))
    if a is None:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    n_old = int(a.size)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    return b / s


def _perturb(a, sigma, rng):
    a = _normalize_nonneg(a)
    if a is None or rng is None:
        return np.empty(0)
    n = int(a.size)
    z = np.log(np.maximum(a, 1e-300))
    z = z + float(sigma) * rng.standard_normal(n)
    z = z - float(np.mean(z))
    return _softmax(z)


def _add_candidate(pool, a, label):
    a = _normalize_nonneg(a)
    if a is None:
        return
    sc = _score_full_norm(a)
    if np.isfinite(sc):
        pool.append((sc, a, str(label)))


def _oracle_score_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    try:
        lst = [float(x) for x in a.tolist()]
        return float(evaluate_sequence(lst))
    except Exception:
        return _score_full_norm(a)


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    reserve = 130.0
    deadline = t0 + max(1.0, budget - reserve)

    rng = np.random.default_rng(int(seed) % ((1 << 63) - 1))
    pool = []

    print("=" * 70)
    print(f"Budget={budget:.1f}s, internal deadline={deadline - t0:.1f}s")

    # Universal safe candidates.
    for n in (4, 16, 64, 256, 1024):
        _add_candidate(pool, np.ones(n, dtype=np.float64), f"uniform_{n}")

    # Load parent construction if available.
    parent = None
    try:
        p_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p_raw.size > 0:
            p_raw = np.nan_to_num(p_raw, nan=0.0, posinf=1000.0, neginf=0.0)
            p_raw = np.clip(p_raw, 0.0, 1000.0)
            s = float(p_raw.sum())
            if s > 0.01:
                parent = p_raw / s
                _add_candidate(pool, parent, "parent")
                print(f"Parent loaded: n={parent.size}, internal score={_score_full_norm(parent):.10f}")
            else:
                parent = None
        else:
            parent = None
    except Exception:
        parent = None

    if parent is not None and 512 <= parent.size <= 16384:
        n_main = int(parent.size)
    elif parent is not None:
        n_main = int(min(16384, max(2048, parent.size)))
    else:
        n_main = 16384

    if n_main < 128:
        n_main = 1024
    if n_main > 16384:
        n_main = 16384

    if n_main >= 4096:
        n_small = 4096
    elif n_main > 1024:
        n_small = max(512, n_main // 2)
    else:
        n_small = n_main

    total = max(0.0, deadline - t0)

    # ------------------------------------------------------------------
    # Phase 1: small-scale exploration / basin finding.
    # ------------------------------------------------------------------
    end_small = t0 + 0.15 * total
    best_small_a = None
    best_small_sc = float("inf")
    second_small_a = None
    second_small_sc = float("inf")

    if end_small > t0 + 20.0:
        kinds = [
            "uniform", "ramp", "exp_ramp", "inv_ramp",
            "cos1", "cos2", "sin1", "sin2",
            "chirp1", "chirp2",
            "gauss_center", "gauss_edge", "bump_asym",
            "pow05", "pow02", "pow2", "pow05_rev",
            "random_smooth",
        ]
        inits = []
        for k in kinds:
            inits.append((k, _make_profile(n_small, k, rng)))
        if parent is not None:
            inits.append(("parent_small", _interpolate_to_n(parent, n_small)))

        candidates_small = []
        for lab, aa in inits:
            sc = _score_full_norm(aa)
            if np.isfinite(sc):
                candidates_small.append((sc, aa, lab))

        # Select best and second-best without using lambda.
        for sc, aa, lab in candidates_small:
            if sc < best_small_sc:
                second_small_sc = best_small_sc
                second_small_a = best_small_a
                best_small_sc = sc
                best_small_a = aa
            elif sc < second_small_sc:
                second_small_sc = sc
                second_small_a = aa

        if best_small_a is not None:
            dt_small = _estimate_lse_dt(n_small)
            a1, s1 = _smooth_lbfgs_full(best_small_a, end_small, min_rem=25.0,
                                        time_est=dt_small, iters_phase=2200)
            _add_candidate(pool, a1, "small_opt1")
            print(f"Small opt1: n={n_small}, internal={s1:.10f}, t={time.monotonic()-t0:.1f}s")

            if second_small_a is not None and time.monotonic() < end_small - 25.0:
                a2, s2 = _smooth_lbfgs_full(second_small_a, end_small, min_rem=20.0,
                                            time_est=dt_small, iters_phase=1500)
                _add_candidate(pool, a2, "small_opt2")
                print(f"Small opt2: n={n_small}, internal={s2:.10f}, t={time.monotonic()-t0:.1f}s")

    # ------------------------------------------------------------------
    # Phase 2: main length optimization.
    # ------------------------------------------------------------------
    end_main = t0 + 0.55 * total
    main_inits = []
    if parent is not None:
        if len(parent) == n_main:
            main_inits.append(("parent_main", parent))
        else:
            main_inits.append(("parent_interp", _interpolate_to_n(parent, n_main)))
    if best_small_a is not None:
        main_inits.append(("small1_main", _interpolate_to_n(best_small_a, n_main)))
    if second_small_a is not None:
        main_inits.append(("small2_main", _interpolate_to_n(second_small_a, n_main)))

    best_main_a = None
    best_main_sc = float("inf")
    if main_inits:
        for lab, aa in main_inits:
            sc = _score_full_norm(aa)
            if np.isfinite(sc) and sc < best_main_sc:
                best_main_sc = sc
                best_main_a = aa

    if best_main_a is None:
        best_main_a = _make_profile(n_main, "uniform")
        best_main_sc = _score_full_norm(best_main_a)

    a_main = best_main_a
    sc_main = best_main_sc

    if end_main > time.monotonic() + 40.0:
        dt_main = _estimate_lse_dt(n_main)
        a_main, sc_main = _smooth_lbfgs_full(
            best_main_a,
            end_main,
            min_rem=max(150.0, 0.25 * total),
            time_est=dt_main
        )
        _add_candidate(pool, a_main, "main_lbfgs")
        print(f"Main L-BFGS: n={n_main}, internal={sc_main:.10f}, t={time.monotonic()-t0:.1f}s")

    # ------------------------------------------------------------------
    # Phase 3: active-set exact-max polishing.
    # ------------------------------------------------------------------
    end_active = t0 + 0.85 * total
    if end_active > time.monotonic() + 30.0:
        a_as = _active_set_polish_full(a_main, end_active)
        _add_candidate(pool, a_as, "active_polish")
        sc_as = _score_full_norm(a_as)
        print(f"Active polish: n={a_as.size}, internal={sc_as:.10f}, t={time.monotonic()-t0:.1f}s")
        if a_as is not None and a_as.size > 0:
            a_main = a_as

    # ------------------------------------------------------------------
    # Phase 4: optional mirror-descent polishing.
    # ------------------------------------------------------------------
    end_mirror = deadline - 10.0
    if (
        a_main is not None
        and a_main.size > 0
        and time.monotonic() < end_mirror - 30.0
    ):
        if a_main.size <= 4096:
            mir_steps = 1800
        elif a_main.size <= 8192:
            mir_steps = 900
        else:
            mir_steps = 450

        a_mir, sc_mir = _mirror_full(a_main, end_mirror, max_steps=mir_steps, eta_init=0.08)
        _add_candidate(pool, a_mir, "mirror")
        print(f"Mirror polish: n={a_mir.size if a_mir is not None else 0}, "
              f"internal={sc_mir:.10f}, t={time.monotonic()-t0:.1f}s")

    # ------------------------------------------------------------------
    # Final validation using the actual evaluator.
    # ------------------------------------------------------------------
    selected = []
    work = pool[:]
    for _ in range(12):
        best_i = -1
        best_val = float("inf")
        for i in range(len(work)):
            if work[i] is not None and work[i][0] < best_val:
                best_val = work[i][0]
                best_i = i
        if best_i < 0:
            break
        selected.append(work[best_i])
        work[best_i] = None

    extra_labels = ("uniform_4", "uniform_16", "uniform_64", "parent")
    for item in pool:
        if item is None:
            continue
        lab = item[2]
        already = False
        for sel in selected:
            if sel is not None and sel[2] == lab:
                already = True
                break
        if (not already) and any(lab == x for x in extra_labels):
            selected.append(item)

    best_seq = None
    best_orac = float("inf")
    evaluated = 0
    limit = 16
    for item in selected:
        if item is None:
            continue
        if evaluated >= limit:
            break
        if time.monotonic() > deadline + 20.0:
            break
        sc_int, aa, lab = item
        osc = _oracle_score_array(aa)
        evaluated += 1
        if np.isfinite(osc) and osc < best_orac:
            best_orac = float(osc)
            best_seq = aa
            print(f"Oracle best: {best_orac:.10f}, n={aa.size}, label={lab}, "
                  f"internal={sc_int:.10f}, t={time.monotonic()-t0:.1f}s")

    if best_seq is None:
        if pool:
            best_seq = pool[0][1]
            best_orac = _oracle_score_array(best_seq)
        else:
            best_seq = np.ones(256, dtype=np.float64) / 256.0
            best_orac = _oracle_score_array(best_seq)

    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if not np.isfinite(s) or s < 0.01:
        best_seq = np.ones(256, dtype=np.float64)
        s = float(best_seq.size)

    best_seq = best_seq / s

    print("=" * 70)

    return [float(x) for x in best_seq]
```
