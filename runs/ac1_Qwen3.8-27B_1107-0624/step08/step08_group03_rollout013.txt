"""

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _ps_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    m = min(len(c), 64)
    top = np.sort(c)[::-1][:m]
    val = (np.mean(top ** p)) ** (1.0 / p)
    return 2.0 * n * val


def _ps_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    m = min(len(c), 64)
    top_idx = np.argsort(c)[::-1][:m]
    top = c[top_idx]
    val = (np.mean(top ** p)) ** (1.0 / p)
    if val < 1e-300:
        w = np.zeros_like(c)
        w[top_idx[0]] = 1.0
    else:
        w = np.zeros_like(c)
        for k in top_idx:
            w[k] = (val ** (1 - p)) * (c[k] ** (p - 1)) / m
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=400, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:3 * n]
            active[top_idx] = True
            n_active = 3 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w_vec = cg_solve(np.ones(n))
        sum_w = float(w_vec.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_vec
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(25):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc

def _peak_equalization(a_init, deadline, max_rounds=100):

def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + t
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t)
        return a / a.sum()
    elif kind == "quadratic_ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 3.0 * t ** 2
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.exp(-((t - 0.3) ** 2) / 0.01) + 0.8 * np.exp(-((t - 0.7) ** 2) / 0.01)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(123)
        a = rng.standard_normal(n)
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _try_n_value(a_best, n_target, deadline, min_remaining=200.0, label=""):
    a_interp = _interpolate_to_n(a_best, n_target)
    sc_init = _score(a_interp)
    print(f"  n={n_target} init: score={sc_init:.10f}")

    a1 = _run_lbfgs_ls(a_interp, deadline, min_time_remaining=min_remaining, beta_schedule="fast_then_slow")
    sc1 = _score(a1)
    print(f"  n={n_target} after L-BFGS-LSE: {sc1:.10f}")

    a2, sc2 = _active_set_newton(a1, deadline, max_iter=200, tol_active=0.0005)
    print(f"  n={n_target} after active-set: {sc2:.10f}")

    a3 = _run_lbfgs_ps(a2, deadline, min_time_remaining=min_remaining * 0.5)
    sc3 = _score(a3)
    print(f"  n={n_target} after power-sum: {sc3:.10f}")

    a4, sc4 = _peak_equalization(a3, deadline, max_rounds=50)
    print(f"  n={n_target} after peak-eq: {sc4:.10f}")

    return a4, sc4


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={parent_sc:.10f}")
                update_best(parent, parent_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "ramp")
        print(f"No parent, using ramp n={n_parent}")
        update_best(parent, _score(parent), "init_ramp")

    # Phase 1: L-BFGS with log-sum-exp on parent (fast_then_slow schedule)
    print(f"\n[Phase 1] L-BFGS LSE on parent n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs_ls(parent, deadline, min_time_remaining=350.0, beta_schedule="fast_then_slow")
        update_best(a1, _score(a1), "lbfgs_ls_parent")

    # Phase 2: Active-set Newton (multiple tolerances)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        for tol in [0.002, 0.001, 0.0005, 0.0002]:
            if time.monotonic() > deadline - 100:
                break
            a_ts, sc_ts = _active_set_newton(best_seq, deadline, max_iter=300, tol_active=tol)
            update_best(a_ts, sc_ts, f"as_newton_tol{tol}")

    # Phase 3: Peak equalization
    print(f"\n[Phase 3] Peak equalization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_pe, sc_pe = _peak_equalization(best_seq, deadline, max_rounds=150)
        update_best(a_pe, sc_pe, "peak_eq")

    # Phase 4: Power-sum L-BFGS
    print(f"\n[Phase 4] Power-sum L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_ps = _run_lbfgs_ps(best_seq, deadline, min_time_remaining=150.0)
        update_best(a_ps, _score(a_ps), "power_sum")

    # Phase 5: Try n=32768
    print(f"\n[Phase 5] Try n=32768 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        n_big = 32768
        a_big_init = _interpolate_to_n(best_seq, n_big)
        sc_big_init = _score(a_big_init)
        print(f"  n={n_big} init: {sc_big_init:.10f}")

        a_big1 = _run_lbfgs_ls(a_big_init, deadline, min_time_remaining=200.0, beta_schedule="geometric")
        update_best(a_big1, _score(a_big1), f"lbfgs_n{n_big}")

        a_big2, sc_big2 = _active_set_newton(a_big1, deadline, max_iter=200, tol_active=0.001)
        update_best(a_big2, sc_big2, f"as_n{n_big}")

        a_big3, sc_big3 = _peak_equalization(a_big2, deadline, max_rounds=50)
        update_best(a_big3, sc_big3, f"peak_eq_n{n_big}")

    # Phase 6: Try n=8192
    print(f"\n[Phase 6] Try n=8192 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        n_small = 8192
        a_small_init = _interpolate_to_n(best_seq, n_small)
        sc_small_init = _score(a_small_init)
        print(f"  n={n_small} init: {sc_small_init:.10f}")

        a_small1 = _run_lbfgs_ls(a_small_init, deadline, min_time_remaining=100.0, beta_schedule="fast_then_slow")
        update_best(a_small1, _score(a_small1), f"lbfgs_n{n_small}")

        a_small2, sc_small2 = _active_set_newton(a_small1, deadline, max_iter=300, tol_active=0.0005)
        update_best(a_small2, sc_small2, f"as_n{n_small}")

        a_small3, sc_small3 = _peak_equalization(a_small2, deadline, max_rounds=80)
        update_best(a_small3, sc_small3, f"peak_eq_n{n_small}")

    # Phase 7: Diverse restarts on best
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        restart_sigs = [0.01, 0.03, 0.07, 0.15]
        for sigma in restart_sigs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            a_r = _run_lbfgs_ls(p_start, deadline, min_time_remaining=10.0, beta_schedule="fast_then_slow")
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_s{sigma}")

            if time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, deadline, max_iter=100, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_s{sigma}_as")

    # Phase 8: Momentum subgradient
    print(f"\n[Phase 8] Momentum subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_md, sc_md = _momentum_subgrad(best_seq, deadline, max_steps=4000, lr_init=0.03, momentum=0.9)
        update_best(a_md, sc_md, "momentum_sg")

    # Phase 9: Final intensive active-set (very tight tolerance)
    print(f"\n[Phase 9] Final active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for tol in [0.0003, 0.0001, 0.00005]:
            if time.monotonic() > deadline - 5:
                break
            a_fin, sc_fin = _active_set_newton(best_seq, deadline, max_iter=200, tol_active=tol)
            update_best(a_fin, sc_fin, f"final_as_{tol}")

    # Phase 10: Final peak equalization + active-set cycle
    print(f"\n[Phase 10] Final polish cycle (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(3):
            if time.monotonic() > deadline - 3:
                break
            a_c, sc_c = _peak_equalization(best_seq, deadline, max_rounds=30)
            update_best(a_c, sc_c, f"final_cycle{cycle}_peak")
            if time.monotonic() > deadline - 3:
                break
            a_c2, sc_c2 = _active_set_newton(a_c, deadline, max_iter=100, tol_active=0.0001)
            update_best(a_c2, sc_c2, f"final_cycle{cycle}_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            n_fallback = 16384
            final = _make_profile(n_fallback, "ramp")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback_n = 16384
    fallback = _make_profile(fallback_n, "ramp")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
