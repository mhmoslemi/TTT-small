    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Slightly reduce edges
        edge = n // 10
        a[:edge] = 0.8
        a[-edge:] = 0.8
        a[:edge//2] = 0.6
        a[-edge//2:] = 0.6
        return a / a.sum()
    else:
        return np.ones(n) / n


# ============================================================
# LSE (Log-Sum-Exp) objective and gradient - adaptive version
# ============================================================

def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Power-sum objective and gradient
# ============================================================

def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    val = np.sum(cp) ** (1.0 / p)
    return 2.0 * n * val


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    cp = np.power(c_safe, p)
    sum_cp = np.sum(cp)
    if sum_cp < 1e-300:
        return np.zeros_like(a)
    w = np.power(c_safe, p - 1)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        return np.zeros_like(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0) * w_sum
    grad_a = factor * h
    return a * (grad_a - np.dot(a, grad_a))


# ============================================================
# Soft active-set Newton (weighted version of LSE gradient at low temp)
# ============================================================

def _hess_matvec(v, weights, n):
    """Compute H*v where (Hv)_i = 2 * sum_j v[j] * weights[i+j]."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(weights, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_func, b, tol=1e-10, max_iter=300):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rn = float(np.dot(r_new, r_new))
        if np.sqrt(rn) < tol * b_norm:
            x = x_new
            break
        beta_cg = rn / rs if rs > 1e-30 else 0.0
        x = x_new
        r = r_new
        p = r + beta_cg * p
        rs = rn
    return x


def _newton_soft_active(a_init, deadline, max_iter=300, beta_target=None):
    """
    Newton method on the soft active set problem.
    Uses weights w[k] = softmax((c[k]-c_max)/beta) for the Hessian.
    Adapts beta during iteration.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    if beta_target is None:
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        beta = max(gap * 0.5, c_max * 0.002)
    else:
        beta = beta_target

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Soft weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        w = e / e.sum()

        # Gradient: 2n * (convolve(w, a_rev)[n-1:2n-1] - dot(w,c))
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        # Hessian matvec
        mu = 1e-10 * max(1.0, float(np.dot(np.abs(w), np.abs(w))))

        def _hv(v):
            return _hess_matvec(v, w, n) + mu * v

        # Solve with CG
        dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=300)
        dx = dx - np.mean(dx)

        # Line search
        f_old = float(np.dot(w, c))
        alpha = 1.0
        a_new = a.copy()
        found = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(w, c_try))
            if f_try < f_old - 1e-4 * alpha * gnorm * np.dot(w, c):
                a_new = a_try
                found = True
                break
            alpha *= 0.5
        if not found:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Adapt beta: decrease when active set is concentrated
        eff_count = 1.0 / (w * w).sum()
        if eff_count < 5.0:
            beta *= 0.5
        elif eff_count < 20.0:
            beta *= 0.8
        else:
            beta *= 0.9

    return best_a, best_sc


def _newton_hard_active(a_init, deadline, max_iter=300, tol_frac=0.002):
    """Classic hard active-set Newton."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-15)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        active_f = active.astype(np.float64)

        def _hv(v):
            return _hess_matvec(v, active_f, n) + 1e-10 * v

        grad = _hess_matvec(a, active_f, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        dx = _cg_solve(_hv, -grad_t, tol=1e-12, max_iter=300)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        found = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-4 * alpha * gnorm:
                a_new = a_try
                found = True
                break
            alpha *= 0.5
        if not found:
            if alpha < 1e-14:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ============================================================
# Variance minimization (pre-conditioning)
# ============================================================

def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    return g - np.mean(g)


def _variance_reduce(a_init, deadline, n_steps=300, lr_init=0.002):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        grad = _variance_grad(a, n)
        grad_z = a * (grad - np.dot(a, grad))
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


# ============================================================
# Adaptive LSE optimization with multiple temperature levels
# ============================================================

def _adaptive_lse(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base_beta = max(gap * 0.4, max_c * 0.005)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Adaptive temperature schedule: start warm, cool gradually
    # Use more levels than before for finer control
    n_levels = 18
    betas = base_beta * np.array([
        1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003,
        0.0015, 0.0007, 0.0003
    ])

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(200, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            # Even if not better, update z toward the solution
            # for continuity
            z = 0.5 * z + 0.5 * result.x
            z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _power_sum_schedule(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    p_values = [4, 8, 16, 32, 64, 128, 256, 512]

    for p in p_values:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _power_sum_obj, z, args=(float(p), n),
                jac=_power_sum_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


# ============================================================
# Subgradient with momentum
# ============================================================

def _subgrad_momentum(a_init, deadline, n_steps=8000, lr_init=0.01, tol_frac=0.001, momentum=0.7):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    v = np.zeros(n)  # momentum
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-15)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))

        step = lr / (1.0 + 0.0002 * i)
        v = momentum * v + (1.0 - momentum) * grad_z
        z = z - step * v
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = max(0.001, lr * 0.7)
                else:
                    lr *= 0.97

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier_phase(a, rng, max_amp=0.3):
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if len(A) > 1 and len(a) % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, len(a))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.05):
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, len(a))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_block(a, rng, n_blocks=8):
    """Randomly permute blocks of the sequence."""
    n = len(a)
    block_size = n // n_blocks
    blocks = [a[i*block_size:(i+1)*block_size] for i in range(n_blocks)]
    new_order = rng.permutation(n_blocks)
    a_new = np.concatenate([blocks[i] for i in new_order])
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_mirror_mix(a, rng, alpha=0.3):
    """Mix with reversed version."""
    a_rev = a[::-1].copy()
    a_new = (1.0 - alpha) * a + alpha * a_rev
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    # Small random perturbation
    z = np.log(np.maximum(a_new, 1e-30))
    z = z + 0.01 * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


# ============================================================
# Block coordinate descent
# ============================================================

def _block_cd(a_init, deadline, block_size=128, n_sweeps=3, lr=0.005):
    """
    Block coordinate descent: for each block, do a few gradient steps
    on just those coordinates.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    n_blocks = n // block_size
    block_starts = list(range(0, n - n_blocks * block_size, block_size))

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        # Shuffle block order
        rng_shuffled = np.random.default_rng(sweep * 7 + 42)
        order = rng_shuffled.permutation(len(block_starts))

        for bi in order:
            if time.monotonic() > deadline:
                break
            start = block_starts[bi]
            end = start + block_size

            # Compute gradient for this block
            c = _full_convolve(a, a)
            c_max = float(c.max())
            tol = max(0.002 * c_max, 1e-15)
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad_a = 2.0 * n * (h - W)

            # Update only the block (with non-negativity)
            block = a[start:end].copy()
            grad_block = grad_a[start:end]
            block -= lr * grad_block
            block = np.maximum(block, 0)
            a[start:end] = block

        # Full evaluation
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Renormalize to prevent drift
        a = a / a.sum()

    return best_a, best_sc

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(16384, "uniform")
        n_parent = 16384
        upd(parent, _score(parent), "init_uniform")

    n_main = n_parent  # Primary working size

    # === Phase 1: Variance pre-conditioning (90s) ===
    print(f"\n[Phase 1] Variance pre-conditioning")
    p1_dead = min(deadline, t_start + 90)
    if time.monotonic() < p1_dead:
        a_var, sc_var = _variance_reduce(best_seq.copy(), p1_dead, n_steps=200, lr_init=0.003)
        upd(a_var, sc_var, "variance")

    # === Phase 2: Adaptive LSE - main (400s) ===
    print(f"\n[Phase 2] Adaptive LSE main")
    p2_dead = min(deadline, t_start + 500)
    if time.monotonic() < p2_dead - 300:
        a_lse, sc_lse = _adaptive_lse(best_seq.copy(), p2_dead, min_remaining=280)
        upd(a_lse, sc_lse, "lse_main")

    # === Phase 3: Soft Newton (200s) ===
    print(f"\n[Phase 3] Soft Newton")
    p3_dead = min(deadline, t_start + 700)
    if time.monotonic() < p3_dead - 180:
        a_sn, sc_sn = _newton_soft_active(best_seq.copy(), p3_dead - 180, max_iter=400)
        upd(a_sn, sc_sn, "soft_newton")

    # === Phase 4: Hard Newton (150s) ===
    print(f"\n[Phase 4] Hard Newton")
    p4_dead = min(deadline, t_start + 850)
    if time.monotonic() < p4_dead - 130:
        a_hn, sc_hn = _newton_hard_active(best_seq.copy(), p4_dead - 130, max_iter=300, tol_frac=0.001)
        upd(a_hn, sc_hn, "hard_newton")

    # === Phase 5: Subgradient with momentum (150s) ===
    print(f"\n[Phase 5] Subgradient momentum")
    p5_dead = min(deadline, t_start + 1000)
    if time.monotonic() < p5_dead - 130:
        a_sg, sc_sg = _subgrad_momentum(best_seq.copy(), p5_dead - 130, n_steps=10000, lr_init=0.008, tol_frac=0.001, momentum=0.6)
        upd(a_sg, sc_sg, "subgrad_mom")

    # === Phase 6: Power-sum LSE (150s) ===
    print(f"\n[Phase 6] Power-sum LSE")
    p6_dead = min(deadline, t_start + 1150)
    if time.monotonic() < p6_dead - 130:
        a_ps, sc_ps = _power_sum_schedule(best_seq.copy(), p6_dead - 130, min_remaining=120)
        upd(a_ps, sc_ps, "power_sum")

    # === Phase 7: Block coordinate descent (120s) ===
    print(f"\n[Phase 7] Block CD")
    p7_dead = min(deadline, t_start + 1270)
    if time.monotonic() < p7_dead - 100:
        a_bcd, sc_bcd = _block_cd(best_seq.copy(), p7_dead - 100, block_size=128, n_sweeps=5, lr=0.003)
        upd(a_bcd, sc_bcd, "block_cd")

    # === Phase 8: Alternating Newton + LSE (150s) ===
    print(f"\n[Phase 8] Alternating Newton + LSE")
    p8_dead = min(deadline, t_start + 1420)
    a_cur = best_seq.copy()
    for rnd in range(3):
        if time.monotonic() > p8_dead - 30:
            break
        # Newton step
        nw_dead = min(p8_dead - 30, time.monotonic() + 30)
        a_nw, sc_nw = _newton_soft_active(a_cur, nw_dead, max_iter=100)
        if sc_nw < _score(a_cur):
            a_cur = a_nw
        upd(a_cur, _score(a_cur), f"alt_newton_{rnd}")
        # LSE step (short)
        if time.monotonic() < p8_dead - 30:
            lse_dead = min(p8_dead - 30, time.monotonic() + 25)
            a_lse2, sc_lse2 = _adaptive_lse(a_cur, lse_dead, min_remaining=20)
            if sc_lse2 < _score(a_cur):
                a_cur = a_lse2
            upd(a_cur, _score(a_cur), f"alt_lse_{rnd}")

    # === Phase 9: Diverse restarts (150s) ===
    print(f"\n[Phase 9] Diverse restarts")
    p9_dead = min(deadline, t_start + 1570)
    restarts = [
        ("log_s1", lambda: _perturb_log(best_seq, 0.02, rng)),
        ("log_s2", lambda: _perturb_log(best_seq, 0.04, rng)),
        ("fourier", lambda: _perturb_fourier_phase(best_seq, rng, max_amp=0.2)),
        ("amp", lambda: _perturb_amplitude(best_seq, rng, sigma=0.04)),
        ("block", lambda: _perturb_block(best_seq, rng, n_blocks=4)),
        ("mirror", lambda: _perturb_mirror_mix(best_seq, rng, alpha=0.2)),
    ]
    for name, gen in restarts:
        if time.monotonic() > p9_dead - 15:
            break
        try:
            p_start = gen()
            sc_start = _score(p_start)
            # Quick LSE
            lse_dead = min(p9_dead - 15, time.monotonic() + 15)
            a_r, sc_r = _adaptive_lse(p_start, lse_dead, min_remaining=10)
            upd(a_r, sc_r, f"restart_{name}")
            # Quick Newton
            if sc_r < best_score_val * 1.005 and time.monotonic() < p9_dead - 10:
                nw_dead2 = min(p9_dead - 10, time.monotonic() + 10)
                a_r2, sc_r2 = _newton_hard_active(a_r, nw_dead2, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{name}_nw")
        except Exception:
            pass

    # === Phase 10: Multi-n exploration (100s) ===
    print(f"\n[Phase 10] Multi-n exploration")
    p10_dead = min(deadline, t_start + 1670)
    n_candidates = [8192, 16384, 32768]
    if n_parent not in n_candidates and n_parent >= 4096:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > p10_dead - 10:
            break
        if len(best_seq) == n_try:
            a_init = best_seq.copy()
        else:
            a_init = _resize_sequence(best_seq, n_try)

        # Quick LSE
        lse_dead = min(p10_dead - 10, time.monotonic() + 25)
        a_opt, sc_opt = _adaptive_lse(a_init, lse_dead, min_remaining=8)
        upd(a_opt, sc_opt, f"n_{n_try}")

        # Quick Newton
        if sc_opt < best_score_val * 1.005 and time.monotonic() < p10_dead - 5:
            nw_dead3 = min(p10_dead - 5, time.monotonic() + 15)
            a_opt2, sc_opt2 = _newton_hard_active(a_opt, nw_dead3, max_iter=50, tol_frac=0.002)
            upd(a_opt2, sc_opt2, f"n_{n_try}_nw")

    # === Phase 11: Final ultra-precision (90s) ===
    print(f"\n[Phase 11] Final ultra-precision")
    p11_dead = deadline - 5

    if best_seq is not None and time.monotonic() < p11_dead:
        # Very tight Newton
        nw_dead4 = min(p11_dead - 30, time.monotonic() + 40)
        a_fn, sc_fn = _newton_hard_active(best_seq.copy(), nw_dead4, max_iter=500, tol_frac=0.0005)
        upd(a_fn, sc_fn, "final_newton_tight")

        # Ultra-low beta LSE
        if time.monotonic() < p11_dead - 20:
            n_fin = len(best_seq)
            c_fin = _full_convolve(best_seq, best_seq)
            mb = float(c_fin.max())
            z_fin = np.log(np.maximum(best_seq, 1e-30))
            z_fin = z_fin - z_fin.mean()
            remaining = p11_dead - 20 - time.monotonic()
            if remaining > 3:
                beta_ultra = mb * 0.00003
                max_iter_fin = min(3000, max(100, int(remaining / 0.004)))
                try:
                    result = minimize(
                        _lse_objective, z_fin, args=(beta_ultra, n_fin),
                        jac=_lse_gradient, method='L-BFGS-B',
                        options={'maxiter': max_iter_fin, 'ftol': 1e-20,
                                 'gtol': 1e-15, 'maxcor': 70, 'maxls': 100}
                    )
                    a_fin = _softmax(result.x)
                    sc_fin = _score(a_fin)
                    upd(a_fin, sc_fin, "final_ultra_lse")
                except Exception:
                    pass

        # Final subgradient with very fine tol
        if time.monotonic() < p11_dead - 5:
            a_fs, sc_fs = _subgrad_momentum(best_seq.copy(), p11_dead - 3, n_steps=5000,
                                            lr_init=0.003, tol_frac=0.0003, momentum=0.5)
            upd(a_fs, sc_fs, "final_subgrad")

        # One more Newton on the absolute best
        if time.monotonic() < p11_dead - 2:
            a_final_nw, sc_final_nw = _newton_soft_active(best_seq.copy(), p11_dead - 1,
                                                           max_iter=200, beta_target=best_score_val * 1e-5)
            upd(a_final_nw, sc_final_nw, "final_soft_nw")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            print("FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
