            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _run_lbfgs_unconstrained(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS without non-negativity constraint (works in original space).
    Allows negative intermediate values, clips at the end."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    current_sc = 2.0 * n * max_c

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.008, 0.004, 0.002,
                                  0.001, 0.0005])

    def obj(x):
        x = x - x.mean()  # Remove mean to fix sum approximately
        c = _full_convolve(x, x)
        c_max = float(c.max())
        shifted = (c - c_max) / self_beta_holder[0]
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return c_max
        return c_max + self_beta_holder[0] * np.log(s)

    def grad(x):
        x = x - x.mean()
        c = _full_convolve(x, x)
        c_max = float(c.max())
        shifted = (c - c_max) / self_beta_holder[0]
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = x[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        # Project out the mean (constraint sum = 0 after centering)
        g = g - g.mean()
        return g

    global self_beta_holder
    self_beta_holder = [base_beta]

    current_x = (a - a.mean()).copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        self_beta_holder[0] = beta
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 15))))
        try:
            result = minimize(obj, current_x, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-14, 'maxcor': 80})
            x_new = result.x - result.x.mean()
            a_new = x_new + a.mean()  # Restore mean
            # Clip to non-negative and normalize
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_sc = sc_new
                current_x = (a_new - a_new.mean()).copy()
        except Exception:
            pass

    # Return the best non-negative version
    x_best = current_x - current_x.mean()
    a_best = np.maximum(x_best + a_init.mean(), 0.0)
    s_best = a_best.sum()
    if s_best > 1e-10:
        a_best = a_best / s_best
    else:
        a_best = a_init.copy()
    return a_best


# Global for the unconstrained gradient
self_beta_holder = [1.0]


def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    """Iterative peak equalization via mass transport.
    Directly redistributes mass from elements contributing to high peaks
    to elements contributing to low peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 0.02
    stall_count = 0
    max_stall = 100

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall_count = 0
            lr = min(0.05, lr * 1.01)
        else:
            stall_count += 1
            if stall_count % 20 == 0:
                lr *= 0.8
            if stall_count >= max_stall:
                a = best_a.copy()
                lr = 0.01
                stall_count = 0
                if lr < 0.001:
                    break

        # Compute weights for peaks (soft-max over top peaks)
        c_sorted = np.sort(c)[::-1]
        n_top = max(10, int(n * 0.05))
        threshold = c_sorted[min(n_top, len(c_sorted)-1)]
        w = (c - threshold) / (c_max - threshold + 1e-30)
        w = np.maximum(w, 0)
        w = w / (w.sum() + 1e-30)

        # Compute gradient of weighted convolution w.r.t. a
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)
        # Remove mean to preserve sum
        grad = grad - grad.mean()

        # Gradient step with non-negativity projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new < current_sc - 1e-14:
            a = a_new
        else:
            # Try smaller step
            a_try = a - lr * 0.1 * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    lr *= 0.5
                else:
                    lr *= 0.7
            else:
                lr *= 0.7

    return best_a, _score(best_a)


def _fourier_perturb(a, n_modes, amplitude, rng):
    """Perturb in Fourier domain to explore different structures."""
    n = len(a)
    N = _next_pow2(2 * n)
    F = np.fft.rfft(a, N)
    # Perturb some non-DC modes
    modes = rng.choice(range(1, len(F)), size=min(n_modes, len(F)-1), replace=False)
    for m in modes:
        phase = rng.uniform(0, 2 * np.pi)
        F[m] += amplitude * np.exp(1j * phase)
    a_new = np.fft.irfft(F, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    """Active-set gradient method focusing on top peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    stall_count = 0

    while time.monotonic() + 5 < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            lr = min(lr_init, lr * 1.01)

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= 300:
                a = best_a.copy()
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.95)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        # U-shaped: large at edges, small in middle
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2 * t - 1) ** 2
        return a / a.sum()
    elif kind == "u_shape_strong":
        t = np.linspace(0, 1, n)
        a = 1.0 + 5.0 * (2 * t - 1) ** 4
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2) + 0.02
        return a / a.sum()
    elif kind == "asymmetric_u":
        t = np.linspace(0, 1, n)
        a = 1.0 + 3.0 * (2 * t - 1) ** 2 + 0.5 * np.cos(2 * np.pi * 3 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spiky":
        a = np.ones(n)
        a[::7] = 3.0
        a[::13] = 5.0
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective support."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _restart_search(a_best, deadline, min_time_remaining=30.0):
    """Try perturbations of the best solution to escape local minima."""
    n = len(a_best)
    rng = np.random.default_rng(12345)
    best_sc = _score(a_best)
    best_a = a_best.copy()

    amplitudes = [0.01, 0.005, 0.02, 0.002, 0.03]
    n_modes_list = [10, 50, 20, 100, 5]

    for amp, n_modes in zip(amplitudes, n_modes_list):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_pert = _fourier_perturb(a_best, n_modes, amp, rng)
        sc_pert = _score(a_pert)
        # Quick L-BFGS from perturbed point
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        a_opt = _run_lbfgs_constrained(a_pert, deadline, min_time_remaining=min_time_remaining)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc - 1e-10:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Restart improved: {best_sc:.10f} (amp={amp}, modes={n_modes})")

    return best_a, best_sc
