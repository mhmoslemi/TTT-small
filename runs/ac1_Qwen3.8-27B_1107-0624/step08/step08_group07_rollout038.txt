"""
Step function optimization using L-BFGS-B with adaptive smoothing,
subgradient polish with active-set equalization, and multi-scale search.
Targets score < 1.5027 by improving upon the parent sequence (1.5057).
"""
import numpy as np
import time
from scipy.optimize import minimize, line_search
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n + 1)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _score(a):
    """Compute the score for sequence a (assumes a >= 0, sum(a) > 0)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_max(a):
    """Return max of linear autocorrelation, assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(c.max()), c


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _subgradient_active(a, c, top_k=5, tol=0.001):
    """Compute subgradient using active set of top peaks."""
    n = len(a)
    c_max = c.max()
    tol_val = c_max * (1 - tol)
    active = np.where(c >= tol_val)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    elif len(active) > top_k:
        # Keep only the top_k
        top_idx = np.argsort(c[active])[::-1][:top_k]
        active = active[top_idx]
    
    # Subgradient: average over active peaks
    # g[j] = (1/|A|) * sum_{k in A} 2*a[k-j] (for valid k-j)
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    
    # Scale by 2*n (the factor in the score)
    g *= 2.0 * n
    return g


def _subgradient_step(a, c, lr, top_k=5, tol=0.001):
    """One subgradient step with projection onto simplex."""
    g = _subgradient_active(a, c, top_k, tol)
    a_new = _project_simplex(a - lr * g)
    return a_new


def _smoothed_obj_and_grad(a, T):
    """Compute smoothed objective and gradient.
    a is on the simplex (sum=1, a>=0).
    Returns (value, gradient) of 2*n*T*log(sum exp(c/T)) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = c.max()
    logits = (c - c_max) / T
    p = np.exp(logits)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = p / p_sum
    
    val = 2.0 * n * T * (c_max / T + np.log(p_sum))
    # = 2.0 * n * (c_max + T * log(p_sum))
    
    # Gradient: df/da[j] = 2*n * sum_k p[k] * 2*a[k-j] = 4*n * (p ⊛ a)[j]
    # where (p ⊛ a)[j] = sum_k p[k] * a[k-j] (cross-correlation)
    N = _next_pow2(3 * n)
    P = rfft(p, N)
    A = rfft(a, N)
    cross = irfft(np.conj(P) * A, N)
    grad = 4.0 * n * cross[:n]
    
    return val, grad


def _lbfgs_smooth(a_init, T, max_iter=200, deadline=None):
    """Run L-BFGS-B on the smoothed objective.
    Parameters are in log-space: z = log(a + eps), centered.
    """
    n = len(a_init)
    eps = 1e-15
    
    def obj_and_grad(z):
        a = np.exp(z)
        a = a / a.sum()
        val, grad_a = _smoothed_obj_and_grad(a, T)
        # Convert gradient from a-space to z-space
        # da[i]/dz[j] = a[i] * (delta_{ij} - a[j])
        # dz[j] = sum_i grad_a[i] * da[i]/dz[j]
        # = sum_i grad_a[i] * a[j] * (delta_{ij} - a[i])
        # = a[j] * (grad_a[j] - sum_i grad_a[i] * a[i])
        W = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - W)
        return val, grad_z
    
    z0 = np.log(a_init + eps)
    z0 = z0 - z0.mean()
    
    result = minimize(
        obj_and_grad, z0,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'gtol': 1e-6, 'ftol': 1e-10}
    )
    
    z_opt = result.x
    a_opt = np.exp(z_opt)
    a_opt = a_opt / a_opt.sum()
    return a_opt


def _subgradient_optimize(a_init, deadline, max_iter=10000, top_k=5, 
                           lr_init=0.01, lr_decay=0.999):
    """Subgradient optimization with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = float(_convolve_max(a)[0]) * 2.0 * n
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            a_cand = _subgradient_step(a, c, alpha, top_k, 0.001)
            c_cand = _fft_convolve(a_cand, a_cand)
            score_cand = 2.0 * n * float(c_cand.max())
            if score_cand < current_score:
                a = a_cand
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            lr = min(lr * 1.01, lr_init)  # slightly increase
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    
    return best_a, best_score


def _active_equalize(a, deadline, max_iter=1000, top_k=5):
    """Equalize the top peaks using a least-squares approach."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(a, a).max())
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        
        if len(active) < 2:
            break
        
        # Check if peaks are already equalized
        if c[active].max() - c[active].min() < c_max * 1e-4:
            # Just try to reduce the max
            pass
        
        # Build the Jacobian: for each active peak k, the gradient of c[k] w.r.t. a
        # J[i, j] = dc[active[i]]/da[j] = 2*a[active[i]-j] (if valid)
        J = np.zeros((len(active), n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                J[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        
        # We want to find d such that:
        # c[k] + J[k,:] @ d ≈ c_max for all active k (equalize)
        # and we want to reduce c_max: c[k] + J[k,:] @ d <= c_max - delta
        # Subject to: sum(d) = 0 (preserve sum)
        # And: a + d >= 0
        
        # Simple approach: minimize ||J @ d - (-delta * ones)||^2 + λ * ||d||^2
        # subject to sum(d) = 0
        # Use Tikhonov regularization for stability
        
        delta = c_max * 0.01
        b = -delta * np.ones(len(active))
        lam = 0.1
        
        # Add sum constraint as a soft penalty
        J_aug = np.vstack([J, np.ones(n) * np.sqrt(lam)])
        b_aug = np.concatenate([b, [0.0]])
        
        # Solve (J_aug^T J_aug + λI) d = J_aug^T b_aug
        ATA = J_aug.T @ J_aug + lam * np.eye(n)
        ATb = J_aug.T @ b_aug
        
        try:
            d = np.linalg.solve(ATA, ATb)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(ATA, ATb, rcond=None)[0]
        
        # Ensure a + alpha*d >= 0 for some alpha
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        
        d = d / d_norm  # normalize to control step size
        
        # Find max step alpha such that a + alpha*d >= 0
        neg_mask = d < 0
        if neg_mask.any():
            max_alpha = np.min(a[neg_mask] / (-d[neg_mask]))
        else:
            max_alpha = 1.0
        max_alpha = min(max_alpha * 0.9, 0.1)  # be conservative
        
        # Try several step sizes
        improved = False
        for alpha in [max_alpha, max_alpha * 0.5, max_alpha * 0.1, 0.01, 0.001]:
            a_cand = a + alpha * d
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            c_cand = _fft_convolve(a_cand, a_cand)
            score_cand = 2.0 * n * float(c_cand.max())
            if score_cand < best_score * (1 - 1e-10):
                best_score = score_cand
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score
