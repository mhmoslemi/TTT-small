"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) FFT-based Newton active set with progressive tightening,
(2) adaptive log-sum-exp smoothing via L-BFGS-B,
(3) KKT-residual-guided strategy switching,
(4) multi-start diversification, (5) peak-shaving local search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hankel_diag(W, n):
    # Diagonal of the Hankel matrix H where H[i,j] = 2*W[i+j] (for the active set)
    # H[i,i] = 2*W[2*i] ... but actually H[i,j] = sum_k w_k * (delta_{i+j+k, ...})
    # From the matvec: H*v where H is defined by the Hankel structure
    # H[i,j] = 2 * sum_k W[k] * delta(i+j+k == n-1+something)
    # Actually from _hankel_matvec: out[i] = 2 * sum_j v[j] * W[i+j] (approximately)
    # Let's just compute diag by matvec with basis vectors... too slow.
    # Instead: the diagonal element H[i,i] = 2*W[2*i] is wrong.
    # Let me just return ones as a simple preconditioner.
    return np.ones(n)


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-40:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-40 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.01:
            # Try half step
            x_new = x + 0.5 * alpha * p
            r_new = r - 0.5 * alpha * Hp
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new) or rs_new > rs:
                break
        x = x_new
        r = r_new
        # CG restart every 50 iterations
        if i > 0 and i % 50 == 0 and rs < 0.1 * rs0:
            p = r.copy()
        else:
            p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _newton_polish(a_init, deadline, max_rounds=200, verbose=False):
    """Newton active set with progressive tightening and KKT monitoring."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    kkt_history = []

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Progressive active set: start loose, tighten over iterations
        if rnd < 5:
            tol_frac = 0.005
        elif rnd < 15:
            tol_frac = 0.001
        elif rnd < 30:
            tol_frac = 0.0001
        elif rnd < 60:
            tol_frac = 1e-5
        else:
            tol_frac = 1e-7

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size to avoid overly ill-conditioned systems
        if n_active > 2 * n:
            # Keep only the top 2n active constraints
            c_sorted_idx = np.argsort(c)[::-1][:2 * n]
            active = np.zeros_like(c, dtype=bool)
            active[c_sorted_idx] = True
            n_active = 2 * n

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute gradient of the active objective
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))

        # KKT residual
        kkt_res = gn / max(c_max * n, 1e-30)
        kkt_history.append(kkt_res)

        if gn < 1e-14 * n:
            if verbose:
                print(f"    Newton: converged at round {rnd}, kkt_res={kkt_res:.2e}")
            break

        # Solve the Newton system
        Hg = _cg_solve(W, n, g, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-25 or not np.isfinite(H1_norm):
            # Fallback to steepest descent direction
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Line search with backtracking (Armijo + curvature)
        alpha = 1.0
        step_done = False
        c_max_old = c_max
        for ls in range(100):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())

            if c_max_try < c_max_old - 1e-16:
                a = a_try
                step_done = True
                break
            alpha *= 0.7
            if alpha < 1e-18:
                break

        if not step_done:
            # Fallback: try multiple step sizes in the gradient direction
            best_alpha = 0
            best_cmdx = c_max_old
            for trial_alpha in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial_alpha * g_t / max(gn, 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                cmdx_try = float(c_try.max())
                if cmdx_try < best_cmdx:
                    best_cmdx = cmdx_try
                    best_alpha = trial_alpha
                    a = a_try
                    step_done = True

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        if verbose and rnd % 10 == 0:
            print(f"    Newton round {rnd}: score={sc:.12f}, kkt={kkt_res:.2e}, "
                  f"active={n_active}, c_max={c_max:.8f}")

        # Early stopping if KKT is very small and no improvement
        if rnd > 20 and kkt_res < 1e-10:
            if verbose:
                print(f"    Newton: KKT satisfied at round {rnd}")
            break

    # Final check
    sc_final = _score(best_a)
    if sc_final < best_sc:
        best_sc = sc_final
    return best_a, best_sc


def _adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """Adaptive L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 1e-10
    prev_sc = current_a.sum()  # placeholder
    prev_sc = current_sc

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, n / 8192.0)
        est_iter_time = 0.004 * scale
        max_iter = min(10000, max(40, int(remaining / (est_iter_time * 20))))

        try:
            obj_fn = lambda z_: _smoobj(z_, beta, n)
            grad_fn = lambda z_: _smograd(z_, beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 100, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS: beta={beta:.2e}, score={sc_new:.12f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")

        # Adaptive beta reduction
        c_new = _full_convolve(current_a, current_a)
        c_max_new = float(c_new.max())
        c_sorted_new = np.sort(c_new)[::-1]
        new_gap = c_sorted_new[0] - c_sorted_new[min(20, len(c_sorted_new) - 1)]

        sc_improved = prev_sc - current_sc
        prev_sc = current_sc

        if new_gap < 1e-9 * c_max_new and sc_improved > 1e-13:
            beta *= 0.1
        elif new_gap < 1e-6 * c_max_new:
            beta *= 0.2
        elif new_gap < 1e-4 * c_max_new:
            beta *= 0.4
        else:
            beta *= 0.5

        beta = min(beta, c_max_new * 0.0003)

    return current_a


def _peak_shaving(a_init, deadline, max_iters=2000):
    """Subgradient-based peak shaving with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.02

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance for active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.06)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t ** 0.6) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "left_heavy":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "right_heavy":
        t = np.linspace(0, 1, n)
        a = t ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        fade = int(n * 0.1)
        for i in range(fade):
            a[i] = i / fade
            a[n - 1 - i] = i / fade
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        sc_p = _score(a_p)
        if sc_p < best_p1_sc:
            best_p1_sc = sc_p
            best_p1_a = a_p

    # Try perturbed parent
    for sigma in [0.03, 0.1, 0.3]:
        if time.monotonic() > phase1_deadline - 5:
            break
        a_pert = _perturb(parent, sigma, rng)
        sc_pert = _score(a_pert)
        if sc_pert < best_p1_sc:
            best_p1_sc = sc_pert
            best_p1_a = a_pert

    # Quick L-BFGS on best profile
    if time.monotonic() < phase1_deadline - 5:
        a_quick = _adaptive_lbfgs(best_p1_a, phase1_deadline, min_time_remaining=10)
        sc_quick = _score(a_quick)
        update_best(a_quick, sc_quick, "p1_quick")
        if sc_quick < best_p1_sc:
            best_p1_a = a_quick
            best_p1_sc = sc_quick

    # Quick Newton on best
    start_a = best_seq if best_seq is not None else best_p1_a
    if time.monotonic() < phase1_deadline - 3:
        a_np, sc_np = _newton_polish(start_a, phase1_deadline, max_rounds=50)
        update_best(a_np, sc_np, "p1_newton")

    # === Phase 2: Main adaptive L-BFGS (~500s) ===
    print(f"\n[Phase 2] Main adaptive L-BFGS", flush=True)
    phase2_deadline = min(deadline - 600, t_start + 700)
    start_a = best_seq if best_seq is not None else best_p1_a
    a_main = _adaptive_lbfgs(start_a, phase2_deadline, min_time_remaining=200)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "p2_lbfgs")

    # Newton after L-BFGS
    if time.monotonic() < phase2_deadline - 10:
        a_np2, sc_np2 = _newton_polish(a_main, phase2_deadline, max_rounds=80)
        update_best(a_np2, sc_np2, "p2_newton")

    # === Phase 3: Intensive Newton polish (~400s) ===
    print(f"\n[Phase 3] Intensive Newton polish", flush=True)
    phase3_deadline = min(deadline - 250, t_start + 1100)
    if best_seq is not None:
        a_int, sc_int = _newton_polish(best_seq, phase3_deadline, max_rounds=300, verbose=True)
        update_best(a_int, sc_int, "p3_newton")

    # Peak shaving
    if best_seq is not None and time.monotonic() < phase3_deadline - 20:
        a_ps, sc_ps = _peak_shaving(best_seq, min(phase3_deadline, time.monotonic() + 100))
        update_best(a_ps, sc_ps, "p3_peakshave")
        if sc_ps < best_score + 0.001:
            a_np3, sc_np3 = _newton_polish(a_ps, phase3_deadline, max_rounds=100)
            update_best(a_np3, sc_np3, "p3_ps_newton")

    # === Phase 4: Diversification and restarts (~200s) ===
    print(f"\n[Phase 4] Diversification", flush=True)
    phase4_deadline = min(deadline - 100, t_start + 1500)
    if best_seq is not None:
        restarts = [
            (0.02, "perturb_small"),
            (0.05, "perturb_med"),
            (0.15, "perturb_large"),
        ]
        for sigma, label in restarts:
            if time.monotonic() > phase4_deadline - 30:
                break
            a_r = _perturb(best_seq, sigma, rng)
            sc_r = _score(a_r)
            # Quick optimize
            if time.monotonic() < phase4_deadline - 20:
                a_ro = _adaptive_lbfgs(a_r, min(phase4_deadline - 20, time.monotonic() + 80),
                                       min_time_remaining=30)
                sc_ro = _score(a_ro)
                update_best(a_ro, sc_ro, f"p4_{label}")
                if sc_ro < best_score + 0.002:
                    a_rn, sc_rn = _newton_polish(a_ro, phase4_deadline - 10, max_rounds=80)
                    update_best(a_rn, sc_rn, f"p4_{label}_np")

    # === Phase 5: Multi-n exploration (~100s) ===
    print(f"\n[Phase 5] Multi-n", flush=True)
    phase5_deadline = min(deadline - 40, t_start + 1650)
    n_base = len(best_seq) if best_seq is not None else n_parent
    n_candidates = []
    for factor in [0.75, 1.5, 2.0]:
        n_try = int(n_base * factor)
        n_try = max(1024, min(65536, n_try))
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > phase5_deadline - 20:
            break
        a_init = _resample(best_seq, n_try)
        a_opt = _adaptive_lbfgs(a_init, min(phase5_deadline - 20, time.monotonic() + 60),
                                min_time_remaining=30)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"p5_n{n_try}")
        if sc_opt < best_score + 0.002 and time.monotonic() < phase5_deadline - 15:
            a_np5, sc_np5 = _newton_polish(a_opt, phase5_deadline - 5, max_rounds=60)
            update_best(a_np5, sc_np5, f"p5_n{n_try}_np")

    # === Phase 6: Final ultra-fine polish (~50s) ===
    print(f"\n[Phase 6] Final polish", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Ultra-tight Newton
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 2, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

        # One more L-BFGS at tiny beta
        if best_seq is not None and time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-11
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(30, int(remaining / 0.005)))
                try:
                    obj_fn = lambda z_: _smoobj(z_, beta_fin, n)
                    grad_fn = lambda z_: _smograd(z_, beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                                 'maxcor': 100, 'maxls': 500}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # Final Newton after last L-BFGS
        if best_seq is not None and time.monotonic() < deadline - 1:
            a_vf, sc_vf = _newton_polish(best_seq, deadline - 0.5, max_rounds=50)
            update_best(a_vf, sc_vf, "very_final")

    # Final validation
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(16384, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
