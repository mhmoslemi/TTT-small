import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (all top-level, no closures)

def _next_pow2(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute the objective score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _corr_a_A(a, A_vec, n):
    """Correlate a with A_vec, returning the n-length 'diagonal' part."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1: 2 * n - 1]

def _cg_solve(matvec_fn, rhs, max_cg, tol, preconditioner=None):
    """Conjugate gradient solver with optional preconditioner."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    if preconditioner is not None:
        z = preconditioner(r)
    else:

    p = z.copy()
    rz = np.dot(r, z)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)

Minimizes 2*n*max(conv(a,a))/sum(a)^2 via:
1. Multi-resolution warm-start (small n -> large n)
2. Adaptive LSE-annealed L-BFGS with gap-based beta scheduling

5. Structured frequency-domain perturbations for diverse restarts

_LBFGS_STATE = {'beta': 1.0, 'n': 0}

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1: 2 * n - 1]

    z = z.copy()
    z -= np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()

        return np.ones_like(z) / len(z)
    return e / s

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

    return 2.0 * n * (c_max + beta * np.log(s))

def _ls_grad(z):
    n = _LBFGS_STATE['n']
    beta = _LBFGS_STATE['beta']
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0

        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs_iter(a_init, beta, deadline, max_iter):
    """Run L-BFGS for the LSE-smoothed problem at given beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    _LBFGS_STATE['beta'] = beta
    _LBFGS_STATE['n'] = n

    result = minimize(
        _ls_obj,
        z,
        jac=_ls_grad,
        method='L-BFGS-B',

            'maxiter': max_iter,

def _adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """Adaptive LSE annealing with gap-based beta scheduling."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    
    # Initial beta based on the gap
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    beta = max(gap * 0.5, c_max * 0.05)
    
    min_beta = c_max * 0.00001
    beta_ratio = 0.4  # More aggressive reduction
    
    while time.monotonic() < deadline - min_time_remaining:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Estimate iterations possible
        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        a_new = _run_lbfgs_iter(a, beta, deadline - min_time_remaining, max_iter)
        sc_new = _score(a_new)
        
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new
        
        # Update beta based on current gap
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        new_beta = max(gap * 0.3, c_max * 0.001)
        
        if new_beta < beta * beta_ratio:
            beta = max(beta * beta_ratio, min_beta)
        else:
            beta = max(new_beta, min_beta)
        
        if beta <= min_beta * 1.1:
            # At minimum beta, do one more round
            beta = min_beta
            a_new = _run_lbfgs_iter(a, beta, deadline - min_time_remaining, max_iter)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
            break

    return best_a, best_sc

def _peak_equalization(a_init, deadline, n_rounds=50):
    """Directly target and reduce the top convolution peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks
        n_peaks = min(5, len(c))
        top_positions = np.argsort(c)[::-1][:n_peaks]

        a_new = a.copy()
        improved = False

        for k_pos in top_positions:
            k = int(k_pos)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi < i_lo:
                continue

            # Find the pair with largest product
            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                j = k - i
                prod = a_new[i] * a_new[j]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i

            if max_i < 0 or max_prod < 1e-20:
                continue

            j = k - max_i
            # Reduce the larger of the pair slightly
            if a_new[max_i] >= a_new[j]:
                reduction = max_i

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    N = _next_pow2(out_n)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    _LBFGS_STATE['beta'] = beta
    _LBFGS_STATE['n'] = n

    result = minimize(
        _ls_obj,
        z,
        jac=_ls_grad,
        method='L-BFGS-B',
        options={
            'maxiter': max_iter,
            'ftol': 1e-18,
            'gtol': 1e-14,
            'maxcor': 80,
            'maxls': 80,
        }
    )
    return _softmax(result.x)


def _adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    beta = max(gap * 0.5, c_max * 0.1)

    min_beta = max(c_max * 1e-6, 1e-12)
    beta_ratio = 0.45

    while time.monotonic() < deadline - min_time_remaining:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 10))))

        a_new = _run_lbfgs_iter(a, beta, deadline - min_time_remaining, max_iter)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
        new_beta = max(gap * 0.3, c_max * 0.002)

        if new_beta < beta * beta_ratio:
            beta = max(beta * beta_ratio, min_beta)
        else:
            beta = max(new_beta, min_beta)

        if beta <= min_beta * 1.05:
            a_new = _run_lbfgs_iter(a, min_beta, deadline - min_time_remaining, max_iter)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
            break

    return best_a, best_sc


def _peak_equalization(a_init, deadline, n_rounds=80):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    for _rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        n_peaks = min(8, len(c))
        top_positions = np.argsort(c)[::-1][:n_peaks]

        a_new = a.copy()

        for k_pos in top_positions:
            k = int(k_pos)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi < i_lo:
                continue

            max_prod = 0.0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                j = k - i
                prod = a_new[i] * a_new[j]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i

            if max_i < 0 or max_prod < 1e-20:
                continue

            j = k - max_i
            if a_new[max_i] >= a_new[j]:
                reduction = max_i
            else:
                reduction = j

            peak_center = k / 2.0
            min_contrib = np.inf
            add_pos = reduction
            for pos in range(0, n, max(1, n // 500)):
                cj = k - pos
                if 0 <= cj < n:
                    contrib = a_new[pos]
                else:
                    contrib = 0.0
                if contrib < min_contrib:
                    min_contrib = contrib
                    add_pos = pos

            amount = a_new[reduction] * 0.015
            a_new[reduction] -= amount
            a_new[add_pos] += amount

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a_small = a.copy()
            for k_pos in top_positions[:4]:
                k = int(k_pos)
                i_lo = max(0, k - n + 1)
                i_hi = min(k, n - 1)
                max_prod = 0.0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    j = k - i
                    prod = a_small[i] * a_small[j]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k - max_i
                    if a_small[max_i] >= a_small[j]:
                        a_small[max_i] *= 0.997
                    else:
                        a_small[j] *= 0.997
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small /= s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-14:
                    best_sc = sc_small
                    best_a = a_small.copy()
                    a = a_small
                else:
                    break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return _normalize(a_new)


def _perturb_structured(a, freq, amplitude, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n, dtype=np.float64)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = amplitude * np.sin(2.0 * np.pi * freq * t / n + phase)
    z += perturbation
    z += 0.005 * rng.standard_normal(n)
    z -= z.mean()
    return _softmax(z)


def _perturb_random(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z += sigma * rng.standard_normal(n)
    z -= z.mean()
    return _softmax(z)


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block2":
        a = np.zeros(n)
        block = max(1, n // 4)
        a[:block] = 1.0
        a[n - block:] = 1.0
        return _normalize(a)
    else:
        return np.ones(n) / n


def _update_best(p, sc, label):
    if p is None:
        return
    p_arr = np.asarray(p, dtype=np.float64).ravel()
    p_arr = np.clip(p_arr, 0.0, 1000.0)
    s = float(p_arr.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = _score(p_arr)
    if np.isfinite(sc) and sc < _BEST['score']:
        prev = _BEST['score']
        _BEST['seq'] = p_arr.copy()
        _BEST['score'] = float(sc)
        print(f"  NEW BEST: {sc:.10f} (n={len(p_arr)}) [{label}] "
              f"improve={prev - sc:.2e} t={time.monotonic() - _T0[0]:.1f}s")


def propose_candidate(seed=42, budget_s=1800):
    _T0[0] = time.monotonic()
    deadline = _T0[0] + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    _BEST['seq'] = None
    _BEST['score'] = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "parent")
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # Phase 1: Multi-resolution warm start
    print(f"\n[Phase 1] Multi-resolution warm start (t={time.monotonic() - _T0[0]:.1f}s)")
    target_n = n_parent if n_parent > 0 else 16384
    small_ns = [1024, 2048, 4096, 8192]
    small_ns = [x for x in small_ns if x < target_n]

    a_warm = parent
    for sn in small_ns:
        if time.monotonic() > deadline - 600:
            break
        a_small = _interpolate_to_n(a_warm, sn)
        sc_small = _score(a_small)
        print(f"  n={sn}: score={sc_small:.10f}")
        # Quick L-BFGS at this scale
        c_s = _full_convolve(a_small, a_small)
        cmax_s = float(c_s.max())
        beta_s = max(cmax_s * 0.05, 1e-6)
        a_opt_s = _run_lbfgs_iter(a_small, beta_s, min(deadline - 580, time.monotonic() + 30), 2000)
        sc_opt_s = _score(a_opt_s)
        if sc_opt_s < sc_small:
            a_warm = a_opt_s
        else:
            a_warm = a_small

    # Interpolate warm start back to target size
    a_target = _interpolate_to_n(a_warm, target_n)
    sc_target = _score(a_target)
    _update_best(a_target, sc_target, f"warm_n{target_n}")

    # Phase 2: Main adaptive LSE L-BFGS at target n
    print(f"\n[Phase 2] Adaptive LSE L-BFGS n={target_n} (t={time.monotonic() - _T0[0]:.1f}s)")
    a_main, sc_main = _adaptive_lbfgs(a_target, deadline, min_time_remaining=400.0)
    _update_best(a_main, sc_main, "lbfgs_main")

    # Phase 3: Active-set Newton refinement
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - _T0[0]:.1f}s)")
    if _BEST['seq'] is not None and time.monotonic() < deadline - 300:
        as_dl = min(deadline - 300, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(_BEST['seq'], as_dl, max_iter=300,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        _update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 200:
            as_dl2 = min(deadline - 200, time.monotonic() + 100)
            a_as2, sc_as2 = _active_set_newton(_BEST['seq'], as_dl2, max_iter=200,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            _update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 4: Peak equalization
    print(f"\n[Phase 4] Peak equalization (t={time.monotonic() - _T0[0]:.1f}s)")
    if _BEST['seq'] is not None and time.monotonic() < deadline - 200:
        pe_dl = min(deadline - 200, time.monotonic() + 80)
        a_pe, sc_pe = _peak_equalization(_BEST['seq'], pe_dl, n_rounds=100)
        _update_best(a_pe, sc_pe, "peak_eq")

        if sc_pe < _BEST['score'] + 1e-10 and time.monotonic() < deadline - 100:
            # Polish after peak equalization
            pol_dl = min(deadline - 100, time.monotonic() + 60)
            a_pol, sc_pol = _active_set_newton(a_pe, pol_dl, max_iter=100,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001])
            _update_best(a_pol, sc_pol, "polish_post_pe")

    # Phase 5: Structured restarts
    print(f"\n[Phase 5] Structured restarts (t={time.monotonic() - _T0[0]:.1f}s)")
    if _BEST['seq'] is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            ("freq1", 1, 0.05),
            ("freq2", 2, 0.04),
            ("freq3", 3, 0.03),
            ("freq5", 5, 0.02),
            ("rand_small", None, 0.03),
            ("rand_mid", None, 0.08),
        ]
        for label, freq, amp in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            if freq is not None:
                p_start = _perturb_structured(_BEST['seq'], freq, amp, rng)
            else:
                p_start = _perturb_random(_BEST['seq'], amp, rng)

            # Quick L-BFGS
            remaining = deadline - 30 - time.monotonic()
            if remaining < 15:
                break
            c_r = _full_convolve(p_start, p_start)
            cmax_r = float(c_r.max())
            beta_r = max(cmax_r * 0.03, 1e-6)
            max_it_r = min(3000, max(100, int(remaining / 0.03)))
            a_r = _run_lbfgs_iter(p_start, beta_r, deadline - 30, max_it_r)
            sc_r = _score(a_r)
            _update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < _BEST['score'] + 0.002 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 25),
                                                  max_iter=80,
                                                  tol_list=[0.002, 0.001, 0.0005, 0.0002])
                _update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 6: Try different n values
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - _T0[0]:.1f}s)")
    for n_try in [8192, 16384, 32768]:
        if time.monotonic() > deadline - 40:
            break
        if n_try == target_n:
            continue
        if n_try > 32768:
            continue

        a_init_n = _interpolate_to_n(_BEST['seq'] if _BEST['seq'] is not None else parent, n_try)
        sc_init_n = _score(a_init_n)
        print(f"  n={n_try}: init score={sc_init_n:.10f}")

        c_n = _full_convolve(a_init_n, a_init_n)
        cmax_n = float(c_n.max())
        beta_n = max(cmax_n * 0.05, 1e-6)
        remaining = deadline - 40 - time.monotonic()
        max_it_n = min(5000, max(200, int(remaining / 0.05)))
        a_n = _run_lbfgs_iter(a_init_n, beta_n, deadline - 40, max_it_n)
        sc_n = _score(a_n)
        _update_best(a_n, sc_n, f"n_{n_try}")

    # Phase 7: Final ultra-tight polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - _T0[0]:.1f}s)")
    if _BEST['seq'] is not None and time.monotonic() < deadline - 5:
        # Tight Newton
        fin_dl = min(deadline - 3, time.monotonic() + 50)
        a_fin, sc_fin = _active_set_newton(_BEST['seq'], fin_dl, max_iter=200,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        _update_best(a_fin, sc_fin, "final_newton")

        # Peak equalization one more time
        if time.monotonic() < deadline - 3:
            a_pe2, sc_pe2 = _peak_equalization(_BEST['seq'], min(deadline - 2, time.monotonic() + 20),
                                               n_rounds=30)
            _update_best(a_pe2, sc_pe2, "final_pe")

        # Ultra-tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(_BEST['seq'])
            c_best = _full_convolve(_BEST['seq'], _BEST['seq'])
            mb = float(c_best.max())
            beta_final = mb * 1e-5
            _LBFGS_STATE['beta'] = beta_final
            _LBFGS_STATE['n'] = n
            z = np.log(np.maximum(_BEST['seq'], 1e-30))
            z = z - z.mean()
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter_fin = min(3000, max(100, int(remaining / 0.04)))
                result = minimize(
                    _ls_obj, z,
                    jac=_ls_grad,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter_fin, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 80, 'maxls': 100}
                )
                a_fin3 = _softmax(result.x)
                _update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")

    # Final validation
    print(f"\n{'=' * 60}")
    if _BEST['seq'] is not None:
        final = np.clip(_BEST['seq'], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - _T0[0]:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
