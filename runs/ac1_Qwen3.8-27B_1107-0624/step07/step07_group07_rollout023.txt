c[k_i](a + δ) ≈ c[k_i](a) + 2 * (M_{k_i} a) · δ = t for all i.

where M_{k_i} a is the vector with components (M_{k_i} a)[j] = a[k_i - j] + a[j - k_i] (approximately).

So we need: 2 * (M_{k_i} a) · δ = t - c[k_i](a) for all i.

This is a system of m equations in n unknowns (with the additional constraint sum δ = 0). We can solve for the minimum-norm δ:

δ = argmin ||δ||^2 s.t. A δ = b, sum δ = 0

where A[i,:] = (M_{k_i} a) and b[i] = (t - c[k_i](a))/2.

The solution is δ = A^T (A A^T + λ I)^{-1} b (approximately, with the sum constraint handled by projecting).

This is a "minimum-norm least squares" problem. For small m, we can solve it directly.

Let me implement this:

```python
    c_max = c.max()
    # Find top peaks

    # Compute gradients
    gr = np.zeros((top_k, n))
    for i, k in enumerate(idx):

    target = c_max

    # Project onto sum=0 subspace
    gr_0 = gr - gr.mean(axis=1, keepdims=True)  # Make each row sum to 0
    # Solve: min ||δ||^2 s.t. gr_0 @ δ = b
```

Wait, I need to be more careful. The gradient of c_k with respect to a is:

But this is O(n) per k, and we have top_k values of k, so O(n * top_k) total. For n=8192 and top_k=20, that's 164K operations, which is fast.

Actually, we can compute this more efficiently using the convolution trick:
∇c_k = conv(e_k, a_rev)[n-1:2n-1] + conv(e_k_rev, a)[n-1:2n-1]

where e_k is a one-hot vector at position k. But actually, conv(e_k, a) is just a shifted version of a. So:
(∇c_k)[j] = a[k-j] + a[j-k] (with bounds checking).

This is simply: shift a by k (forward) + shift a by k (backward). We can compute this with numpy:

```python
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:

    # a[j-k] for j in range(n): valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2+1] += a[lo2-k:hi2-k+1]
    return g
```

```python
"""
Step sequence optimizer v2: Ultra-fine beta continuation, multi-peak Newton
equalization, Nesterov-accelerated subgradient, multi-length exploration (n=8192,
16384), and intensive coordinate descent polishing.
Key new ideas: (1) Multi-peak Newton direction that solves a small LSQ to equalize
the top m convolution peaks simultaneously; (2) Nesterov momentum for gradient
phases; (3) Adaptive active-set threshold that tightens over iterations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _conv_grad_k(a, k, n):
    """Compute gradient of c_k = conv(a,a)[k] with respect to a."""
    idx1 = k - np.arange(n)
    g1 = np.zeros(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    g1[mask1] = a[idx1[mask1]]
    idx2 = np.arange(n) - k
    g2 = np.zeros(n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    g2[mask2] = a[idx2[mask2]]
    return g1 + g2


def _multi_peak_newton_direction(a, n, top_m=15):
    """Compute a Newton direction that equalizes the top m peaks."""
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n), c
    idx = np.argsort(c)[::-1][:top_m]
    # Build gradient matrix G (top_m x n)
    G = np.zeros((top_m, n))
    for i, k in enumerate(idx):
        G[i] = _conv_grad_k(a, k, n)
    # Target: reduce each peak to c_max * (1 - small_factor)
    # Actually, we want to equalize: set target = mean of top peaks
    target = float(np.mean(c[idx]))
    b = (target - c[idx]) * 0.5  # factor 0.5 because d(c_k)/da = 2*M_k*a approximately
    # Project G rows to sum=0 subspace (since we constrain sum(delta)=0)
    G_proj = G - G.mean(axis=1, keepdims=True) * 1.0  # each row minus its mean
    # Wait: we need sum(delta)=0, so we project the solution.
    # Minimum norm solution: delta = G^T (G G^T)^{-1} b
    # Then project to sum=0: delta = delta - mean(delta)
    try:
        GGt = G_proj @ G_proj.T + 1e-12 * np.eye(top_m)
        sol = np.linalg.solve(GGt, b)
        delta = G_proj.T @ sol
    except np.linalg.LinAlgError:
        delta = G_proj.T @ np.linalg.lstsq(G_proj, b, rcond=None)[0]
    # Project to sum=0
    delta = delta - delta.mean()
    # Clip to ensure a+delta >= 0
    if a.min() > 0:
        max_neg = -delta.min()
        if max_neg > a.min() * 0.5:
            delta = delta * (a.min() * 0.5) / max_neg
    return delta, c


def _nesterov_subgrad(a_init, deadline, max_steps=50000, lr_init=0.02):
    """Nesterov-accelerated subgradient in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum parameter
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        z_y = z + mu * v
        a_y = _softmax(z_y)
        c = _full_convolve(a_y, a_y)
        c_max = c.max()

        # Subgradient: active set = top peaks
        n_active = max(3, min(20, int(0.01 * len(c))))
        top_idx = np.argsort(c)[::-1][:n_active]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / n_active

        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_y * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step
        step_size = lr / (1.0 + 0.01 * i)
        z = z - step_size * grad
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        v = (z - y) / (step_size + 1e-30) * mu + (1 - mu) * v
        # Simpler momentum:
        v = mu * v - step_size * grad
        y = z + mu * v

        # Periodic evaluation
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                lr *= 1.02
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.8
                    mu = max(0.5, mu * 0.95)
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_equalize_iter(a_init, deadline, max_iters=20000, top_m=15):
    """Iterative multi-peak Newton equalization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    step_scale = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        delta, c = _multi_peak_newton_direction(a, n, top_m)
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-15:
            break
        delta = delta / delta_norm

        # Line search
        best_local = a.copy()
        best_local_sc = best_sc
        improved = False
        for scale in [step_scale, step_scale * 0.5, step_scale * 0.25, step_scale * 0.1]:
            a_try = a + scale * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_local_sc - 1e-15:
                best_local_sc = sc_try
                best_local = a_try.copy()
                improved = True
                break

        if improved:
            best_sc = best_local_sc
            best_a = best_local.copy()
            a = best_local
            step_scale = min(step_scale * 1.05, 0.1)
            stall = 0
        else:
            step_scale *= 0.8
            stall += 1
            if stall > 50:
                break
            if step_scale < 1e-8:
                break

    return best_a, best_sc


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxcor': 150,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

    if max_c < 1e-30:
        return a_start

    # Longer, finer schedule

                              7e-5, 3.5e-5, 1.5e-5, 7e-6, 3e-6, 1.2e-6,
                              5e-7, 2e-7, 8e-8, 3e-8, 1e-8, 4e-9, 1.5e-9,
                              5e-10])
    current_a = a_start.copy()

def _coordinate_descent(a_init, deadline, max_rounds=20000):
    """Intensive coordinate descent with smart pairing."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Random pair
        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        # Golden section or uniform search over the 1D subproblem
        for frac in np.linspace(0.02, 0.98, 25):
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            if abs(new_i - a[idx]) < 1e-15:
                continue
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        # Occasionally try triple
        if rnd % 7 == 0:
            idx3 = int(rng.integers(0, n))
            if idx3 != idx and idx3 != jdx:
                s3 = a[idx] + a[jdx] + a[idx3]
                if s3 > 1e-15:
                    f1, f2, f3 = 0.33, 0.33, 0.34
                    for (ff1, ff2, ff3) in [(0.2, 0.4, 0.4), (0.4, 0.3, 0.3),
                                            (0.3, 0.5, 0.2), (0.5, 0.25, 0.25)]:
                        a_try3 = a.copy()
                        a_try3[idx] = s3 * ff1
                        a_try3[jdx] = s3 * ff2
                        a_try3[idx3] = s3 * ff3
                        sc_try3 = _score(a_try3)
                        if sc_try3 < best_sc - 1e-14:
                            best_sc = sc_try3
                            best_a = a_try3.copy()
                            a = a_try3
                            improved = True
                            break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.08 * np.cos(2.0 * np.pi * 5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_for_restart(a, rng, mode="gauss", scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.002 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(16, n // 32)
        for _ in range(12):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.005, 0.005)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: ultra-fine L-BFGS continuation, multi-peak Newton
    equalization, Nesterov subgradient, multi-length exploration, and
    intensive coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(8192, "uniform")
        print(f"No parent. Using uniform n=8192, score={_score(a0):.10f}")
        try_update(a0, "init")

    start_n = len(best_seq) if best_seq is not None else 8192

    # === Phase 1: Fine L-BFGS on original length ===
    print(f"\n=== Phase 1: Fine L-BFGS on n={start_n} ===")
    t_phase = time.monotonic()
    a_p1 = _lbfgs_schedule(best_seq, deadline - 800, min_time_remaining=700.0)
    try_update(a_p1, "lbfgs_fine")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: Multi-peak Newton equalization ===
    print(f"\n=== Phase 2: Multi-peak Newton ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 600:
        a_mp, sc_mp = _peak_equalize_iter(best_seq, deadline - 600, max_iters=30000, top_m=12)
        try_update(a_mp, "multi_peak_newton")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: More L-BFGS (second pass) ===
    print(f"\n=== Phase 3: Second L-BFGS pass ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 500:
        a_p3 = _lbfgs_schedule(best_seq, deadline - 500, min_time_remaining=400.0)
        try_update(a_p3, "lbfgs_p2")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov subgradient ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400:
        a_ng, sc_ng = _nesterov_subgrad(best_seq, deadline - 400, max_steps=40000)
        try_update(a_ng, "nesterov")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Multi-peak Newton again ===
    print(f"\n=== Phase 5: Multi-peak Newton (2nd) ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_mp2, _ = _peak_equalize_iter(best_seq, deadline - 300, max_iters=20000, top_m=20)
        try_update(a_mp2, "multi_peak_2")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Try larger n (16384) ===
    print(f"\n=== Phase 6: Try n=16384 ===")
    t_phase = time.monotonic()
    if start_n < 16384 and time.monotonic() < deadline - 250:
        print("  Interpolating to n=16384...")
        a_big = _interpolate_to_n(best_seq, 16384)
        sc_big = _score(a_big)
        print(f"    Interpolated: score={sc_big:.10f}")
        a_big_opt = _lbfgs_schedule(a_big, deadline - 200, min_time_remaining=100.0)
        try_update(a_big_opt, "n=16384")
        if time.monotonic() < deadline - 100:
            a_big_mp, _ = _peak_equalize_iter(a_big_opt, deadline - 100, max_iters=10000, top_m=10)
            try_update(a_big_mp, "n=16384_mp")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Ultra-fine L-BFGS ===
    print(f"\n=== Phase 7: Ultra-fine L-BFGS ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_ultra = _lbfgs_ultra_fine(best_seq, deadline - 80, min_time_remaining=50.0)
        try_update(a_ultra, "ultra_lbfgs")
    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Multi-peak Newton (final) ===
    print(f"\n=== Phase 8: Final multi-peak Newton ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_final_mp, _ = _peak_equalize_iter(best_seq, deadline - 60, max_iters=15000, top_m=25)
        try_update(a_final_mp, "final_mp")
    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Coordinate descent ===
    print(f"\n=== Phase 9: Coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 30, max_rounds=30000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 10: Multi-restart ===
    print(f"\n=== Phase 10: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("gauss_s1", "gauss", 0.01),
            ("gauss_s3", "gauss", 0.03),
            ("fourier", "fourier", 0.0),
            ("block", "block", 0.0),
        ]
        for name, mode, scale in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb_for_restart(best_seq, rng, mode, scale)
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")
            # Quick L-BFGS
            n_r = len(p_start)
            c_r = _full_convolve(p_start, p_start)
            mb_r = float(c_r.max())
            betas_r = mb_r * np.array([0.05, 0.01, 0.002, 5e-4, 1e-4])
            a_r = p_start
            for br in betas_r:
                if time.monotonic() > deadline - 15:
                    break
                rem = deadline - 15 - time.monotonic()
                if rem < 5:
                    break
                mi = min(3000, max(200, int(rem / 0.002)))
                a_r, _, _ = _lbfgs_run(a_r, br, n_r, mi)
            try_update(a_r, f"restart_{name}")
            # Quick peak equalize
            if time.monotonic() < deadline - 10:
                a_r2, _ = _peak_equalize_iter(a_r, deadline - 10, max_iters=3000, top_m=10)
                try_update(a_r2, f"restart_{name}_mp")

    print(f"  Phase 10 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 11: Final intensive cycle ===
    print(f"\n=== Phase 11: Final polish cycle ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(4):
            if time.monotonic() > deadline - 5:
                break

            # Peak equalize
            if time.monotonic() < deadline - 5:
                a_f1, _ = _peak_equalize_iter(best_seq, deadline - 5, max_iters=5000, top_m=15)
                try_update(a_f1, f"final_polish_mp_{cycle}")

            # Ultra L-BFGS
            if time.monotonic() < deadline - 5 and best_seq is not None:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-9
                max_iter = min(5000, max(200, int((deadline - 5 - time.monotonic()) / 0.002)))
                a_f2, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                try_update(a_f2, f"final_polish_lbfgs_{cycle}")

            # Coordinate descent
            if time.monotonic() < deadline - 5:
                a_f3, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=5000)
                try_update(a_f3, f"final_polish_cd_{cycle}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
