import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        if not np.isfinite(alpha):
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.01:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_polish(a_init, deadline, max_rounds=80):
    """CG-Newton polish on the active set with multi-scale tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Multi-scale active set: start wide, tighten progressively
    tol_schedule = [0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                    0.00002, 0.00001, 0.000005]
    tol_idx = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)] * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            # Try tightening active set
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
            continue

        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Damped Newton with backtracking
        alpha = 1.0
        step_done = False
        c_max_best = c_max
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max_best:
                c_max_best = c_max_try
                a = a_try
                step_done = True
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    c_max_best = float(c_try.max())
                    step_done = True
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
        elif not step_done:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very aggressive adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    # Much more aggressive: 40 levels of beta reduction
    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                       0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                       0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                       0.0000005, 0.0000002, 0.0000001, 0.00000005,
                       0.00000002, 0.00000001, 0.000000005, 0.000000002,
                       0.000000001, 0.0000000005])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_iter_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 20))))

        try:
            result = minimize(
                _make_smoobj_wrapper(z, beta, n) if False else _smoobj,
                z,
                args=(beta, n),
                jac=_make_smograd_wrapper if False else _smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement, try a few more iterations with same beta
                if i > 5 and i % 3 == 0:
                    continue
        except Exception:
            pass

    return current_a


def _smograd_wrapper(z, beta, n):
    return _smograd(z, beta, n)


def _subgrad_nesterov(a_init, deadline, max_steps=8000):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.03
    momentum = 0.9
    z_prev = z.copy()
    z_mom = z.copy()

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        z_look = z_mom + momentum * (z_mom - z_prev)
        a = _softmax(z_look)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 15 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.08)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    z_mom = z.copy()
                    patience = 0
                    lr = 0.02
                    momentum = 0.8
            z = z_new
        else:
            z = z_new

        z_prev = z_mom.copy()
        z_mom = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.08, 0.12, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs_aggressive(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2 * t - 1) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.2) ** 2) + np.exp(-6.0 * (t - 0.8) ** 2) + 0.03
        return a / a.sum()
    elif kind == "quadrupodal":
        t = np.linspace(0, 1, n)
        a = (np.exp(-8.0 * (t - 0.15) ** 2) + np.exp(-8.0 * (t - 0.5) ** 2)
             + np.exp(-8.0 * (t - 0.85) ** 2) + 0.02)
        return a / a.sum()
    elif kind == "parabola":
        t = np.linspace(0, 1, n)
        a = t * (1 - t) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_perturb(a, rng, strength=0.1):
    """Perturb the spectrum of the sequence to explore different structures."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Add random phase perturbation to non-DC components
    phase = rng.uniform(0, 2 * np.pi, size=len(A))
    A_new = A * (1 + strength * np.exp(1j * phase))
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search with aggressive optimization:
    - Main L-BFGS with 30-level beta continuation
    - Multi-scale CG-Newton active set polishing
    - Nesterov subgradient
    - Support reduction with re-optimization
    - Multi-n exploration (4096, 8192, 16384)
    - Diverse restarts with spectral and structural perturbations
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "edge")
        print(f"No parent, using edge profile n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Also try different initializations
    alt_inits = {}
    for kind in ["edge", "bimodal", "tapered", "parabola"]:
        if kind == "edge" and n_parent == 8192:
            continue
        alt = _make_profile(n_parent, kind)
        alt_sc = _score(alt)
        alt_inits[kind] = (alt, alt_sc)
        print(f"  Alt init '{kind}': score={alt_sc:.10f}")
        if alt_sc < best_score:
            update_best(alt, alt_sc, f"init_{kind}")

    # Phase 1: Main L-BFGS from best initialization (60% of time)
    t_phase1_end = t_start + 0.50 * (budget_s - 20)
    print(f"\n[Phase 1] Main L-BFGS aggressive (deadline={t_phase1_end - t_start:.0f}s)")
    a1 = _run_lbfgs_aggressive(parent, t_phase1_end, min_time_remaining=300.0)
    sc1 = _score(a1)
    update_best(a1, sc1, "lbfgs_main")

    # Also optimize from the best alternative if it was competitive
    for kind, (alt, alt_sc) in alt_inits.items():
        if alt_sc < best_score + 0.01 and time.monotonic() < t_phase1_end - 60:
            a_alt = _run_lbfgs_aggressive(alt, min(t_phase1_end - 30, time.monotonic() + 120),
                                          min_time_remaining=200.0)
            update_best(a_alt, _score(a_alt), f"lbfgs_{kind}")

    # Phase 2: CG-Newton active set polish (15% of time)
    t_phase2_end = t_start + 0.65 * (budget_s - 20)
    print(f"\n[Phase 2] CG-Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < t_phase2_end:
        a2, sc2 = _newton_active_polish(best_seq, min(t_phase2_end, time.monotonic() + 200),
                                         max_rounds=100)
        update_best(a2, sc2, "newton_1")

    # Phase 3: Nesterov subgradient (10% of time)
    t_phase3_end = t_start + 0.75 * (budget_s - 20)
    print(f"\n[Phase 3] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < t_phase3_end:
        a3, sc3 = _subgrad_nesterov(best_seq, min(t_phase3_end, time.monotonic() + 150),
                                     max_steps=10000)
        update_best(a3, sc3, "nesterov_1")

    # Phase 4: Second Newton polish after subgradient
    t_phase4_end = t_start + 0.80 * (budget_s - 20)
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < t_phase4_end:
        a4, sc4 = _newton_active_polish(best_seq, min(t_phase4_end, time.monotonic() + 100),
                                         max_rounds=80)
        update_best(a4, sc4, "newton_2")

    # Phase 5: Support reduction + re-optimize (10% of time)
    t_phase5_end = t_start + 0.88 * (budget_s - 20)
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < t_phase5_end:
        a5, sc5 = _shrinkage_and_reopt(best_seq, t_phase5_end, min_time_remaining=100.0)
        update_best(a5, sc5, "shrinkage")

    # Phase 6: Multi-n exploration
    t_phase6_end = t_start + 0.93 * (budget_s - 20)
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384]
    for n_try in n_candidates:
        if time.monotonic() > t_phase6_end - 60:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt = _run_lbfgs_aggressive(a_init, min(t_phase6_end - 30, time.monotonic() + 120),
                                       min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < t_phase6_end - 20:
            a_np, sc_np = _newton_active_polish(a_opt, min(t_phase6_end - 5, time.monotonic() + 60),
                                                 max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 7: Diverse restarts (5% of time)
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        restart_profiles = []
        restart_profiles.append(_perturb(best_seq, 0.05, rng))
        restart_profiles.append(_perturb(best_seq, 0.15, rng))
        restart_profiles.append(_spectral_perturb(best_seq, rng, strength=0.15))
        restart_profiles.append(_make_profile(n, "quadrupodal"))

        for i, p_start in enumerate(restart_profiles):
            if time.monotonic() > deadline - 25:
                break
            a_r = _run_lbfgs_aggressive(p_start, min(deadline - 15, time.monotonic() + 60),
                                         min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 5, time.monotonic() + 30),
                                                     max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS with extremely small beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        _smoobj, z, args=(beta_fin, n),
                        jac=_smograd,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "edge")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
