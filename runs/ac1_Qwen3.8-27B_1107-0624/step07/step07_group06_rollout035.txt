"""
Optimized step-function sequence generator for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses aggressive L-BFGS with smooth-max continuation, Newton active-set polish,
KKT-based fixed-point iteration, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=0.003):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a), np.zeros_like(c)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad, w


def _hess_matvec(v, active, n, fa_cached=None):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N)
    if fa_cached is not None:
        fa = fa_cached
    else:
        fa = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fv * fa, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if not np.isfinite(alpha) or alpha <= 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        if rs > 1e-30:
            beta_cg = r_new / rs
        else:
            beta_cg = 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active(a_init, deadline, max_iter=300):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        fa_cache = np.fft.rfft(active.astype(np.float64), _next_pow2(4 * n))

        def hess_vec(v):
            return _hess_matvec(v, active, n, fa_cached=fa_cache) + 1e-9 * v

        grad = hess_vec(a) - 1e-9 * a
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=200, tol=1e-11)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for _ in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * max(alpha * gnorm, 1e-15):
                a_new = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.003 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        fa_cache = np.fft.rfft(active.astype(np.float64), _next_pow2(4 * n))

        def hess_vec(v):
            return _hess_matvec(v, active, n, fa_cached=fa_cache) + 1e-8 * v

        target = np.full(n, 1.0)
        dx = _cg_solve(hess_vec, target - hess_vec(a) + 1e-8 * a, n, max_iter=150, tol=1e-10)
        a_new = a + dx
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_check = _full_convolve(a_new, a_new)
        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            sc
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0] - c_sorted[-1]
    base_beta = max(gap * 0.5, max_c * 0.05)

    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001,
         0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(12000, max(100, int(remaining / (est_time * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad, _ = _active_grad(a, c, n, tol_frac=0.002)
        step = lr / (1.0 + 0.002 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.02 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: aggressive L-BFGS continuation + Newton active-set
    + KKT fixed-point + subgradient polish + multi-n exploration + restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        _update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update_best(parent, _score(parent), "init")

    n_work = len(best_seq) if best_seq is not None else 8192

    # === Phase 1: Aggressive L-BFGS from parent ===
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_main = _run_lbfgs(best_seq, deadline, min_time_remaining=400)
        _update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 2: Newton active-set ===
    print(f"\n[Phase 2] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_ne, sc_ne = _newton_active(best_seq, min(deadline - 300, time.monotonic() + 200), max_iter=300)
        _update_best(a_ne, sc_ne, "newton")

    # === Phase 3: KKT fixed-point ===
    print(f"\n[Phase 3] KKT fixed-point (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_kkt, sc_kkt = _kkt_fixed_point(best_seq, min(deadline - 250, time.monotonic() + 150), max_iter=200)
        _update_best(a_kkt, sc_kkt, "kkt_fp")

    # === Phase 4: Second L-BFGS pass (lower betas) ===
    print(f"\n[Phase 4] L-BFGS second pass (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_l2 = _run_lbfgs(best_seq, deadline, min_time_remaining=200)
        _update_best(a_l2, _score(a_l2), "lbfgs_pass2")

    # === Phase 5: Subgradient polish ===
    print(f"\n[Phase 5] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 100),
                                      n_max_steps=5000, lr_init=0.012)
        _update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60)
        _update_best(a_opt, _score(a_opt), f"n_{n_try}")

        if time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _newton_active(a_opt, min(deadline - 40, time.monotonic() + 50), max_iter=100)
            _update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restarts = [
            (lambda: _perturb_log(best_seq, 0.02, rng), "log"),
            (lambda: _perturb_fourier(best_seq, rng, n_modes=3), "fourier"),
            (lambda: _perturb_log(best_seq, 0.05, rng), "log_big"),
            (lambda: _make_profile(len(best_seq), "cos2"), "cos2"),
        ]
        for i, (resample, label) in enumerate(restarts):
            if time.monotonic() > deadline - 20:
                break
            try:
                a_start = resample()
            except Exception:
                continue
            a_r = _run_lbfgs(a_start, deadline, min_time_remaining=10)
            _update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 8: Final intense polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Newton
        a_f1, sc_f1 = _newton_active(best_seq, deadline - 5, max_iter=200)
        _update_best(a_f1, sc_f1, "final_newton")

        # KKT
        if time.monotonic() < deadline - 5:
            a_f2, sc_f2 = _kkt_fixed_point(best_seq, deadline - 5, max_iter=150)
            _update_best(a_f2, sc_f2, "final_kkt")

        # Ultra-low beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.000003
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(200, int(remaining / 0.01)))

                def fun_final(z_local):
                    return _smoobj(z_local, beta_final, n)

                def jac_final(z_local):
                    return _smograd(z_local, beta_final, n)

                try:
                    result = minimize(
                        fun_final, z, jac=jac_final, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 60}
                    )
                    a_f3 = _softmax(result.x)
                    _update_best(a_f3, _score(a_f3), "final_ultra")
                except Exception:
                    pass

        # Subgradient final
        if time.monotonic() < deadline - 3:
            a_f4, sc_f4 = _subgrad_polish(best_seq, deadline - 3, n_max_steps=3000, lr_init=0.005)
            _update_best(a_f4, sc_f4, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
