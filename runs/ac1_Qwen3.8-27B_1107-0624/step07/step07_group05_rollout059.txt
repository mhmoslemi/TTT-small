import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm1(a):
    """Score for a with sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=0.001):

    w = np.zeros_like(c)

    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

    best_a = a.copy()
    best_sc = _score_norm1(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.008 * c_max
        active = c >= (c_max - tol)
        n_act = active.sum()
        if n_act < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = 5

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 50 == 0:
            sc = _score_norm1(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm1(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=10000, lr_init=0.02):
    """Subgradient descent in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_grad(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + 0.0001 * i)
        z = z - step * grad
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=20.0, label=""):
    """L-BFGS with very aggressive beta continuation."""
    n = len(a_init)
    a_init = a_init.copy() / max(a_init.sum(), 1e-15)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    # Very aggressive: more steps at small beta
    betas = base_beta * np.array([
        1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
        0.0002, 0.0001, 0.00005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 2:
            break

        # Estimate per-iteration time
        est = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est * 10))))

        try:
            result = minimize(
                _smoobj, z.copy(), args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200, 'maxfun': max_iter * 5}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_perturb(a, rng, n_modes=8, amplitude=0.02):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    nfft = len(z_hat)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, nfft - 1)))
        if freq < nfft:
            delta = amplitude * rng.uniform(0.5, 2.5) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if nfft > 1 and (nfft - 1 - freq) < nfft:
                z_hat[nfft - 1 - freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_concentrated":
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * t) + np.exp(-8 * (1 - t)) + 0.1
        return a / a.sum()
    elif kind == "double_peak":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * t) + 0.15 * np.cos(2 * np.pi * 19 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_positive":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 5 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_positive":
        rng = np.random.default_rng(0)
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _multi_n_explore(parent, best_seq, best_score, deadline, rng):
    """Try a few different n values to see if any give better results."""
    results = []
    n_parent = len(parent) if parent is not None else 8192

    # Try a few n values near the parent
    candidates = []
    for delta in [-256, -128, 128, 256, 512]:
        n_try = n_parent + delta
        if 1024 <= n_try <= 16384:
            candidates.append(n_try)
    # Also try some standard sizes
    for n_std in [4096, 6144, 10240, 12288]:
        if n_std != n_parent and 1024 <= n_std <= 16384:
            candidates.append(n_std)

    candidates = list(set(candidates))
    candidates.sort(reverse=True)
    candidates = candidates[:4]  # Limit to 4

    for n_try in candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 20:
            break

        # Interpolate parent to new length
        if parent is not None:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0.0)
            s = a_init.sum()
            if s < 1e-10:
                a_init = _make_profile(n_try, "tapered")
            else:
                a_init = a_init / s
        else:
            a_init = _make_profile(n_try, "tapered")

        # Quick L-BFGS
        time_budget = min(remaining * 0.7, 120.0)
        a_opt = _run_lbfgs(a_init, time.monotonic() + time_budget, min_time_remaining=5)
        sc = _score(a_opt)
        results.append((sc, a_opt))
        print(f"  n={n_try}: score={sc:.10f}")

    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: aggressive L-BFGS continuation, extended pressure
    equalization, subgradient polish, multi-n exploration, and spectral restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        sc_parent = _score(parent)
        print(f"No parent, using uniform n={n_parent}, score={sc_parent:.10f}")
        _update(parent)

    # === Phase 1: Aggressive L-BFGS (main workhorse) ===
    print(f"\n[Phase 1] Aggressive L-BFGS on parent (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = t_start + (deadline - t_start) * 0.45
    a1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=10, label="main")
    sc1 = _score(a1)
    _update(a1, sc1)
    print(f"  After Phase 1: score={sc1:.10f}")

    # === Phase 2: Extended pressure equalization ===
    print(f"\n[Phase 2] Pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        pe_deadline = min(deadline - 600, time.monotonic() + 180)
        a_pe, sc_pe = _pressure_equalize(best_seq, pe_deadline, n_iters=1000, gamma=0.35)
        sc_pe_full = _score(a_pe)
        _update(a_pe, sc_pe_full)
        print(f"  After Phase 2: score={sc_pe_full:.10f}")

        # Follow up with L-BFGS if improved
        if sc_pe_full < best_score + 0.001 and time.monotonic() < deadline - 500:
            a_follow = _run_lbfgs(a_pe, deadline, min_time_remaining=500, label="pe_follow")
            sc_follow = _score(a_follow)
            _update(a_follow, sc_follow)
            print(f"  After Phase 2 follow-up: score={sc_follow:.10f}")

    # === Phase 3: Second L-BFGS round (finer) ===
    print(f"\n[Phase 3] Second L-BFGS round (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3 = _run_lbfgs(best_seq, deadline, min_time_remaining=400, label="second")
        sc3 = _score(a3)
        _update(a3, sc3)
        print(f"  After Phase 3: score={sc3:.10f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        mn_results = _multi_n_explore(parent, best_seq, best_score,
                                      deadline - 150, rng)
        for sc_mn, a_mn in mn_results:
            _update(a_mn, sc_mn)

    # === Phase 5: Spectral restarts from best ===
    print(f"\n[Phase 5] Spectral restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        for i in range(4):
            if time.monotonic() > deadline - 50:
                break
            # Create diverse perturbation
            if i == 0:
                a_init = _spectral_perturb(best_seq, rng, n_modes=6, amplitude=0.015)
            elif i == 1:
                # Blend with a structured profile
                prof = _make_profile(n_best, "cosine_mod")
                a_init = 0.7 * best_seq + 0.3 * prof
                a_init = np.maximum(a_init, 0)
                a_init = a_init / a_init.sum()
                a_init = _spectral_perturb(a_init, rng, n_modes=4, amplitude=0.01)
            elif i == 2:
                prof = _make_profile(n_best, "double_peak")
                a_init = 0.6 * best_seq + 0.4 * prof
                a_init = np.maximum(a_init, 0)
                a_init = a_init / a_init.sum()
            else:
                a_init = _spectral_perturb(best_seq, rng, n_modes=10, amplitude=0.025)

            a_r = _run_lbfgs(a_init, deadline, min_time_remaining=50, label=f"restart_{i}")
            _update(a_r, _score(a_r))

    # === Phase 6: Subgradient polish ===
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        sg_deadline = min(deadline - 10, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline,
                                       n_max_steps=20000, lr_init=0.015)
        _update(a_sg, sc_sg)
        print(f"  After Phase 6: score={sc_sg:.10f}")

    # === Phase 7: Final intensive L-BFGS ===
    print(f"\n[Phase 7] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=3, label="final")
        _update(a_fin, _score(a_final) if 'a_final' in dir() else _score(a_fin))

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
