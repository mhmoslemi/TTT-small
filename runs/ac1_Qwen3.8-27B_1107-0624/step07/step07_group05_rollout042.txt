multi-n exploration. Designed to push below 1.5027.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _np2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    N = _np2(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    return float(2.0 * n * _conv(a, a).max() / (s * s))


def _score_n(a):
    return float(2.0 * len(a) * _conv(a, a).max())


def _sm(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    return np.ones_like(z) / len(z) if s < 1e-300 else e / s


def _obj(z, beta, n):
    a = _sm(z)
    c = _conv(a, a)
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _grad(z, beta, n):
    a = _sm(z)
    c = _conv(a, a)
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    h = _conv(w, a[::-1])[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _agrad(a, c, n, k=3):
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros_like(a)
    idx = np.argsort(c)[-k:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    h = _conv(w, a[::-1])[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _pequal(a, dline, nit=200, gam=0.35):
    n = len(a)
    a = a.copy()
    ba, bs = a.copy(), _score_n(a)
    for it in range(nit):
        if time.monotonic() > dline:
            break
        c = _conv(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break
        act = c >= cm - 0.02 * cm
        if act.sum() < 3:
            act = np.zeros_like(c, dtype=bool)
            act[np.argsort(c)[-5:]] = True
        p = _conv(act.astype(float), a[::-1])[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        a = a * np.power(np.mean(p) / p, gam)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s
        if it % 25 == 0:
            sc = _score_n(a)
            if sc < bs:
                bs, ba = sc, a.copy()
    sc = _score_n(a)
    if sc < bs:
        bs, ba = sc, a.copy()
    return ba, bs


def _lbfgs(a0, dline, mrem=10.0):
    n = len(a0)
    ca, cs = a0.copy(), _score(a0)
    z = np.log(np.maximum(ca, 1e-30))
    z -= z.mean()
    cc = _conv(a0, a0)
    cm = float(cc.max())
    if cm < 1e-15:
        return ca
    csrt = np.sort(cc)[::-1]
    gap = csrt[0] - csrt[min(20, len(csrt) - 1)]
    bb = max(gap * 0.5, cm * 0.01)
    betas = bb * np.array([1.0, .5, .25, .125, .062, .031, .016, .008,
                           .004, .002, .001, 5e-4, 2e-4, 1e-4, 5e-5,
                           2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 1e-8])
    et = 4e-4 * max(1.0, n / 4096.0)
    for i, beta in enumerate(betas):
        if time.monotonic() + mrem > dline:
            break
        rem = dline - mrem - time.monotonic()
        if rem < 3:
            break
        mi = int(min(50000, max(200, rem / (et * 8.0))))
        try:
            r = minimize(_obj, z, args=(beta, n), jac=_grad, method='L-BFGS-B',
                         options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-14,
                                  'maxcor': 100, 'maxls': 50})
            an = _sm(r.x)
            sn = _score(an)
            if sn < cs - 1e-15:
                ca, cs = an, sn
                z = np.log(np.maximum(an, 1e-30))
                z -= z.mean()
        except Exception:
            pass
    return ca


def _subpol(a0, dline, nst=5000, lr0=0.015):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z -= z.mean()
    ba, bs, lr, pat, stl = a0.copy(), _score(a0), lr0, 0, 0
    for i in range(nst):
        if time.monotonic() > dline:
            break
        a = _sm(z)
        c = _conv(a, a)
        g = _agrad(a, c, n, 3)
        if np.linalg.norm(g) < 1e-15:
            break
        st = lr / (1.0 + 1e-3 * i)
        z = np.clip(z - st * g, -60, 60)
        z -= z.mean()
        if i % 40 == 0:
            ac = _sm(z)
            sc = _score(ac)
            if sc < bs - 1e-13:
                bs, ba, pat, stl, lr = sc, ac.copy(), 0, 0, min(lr * 1.02, lr0)
            else:
                pat += 1
                stl += 1
                if pat % 4 == 0:
                    lr *= 0.88
                if pat > 25:
                    z = np.log(np.maximum(ba, 1e-30))
                    z -= z.mean()
                    pat, stl = 0, 0
    sc = _score(_sm(z))
    if sc < bs:
        bs, ba = sc, _sm(z).copy()
    return ba, bs


def _cdev(a0, dline, nrnd=80):
    n = len(a0)
    a = a0.copy()
    ba, bs = a.copy(), _score_n(a)
    for r in range(nrnd):
        if time.monotonic() > dline:
            break
        c = _conv(a, a)
        cm = float(c.max())
        act = c >= cm - 0.05 * cm
        contrib = _conv(act.astype(float), a[::-1])[n - 1:2 * n - 1] * a
        top = np.argsort(contrib)[-max(10, n // 80):]
        np.random.shuffle(top)
        for i in top[:min(len(top), max(5, n // 200))]:
            if time.monotonic() > dline:
                break
            ov, co = a[i], _conv(a, a).max()
            for f in [0.5, 0.7, 0.9, 1.1, 1.4]:
                a[i] = ov * f
                if a[i] < 0:
                    a[i] = ov
                    break
                ct = _conv(a, a).max()
                if ct < co - 1e-15:
                    pass
                else:
                    a[i] = ov
                    break
        s = a.sum()
        if s > 1e-15:
            a /= s
            sc = _score_n(a)
            if sc < bs:
                bs, ba = sc, a.copy()
    return ba, bs


def _profile(n, kind="uni"):
    if kind == "uni":
        return np.ones(n) / n
    t = np.linspace(0, 1, n)
    if kind == "tap":
        a = np.maximum(np.sin(np.pi * t) ** 0.6, 1e-10)
    elif kind == "exp":
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
    else:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 1e-10)
    return a / a.sum()


def _fourier_pert(a, rng, nm=6, amp=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    zh = np.fft.rfft(z)
    for _ in range(nm):
        f = int(rng.integers(1, max(2, n // 4)))
        if f < len(zh):
            d = amp * rng.uniform(0.5, 2.0) * rng.standard_normal()
            ph = rng.uniform(0, 2 * np.pi)
            zh[f] += d * np.exp(1j * ph)
            if f > 0 and f < len(zh) - 1:
                zh[-f] = np.conj(zh[f])
    zn = np.fft.irfft(zh, n)
    zn -= zn.mean()
    return _sm(zn)


def _upscale(a_old, n_new):
    x_old = np.linspace(0, 1, len(a_old))
    x_new = np.linspace(0, 1, n_new)
    a = np.interp(x_new, x_old, a_old)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else _profile(n_new, "tap")


def _pipeline(a0, dline, mrem=30.0):
    n = len(a0)
    ca, cs = a0.copy(), _score(a0)
    # L-BFGS
    if time.monotonic() < dline - mrem - 20:
        a1 = _lbfgs(ca, dline, mrem + 20)
        s1 = _score(a1)
        if s1 < cs:
            ca, cs = a1, s1
    # Pressure equalization
    if time.monotonic() < dline - mrem - 10:
        a2, sn2 = _pequal(ca, min(dline - mrem - 5, time.monotonic() + 90), 250, 0.3)
        s2 = _score(a2)
        if s2 < cs:
            ca, cs = a2, s2
    # L-BFGS again
    if time.monotonic() < dline - mrem - 5:
        a3 = _lbfgs(ca, dline, mrem + 5)
        s3 = _score(a3)
        if s3 < cs:
            ca, cs = a3, s3
    # Subgradient
    if time.monotonic() < dline - mrem:
        a4, s4 = _subpol(ca, dline - mrem, 3000, 0.01)
        if s4 < cs:
            ca, cs = a4, s4
    return ca, cs
