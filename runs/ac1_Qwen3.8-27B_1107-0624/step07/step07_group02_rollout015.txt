            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.4
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.01)

    return best_a, _score(best_a)


def _peak_shave(a_init, deadline, n_peaks=10, lr_init=0.01):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_init
    stall = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0

        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        peak_threshold = c[sorted_idx[0]]

        w = np.zeros_like(c)
        peak_vals = c[sorted_idx]
        weights = (peak_vals - c_max * 0.999 + 1e-15) ** 2
        w[sorted_idx] = weights / weights.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W * a)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            stall += 1
            lr *= 0.7
            if stall > 100:
                break
        else:
            lr = min(lr_init, lr * 1.005)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.004, 0.006, 0.008, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.08
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 17.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp_left":
        t = np.linspace(0, 1, n)
        a = 0.5 + t + 0.1 * np.sin(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(-1, 1, n)
        a = np.exp(-3.0 * t ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 6))
        amp = 0.04 * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _subgradient_polish(a_init, deadline, n_substeps=200):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    steps = min(n_substeps, max(10, int((deadline - time.monotonic()) * 5)))
    current_t = 1.0 / (steps + 10)

    for step in range(steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        peak_idx = int(np.argmax(c))

        a_rev = a[::-1]
        e_peak = np.zeros_like(c)
        e_peak[peak_idx] = 1.0
        wc = _full_convolve(e_peak, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-16:
            break

        alpha = current_t / (gnorm + 1e-15)
        a_try = a - alpha * grad_t
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        a = a_try
        current_t /= 1.005

    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.12f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n_parent}, score={best_score:.12f}")

    # Phase 1: Main adaptive L-BFGS
    print(f"\n[P1] Adaptive L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, time.monotonic() + 400)
    a_main = _run_lbfgs_adaptive(parent, p1_deadline, min_time_remaining=300.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.12f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement from L-BFGS. Score remains {best_score:.12f}")

    # Phase 2: Peak shaving
    print(f"\n[P2] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p2_deadline = min(deadline - 400, time.monotonic() + 120)
        a_ps, sc_ps = _peak_shave(best_seq, p2_deadline, n_peaks=15, lr_init=0.008)
        if sc_ps < best_score:
            best_seq = a_ps.copy()
            best_score = sc_ps
            print(f"  NEW BEST: {best_score:.12f} (n={len(best_seq)}) [peak_shave] t={time.monotonic()-t_start:.1f}s")

    # Phase 3: Active-set GD coarse
    print(f"\n[P3] Active-set GD coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, time.monotonic() + 150)
        a_ag, sc_ag = _active_set_gd_multi(best_seq, p3_deadline,
                                           tol_frac_start=0.008, tol_frac_end=0.0001,
                                           lr_init=0.035, top_k_frac=0.003)
        if sc_ag < best_score:
            best_seq = a_ag.copy()
            best_score = sc_ag
            print(f"  NEW BEST: {best_score:.12f} (n={len(best_seq)}) [active_coarse] t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Active-set GD fine
    print(f"\n[P4] Active-set GD fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, time.monotonic() + 120)
        a_ag2, sc_ag2 = _active_set_gd_multi(best_seq, p4_deadline,
                                             tol_frac_start=0.002, tol_frac_end=0.00003,
                                             lr_init=0.015, top_k_frac=0.001)
        if sc_ag2 < best_score:
            best_seq = a_ag2.copy()
            best_score = sc_ag2
            print(f"  NEW BEST: {best_score:.12f} (n={len(best_seq)}) [active_fine] t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Shrinkage
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 150)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.12f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 120:
                a_shr2 = _run_lbfgs_adaptive(best_seq, min(deadline - 100, time.monotonic() + 80),
                                             min_time_remaining=80.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.12f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Multi-n exploration
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_base = len(best_seq)
    n_candidates = sorted(set([n_base, n_base // 2, n_base * 2, 4096, 6144, 8192, 10240, 12288, 16384]))
    n_candidates = [x for x in n_candidates if 1024 <= x <= 16384]
    n_candidates.sort(key=lambda x: abs(x - n_base))

    n_tried = 0
    for n_try in n_candidates[:6]:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_base and n_tried > 0:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs_adaptive(a_init, min(deadline - 60, time.monotonic() + 60),
                                    min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.12f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        n_tried += 1

    # Phase 7: Diverse restarts
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("taper", _make_profile(n_best, "tapered")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 20:
                break
            a_opt = _run_lbfgs_adaptive(a_r, min(deadline - 15, time.monotonic() + 30),
                                        min_time_remaining=12.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.12f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # Phase 8: Final subgradient polish
    print(f"\n[P8] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_sub, sc_sub = _subgradient_polish(best_seq, min(deadline - 3, time.monotonic() + 30))
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"  NEW BEST: {best_score:.12f} [subgradient] t={time.monotonic()-t_start:.1f}s")

    # Phase 9: Ultra-fine active set
    print(f"\n[P9] Ultra-fine active set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_gd_multi(best_seq, deadline - 2,
                                             tol_frac_start=0.0003, tol_frac_end=0.000005,
                                             lr_init=0.004, top_k_frac=0.0005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.12f} [final_fine] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
