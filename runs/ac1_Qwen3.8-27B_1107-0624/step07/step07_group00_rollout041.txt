Uses smoothed gradient descent in log-space with adaptive temperature,

import numpy as np
import time
import sys

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, c, T):
    n = len(a)
    c_shift = (c - c.max()) / T
    e = np.exp(c_shift)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    grad = 2.0 * conv_full[n - 2:2 * n - 1][::-1]
    return grad

def _compute_active_gradient(a, c, top_k=8, threshold_frac=0.95):
    n = len(a)
    c_max = float(c.max())
    threshold = c_max * threshold_frac
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(c)[::-1][:top_k]
    if len(active) > top_k:
        active = active[:top_k]
    p = np.zeros_like(c)
    p[active] = 1.0 / len(active)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    grad = 2.0 * conv_full[n - 2:2 * n - 1][::-1]
    return grad

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0

        if use_active and frac > active_start_frac:

        z_new = z_new - z_new.mean()

        if use_backtrack and t % 30 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.05:
                z_new = z
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and lr_mult > 0.03:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_polish(a_init, deadline, max_iter=500):
    from scipy.optimize import minimize
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    def objective(z):
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        T = 0.001 * float(c.max())
        c_shift = (c - c.max()) / T
        e = np.exp(c_shift)
        s = e.sum()
        if s < 1e-300:
            return float(c.max())
        return float(T * np.log(s) + c.max())

    def gradient(z):
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        T = 0.001 * float(c.max())
        g = _compute_gradient(a, c, T)
        W = float(np.dot(a, g))
        return a * (g - W)

    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    try:
        result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'gtol': 1e-10,
                                  'maxfun': 2000, 'disp': False})
        a_opt = _softmax_stable(result.x)
        sc = _score_norm(a_opt)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
    except Exception:
        pass

    if time.monotonic() < deadline:
        try:
            result2 = minimize(objective, np.log(np.maximum(best_a, 1e-15)) - 0,
                             jac=gradient, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-12,
                                      'maxfun': 2000, 'disp': False})
            a_opt2 = _softmax_stable(result2.x)
            sc2 = _score_norm(a_opt2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_a = a_opt2.copy()
        except Exception:
            pass

    return best_a, _score_norm(best_a)

def _coord_descent_polish(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]

    improved_total = 0
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for k in active:
            if k < n:
                js = range(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = range(k - (n - 1), n)
            js_list = list(js)
            if len(js_list) == 0:
                continue
            for j in js_list:
                for scale in [1.1, 1.2, 0.85, 0.75, 0.6, 1.3, 0.5]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    if a_cand[j] < 0:
                        continue
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            if improved_total < 3:
                improved_total += 1
            else:
                break

    return best_a, _score_norm(best_a)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _generate_gaussian_init(n, center=0.5, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _generate_two_bump_init(n, frac=0.3):
    a = np.zeros(n)
    m = max(1, int(n * (1 - frac)))
    a[:m//4] = 1.0
    a[-m//4:] = 1.0
    a[m//4:m - m//4] = 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _generate_random_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, size=n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def propose_candidate(seed=42, budget_s=1800):

    best_a = None
    best_score = np.inf
    best_n = 0

    parent = None

        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            parent /= parent.sum()
            parent_score = _score_full(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
            if np.isfinite(parent_score) and parent_score < best_score:
                best_score = parent_score
                best_a = parent.copy()
                best_n = len(parent)
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Define initializations to try
    inits = []
    n_target = best_n if best_n > 0 else 8192

    if parent is not None:
        inits.append(('parent', parent))

    for n_try in [2048, 4096, 8192]:
        if n_try != n_target:
            if parent is not None:
                a_interp = _interp_to_length(parent, n_try)
                inits.append((f'parent_n{n_try}', a_interp))
            inits.append((f'gauss_n{n_try}', _generate_gaussian_init(n_try)))
            inits.append((f'random_n{n_try}', _generate_random_init(n_try, seed=n_try)))

    # Also try two-bump at target n
    inits.append((f'twobump_n{n_target}', _generate_two_bump_init(n_target)))

    print(f"Trying {len(inits)} initializations, deadline at t+{deadline-t_start:.0f}s")

    # Phase 1: Quick screen of all initializations (10s each)
    screen_time = min(10.0, (deadline - t_start) * 0.05)
    if screen_time > 3:
        for name, a_init in inits:
            if time.monotonic() > deadline - 60:
                break
            sc = _score_full(a_init)
            print(f"  Screen {name}: n={len(a_init)}, score={sc:.10f}")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_init.copy()
                best_n = len(a_init)

    # Phase 2: Adam optimization on best initialization (60s)
    if time.monotonic() < deadline - 60:
        a_work = best_a.copy()
        n_work = len(a_work)
        c_work = _fft_convolve(a_work, a_work)
        max_work = float(c_work.max())
        phase2_end = min(deadline - 50, t_start + 60.0)

                                   T_start=0.005 * max_work, T_end=0.0001 * max_work,

        n_target = best_n

        c_work = _fft_convolve(a_work, a_work)
