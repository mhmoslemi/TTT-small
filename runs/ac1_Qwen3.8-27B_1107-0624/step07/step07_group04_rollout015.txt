- p-norm smoothing (alternative to log-sum-exp) for finding different local minima
- Nesterov-accelerated subgradient with momentum
- Larger adaptive active set for CG-Newton polish
- Deflation steps to break out of plateaus
- Structured perturbations (sinusoidal) for restarts
- Better time management

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            # CG broke down (indefinite Hessian), use steepest descent step
            alpha_sd = rs / (float(p @ p) + 1e-30)
            x = x + alpha_sd * p
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _pnorm_obj(z, p, n):
    """p-norm smoothing objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(c ** p) ** (1.0 / p))


def _pnorm_grad(z, p, n):
    """Gradient of p-norm smoothing."""
    a = _softmax(z)
    a_rev = a[::-1]
    c = _full_convolve(a, a)
    cp = c ** (p - 1)
    S = float(np.sum(c ** p))
    if S < 1e-300:
        return np.zeros(n)
    factor = (1.0 / p) * S ** (1.0 / p - 1.0)
    hc = _full_convolve(cp, a_rev)
    h = hc[n - 1:2 * n - 1]
    return 2.0 * factor * h


def _lnorm_obj(z, n):
    """L1 norm of convolution as proxy (not great but fast)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(np.sum(np.abs(c)))


def _lse_obj(z, beta, n):
    """Log-sum-exp smoothing."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)


def _lse_grad(z, beta, n):
    """Gradient of log-sum-exp smoothing."""
    a = _softmax(z)
    a_rev = a[::-1]
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)

        if rs < tol * tol * rs0:
            break
    return x


def _pnorm_obj(z):
    p = _OPT['p']
    n = _OPT['n']
    a = _softmax(z)
    c = _full_convolve(a, a)
    s = float(np.sum(c ** p))
    if s < 1e-300:
        return 0.0
    return s ** (1.0 / p)


def _pnorm_grad(z):
    p = _OPT['p']
    n = _OPT['n']
    a = _softmax(z)
    a_rev = a[::-1]
    c = _full_convolve(a, a)
    cp = c ** (p - 1)
    S = float(np.sum(c ** p))
    if S < 1e-300:
        return np.zeros(n)
    factor = (1.0 / p) * S ** (1.0 / p - 1.0)
    hc = _full_convolve(cp, a_rev)
    h = hc[n - 1:2 * n - 1]
    return 2.0 * factor * h


def _lse_obj(z):
    beta = _OPT['beta']
    n = _OPT['n']
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)


def _lse_grad(z):
    beta = _OPT['beta']
    n = _OPT['n']
    a = _softmax(z)
    a_rev = a[::-1]
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _run_pnorm_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    _OPT['n'] = n
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    p_values = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    for p in p_values:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        _OPT['p'] = p
        scale = max(1.0, n / 4096.0)
        est = 0.006 * scale
        max_iter = min(2000, max(30, int(remaining / (est * 20))))
        try:
            res = sp_minimize(_pnorm_obj, z, jac=_pnorm_grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                       'maxcor': 50, 'maxls': 50})
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    pnorm p={p}: score={current_sc:.10f} nit={res.nit}")
        except Exception:
            pass
    return current_a


def _run_lse_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    _OPT['n'] = n
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    betas = base_beta * np.array(ratios)
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        _OPT['beta'] = float(beta)
        scale = max(1.0, n / 4096.0)
        est = 0.005 * scale
        max_iter = min(3000, max(20, int(remaining / (est * 15))))
        try:
            res = sp_minimize(_lse_obj, z, jac=_lse_grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                       'maxcor': 50, 'maxls': 100})
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i < 6 or i % 4 == 0:
                print(f"    lse beta={beta:.2e}: score={current_sc:.10f} nit={res.nit}")
        except Exception:
            pass
    return current_a


def _newton_active_polish(a_init, deadline, max_rounds=80):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    tol_frac = 0.005
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / (float(np.ones(n) @ H1) + 1e-30)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1.0 - 1e-10):
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        if not step_done:
            alpha = 0.005
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                if float(_full_convolve(a_try, a_try).max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5
        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
        tol_frac *= 0.95
    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, max_steps=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.03
    mu = 0.9
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        mu = 0.9 + 0.1 * (i / max_steps)
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        v = mu * v + (1.0 - mu) * (z_new - z)
        z = z_new + v
        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.06)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = 0.02
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_reopt(a, deadline, min_time_remaining=60.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.01, 0.03, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_r = _run_lse_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_r = _score(a_r)
            if sc_r < best_sc:
                best_sc = sc_r
                best_a = a_r
            a_r2 = _newton_active_polish(best_a, deadline)
            sc_r2 = _score(a_r2)
            if sc_r2 < best_sc:
                best_sc = sc_r2
                best_a = a_r2
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_center":
        a = np.ones(n)
        center = n // 2
        width = n // 4
        a[center - width:center + width] = 0.4
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_sinusoidal(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    k = rng.integers(1, max(2, n // 50))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.1)
    t = np.arange(n)
    z = z + amp * np.sin(2 * np.pi * k * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(8192, "tapered")
        print(f"No parent, using tapered n=8192, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init")

    nparent = len(parent)

    # Phase 1: p-norm L-BFGS
    print(f"\n[Phase 1] p-norm L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_pnorm_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "pnorm")

    # Phase 2: log-sum-exp L-BFGS (from p-norm result)
    print(f"\n[Phase 2] LSE L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 450:
        a2 = _run_lse_lbfgs(best_seq, deadline, min_time_remaining=400.0)
        update_best(a2, _score(a2), "lse")

    # Phase 3: CG-Newton active set
    print(f"\n[Phase 3] CG-Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a3, sc3 = _newton_active_polish(best_seq, deadline)
        update_best(a3, sc3, "newton1")

    # Phase 4: Nesterov subgradient
    print(f"\n[Phase 4] Nesterov subgrad (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a4, sc4 = _nesterov_subgrad(best_seq, deadline, max_steps=2000)
        update_best(a4, sc4, "nesterov1")

    # Phase 5: Another round of p-norm from improved point
    print(f"\n[Phase 5] p-norm round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        # Slightly perturb to escape potential plateau
        a5_init = _perturb(best_seq, 0.01, rng)
        a5 = _run_pnorm_lbfgs(a5_init, deadline, min_time_remaining=100.0)
        update_best(a5, _score(a5), "pnorm2")

    # Phase 6: CG-Newton again
    print(f"\n[Phase 6] CG-Newton 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a6, sc6 = _newton_active_polish(best_seq, deadline, max_rounds=100)
        update_best(a6, sc6, "newton2")

    # Phase 7: Shrinkage
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a7, sc7 = _shrinkage_reopt(best_seq, deadline, min_time_remaining=60.0)
        update_best(a7, sc7, "shrink")

    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == nparent:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lse_lbfgs(a_init, deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.001 and time.monotonic() < deadline - 40:
            a_np, sc_np = _newton_active_polish(a_opt, deadline, max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 9: Diverse restarts
    print(f"\n[Phase 9] Restarts (t={time.monotonic()-t_start:.1f}s)")
    for i in range(5):
        if time.monotonic() > deadline - 20:
            break
        if i == 0:
            p_start = _perturb(best_seq, 0.05, rng)
        elif i == 1:
            p_start = _perturb_sinusoidal(best_seq, rng)
        elif i == 2:
            p_start = _perturb_sinusoidal(best_seq, rng)
        elif i == 3:
            p_start = _make_profile(nparent, "gaussian")
        else:
            p_start = _make_profile(nparent, "bimodal")
        if time.monotonic() > deadline - 15:
            break
        a_r = _run_lse_lbfgs(p_start, deadline, min_time_remaining=10.0)
        sc_r = _score(a_r)
        update_best(a_r, sc_r, f"restart_{i}")
        if sc_r < best_score + 0.0005 and time.monotonic() < deadline - 10:
            a_rp, sc_rp = _newton_active_polish(a_r, deadline, max_rounds=40)
            update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 10: Final ultra-fine polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=150)
        update_best(a_fin, sc_final if False else sc_fin, "final_newton")
        # One more LSE pass with very small beta
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            _OPT['n'] = n
            _OPT['beta'] = mb * 1e-6
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(500, max(20, int(remaining / 0.02)))
                try:
                    res = sp_minimize(_lse_obj, z, jac=_lse_grad, method='L-BFGS-B',
                                      options={'maxiter': max_iter, 'ftol': 1e-18,
                                               'gtol': 1e-14, 'maxcor': 50})
                    a_flb = _softmax(res.x)
                    update_best(a_flb, _score(a_flb), "final_lse")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
