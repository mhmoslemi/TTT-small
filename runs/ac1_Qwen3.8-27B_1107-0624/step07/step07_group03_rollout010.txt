"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-scale log-sum-exp smoothing, tight active-set Newton with peeling,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton_v2(a_init, deadline, max_iter=500, tol_active=0.0005, tol_converge=1e-14):
    """Improved active-set Newton with peeling and tighter convergence."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            n_active = 3 * n
            active_new = np.zeros_like(c, dtype=bool)

            active_new[top_idx] = True
            active = active_new

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of f(a) = (1/m) * sum_{k in active} c[k]
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)

            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))

        else:
            lam = float(u.sum()) / sum_w

        t_max = max(t_max, 1e-12)

        f_old = float(c.max())
        accepted = False

            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-14 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < tol_converge:

    return best_a, best_sc


def _run_lbfgs_v2(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation, more aggressive schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.2, 0.1, 0.05, 0.02,
                                   0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, max_rounds=200):
    """Coordinate descent: try updating small subsets of coordinates."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Generate coordinate groups
    group_size = max(4, n // 512)
    n_groups = n // group_size

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick a random group of coordinates to update
        start = (rnd * group_size * 7) % (n - group_size)
        indices = np.arange(start, start + group_size)

        a_sub = a[indices].copy()
        # Try to optimize these coordinates while keeping others fixed
        # Use a few gradient steps on the subproblem
        for inner in range(10):
            c = _full_convolve(a, a)
            c_max = float(c.max())
            active = c >= c_max - 0.001 * c_max
            A_vec = active.astype(np.float64)
            m = max(int(active.sum()), 1)

            # Gradient w.r.t. a[indices]
            g_full = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_sub = g_full[indices] - np.mean(g_full[indices])

            # Newton step on the subproblem
            a_try = a.copy()
            a_try[indices] = a_sub - 0.1 * g_sub
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                a = a_try
                a_sub = a[indices].copy()
                best_sc = sc_try
                best_a = a.copy()
            else:
                a_sub = a[indices].copy()
                # Reset
                a = a.copy()

    return best_a, best_sc


def _peeling_refine(a_init, deadline, tol_active=0.0003):

    active_idx = np.where(active)[0]

    if n_active < 2:
        return best_a, best_sc

    for i in range(n_active):

        active_mod = active.copy()

        if int(active_mod.sum()) == 0:

        A_vec = active_mod.astype(np.float64)

                rs_w = rs_w_new

            sum_w = float(w.sum())
            if abs(sum_w) < 1e-15:

                dx = dx - np.mean(dx)
            else:
                lam = float(x.sum()) / sum_w
                dx = -x + lam * w

            neg_mask = dx < -1e-15
            if neg_mask.any():
                t_max = np.min(a_try[neg_mask] / (-dx[neg_mask])) * 0.99

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Subgradient method with active-set tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-scale log-sum-exp smoothing, tight active-set Newton with peeling,
coordinate descent, and n-value exploration.
Key improvements: tighter active set identification, peeling strategy for constraint
removal, more aggressive beta schedule, better time allocation, and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lsobj_wrapper(z, beta, n):
    return _lsobj(z, beta, n)


def _lsgrad_wrapper(z, beta, n):
    return _lsgrad(z, beta, n)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _cg_solve(hess_matvec, rhs, max_cg=300):
    x = np.zeros_like(rhs)
    r = rhs - hess_matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    for cg_it in range(max_cg):
        Ap = hess_matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x


def _make_hess_matvec(A_vec, m, mu):
    def hess_matvec(v):
        return (2.0 / m) * _corr_a_A(v, A_vec, len(A_vec) - (len(v)) + 1 - (len(A_vec) - len(v))) + mu * v
    return hess_matvec

            break

        active_mod = active.copy()
        active_mod[active_idx[i]] = False
        if int(active_mod.sum()) == 0:
            continue

        A_vec = active_mod.astype(np.float64)
        m = int(A_vec.sum())
        mu = 1e-10

        def hess_vp_peel(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        a_try = a.copy()
        for _ in range(25):
            if time.monotonic() > deadline:
                break
            g = (2.0 / m) * _corr_a_A(a_try, A_vec, n)
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-15:
                break

            x = _cg_solve(hess_vp_peel, g_t, max_cg=150)
            w = _cg_solve(hess_vp_peel, np.ones(n), max_cg=150)
            sum_w = float(w.sum())
            if abs(sum_w) < 1e-15:
                dx = -x
                dx = dx - np.mean(dx)
            else:
                lam = float(x.sum()) / sum_w
                dx = -x + lam * w
                dx = dx - np.mean(dx)

            neg_mask = dx < -1e-15
            if neg_mask.any():
                t_max = np.min(a_try[neg_mask] / (-dx[neg_mask])) * 0.99
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-12)

            t = min(1.0, t_max)
            f_old = float(_full_convolve(a_try, a_try).max())
            accepted = False
            for ls in range(30):
                a_ls = a_try + t * dx
                a_ls = np.maximum(a_ls, 0)
                s_ls = a_ls.sum()
                if s_ls < 1e-15:
                    t *= 0.5
                    continue
                a_ls = a_ls / s_ls
                f_ls = float(_full_convolve(a_ls, a_ls).max())
                if f_ls < f_old - 1e-14 or ls == 29:
                    if f_ls <= f_old + 1e-12:
                        a_try = a_ls
                        accepted = True
                        break
                t *= 0.5
            if not accepted:
                break

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    group_size = max(8, n // 256)
    offset = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        offset = (offset + group_size * 3) % (n - group_size)
        indices = np.arange(offset, offset + group_size)

        for inner in range(8):
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            active = c >= c_max - 0.002 * c_max
            A_vec = active.astype(np.float64)
            m = max(int(active.sum()), 1)

            g_full = (2.0 / m) * _corr_a_A(a, A_vec, n)
            g_sub = g_full[indices] - np.mean(g_full[indices])

            for step_scale in [0.1, 0.05, 0.02, 0.01]:
                a_try = a.copy()
                a_try[indices] = a[indices] - step_scale * g_sub
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer with peeling refinement.
    Phases: L-BFGS -> Active-Set Newton -> Peeling -> Subgrad -> CoordDescent -> Multi-n -> Restarts -> Final
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={parent_sc:.10f}")
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS from parent
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs_v2(parent, deadline, min_time_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Tight active-set Newton (multiple rounds)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_deadline = min(deadline - 500, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton_v2(best_seq, as_deadline, max_iter=400, tol_active=0.0005)
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 440:
            as_deadline2 = min(deadline - 440, time.monotonic() + 120)
            a_as2, sc_as2 = _active_set_newton_v2(best_seq, as_deadline2, max_iter=300, tol_active=0.0002)
            update_best(a_as2, sc_as2, "as_newton_2")

        if time.monotonic() < deadline - 400:
            as_deadline3 = min(deadline - 400, time.monotonic() + 80)
            a_as3, sc_as3 = _active_set_newton_v2(best_seq, as_deadline3, max_iter=200, tol_active=0.0001)
            update_best(a_as3, sc_as3, "as_newton_3")

    # Phase 3: Peeling refinement
    print(f"\n[Phase 3] Peeling (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 380:
        peel_deadline = min(deadline - 380, time.monotonic() + 150)
        a_peel, sc_peel = _peeling_refine(best_seq, peel_deadline, tol_active=0.0003)
        update_best(a_peel, sc_peel, "peeling_1")

        if time.monotonic() < deadline - 340:
            peel_deadline2 = min(deadline - 340, time.monotonic() + 80)
            a_peel2, sc_peel2 = _peeling_refine(best_seq, peel_deadline2, tol_active=0.0001)
            update_best(a_peel2, sc_peel2, "peeling_2")

    # Phase 4: Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 80)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=4000, lr_init=0.012)
        update_best(a_sg, sc_sg, "sg_polish")

    # Phase 5: Coordinate descent
    print(f"\n[Phase 5] Coord descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 260:
        cd_deadline = min(deadline - 260, time.monotonic() + 60)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_deadline, max_rounds=400)
        update_best(a_cd, sc_cd, "coord_descent")

    # Phase 6: Multi-n search
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 12288, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent:
            continue

        if parent is not None and n_parent != n_try:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        a_opt = _run_lbfgs_v2(a_init, deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_newton_v2(a_opt, min(deadline - 20, time.monotonic() + 40),
                                                      max_iter=200, tol_active=0.0005)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 25:
        restarts_configs = [
            ("perturb_small", 0.02),
            ("perturb_mid", 0.05),
            ("perturb_large", 0.10),
        ]
        for label, sigma in restarts_configs:
            if time.monotonic() > deadline - 12:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs_v2(p_start, deadline, min_time_remaining=8.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 8:
                a_r2, sc_r2 = _active_set_newton_v2(a_r, min(deadline - 4, time.monotonic() + 20),
                                                      max_iter=100, tol_active=0.0003)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

        if time.monotonic() < deadline - 8:
            p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            a_r = _run_lbfgs_v2(p_start, deadline, min_time_remaining=4.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, "restart_fourier")

    # Phase 8: Final ultra-tight refinement
    print(f"\n[Phase 8] Final (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        as_final = min(deadline - 1, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_newton_v2(best_seq, as_final, max_iter=400, tol_active=0.0001)
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _peeling_refine(best_seq, deadline - 1, tol_active=0.0001)
            update_best(a_fin2, sc_fin2, "final_peel")

        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _active_set_newton_v2(best_seq, deadline - 1, max_iter=200, tol_active=0.00005)
            update_best(a_fin3, sc_fin3, "final_as_ultra")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
