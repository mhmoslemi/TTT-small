"""
Flat-Convolution Optimizer using Multiplicative Weights, Power Smoothing,
and Aggressive Peeling with KKT-based support identification.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _convolve(a):
    return _full_convolve(a, a)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = eta_init
    patience = 0
    max_patience = 2000
    
    while time.monotonic() < deadline:
        c = _convolve(a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:

        if w_sum < 1e-300:

    """Identify the support using KKT conditions.
    
    At optimum with sum(a)=1, a>=0:
    - For i with a_i > 0: (grad)_i = lambda (constant)
    - For i with a_i = 0: (grad)_i >= lambda

    c_max = float(c.max())

    w = np.exp(np.clip(shifted, -700, 0))

    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]

    # Coordinates where a_i is very small AND h_i > mean(h) should be zero
    active_coords = a > 1e-10

        return np.ones(n, dtype=bool)

    h_std = float(np.std(h_active))

    support = np.ones(n, dtype=bool)
    for i in range(n):
        if a[i] < 1e-8 and h[i] > h_mean + 2.0 * h_std:
            support[i] = False

def _aggressive_peel(a, deadline):
    """Iteratively peel coordinates and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Sort coordinates by value
    order = np.argsort(a)
    
    # Try different peeling fractions
    fractions = [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15, 0.20]
    a_current = a.copy()
    
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a_current.copy()
        # Zero out the smallest coordinates
        a_try[order[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a_current = a_try
            # Re-sort
            order = np.argsort(a_current)
    
    return best_a, best_sc

        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        
        # Projected gradient (maintain sum=1 and a>=0)
        grad_t = h - np.mean(h)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-16))
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _convolve(a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < _convolve(a).max() * 2.0 * n - 1e-14:
                # Better: compare to current_sc
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.8
            tol_frac = max(0.00001, tol_frac * 0.9)
            if lr < 1e-8:
                break
        else:
            if tol_frac > 0.00001:
                tol_frac = max(0.00001, tol_frac * 0.98)
            lr = min(0.02, lr * 1.01)
    
    return best_a, best_sc

# Profile generators
def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_sides":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = n // 8
        a[:taper] = np.sin(np.pi * np.arange(taper) / (2*taper)) ** 2
        a[-taper:] = np.sin(np.pi * np.arange(taper) / (2*taper)) ** 2[::-1]
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.02
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7*t
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _convolve(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _mwua_optimize(a_init, deadline, eta_init=0.05, eta_min=0.0001, eta_decay=0.9995):
    """Multiplicative Weights Update Algorithm for non-negative optimization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_convolve(a).max())

    eta = eta_init
    no_improve = 0
    max_no_improve = 3000

    while time.monotonic() < deadline:
        c = _convolve(a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break
        if eta < eta_min:
            break

        # Soft argmax weighting for robustness
        beta = c_max * 0.08
        shifted = (c - c_max) / beta
        w = np.exp(np.clip(shifted, -700, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Subgradient: h_i = sum_k w_k * a_{k-i}
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        # Multiplicative update in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * h
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-300 or not np.isfinite(s_new):
            eta *= 0.5
            continue
        a = a_new / s_new
        eta *= eta_decay

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_remaining=30.0):
    """L-BFGS on log-sum-exp smoothed objective with adaptive beta schedule."""
    n = len(a_init)
    c = _convolve(a_init)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a_init

    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.004)

    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.1,
                                  0.06, 0.04, 0.025, 0.015, 0.01,
                                  0.006, 0.004, 0.002, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        def _obj(z):
            a = _softmax(z)
            c = _convolve(a)
            c_m = float(c.max())
            shifted = (c - c_m) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_m
            return 2.0 * n * (c_m + beta * np.log(s))

        def _grad(z):
            a = _softmax(z)
            c = _convolve(a)
            c_m = float(c.max())
            shifted = (c - c_m) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)

        est_time = 0.002 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 8))))

        try:
            result = minimize(_obj, z, jac=_grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-18,
                                      'gtol': 1e-15, 'maxcor': 100, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _fista_smooth(a_init, deadline, beta=None):
    """FISTA (accelerated proximal gradient) on smoothed objective."""
    n = len(a_init)
    c = _convolve(a_init)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a_init
    if beta is None:
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
        beta = max(gap, c_max * 0.02)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()
    t_k = 1.0
    best_z = z.copy()
    best_val = _lbfgs_obj_val(z, beta, n)

    # Lipschitz constant estimate
    L = 4.0 * n * c_max * 4.0 * len(c)  # rough upper bound
    step = 1.0 / max(L, 1e-10)

    for it in range(5000):
        if time.monotonic() > deadline:
            break
        # Gradient step
        g = _lbfgs_grad_val(y, beta, n)
        z_new = y - step * g
        # Soft projection (softmax handles this implicitly)
        # Backtracking
        f_new = _lbfgs_obj_val(z_new, beta, n)
        while f_new > _lbfgs_obj_val(y, beta, n) + 0.5 * step * np.dot(g, z_new - y):
            step *= 0.5
            z_new = y - step * g
            f_new = _lbfgs_obj_val(z_new, beta, n)
            if step < 1e-20:
                break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z_new + ((t_k - 1.0) / t_new) * (z_new - z)
        z, t_k = z_new, t_new

        if f_new < best_val - 1e-15:
            best_val = f_new
            best_z = z_new.copy()

    a_best = _softmax(best_z)
    return a_best


def _lbfgs_obj_val(z, beta, n):

    c = _convolve(a)
    c_m = float(c.max())
    shifted = (c - c_m) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_m
    return 2.0 * n * (c_m + beta * np.log(s))


def _lbfgs_grad_val(z, beta, n):

    c = _convolve(a)
    c_m = float(c.max())
    shifted = (c - c_m) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _ultra_polish(a_init, deadline):
    """Ultra-fine active-set polish with geometric line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Multiple passes with decreasing tolerance
    tol_fracs = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for tol_frac in tol_fracs:
        if time.monotonic() > deadline:
            break

        lr = 0.03
        no_improve = 0
        max_stall = 300

        while time.monotonic() < deadline and no_improve < max_stall:
            c = _convolve(a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max

            if current_sc < best_sc - 1e-15:
                best_sc = current_sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Active set
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active == 0:
                idx = int(np.argmax(c))
                active = np.zeros_like(c, dtype=bool)
                active[idx] = True
                n_active = 1
            if n_active > 50:
                # Only keep top 50
                thresh = np.sort(c)[::-1][min(49, len(c) - 1)]
                active = c >= thresh
                n_active = int(active.sum())

            w = np.zeros_like(c)
            w[active] = 1.0 / n_active
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]

            grad_t = h - np.mean(h)
            gnorm = np.linalg.norm(grad_t)
            if gnorm < 1e-16:
                break

            alpha = min(lr, 0.8 / (gnorm + 1e-16))
            improved = False
            for _ in range(100):
                a_try = a - alpha * grad_t
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = 2.0 * n * float(_convolve(a_try).max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                lr *= 0.7
                if lr < 1e-12:
                    break
            else:
                lr = min(0.05, lr * 1.01)

    return best_a, best_sc


def _kkt_peel(a, deadline):
    """KKT-based support identification and peeling."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    # Use tight log-sum-exp to get gradient
    c = _convolve(a)
    c_max = float(c.max())
    beta = c_max * 0.005
    shifted = (c - c_max) / beta
    w = np.exp(np.clip(shifted, -700, 0))
    w = w / w.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

    # KKT: at optimum, h_i should be ~constant for all i with a_i > 0
    # Coordinates with very small a_i and h_i >> mean should be zero
    active_mask = a > 1e-10
    n_active = int(active_mask.sum())
    if n_active < 2:
        return a, _score(a)

    h_active = h[active_mask]
    h_mean = float(np.mean(h_active))
    h_std = float(np.std(h_active))
    if h_std < 1e-15:
        return a, _score(a)

    # Identify candidates for removal: small a_i, large h_i
    z_scores = (h - h_mean) / (h_std + 1e-15)
    candidates = (a < np.percentile(a, 5)) & (z_scores > 2.0)
    n_cand = int(candidates.sum())
    if n_cand == 0 or n_cand > n * 0.3:
        return a, _score(a)

    # Try removing candidates
    a_try = a.copy()
    a_try[candidates] = 0.0
    s_try = a_try.sum()
    if s_try < 1e-10:
        return a, _score(a)
    a_try = a_try / s_try
    sc_try = _score(a_try)

    if sc_try < best_sc - 1e-12:
        best_a = a_try
        best_sc = sc_try

    return best_a, best_sc


def _aggressive_peel(a, deadline):
    """Iterative magnitude-based peeling with re-optimization potential."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_current = a.copy()
    order = np.argsort(a_current)

    fractions = [0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.04, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        if n_zero >= n:
            break
        a_try = a_current.copy()
        a_try[order[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a_current = a_try
            order = np.argsort(a_current)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_sides":
        t = np.ones(n)
        taper = max(2, n // 10)
        ramp = np.sin(np.pi * np.arange(taper) / (2.0 * taper)) ** 2
        t[:taper] = ramp
        t[-taper:] = ramp[::-1]
        return t / t.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-25 * (t - 0.3) ** 2) + np.exp(-25 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "broad_taper":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.4 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 10))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer:
    P1: MWUA coarse optimization
    P2: L-BFGS log-sum-exp smoothing
    P3: FISTA acceleration
    P4: KKT + magnitude peeling
    P5: Ultra-fine active-set polish
    P6: Multi-n exploration
    P7: Diverse restarts with re-optimization
    P8: Final polish sweep
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 18.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === P1: MWUA ===
    print(f"\n[P1] MWUA (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200, time.monotonic() + 250)
    a1, sc1 = _mwua_optimize(best_seq, p1_deadline, eta_init=0.04, eta_decay=0.9997)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [mwua_coarse] t={time.monotonic()-t_start:.1f}s")

    # === P2: L-BFGS main ===
    print(f"\n[P2] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800, time.monotonic() + 450)
    a2 = _lbfgs_smooth(best_seq, p2_deadline, min_remaining=300.0)
    sc2 = _score(a2)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === P3: FISTA ===
    print(f"\n[P3] FISTA (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        p3_deadline = min(deadline - 600, time.monotonic() + 120)
        a3 = _fista_smooth(best_seq, p3_deadline, beta=_convolve(best_seq).max() * 0.01)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [fista] t={time.monotonic()-t_start:.1f}s")

    # === P4: Peeling (magnitude + KKT) ===
    print(f"\n[P4] Peeling (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        p4_deadline = min(deadline - 500, time.monotonic() + 100)
        a4, sc4 = _aggressive_peel(best_seq, p4_deadline)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [mag_peel] t={time.monotonic()-t_start:.1f}s")

        # KKT peeling
        if time.monotonic() < deadline - 450:
            a4b, sc4b = _kkt_peel(best_seq, min(deadline - 450, time.monotonic() + 30))
            if sc4b < best_score:
                best_seq = a4b.copy()
                best_score = sc4b
                print(f"  NEW BEST: {best_score:.10f} [kkt_peel] t={time.monotonic()-t_start:.1f}s")

        # Re-optimize after peeling
        if best_score < np.inf and time.monotonic() < deadline - 400:
            a4c = _lbfgs_smooth(best_seq, min(deadline - 400, time.monotonic() + 120), min_remaining=80.0)
            sc4c = _score(a4c)
            if sc4c < best_score:
                best_seq = a4c.copy()
                best_score = sc4c
                print(f"  NEW BEST: {best_score:.10f} [reopt_peel] t={time.monotonic()-t_start:.1f}s")

    # === P5: Ultra-fine polish ===
    print(f"\n[P5] Ultra-fine polish (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 350, time.monotonic() + 200)
    a5, sc5 = _ultra_polish(best_seq, p5_deadline)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} [ultra_polish] t={time.monotonic()-t_start:.1f}s")

    # === P6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([4096, 6144, 8192, 10240, 12288, 14336, 16384]))
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 60:
            continue
        a_init = _resample(best_seq, n_try)
        # Quick optimization
        p6_deadline = min(deadline - 100, time.monotonic() + 60)
        a6 = _lbfgs_smooth(a_init, p6_deadline, min_remaining=20.0)
        sc6 = _score(a6)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
            # Quick polish on new best
            if time.monotonic() < deadline - 80:
                a6p, sc6p = _ultra_polish(best_seq, min(deadline - 70, time.monotonic() + 30))
                if sc6p < best_score:
                    best_seq = a6p.copy()
                    best_score = sc6p
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === P7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restarts = [
            ("pert3", _perturb(best_seq, 0.03, rng)),
            ("pert6", _perturb(best_seq, 0.06, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("flat", _make_profile(n_best, "flat_sides")),
            ("broad", _make_profile(n_best, "broad_taper")),
            ("peri", _make_profile(n_best, "periodic")),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 20:
                break
            p7_deadline = min(deadline - 15, time.monotonic() + 25)
            a7 = _lbfgs_smooth(a_r, p7_deadline, min_remaining=8.0)
            sc7 = _score(a7)
            if sc7 < best_score:
                best_seq = a7.copy()
                best_score = sc7
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")
            elif sc7 < best_score + 0.001 and time.monotonic() < deadline - 15:
                a7p, sc7p = _ultra_polish(a7, min(deadline - 12, time.monotonic() + 12))
                if sc7p < best_score:
                    best_seq = a7p.copy()
                    best_score = sc7p
                    print(f"  NEW BEST: {best_score:.10f} [restart_{label}_polish] t={time.monotonic()-t_start:.1f}s")

    # === P8: Final polish sweep ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        # Ultra tight polish
        a8, sc8 = _ultra_polish(best_seq, deadline - 2)
        if sc8 < best_score:
            best_seq = a8.copy()
            best_score = sc8
            print(f"  NEW BEST: {best_score:.10f} [final_ultra] t={time.monotonic()-t_start:.1f}s")

        # Tiny MWUA
        remaining = deadline - 2 - time.monotonic()
        if remaining > 8:
            a8b, sc8b = _mwua_optimize(best_seq, deadline - 1, eta_init=0.005, eta_decay=0.9999)
            if sc8b < best_score:
                best_seq = a8b.copy()
                best_score = sc8b
                print(f"  NEW BEST: {best_score:.10f} [final_mwua] t={time.monotonic()-t_start:.1f}s")

        # One more L-BFGS with tiny beta
        remaining = deadline - 1 - time.monotonic()
        if remaining > 10:
            c_best = _convolve(best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 0.00002

            def _obj_fin(z):
                a = _softmax(z)
                c = _convolve(a)
                c_m = float(c.max())
                shifted = (c - c_m) / beta_fin
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300:
                    return 2.0 * n * c_m
                return 2.0 * n * (c_m + beta_fin * np.log(s))

            def _grad_fin(z):
                a = _softmax(z)
                c = _convolve(a)
                c_m = float(c.max())
                shifted = (c - c_m) / beta_fin
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300:
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s
                a_rev = a[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W = float(np.dot(w, c))
                return 4.0 * n * a * (h - W)

            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(_obj_fin, z, jac=_grad_fin, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-16, 'maxcor': 100})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs_tiny] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Scale so sum is reasonable (not required but safe)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
