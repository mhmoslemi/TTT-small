KKT:

def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

def _softmax(z):

def _autocorr(a):
    """Compute linear autocorrelation R[m] = sum_i a_i * a_{i+m} for m in [-(n-1), n-1]."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    power = fa * np.conj(fa)
    r = np.fft.irfft(power, N)

    out = np.empty(2 * n - 1)
    out[:n] = r[:n]
    out[n:] = r[N - 1: N - (n - 1): -1]
    return out

        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Backtracking step in z-space
        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 50 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.1)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.03
            z = z_new
        else:
            z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs_extended(a_init, deadline, min_time_remaining=60.0, n_beta_steps=35):
    """
    L-BFGS with very aggressive beta continuation.
    Uses more beta steps and tighter tolerances than standard.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Compute initial beta from the gap between top values
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.02)

    # Very aggressive schedule: many steps, geometric decrease
    # Start at base_beta, decrease by factor ~1.5 each step
    ratio = (max_c * 1e-9 / base_beta) ** (1.0 / max(1, n_beta_steps - 1))
    ratio = max(0.5, min(0.85, ratio))  # Clamp ratio
    betas = [base_beta * (ratio ** i) for i in range(n_beta_steps)]
    # Ensure the last beta is very small
    if betas[-1] > max_c * 1e-8:
        betas[-1] = max_c * 1e-8
    if betas[-2] > max_c * 1e-6:
        betas[-2] = max_c * 1e-6

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations: more iterations at small beta
        if i < n_beta_steps // 3:
            # Early: fewer iterations (beta is large, problem is smooth)
            est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
            max_iter = min(2000, max(30, int(remaining / 8 / (est_time_per_iter * 10))))
        elif i < 2 * n_beta_steps // 3:
            # Middle: moderate iterations
            est_time_per_iter = 0.005 * max(1.0, n / 4096.0)
            max_iter = min(4000, max(50, int(remaining / 6 / (est_time_per_iter * 10))))
        else:
            # Late: many iterations (beta is small, need to converge)
            est_time_per_iter = 0.006 * max(1.0, n / 4096.0)
            max_iter = min(8000, max(100, int(remaining / 4 / (est_time_per_iter * 8))))

        gtol = 1e-10 if i > n_beta_steps // 2 else 1e-8

        try:
            result = minimize(
                _smoobj_wrapper,
                z,
                args=(beta, n),
                jac=_smograd_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': gtol,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i >= n_beta_steps // 2:
                # Even if score didn't improve, keep the new z for next beta
                z = np.log(np.maximum(_softmax(result.x), 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _smoobj_wrapper(z, beta, n):
    return _smoobj(z, beta, n)


def _smograd_wrapper(z, beta, n):
    return _smograd(z, beta, n)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: aggressive L-BFGS + flat-top equalization +
    subgradient + multi-n + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Main L-BFGS from parent (most aggressive) ===
    print(f"\n[Phase 1] Main L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs_extended(parent, deadline, min_time_remaining=400.0, n_beta_steps=35)
    update_best(a1, _score(a1), "lbfgs_main")

    # === Phase 2: Flat-top equalization ===
    print(f"\n[Phase 2] Flat-top equalize (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a2, sc2 = _flat_top_equalize(best_seq, min(deadline - 250, time.monotonic() + 200),
                                     max_iter=150, tol_active=0.002)
        update_best(a2, sc2, "flat_top_1")

    # === Phase 3: Another L-BFGS round from improved point ===
    print(f"\n[Phase 3] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a3 = _run_lbfgs_extended(best_seq, deadline, min_time_remaining=180.0, n_beta_steps=25)
        update_best(a3, _score(a3), "lbfgs_2")

    # === Phase 4: Flat-top equalization again ===
    print(f"\n[Phase 4] Flat-top equalize 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a4, sc4 = _flat_top_equalize(best_seq, min(deadline - 150, time.monotonic() + 100),
                                     max_iter=200, tol_active=0.001)
        update_best(a4, sc4, "flat_top_2")

    # === Phase 5: Subgradient polish ===
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a5, sc5 = _subgrad_polish(best_seq, min(deadline - 120, time.monotonic() + 80), max_steps=2000)
        update_best(a5, sc5, "subgrad_1")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 12288, 16384, 6144]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent and abs(best_score - _score(parent)) < 1e-10:
            continue  # Already optimized at this n
        if time.monotonic() > deadline - 60:
            break

        a_init = _resample(parent, n_try)
        # Quick L-BFGS
        if time.monotonic() < deadline - 40:
            a_opt = _run_lbfgs_extended(a_init, deadline, min_time_remaining=35.0, n_beta_steps=20)
            update_best(a_opt, _score(a_opt), f"n_{n_try}")
            if _score(a_opt) < best_score + 0.003 and time.monotonic() < deadline - 20:
                a_fp, sc_fp = _flat_top_equalize(a_opt, min(deadline - 18, time.monotonic() + 30),
                                                 max_iter=80, tol_active=0.002)
                update_best(a_fp, sc_fp, f"n_{n_try}_ft")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            (0.02, "perturb_small"),
            (0.08, "perturb_med"),
            (0.0, "shrink_perturb"),
        ]
        for i, (sigma, label) in enumerate(restart_configs):
            if time.monotonic() > deadline - 20:
                break
            if label == "shrink_perturb":
                n_shrink = len(best_seq)
                p_shrunk = best_seq.copy()
                n_zero = n_shrink // 8
                p_shrunk[np.argsort(p_shrunk)[:n_zero]] = 0
                p_shrunk = p_shrunk / max(p_shrunk.sum(), 1e-15)
                p_start = _perturb(p_shrunk, 0.04, rng)
            else:
                p_start = _perturb(best_seq, sigma, rng)

            if time.monotonic() > deadline - 15:
                break
            a_r = _run_lbfgs_extended(p_start, deadline, min_time_remaining=12.0, n_beta_steps=15)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Flat-top with very tight tolerance
        a_fin, sc_fin = _flat_top_equalize(best_seq, deadline - 2, max_iter=100, tol_active=0.0005)
        update_best(a_fin, sc_fin, "final_flat")

        # One more quick L-BFGS at tiny beta
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 1:
                max_iter = min(5000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _smoobj_wrapper,
                        z,
                        args=(beta_fin, n),
                        jac=_smograd_wrapper,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                                 'maxcor': 60, 'maxls': 200}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception as e:
                    print(f"  Final L-BFGS error: {e}")

        # One last flat-top
        if time.monotonic() < deadline - 1 and best_seq is not None:
            a_last, sc_last = _flat_top_equalize(best_seq, deadline - 0.5, max_iter=50, tol_active=0.0003)
            update_best(a_last, sc_last, "final_ft2")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        if sc_final < best_score:
            best_score = sc_final
        print(f"FINAL: score={best_score:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
