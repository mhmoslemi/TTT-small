import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

def _soft_pressure_equalize(a_init, deadline, n_iters=300, gamma=0.4, beta0=0.02, beta_decay=0.97):
    """
    Soft active-set pressure equalization using Gaussian weights.
    At KKT optimum: sum_k w_k * a_{k-i} = const for all i (with a_i > 0).
    Uses multiplicative update: a_i <- a_i * (p_mean / p_i)^gamma.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score_norm(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)
    beta = beta0
    momentum = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Soft weights: Gaussian kernel centered at max
        shifted = (c - c_max) / beta
        w = np.exp(np.clip(shifted, -500, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Compute pressure: p_i = sum_k w_k * a_{k-i} (cross-correlation)
        a_rev = a[::-1]
        p = _full_convolve(w, a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Multiplicative update with momentum
        p_mean = np.mean(p)
        log_ratio = np.log(p_mean / p)
        momentum = 0.7 * momentum + 0.3 * log_ratio
        update = np.exp(np.clip(gamma * momentum, -3.0, 3.0))
        a = a * update
        a = np.maximum(a, 1e-30)
        s = a.sum()
        a = a / s

            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find coordinates that contribute most to peaks
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]
        try_order = top_idx[rng.permutation(len(top_idx))[:len(top_idx)//4]]

        improved = False
        for i in try_order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            c_old = c_max  # reuse

            for factor in [0.6, 0.8, 1.0, 1.25, 1.5]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-14:
                    improved = True
                    c_old = c_try
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 4 * t) + 0.15 * np.cos(2 * np.pi * 11 * t + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_amp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.35) ** 2) + 0.2 * np.exp(-4.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "wavelet":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.sin(6 * np.pi * t) * np.exp(-6 * (t - 0.3) ** 2)
        a = a + 0.15 * np.sin(10 * np.pi * t) * np.exp(-6 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_init(n, rng, n_modes=6):
    """Generate initial sequence as sum of random Fourier modes (abs value)."""
    z = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        z += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.abs(z) + np.exp(rng.uniform(0.5, 2.0))
    return a / a.sum()


def _fourier_perturb(a, rng, n_modes=4, amplitude=0.03):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 3)))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _multi_stage_optimize(a_init, deadline, min_remaining=50.0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    s = current_a.sum()
    if s < 1e-15:
        return current_a, np.inf
    current_a = current_a / s
    current_sc = _score(current_a)

    # Stage 1: L-BFGS
    if time.monotonic() < deadline - min_remaining - 40:
        a1 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 40)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Soft pressure equalization
    if time.monotonic() < deadline - min_remaining - 10:
        pe_deadline = min(deadline - min_remaining - 5, time.monotonic() + 90)
        a2, sc2n = _soft_pressure_equalize(current_a, pe_deadline, n_iters=250, gamma=0.4, beta0=0.015)
        sc2 = _score(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: L-BFGS again
    if time.monotonic() < deadline - min_remaining - 10:
        a3 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 10)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Hard pressure equalization
    if time.monotonic() < deadline - min_remaining - 5:
        pe_deadline2 = min(deadline - min_remaining, time.monotonic() + 60)
        a4, sc4n = _hard_pressure_equalize(current_a, pe_deadline2, n_iters=150, gamma=0.3)
        sc4 = _score(a4)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 50)
        a5, sc5 = _subgrad_polish(current_a, sg_deadline, n_max_steps=2000, lr_init=0.02)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: adaptive L-BFGS, soft/hard pressure equalization,
    coordinate descent, multi-n exploration, diverse Fourier initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=800)
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure equalization ---
    print(f"\n[Phase 2] Extended soft PE (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        pe_dead = min(deadline - 500, time.monotonic() + 180)
        a_pe, _ = _soft_pressure_equalize(best_seq, pe_dead, n_iters=400, gamma=0.35, beta0=0.01, beta_decay=0.98)
        _update(a_pe)

    # --- Phase 3: Follow-up L-BFGS after PE ---
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_f = _run_lbfgs(best_seq, deadline, min_time_remaining=400)
        _update(a_f, _score(a_f))

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    # Try larger n first for exploration
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 250:
            break
        remaining_for_this = deadline - 250 - time.monotonic()
        if remaining_for_this < 40:
            break

        # Initialize from parent via resampling
        if parent is not None and n_parent != n_try:
            a_init = _resample(parent, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(20, remaining_for_this * 0.25)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 5: Diverse restarts ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)

        # Try different profile blends
        restart_kinds = ["ramp", "exp_ramp", "double_cos", "gauss_asym", "wavelet", "chirp_amp"]
        for i, prof in enumerate(restart_kinds):
            if time.monotonic() > deadline - 40:
                break

            a_profile = _make_profile(n_best, prof)
            # Blend: 70% best + 30% profile
            a_blend = 0.7 * best_seq + 0.3 * a_profile
            a_blend = np.maximum(a_blend, 1e-10)
            a_blend = a_blend / a_blend.sum()

            # Fourier perturbation
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=0.04)

            a_r, sc_r = _multi_stage_optimize(a_pert, deadline, min_remaining=30)
            _update(a_r, sc_r)

    # --- Phase 6: Fourier-domain random initialization ---
    print(f"\n[Phase 6] Fourier random init (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        n_best = len(best_seq) if best_seq is not None else 8192
        for attempt in range(3):
            if time.monotonic() > deadline - 50:
                break
            a_f = _fourier_init(n_best, rng, n_modes=rng.integers(4, 8))
            a_opt, sc_opt = _multi_stage_optimize(a_f, deadline, min_remaining=30)
            _update(a_opt, sc_opt)

    # --- Phase 7: Coordinate descent polish ---
    print(f"\n[Phase 7] Coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        cd_dead = min(deadline - 10, time.monotonic() + 120)
        a_cd, _ = _coordinate_polish(best_seq, cd_dead, n_rounds=60)
        _update(a_cd)

    # --- Phase 8: Final L-BFGS + subgrad ---
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=3)
        _update(a_fin, _score(a_fin))

        if time.monotonic() < deadline - 2:
            a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.005)
            _update(a_sg, sc_sg)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Ensure all values are finite and valid
        final = np.nan_to_num(final, nan=0.0, posinf=1000.0, neginf=0.0)
        final = np.maximum(final, 0.0)
        sc_final = _score(final)
        # Cross-check with evaluate_sequence
        try:
            ev_sc = evaluate_sequence([float(x) for x in final])
            if np.isfinite(ev_sc):
                sc_final = ev_sc
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
