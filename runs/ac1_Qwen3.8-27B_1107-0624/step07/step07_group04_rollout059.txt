"""
Symmetric L-BFGS optimizer with adaptive log-sum-exp continuation and CG-Newton active set polish.
Key innovation: exploits the symmetry of the optimal solution (a_i = a_{n-1-i}) to halve the
number of free variables, enabling faster iterations and more aggressive optimization within the time budget.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_full(z_full):
    z_full = z_full - np.max(z_full)
    e = np.exp(np.clip(z_full, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z_full) / len(z_full)
    return e / s


def _sym_to_full(z_free, n):
    """Convert symmetric free variables to full vector."""
    m = n // 2
    z_full = np.empty(n)
    z_full[:m] = z_free
    z_full[m:] = z_free[::-1]
    return z_full


def _smoobj_args(args):
    z_free, beta, n = args
    z_full = _sym_to_full(z_free, n)
    a = _softmax_full(z_full)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoobj_wrapper(z_free, beta, n):
    return _smoobj_args((z_free, beta, n))


def _smograd_args(args):
    z_free, beta, n = args
    m = n // 2
    z_full = _sym_to_full(z_free, n)
    a = _softmax_full(z_full)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # Compute h_i = sum_k w_k * a_{k-i} for i = 0..n-1
    # h = conv(w[::-1], a) extracted appropriately
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    # wc has length (2n-1) + n - 1 = 3n-2
    # h_i = wc[i + (len(w)-1)] = wc[i + 2n-2]
    h = wc[2*n-2:3*n-2]

    W = float(np.dot(w, c))

    # Gradient w.r.t. full z: 4n * a_i * (h_i - W)
    grad_full = 4.0 * n * a * (h - W)

    # Gradient w.r.t. free z: sum of grad at position j and n-1-j
    grad_free = grad_full[:m] + grad_full[m:][::-1]

    return grad_free


def _smograd_wrapper(z_free, beta, n):
    return _smograd_args((z_free, beta, n))


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):
    """Solve H*x = rhs using CG with the Hankel matrix structure."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_polish(a_init, deadline, max_rounds=100):
    """CG-Newton polish on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set tolerance
        tol = max(0.0002 * c_max, 1e-10)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-12 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            if not np.isfinite(np.linalg.norm(dx)) or np.linalg.norm(dx) < 1e-30:
                dx = -g_t / (gn + 1e-30)

        # Backtracking line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try <= c_max - 0.05 * alpha * gn * 0.01:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient step fallback
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 4)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) * (1 + 0.3 * np.cos(4 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _make_symmetric(a):
    """Make a sequence symmetric."""
    n = len(a)
    m = n // 2
    a_sym = np.empty(n)
    a_sym[:m] = (a[:m] + a[m:][::-1]) / 2
    a_sym[m:] = a_sym[:m][::-1]
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        a_sym = np.ones(n) / n
    else:
        a_sym = a_sym / s
    return a_sym


def _perturb_sym(a, sigma, rng):
    """Symmetric perturbation in log space."""
    n = len(a)
    m = n // 2
    z = np.log(np.maximum(a[:m], 1e-30))
    z = z + sigma * rng.standard_normal(m)
    z = z - z.mean()
    z_full = _sym_to_full(z, n)
    return _softmax_full(z_full)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n_new) / n_new
    else:
        a_new = a_new / s
    return a_new


def _subgrad_polish_sym(a_init, deadline, max_steps=3000):
    """Subgradient polish using symmetric parameterization."""
    n = len(a_init)
    m = n // 2
    z = np.log(np.maximum(a_init[:m], 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Use a medium beta for the gradient
        a = _softmax_full(_sym_to_full(z, n))
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set for gradient
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Compute gradient using the smooth formula with this fixed w
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_full = 4.0 * n * a * (h - W)
        grad_free = grad_full[:m] + grad_full[m:][::-1]

        grad_norm = np.linalg.norm(grad_free)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad_free
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()

        if i % 30 == 0:
            a_check = _softmax_full(_sym_to_full(z_new, n))
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a[:m], 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.012
            z = z_new
        else:
            z = z_new

    a_final = _softmax_full(_sym_to_full(z, n))
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Symmetric optimizer: exploits a_i = a_{n-1-i} to halve variables.
    Multi-phase: L-BFGS (adaptive beta) -> Newton polish -> subgradient -> rest
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Ensure parent is symmetric
    if n_parent % 2 == 0:
        parent = _make_symmetric(parent)
        par_sc = _score(parent)
        update_best(parent, par_sc, "parent_sym")

    # Phase 1: Main L-BFGS from parent (symmetric)
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    n_main = n_parent if n_parent % 2 == 0 else (n_parent // 2) * 2
    if n_main < 4:
        n_main = 4096
    m_main = n_main // 2
    a_cur = parent if len(parent) == n_main else _resample(parent, n_main)
    a_cur = _make_symmetric(a_cur)
    z_init = np.log(np.maximum(a_cur[:m_main], 1e-30))
    z_init = z_init - z_init.mean()

    a1 = _run_lbfgs_sym_v2(z_init, n_main, deadline, min_time_remaining=400.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Newton polish
    print(f"\n[Phase 2] Newton polish 1 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 200), max_rounds=80)
        update_best(a2, sc2, "newton_1")

    # Phase 3: Subgradient
    print(f"\n[Phase 3] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a3, sc3 = _subgrad_polish_sym(best_seq, min(deadline - 200, time.monotonic() + 120), max_steps=5000)
        update_best(a3, sc3, "subgrad_1")

    # Phase 4: Newton polish 2
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 150, time.monotonic() + 100), max_rounds=100)
        update_best(a4, sc4, "newton_2")

    # Phase 5: Try different n values
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_tries = [4096, 8192, 16384]
    if n_parent % 2 == 0 and n_parent in n_tries:
        n_tries.remove(n_parent)
    n_tries.insert(0, n_parent if n_parent % 2 == 0 else 8192)

    for n_try in n_tries:
        if time.monotonic() > deadline - 80:
            break
        if n_try < 4 or n_try > 32768:
            continue
        if n_try == n_main:
            continue  # Already optimized this n
        a_init_try = _resample(parent, n_try)
        a_init_try = _make_symmetric(a_init_try)
        m_try = n_try // 2
        z_try = np.log(np.maximum(a_init_try[:m_try], 1e-30))
        z_try = z_try - z_try.mean()

        # Time budget per n
        remaining_budget = max(60, (deadline - 60 - time.monotonic()) / max(1, len(n_tries)))
        a_opt = _run_lbfgs_sym_v2(z_try, n_try, time.monotonic() + remaining_budget, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 50:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 40, time.monotonic() + 60), max_rounds=60)
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restarts = [
            ("perturb_small", _perturb_sym(best_seq, 0.03, rng)),
            ("perturb_med", _perturb_sym(best_seq, 0.1, rng)),
            ("gaussian", _make_profile(n_best, "gaussian")),
            ("super_gauss", _make_profile(n_best, "super_gauss")),
            ("plateau", _make_profile(n_best, "plateau")),
        ]
        for name, p_start in restarts:
            if time.monotonic() > deadline - 20:
                break
            p_start = _make_symmetric(p_start)
            m_r = len(p_start) // 2
            z_r = np.log(np.maximum(p_start[:m_r], 1e-30))
            z_r = z_r - z_r.mean()
            remaining = deadline - 15 - time.monotonic()
            if remaining < 10:
                break
            a_r = _run_lbfgs_sym_v2(z_r, len(p_start), time.monotonic() + remaining * 0.6, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{name}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 8, time.monotonic() + 20), max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{name}_np")

    # Phase 7: Final ultra-fine polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
