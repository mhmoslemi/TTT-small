```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses flat-spectrum (Gerchberg-Saxton) initialization, adaptive smoothed
subgradient with Adam in log-space, scipy L-BFGS-B for small n,
coordinate descent polishing, multi-scale exploration (n=4096-16384),
and diverse perturbation restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize as _scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smoothed_grad(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    conv_full = _fft_convolve(a, p[::-1])
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_topk_grad(a, c, top_k=5, decay=0.7):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    ps = p.sum()
    if ps < 1e-300:
        return np.zeros(n)
    p = p / ps
    conv_full = _fft_convolve(a, p[::-1])
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _flat_spectrum_init(n, rng, n_iter=80):
    L = _next_pow2(2 * n - 1)
    a = rng.uniform(0.1, 1.0, n)
    for _ in range(n_iter):
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        target_mag = abs(A[0]) / np.sqrt(L)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_padded_new = np.fft.irfft(A_new, L)
        a_new = np.maximum(a_padded_new[:n], 0.0)
        if a_new.sum() < 1e-10:
            break
        a = a_new
    return a / a.sum()


def _gaussian_init(n, sigma_frac=0.4):
    j = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    return a / a.sum()


def _cosine_init(n):
    j = np.arange(n)
    a = np.cos(np.pi * j / (2.0 * n)) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=3, decay=0.6,
                  eval_interval=200, clip_norm=0.8, use_backtrack=True,
                  plateau_frac=0.2):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / (1.0 - plateau_frac)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if T < T_start * 0.01:
            g = _compute_topk_grad(a, c, active_k, decay)
        else:
            g = _compute_smoothed_grad(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 50 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.02:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _scipy_lbfgs_optimize(a_init, deadline, max_iter=5000):
    n = len(a_init)
    if n > 2048:
        return a_init.copy(), _score_norm(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    def _obj_grad(z):
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        T = 0.001 * c.max() + 1e-15
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            return best_score, np.zeros(n)
        p = e / s
        conv_full = _fft_convolve(a, p[::-1])
        g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        return 2.0 * n * (c.max() + T * np.log(s)), z_grad

    result = _scipy_minimize(_obj_grad, z0, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                     'gtol': 1e-6})
    z_opt = result.x
    z_opt = z_opt - z_opt.mean()
    a_opt = _softmax_stable(z_opt)
    sc = _score_norm(a_opt)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_opt.copy()
    return best_a, best_score


def _fast_cd_polish(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]

    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for k in active[:30]:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue
            for j in js[:10]:
                for scale in [1.2, 1.4, 0.8, 0.6, 0.4]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _stochastic_local_search(a, deadline, rng, n_iters=5000, n_perturb=3):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.001
    active = np.where(c > tol)[0]

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        a_cand = best_a.copy()
        for _ in range(n_perturb):
            if len(active) > 0:
                k = rng.integers(len(active))
                kk = active[k]
                if kk < n:
                    js = list(range(max(0, kk - n + 1), min(kk, n - 1) + 1))
                else:
                    js = list(range(kk - (n - 1), n))
                if js:
                    j = js[rng.integers(len(js))]
            else:
                j = rng.integers(n)
            factor = rng.choice([0.5, 0.7, 1.3, 1.5, 2.0])
            a_cand[j] *= factor
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand /= s
        c_cand = _fft_convolve(a_cand, a_cand)
        sc = 2.0 * n * float(c_cand.max())
        if sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            c = c_cand
            c_max = float(c.max())
            tol = c_max * 1.001
            active = np.where(c > tol)[0]
    return best_a, best_score
