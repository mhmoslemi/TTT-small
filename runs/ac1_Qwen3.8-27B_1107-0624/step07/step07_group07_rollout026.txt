"""
LP-based step function optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2
by iteratively solving LPs to find the best direction that keeps the
convolution bounded, combined with spectral shaping and local polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_direction(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0. Returns optimal g."""
    n = len(a)
    g = cp.Variable(n, non_neg=True)
    
    # Build the constraint matrix (Toeplitz)
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    # This is a convolution, so the constraint matrix is Toeplitz
    
    # For efficiency, build the matrix explicitly
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    
    # Actually, for very large n, building the full matrix might be expensive.
    # Let's use a sparse representation or a different approach.
    
    # Alternative: express the constraint as a series of linear combinations
    # (a*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * g[j]
    
    # For n=4096, the matrix A is (2n-1) x n = 8191 x 4096, which has about
    # n^2/2 nonzeros. That's about 8 million nonzeros, which is large but manageable
    # for sparse solvers.
    
    # Let's build it as a sparse matrix
    from scipy import sparse
    
    # Build rows of A
    # For row k (k=0..2n-2), the nonzeros are at columns j where 0<=k-j<n
    # i.e., max(0,k-n+1) <= j <= min(n-1, k)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        # or use other solvers
    except:
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                return None
    
    if g.value is not None:
        return np.asarray(g.value).ravel()
    return None

Minimize `t`

LP-based step function optimizer using iterative linear programming to
minimize 2*n*max(conv(a,a))/sum(a)^2. Combines LP direction finding,
adaptive blending, spectral analysis, and local polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy import sparse
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
