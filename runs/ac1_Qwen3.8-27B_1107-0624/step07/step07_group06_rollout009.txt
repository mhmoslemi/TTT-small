def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _solve_qp_active(active_set_bool, n, max_outer=200, tol=1e-14, reg_init=1e-10):
    """Solve: min a^T H_S a s.t. sum(a)=1, a>=0
    where H_S is Hankel with (H_S)_{i,j}=1 if i+j in active_set.
    Returns the optimal a and the objective value."""
    # Support set (indices where a_i > 0)
    support = np.ones(n, dtype=bool)
    reg = reg_init
    
    for outer in range(max_outer):
        I = np.where(support)[0]
        nI = len(I)
        if nI == 0:
            return np.ones(n) / n, np.inf
        if nI == 1:
            a = np.zeros(n)
            a[I[0]] = 1.0
            return a, _score(a)
        
        # Build reduced Hessian function
        active_red = active_set_bool[I[:, None] + I[None, :]]
        # (H_red v)_i = sum_j v[j] * active_red[i,j]
        # = sum_j v[j] * S[I[i] + I[j]]
        # This is trickier since I is not contiguous.
        # Use a different approach: compute H @ v for full vector, then restrict.
        
        def hess_full(v_full):
            return _hess_vec(v_full, active_set_bool, n)
        
        def hess_reduced(v_red):
            # v_red is of length nI, corresponding to indices I
            v_full = np.zeros(n)
            v_full[I] = v_red
            result = hess_full(v_full)
            return result[I]
        
        # Solve: H_reduced @ a_red = lambda * 1, sum(a_red) = 1
        # Using augmented system:
        # [H  1] [a]   [lam]
        # [1' 0] [lam] = [1]
        # But lam is unknown. Instead:
        # From H @ a = (lam/2) * 1:  a = (lam/2) * H^{-1} @ 1
        # From 1' @ a = 1:  (lam/2) * 1' @ H^{-1} @ 1 = 1
        # So: a = H^{-1} @ 1 / (1' @ H^{-1} @ 1)
        
        # Compute H^{-1} @ 1 using CG
        ones = np.ones(nI)
        try:
            x = _cg_solve(hess_reduced, ones, nI, max_iter=10000, tol=1e-14, reg=reg)
        except Exception:
            reg *= 10
            continue
        
        denom = float(np.dot(ones, x))
        if abs(denom) < 1e-30:
            # H is (nearly) singular on this support; try regularization
            reg *= 10
            if reg > 1.0:
                # Give up on this support
                support = support & (np.ones(n, dtype=bool))
                break
            continue
        
        a_red = x / denom
        
        # Check non-negativity
        if np.all(a_red >= -1e-10):
            a_red = np.maximum(a_red, 0)
            a_red = a_red / a_red.sum()
            # Check KKT for inactive constraints
            a_full = np.zeros(n)
            a_full[I] = a_red
            H_a = hess_full(a_full)
            # For indices not in support: need (H_a)_i >= lambda
            lam = float(np.mean(H_a[I]))
            violations = H_a[~support] < lam - 1e-10
            if not np.any(violations):
                # KKT satisfied!
                a_full = np.maximum(a_full, 0)
                a_full = a_full / a_full.sum()
                return a_full, _score(a_full)
            else:
                # Deactivate the worst violators
                worst_idx = np.where(violations)[0]
                # Remove the one with most negative violation
                worst_score = H_a[worst_idx]
                worst = worst_idx[np.argmin(worst_score)]
                support[worst] = False
                continue
        else:
            # Some a_red[i] < 0; deactivate the most negative
            min_idx = np.argmin(a_red)
            I_deactivate = I[min_idx]
            support[I_deactivate] = False
            continue
    
    # Fallback: return uniform on support
    a = np.zeros(n)
    a[support] = 1.0 / support.sum()
    return a, _score(a)


def _active_set_refine(a, deadline, max_rounds=50):
    """Iteratively refine using active-set QP."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set with adaptive tolerance
        # Use a relative tolerance that tightens over iterations
        base_tol = max(1e-6, 0.001 * c_max / (1 + 0.1 * round_num))
        
        for tol_frac in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.monotonic() > deadline:
                break
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            if active.sum() < 2:
                active[int(np.argmax(c))] = True
                active[int(np.argsort(c)[-2])] = True
            if active.sum() > n * 0.8:
                continue
            
            a_new, sc_new = _solve_qp_active(active, n, max_outer=100, tol=1e-13, reg_init=1e-12)
            
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
                c = _full_convolve(a, a)
                c_max = float(c.max())
                break
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n)
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a
    elif kind == "truncated_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a
    elif kind == "random_positive":
        rng = np.random.default_rng(12345)
        return rng.exponential(1.0, n)
    else:
        return np.ones(n)


def _run_lbfgs_smooth(a_init, deadline, min_time_remaining=30.0, aggressive=True):
    """L-BFGS with smooth-max objective and adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                       0.007, 0.003, 0.001, 0.0005, 0.0002,
                                       0.0001, 0.00005, 0.00002])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        est_time_per_iter = 0.002 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 10))))
        
        def fun_local(z_local):
            return _smoobj(z_local, beta, n)
        
        def jac_local(z_local):
            return _smograd(z_local, beta, n)
        
        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _fourier_optimize(a_init, n_modes, deadline, min_time_remaining=30.0):
    """Optimize in a Fourier (cosine) subspace for global search."""
    n = len(a_init)
    t = np.arange(n)
    
    # Build cosine basis
    basis = np.zeros((n, n_modes))
    for k in range(n_modes):
        basis[:, k] = np.cos(np.pi * k * t / n)
    basis[:, 0] = 1.0  # DC component
    
    # Initial coefficients
    coeffs = np.linalg.lstsq(basis, a_init, rcond=None)[0]
    
    best_coeffs = coeffs.copy()
    best_sc = _score(a_init)
    
    def fun_fourier(c):
        a = basis @ c
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        return _score(a)
    
    x0 = coeffs
    x0 = x0 - x0.mean() + x0.mean()  # keep as is
    
    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 5:
        return a_init
    
    max_iter = min(5000, max(100, int(remaining / 0.05)))
    
    try:
        result = minimize(
            fun_fourier, x0, method='Nelder-Mead',
            options={'maxiter': max_iter, 'xatol': 1e-12, 'fatol': 1e-14}
        )
        a_new = basis @ result.x
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            sc_new = _score(a_new)
            if sc_new < best_sc:
                return a_new
    except Exception:
        pass
    
    # Try L-BFGS in Fourier space with smooth objective
    if time.monotonic() + min_time_remaining < deadline:
        c_cur = _full_convolve(a_init, a_init)
        max_c = float(c_cur.max())
        beta = max_c * 0.001
        
        def fun_fourier_smooth(c):
            a = basis @ c
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            cconv = _full_convolve(a, a)
            c_max = float(cconv.max())
            shifted = (cconv - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining > 5:
            max_iter2 = min(2000, max(50, int(remaining / 0.1)))
            try:
                result2 = minimize(
                    fun_fourier_smooth, x0, method='Nelder-Mead',
                    options={'maxiter': max_iter2, 'xatol': 1e-10, 'fatol': 1e-12}
                )
                a_new = basis @ result2.x
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new = a_new / s
                    sc_new = _score(a_new)
                    if sc_new < best_sc:
                        return a_new
            except Exception:
                pass
    
    return a_init


def _newton_polish(a_init, deadline, max_iter=500):
    """Newton's method on the active-set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        tol = max(0.0005 * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        if active.sum() > 2 * n:
            # Too many active; tighten
            tol = 0.001 * c_max
            active = c >= (c_max - tol)
        
        # Gradient of sum_{k in active} c_k
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * a * (h - W)
        
        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break
        
        # Newton step with CG
        def hess_vec(v):
            return _hess_vec(v, active, n) + 1e-10 * v
        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=2000, tol=1e-12)
        dx = dx - np.mean(dx)
        
        # Line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a
        
        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.01):
    """Subgradient descent on the active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.001)
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.95
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _palindromic_score(a):
    """Score of the palindromic version of a (for symmetry exploitation)."""
    n = len(a)
    a_pal = np.concatenate([a, a[:n//2][::-1]]) if n % 2 == 0 else np.concatenate([a, a[:n//2+1][::-1]])
    return _score(a_pal)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining smooth-max L-BFGS, active-set QP,
    Newton polishing, Fourier subspace search, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform") / _make_profile(n_parent, "uniform").sum()
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: L-BFGS smooth-max from parent ===
    print(f"\n[Phase 1] L-BFGS smooth-max (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + min(400, budget_s * 0.22)
    a1 = _run_lbfgs_smooth(best_seq, p1_deadline, min_time_remaining=50, aggressive=True)
    update_best(a1, _score(a1), "lbfgs_smooth")
    
    # === Phase 2: Newton polish ===
    print(f"\n[Phase 2] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(t_start + 600, deadline - 800)
    if time.monotonic() < p2_deadline:
        a2, sc2 = _newton_polish(best_seq, p2_deadline, max_iter=300)
        update_best(a2, sc2, "newton")
    
    # === Phase 3: Active-set QP refinement ===
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(t_start + 900, deadline - 500)
    if time.monotonic() < p3_deadline:
        a3, sc3 = _active_set_refine(best_seq, p3_deadline, max_rounds=30)
        update_best(a3, sc3, "active_qp")
    
    # === Phase 4: L-BFGS again from improved point ===
    print(f"\n[Phase 4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(t_start + 1100, deadline - 400)
    if time.monotonic() < p4_deadline:
        a4 = _run_lbfgs_smooth(best_seq, p4_deadline, min_time_remaining=300, aggressive=True)
        update_best(a4, _score(a4), "lbfgs_round2")
    
    # === Phase 5: Newton + Active-set QP final polish ===
    print(f"\n[Phase 5] Final Newton + QP (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(t_start + 1400, deadline - 300)
    if time.monotonic() < p5_deadline:
        a5, sc5 = _newton_polish(best_seq, p5_deadline, max_iter=500)
        update_best(a5, sc5, "final_newton")
        
    p5b_deadline = min(t_start + 1600, deadline - 150)
    if time.monotonic() < p5b_deadline:
        a5b, sc5b = _active_set_refine(best_seq, p5b_deadline, max_rounds=40)
        update_best(a5b, sc5b, "final_qp")
    
    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates and 1024 <= n_parent <= 32768:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        else:
            a_init = best_seq.copy()
        
        # L-BFGS
        n_dl = min(deadline - 80, time.monotonic() + 120)
        a_opt = _run_lbfgs_smooth(a_init, n_dl, min_time_remaining=30, aggressive=True)
        update_best(a_opt, _score(a_opt), f"n_{n_try}")
        
        # Newton
        if time.monotonic() < deadline - 60:
            n_dl2 = min(deadline - 50, time.monotonic() + 60)
            a_opt2, sc_opt2 = _newton_polish(a_opt, n_dl2, max_iter=100)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")
    
    # === Phase 7: Subgradient polish ===
    print(f"\n[Phase 7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 30, n_max_steps=3000, lr_init=0.008)
        update_best(a_sg, sc_sg, "sg_polish")
    
    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_restarts = 3
    for i in range(n_restarts):
        if time.monotonic() > deadline - 20:
            break
        if i == 0:
            p_start = _perturb(best_seq, 0.02, rng)
        elif i == 1:
            p_start = _perturb(best_seq, 0.05, rng)
        else:
            p_start = _perturb(best_seq, 0.08, rng)
        
        r_dl = min(deadline - 15, time.monotonic() + 40)
        a_r = _run_lbfgs_smooth(p_start, r_dl, min_time_remaining=10, aggressive=False)
        sc_r = _score(a_r)
        update_best(a_r, sc_r, f"restart_{i}")
        
        if sc_r < best_score * 1.002 and time.monotonic() < deadline - 10:
            a_r2, sc_r2 = _newton_polish(a_r, deadline - 5, max_iter=50)
            update_best(a_r2, sc_r2, f"restart_{i}_newton")
    
    # === Phase 9: Final ultra-low beta L-BFGS ===
    print(f"\n[Phase 9] Final ultra L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00001
        remaining = deadline - 3 - time.monotonic()
        if remaining > 3:
            max_iter = min(5000, max(200, int(remaining / 0.005)))
            
            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)
            
            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)
            
            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-15,
                             'maxcor': 100, 'maxls': 200}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra")
            except Exception:
                pass
    
    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            final = final / final.sum()
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
