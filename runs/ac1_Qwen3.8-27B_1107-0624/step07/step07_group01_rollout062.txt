    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _lbfgs_smooth_block(a_full, start, end, beta, max_iter=300):
    """
    L-BFGS optimization of a single block [start:end] of a_full.
    Uses softmax parameterization for the block.
    Returns the optimized block.
    """
    n = len(a_full)
    bs = end - start
    block = a_full[start:end].copy()
    z0 = np.log(np.maximum(block, 1e-18))

    def block_obj(z_b):
        new_block = _softmax_z(z_b)
        a_copy = a_full.copy()
        a_copy[start:end] = new_block
        c = _full_convolve(a_copy, a_copy)
        c_max = float(c.max())
        sh = (c - c_max) / beta
        e = np.exp(sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def block_grad(z_b):
        new_block = _softmax_z(z_b)
        a_copy = a_full.copy()
        a_copy[start:end] = new_block
        c = _full_convolve(a_copy, a_copy)
        c_max = float(c.max())
        sh = (c - c_max) / beta
        e = np.exp(sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a_copy[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        full_grad = 4.0 * n * a_copy * (h - W)
        bg = full_grad[start:end]
        grad_z = new_block * (bg - float(np.dot(bg, new_block)))
        return grad_z

    try:
        result = minimize(block_obj, z0, jac=block_grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 30, 'maxls': 50})
        new_block = _softmax_z(result.x)
        return new_block
    except Exception:
        return block


def _block_coord_descent(a_init, deadline, block_size=256, reserve=30.0):
    """
    Block coordinate descent: cycle through blocks, optimizing each with L-BFGS.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = int(np.ceil(n / float(block_size)))
    sweep = 0
    stall_count = 0
    stale_threshold = 8

    while time.monotonic() < deadline - reserve:
        sweep += 1
        sweep_improved = False

        # Determine block order (randomized for diversity)
        order = list(range(n_blocks))
        if sweep % 3 == 0:
            np.random.shuffle(order)

        for j in order:
            if time.monotonic() > deadline - reserve:
                break
            start = j * block_size
            end = min(n, start + block_size)
            if start >= n:
                continue

            # Adaptive beta: start larger, decrease as we converge
            c_cur = _full_convolve(a, a)
            c_max = float(c_cur.max())
            beta = c_max * max(0.005, 0.05 * (0.95 ** sweep))

            new_block = _lbfgs_smooth_block(a, start, end, beta, max_iter=200)
            a[start:end] = new_block

        # Evaluate after full sweep
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            sweep_improved = True
            stall_count = 0
            if sweep % 10 == 0:
                print(f"    BCD sweep {sweep}: score={best_sc:.10f}")
        else:
            stall_count += 1
            # If stuck, try smaller blocks
            if stall_count >= stale_threshold:
                if block_size > 64:
                    block_size = max(64, block_size // 2)
                    n_blocks = int(np.ceil(n / float(block_size)))
                    print(f"    BCD: reducing block size to {block_size}")
                stall_count = 0

        if sweep % 20 == 0:
            print(f"    BCD sweep {sweep}: best={best_sc:.10f}, block_size={block_size}")

        # Early stop if very good
        if best_sc < 1.5005:
            break

    return best_a, best_sc


def _adaptive_lbfgs_full(a_start, deadline, reserve_time=30.0):
    """Full-sequence L-BFGS with decreasing beta."""
    n = len(a_start)
    current_a = a_start.copy()
    current_sc = _score(current_a)
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return current_a

    z0 = np.log(np.maximum(current_a, 1e-18))
    z0 = z0 - z0.mean()

    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                              0.0005, 0.0002, 0.0001])

    for beta in betas:
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(5000, max(200, int(remaining / (0.003 * (n / 4096.0) ** 1.3 * 20))))

        def fun(z):
            a = _softmax_z(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            sh = (c - cm) / beta
            e = np.exp(sh)
            s = float(e.sum())
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))

        def jac(z):
            a = _softmax_z(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            sh = (c - cm) / beta
            e = np.exp(sh)
            s = float(e.sum())
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            full_g = 4.0 * n * a * (h - W)
            return full_g - float(np.dot(full_g, a)) * a

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 50, 'maxls': 60})
            a_new = _softmax_z(result.x)
            sc = _score(a_new)
            if sc < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc
                z0 = np.log(np.maximum(current_a, 1e-18))
                z0 = z0 - z0.mean()
            if current_sc < 1.5005:
                break
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=30000, lr_init=0.04):
    """Subgradient method in log-space with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax_z(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = lr * a.sum() / n
        z = z - step * grad
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 1000 == 0:
            a_check = _softmax_z(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.05
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax_z(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.4):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.3):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _peak_reduction(a_init, deadline, n_iterations=500):
    """
    Direct peak reduction: identify peaks and redistribute mass.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.01

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find peaks above threshold
        threshold = c_max * 0.95
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            break

        # Compute gradient contribution from peaks
        w = np.zeros_like(c)
        w[peak_mask] = 1.0 / n_peaks

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()

        # Line search
        improved = False
        alpha = step_size
        for _ in range(10):
            a_try = a - alpha * g / n
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                step_size = min(alpha * 1.1, 0.05)
                break
            alpha *= 0.7

        if not improved:
            step_size *= 0.8
            if step_size < 1e-8:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using block-coordinate descent as the main method,
    combined with full-sequence L-BFGS, subgradient polishing, and peak reduction.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 0: Get initial candidates ===
    print("=== Phase 0: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    if best_seq is None:
        n_init = 4096
        a_init = _make_chirp(n_init, freq=1.0, eps=0.4)
        try_update(a_init, "chirp_init")
        print(f"  Chirp init: n={n_init}, score={_score(a_init):.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 0 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 1: Block-coordinate descent (main workhorse) ===
    n_target = len(best_seq)
    print(f"\n=== Phase 1: Block-coordinate descent (n={n_target}) ===")
    bcd_deadline = deadline - 300.0
    a_bcd, sc_bcd = _block_coord_descent(best_seq, bcd_deadline, block_size=256, reserve=60.0)
    try_update(a_bcd, "bcd_bs256")

    # === Phase 2: Smaller blocks for more precision ===
    print(f"\n=== Phase 2: Fine block descent (n={n_target}) ===")
    if time.monotonic() < deadline - 200:
        a_bcd2, sc_bcd2 = _block_coord_descent(best_seq, deadline - 200,
                                               block_size=128, reserve=50.0)
        try_update(a_bcd2, "bcd_bs128")

    # === Phase 3: Full L-BFGS refinement ===
    print(f"\n=== Phase 3: Full L-BFGS ===")
    if time.monotonic() < deadline - 150:
        a_lf = _adaptive_lbfgs_full(best_seq, deadline - 150, reserve_time=30.0)
        try_update(a_lf, "full_lbfgs")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient polishing ===")
    if time.monotonic() < deadline - 100:
        a_sg, _ = _subgrad_polish(best_seq, deadline - 100, k_active=5,
                                  max_steps=20000, lr_init=0.03)
        try_update(a_sg, "sg_k5")

    if time.monotonic() < deadline - 60:
        a_sg2, _ = _subgrad_polish(best_seq, deadline - 60, k_active=3,
                                   max_steps=15000, lr_init=0.02)
        try_update(a_sg2, "sg_k3")

    # === Phase 5: Peak reduction ===
    print(f"\n=== Phase 5: Peak reduction ===")
    if time.monotonic() < deadline - 40:
        a_pr, _ = _peak_reduction(best_seq, deadline - 40, n_iterations=300)
        try_update(a_pr, "peak_red")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Length exploration ===")
    for target_n in [n_target * 2, n_target // 2]:
        if time.monotonic() > deadline - 60:
            break
        if target_n < 256 or target_n > 65536:
            continue
        if target_n == n_target:
            continue
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        try_update(a_interp, f"interp_{target_n}")
        if time.monotonic() < deadline - 50:
            a_opt, _ = _block_coord_descent(a_interp, deadline - 50,
                                            block_size=128, reserve=20.0)
            try_update(a_opt, f"bcd_{target_n}")

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final polishing ===")
    if time.monotonic() < deadline - 5:
        # One more round of small-block BCD
        if time.monotonic() < deadline - 10:
            a_final, _ = _block_coord_descent(best_seq, deadline - 10,
                                              block_size=64, reserve=5.0)
            try_update(a_final, "final_bcd_bs64")

        # Quick subgradient
        if time.monotonic() < deadline - 5:
            a_fs, _ = _subgrad_polish(best_seq, deadline - 5, k_active=3,
                                      max_steps=5000, lr_init=0.01)
            try_update(a_fs, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
