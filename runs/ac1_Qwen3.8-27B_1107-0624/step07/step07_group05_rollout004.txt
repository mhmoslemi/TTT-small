def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

def _make_msequence_profile(n, eps=0.05):
    """Create a non-negative profile from an m-sequence of appropriate length."""
    # Find m such that 2^m - 1 is close to n
    m = max(4, int(np.log2(n)) + 1)
    mlen = 2**m - 1
    raw = _lfsr_msequence(m)
    # Map binary to [eps, 1]
    mapped = np.where(raw > 0.5, 1.0, eps)
    # Tile or truncate to length n
    if mlen >= n:
        profile = mapped[:n]
    else:
        repeats = int(np.ceil(n / mlen))
        profile = np.tile(mapped, repeats)[:n]
    profile = np.maximum(profile, 0.0)
    s = profile.sum()
    if s > 0:
        profile = profile / s
    return profile


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 2 * t * t) + 1.5
        return np.maximum(a, 1e-10) / np.maximum(a, 1e-10).sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 13 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2 * 8) + 0.05
        return a / a.sum()
    elif kind == "mseq":
        return _make_msequence_profile(n)
    elif kind == "mseq_blend":
        mseq = _make_msequence_profile(n)
        uniform = np.ones(n) / n
        return 0.6 * mseq + 0.4 * uniform
    else:
        return np.ones(n) / n


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_active(a, c, n, top_k=10):
    """Subgradient using top-k peaks."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = 0.02 * c_max
    active = c >= (c_max - tol)
    if active.sum() < top_k:
        idx = np.argsort(c)[-top_k:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    elif active.sum() > top_k * 5:

        active = np.zeros_like(c, dtype=bool)

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _peak_shave(a, deadline, n_rounds=50, n_try=20):
    """
    Peak shaving: identify top contributors to the peak and reduce them.
    Uses analytical insight: reducing a_i by delta reduces c_k by ~delta * a_{k-i}.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top peaks
        n_peaks = min(10, len(c))
        peak_indices = np.argsort(c)[-n_peaks:]

        # For each peak, find the top contributing pair (i, k-i)
        # Contribution of a_i to c_k is a_i * a_{k-i}
        # Total contribution of a_i across all peaks: sum_{k in peaks} a_i * a_{k-i}
        contrib = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j]

        # Try reducing top contributors
        n_modify = min(n_try, max(5, n // 200))
        top_idx = np.argsort(contrib)[-n_modify:]
        rng.shuffle(top_idx)

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            c_before = c.max()

            # Try several reduction factors
            for factor in [0.6, 0.75, 0.85, 0.95, 1.1, 1.3]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    break
                c_new = _full_convolve(a, a)
                c_after = c_new.max()
                if c_after < c_before - 1e-14:
                    improved = True
                    break
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _pressure_equalize(a, deadline, n_iters=300, gamma=0.35):
    """Multiplicative update to equalize pressure at active peaks."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:

        c_max = float(c.max())

        tol = frac * c_max
        active = c >= (c_max - tol)

            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Multiplicative update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 25 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, max_beta_stages=15):

    if max_c < 1e-15:
        return a_init

    if len(c_sorted) > 20:

        gap = c_sorted[0] * 0.5
    base_beta = max(gap * 0.5, max_c * 0.03)

        1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):

            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

            a_new = _softmax(result.x)

def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 3)))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        _update(parent, parent_sc)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: M-sequence based initialization and optimization ---
    print(f"\n[Phase 1] M-sequence initialization (t={time.monotonic()-t_start:.1f}s)")
    mseq_n = n_parent if n_parent > 0 else 8192
    a_mseq = _make_profile(mseq_n, "mseq")
    a_mseq_bl = _make_profile(mseq_n, "mseq_blend")
    _update(a_mseq)
    _update(a_mseq_bl)

    # Optimize m-sequence if it's competitive
    best_init = a_mseq if _score(a_mseq) < _score(a_mseq_bl) else a_mseq_bl
    if _score(best_init) < best_score * 1.05:
        rem = deadline - 600 - time.monotonic()
        if rem > 60:
            a_opt, sc_opt = _multi_stage_optimize(best_init, deadline - 600, min_remaining=600)
            _update(a_opt, sc_opt)

    # --- Phase 2: Main optimization on parent/best ---
    print(f"\n[Phase 2] Main optimization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a_main, sc_main = _multi_stage_optimize(best_seq, deadline, min_remaining=400)
        _update(a_main, sc_main)
    elif parent is not None:
        a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=400)
        _update(a_main, sc_main)

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288]
    if n_parent > 0:
        n_candidates = [x for x in n_candidates if abs(x - n_parent) > 512]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 45:
            break

        # Multiple initializations for this n
        inits = []
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, parent)
            a_interp = np.maximum(a_interp, 0)
            s_int = a_interp.sum()
            if s_int > 1e-10:
                inits.append(a_interp / s_int)
        inits.append(_make_profile(n_try, "mseq"))
        inits.append(_make_profile(n_try, "tapered"))

        best_for_n = None
        best_sc_n = np.inf
        for a_init in inits:
            if time.monotonic() > deadline - 120:
                break
            sc_init = _score(a_init)
            if best_for_n is None or sc_init < best_sc_n:
                best_for_n = a_init
                best_sc_n = sc_init

        if best_for_n is not None:
            min_rem = max(30, (deadline - 120 - time.monotonic()) * 0.2)
            a_opt, sc_opt = _multi_stage_optimize(best_for_n, deadline - 120, min_remaining=min_rem)
            _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts from best ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_kinds = ["chirp", "double_cos", "exp_taper", "mseq"]

        for i, prof in enumerate(restart_kinds):
            if time.monotonic() > deadline - 25:
                break

            a_blend = 0.4 * best_seq + 0.6 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            s = a_blend.sum()
            if s < 1e-10:
                continue
            a_blend = a_blend / s
            a_pert = _fourier_perturb(a_blend, rng, n_modes=4, amplitude=0.02)

            rem = min(80, deadline - 25 - time.monotonic())
            if rem > 20:
                a_r, sc_r = _multi_stage_optimize(a_pert, time.monotonic() + rem, min_remaining=5)
                _update(a_r, sc_r)

    # --- Phase 5: Focused final polishing on best ---
    print(f"\n[Phase 5] Final polishing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Extended L-BFGS with very fine beta
        a_fin = _run_lbfgs(best_seq, deadline - 5, min_time_remaining=3, max_beta_stages=15)
        _update(a_fin, _score(a_fin))

    # --- Phase 6: Peak shaving on absolute best ---
    print(f"\n[Phase 6] Final peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_ps, sc_ps = _peak_shave(best_seq, deadline - 3, n_rounds=80, n_try=40)
        _update(a_ps, sc_ps)

    # --- Phase 7: Final subgradient ---
    print(f"\n[Phase 7] Final subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 1, n_max_steps=2000, lr_init=0.005)
        _update(a_sg, sc_sg)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Normalize to a reasonable scale (evaluate_sequence clips to [0, 1000])
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
