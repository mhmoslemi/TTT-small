```python
def lp_step(f, M, K, solver='ECOS'):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b)[k] <= M for k in K"""
    n = len(f)
    import cvxpy as cp
    
    b = cp.Variable(n)
    constraints = [b >= 0]
    for k in K:
        # conv(f, b)[k] = sum_i f[i] * b[k-i]
        # This is a linear function of b
        coeffs = np.zeros(2*n-1)
        for i in range(n):
            if 0 <= k-i < n:
                coeffs[k-i] = f[i]
        # constraints: coeffs @ b <= M
        constraints.append(np.dot(coeffs[:n], b) <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)
    
    if b.value is not None:
        return np.maximum(b.value, 0)
    return None
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    c = _full_convolve(a, a)
```

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP-based direction finding (primary new approach)
2. L-BFGS with adaptive log-sum-exp smoothing
3. Threshold-based exact subgradient with adaptive step size
4. Projected subgradient as fallback
5. KKT-inspired equalization
6. Multi-length exploration
Key innovation: LP-based direction that finds the "best" next state
compatible with the current convolution bound, solved via cutting-plane.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _build_constraint_matrix(f, K, n):
    """Build LP constraint matrix A where A[p,j] = f[K[p]-j] if valid."""
    A = np.zeros((len(K), n))
    for p, k in enumerate(K):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[p, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
    return A


def _lp_direction(f, M, K, verbose=False):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(f,b)[k] <= M for k in K.
    Returns b (unrescaled) or None.
    """
    n = len(f)
    if len(K) == 0:
        return f.copy()

    A = _build_constraint_matrix(f, K, n)
    b_var = cp.Variable(n)
    constraints = [b_var >= 0]
    constraints.append(A @ b_var <= M * np.ones(len(K)))
    # Bounding constraint
    constraints.append(cp.sum(b_var) <= 3.0 * float(f.sum()))

    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)

    solvers_to_try = [cp.SCS, cp.ECOS, cp.CLARABEL]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None:
                return np.maximum(b_var.value, 0.0)
        except Exception:
            continue
    return None


def _lp_optimize(f_init, deadline, max_iters=200, K_size=100, reserve=30.0):
    """
    LP-based optimization loop.
    At each step:
    1. Compute M = max conv(f,f)
    2. Find active set K (top K_size lags)
    3. Solve LP to get direction b
    4. Rescale b, line search for step size
    5. Update f
    """
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    it = 0

    while it < max_iters and time.monotonic() < deadline - reserve:
        c = _full_convolve(f, f)
        c_max = float(c.max())

        # Active set: top K_size lags
        idx = np.argsort(c)[::-1][:K_size]
        K = sorted(idx.tolist())

        # Solve LP
        b = _lp_direction(f, c_max, K)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-10:
            break

        # Rescale to match f's sum
        g = b / b_sum * s

        # Line search
        best_t = 0.0
        best_t_sc = _score(f)
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            f_try = (1.0 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            st = f_try.sum()
            if st < 1e-10:
                continue
            f_try = f_try / st * s
            sc_try = _score(f_try)
            if sc_try < best_t_sc - 1e-12:
                best_t_sc = sc_try
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            st = f.sum()
            f = f / st * s
            sc = _score(f)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    elapsed = time.monotonic()
                    print(f"    LP iter {it}: score={sc:.10f}, t={best_t:.3f}, K_size={len(K)}")
        else:
            # Try smaller step sizes
            improved = False
            for t in [0.02, 0.01, 0.005]:
                f_try = (1.0 - t) * f + t * g
                f_try = np.maximum(f_try, 0.0)
                st = f_try.sum()
                if st < 1e-10:
                    continue
                f_try = f_try / st * s
                sc_try = _score(f_try)
                if sc_try < best_sc - 1e-14:
                    f = f_try
                    best_sc = sc_try
                    best_f = f.copy()
                    improved = True
                    break
            if not improved:
                # Stuck, try expanding active set
                K_size = min(K_size * 2, 500)
                if K_size > 500:
                    break

        it += 1

    return best_f, best_sc


def _subgrad_threshold(a_init, deadline, threshold_frac=0.01, max_steps=100000,
                       lr_init=0.05, reserve=10.0):
    """
    Subgradient method using threshold-based active set.
    Uses all lags where c[k] > (1-threshold_frac)*c_max.
    """
    n = len(a_init)
    f = a_init.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline - reserve:
            break

        c = _full_convolve(f, f)
        c_max = float(c.max())
        threshold = c_max * (1.0 - threshold_frac)
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient: average of gradients at active lags
        # grad[j] = (1/|active|) * sum_{k in active} 2*f[k-j]
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                grad[j_min:j_max + 1] += 2.0 * f[k - j_max:k - j_min + 1][::-1]
        grad /= len(active)

        # Projected subgradient step
        f_try = f - lr * grad / n
        f_try = np.maximum(f_try, 0.0)
        st = f_try.sum()
        if st < 1e-15:
            lr *= 0.5
            stall += 1
            if stall > 50:
                break
            continue
        f_try = f_try / st * s

        sc_try = _score(f_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_f = f_try.copy()
            f = f_try
            stall = 0
            lr = min(lr * 1.02, 0.5)
        else:
            f = f_try  # Still move (subgradient method)
            stall += 1
            if stall > 20:
                lr *= 0.7
                stall = 0
            if lr < 1e-10:
                break

        if i % 5000 == 0 and i > 0:
            print(f"    SG-thresh iter {i}: score={best_sc:.10f}, lr={lr:.6f}, "
                  f"|active|={len(active)}")

    return best_f, best_sc


def _projected_subgrad(a_init, deadline, max_steps=50000, lr_init=0.1,

    """Projected subgradient with top-k active set."""

    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline - reserve:
            break

        c = _full_convolve(f, f)
        idx = np.argsort(c)[::-1][:k_active]

        grad = np.zeros(n)
        for k in idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                grad[j_min:j_max + 1] += 2.0 * f[k - j_max:k - j_min + 1][::-1]
        grad /= len(idx)

        f_try = f - lr * grad / n
        f_try = np.maximum(f_try, 0.0)
        st = f_try.sum()
        if st < 1e-15:
            lr *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        f_try = f_try / st * s

        sc_try = _score(f_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_f = f_try.copy()
            f = f_try
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            f = f_try
            stall += 1
            if stall > 30:
                lr *= 0.8
                stall = 0
            if lr < 1e-10:
                break

    return best_f, best_sc


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    """Adaptive L-BFGS with geometrically decreasing beta."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(15000, max(200, int(remaining / (0.005 * scale * 25))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _kkt_equalize(a, deadline, k_active=5, max_iter=100, reserve=5.0):
    """KKT-inspired: try to equalize active convolution values."""
    n = len(a)
    f = a.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)

    for it in range(max_iter):
        if time.monotonic() > deadline - reserve:
            break

        c = _full_convolve(f, f)
        c_max = float(c.max())
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        spread = active_vals.max() - active_vals.min()

        if spread < 1e-15:
            break

        # Compute gradient to reduce spread
        grad = np.zeros(n)
        for k in idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                grad[j_min:j_max + 1] += 2.0 * f[k - j_max:k - j_min + 1][::-1]
        grad /= len(idx)
        grad -= grad.mean()

        # Line search
        alpha = 0.5
        improved = False
        for _ in range(15):
            f_try = f - alpha * grad / n
            f_try = np.maximum(f_try, 0.0)
            st = f_try.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            f_try = f_try / st * s
            sc_try = _score(f_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_f = f_try.copy()
                f = f_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            alpha *= 0.5
            if alpha < 1e-12:
                break

    return best_f, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=30.0):
    """Simple basin hopping with L-BFGS from perturbed starts."""
    n = len(a_current)
    best_f = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.8

        # Perturb in log-space
        sigma = 0.03 + 0.02 * h
        z = np.log(np.maximum(best_f, 1e-18))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.01
        remaining = max(5, hop_deadline - time.monotonic())
        max_iter = min(3000, max(100, int(remaining / 0.01)))
        a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_f = a_opt.copy()
            print(f"    Basin hop {h}: improved to {sc_opt:.10f}")

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Best initialization selection
    Phase 2: LP-based optimization (primary new method)
    Phase 3: L-BFGS with adaptive smoothing
    Phase 4: Exact subgradient polishing (threshold-based)
    Phase 5: KKT equalization
    Phase 6: More L-BFGS rounds
    Phase 7: Basin hopping
    Phase 8: Multi-length exploration
    Phase 9: Final aggressive polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Also try structured inits at the parent's length
    init_n = len(parent) if parent is not None else 4096

    structured = [
        ("chirp1", _make_chirp(init_n, 1.0, 0.3)),
        ("chirp2", _make_chirp(init_n, 2.0, 0.25)),
        ("multi13", _make_multiband(init_n, [1, 3], 0.25)),
        ("tapered", _make_tapered(init_n, 0.5)),
    ]
    for name, a_s in structured:
        sc = _score(a_s)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_s, name)

    # Random
    for r in range(2):
        a_r = _make_random_positive(init_n, rng)
        sc = _score(a_r)
        print(f"  random_{r}: score={sc:.10f}")
        try_update(a_r, f"random_{r}")

    if best_seq is None:
        best_seq = _make_chirp(4096, 1.0, 0.3)
        best_score_val = _score(best_seq)
        best_n = len(best_seq)

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: LP-based optimization ===
    print(f"\n=== Phase 2: LP optimization (t={time.monotonic()-t_start:.1f}s) ===")
    lp_deadline = t_start + 500.0
    if time.monotonic() < lp_deadline:
        a_lp, sc_lp = _lp_optimize(best_seq, lp_deadline, max_iters=300,
                                    K_size=80, reserve=30.0)
        try_update(a_lp, "LP_main")

    # Try LP with larger active set
    if best_seq is not None and time.monotonic() < deadline - 600:
        a_lp2, sc_lp2 = _lp_optimize(best_seq, deadline - 600, max_iters=150,
                                      K_size=200, reserve=30.0)
        try_update(a_lp2, "LP_K200")

    # === Phase 3: L-BFGS with adaptive smoothing ===
    print(f"\n=== Phase 3: L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        a_lbfgs = _adaptive_lbfgs(best_seq, deadline, reserve_time=400.0)
        try_update(a_lbfgs, "lbfgs_main")

    # === Phase 4: Exact subgradient (threshold-based) ===
    print(f"\n=== Phase 4: Subgradient threshold (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_sg, sc_sg = _subgrad_threshold(best_seq, deadline - 400,
                                          threshold_frac=0.02, max_steps=80000,
                                          lr_init=0.04, reserve=20.0)
        try_update(a_sg, "sg_thresh")

    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg2, sc_sg2 = _subgrad_threshold(best_seq, deadline - 300,
                                            threshold_frac=0.005, max_steps=60000,
                                            lr_init=0.03, reserve=20.0)
        try_update(a_sg2, "sg_thresh_tight")

    # Projected subgradient with top-k
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_psg, sc_psg = _projected_subgrad(best_seq, deadline - 250,
                                            max_steps=40000, lr_init=0.08,
                                            k_active=5, reserve=20.0)
        try_update(a_psg, "proj_sg_k5")

    # === Phase 5: KKT equalization ===
    print(f"\n=== Phase 5: KKT (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_kkt, sc_kkt = _kkt_equalize(best_seq, deadline - 250, k_active=5,
                                       max_iter=200, reserve=20.0)
        try_update(a_kkt, "kkt_eq")

    # === Phase 6: More L-BFGS ===
    print(f"\n=== Phase 6: L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_lb2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=100.0)
        try_update(a_lb2, "lbfgs_r2")

    # === Phase 7: Basin hopping ===
    print(f"\n=== Phase 7: Basin hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 100, rng,
                                    n_hops=5, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Multi-length exploration ===
    print(f"\n=== Phase 8: Multi-length (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        lengths_to_try = []
        bn = best_n
        for ln in [2 * bn, bn // 2, bn + 2048, bn - 2048]:
            if 1024 <= ln <= 32768 and ln != bn:
                lengths_to_try.append(ln)
        seen = set()
        unique_lengths = []
        for ln in lengths_to_try:
            if ln not in seen:
                seen.add(ln)
                unique_lengths.append(ln)

        for target_n in unique_lengths[:3]:
            if time.monotonic() > deadline - 50:
                break
            print(f"  Trying n={target_n} (currently best n={best_n})")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")

            if time.monotonic() < deadline - 40:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"n_{target_n}")

            if time.monotonic() < deadline - 30:
                a_lpN, _ = _lp_optimize(a_interp, deadline - 20, max_iters=50,
                                         K_size=60, reserve=15.0)
                try_update(a_lpN, f"lp_n_{target_n}")

    # === Phase 9: Final aggressive polishing ===
    print(f"\n=== Phase 9: Final polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(8):
            if time.monotonic() > deadline - 3:
                break

            # Very tight L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-8
            remaining = max(2, deadline - 3 - time.monotonic())
            max_iter = min(5000, max(200, int(remaining / 0.004)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 3:
                a_s, _ = _subgrad_threshold(best_seq, deadline - 3,
                                             threshold_frac=0.001, max_steps=20000,
                                             lr_init=0.01, reserve=2.0)
                try_update(a_s, f"final_sg_{rnd}")

            # KKT
            if time.monotonic() < deadline - 3:
                a_k, _ = _kkt_equalize(best_seq, deadline - 3, k_active=3,
                                        max_iter=50, reserve=2.0)
                try_update(a_k, f"final_kkt_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
