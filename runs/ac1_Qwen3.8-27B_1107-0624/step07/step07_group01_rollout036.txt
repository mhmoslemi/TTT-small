def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive multi-scale L-BFGS with log-sum-exp smoothing
2. Polyak subgradient method with adaptive active sets
3. Autocorrelation flattening (variance minimization) via L-BFGS
4. Fourier-domain diversification
5. Simplex projection subgradient
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    try:
        result = minimize(_smoobj, z0, jac=_smograd, args=(beta, n),
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxcor': 80,
                                   'maxls': 100})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = v.copy()
    # Subtract the largest tau such that sum(max(v - tau, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, len(v) + 1) > (cv[-1] - target_sum))[0][-1]
    tau = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def _subgrad_polyak(a_init, deadline, k_active=1, max_steps=50000, f_lower=1.0):
    """Subgradient method with Polyak step size on the simplex."""
    n = len(a_init)
    a = _project_simplex(a_init.copy(), 1.0)
    best_a = a.copy()
    best_sc = _score(a)
    stall = 0
    f_best_lower = max(f_lower, best_sc - 0.05)  # optimistic lower bound

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        n_active = len(c)  # 2n-1
        mean_c = c_max  # we'll use f(a) directly
        
        # Find active set
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active

        # Compute subgradient: g = 2n * sum_k w[k] * dc[k]/da
        # dc[k]/da[i] = 2*a[k-i] (with bounds)
        # So g[i] = 4n * sum_k w[k] * a[k-i]
        # = 4n * cross_corr(w, a)[i]
        N = _next_pow2(3 * n - 2)
        W_ft = np.fft.rfft(w, N)
        A_ft = np.fft.rfft(a, N)
        xc = np.fft.irfft(np.conj(W_ft) * A_ft, N)
        g = 4.0 * n * xc[:n]
        
        # Normalize by (sum a)^2 = 1 (since we're on the simplex)
        # f(a) = 2n * c_max (since sum a = 1)
        f_cur = 2.0 * n * c_max
        
        # Polyak step size
        gnorm2 = float(np.dot(g, g))
        if gnorm2 < 1e-30:
            break
        alpha = (f_cur - f_best_lower) / gnorm2
        alpha = min(alpha, 0.5)  # cap step size
        
        # Take step and project
        a_new = a - alpha * g
        a_new = _project_simplex(a_new, 1.0)
        
        sc_new = _score(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            f_best_lower = max(f_lower, best_sc - 0.02)
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Reduce optimism of lower bound
                f_best_lower = max(f_lower, best_sc - 0.01)
                stall = 0
            if stall > 5000:
                break
        
        a = a_new

    return best_a, best_sc


def _flatten_obj(z, n):
    """Objective: variance of autocorrelation (in log-space)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    mean_c = c.mean()
    return float(np.sum((c - mean_c) ** 2))


def _flatten_grad(z, n):
    """Gradient of variance of autocorrelation w.r.t. log-space parameters."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    mean_c = c.mean()
    e = c - mean_c
    # d(sum e^2)/da[i] = 4 * cross_corr(c, a)[i] (since gradient of sum c^2)
    # Actually: d(sum e^2)/da[i] = 2 * sum_k e[k] * dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] (bounds)
    # = 4 * sum_k e[k] * a[k-i] = 4 * cross_corr(e, a)[i]
    N = _next_pow2(3 * n - 2)
    E_ft = np.fft.rfft(e, N)
    A_ft = np.fft.rfft(a, N)
    xc = np.fft.irfft(np.conj(E_ft) * A_ft, N)
    grad_a = 4.0 * xc[:n]
    # Chain rule: d/dz = d/da * da/dz = grad_a * diag(a) - grad_a * a * sum(...)
    # da/dz[i] = a[i] * (delta[i,j] - a[j])
    # d/dz_j = sum_i grad_a[i] * a[i] * (delta[i,j] - a[j])
    #         = grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
    ga = float(np.dot(grad_a, a))
    return a * (grad_a - ga)


def _lbfgs_flatten(a_init, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    try:
        result = minimize(_flatten_obj, z0, jac=_flatten_grad, args=(n,),
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 60})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _fourier_perturb(a, rng, n_perturb=5, amp=0.05):
    """Perturb a in the Fourier domain."""
    n = len(a)
    A = np.fft.rfft(a)
    # Perturb random frequency bins
    n_freqs = len(A)
    for _ in range(n_perturb):
        k = int(rng.integers(1, n_freqs - 1))
        dA = amp * rng.standard_normal() + 0j * amp * rng.standard_normal()
        A[k] += dA
        if k > 0 and k < n_freqs - 1:
            A[-k] = np.conj(A[k])  # maintain real-valued... wait, A is rfft
    # Actually, rfft doesn't have the conjugate symmetry stored explicitly
    # Let me use full fft instead
    A_full = np.fft.fft(a)
    for _ in range(n_perturb):
        k = int(rng.integers(1, n - 1))
        dA = amp * (rng.standard_normal() + 1j * rng.standard_normal())
        A_full[k] += dA
        A_full[n - k] += np.conj(dA)
    a_new = np.fft.ifft(A_full).real
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # More aggressive beta schedule going lower
    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              0.00005, 0.00002, 0.00001, 0.000005,
                              0.000002, 0.000001, 0.0000005, 0.0000002,
                              0.0000001, 0.00000005])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        # Scale iterations based on n
        scale = max(1.0, (n / 4096.0) ** 1.3)
        # Give more iterations to tighter betas (they need more work per iteration)
        beta_factor = 1.0 + 2.0 * (i / len(betas))
        max_iter = min(20000, max(200, int(remaining / (0.003 * scale * beta_factor * 15))))

            print(f"    beta={beta:.2e}: score={sc:.10f} (improved)")
        else:
            print(f"    beta={beta:.2e}: no improvement (score={sc:.10f})")

        if current_sc < 1.5005:
            break

    return current_a


def _kkt_refine(a, deadline, k_active=3, max_iter=100):
    """Refine using KKT-inspired iterative method with line search."""
    n = len(a)
    a = _project_simplex(a.copy(), 1.0)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        w = np.zeros(len(c))
        w[idx] = 1.0 / k_active

        # Gradient direction
        N = _next_pow2(3 * n - 2)
        W_ft = np.fft.rfft(w, N)
        A_ft = np.fft.rfft(a, N)
        xc = np.fft.irfft(np.conj(W_ft) * A_ft, N)
        g = 4.0 * n * xc[:n]
        g = g - g.mean()  # center to maintain sum=1 after step

        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a - alpha * g / (n * np.linalg.norm(g) + 1e-30)
            a_try = _project_simplex(a_try, 1.0)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            break

    return best_a, best_sc

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()

    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _project_simplex(v, target_sum=1.0):
    v = v.copy()
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho_arr = np.arange(1, len(v) + 1)
    cond = v_sorted * rho_arr > (cv[-1] - target_sum)
    if not np.any(cond):
        return np.full(len(v), target_sum / len(v))
    rho = int(np.nonzero(cond)[0][-1])
    tau = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)
