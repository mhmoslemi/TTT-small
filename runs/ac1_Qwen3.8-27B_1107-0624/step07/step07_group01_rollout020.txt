    """Gradient using top-k active set (exact subgradient of max)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:k_active]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k_active
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W), c


def _adam_lse_optimize(a_init, deadline, reserve=30.0):
    """Adam optimizer with LSE smoothing and adaptive beta cooling."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_c0 < 1e-30:
        return best_a

    betas = max_c0 * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                                0.001, 0.0005, 0.0002, 0.0001, 5e-5,
                                2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7,
                                2e-7, 1e-7, 5e-8])

    for bi, beta in enumerate(betas):
        if time.monotonic() + reserve > deadline:
            break

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        lr = 0.5 / np.sqrt(n)
        beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
        stall = 0
        best_local_sc = best_sc
        best_local_a = best_a.copy()

        scale_factor = (n / 4096.0) ** 0.7
        max_iters = int(max(200, (deadline - reserve - time.monotonic()) / (0.002 * scale_factor)))
        max_iters = min(max_iters, 50000)

        for it in range(max_iters):
            if time.monotonic() + reserve > deadline:
                break
            t_adam += 1
            grad = _lse_smooth_grad(z, beta, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                break

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z - z.mean()
            z = np.clip(z, -100.0, 100.0)

            if it % 200 == 199:
                a_check = _softmax(z)
                sc = _score(a_check)
                if sc < best_local_sc - 1e-13:
                    best_local_sc = sc
                    best_local_a = a_check.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 50:
                        break

        a_local = _softmax(z)
        sc_local = _score(a_local)
        if sc_local < best_sc:
            best_sc = sc_local
            best_a = a_local.copy()
            print(f"    Adam beta={beta:.2e}: score={best_sc:.10f} (n={n})")

    return best_a


def _nesterov_direct(a_init, deadline, reserve=30.0, k_active=5):
    """Nesterov-accelerated subgradient in direct (non-log) space."""
    n = len(a_init)
    a = a_init.copy()
    a_y = a.copy()
    best_a = a.copy()
    best_sc = _score(a_init)
    lr0 = 0.8 / np.sqrt(n)
    mu = 1.0

    for t in range(1, 2000000):
        if time.monotonic() + reserve > deadline:
            break
        alpha_t = 2.0 / (t + 1.0 + mu)
        a_new = a_y + (alpha_t - 1.0) * (a_y - a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        c = _full_convolve(a_new, a_new)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active
        a_rev = a_new[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break
        lr_t = lr0 / np.sqrt(t)
        a_y = a_new - lr_t * g / gnorm
        a_y = np.maximum(a_y, 0.0)
        s = a_y.sum()
        if s < 1e-15:
            a_y = a_new.copy()
        else:
            a_y = a_y / s

        if t % 500 == 0:
            sc = _score(a_y)
            if sc < best_sc:
                best_sc = sc
                best_a = a_y.copy()
        a = a_new

    sc_final = _score(a_y)
    if sc_final < best_sc:
        best_a = a_y.copy()
    return best_a


def _mirror_polish(a_init, deadline, reserve=10.0, k_active=3):
    """Mirror descent polishing in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.02
    stall = 0

    for i in range(1000000):
        if time.monotonic() + reserve > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        step = lr / (a * n + 1e-10)
        z = z - step * grad
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 100 == 99:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                lr = min(lr * 1.05, 0.1)
            else:
                stall += 1
                if stall > 40:
                    lr *= 0.7
                    if lr < 1e-12:
                        break
                else:
                    lr *= 0.97

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
    return best_a


def _kkt_equality(a_init, deadline, reserve=30.0, k_active=5):
    """Try to equalize active convolution values using Newton iteration."""
    n = a_init.size
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(500):
        if time.monotonic() + reserve > deadline:
            break
        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        spread = active_vals.max() - active_vals.min()
        if spread < 1e-18:
            break

        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()

        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a - alpha * g / (n * 4.0)
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
            alpha *= 0.6
        if not improved:
            break

    return best_a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands, eps=0.2):
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=30.0):
    """Basin hopping with diverse perturbation types."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_dl = time.monotonic() + min(time_per_hop * 0.8, deadline - reserve - time.monotonic())
        pert_type = h % 4

        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            t_arr = np.arange(n)
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 16)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 4 + h
            block_sz = n // n_blocks
            for _ in range(n_blocks):
                start = int(rng.integers(0, n - block_sz))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            a_pert = best_a + 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()

        a_opt = _adam_lse_optimize(a_pert, hop_dl, reserve=5.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Diverse initialization (parent + structured + random)
    Phase 2: Adam+LSE main optimization
    Phase 3: Nesterov subgradient in direct space
    Phase 4: Mirror descent polishing
    Phase 5: KKT active-set equalization
    Phase 6: Basin hopping diversification
    Phase 7: Multi-length exploration
    Phase 8: Final aggressive polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
                try_update(parent, "parent")
    except Exception:
        pass

    n_init = 16384
    inits = []
    if parent is not None:
        inits.append(("parent", parent))

    n_struct = 4096
    struct_list = [
        ("chirp1", _make_chirp(n_struct, 1.0, 0.3)),
        ("chirp2", _make_chirp(n_struct, 2.0, 0.25)),
        ("chirp3", _make_chirp(n_struct, 3.0, 0.2)),
        ("mb_13", _make_multiband(n_struct, [1, 3], 0.25)),
        ("mb_12", _make_multiband(n_struct, [1, 2], 0.3)),
        ("mb_25", _make_multiband(n_struct, [2, 5], 0.2)),
    ]
    for name, a_s in struct_list:
        inits.append((name, a_s))
        try_update(a_s, name)

    for r in range(3):
        a_r = _make_random_positive(n_struct, rng)
        inits.append((f"rand{r}", a_r))
        try_update(a_r, f"rand{r}")

    if best_seq is None:
        best_seq = inits[0][1].copy()
        best_score_val = _score(best_seq)
    print(f"  Phase 1 done, best={best_score_val:.10f} (n={len(best_seq)})")

    # === Phase 2: Main Adam+LSE on best ===
    print(f"=== Phase 2: Adam+LSE main (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 600:
        a_p2 = _adam_lse_optimize(best_seq, deadline, reserve=600.0)
        try_update(a_p2, "adam_main")

    # Also run Adam on parent separately if different
    if parent is not None and len(parent) != len(best_seq) and time.monotonic() < deadline - 500:
        a_p2b = _adam_lse_optimize(parent, deadline, reserve=500.0)
        try_update(a_p2b, "adam_parent")

    # === Phase 3: Nesterov subgradient ===
    print(f"=== Phase 3: Nesterov subgrad (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_ng = _nesterov_direct(best_seq, deadline - 400, reserve=30.0, k_active=5)
        try_update(a_ng, "nesterov_k5")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_ng2 = _nesterov_direct(best_seq, deadline - 300, reserve=30.0, k_active=3)
        try_update(a_ng2, "nesterov_k3")

    # === Phase 4: Mirror descent polish ===
    print(f"=== Phase 4: Mirror polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_mp = _mirror_polish(best_seq, deadline - 200, reserve=10.0, k_active=3)
        try_update(a_mp, "mirror_k3")

    # === Phase 5: KKT refinement ===
    print(f"=== Phase 5: KKT (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_kkt = _kkt_equality(best_seq, deadline - 150, reserve=30.0, k_active=4)
        try_update(a_kkt, "kkt_k4")

    # === Phase 6: Second Adam round ===
    print(f"=== Phase 6: Adam round 2 (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_a2 = _adam_lse_optimize(best_seq, deadline, reserve=150.0)
        try_update(a_a2, "adam_r2")

    # === Phase 7: Basin hopping ===
    print(f"=== Phase 7: Basin hop (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_hop = _basin_hop(best_seq, deadline - 100, rng, n_hops=4, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Multi-length exploration ===
    print(f"=== Phase 8: Length exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        cur_n = len(best_seq)
        lengths_to_try = [8192, 32768, 10240]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 40:
                break
            if target_n == cur_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interp score: {sc_interp:.10f}")
            if time.monotonic() < deadline - 30:
                a_opt = _adam_lse_optimize(a_interp, deadline, reserve=30.0)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 9: Final aggressive polishing ===
    print(f"=== Phase 9: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        for rnd in range(4):
            if time.monotonic() > deadline - 5:
                break
            a_f1 = _mirror_polish(best_seq, deadline, reserve=5.0, k_active=3)
            try_update(a_f1, f"final_mp_{rnd}")
            if time.monotonic() > deadline - 5:
                break
            a_f2 = _kkt_equality(best_seq, deadline, reserve=5.0, k_active=3)
            try_update(a_f2, f"final_kkt_{rnd}")
            if time.monotonic() > deadline - 5:
                break
            a_f3 = _nesterov_direct(best_seq, deadline, reserve=5.0, k_active=3)
            try_update(a_f3, f"final_ng_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
