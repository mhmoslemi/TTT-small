"""
Adam-based optimizer with log-sum-exp smoothing and KKT pressure equalization
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_and_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _adam_optimize(z_init, beta_schedule, n, deadline, min_remaining=30.0):
    """Adam optimizer on log-space variables with log-sum-exp smoothing."""
    z = z_init.copy()
    n_z = len(z)
    m = np.zeros(n_z)  # first moment
    v = np.zeros(n_z)  # second moment
    t = 0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_z = z.copy()
    best_obj = _smoobj_and_grad(z, beta_schedule[0], n)[0]
    
    for beta in beta_schedule:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Number of iterations at this beta
        # Each iteration takes roughly O(n log n) for the FFT
        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iters = min(3000, max(100, int(remaining / est_time_per_iter * 0.7)))
        
        for it in range(max_iters):
            if time.monotonic() + min_remaining > deadline:
                break
            
            t += 1
            obj, grad = _smoobj_and_grad(z, beta, n)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()  # re-center for numerical stability
            
            # Periodic check
            if it % 50 == 0 or it == max_iters - 1:
                obj_check = _smoobj_and_grad(z, beta, n)[0]
                if obj_check < best_obj - 1e-12:
                    best_obj = obj_check
                    best_z = z.copy()
        
        # Reset momentum when beta changes
        m = np.zeros(n_z)
        v = np.zeros(n_z)
        t = 0
        # Adjust learning rate for new beta
        lr = 0.01 * (beta / beta_schedule[0]) ** 0.3
    
    return best_z


def _lbfgs_refine(a_init, deadline, min_time_remaining=10.0):

    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 8))))
        
        try:
            def obj_func(z2):
                return _smoobj_and_grad(z2, beta, n)[0]
            
            def grad_func(z2):
                return _smoobj_and_grad(z2, beta, n)[1]
            
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _kkt_pressure_equalize(a, deadline, n_iters=300, gamma=0.3):
    """KKT-motivated pressure equalization."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 2:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 5
        
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 10 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        # Flat top with smooth edges
        t = np.linspace(0, 1, n)
        edge = 0.1
        a = np.ones(n)
        mask_left = t < edge
        mask_right = t > 1 - edge
        a[mask_left] = np.sin(np.pi * t[mask_left] / (2 * edge)) ** 2
        a[mask_right] = np.sin(np.pi * (t[mask_right] - (1-edge)) / (2 * edge)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 0.5 * np.exp(-20*(t-0.3)**2) + 0.5 * np.exp(-20*(t-0.7)**2) + 0.05
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 4 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    n_half = len(z_hat)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, min(n_half - 1, n // 2)))
        delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
        phase = rng.uniform(0, 2 * np.pi)
        z_hat[freq] += delta * np.exp(1j * phase)
        if freq < n_half - 1:
            z_hat[n_half - 1 - freq] = np.conj(z_hat[n_half - 1 + freq]) if (n_half - 1 + freq) < n_half else z_hat[freq].conj()
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

"""
Adam-based optimizer with log-sum-exp smoothing and KKT pressure equalization
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key improvements:
1. Adam optimizer with aggressive beta continuation (faster than L-BFGS for this problem)
2. Warm-started momentum across beta levels
3. Multi-n exploration with smart resampling
4. KKT pressure equalization for fine convergence
5. Diverse restarts and coordinate descent for final polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_and_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700.0, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _adam_run(z_init, beta, n, deadline, min_remaining=30.0, lr_init=0.005, max_iters=5000):
    """Run Adam optimizer at a fixed beta level."""
    z = z_init.copy()
    nz = len(z)
    m = np.zeros(nz)
    v = np.zeros(nz)
    t = 0
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    obj0, _ = _smoobj_and_grad(z, beta, n)
    best_z = z.copy()
    best_obj = obj0
    patience = 0
    best_patience_obj = best_obj

    est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
    max_iters = min(max_iters, max(100, int((deadline - min_remaining - time.monotonic()) / est_time_per_iter)))

    for it in range(max_iters):
        if time.monotonic() + min_remaining > deadline:
            break

        t += 1
        obj, grad = _smoobj_and_grad(z, beta, n)

        if not np.isfinite(obj) or not np.isfinite(grad).all():
            break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step_norm = np.linalg.norm(step)
        if step_norm > 5.0:
            step = step * 5.0 / step_norm

        z = z - step
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

                best_z = z.copy()
                patience = 0
            else:
                patience += 1
                if patience > 8:
                    lr *= 0.8
                    if patience > 16:
                        break

    return best_z

                return _smoobj_and_grad(z2, beta, n)[0]

            def grad_func(z2):
                return _smoobj_and_grad(z2, beta, n)[1]

                obj_func, z, jac=grad_func,
                method='L-BFGS-B',

            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _kkt_pressure_equalize(a, deadline, n_iters=300, gamma=0.25):
    """KKT-motivated pressure equalization for fine convergence."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 5

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score_normalized(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        edge = 0.08
        a = np.ones(n)
        mask_left = t < edge
        mask_right = t > 1.0 - edge
        a[mask_left] = np.sin(np.pi * t[mask_left] / (2.0 * edge)) ** 2
        a[mask_right] = np.sin(np.pi * (t[mask_right] - (1.0 - edge)) / (2.0 * edge)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 0.5 * np.exp(-25.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-25.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_mixture":
        t = np.linspace(0, 1, n)
        a = (np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.5) ** 2) +
             np.exp(-15.0 * (t - 0.75) ** 2)) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    n_half = len(z_hat)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, min(n_half - 1, n // 3)))
        delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
        phase = rng.uniform(0.0, 2.0 * np.pi)
        z_hat[freq] += delta * np.exp(1j * phase)
        mirror = n_half - 1 - freq
        if 0 < mirror < n_half:
            z_hat[mirror] += delta * np.exp(-1j * phase)
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _coordinate_descent_polish(a, deadline, n_rounds=150):
    """Targeted coordinate descent on high-contribution coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        active = c >= (c_max - 0.015 * c_max)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(15, n // 80))
        top_idx = np.argsort(contrib)[-n_try:]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue

            for factor in [0.75, 0.85, 0.93, 1.08]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < c_max - 1e-15:
                    c_max = float(c_try)
                    improved = True
                    break
                a[i] = old_val

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _quick_optimize_at_n(a_init, deadline, min_remaining=20.0):
    """Quick Adam + KKT optimization at a given length."""
    n = len(a_init)
    a_adam, sc_adam = _adam_continuation(a_init, deadline, min_remaining=min_remaining + 20)

    a_kkt, _ = _kkt_pressure_equalize(a_adam, min(deadline - min_remaining - 5, time.monotonic() + 40),
                                      n_iters=150, gamma=0.2)
    sc_kkt = _score(a_kkt)

    if sc_kkt < sc_adam:
        return a_kkt, sc_kkt
    return a_adam, sc_adam


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy Adam optimizer with KKT polish for minimizing the autocorrelation score.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None or len(p) == 0:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={parent_sc:.10f}")
                _update(parent, parent_sc)
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent available, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Adam on parent (main workhorse) ---
    print(f"\n[Phase 1] Adam continuation on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a1, sc1 = _adam_continuation(parent, deadline, min_remaining=700)
    _update(a1, sc1)

    # --- Phase 2: KKT pressure equalization ---
    print(f"[Phase 2] KKT pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, _ = _kkt_pressure_equalize(best_seq, min(deadline - 400, time.monotonic() + 100),
                                       n_iters=400, gamma=0.2)
        sc2 = _score(a2)
        _update(a2, sc2)

    # --- Phase 3: L-BFGS fine-tuning ---
    print(f"[Phase 3] L-BFGS fine-tuning (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3 = _lbfgs_refine(best_seq, deadline, min_time_remaining=300)
        sc3 = _score(a3)
        _update(a3, sc3)

    # --- Phase 4: Multi-n exploration ---
    print(f"[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 5120, 6144, 7168, 7680, 8192, 8704, 9216, 10240, 11264, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(key=lambda x: -abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - 120 - time.monotonic()
        if remaining < 40:
            break

        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, parent)
        a_init = np.maximum(a_init, 0.0)
        s_init = a_init.sum()
        if s_init > 1e-10:
            a_init = a_init / s_init
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(20, (deadline - 120 - time.monotonic()) * 0.35)
        a_opt, sc_opt = _quick_optimize_at_n(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 80:
            a_lb = _lbfgs_refine(a_opt, deadline, min_time_remaining=60)
            _update(a_lb, _score(a_lb))

    # --- Phase 5: Diverse restarts from best ---
    print(f"[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        profiles = ["tapered", "flat_top", "bimodal", "chirp", "gaussian_mixture"]

        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 40:
                break

            weight = 0.5 + 0.1 * i
            a_blend = weight * best_seq + (1.0 - weight) * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0.0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=0.015)

            a_r, sc_r = _quick_optimize_at_n(a_pert, deadline, min_remaining=40)
            _update(a_r, sc_r)

    # --- Phase 6: Final KKT + coordinate descent ---
    print(f"[Phase 6] Final KKT + coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_kkt_fin, _ = _kkt_pressure_equalize(best_seq, min(deadline - 8, time.monotonic() + 50),
                                              n_iters=300, gamma=0.12)
        sc_fin = _score(a_kkt_fin)
        _update(a_kkt_fin, sc_fin)

        if time.monotonic() < deadline - 4:
            a_cd, _ = _coordinate_descent_polish(a_kkt_fin, deadline - 3, n_rounds=100)
            sc_cd = _score(a_cd)
            _update(a_cd, sc_cd)

    # --- Phase 7: Final Adam burst ---
    print(f"[Phase 7] Final Adam burst (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _adam_continuation(best_seq, deadline, min_remaining=1)
        _update(a_fin, sc_fin)

        if time.monotonic() < deadline - 1:
            a_kkt_last, _ = _kkt_pressure_equalize(a_fin, deadline - 1, n_iters=100, gamma=0.1)
            _update(a_kkt_last, _score(a_kkt_last))

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
