   Wait, but for uniform, ALL positions near the center have values close to 1/n. So the "active set" (positions where b is close to the max) is large. The perturbation needs to reduce the values at ALL these positions simultaneously.

   This is the key challenge: we need a perturbation that simultaneously reduces b_k for all k near the center.

   The gradient of b_k with respect to a_i is 2*a_{k-i} (as computed earlier). For the uniform sequence, this is 2/n for all valid (i,k) pairs. So the gradient of the max (or log-sum-exp) is approximately uniform, meaning we should reduce all a_i equally, which contradicts the constraint sum(a) = 1.

   This is why the problem is hard: the "descent direction" for the peak is to make all a_i smaller, but we can't do that while maintaining sum(a) = 1.

   The resolution is that we need to exploit the second-order effects. By making some a_i slightly larger and others slightly smaller (in a correlated way), we can reduce the peak while maintaining the sum.

   This is exactly what the L-BFGS and active-set methods do: they find the optimal perturbation direction.

   4. **Multi-resolution**: Solve at n/4, n/2, n, and 2n, using solutions as initialization for the next scale.

   5. **Different optimization landscapes**: Try both log-sum-exp and power-sum smoothing.

   One more thought: the **power-sum** approximation might work better than log-sum-exp for this problem. The p-th power sum:
   F_p(a) = (sum_k b_k^p)^(1/p)
   
   For large p, this approximates max(b). The gradient is:
   dF_p/da_i = (1/p) * (sum b_k^p)^{(1/p)-1} * sum_k p*b_k^{p-1} * db_k/da_i
   = (sum b_k^p)^{(1/p)-1} * sum_k b_k^{p-1} * 2*a_{k-i}
   
   This is computable via the same Hankel matvec with weights w_k = b_k^{p-1}.

   The advantage of power-sum over log-sum-exp:
   - No temperature parameter (p plays the role of 1/beta)
   - The gradient naturally weights positions by b_k^{p-1}, which for large p focuses on the peaks
   - No underflow/overflow issues (unlike exp)
   - Smoother transition as p increases

   Actually, for the power-sum, there's a subtlety: we need b_k > 0 for all k (which is true for a > 0), and p should be large enough. If p is too small, the approximation is poor; if too large, numerical issues.

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses power-sum smoothing, log-sum-exp smoothing, weighted active-set descent,
multi-resolution refinement, and adaptive schedules.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _higher_order_derivative_correction(a, beta, n):
    """Compute correction term for second-order effects in the smoothed objective."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 0.0
    w = e / s
    # Second-order correction: variance of weighted convolution
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    # Variance term
    var = float(np.dot(w, (c - W)**2))
    return var

def _smoobj_lse(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd_lse(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _smoobj_ps(z, p, n):
    """Power-sum smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    ps = np.sum(c_pos ** p)
    if ps < 1e-300 or not np.isfinite(ps):
        return 2.0 * n * float(c.max())
    return 2.0 * n * ps ** (1.0 / p)

def _smograd_ps(z, p, n):
    """Gradient of power-sum smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** (p - 1)
    W = float(np.sum(w * c))
    if W < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / W
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 4.0 * n * a * h

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _active_set_polish_weighted(a_init, deadline, tol_frac_start=0.01,
                                  tol_frac_end=0.00005, lr_init=0.05):
    """Active-set gradient descent with exponential weighting."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        tol = tol_frac * c_max
        # Exponential weighting instead of binary
        weights = np.exp(-(c_max - c) / max(tol, 1e-30))
        w_sum = weights.sum()
        if w_sum < 1e-300:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
            weights = active.astype(float)
        else:
            weights = weights / w_sum
        
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-15:
            break
        
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.8)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end * 2:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)
    
    return best_a, _score(best_a)

def _run_smoothed_lbfgs(a_init, deadline, min_time_remaining=30.0, method='lse'):
    """Run L-BFGS with adaptive smoothing (log-sum-exp or power-sum)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    if method == 'lse':
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                       0.03, 0.015, 0.0075, 0.004,
                                       0.002, 0.001, 0.0005, 0.0002,
                                       0.0001, 0.00005, 0.00002, 0.00001])
        for i, beta in enumerate(betas):
            if time.monotonic() + min_time_remaining > deadline:
                break
            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining < 5:
                break
            
            est_time = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(30, int(remaining / (est_time * 10))))
            
            class Prob:
                def __call__(self, z_in):
                    return _smoobj_lse(z_in, beta, n)
                def grad(self, z_in):
                    return _smograd_lse(z_in, beta, n)
            
            prob = Prob()
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 80, 'maxls': 50})
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass
    else:  # power-sum
        ps_values = [10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120]
        for p in ps_values:
            if time.monotonic() + min_time_remaining > deadline:
                break
            remaining = deadline - min_time_remaining - time.monotonic()
            if remaining < 5:
                break
            
            est_time = 0.003 * max(1.0, n / 2048.0)
            max_iter = min(5000, max(30, int(remaining / (est_time * 10))))
            
            class ProbPS:
                def __call__(self, z_in):
                    return _smoobj_ps(z_in, p, n)
                def grad(self, z_in):
                    return _smograd_ps(z_in, p, n)
            
            prob = ProbPS()
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 80, 'maxls': 50})
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            except Exception:
                pass
    
    return current_a

def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to improve score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Try different fractions to zero out
    fractions = [0.001, 0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_sorted_idx = np.argsort(a)
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Taper the edges
        taper = max(1, n // 10)
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n-1-i] *= (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):

        amp = 0.02 * rng.uniform(0.5, 3.0)

        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

def _peak_aware_perturb(a, rng):

    z = z - 0.05 * h_norm + 0.03 * rng.standard_normal(n)
    z = z - z.mean()
