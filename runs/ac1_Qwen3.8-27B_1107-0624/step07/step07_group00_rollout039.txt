"""
Hybrid LP-guided optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast greedy LP directions, smoothed gradient descent with adaptive
temperature, animated active-set focusing, and 1D minimax coordinate descent
polishing. Multi-phase strategy with random restarts on n=4096/8192.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2n*max(c)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_grad(a, c, T):
    """Compute smoothed gradient of 2n*max(c) w.r.t. a."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _active_grad(a, c, top_k=5, decay=0.7):
    """Weighted active-set gradient."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _fast_lp_direction(f, M, max_iters=200000):
    """
    Fast greedy LP: find b >= 0, conv(f,b) <= M, maximizing sum(b).
    Returns normalized direction (sum=1).
    """
    n = len(f)
    b = np.ones(n)
    h = _fft_convolve(f, b)
    h = np.maximum(h, 0)
    
    for _ in range(max_iters):
        k_star = int(np.argmax(h))
        if h[k_star] <= M * (1.0 + 1e-12):
            break
        # Find j that most contributes to h[k_star]
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_lo > j_hi:
            break
        contrib = f[j_lo:j_hi+1] * b[j_lo:j_hi+1]
        j_rel = int(np.argmax(contrib))
        j_star = j_lo + j_rel
        f_val = f[k_star - j_star]
        if f_val < 1e-30:
            break
        delta = min(b[j_star], (h[k_star] - M) / (f_val + 1e-30))
        if delta < 1e-15:
            break
        b[j_star] -= delta
        # Update h: h[k] -= f[k - j_star] * delta for k in [j_star, j_star+n-1]
        k_start = j_star
        k_end = min(j_star + n, 2 * n - 1)
        f_slice = f[:k_end - k_start]
        h[k_start:k_end] -= f_slice * delta
        h = np.maximum(h, 0)
    
    s = b.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b / s


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=3, decay=0.6,
                  eval_interval=200, clip_norm=0.8, use_backtrack=True,
                  plateau_frac=0.3):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / (1.0 - plateau_frac)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if T < T_start * 0.01:
            g = _active_grad(a, c, active_k, decay)
        else:
            g = _smoothed_grad(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 50 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 4 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_optimize(f_init, deadline, max_rounds=200):
    """
    LP-guided optimization: repeatedly solve LP for direction, take step.
    """
    n = len(f_init)
    f = f_init.copy() / f_init.sum()
    best_a = f.copy()
    best_score = _score_norm(f)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(f, f)
        M = float(c.max())
        
        # Get LP direction
        g = _fast_lp_direction(f, M)
        g = g / g.sum()
        
        # Line search: try several step sizes
        best_t = 0.0
        best_sc = best_score
        for t in [0.8, 0.5, 0.3, 0.15, 0.08, 0.04]:
            f_try = (1.0 - t) * f + t * g
            f_try = np.maximum(f_try, 0)
            s = f_try.sum()
            if s < 1e-15:
                continue
            f_try = f_try / s
            sc = _score_norm(f_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 1e-15:
                f = f / s
            if best_sc < best_score:
                best_score = best_sc
                best_a = f.copy()
        else:
            # Try small step
            f_try = 0.98 * f + 0.02 * g
            f_try = np.maximum(f_try, 0)
            s = f_try.sum()
            if s > 1e-15:
                f_try = f_try / s
            sc = _score_norm(f_try)
            if sc < best_score:
                f = f_try
                best_score = sc
                best_a = f.copy()
    
    return best_a, best_score


def _cd_polish_1d(a, deadline, max_rounds=100):
    """
    Coordinate descent with 1D minimax optimization.
    For each coordinate j, find the optimal value that minimizes max(conv).
    """
    n = len(a)
    f = a.copy() / a.sum()
    best_a = f.copy()
    best_score = _score_norm(f)
    c = _fft_convolve(f, f)
    c_max = float(c.max())
    tol = c_max * 1.005
    active = np.where(c > tol)[0]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Shuffle active positions for diversity
        order = np.random.default_rng(rnd * 7919 + 13).permutation(len(active))
        for idx in order:
            k = active[idx]
            # Determine which j's affect c[k]
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo > j_hi:
                continue
            # Try a few candidates
            for j in range(j_lo, j_hi + 1):
                # 1D minimax: find optimal f[j]
                # c_new[k'] = c[k'] + 2*delta*f[k'-j] + delta^2*[k'=2j]
                # We want to minimize max over k' of c_new[k']
                # Use bisection on delta
                old_val = f[j]
                # The range of delta: -old_val to +inf (but we cap it)
                # Try a grid of multipliers
                for mult in [0.1, 0.3, 0.5, 0.7, 1.5, 2.0, 3.0]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    new_val = min(new_val, 10.0)  # cap
                    delta = new_val - old_val
                    if abs(delta) < 1e-15:
                        continue
                    # Update c efficiently
                    # c[k'] changes by 2*delta*f[k'-j] + delta^2*[k'=2j]
                    k_start = j
                    k_end = min(j + n, 2 * n - 1)
                    f_slice = f[:k_end - k_start].copy()
                    # f_slice[i] = f[i] for i = 0..k_end-k_start-1
                    # The change in c[k'] for k' in [j, j+n-1] is 2*delta*f[k'-j]
                    # Plus delta^2 if k' = 2j
                    c_change = 2.0 * delta * f_slice
                    if 2 * j < 2 * n - 1:
                        c_change[2 * j - j] += delta * delta  # c[2j] gets delta^2
                    c_new = c.copy()
                    c_new[k_start:k_end] += c_change
                    c_new = np.maximum(c_new, 0)
                    new_max = float(c_new.max())
                    new_sc = 2.0 * n * new_max
                    if new_sc < best_score * (1.0 - 1e-12):
                        best_score = new_sc
                        f[j] = new_val
                        s = f.sum()
                        if s > 1e-15:
                            f = f / s
                        # Recompute c after renormalization (approximate)
                        c = _fft_convolve(f, f)
                        c_max = float(c.max())
                        tol = c_max * 1.005
                        active = np.where(c > tol)[0]
                        best_a = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_a, best_score


def _fast_cd_polish(a, deadline, max_rounds=100):
    """Faster coordinate descent: only try active coordinates with simple multipliers."""
    n = len(a)
    f = a.copy() / a.sum()
    best_a = f.copy()
    best_score = _score_norm(f)
    c = _fft_convolve(f, f)
    c_max = float(c.max())
    tol = c_max * 1.002
    active = np.where(c > tol)[0]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                for mult in [1.2, 1.5, 0.8, 0.6, 0.4]:
                    f_try = f.copy()
                    f_try[j] *= mult
                    if f_try[j] < 0:
                        continue
                    s = f_try.sum()
                    if s < 1e-15:
                        continue
                    f_try /= s
                    c_try = _fft_convolve(f_try, f_try)
                    sc = 2.0 * n * float(c_try.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        f = f_try
                        best_a = f.copy()
                        c = c_try
                        c_max = float(c.max())
                        tol = c_max * 1.002
                        active = np.where(c > tol)[0]
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)
