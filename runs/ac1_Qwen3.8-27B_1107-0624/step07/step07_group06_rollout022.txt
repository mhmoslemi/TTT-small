```python
def cg_solve_hankel(S, I, n, rhs, max_iter=500):
    m = len(I)
    x = np.zeros(m)
    r = rhs.copy()
    p = r.copy()
    rs = r @ r
    
    for _ in range(max_iter):
        # Compute Q[I,I] * p
        # (Q_S p_full)[i] = sum_j p_full[j] * S[i+j] for i in I
        p_full = np.zeros(n)
        p_full[I] = p
        Qp = hankel_matvec_full(p_full, S, n)[I]
        
        pQp = p @ Qp
        if abs(pQp) < 1e-30:
            break
        alpha = rs / pQp
        x = x + alpha * p
        r = r - alpha * Qp
        r_new = r @ r
        if np.sqrt(r_new) < 1e-12 * np.linalg.norm(rhs):
            break
        beta = r_new / rs if rs > 1e-30 else 0
        p = r + beta * p
        rs = r_new
    
    return x
```

"""
Multi-resolution active-set QP optimizer for minimizing the peak of 
autocorrelation of non-negative sequences. Uses exact active-set identification
with Hankel-structured QP subproblems solved via conjugate gradient.
"""
import numpy as np
import time
from scipy.optimize import minimize

# --- Helper functions ---

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, S_indicator, n):
    """Compute (Q_S * v) where Q_S[i,j] = S_indicator[i+j].
    Result has length n.
    """
    N = _next_pow2(2 * n)
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N)
    fs = np.fft.rfft(S_indicator.astype(np.float64), N)
    r = np.fft.irfft(fv * fs, N)
    return r[n - 1:2 * n - 1]

def _cg_solve_hankel(S_indicator, I, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    """Solve Q_S[I,I] x = rhs using CG.
    Q_S[i,j] = S_indicator[i+j]
    """
    m = len(I)
    if m == 0:
        return np.array([])
    
    x = np.zeros(m)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rhs_norm = float(np.linalg.norm(rhs))
    if rhs_norm < 1e-30:
        return x
    
    for it in range(max_iter):
        # Compute Q[I,I] * p
        p_full = np.zeros(n)
        p_full[I] = p
        Qp_full = _hankel_matvec(p_full, S_indicator, n)
        Qp = Qp_full[I]
        
        if reg > 0:
            Qp = Qp + reg * p
        
        pQp = float(p @ Qp)
        if abs(pQp) < 1e-30:
            break
        alpha = rs / pQp
        x = x + alpha * p
        r = r - alpha * Qp
        r_new = float(r @ r)
        if np.sqrt(r_new) < tol * rhs_norm:
            break
        beta = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta * p
        rs = r_new
    
    return x

# --- Core QP solver ---

def _solve_active_qp(S_indicator, a_init, n, max_outer=50, max_cg=300):
    """Solve min a^T Q_S a s.t. a >= 0, sum(a) = 1.
    Q_S[i,j] = S_indicator[i+j]
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    
    # Initial support
    I = np.where(a > 1e-12)[0]
    if len(I) == 0:
        I = np.array([0])
    
    best_a = a.copy()
    best_obj = float(a @ _hankel_matvec(a, S_indicator, n))
    
    for outer in range(max_outer):
        m = len(I)
        rhs = np.ones(m)
        
        # Try CG solve
        reg = 0.0
        x = _cg_solve_hankel(S_indicator, I, n, rhs, max_iter=max_cg, tol=1e-12, reg=reg)
        
        # Check if solution is valid
        if not np.all(np.isfinite(x)):
            x = np.ones(m) / m
        
        denom = float(x.sum())
        if denom < 1e-15:
            # Add regularization and retry
            reg = 1e-8
            x = _cg_solve_hankel(S_indicator, I, n, rhs, max_iter=max_cg, tol=1e-12, reg=reg)
            denom = float(x.sum())
            if denom < 1e-15:
                break
        
        # Construct solution
        a_new = np.zeros(n)
        a_new[I] = x / denom
        mu = 2.0 / denom
        
        # Check objective
        Qa = _hankel_matvec(a_new, S_indicator, n)
        obj = float(a_new @ Qa)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # KKT check: for i not in I, need (2*Qa[i] - mu) >= 0
        inactive_mask = np.ones(n, dtype=bool)
        inactive_mask[I] = False
        inactive_idx = np.where(inactive_mask)[0]
        
        if len(inactive_idx) > 0:
            violations = 2.0 * Qa[inactive_idx] - mu
            min_viol = float(violations.min())
            if min_viol >= -1e-10:
                # KKT satisfied
                a_new = best_a  # Use best found
                break
            # Add most violated
            i_new = inactive_idx[np.argmin(violations)]
            I = np.sort(np.append(I, i_new))
        else:
            break
        
        # Remove indices with very small values (deflation)
        if len(I) > 2:
            small_mask = a_new <= 1e-15
            if small_mask.sum() > len(I) * 0.5 and small_mask.sum() < n:
                # Keep only indices with significant values
                keep = I[~small_mask[I]]
                if len(keep) >= 2:
                    I = keep
                    a_new[~np.isin(np.arange(n), I)] = 0
                    s = a_new.sum()
                    if s > 1e-15:
                        a_new = a_new / s
    
    return best_a

# --- Multi-resolution optimizer ---

def _upscale(a, n_target):
    """Upscale a to length n_target using interpolation."""
    if len(a) == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n_target) / n_target
    return a_new

def _make_u_shaped(n, alpha=1.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_tapered(n, power=1.0):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_ramp(n, direction=1):
    t = np.linspace(0, 1, n)
    if direction == 1:
        a = 1.0 + t
    else:
        a = 1.0 + (1.0 - t)
    return a / a.sum()

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Main loop: iterate active set and QP solve
    for outer in range(200):
        if time.monotonic() > deadline or (time.monotonic() - t_start) > max_time:
