import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _active_grad(a, c, n, tol_frac=0.005):
    """Compute gradient of max(conv) using active set."""
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    from scipy.optimize import linprog
    
    c_vec = -np.ones(n)  # minimize -sum(b)
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # b[j] contributes to conv[k] with weight a[k-j]
        # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    b_vec = np.full(2*n-1, M)
    
    # Add non-negativity: -b <= 0
    A_full = np.vstack([A, -np.eye(n)])
    b_full = np.concatenate([b_vec, np.zeros(n)])
    
    res = linprog(c_vec, A_ub=A_full, b_ub=b_full, method='highs')

```python
def _solve_kkt_system(S, n):
    """Solve the KKT linear system for a given active set S."""
    # S is a set of lag indices where the max is achieved
    S_set = set(S)
    
    # Build B: n x n sparse matrix
    # B[i, j] = 1 if (j + i) in S_set and 0 <= j < n and 0 <= j+i < 2n-1
    rows = []
    cols = []
    data = []
    for i in range(n):
        for k in S_set:
            j = k - i
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1.0)
    
    from scipy.sparse import csr_matrix
    B = csr_matrix((data, (rows, cols)), shape=(n, n))
    
    # Build the augmented system: [B  -1; 1^T  0] [a; lambda] = [0; 1]
    # But this is (n+1) x (n+1), which is large.
    # Instead, use the fact that we can eliminate lambda:
    # From B a = lambda * 1, we get lambda = (sum of entries of a) ... no.
    # Actually, sum over i: sum_i (B a)_i = sum_i lambda = n * lambda
    # sum_i (B a)_i = (1^T B) a
    # So lambda = (1^T B a) / n
    # And the constraint is 1^T a = 1.
    # So: B a = ((1^T B a)/n) * 1
    # => (n*B - 1^T B) a = 0
    # Plus: 1^T a = 1
    
    # This is: [(n*B - 1^T B) | -1] [a; c] = 0 where c is a free variable
    # And 1^T a = 1
    
    # Alternative: just solve the (n+1) x (n+1) system directly.
    # For n=8192, this is 8193 x 8193. With sparse B, this should be fast.
    
    from scipy.sparse import eye as speye
    from scipy.sparse.linalg import spsolve
    
    # Augmented system
    # [B    -1_vec ] [a]   = [0]
    # [1^T    0    ] [lam]     [1]
    
    one_vec = csr_matrix(np.ones((1, n)))
    zero_scalar = csr_matrix(np.zeros((1, 1)))
    
    # Top block: [B, -1]
    # Bottom block: [1^T, 0]
    
    top_right = csr_matrix(-np.ones((n, 1)))
    bottom_left = csr_matrix(np.ones((1, n)))
    bottom_right = csr_matrix(np.array([[0.0]]))
    
    A_aug = sparse.block_array([[B, top_right], [bottom_left, bottom_right]], format='csr')
    rhs = np.zeros(n + 1)
    rhs[-1] = 1.0
    
    try:
        sol = spsolve(A_aug, rhs)
        a_sol = sol[:n]
        lam = sol[n]
    except Exception:
        return None
    
    # Check feasibility
    if np.min(a_sol) < -1e-8:
        return None  # Not all non-negative
    
    a_sol = np.maximum(a_sol, 0)
    s = a_sol.sum()
    if s < 1e-10:
        return None
    a_sol = a_sol / s
    
    return a_sol, lam
```

"""
Multi-strategy optimizer with KKT active-set insights and aggressive beta continuation.
Key improvements over previous:
1. Longer, more aggressive log-sum-exp L-BFGS schedule (more time at small beta)
2. KKT linear system solver for exact pressure computation
3. Adaptive pressure equalization with decreasing gamma
4. Active-set gradient "squeezing" step for final polish
5. Diverse asymmetric initializations and multi-n exploration
6. LP-based best response for moderate n as additional refinement
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix, vstack as sp_vstack, eye as sp_eye
from scipy.sparse.linalg import spsolve


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=0.003):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _solve_pressure_system(S, n):
    """Solve the KKT linear system B*a = mu*1, sum(a)=1 where B[i,j]=1_{j+i in S}."""
    S_set = set(int(k) for k in S)
    rows = []
    cols = []
    data = []
    for i in range(n):
        for k in S_set:
            j = k - i
            if 0 <= j < n:
                rows.append(i)
                cols.append(j)
                data.append(1.0)
    B = csr_matrix((data, (rows, cols)), shape=(n, n))
    one_col = csr_matrix(-np.ones((n, 1)))
    one_row = csr_matrix(np.ones((1, n)))
    zero_11 = csr_matrix(np.zeros((1, 1)))
    top = sp_vstack([B, one_col], format='csr')
    bot = sp_vstack([one_row, zero_11], format='csr')
    A_aug = sp_vstack([top, bot], format='csr')
    rhs = np.zeros(n + 1)
    rhs[-1] = 1.0
    try:
        sol = spsolve(A_aug + 1e-12 * sp_eye(n + 1, format='csr'), rhs)
    except Exception:
        try:
            sol = spsolve(A_aug, rhs)
        except Exception:
            return None
    if sol is None or not np.all(np.isfinite(sol)):
        return None
    a_sol = sol[:n]
    if np.min(a_sol) < -1e-4:
        a_sol = np.maximum(a_sol, 0.0)
    s = a_sol.sum()
    if s < 1e-10:
        return None
    return a_sol / s


def _pressure_equalize(a, deadline, n_iters=300, gamma_init=0.5, gamma_end=0.1):
    """Adaptive pressure equalization with decreasing gamma."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score_norm(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        gamma = gamma_init * (gamma_end / gamma_init) ** (it / max(1, n_iters))
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        a_rev = a[::-1]
        p = _full_convolve(active.astype(np.float64), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        if it % 30 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _kkt_refine(a, deadline, max_iters=30, S_size=15):
    """Use KKT system to refine the solution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[-S_size:]
        S = set(int(k) for k in top_idx)
        a_kkt = _solve_pressure_system(S, n)
        if a_kkt is None:
            break
        sc_kkt = _score_norm(a_kkt)
        if sc_kkt < best_sc * 0.999:
            a = a_kkt
            best_a = a.copy()
            best_sc = sc_kkt
        elif sc_kkt < best_sc:
            a = 0.5 * a + 0.5 * a_kkt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            sc_new = _score_norm(a)
            if sc_new < best_sc:
                best_a = a.copy()
                best_sc = sc_new
        else:
            break
    return best_a


def _squeezing_step(a, deadline, n_steps=100, lr=0.001):
    """Exact gradient steps on the true max (no smoothing)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)
    current_lr = lr

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        idx_max = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx_max] = 1.0
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = a[idx_max - (n - 1)] if 0 <= idx_max - (n - 1) < n else 0.0
        # More carefully:
        W = float(a[idx_max] * 0)  # simplified: W = dot(w, c) = c[idx_max] = c_max
        W = c_max
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        z = z - current_lr * grad / grad_norm
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if step % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                current_lr = min(current_lr * 1.01, lr * 2)
            else:
                current_lr *= 0.9
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0] * 0.1
    base_beta = max(gap, max_c * 0.01)
    if base_beta < 1e-10:
        base_beta = max_c * 0.01

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.001953125,
        0.0009765625, 0.0004882812, 0.0002441406,
        0.0001220703, 0.0000610352, 0.0000305176,
        0.0000152588, 0.0000076294
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        est_time = 0.004 * max(1.0, n / 2048.0)
        # More iterations at small beta
        if beta < base_beta * 0.01:
            max_iter = min(15000, max(100, int(remaining / (est_time * 6))))
        elif beta < base_beta * 0.1:
            max_iter = min(10000, max(80, int(remaining / (est_time * 7))))
        else:
            max_iter = min(5000, max(50, int(remaining / (est_time * 8))))
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 150}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_grad(a, c, n, tol_frac=0.005)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.015, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_polish(a, deadline, n_rounds=80):
    """Coordinate descent focusing on peak-contributing coordinates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(98765)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(np.float64), a_rev)
        contrib = contrib[n - 1:2 * n - 1]
        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = rng.permutation(top_idx[:min(len(top_idx), n_try)])

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            c_old = c_max
            for factor in [0.5, 0.7, 0.85, 1.15, 1.4]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-15:
                    improved = True
                    c_old = c_try
                else:
                    a[i] = old_val
                    break
        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):

        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_incr":
        t = np.linspace(0, 1, n)
        a = np.exp(3.0 * t) + 0.1
        return a / a.sum()
    elif kind == "asym_decr":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.1
        return a / a.sum()

        return a / a.sum()
