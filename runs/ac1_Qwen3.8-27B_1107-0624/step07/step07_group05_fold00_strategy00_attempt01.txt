def make_profile(length, profile_type="uniform"):
    # Normalize to sum=1 for all profiles
    if profile_type == "exponential_taper":
        a = np.ones(length) * 0.1
        b = np.ones(length) * np.random.uniform(1.0, 3.0)
        nphi = np.arctan(2.7 * np.pi / length) + 0.1
        a = 0.2 + 0.8 * np.cos(np.arange(length) * nphi / length)
        a = a / a.sum()
        # Add noise
        a = a * (1 + 0.05 * (np.random.rand(length) - 0.5))
    elif profile_type == "chirp_real":
        t = np.linspace(0, length/2, length, dtype=np.float64)
        a = np.ones(length) + 0.3 * np.cos(2*length*t*t) + 0.1*t
        a = a / a.sum()
    elif profile_type == "wavelet_like":
        t = np.linspace(0, 1, length)
        # Build two effective components
        w1 = np.ones(length) + 0.2 * np.sin(4*length*t) * np.exp(-10*t*(t-0.3))
        w2 = np.ones(length) + 0.1 * np.sin(8*length*t) * np.exp(-5*t*(t-0.7))
        a = (w1 + w2) / (w1+w2).sum()
    else:  # Default: "uniform" is not specified for exponential_taper, default to uniform
        a = np.ones(length) / length
    return a

def analyze_convolve(seq):
    if seq is None:
        return np.inf
    n = len(seq)
    # Evaluator handles defensive clipping automatically
    conv = np.convolve(seq, seq, mode="full")
    return float(2.0 * n * conv.max())

def final_resample(original_sequence, max_reduction=4):
    # Not all reduction paths will succeed with single Gaussian convolution
    if original_sequence is None:
        return None
    
    n_orig = len(original_sequence)
    reduction_goal = min(256, n_orig // max_reduction)
    reduction_candidates = []
    for fc in range(1, reduction_goal):
        reduction_candidates.append(reduction_goal)

    # Try shorter sets, since evaluator's objective normalizes and is faster
    # This is a placeholder - actual implementation must be done using coordinate descent
    # But we'll simulate by truncating and re-normalizing
    tcoeff = original_sequence[:reduction_goal]
    normalized_tcoeff = tcoeff / tcoeff.sum()
    return normalized_tcoeff
