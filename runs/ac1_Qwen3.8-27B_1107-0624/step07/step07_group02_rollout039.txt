"""
Optimized step function search using adaptive gradient flow with momentum,
log-sum-exp L-BFGS refinement, and local perturbation search.
Key innovations: adaptive active-set temperature, Nesterov momentum,
backtracking line search, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _grad_flow(a_init, deadline, temp_start=0.05, temp_end=0.001,
               lr_start=0.02, momentum_start=0.85, max_steps=None):
    """Adaptive gradient flow with Nesterov momentum and backtracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    
    momentum = np.zeros(n)
    lr = lr_start
    mu = momentum_start
    no_improve = 0
    max_stall = 2000
    
    if max_steps is None:
        max_steps = 10_000_000
    steps = 0
    
    while time.monotonic() < deadline:
        steps += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        # Adaptive temperature based on the gap between max and mean
        gap = c_max - c_mean
        if gap < c_max * 1e-6:
            break
        
        # Interpolate temperature
        progress = min(1.0, steps / max_steps)
        temp = temp_start * (temp_end / temp_start) ** progress * c_max
        temp = max(temp, c_max * 1e-5)
        
        # Compute softmax weights
        shifted = (c - c_max) / temp
        w = np.exp(np.clip(shifted, -700, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient: grad[i] = sum_k w[k] * 2 * a[k-i]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        
        # Center the gradient to preserve sum
        grad = grad - np.mean(grad)
        
        # Nesterov momentum
        momentum = mu * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            momentum *= 0.5
            continue
        a_new = a_new / s_new
        
        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        elif sc_new < 2.0 * n * c_max - 1e-14:
            no_improve = 0
        else:
            no_improve += 1
        
        # Backtracking if not improved
        if sc_new >= 2.0 * n * c_max:
            # Try smaller step
            a_try = a + 0.5 * momentum
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < 2.0 * n * c_max:
                    a = a_try
                    momentum *= 0.5
                else:
                    lr *= 0.7
                    momentum *= 0.3
            else:
                lr *= 0.5
                momentum = np.zeros(n)
        else:
            a = a_new
            # Slightly increase learning rate on success
            lr = min(lr * 1.01, 0.5)
        
        # Stall handling
        if no_improve >= max_stall:
            # Perturb to escape
            a = a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            momentum = np.zeros(n)
            lr = lr_start * 0.5
            no_improve = 0
        
        # Decrease momentum slightly over time for convergence
        mu = max(0.5, momentum_start - 0.01 * progress)
    
    return best_a, best_sc

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.005:
        gap = max_c * 0.005
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004, 0.002, 0.001, 0.0005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    class _Prob:
        def __init__(self, beta, n):
            self.beta = beta
            self.n = n
        def __call__(self, z):
            return _smoobj(z, self.beta, self.n)
        def grad(self, z):
            return _smograd(z, self.beta, self.n)
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 15))))
        
        prob = _Prob(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-16,
                                      'gtol': 1e-14, 'maxcor': 50, 'maxls': 30})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        idx = rng.choice(n, size=subset_size, replace=False)
        
        # Choose modification: scale by random factor
        factors = 1.0 + rng.standard_normal(subset_size) * 0.1
        factors = np.maximum(factors, 0.1)
        
        a_try = a.copy()
        a_try[idx] *= factors

        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

            c = _full_convolve(a, a)
            peak_idx = int(np.argmax(c))

"""
Adaptive gradient flow with Nesterov momentum, log-sum-exp L-BFGS refinement,

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _compute_grad(a, w):
    """Compute weighted gradient of score (with sum=1 normalization)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    grad = grad - np.mean(grad)
    return grad


def _grad_flow(a_init, deadline, temp_start=0.04, temp_end=0.003,

    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)

    best_a = a.copy()

    c_max = float(c.max())
    best_sc = 2.0 * n * c_max

    lr = lr_start

        c = _full_convolve(a, a)

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 20))))

        class _Prob:
            def __init__(self, beta_val, n_val):
                self.beta_val = beta_val
                self.n_val = n_val

            def __call__(self, z_in):
                return _smoobj(z_in, self.beta_val, self.n_val)

            def grad(self, z_in):
                return _smograd(z_in, self.beta_val, self.n_val)

        prob = _Prob(beta, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-17,
                                       'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _local_perturb_search(a_init, deadline, rng, n_max_trials=2000):
    """Local search with structured perturbations targeting peak lags."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    trials = 0

    while time.monotonic() < deadline and trials < n_max_trials:
        trials += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Strategy 1: Perturb entries that contribute to the peak
        # grad[i] = 2*a[peak_idx - i], so large grad means large a[peak_idx - i]
        partners = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                partners[i] = a[j]

        # Select top contributors to the peak
        top_k = max(1, n // 128)
        top_idx = np.argsort(partners)[::-1][:top_k]

        # Randomly scale some of the top contributors down
        n_modify = max(1, top_k // 4)
        modify_idx = rng.choice(top_idx, size=n_modify, replace=False)
        a_try = a.copy()
        a_try[modify_idx] *= (1.0 - 0.05 * rng.random(n_modify))
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            continue

        # Strategy 2: Random subset scaling
        subset_size = rng.integers(max(1, n // 512), max(2, n // 32))
        idx = rng.choice(n, size=subset_size, replace=False)
        a_try = a.copy()
        a_try[idx] *= (1.0 + 0.03 * rng.standard_normal(subset_size))
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: adaptive gradient flow + L-BFGS + local search."""
    t_start = time.monotonic()
    total_time = float(budget_s) - 12.0
    deadline = t_start + total_time
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_sc = np.inf

    # Load parent
    parent = None
    n_parent = 8192
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={best_sc:.10f}")
    else:
        n = 8192
        best_a = np.ones(n) / n
        best_sc = _score(best_a)
        print(f"No parent, using uniform n={n}, score={best_sc:.10f}")

    n = len(best_a)

    # === Phase 1: Main gradient flow (40% of time) ===
    t1_end = t_start + 0.40 * total_time
    print(f"\n[Phase 1] Gradient flow (t={time.monotonic()-t_start:.1f}s to {t1_end-t_start:.1f}s)")
    a1, sc1 = _grad_flow(best_a, t1_end, temp_start=0.04, temp_end=0.003,
                         lr_start=0.04, mu_start=0.92)
    if sc1 < best_sc:
        best_a = a1
        best_sc = sc1
        print(f"  Phase 1: {best_sc:.10f} (improved by {0 if np.isinf(best_sc) else 0:.6f})")
    else:
        print(f"  Phase 1: no improvement (got {sc1:.10f}, best {best_sc:.10f})")

    # === Phase 2: L-BFGS refinement (25% of time) ===
    t2_end = t_start + 0.65 * total_time
    print(f"\n[Phase 2] L-BFGS (t={time.monotonic()-t_start:.1f}s to {t2_end-t_start:.1f}s)")
    a2 = _run_lbfgs(best_a, t2_end, min_time_remaining=200)
    sc2 = _score(a2)
    if sc2 < best_sc:
        best_a = a2
        best_sc = sc2
        print(f"  Phase 2: {best_sc:.10f}")
    else:
        print(f"  Phase 2: no improvement (got {sc2:.10f})")

    # === Phase 3: Fine gradient flow (20% of time) ===
    t3_end = t_start + 0.85 * total_time
    print(f"\n[Phase 3] Fine gradient flow (t={time.monotonic()-t_start:.1f}s to {t3_end-t_start:.1f}s)")
    a3, sc3 = _grad_flow_fine(best_a, t3_end, temp_start=0.004, temp_end=0.0003,
                              lr_start=0.006, mu_start=0.85)
    if sc3 < best_sc:
        best_a = a3
        best_sc = sc3
        print(f"  Phase 3: {best_sc:.10f}")
    else:
        print(f"  Phase 3: no improvement (got {sc3:.10f})")

    # === Phase 4: Second L-BFGS pass (8% of time) ===
    t4_end = t_start + 0.93 * total_time
    print(f"\n[Phase 4] L-BFGS pass 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < t4_end:
        a4 = _run_lbfgs(best_a, t4_end, min_time_remaining=50)
        sc4 = _score(a4)
        if sc4 < best_sc:
            best_a = a4
            best_sc = sc4
            print(f"  Phase 4: {best_sc:.10f}")
        else:
            print(f"  Phase 4: no improvement (got {sc4:.10f})")

    # === Phase 5: Local perturbation search (7% of time) ===
    t5_end = deadline
    print(f"\n[Phase 5] Local search (t={time.monotonic()-t_start:.1f}s)")
    a5, sc5 = _local_perturb_search(best_a, t5_end, rng)
    if sc5 < best_sc:
        best_a = a5
        best_sc = sc5
        print(f"  Phase 5: {best_sc:.10f}")
    else:
        print(f"  Phase 5: no improvement (got {sc5:.10f})")

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        ev = evaluate_sequence([float(x) for x in final])
        print(f"evaluate_sequence: {ev}")
    except Exception:
        pass

    return [float(x) for x in final]
