"""
Optimized step-function search using aggressive L-BFGS annealing, active-set
polishing, multi-n exploration, peak-shaving heuristics, and diverse restarts.
Targets improvement below 1.5027 by exploiting larger sequence lengths and
sharper optimization schedules.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LseProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad(z, self.beta, self.n)


def _run_lbfgs_aggressive(a_init, deadline, min_remaining=20.0):
    """L-BFGS with very aggressive beta annealing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.002)

    # More aggressive schedule: go down to very small betas
    betas = base_beta * np.array(
        [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
         0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
         0.00005, 0.00002, 0.00001, 0.000005]
    )

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5.0:
            break

        # Estimate time per iteration based on n
        scale = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * scale
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 15))))

        prob = _LseProblem(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish_v2(a_init, deadline, tol_start=0.005, tol_end=0.00001,
                          lr_start=0.03, top_k=5):
    """Improved active-set GD with top-k peaks and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_start
    tol = tol_start
    no_improve = 0
    stall = 0
    max_stall = 300

    while time.monotonic() < deadline - 2.0:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Select top-k peaks
        k_eff = min(top_k, len(c))
        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        if n_active > k_eff * 10:
            # Use top-k_eff instead
            top_indices = np.argsort(c)[::-1][:k_eff]
            active = np.zeros_like(c, dtype=bool)
            active[top_indices] = True
            n_active = k_eff

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Backtracking line search
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        base_sc = current_sc
        improved = False

        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < base_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol = max(tol_end, tol * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                tol = max(tol_end, tol * 0.5)
                lr = lr_start * 0.4
                no_improve = 0
                stall += 1
                if stall > 12:
                    break
        else:
            if tol > tol_end:
                tol = max(tol_end, tol * 0.985)
            lr = min(lr_start, lr * 1.01)

    return best_a, _score(best_a)


def _peak_shave(a, deadline, n_trials=200):
    """Direct peak shaving: identify top contributors and reduce them."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find the elements contributing most to the peak
        # conv(a,a)[k] = sum_i a[i] * a[k-i]
        # Gradient wrt a[i] is a[k-i] (approximately)
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        if k_lo >= k_hi:
            idx = int(np.argmax(a))
            contrib = a.copy() * a.copy()
            # Reduce top contributor
            a_try = a.copy()
            a_try[idx] *= 0.95
            # Distribute the reduction to others
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            continue

        # Contribution vector: a[i] * a[k_star - i]
        i_range = np.arange(k_lo, k_hi + 1)
        j_range = k_star - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            continue

        # Reduce the top contributors
        frac = rng.uniform(0.01, 0.08)
        n_reduce = max(1, int(len(contrib) * frac))
        top_contrib_idx = i_range[np.argsort(contrib)[::-1][:n_reduce]]

        a_try = a.copy()
        reduction = 0.0
        for idx in top_contrib_idx:
            red = a_try[idx] * 0.1
            a_try[idx] -= red
            reduction += red

        # Distribute reduction to low-contribution elements
        low_mask = np.zeros(n, dtype=bool)
        low_mask[:] = True
        for idx in top_contrib_idx:
            low_mask[idx] = False
        j_set = set(int(j) for j in j_range)
        for idx in top_contrib_idx:
            j_val = k_star - idx
            if 0 <= j_val < n:
                low_mask[j_val] = False

        if low_mask.sum() == 0:
            low_mask = np.ones(n, dtype=bool)
            for idx in top_contrib_idx:
                low_mask[idx] = False
            if low_mask.sum() == 0:
                continue

        a_try[low_mask] += reduction / low_mask.sum()
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wedge":
        t = np.linspace(0, 1, n)
        a = 0.5 + np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 13.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(1, n + 1)
        a = 1.0 / np.sqrt(t.astype(float))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Try larger n (16384) ===
    print(f"\n[P4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [16384, 12288, 20480]
    if n_parent not in n_targets:
        n_targets.append(n_parent)

    for n_try in sorted(n_targets):
        if time.monotonic() > deadline - 150.0:
            break
        if n_try == len(best_a):
            continue
        a_init = _resample(best_a, n_try)
        sc_init = _score(a_init)
        print(f"  Resampled to n={n_try}, init score={sc_init:.10f}")
        remaining_for_n = min(250.0, deadline - time.monotonic() - 100.0)
        if remaining_for_n < 30:
            continue
        a_opt = _run_lbfgs_aggressive(a_init, t_start + time.monotonic() - t_start, min_remaining=20.0)
        # Fix: use absolute deadline
        a_opt = _run_lbfgs_aggressive(a_init, deadline - 100.0, min_remaining=20.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        # Quick polish on this n
        if time.monotonic() < deadline - 80.0:
            a_pol, sc_pol = _active_set_polish_v2(a_opt, min(deadline - 60.0, time.monotonic() + 80.0),
                                                   tol_start=0.003, tol_end=0.0001, lr_start=0.015)
            if sc_pol < best_score:
                best_a = a_pol.copy()
                best_score = sc_pol
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Diverse restarts ===
    print(f"\n[P5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80.0:
        n_best = len(best_a)
        restarts = [
            ("pert_small", _perturb(best_a, 0.02, rng)),
            ("pert_med", _perturb(best_a, 0.05, rng)),
            ("tapered", _make_profile(n_best, "tapered")),
            ("wedge", _make_profile(n_best, "wedge")),
            ("bimodal", _make_profile(n_best, "bimodal")),
            ("cos_mod", _make_profile(n_best, "cosine_mod")),
        ]
        # Also try resampled parent with different profiles
        if n_parent != n_best:
            restarts.append(("parent_resample", _resample(parent, n_best) if parent is not None else _make_profile(n_best, "uniform")))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 40.0:
                break
            remaining = deadline - time.monotonic() - 30.0
            if remaining < 15.0:
                break
            a_opt = _run_lbfgs_aggressive(a_r, deadline - 30.0, min_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polishing ===
    print(f"\n[P6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_polish_v2(best_a, deadline - 2.0,
                                              tol_start=0.0005, tol_end=0.000005,
                                              lr_start=0.008, top_k=3)
        if sc_fin < best_score:
            best_a = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Peak shave at the end
        if time.monotonic() < deadline - 2.0:
            a_ps, sc_ps = _peak_shave(best_a, deadline - 1.0, n_trials=50)
            if sc_ps < best_score:
                best_a = a_ps.copy()
                best_score = sc_ps
                print(f"  NEW BEST: {best_score:.10f} [final_peak_shave] t={time.monotonic()-t_start:.1f}s")

        # One more tiny L-BFGS
        remaining = deadline - 2.0 - time.monotonic()
        if remaining > 5.0:
            n = len(best_a)
            c_best = _full_convolve(best_a, best_a)
            mb = float(c_best.max())
            beta_fin = mb * 0.00001
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
            prob = _LseProblem(beta_fin, n)
            max_iter = min(1000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 50})
                a_fin2 = _softmax(result.x)
                a_fin2 = np.maximum(a_fin2, 0.0)
                s_fin2 = a_fin2.sum()
                if s_fin2 > 1e-10:
                    a_fin2 = a_fin2 / s_fin2
                    sc_fin2 = _score(a_fin2)
                    if sc_fin2 < best_score:
                        best_a = a_fin2.copy()
                        best_score = sc_fin2
                        print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
