"""
Convex optimization approach to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key insight: the problem is convex (max of quadratic forms subject to simplex constraint).
Uses FISTA with exact subgradients, adaptive active sets, Adam in log-space,
and coordinate descent polishing. Multi-n exploration included.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _exact_subgrad(a, c, active):
    """Exact subgradient of mean(c[active]) w.r.t. a, using FFT."""
    n = len(a)
    m = len(c)
    a_rev = a[::-1]
    u = np.zeros(m)
    u[active] = 1.0
    conv_res = _fft_convolve(a_rev, u)
    g = 2.0 / len(active) * conv_res[n - 1:2 * n - 1]
    return g

    t_k = 1.0

    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break

        frac = t / max_iter
        lr = lr_start * (lr_end / lr_start) ** frac
        k_active = max(1, int(active_top * active_shrink ** (t / max(1, max_iter // 10))))

        c_y = _fft_convolve(y, y)
        c_max = float(c_y.max())
        top_idx = np.argsort(c_y)[::-1][:k_active]
        active = top_idx

        g = _exact_subgrad(y, c_y, active)

        a_new = _project_simplex(y - lr * g)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new

        if t % eval_every == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if t % (eval_every * 10) == 0:
                    print(f"  FISTA t={t}, score={best_score:.10f}, k={k_active}, lr={lr:.6f}")

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, T_start=0.01, T_end=0.0001,
                   lr_start=0.03, lr_end=0.001, max_iter=200000,
                   top_k=5, eval_every=200):
    """Adam optimizer in log-space with smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Smoothed gradient: softmax-weighted subgradient
        c_shifted = (c - c_max) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s

        a_rev = a[::-1]
        u = p.copy()
        conv_res = _fft_convolve(a_rev, u)
        g_conv = 2.0 * conv_res[n - 1:2 * n - 1]

        # Gradient in z-space: a * (g - mean(a*g))
        W = float(np.dot(a, g_conv))
        z_grad = a * (g_conv - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        upd_norm = np.linalg.norm(update)
        if upd_norm > 1.0:
            update = update * (1.0 / upd_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % eval_every == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                if t % (eval_every * 5) == 0:
                    print(f"  Adam t={t}, score={best_score:.10f}, T={T:.6f}")
            else:
                patience += 1
            if patience > 3 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=200, top_k=5):
    """Fast coordinate descent: adjust individual a[j] to reduce peak."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        peak_idx = np.argsort(c)[::-1][:top_k]

        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            best_local = best_score
            best_local_a = None

            for j in range(i_start, i_end + 1):
                partner = k - j
                if partner < 0 or partner >= n:
                    continue
                for scale in [0.5, 0.3, 0.1, 2.0, 0.7]:
                    a_cand = best_a.copy()
                    old = a_cand[j]
                    a_cand[j] = old * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_local - 1e-12:
                        best_local = sc
                        best_local_a = a_cand.copy()

            if best_local_a is not None:
                best_a = best_local_a
                best_score = best_local
                improved = True

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fista_adaptive(a_init, deadline, max_iter=100000):
    """FISTA with gradually tightening active set and adaptive lr."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = _score_norm(y)
    t_k = 1.0
    lr = 0.02

    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break

        phase = min(1.0, t / max_iter)
        if phase < 0.3:
            k_act = 20
            lr = 0.02 * (1.0 - phase / 0.3 * 0.5)
        elif phase < 0.7:
            k_act = 8
            lr = 0.01 * (1.0 - (phase - 0.3) / 0.4 * 0.5)
        else:
            k_act = 3
            lr = 0.005 * (1.0 - (phase - 0.7) / 0.3 * 0.5)

        c_y = _fft_convolve(y, y)
        c_max = float(c_y.max())
        top_k = min(k_act, len(c_y))
        active = np.argsort(c_y)[::-1][:top_k]
        g = _exact_subgrad(y, c_y, active)

        a_new = _project_simplex(y - lr * g)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y_new = _project_simplex(y_new)

        # Check if y_new improves
        c_ynew = _fft_convolve(y_new, y_new)
        if float(c_ynew.max()) < c_max:
            y = y_new
        else:
            y = a_new

        t_k = t_new
        a = a_new

        if t % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if t % 2000 == 0:
                    print(f"  FistaAdapt t={t}, score={best_score:.10f}, k={k_act}")

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimization:
    1. FISTA with exact subgradients (broad active set)
    2. Adam in log-space (smooth optimization)
    3. FISTA with tight active set (fine optimization)
    4. Try larger n (16384)
    5. Final coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Phase 1: FISTA with broad active set (0-200s)
    p1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < p1_end:
        a1, sc1 = _fista_subgrad(a, p1_end, max_iter=200000,
                                  lr_start=0.04, lr_end=0.002,
                                  active_top=15, active_shrink=0.6,
                                  eval_every=200)
        full_sc1 = _score(a1)
        print(f"Phase 1 FISTA: score={full_sc1:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc1) and full_sc1 < best_score:
            best_score = full_sc1
            best_a = a1.copy()

    # Phase 2: Adam in log-space (200-600s)
    p2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < p2_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a2, sc2 = _adam_logspace(best_a, p2_end,
                                  T_start=0.005 * max_c, T_end=0.0001 * max_c,
                                  lr_start=0.03, lr_end=0.001,
                                  max_iter=500000, top_k=5, eval_every=300)
        full_sc2 = _score(a2)
        print(f"Phase 2 Adam: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a2.copy()

    # Phase 3: FISTA with tight active set (600-1000s)
    p3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < p3_end:
        a3, sc3 = _fista_adaptive(best_a, p3_end, max_iter=300000)
        full_sc3 = _score(a3)
        print(f"Phase 3 FistaAdapt: score={full_sc3:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc3) and full_sc3 < best_score:
            best_score = full_sc3
            best_a = a3.copy()

    # Phase 4: Try n=16384 (1000-1300s)
    p4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < p4_end - 60:
        a_16k = _interp_to_length(best_a, 16384)
        c_16k = _fft_convolve(a_16k, a_16k)
        max_16k = float(c_16k.max())
        a4, sc4 = _adam_logspace(a_16k, min(p4_end - 30, time.monotonic() + 250),
                                  T_start=0.003 * max_16k, T_end=0.00005 * max_16k,
                                  lr_start=0.02, lr_end=0.001,
                                  max_iter=200000, top_k=3, eval_every=300)
        full_sc4 = _score(a4)
        print(f"Phase 4 n=16384: score={full_sc4:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc4) and full_sc4 < best_score:
            best_score = full_sc4
            best_a = a4.copy()

    # Phase 5: Come back to best-n and do final FISTA (1300-1550s)
    p5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p5_end - 30:
        n_best = len(best_a)
        a5, sc5 = _fista_subgrad(best_a, min(p5_end - 10, time.monotonic() + 220),
                                  max_iter=150000,
                                  lr_start=0.01, lr_end=0.0001,
                                  active_top=5, active_shrink=0.4,
                                  eval_every=200)
        full_sc5 = _score(a5)
        print(f"Phase 5 Final FISTA (n={n_best}): score={full_sc5:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc5) and full_sc5 < best_score:
            best_score = full_sc5
            best_a = a5.copy()

    # Phase 6: Adam fine-tune (1550-1720s)
    p6_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < p6_end - 20:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a6, sc6 = _adam_logspace(best_a, min(p6_end - 10, time.monotonic() + 150),
                                  T_start=0.0005 * max_cf, T_end=0.00002 * max_cf,
                                  lr_start=0.005, lr_end=0.0002,
                                  max_iter=100000, top_k=3, eval_every=200)
        full_sc6 = _score(a6)
        print(f"Phase 6 Adam fine: score={full_sc6:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc6) and full_sc6 < best_score:
            best_score = full_sc6
            best_a = a6.copy()

    # Phase 7: Coordinate descent polish (1720-1785s)
    p7_end = min(deadline, t_start + 1785.0)
    if time.monotonic() < p7_end - 5:
        a7, sc7 = _cd_polish(best_a, min(p7_end - 3, time.monotonic() + 60),
                              max_rounds=100, top_k=5)
        full_sc7 = _score(a7)
        print(f"Phase 7 CD: score={full_sc7:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc7) and full_sc7 < best_score:
            best_score = full_sc7
            best_a = a7.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
