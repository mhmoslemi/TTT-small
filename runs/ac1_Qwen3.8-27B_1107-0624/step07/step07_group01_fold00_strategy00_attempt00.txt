def setup_optimization(sequence):
    """Setup the optimization problem.

    Args:
        sequence (list[float]): The initial sequence to start the optimization.

    Returns:
        tuple: (Objective, Gradient function) for the PyROPBSVM optimization.
    """
    # Convert to numpy for efficiency
    sequence_np = np.array(sequence, dtype=np.float64)
    n = len(sequence_np)

    # Define the objective function
    def objective(z):
        # z is the vector of log coefficients, then exp-normalize to get sequence
        coeffs = np.exp(z)
        s = np.sum(coeffs)
        # Normalize c_exp: but this might interfere with the constraint, so we maintain log space
        # Instead, we compute the scaled objective without the sum adjustment
        score = (2.0 * n * full_convolve_man(coeffs)[-1] if len(z)>0 else np.inf) / s**2.0
        return score

    # Define the gradient function (if we had a proper version, would use it)
    def gradient(z):
        """Gradient computed using active-set analysis."""
        # Placeholder: This would require the active gradient in our previous framework.
        # We compute a numerical gradient here, which is inefficient.
        grad = []
        for idx in range(len(z)):
            # Small perturbation
            z_plus = np.array(z, dtype=np.float64)
            z_plus[idx] += 1e-5
            z_minus = np.array(z, dtype=np.float64)
            z_minus[idx] -= 1e-5

            # Outer function: objective(z)
            f_plus = objective(z_plus)
            f_minus = objective(z_minus)

            # Central difference: (f_plus - f_minus) / (2*dx)
            grad_here = (f_plus - f_minus) / (2e-5)

            # Scale to log space? Note: the objective f(z)= (2n * max_c) / (sum_z)^2, sum_z is the log sum of exp(z_i)
            # But the previous framework used an analytic gradient, we don't.
            grad.append(grad_here)
        return np.array(grad)

    # Helper function for full convolution (mode='full') via standard method, since we cannot rely on the previous program's helper
    def full_convolve_man(coeffs):
        """Perform a full convolution of the sequence array."""
        n_len = len(coeffs)
        # Default mode for np.convolve is 'linear', but we want 'full'
        b = np.convolve(coeffs, coeffs, mode='full')
        return b

    # Setup for L-BFGS minimization in log space (PyCHORAL nature, but we're using a fallback)
    return objective, gradient, n, sequence_np

    Uses the function and its gradient if available.

    Args:
        sequence (list[float]): Starting sequence.
        time_budget (float): Total time budget for optimization.
        start_time (float): Current start time (seconds since epoch or wall-time).

    # We'll define a helper function and its gradient for this particular length

    # Adjust time_budget to be the remaining time (starting_time + reserve <= deadline)
    current_time = time.time()

    # Optimization in log space

    # Define objective function for L-BFGS
    def objective(z):

        s = np.sum(coefficients)
        conv_result = np.convolve(coefficients, coefficients, mode='full')
        conv_max = max(conv_result) if len(z) > 0 else np.inf
        result = (2 * len(sequence) * conv_max) / (s ** 2)
        return float(result) if not np.isnan(result) else np.inf

    # Define the gradient for L-BFGS

        grad = []
        for idx in range(len(z)):
            z_plus = np.array(z, dtype=np.float64)
            z_plus[idx] += 1e-7

        # Lower bound is determined by the derivative: note that L-BFGS requires derivatives
        # But if the function is not safe to use, we return a gradient of zeros
