"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) L-BFGS with log-sum-exp smoothing, (2) Active-set Newton with CG,
(3) Coordinate descent polish, (4) Multi-n exploration with adaptive active-set refinement.
Key improvements: tighter Newton convergence, adaptive active-set tolerance,
coordinate descent for final polish, and n=16384 exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _make_lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_matvec(v, A_vec, n, mu, m):
    return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(rhs, A_vec, n, mu, m, max_cg=300):
    x = np.zeros_like(rhs)
    r = rhs - _hess_matvec(x, A_vec, n, mu, m)
    p = r.copy()
    rs = np.dot(r, r)
    for _ in range(max_cg):
        Ap = _hess_matvec(p, A_vec, n, mu, m)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(rhs)):
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x


def _active_set_newton(a_init, deadline, max_iter=400, tol_active=0.001):
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = int(active.sum())

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-15:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()
            break

        mu = 1e-8 * c_max + 1e-10

        u = _cg_solve(g_t, A_vec, n, mu, m, max_cg=300)
        w = _cg_solve(np.ones(n), A_vec, n, mu, m, max_cg=300)
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-11:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(20):
                a_try = np.maximum(a - t * g_t, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(best_a, best_a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1, n / 2048.0)
        max_iter = min(10000, max(100, int(remaining / (0.008 * scale * 10))))

        try:
            result = minimize(
                _make_lsobj,
                z,
                args=(beta, n),
                jac=_make_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coord_descent(a_init, deadline, n_passes=50):
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            old_x = a[i]
            if old_x < 1e-15:
                continue

            B = np.zeros(2 * n - 1)
            valid_k = np.arange(max(0, i), min(2 * n - 1, n + i))
            B[valid_k] = a[valid_k - i]

            C = c - 2.0 * old_x * B - old_x * old_x * (1.0 if 2 * i < len(c) else 0.0)
            if 2 * i < len(c):
                C[2 * i] = c[2 * i] - old_x * B[2 * i]

            # Minimize max_k {C[k] + 2*x*B[k] + x^2*(k==2i)} over x >= 0
            # Use bisection on subgradient
            x_lo = 0.0
            x_hi = old_x * 3.0 + 1e-10

            for _ in range(40):
                x_mid = 0.5 * (x_lo + x_hi)
                vals = C + 2.0 * x_mid * B
                if 2 * i < len(c):
                    vals[2 * i] += x_mid * x_mid

                k_star = int(np.argmax(vals))
                f_val = vals[k_star]

                if k_star == 2 * i:
                    subgrad = 2.0 * B[k_star] + 2.0 * x_mid
                else:
                    subgrad = 2.0 * B[k_star]

                if subgrad > 0:
                    x_hi = x_mid
                else:
                    x_lo = x_mid

            x_new = max(0.0, 0.5 * (x_lo + x_hi))

            if abs(x_new - old_x) > 1e-15:
                a[i] = x_new
                diff = x_new - old_x
                c += diff * (2.0 * B)
                if 2 * i < len(c):
                    c[2 * i] += diff * old_x
                improved = True

        if improved:
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            c = _full_convolve(a, a)
            c_max_new = float(c.max())
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                c_max = c_max_new

        if not improved:
            break

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _refine_active_set(a, deadline, max_rounds=5):
    """Iteratively refine the active set with tightening tolerance."""
    n = len(a)
    current_tol = 0.005
    best_a = a.copy()
    best_sc = _score(a)

    for round_i in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_ref, sc_ref = _active_set_newton(best_a, min(deadline, time.monotonic() + 30),

        if sc_ref < best_sc - 1e-15:
            best_a = a_ref
            best_sc = sc_ref

        current_tol *= 0.5

    return best_a, best_sc

        if time.monotonic() < deadline - 100:
            cd_deadline2 = min(deadline - 100, time.monotonic() + 30)
            a_cd2, sc_cd2 = _coord_descent(best_seq, cd_deadline2, n_passes=50)
            update_best(a_cd2, sc_cd2, "coord_descent_2")

    # Phase 5: Multi-n search
    print(f"\n[Phase 5] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 8192]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = np.ones(n_try) / n_try

        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}, t={time.monotonic() - t_start:.1f}s")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 30, time.monotonic() + 40),
                                                  max_iter=100, tol_active=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_asian")

            if time.monotonic() < deadline - 20:
                a_opt3, sc_opt3 = _coord_descent(a_opt2, min(deadline - 15, time.monotonic() + 20),
                                                 n_passes=30)
                update_best(a_opt3, sc_opt3, f"n_{n_try}_cd")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("perturb_small", 0.02),
            ("perturb_mid", 0.05),
            ("perturb_large", 0.10),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 20),
                                                  max_iter=80, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_asian")

        # Fourier perturbation
        if time.monotonic() < deadline - 15:
            p_start = _perturb_fourier(best_seq, rng, n_modes=7)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, "restart_fourier")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 3, time.monotonic() + 10),
                                                  max_iter=50, tol_active=0.001)
                update_best(a_r2, sc_r2, "restart_fourier_asian")

    # Phase 7: Final polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very tight active-set Newton
        as_final = min(deadline - 2, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=300, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_asian")

        # Coordinate descent final
        if time.monotonic() < deadline - 2:
            cd_final = min(deadline - 1, time.monotonic() + 30)
            a_fin2, sc_fin2 = _coord_descent(best_seq, cd_final, n_passes=200)
            update_best(a_fin2, sc_fin2, "final_cd")

        # One more very tight Newton
        if time.monotonic() < deadline - 1:
            as_final2 = min(deadline - 0.5, time.monotonic() + 20)
            a_fin3, sc_fin3 = _active_set_newton(best_seq, as_final2, max_iter=200, tol_active=0.0002)
            update_best(a_fin3, sc_fin3, "final_asian2")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
