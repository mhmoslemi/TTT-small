```python
    n, m = len(a), len(b)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]
```

```python
"""
Active-set KKT optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary: Solve KKT linear system with adaptive active set identification.
Secondary: Adam in log-space with FFT-based smoothed gradients.
Tertiary: Coordinate descent and spectral equalization polishing.
Exploits symmetry and sparse linear algebra for efficiency.
"""
import numpy as np
from scipy import sparse
from scipy.sparse.linalg import spsolve, cg
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0


def _build_T_S_sparse(n, S):
    """Build sparse T_S where T_S[i,j] = 1 if (i+j) in S, 0<=i,j<n."""
    all_rows = []
    all_cols = []
    for s in S:
        s = int(s)
        if s < 0 or s > 2 * n - 2:
            continue
        i_min = max(0, s - (n - 1))
        i_max = min(s, n - 1)
        if i_min <= i_max:
            i_vals = np.arange(i_min, i_max + 1)
            j_vals = s - i_vals
            all_rows.append(i_vals)
            all_cols.append(j_vals)
    if not all_rows:
        return sparse.csr_matrix((n, n), dtype=np.float64)
    rows = np.concatenate(all_rows)
    cols = np.concatenate(all_cols)
    data = np.ones(len(rows), dtype=np.float64)
    M = sparse.coo_matrix((data, (rows, cols)), shape=(n, n), dtype=np.float64)
    return M.tocsr()


def _solve_kkt_system(n, S, support=None):
    """
    Solve the KKT linear system:
      T_S @ a = lambda * 1 (on support)
      sum(a) = 1
      a[i] = 0 for i not in support
    Returns (a, lambda) or None if infeasible.
    """
    if support is None:
        support = np.arange(n)
    m = len(support)
    if m == 0:
        return None
    T_S = _build_T_S_sparse(n, S)
    # Extract sub-matrix for support
    T_sub = T_S[support, :][:, support].tocsc()
    # Build augmented system (m+1) x (m+1):
    # [T_sub  -1] [a_sub]   [0]
    # [1^T     0] [lambda] = [1]
    one_col = sparse.csc_matrix(-np.ones((m, 1)), dtype=np.float64)
    one_row = sparse.csc_matrix(np.ones((1, m)), dtype=np.float64)
    zero = sparse.csc_matrix(np.zeros((1, 1)), dtype=np.float64)
    M_aug = sparse.bmat([[T_sub, one_col], [one_row, zero]], format='csc')
    b_aug = np.zeros(m + 1, dtype=np.float64)
    b_aug[-1] = 1.0
    try:
        x = spsolve(M_aug, b_aug)
    except Exception:
        try:
            x, _ = cg(M_aug, b_aug, tol=1e-10, maxiter=5000)
        except Exception:
            return None
    if not np.all(np.isfinite(x)):
        return None
    a_sub = x[:-1]
    lam = x[-1]
    a_full = np.zeros(n, dtype=np.float64)
    a_full[support] = a_sub
    return a_full, lam


def _active_set_optimize(a_init, deadline, max_iter=200, verbose=True):
    """
    Active-set KKT optimization:
    1. Identify active set S from current convolution peaks.
    2. Solve KKT linear system.
    3. Check feasibility and consistency.
    4. Update and iterate.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        score = _score_norm(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Determine active set with adaptive threshold
        # Start broad, narrow as we converge
        if it < 10:
            frac = 0.90
        elif it < 30:
            frac = 0.95
        elif it < 60:
            frac = 0.98
        else:
            frac = 0.995
        S = np.where(c >= c_max * frac)[0]
        if len(S) == 0:
            S = np.array([int(c.argmax())])
        if len(S) > 5000:
            # Too many; keep only top
            top_idx = np.argsort(c)[::-1][:2000]
            S = top_idx

        # Solve KKT system
        a_sol, lam = _solve_kkt_system(n, S)
        if a_sol is None:
            # Try with reduced support
            current_support = np.where(a > 1e-10)[0]
            a_sol, lam = _solve_kkt_system(n, S, support=current_support)
        if a_sol is None:
            if verbose and it % 20 == 0:
                print(f"  AS iter {it}: solve failed, skipping")
            continue

        # Check non-negativity
        min_val = float(a_sol.min())
        if min_val < -1e-8:
            # Project: set negative to 0, normalize
            a_sol = np.maximum(a_sol, 0)
            s = a_sol.sum()
            if s < 1e-15:
                continue
            a_sol = a_sol / s

        # Line search: blend current with solution
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_try = (1 - t) * a + t * a_sol
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc_try = _score_norm(a_try)
            if sc_try < score * (1 - 1e-12):
                a = a_try
                improved = True
                break

        if not improved and it % 10 == 0 and verbose:
            print(f"  AS iter {it}: no improvement, score={score:.10f}, |S|={len(S)}")

        # Convergence check
        if it > 0 and abs(score - best_score) < 1e-14:
            if verbose and it > 20:
                print(f"  AS converged at iter {it}, score={best_score:.10f}")
            break

    return best_a, best_score


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=5, eval_interval=200,
                  clip_norm=0.8, symmetric=False):
    """Adam optimizer in log-space with FFT-based smoothed gradient."""
    n = len(a_init)
    if symmetric:
        half_n = (n + 1) // 2
        a_half = a_init[:half_n].copy()
        z = np.log(np.maximum(a_half, 1e-15))
        z = z - z.mean()

        def full_a_from_z(z_h):
            e = np.exp(z_h - z_h.max())
            e = e / e.sum()
            if n % 2 == 0:
                return np.concatenate([e, e[::-1]])
            else:
                return np.concatenate([e[:-1], e[-1:], e[-2::-1]])
    else:
        z = np.log(np.maximum(a_init, 1e-15))
        z = z - z.mean()

        def full_a_from_z(z_h):
            e = np.exp(z_h - z_h.max())
            return e / e.sum()

    m_buf = np.zeros_like(z)
    v_buf = np.zeros_like(z)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = full_a_from_z(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Smoothed gradient via softmax weighting
        c_shifted = (c - c_max) / max(T, 1e-15)
        e = np.exp(c_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s_e

        # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (valid indices)
        # This is 2 * (a * p_reversed) circular conv, but we need linear
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        # The gradient at position i is 2 * conv_full at appropriate index
        # conv_full[k] = sum_j a[j] * p_rev[k-j] = sum_j a[j] * p[len(p)-1-(k-j)]
        # We want sum_k p[k] * a[k-i] which is the correlation
        # Let's just compute it directly via the FFT approach
        # Actually: sum_k p[k] * a[k-i] = (p * a)[i] where * is cross-correlation
        # cross_correlation(p, a)[i] = sum_k p[k] * conjugate(a[k-i]) = sum_k p[k] * a[k-i] (real)
        # This equals conv(p_rev, a)[i + len(p) - 1 - (len(a)-1)]... 
        # Let me just use: g[i] = 2 * sum_{k: 0<=k-i<n} p[k]*a[k-i]
        # = 2 * (linear cross-correlation of p and a) [i]
        # cross_corr[p, a][i] = sum_k p[k] * a[k-i]
        # = conv(p, reverse(a))[i + len(a) - 1]
        a_rev = a[::-1]
        conv_pa = _fft_convolve(p, a_rev)
        # cross_corr[i] = conv_pa[i + n - 1] for i = 0..n-1
        g_full = 2.0 * conv_pa[n - 1:2 * n - 1]

        if symmetric:
            # Project gradient to symmetric subspace
            g_sym = (g_full[:half_n] + g_full[:half_n][::-1]) / 2.0
            # For odd n, center element gets special handling
            if n % 2 == 0:
                z_grad = a_half * (g_sym - np.dot(a_half, g_sym))
            else:
                # a = [a_half[:-1], a_half[-1], a_half[-2::-1]]
                # Sum constraint: 2*sum(a_half[:-1]) + a_half[-1] = 1
                grad_center = g_full[half_n - 1]
                grad_side = (g_full[:half_n - 1] + g_full[:half_n - 1][::-1]) / 2.0
                W = 2.0 * np.dot(a_half[:-1], grad_side) + a_half[-1] * grad_center
                z_grad_side = a_half[:-1] * (grad_side - W)
                z_grad_center = a_half[-1] * (grad_center - W)
                z_grad = np.concatenate([z_grad_side, [z_grad_center]])
        else:
            W = float(np.dot(a, g_full))
            z_grad_full = a * (g_full - W)
            z_grad = z_grad_full

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros_like(z)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if t % eval_interval == 0:
            a_cur = full_a_from_z(z)
            a_cur = np.maximum(a_cur, 0)
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                if symmetric:
                    z_init = np.log(np.maximum(best_a[:half_n], 1e-15))
                else:
                    z_init = np.log(np.maximum(best_a, 1e-15))
                z = z_init - z_init.mean()
                m_buf = np.zeros_like(z)
                v_buf = np.zeros_like(z)

    a_final = full_a_from_z(z)
    a_final = np.maximum(a_final, 0)
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=300):
    """Coordinate descent polishing on active peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        tol = c_max * 0.99
        active_k = np.where(c > tol)[0]
        if len(active_k) == 0:
            active_k = np.array([c.argmax()])
        active_k = active_k[:200]  # Limit for speed

        improved = False
        for k in active_k:
            if k < n:
                js = range(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = range(k - (n - 1), n)
            js_list = list(js)[:50]  # Limit
            for j in js_list:
                for scale in [0.85, 0.7, 1.15, 1.3, 0.5, 1.5]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    a_cand = np.maximum(a_cand, 0)
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_a, best_score


def _spectral_equalize(a, deadline, max_iter=1000):
    """
    Spectral equalization: flatten the power spectrum of the zero-padded sequence.
    Motivated by: flat |A(f)|^2 => delta-like autocorrelation => minimal peak.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_cur = a.copy()
    best_a = a_cur.copy()
    best_score = _score_norm(a_cur)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        # Compute zero-padded DFT
        A = np.fft.fft(a_cur, N)
        power = np.abs(A) ** 2
        # Target: flat power spectrum
        target = power.mean()
        # Compute gradient: we want to reduce variance of power spectrum
        # d/power/d(a_j) involves phase of A
        # Simplified: adjust a to reduce (power - target)^2
        # Use: a_new = a - lr * grad where grad comes from spectral error
        spectral_error = power - target
        # Gradient of sum(spectral_error^2) w.r.t. a:
        # This is complex; use a simpler heuristic:
        # Invert the spectral error, add to signal
        err_time = np.fft.ifft(spectral_error * np.conj(A) / (np.abs(A) ** 2 + 1e-30), N)[:n]
        err_time = np.real(err_time)
        lr = 0.01 * (1.0 - it / max_iter)
        a_new = a_cur - lr * err_time
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a_cur = a_new

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Active-set KKT linear system (novel approach)
    2. Adam in log-space with symmetry
    3. Spectral equalization
    4. Coordinate descent polishing
    5. Multi-length exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # --- Phase 0: Initialization ---
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
        else:
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        a_init = parent.copy()
        print(f"Loaded parent: n={n_parent}, score={_score(a_init):.10f}")
    else:
        n_parent = 4096
        a_init = np.ones(n_parent) / n_parent
        print("Using uniform initialization, n=4096")

    best_a = a_init.copy()
    best_score = _score(a_init)

    # Symmetrize
    a_sym = _symmetrize(a_init)
    a_sym = a_sym / a_sym.sum()
    sc_sym = _score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_a = a_sym.copy()
        a_init = a_sym.copy()
        print(f"Symmetrized: score={best_score:.10f}")

    # Also try a few other initializations
    for name, init_fn in [
        ("triangular", lambda n: _symmetrize(np.abs(np.linspace(-1, 1, n)) + 0.1)),
        ("sinusoidal", lambda n: _symmetrize(np.abs(np.sin(np.pi * np.arange(n) / n)) + 0.1)),
        ("bimodal", lambda n: _symmetrize(np.exp(-20 * (np.arange(n) / n - 0.3) ** 2) + np.exp(-20 * (np.arange(n) / n - 0.7) ** 2) + 0.05)),
    ]:
        a_try = init_fn(n_parent)
        a_try = np.maximum(a_try, 0)
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            a_init = a_try.copy()
            print(f"  {name} init better: score={best_score:.10f}")

    print(f"After init: n={len(best_a)}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # --- Phase 1: Active-set KKT optimization (0-400s) ---
    phase1_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase1_end:
        print(f"Phase 1: Active-set KKT (n={len(best_a)})")
        a_as, sc_as = _active_set_optimize(best_a, phase1_end, max_iter=300, verbose=True)
        full_sc = _score(a_as)
        print(f"  AS result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_as.copy()

    # --- Phase 2: Adam with symmetry (400s-700s) ---
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        n_cur = len(best_a)
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        print(f"Phase 2: Adam symmetric (n={n_cur})")
        a_adam, sc_adam = _adam_z_space(best_a, phase2_end,
                                         T_start=0.003 * max_c, T_end=0.00003 * max_c,
                                         lr_start=0.03, lr_end=0.001,
                                         max_iter=500000, active_k=5,
                                         eval_interval=150, symmetric=True)
        full_sc = _score(a_adam)
        print(f"  Adam result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # --- Phase 3: Spectral equalization (700s-900s) ---
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Spectral equalization (n={len(best_a)})")
        a_spec, sc_spec = _spectral_equalize(best_a, phase3_end, max_iter=2000)
        full_sc = _score(a_spec)
        print(f"  Spectral result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_spec.copy()

    # --- Phase 4: Another round of active-set on improved solution (900s-1100s) ---
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Active-set round 2 (n={len(best_a)})")
        a_as2, sc_as2 = _active_set_optimize(best_a, phase4_end, max_iter=200, verbose=True)
        full_sc = _score(a_as2)
        print(f"  AS2 result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_as2.copy()

    # --- Phase 5: Multi-length exploration (1100s-1500s) ---
    phase5_end = min(deadline, t_start + 1500.0)
    n_best = len(best_a)
    for n_try in [4096, 8192, 16384]:
        if time.monotonic() > phase5_end - 60:
            break
        if n_try == n_best:
            continue
        print(f"Phase 5: Trying n={n_try}")
        a_try = _interp_to_length(best_a, n_try)
        # Quick Adam on new length
        sub_dl = min(phase5_end - 10, time.monotonic() + 100.0)
        c_try = _fft_convolve(a_try, a_try)
        max_try = float(c_try.max())
        a_try_opt, _ = _adam_z_space(a_try, sub_dl,
                                      T_start=0.002 * max_try, T_end=0.00002 * max_try,
                                      lr_start=0.02, lr_end=0.001,
                                      max_iter=200000, active_k=5,
                                      eval_interval=200, symmetric=True)
        full_sc = _score(a_try_opt)
        print(f"  n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_try_opt.copy()
            n_best = n_try

    # --- Phase 6: CD polish (1500s-1700s) ---
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end:
        print(f"Phase 6: CD polish (n={len(best_a)})")
        a_cd, sc_cd = _cd_polish(best_a, phase6_end, max_rounds=200)
        full_sc = _score(a_cd)
        print(f"  CD result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # --- Phase 7: Final ultra-fine Adam (1700s-deadline) ---
    if time.monotonic() < deadline - 10:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        print(f"Phase 7: Final Adam (n={len(best_a)})")
        a_final, _ = _adam_z_space(best_a, deadline - 8,
                                    T_start=0.0005 * max_cf, T_end=0.000005 * max_cf,
                                    lr_start=0.005, lr_end=0.0002,
                                    max_iter=200000, active_k=5,
                                    eval_interval=100, symmetric=True)
        full_sc = _score(a_final)
        print(f"  Final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # Quick final CD
        if time.monotonic() < deadline - 3:
            a_cd_f, _ = _cd_polish(best_a, deadline - 2, max_rounds=30)
            sc_f = _score(a_cd_f)
            if sc_f < best_score:
                best_score = sc_f
                best_a = a_cd_f.copy()

    # --- Final validation ---
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
