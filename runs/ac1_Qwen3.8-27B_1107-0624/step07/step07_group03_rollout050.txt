def _next_pow2(n):

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        c_new_max = float(c_new.max())
        if abs(c_new_max - prev_cmax) < 1e-13:
            break
        prev_cmax = c_new_max

    return best_a, best_sc


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03,
                                  0.015, 0.008, 0.004, 0.002, 0.001, 0.0005,
                                  0.00025, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(15000, max(100, int(remaining / (est_time_per_iter * 10))))

        prev_obj = None
        stall_count = 0

        try:
            result = minimize(
                _lsobj_wrapper,
                z,
                _lsgrad_wrapper,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 80,
                    'args': (beta, n),
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lsobj_wrapper(z, beta, n):
    return _lsobj(z, beta, n)


def _lsgrad_wrapper(z, beta, n):
    return _lsgrad(z, beta, n)


def _subgrad_polish(a_init, deadline, n_max_steps=8000, lr_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    lr_min = lr_init * 0.01

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if lr < lr_min:
                    lr = lr_min
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * t
        return a / a.sum()
    elif kind == "asym_ramp2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 6))
        amp = 0.015 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _block_reoptimization(a_init, deadline, block_size=64):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    n_blocks = (n + block_size - 1) // block_size
    order = np.random.default_rng(42).permutation(n_blocks)

    for passes in range(20):
        if time.monotonic() > deadline:
            break
        for bi in order:
            if time.monotonic() > deadline:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            if start >= n:
                continue

            c = _full_convolve(a, a)
            c_max = float(c.max())

            # For the block [start:end], try to optimize these coordinates
            # using projected gradient on the block
            z_block = np.log(np.maximum(a[start:end], 1e-30))
            z_block = z_block - z_block.mean()

            for sub_it in range(50):
                a_block = _softmax(z_block) * a[start:end].sum()
                a_try = a.copy()
                a_try[start:end] = a_block
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())

                if f_try < c_max - 1e-14:
                    # Compute subgradient for the block
                    tol = 0.002 * c_try.max()
                    active = c_try >= (c_try.max() - tol)
                    if active.sum() == 0:
                        active[int(np.argmax(c_try))] = True
                    w = np.zeros_like(c_try)
                    w[active] = 1.0 / active.sum()
                    a_rev = a_try[::-1]
                    wc = _full_convolve(w, a_rev)
                    h = wc[n - 1:2 * n - 1]
                    W = float(np.dot(w, c_try))
                    grad = 4.0 * n * a_try * (h - W)
                    grad_block = grad[start:end]
                    lr = 0.01 / (1.0 + 0.01 * sub_it)
                    z_block = z_block - lr * grad_block
                    z_block = np.clip(z_block, -40, 40)
                    z_block = z_block - z_block.mean()
                    a = a_try
                    c_max = f_try
                    if best_sc is not None and f_try * 2.0 * n < best_sc:
                        best_sc = _score(a)
                        best_a = a.copy()
                else:
                    break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init_uniform")

    # Phase 1: Main L-BFGS from parent (allocate ~60% of time)
    print(f"\n[Phase 1] Main L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    p1_deadline = t_start + (deadline - t_start) * 0.55
    a_main = _run_lbfgs_adaptive(parent, p1_deadline, min_time_remaining=50.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton refinement
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        # Multiple tolerances
        for tol_val, max_it in [(0.005, 150), (0.002, 200), (0.001, 200), (0.0005, 150)]:
            if time.monotonic() > deadline - 350:
                break
            as_dl = min(deadline - 350, time.monotonic() + 80)
            a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=max_it, tol_active=tol_val)
            update_best(a_as, sc_as, f"as_newton_tol{tol_val}")

    # Phase 3: Subgradient polish
    print(f"\n[Phase 3] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_dl = min(deadline - 300, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_dl, n_max_steps=5000, lr_init=0.01)
        update_best(a_sg, sc_sg, "sg_polish")

        if sc_sg < best_score and time.monotonic() < deadline - 200:
            a_sg2, sc_sg2 = _active_set_newton(a_sg, min(deadline - 200, time.monotonic() + 60),
                                                max_iter=100, tol_active=0.001)
            update_best(a_sg2, sc_sg2, "sg_polish_as")

    # Phase 4: Try larger n
    print(f"\n[Phase 4] Larger n exploration (t={time.monotonic() - t_start:.1f}s)")
    larger_ns = [16384]
    if n_parent != 4096:
        larger_ns.append(4096)

    for n_try in larger_ns:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        p4_dl = min(deadline - 60, time.monotonic() + 180)
        a_opt = _run_lbfgs_adaptive(a_init, p4_dl, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if sc_opt < best_score + 0.02 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 40, time.monotonic() + 60),
                                                  max_iter=80, tol_active=0.002)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 5: Diverse restarts from best
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restarts = [
            ("perturb_001", lambda: _perturb(best_seq, 0.005, rng)),
            ("perturb_003", lambda: _perturb(best_seq, 0.01, rng)),
            ("perturb_010", lambda: _perturb(best_seq, 0.03, rng)),
            ("fourier_3", lambda: _perturb_fourier(best_seq, rng, n_modes=3)),
            ("fourier_8", lambda: _perturb_fourier(best_seq, rng, n_modes=8)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 20:
                break
            p_start = gen_fn()
            sc_start = _score(p_start)
            if sc_start >= best_score + 0.01:
                continue
            a_r = _run_lbfgs_adaptive(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 20),
                                                  max_iter=50, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 6: Final intensive polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight active-set Newton
        as_dl = min(deadline - 2, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_newton(best_seq, as_dl, max_iter=300, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as_tight")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj_wrapper,
                        z,
                        _lsgrad_wrapper,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 60, 'args': (beta_final, n)}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # One more active-set pass
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _active_set_newton(best_seq, deadline - 1, max_iter=100, tol_active=0.0001)
            update_best(a_fin4, sc_fin4, "final_as_ultra")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
