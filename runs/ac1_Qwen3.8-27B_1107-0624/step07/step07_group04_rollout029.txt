def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

"""
QP-based optimizer using active set method with exact dense solves on reduced support.
Key innovation: Direct QP formulation exploiting the Hankel structure, with
support reduction to enable exact solves, and alternating active set updates.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _ma_vec(a, W, n):
    """Compute M @ a where M[i,j] = W[i+j]. Uses FFT."""
    a_rev = a[::-1]
    out_len = len(W) + len(a_rev) - 1
    N = _next_pow2(out_len)
    F_W = np.fft.rfft(W, N)
    F_a = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(F_W * F_a, N)[:out_len]
    return conv[n-1:2*n-1]

    p = len(indices)

"""
QP-based optimizer exploiting the quadratic structure of the autocorrelation minimization.
Key innovations: (1) Direct QP formulation with active set method using exact dense solves
on reduced support, (2) Hankel structure exploitation for efficient matrix formation,
(3) Alternating active set (convolution peak) updates with QP solves, (4) Multi-scale
support reduction, (5) Parametric restarts to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_len] if False else np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _ma_vec(a, W, n):
    """Compute M @ a where M[i,j] = W[i+j], using FFT (cross-correlation trick)."""
    a_rev = a[::-1]
    out_len = len(W) + len(a_rev) - 1
    N = _next_pow2(out_len)
    F_W = np.fft.rfft(W, N)
    F_a = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(F_W * F_a, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _form_hankel_block(W, start, p):
    """Form p×p matrix with entries H[i,j] = W[start + i + start + j] = W[2*start + i + j].
    This is a Hankel matrix."""
    M = np.zeros((p, p))
    for d in range(2 * p - 1):
        val = W[2 * start + d]
        if abs(val) < 1e-18:
            continue
        if d < p:
            i = np.arange(d + 1)
            j = d - i
            M[i, j] = val
        else:
            i = np.arange(d - p + 1, p)
            j = d - i
            M[i, j] = val
    return M


def _form_general_submatrix(W, indices):
    """Form submatrix M[indices, indices] where M[i,j] = W[i+j]."""
    p = len(indices)
    M = np.zeros((p, p))
    for i in range(p):
        M[i, :] = W[indices[i] + indices]
    return M


def _solve_qp_active_set(M, p, max_iter=None):
    """
    Solve: minimize x^T M x subject to sum(x) = 1, x >= 0.
    Active set method with dense solves.
    Returns x (length p) or None.
    """
    if max_iter is None:
        max_iter = 2 * p
    if p == 0:
        return None
    if p == 1:
        return np.array([1.0])

    active = np.ones(p, dtype=bool)
    x = np.ones(p) / p

    for it in range(max_iter):
        idx = np.where(active)[0]
        n_act = len(idx)
        if n_act == 0:
            return None

        M_aa = M[np.ix_(idx, idx)]
        try:
            # Try Cholesky with small regularization
            reg = 1e-12
            try:
                L = np.linalg.cholesky(M_aa + reg * np.eye(n_act))
                rhs = np.ones(n_act)
                y = np.linalg.solve(L, rhs)
                sol = np.linalg.solve(L.T, y)
            except np.linalg.LinAlgError:
                reg = 1e-8
                M_aa_r = M_aa + reg * np.eye(n_act)
                try:
                    L = np.linalg.cholesky(M_aa_r)
                    rhs = np.ones(n_act)
                    y = np.linalg.solve(L, rhs)
                    sol = np.linalg.solve(L.T, y)
                except np.linalg.LinAlgError:
                    sol = np.linalg.lstsq(M_aa, np.ones(n_act), rcond=None)[0]
        except Exception:
            break

        s = sol.sum()
        if abs(s) < 1e-15:
            break

        sol = sol / s  # Normalize: sum = 1

        # Check non-negativity
        min_val = np.min(sol)
        if min_val < -1e-8:
            # Remove most negative
            worst_local = idx[np.argmin(sol)]
            active[worst_local] = False
            continue
        elif min_val < 0:
            # Clamp and treat as boundary
            sol = np.maximum(sol, 0)
            sol = sol / sol.sum()
            # Mark zeros as potentially inactive
            for k in range(n_act):
                if sol[k] < 1e-12:
                    active[idx[k]] = False
                    idx = np.where(active)[0]
                    n_act = len(idx)
                    if n_act == 0:
                        return None
                    break
            x[idx[np.where(active)[0]]] = 0  # reset
            # Re-solve with updated active
            continue

        # Build candidate
        x_cand = np.zeros(p)
        x_cand[idx] = sol

        # KKT check for inactive elements
        if n_act < p:
            Ma = M @ x_cand
            lam_half = np.mean(Ma[idx])
            inactive_idx = np.where(~active)[0]
            if len(inactive_idx) > 0:
                violations = Ma[inactive_idx] - lam_half
                min_viol = np.min(violations)
                if min_viol < -1e-10:
                    worst = inactive_idx[np.argmin(violations)]
                    active[worst] = True
                    continue

        x = x_cand
        break

    return x


def _solve_qp_on_support(W, n, support, max_qp_iters=None):
    """
    Solve QP: min a^T M a s.t. sum(a)=1, a>=0, a zero outside support.
    M[i,j] = W[i+j].
    Returns full-length a or None.
    """
    p = len(support)
    if p == 0:
        return None
    if p == 1:
        a = np.zeros(n)
        a[support[0]] = 1.0
        return a

    # Check if support is contiguous for faster Hankel formation
    support_sorted = np.sort(support)
    is_contiguous = np.all(np.diff(support_sorted) == 1)

    if is_contiguous and p <= 4000:
        start = support_sorted[0]
        M = _form_hankel_block(W, start, p)
    else:
        M = _form_general_submatrix(W, np.sort(support))

    x = _solve_qp_active_set(M, p, max_iter=max_qp_iters)
    if x is None:
        return None

    a = np.zeros(n)
    a[support_sorted] = x
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s

            if time.monotonic() + min_time > deadline:
                break

            W = _update_active_set(c, n, top_k=top_k)
            a_new = _solve_qp_on_support(W, n, support, max_qp_iters=max(50, 2 * p_size))
            if a_new is None:
                continue

            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-12:
                best_sc = sc_new
                best_a = a_new.copy()
                print(f"    QP: score={best_sc:.10f} (p={p_size}, top_k={top_k}) t={time.monotonic()-_qp_multi_t0:.1f}s")

            # If improved, re-convolve for next iteration
            if sc_new < best_sc + 1e-10:
                a = a_new.copy()
                c = _full_convolve(a, a)

        # Update a to best
        a = best_a.copy()

    return best_a, best_sc


_qp_multi_t0 = 0.0


def _refine_qp(a_init, n, deadline, min_time=30):
    """Refine with alternating active set updates on fixed support."""
    global _qp_multi_t0
    _qp_multi_t0 = time.monotonic()

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    # Determine support
    thresh = a.max() * 1e-4
    support = np.where(a > thresh)[0]
    if len(support) < 50:
        support = np.where(a > 0)[0]
    if len(support) == 0:
        support = np.arange(n)
    # Limit support size
    if len(support) > 3000:
        a_sorted_idx = np.argsort(a)[::-1]
        support = a_sorted_idx[:3000]

    # Alternating: update W, solve QP, repeat
    for top_k in [3, 5, 8, 12, 20, 30]:
        if time.monotonic() + min_time > deadline:
            break

        for outer in range(5):
            if time.monotonic() + min_time > deadline:
                break

            c = _full_convolve(a, a)
            W = _update_active_set(c, n, top_k=top_k)

            a_new = _solve_qp_on_support(W, n, support, max_qp_iters=max(100, 2 * len(support)))
            if a_new is None:
                break

            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-12:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new.copy()

            # Check convergence
            if sc_new > best_sc - 1e-14:
                break

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time_remaining=60):
    """L-BFGS on log-sum-exp smoothed objective (as fallback/complement)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Beta schedule
    gaps = np.sort(c_cur)[::-1]
    gap = float(gaps[0] - gaps[min(10, len(gaps) - 1)])
    base_beta = max(gap * 0.5, max_c * 0.01)
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    betas = base_beta * np.array(ratios)

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(20, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _smoobj_wrapper,
                z,
                args=(beta, n),
                jac=_smograd_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxcor': 40}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _smoobj_wrapper(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_wrapper(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _ma_vec(a, w, n)
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_center":
        a = np.ones(n)
        center = n // 2
        width = n // 3
        a[center - width:center + width] = 2.0
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase QP-based optimizer:
    Phase 1: L-BFGS initialization
    Phase 2: Multi-scale QP with support reduction
    Phase 3: Refined QP with alternating active set updates
    Phase 4: Diverse restarts + final polish
    """
    global _qp_multi_t0
    t_start = time.monotonic()
    _qp_multi_t0 = t_start
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n = len(parent)

    # Phase 1: L-BFGS initialization
    print(f"\n[Phase 1] L-BFGS init (t={time.monotonic() - t_start:.1f}s)")
    a1 = _lbfgs_smooth(parent, deadline, min_time_remaining=600)
    update_best(a1, _score(a1), "lbfgs_init")

    # Phase 2: Multi-scale QP optimization
    print(f"\n[Phase 2] Multi-scale QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a2, sc2 = _multi_scale_qp(best_seq, n, deadline, min_time=400)
        update_best(a2, sc2, "multi_scale_qp")

    # Phase 3: Refined QP with alternating updates
    print(f"\n[Phase 3] Refined QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a3, sc3 = _refine_qp(best_seq, n, deadline, min_time=150)
        update_best(a3, sc3, "refined_qp")

    # Phase 4: Try different n values
    print(f"\n[Phase 4] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 12288, 6144]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n:
            continue
        a_init = _resample(parent, n_try)
        # Quick L-BFGS
        a_opt = _lbfgs_smooth(a_init, deadline, min_time_remaining=100)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        # QP polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 80:
            a_q, sc_q = _multi_scale_qp(a_opt, n_try, deadline, min_time=60)
            update_best(a_q, sc_q, f"n_{n_try}_qp")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_profiles = ["bimodal", "gaussian", "cos", "flat_center"]
        for i, prof in enumerate(restart_profiles):
            if time.monotonic() > deadline - 40:
                break

            if i == 0:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.15, rng)
            else:
                p_start = _make_profile(n, prof)
                # Blend with best
                p_start = 0.7 * best_seq + 0.3 * p_start
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()

            a_r = _lbfgs_smooth(p_start, deadline, min_time_remaining=40)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 30:
                a_rq, sc_rq = _multi_scale_qp(a_r, n, deadline, min_time=25)
                update_best(a_rq, sc_rq, f"restart_{i}_qp")

    # Phase 6: Final ultra-fine QP polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Use smallest active set for maximum selectivity
        c_best = _full_convolve(best_seq, best_seq)
        c_max = float(c_best.max())

        # Try very small active sets
        for top_k in [2, 3, 4, 5, 6, 7, 8]:
            if time.monotonic() > deadline - 3:
                break
            W = _update_active_set(c_best, n, top_k=top_k)

            # Use full support (or top 2000)
            a_sorted_idx = np.argsort(best_seq)[::-1]
            support = a_sorted_idx[:min(2000, n)]

            a_new = _solve_qp_on_support(W, n, support, max_qp_iters=300)
            if a_new is not None:
                sc_new = _score(a_new)
                update_best(a_new, sc_new, f"final_top{k}")

        # Also try with tolerance-based active set
        for tol in [0.0001, 0.0005, 0.001, 0.003, 0.01]:
            if time.monotonic() > deadline - 2:
                break
            W = _update_active_set(c_best, n, tol_frac=tol)
            a_sorted_idx = np.argsort(best_seq)[::-1]
            support = a_sorted_idx[:min(2000, n)]
            a_new = _solve_qp_on_support(W, n, support, max_qp_iters=200)
            if a_new is not None:
                sc_new = _score(a_new)
                update_best(a_new, sc_new, f"final_tol{tol}")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
