```python
    n = len(f)
    # Build Toeplitz matrix A of shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a lower-Hessenberg-like Toeplitz matrix
    
    # For memory efficiency with large n, I'll build it as a dense matrix
    # For n=1024: 2047*1024 = 2,096,128 entries ≈ 16.8 MB - fine
    # For n=2048: 4095*2048 = 8,386,560 entries ≈ 67 MB - fine
    # For n=4096: 8191*4096 = 33,540,336 entries ≈ 268 MB - borderline
    # For n=8192: 16383*8192 = 134,215,936 entries ≈ 1.07 GB - too large
    
    # Build the matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # = f[k-j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = f[k-j_max:k-j_min+1][::-1]

    # Subject to A @ g <= M
    # Subject to g >= 0
    
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:

    else:
```

```python
"""
Optimizer combining LP-based refinement, log-sum-exp continuation,
adaptive Newton active-set, and diverse restarts for minimizing
the peak autocorrelation score.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _build_conv_matrix(f):
    n = len(f)
    rows = np.arange(2 * n - 1)[:, np.newaxis]
    cols = np.arange(n)[np.newaxis, :]
    diff = rows - cols
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[valid] = f[diff[valid].astype(int)]
    return A


def _lp_solve(f, M):
    n = len(f)
    A = _build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_iterations(f, deadline, max_iter=200):
    n = len(f)
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, _score(f)
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(f, f)
        M = float(c.max())
        g = _lp_solve(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = np.maximum(g, 0)
        g = g / g.sum()
        
        # Check if g is similar to f (no progress)
        cos_sim = float(np.dot(f, g) / (np.linalg.norm(f) * np.linalg.norm(g)))
        if cos_sim > 0.9999:
            break
        
        # Line search
        improved = False
        for t in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2]:
            f_try = (1 - t) * f + t * g
            f_try = np.maximum(f_try, 0)
            st = f_try.sum()
            if st < 1e-10:
                continue
            f_try = f_try / st
            sc_try = _score(f_try)
            if sc_try < best_sc - 1e-12:
                f = f_try
                best_sc = sc_try
                best_f = f.copy()
                improved = True
                break

                st = f_try.sum()
                if st < 1e-10:
```

        try:
            result = minimize(
                _fun_wrapper, z, jac=_jac_wrapper, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    a_final = _softmax(z)
    return a_final, _score(a_final)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "truncated_gauss":
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bilinear":
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 - 0.3 * np.cos(2.0 * np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.04 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, max(1, n - 2 * block_size))
        j = rng.integers(0, max(1, n - 2 * block_size))
        if abs(int(i) - int(j)) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _interpolate(a_old, n_new):
    x_old = np.linspace(0.0, 1.0, len(a_old))
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    g = g - np.mean(g)
    return g
