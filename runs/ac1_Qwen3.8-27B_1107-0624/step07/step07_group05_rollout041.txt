def _next_pow2(n):
    k = 1

def _full_convolve(a, b):
    n, m = len(a), len(b)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

                a_b2, sc_b2 = _barrier_optimize(a_lbf, deadline, min_remaining=400)
                _update(a_b2, sc_b2)
                if time.monotonic() < deadline - 350:
                    a_l2 = _lbfgs_polish(a_b2, deadline, min_time_remaining=350)
                    _update(a_l2, _score(a_l2))

    # === Phase 3: Pressure equalization + subgradient ===
    print(f"\n[Phase 3] Pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_pe = _pressure_equalize(best_seq, min(deadline - 300, time.monotonic() + 60),
                                  n_iters=80, gamma=0.35)
        sc_pe = _score(a_pe)
        _update(a_pe, sc_pe)
        if sc_pe < best_score + 1e-6 and time.monotonic() < deadline - 250:
            a_pg, sc_pg = _subgrad_polish(a_pe, min(deadline - 250, time.monotonic() + 60),
                                          n_max_steps=2000, lr_init=0.015)
            _update(a_pg, sc_pg)

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    a_multi, sc_multi = _multi_n_optimize(best_seq, best_score, deadline, min_remaining=200)
    _update(a_multi, sc_multi)

    # === Phase 5: Final combined polish on best ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        # Barrier
        if time.monotonic() < deadline - 40:
            a_fb, sc_fb = _barrier_optimize(best_seq, min(deadline - 20, time.monotonic() + 60),
                                            min_remaining=20)
            _update(a_fb, sc_fb)
        # L-BFGS
        if time.monotonic() < deadline - 10:
            a_fl = _lbfgs_polish(best_seq, min(deadline - 5, time.monotonic() + 60),
                                 min_time_remaining=5)
            _update(a_fl, _score(a_fl))
        # Subgradient
        if time.monotonic() < deadline - 5:
            a_fs, sc_fs = _subgrad_polish(best_seq, deadline - 3, n_max_steps=1000, lr_init=0.008)
            _update(a_fs, sc_fs)

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
