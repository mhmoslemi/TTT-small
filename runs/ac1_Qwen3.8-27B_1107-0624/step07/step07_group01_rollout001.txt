def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _peak_reduce(a, k, alpha):
    """Reduce the autocorrelation peak at lag k."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max_idx = int(np.argmax(c))
    lag = c_max_idx - (n - 1)  # actual lag
    
    if lag == 0:
        # Peak at DC: this is sum(a_i^2), minimized by uniform a
        # But we can't make it uniform if we want to reduce other peaks
        return a
    
    # The contribution to c[lag] is sum_i a[i] * a[i+lag]
    # Gradient w.r.t. a[i] of c[lag] is a[i-lag] + a[i+lag] (with boundary handling)
    grad = np.zeros(n)
    if lag > 0:
        grad[lag:] += a[:-lag]
        grad[:n-lag] += a[lag:]
    elif lag < 0:
        lag = -lag
        grad[lag:] += a[:-lag]
        grad[:n-lag] += a[lag:]
    
    # Normalize and take step
    grad = grad / (grad.max() + 1e-30)
    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with adaptive log-sum-exp smoothing (primary)
2. Multi-scale active-set subgradient with backtracking line search
3. Peak equalization via projected gradient
4. Diverse initializations including parent, chirp, multi-harmonic
5. Adaptive length selection and basin hopping
6. Newton-KKT refinement for final polish
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(z, beta, n):
    """Compute gradient of smooth max objective w.r.t. log-parameters z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    smooth_val = c_max + beta * np.log(s) if s > 1e-300 else c_max
    return grad, smooth_val, a, c


def _adam_smooth(a_init, deadline, n_levels=25, iters_per_level=3000):
    """Adam optimization with geometric beta schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())

    # Geometric beta schedule - more levels for better convergence
    ratio = (1e-8) ** (1.0 / n_levels)
    betas = max_c_init * (1e-2 * ratio ** np.arange(n_levels + 1))
    betas = np.array([max_c_init * 0.05] + list(betas[1:]))

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 0.08
    t_global = 0

    for level, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        # Scale iterations: fewer for large beta, more for small
        frac = max(0.3, 1.0 - level / (n_levels * 1.5))
        iters = int(iters_per_level * frac)
        iters = max(500, min(iters, 5000))

        m = np.zeros(n)
        v = np.zeros(n)

        for i in range(iters):
            if time.monotonic() > deadline:
                break
            t_global += 1

            grad, smooth_val, a, c = _compute_grad_z(z, beta, n)

            # Adam update
            lr_t = lr_base * 0.95 ** (level * 0.3)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** (i + 1))
            v_hat = v / (1 - beta2 ** (i + 1))
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.mean()
            z = np.clip(z, -80.0, 80.0)

            # Check score periodically
            if i % 200 == 0 or i == iters - 1:
                a_check = _softmax(z)
                sc = _score(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()

    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=30000):
    """Subgradient method with adaptive step size and backtracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    step_size = 0.04
    stall = 0
    best_step = step_size
    improve_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        if c_max < 1e-15:
            break

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1: 2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Normalize gradient
        grad = grad / (gnorm / n + 1e-30)

        # Try step with backtracking
        a_improved = False
        for attempt in range(10):
            z_try = z - step_size * grad
            z_try = z_try - z_try.mean()
            z_try = np.clip(z_try, -80.0, 80.0)
            a_try = _softmax(z_try)
            sc_try = _score(a_try)

            if sc_try < best_score - 1e-15:
                z = z_try
                best_score = sc_try
                best_a = a_try.copy()
                stall = 0
                improve_count += 1
                a_improved = True
                break
            step_size *= 0.7

        if a_improved:
            best_step = step_size
            step_size = min(step_size * 1.02, 0.1)
        else:
            stall += 1
            step_size = max(0.005, best_step * 0.5)
            if stall > 50:
                # Perturb to escape
                z = z + 0.01 * np.random.standard_normal(n)
                z = z - z.mean()
                stall = 0
            if step_size < 1e-6:
                break

    return best_a, best_score


def _peak_equalize(a_init, deadline, max_iter=1000):
    """Iteratively equalize the top autocorrelation peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(a)
    idx_center = n - 1

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:10]
        top_vals = c[top_idx]
        spread = top_vals.max() - top_vals.min()

        if spread < 1e-10 * c_max:
            break

        # For each top peak at lag k, compute the "pressure" vector
        # and try to equalize
        total_grad = np.zeros(n)
        for k in top_idx:
            lag = k - idx_center
            if lag == 0:
                # DC component: reduce sum(a^2) by spreading out
                total_grad += a
            else:
                # Cross-correlation gradient at this lag
                if lag > 0:
                    g = np.zeros(n)
                    g[lag:] += a[:-lag]
                    g[:n - lag] += a[lag:]
                else:
                    lag = -lag
                    g = np.zeros(n)
                    g[lag:] += a[:-lag]
                    g[:n - lag] += a[lag:]
                # Weight by how much this peak exceeds the target
                excess = top_vals[top_idx == k] - top_vals.min()
                if excess.ndim == 0:
                    excess = float(excess)
                else:
                    excess = float(excess[0])
                total_grad += max(excess, 0) * g

        gnorm = np.linalg.norm(total_grad)
        if gnorm < 1e-15:
            break

        # Line search
        alpha = 0.5 / n
        improved = False
        for _ in range(15):
            a_try = a - alpha * total_grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            alpha *= 0.5
            if alpha < 1e-10:
                break

    return best_a, best_score


def _make_chirp(n, freq=1.5, eps=0.4):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiharmonic(n, harmonics=None, eps=0.3):
    if harmonics is None:
        harmonics = [(1, 0.4), (3, 0.2), (5, 0.1)]
    t = np.arange(n)
    a = np.ones(n)
    for f, amp in harmonics:
        a += amp * eps * np.cos(2.0 * np.pi * f * t / n)
    # Apply Hann taper
    taper = 0.5 + 0.5 * np.cos(2.0 * np.pi * t / n)
    a = a * taper
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_carrier(n, freq=3.0, taper_power=1.5):
    t = np.arange(n) / n
    taper = np.sin(np.pi * t) ** taper_power
    a = taper * (1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t * n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")

            if time.monotonic() < deadline - 50:
                a_opt_n, sc_opt_n = _adam_smooth(a_interp, deadline - 50,
                                                  n_levels=12, iters_per_level=2000)
                try_update(a_opt_n, f"n_{target_n}")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(4):
            if time.monotonic() > deadline - 5:
                break

            # Subgradient with very small steps
            a_s, _ = _subgrad_polish(best_seq, deadline - 5, k_active=3, max_steps=15000)
            try_update(a_s, f"final_sg_{rnd}")

            if time.monotonic() > deadline - 5:
                break

            # Peak equalization
            a_pe, _ = _peak_equalize(best_seq, deadline - 5, max_iter=1000)
            try_update(a_pe, f"final_pe_{rnd}")

            if time.monotonic() > deadline - 5:
                break

            # Tight L-BFGS
            a_lb, _ = _lbfgs_polish(best_seq, deadline - 5, beta_frac=1e-8)
            try_update(a_lb, f"final_lbfgs_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_chirp(2048, 1.5, 0.4)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
