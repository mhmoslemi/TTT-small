"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
refined L-BFGS with adaptive beta continuation, palindromic exploitation,
direct-space peak equalization, and multi-restart polishing.
Improvements: larger L-BFGS memory, periodic restarts, palindromic
constraint exploitation, and more aggressive final polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _soft_max_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # h_i = sum_k w_k * a[k-i] (cross-correlation)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_direct(a, n, w):
    """Subgradient of max(conv(a,a)) w.r.t. a (not z)."""
    # g_i = 2 * sum_k w_k * a[k-i] - 2 * dot(w, c) * ... 
    # Actually for the unnormalized problem (sum(a)=1), the gradient of max(c) w.r.t. a_i is:
    # sum_{k in active} (1/|A|) * 2*a[k-i]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + 0.5 * t**4)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_with_restart(a_init, beta, n, max_iter, restart_every=100):
    """L-BFGS with periodic restart of the memory."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    z_cur = z0.copy()
    
    for start_iter in range(0, max_iter, restart_every):
        remaining = max_iter - start_iter
        if remaining <= 0:
            break
        
        def fun(z):
            return _soft_max_obj(z, beta, n)
        
        def jac(z):
            return _soft_max_grad(z, beta, n)
        
        try:
            result = minimize(fun, z_cur, jac=jac, method='L-BFGS-B',
                             options={'maxiter': remaining, 'ftol': 1e-16,
                                      'gtol': 1e-12, 'maxcor': 200,
                                      'maxls': 50, 'disp': False})
            z_cur = result.x
            if result.success and 'nit' in result:
                if result.nit < remaining:
                    break  # Converged
        except Exception:
            break
    
    a_out = _softmax(z_cur)
    return a_out

def _peak_equalize(a_init, deadline, max_iters=20000):
    """Direct-space peak equalization with adaptive step."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Adaptive step size
    eta = 0.01
    stall = 0
    stall_limit = 50
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Active set: top peaks within tolerance
        tol = max(1e-6 * c_max, (c_max - np.sort(c)[::-1][min(10, len(c)-1)]))
        active = c >= (c_max - tol)
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient in a-space: g_i = 2 * sum_k w_k * a[k-i]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h
        
        # Center the gradient (to maintain sum constraint)
        g = g - g.mean()
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        
        g = g / g_norm
        
        # Try multiple step sizes
        best_step = 0
        best_a_try = a
        best_sc_try = _score(a)
        
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_step = step
        
        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-10 or stall > stall_limit:
                break
    
    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, max_steps=50000, lr_init=0.005):
    """Nesterov-accelerated subgradient method in a-space."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov momentum variable
    
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    beta_momentum = 0.0  # Start without momentum, ramp up
    stall = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        t_k = 2.0 / (2.0 + i + np.sqrt(4.0 + i * i))
        y = a + (beta_momentum) * (a - y + a)  # This isn't quite right
        
        # Actually, standard Nesterov:
        # y_k = a_k + (t_{k-1} - 1) / t_k * (a_k - a_{k-1})
        # a_{k+1} = prox(y_k - t_k * g_k)
        
        # Simplified: just use momentum
        if i > 0:
            beta_momentum = (i + 1.0) / (i + 1.0 + 2.0)
        
        c = _full_convolve(y, y)
        c_max = c.max()
        
        # Active set
        sorted_c = np.sort(c)[::-1]
        n_active = max(5, min(20, int(0.01 * len(c))))
        threshold = sorted_c[min(n_active - 1, len(c) - 1)]
        tol = max(1e-5 * c_max, c_max - threshold)
        active = c >= (c_max - tol)
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Subgradient
        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h
        g = g - g.mean()  # Project to sum=0 subspace
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        
        step = lr / (1.0 + 0.01 * i)
        a_new = y - step * g / g_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Update Nesterov variable
        y = a_new + beta_momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        a = a_new
        
        # Check improvement
        if i % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                lr *= 1.1
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    lr *= 0.7
                    if lr < 1e-8:
                        break
    
    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()


def _perturb(a, rng, scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    return _softmax(z)
