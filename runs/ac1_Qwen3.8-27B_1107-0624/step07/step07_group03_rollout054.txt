"""
Optimized step function search using active-set linear system solving,
L-BFGS with aggressive smoothing, and spectral shaping.
Target: minimize 2*n*max(conv(a,a))/sum(a)^2
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

    # g[i] = sum_j a[j] * active_mask[i-j+n-1] (convolution-like)

    # f[i] = sum_j a[j] * active_mask[j+i] for j+i in [0, 2n-2]
    # This is the correlation, which can be done as convolution with reversed signal

    am_rev = active_mask[::-1].astype(np.float64)
    f_full = _full_convolve(a, am_rev)

    # f[i] = f_full[i + (2n-2)] ... need to figure out the indexing
    
    # Actually, let me think about this more carefully.
    # active_mask has length 2n-1, indexed 0 to 2n-2.
    # f[i] = sum_{j: 0<=j<n, 0<=j+i<=2n-2} a[j] * active_mask[j+i]
    # 
    # Let's use: f = np.correlate(a, active_mask, 'full')
    # np.correlate(a, b, 'full')[k] = sum_j a[j] * b[j+k-n_b+1] ... 
    # Actually numpy's correlate is:
    # np.correlate(a, v, 'full')[k] = sum_j a[j+k] * conj(v[j]) for valid k
    # 
    # This is getting confusing. Let me just use the direct FFT approach.
    
    # Method: pad a and active_mask to avoid circular effects
    # f[i] = sum_j a[j] * am[j+i]
    # Pad am to length N, compute FFT, multiply by FFT(a) shifted
    # 
    # Actually, the cleanest way:
    # f[i] = <a, T_i(am)> where T_i shifts am by i
    # In frequency domain: f[i] = ifft(fft(a) * conj(fft(T_i(am))))[0]
    # fft(T_i(am)) = exp(-2πi*i/N) * fft(am) (for circular shift)
    # But we want linear shift, so we need to be careful.
    
    # Let me just use the convolution approach:
    # f[i] = sum_j a[j] * am[j+i]
    # = sum_j a[j] * am[j+i]
    # If we reverse a: a_rev[j] = a[n-1-j]
    # Then f[i] = sum_j a[n-1-j] * am[n-1-j+i] = sum_j a_rev[j] * am[i + n-1-j]
    # This is the cross-correlation of a_rev and am at shift i+n-1
    # = convolve(a_rev, am)[i + n-1] (approximately)
    
    a_rev = a[::-1]
    conv_ra = _full_convolve(a_rev, active_mask.astype(np.float64))
    # conv_ra[k] = sum_j a_rev[j] * am[k-j] = sum_j a[n-1-j] * am[k-j]
    # Let p = n-1-j, j = n-1-p:
    # = sum_p a[p] * am[k-(n-1-p)] = sum_p a[p] * am[k-n+1+p]
    # We want f[i] = sum_p a[p] * am[p+i]
    # So k-n+1 = i, k = i+n-1
    f = conv_ra[n-1:n-1+n]  # length n
    
    # Now g[i] = sum_j a[j] * am[i-j+n-1] (convolution)
    # g[i] = (a * am_pad)[i] where am_pad is am shifted
    # g[i] = sum_j a[j] * am[i-j] (with am[-m] = 0)
    # = convolve(a, am)[i + (n-1)] ... 
    # Actually: convolve(a, am)[k] = sum_j a[j] * am[k-j]
    # We want g[i] = sum_j a[j] * am[i-j] where am is indexed 0 to 2n-2
    # But i-j can be negative (when j > i), so we need am[i-j] = 0 for i-j < 0
    # And i-j can be up to i (when j=0), which is at most n-1 < 2n-1, so no issue on the upper end.
    # The issue is: when j > i, i-j < 0, and am[i-j] should be 0.
    # But in the standard convolution, convolve(a, am)[i] = sum_{j: 0<=j<n, 0<=i-j<len(am)} a[j]*am[i-j]
    # = sum_{j: max(0, i-len(am)+1) <= j <= min(i, n-1)} a[j]*am[i-j]
    # = sum_{j: max(0, i-2n+2) <= j <= min(i, n-1)} a[j]*am[i-j]
    # For i in 0..n-1: this is sum_{j=0}^{min(i,n-1)} a[j]*am[i-j] = sum_{j=0}^{i} a[j]*am[i-j]
    # Because i < n <= len(am) = 2n-1 (for n >= 1)
    # So g[i] = convolve(a, am)[i] for i = 0, ..., n-1
    
    conv_aa = _full_convolve(a, active_mask.astype(np.float64))
    g = conv_aa[0:n]
    
    # KKT residual: f[i] + g[i] should be constant (= 2*lambda) for all i in support
    total = f + g
    return total

def _active_set_solve(a_init, deadline, max_iter=200, tol=0.005):
    """Solve the active-set KKT system iteratively."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify active set
        threshold = c_max - tol * c_max
        active_mask = (c >= threshold).astype(np.float64)
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask[int(np.argmax(c))] = 1.0
            n_active = 1
        
        # Compute KKT residual
        total = _kkt_residual(a, active_mask)
        
        # The target is total[i] = 2*lambda for all i
        # lambda = mean of total (approximately)
        lam = float(total.mean())
        
        # Residual: total[i] - lam (should be 0 at optimum)
        residual = total - lam
        res_norm = np.linalg.norm(residual)
        
        if res_norm < 1e-10 * c_max:
            break
        
        # Use the residual as a descent direction
        # We want to move a to reduce the residual
        # The "Hessian" of the KKT objective with respect to a is related to the active set
        # For now, use a simple gradient step on the smoothed objective
        
        # Alternative: use the residual to directly update a
        # a_new[i] = a[i] - step * residual[i] (projected)
        step = 0.1 / (1 + 0.01 * it)
        a_new = a - step * residual
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        if sc_new < _score(a) + 1e-12:
            a = a_new
        else:
            # Try smaller step
            step *= 0.5
            a_new = a - step * residual
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            a = a_new
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc
