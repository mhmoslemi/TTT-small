"""
Advanced optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS smoothing, LP-based iterative refinement (active-set LP),
improved active-set gradient descent, and multi-n exploration.
Key innovation: LP-based direction finding using the convolution constraint structure.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)
    
    # Adaptive beta schedule: start with larger beta, decrease
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        # More iterations for larger beta (smoother), fewer for smaller
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))
        
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

**Projected Gradient Ascent for the LP:**
- Maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
- Gradient of objective: 1 (all ones).
- At each step: g_new = g + alpha * 1, then project onto the feasible set.

The projection onto {(g) : (a*g)[k] <= M for all k, g >= 0} is not trivial. But we can approximate it with a few iterations of the **projected subgradient** or **Alternating Direction Method**.

Actually, a simpler approach: at each iteration, find the most violated constraint and project onto it.

**Algorithm:**
1. g = uniform (or any feasible point).
2. Repeat:
   a. g = g + alpha (gradient step in direction of maximizing sum).
   b. g = max(g, 0) (project onto non-negativity).
   c. Find k* = argmax_k (a*g)[k].
   d. If (a*g)[k*] > M, project g onto the hyperplane (a*g)[k*] = M.
      The projection is: g = g - ((a*g)[k*] - M) / ||a_shift||^2 * a_shift
      where a_shift is the vector (a[k*-i]) for valid i.
   e. Iterate b-d a few times.

This is essentially the **Hitchcock-Karush** or **successive projection** algorithm.

Let me implement this more carefully:

For constraint k, the vector in the constraint (a*g)[k] = sum_i a[k-i]*g[i] has coefficient vector v_k where v_k[i] = a[k-i] if 0 <= k-i < n, else 0.

Projecting g onto the halfspace v_k^T g <= M:
- If v_k^T g > M: g = g - (v_k^T g - M) / ||v_k||^2 * v_k
- Then clip to g >= 0.

This is a projection onto the intersection of one halfspace and the positive orthant, done iteratively.

"""
Advanced optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous attempt:
1. LP-based iterative direction finding (projected gradient on convolution constraints)
2. Multi-scale optimization (coarse-to-fine)
3. Improved adaptive beta schedule for L-BFGS smoothing
4. Momentum-based active-set polishing
5. Aggressive multi-n exploration with truncation

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)
    def grad(self, z):
        return _smograd(z, self.beta, self.n)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)
    # More aggressive beta reduction schedule
    betas = base_beta * np.array([1.0, 0.6, 0.36, 0.216, 0.13, 0.078,
                                  0.047, 0.028, 0.017, 0.010, 0.006,
                                  0.0036, 0.0022, 0.0013, 0.0008, 0.0005, 0.0003])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(40, int(remaining / (est_time_per_iter * 8))))
        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a

def _lp_direction_fast(a, M, n_steps=80):
    """Find LP direction via projected gradient: max sum(g) s.t. (a*g)<=M, g>=0."""
    n = len(a)
    g = np.ones(n) / n
    c = _full_convolve(a, g)
    cmax = c.max()
    if cmax > M * 1.001:
        g = g * (M / cmax)
    lr = 0.5 / (n * max(a.max(), 1e-15))
    for _ in range(n_steps):
        g = g + lr
        g = np.maximum(g, 0)
        c = _full_convolve(a, g)
        if c.max() > M:
            k = int(np.argmax(c))
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            v = np.zeros(n)
            v[i_start:i_end+1] = a[k - i_end:k - i_start+1][::-1]
            vns = np.dot(v, v)
            if vns > 1e-30:
                g = g - (c[k] - M) / vns * v
            g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def _lp_iteration(a, deadline, max_rounds=30, blend_rate=0.15):
    """LP-based iterative refinement."""
    n = len(a)
    a_cur = a.copy()
    a_cur = a_cur / a_cur.sum()
    best_a = a_cur.copy()
    best_sc = _score(a_cur)
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())
        g = _lp_direction_fast(a_cur, M, n_steps=60)
        if g is None:
            break
        # Try different blend rates
        best_blend = None
        for t in [blend_rate, blend_rate * 0.5, blend_rate * 0.25, blend_rate * 1.5]:
            a_try = (1 - t) * a_cur + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if best_blend is None or sc_try < best_blend[1]:
                best_blend = (a_try, sc_try, t)
        if best_blend is None:
            break
        a_new, sc_new, t_used = best_blend
        if sc_new < best_sc - 1e-12:
            best_a = a_new.copy()
            best_sc = sc_new
            a_cur = a_new
            blend_rate = min(0.5, blend_rate * 1.1)
        else:
            blend_rate *= 0.8
            if blend_rate < 0.01:
                break
    return best_a, best_sc

def _active_set_polish(a_init, deadline, tol_frac_start=0.01, tol_frac_end=0.00003, lr_init=0.05):
    """Active-set gradient descent with momentum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()

    best_sc = 2.0 * n * float(c0.max())

    tol_frac = tol_frac_start
    no_improve = 0
    # Momentum
    momentum = np.zeros(n)
    mu = 0.5

        if time.monotonic() > deadline:
            break

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        # Center gradient
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break
        grad_t = grad_t / gnorm
        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad_t
        alpha = lr
        improved = False
        for _ in range(40):
            a_try = a - alpha * momentum
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.8)
                lr *= 0.8
                mu = max(0.1, mu * 0.9)
            if no_improve >= 300:
                a = best_a.copy()
                momentum = np.zeros(n)
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                mu = 0.3
                no_improve = 0

                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.01)
            mu = min(0.7, mu * 1.005)
    return best_a, _score(best_a)

def _multiscale_optimize(a_init, deadline, min_time_remaining=100.0):
    """Coarse-to-fine multi-scale optimization."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    # Go to coarse
    scales = []
    cur_n = n
    while cur_n > 512 and time.monotonic() + min_time_remaining < deadline:
        cur_n = max(512, cur_n // 2)
        scales.append(cur_n)
    # Optimize at each scale from coarse to fine
    cur_a = _resample(a_init, scales[0]) if scales else a_init.copy()
    for i, target_n in enumerate(scales):
        if time.monotonic() + min_time_remaining > deadline:
            break
        sub_deadline = min(deadline - min_time_remaining, time.monotonic() + 120)
        cur_a = _run_lbfgs(cur_a, sub_deadline, min_time_remaining=20.0)
        sc_cur = _score(cur_a)
        # Upsample to next scale
        if i + 1 < len(scales):
            cur_a = _resample(cur_a, scales[i+1])
        else:
            cur_a = _resample(cur_a, n)
        sc_resampled = _score(cur_a)
        if sc_resampled < best_sc:
            best_a = cur_a.copy()
            best_sc = sc_resampled
    return best_a, best_sc

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _trim_zeros(a):
    """Remove leading/trailing near-zero elements."""
    threshold = 1e-8 * a.max()
    mask = a > threshold
    if not mask.any():
        return a
    first = np.argmax(mask)
    last = len(mask) - 1 - np.argmax(mask[::-1])
    trimmed = a[first:last+1]
    return trimmed

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_mix":
        t = np.linspace(0, 1, n)
        a = np.exp(-30*(t-0.3)**2) + 0.7*np.exp(-30*(t-0.7)**2) + 0.3*np.exp(-20*(t-0.5)**2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _fourier_perturb(a, rng, n_modes=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 6))
        amp = 0.04 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to improve score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    return best_a, best_sc
