"""
Adam-based optimizer with active-set polishing for minimizing
2*n*max(convolve(a,a))/sum(a)^2.
Uses log-sum-exp smoothing in log-space with Adam adaptive gradients,
interleaved active-set GD, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    result = 1
    while result < n:
        result <<= 1
    return max(2, result)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max, a, c, c_max
    return 2.0 * n * (c_max + beta * np.log(s)), a, c, c_max


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(a_init, deadline, min_time_remaining=30.0):
    """Adam optimizer in log-space with adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    c_max_init = float(c_cur.max())
    if c_max_init < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max_init * 0.002:
        gap = c_max_init * 0.002
    base_beta = max(gap, c_max_init * 0.005)

    # Beta schedule: geometric decay
    num_phases = 12
    betas = base_beta * (0.65 ** np.arange(num_phases))

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    lr_init = 0.01
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    current_a = a_init.copy()
    current_sc = _score(current_a)

    for phase, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iters = min(2000, max(20, int(remaining / (est_time * 5))))

        # Reset or continue Adam state
        if phase == 0:
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            t = 0
            lr = lr_init
        else:
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            t = 0
            lr = lr_init * max(0.3, 0.7 ** phase)

        for it in range(max_iters):
            if time.monotonic() + 2.0 > deadline - min_time_remaining:
                break

            t += 1
            obj, a, c, c_max = _smooth_obj(z, beta, n)
            grad = _smooth_grad(z, beta, n)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.mean()  # recenter to maintain gauge freedom

            # Check improvement every few iterations
            if it % 50 == 0 or it == max_iters - 1:
                a_new = _softmax(z)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new

        # After each phase, also try a few steps of active-set
        if time.monotonic() < deadline - min_time_remaining:
            current_a = _active_set_steps(current_a, 20, deadline - min_time_remaining)
            current_sc = _score(current_a)
            z = np.log(np.maximum(current_a, 1e-30))
            z = z - z.mean()

    return current_a


def _active_set_steps(a_init, num_steps, deadline):
    """Quick active-set gradient descent steps."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    lr = 0.03

    for step in range(num_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Select active set
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)

        # Line search
        alpha = lr
        improved = False
        for _ in range(10):
            a_try = np.maximum(a - alpha * grad_t, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.8
            if lr < 0.001:
                break

    return best_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01, tol_frac_end=0.00003, lr_init=0.05):
    """Extended active-set polishing with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < best_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.8)
                lr *= 0.85
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.5
                no_improve = 0
                if tol_frac <= tol_frac_end * 1.1 and lr < lr_init * 0.1:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.03)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage_optimize(a, deadline):
    """Try setting small elements to zero."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(0, 1, n)
        a = 1.0
        mask = (t > 0.1) & (t < 0.9)
        a[~mask] = 0.3
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Hybrid optimizer combining Adam in log-space, LP-based water-filling,
weighted active-set GD, peak deflation, and multi-n exploration to minimize
2*n*max(convolve(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    result = 1
    while result < n:
        result <<= 1
    return max(2, result)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    z = np.log(np.maximum(a_init, 1e-30))

def _water_filling_lp(a_init, deadline, lp_n=None):
    """Water-filling using LP relaxation at smaller n."""
    try:
        import cvxpy as cp
    except ImportError:
        return a_init, _score(a_init), False

    n_full = len(a_init)
    if lp_n is None:
        lp_n = min(2048, n_full)
    a_small = _resample_simple(a_init, lp_n)
    n = lp_n
    a = a_small.astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init), False
    a = a / s
    best_a = a.copy()
    c = _full_convolve(a, a)
    M = float(c.max())
    best_sc = 2.0 * n * M
    t_mix = 0.3
    it_count = 0
    while time.monotonic() < deadline - 5:
        it_count += 1
        c = _full_convolve(a, a)
        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        # Build LP: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k
        b = cp.Variable(n)
        # Convolution a*b has 2n-1 entries
        # (a*b)[k] = sum_j a[j] * b[k-j] for valid j, k-j
        # We need this <= M for k = 0, ..., 2n-2
        # This is: A_lp @ b <= M * ones(2n-1)
        # where A_lp[k, j] = a[j] if 0 <= k-j < n, else 0
        # Build asToeplitz-like matrix
        constraints = [b >= 0]
        # (a*b)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * b_{k-j}
        # Let's build the constraint matrix
        # For k = 0: a[0]*b[0] <= M
        # For k = 1: a[0]*b[1] + a[1]*b[0] <= M
        # ...
        # For k = n-1: a[0]*b[n-1] + a[1]*b[n-2] + ... + a[n-1]*b[0] <= M
        # For k = n: a[1]*b[n-1] + ... + a[n-1]*b[0] <= M
        # ...
        # For k = 2n-2: a[n-1]*b[n-1] <= M
        # We can express this using a Toeplitz-like structure
        # (a * b)_k = sum_j a[j] * b[k-j]
        # = sum_j a[j] * b_j * delta(k-j, j') -- this is convolution
        # Using FFT-based approach won't work for LP, need explicit matrix
        # For n=2048, the matrix is (2n-1) x n = 4095 x 2048, which is fine
        n_constraints = 2 * n - 1
        # A_lp[k, j] = a[j] if 0 <= k-j < n and 0 <= j < n
        # i.e., max(0, k-n+1) <= j <= min(n-1, k)
        # Build sparsely
        rows = []
        cols = []
        vals = []
        for k in range(n_constraints):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
        A_lp = np.zeros((n_constraints, n))
        for idx in range(len(rows)):
            A_lp[rows[idx], cols[idx]] = vals[idx]
        constraints.append(A_lp @ b <= M)
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-8, reltol=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=5000)
            except Exception:
                break
        if b.value is None:
            break
        b_val = np.maximum(b.value, 0.0)
        s_b = b_val.sum()
        if s_b < 1e-10:
            break
        b_val = b_val / s_b
        # Mix
        a = (1.0 - t_mix) * a + t_mix * b_val
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-10:
            a = best_a.copy()
            break
        a = a / s_a
        # Gradually increase mixing
        t_mix = min(0.8, t_mix * 1.05)
    # Interpolate back
    if len(best_a) != n_full:
        result = _resample_simple(best_a, n_full)
    else:
        result = best_a
    sc_result = _score(result)
    return result, sc_result, True


def _resample_simple(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main optimization ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_main = _lbfgs_optimize(parent, deadline, min_time_remaining=500.0)
        sc_main = _score(a_main)
        if sc_main < best_score:
            best_seq = a_main.copy()
            best_score = sc_main
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam optimization ===
    print(f"\n[P2] Adam optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_adam = _adam_optimize(best_seq, deadline, min_time_remaining=400.0)
        sc_adam = _score(a_adam)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [adam] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Water-filling LP ===
    print(f"\n[P3] Water-filling LP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        wf_deadline = min(deadline - 350, time.monotonic() + 120)
        a_wf, sc_wf, wf_ok = _water_filling_lp(best_seq, wf_deadline, lp_n=1024)
        if wf_ok and sc_wf < best_score:
            best_seq = a_wf.copy()
            best_score = sc_wf
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [waterfill] t={time.monotonic()-t_start:.1f}s")
        elif wf_ok:
            print(f"  Water-filling no improvement: {sc_wf:.10f}")

    # === Phase 4: Active-set polish coarse ===
    print(f"\n[P4] Active-set polish coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        pol_dl = min(deadline - 250, time.monotonic() + 150)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_dl,
                                           tol_frac_start=0.006, tol_frac_end=0.0001,
                                           lr_init=0.04)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish_coarse] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak deflation ===
    print(f"\n[P5] Peak deflation (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        defl_dl = min(deadline - 200, time.monotonic() + 100)
        a_defl = _peak_deflation(best_seq, defl_dl)
        sc_defl = _score(a_defl)
        if sc_defl < best_score:
            best_seq = a_defl.copy()
            best_score = sc_defl
            print(f"  NEW BEST: {best_score:.10f} [deflation] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 180:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 180)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 120:
                a_r = _lbfgs_optimize(best_seq, deadline, min_time_remaining=100.0)
                sc_r = _score(a_r)
                if sc_r < best_score:
                    best_seq = a_r.copy()
                    best_score = sc_r
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    n_candidates = []
    for candidate in [n_best, n_best * 2, n_best * 3 // 2, n_best // 2]:
        if 1024 <= candidate <= 32768 and candidate not in n_candidates:
            n_candidates.append(candidate)
    n_candidates = sorted(set(n_candidates))
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_best and time.monotonic() > deadline - 50:
            continue
        a_init = _resample_simple(best_seq, n_try)
        a_opt = _lbfgs_optimize(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.003 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 25, time.monotonic() + 25),
                                                 tol_frac_start=0.003, tol_frac_end=0.00005,
                                                 lr_init=0.015)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final active-set polish fine ===
    print(f"\n[P8] Final polish fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        pol_dl2 = min(deadline - 35, time.monotonic() + 120)
        a_fin, sc_fin = _active_set_polish(best_seq, pol_dl2,
                                           tol_frac_start=0.002, tol_frac_end=0.00002,
                                           lr_init=0.02)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Ultra-fine final polish ===
    print(f"\n[P9] Ultra-fine final (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_ultra, sc_ultra = _active_set_polish(best_seq, deadline - 3,
                                               tol_frac_start=0.0008, tol_frac_end=0.000003,
                                               lr_init=0.008)
        if sc_ultra < best_score:
            best_seq = a_ultra.copy()
            best_score = sc_ultra
            print(f"  NEW BEST: {best_score:.10f} [ultra_fine] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
