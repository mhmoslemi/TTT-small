"""
Advanced optimizer using adaptive L-BFGS, LP-based active-set methods,
and coordinate descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s
    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]

def _lp_direction(a, M, top_k=100):
    """
    Find optimal direction via LP: max sum(g) s.t. (a*g)_k <= M, g >= 0.
    Uses only the top_k most active constraints.
    Returns the direction vector g (normalized to sum=1) or None.
    """
    n = len(a)
    c = _full_convolve(a, a)
    
    # Select top_k most active constraints
    idx = np.argsort(c)[::-1][:top_k]
    idx = np.sort(idx)
    
    # Build constraint matrix for selected constraints
    # A[i, j] = a[j] if i - j in [0, n-1], else 0
    rows = []
    cols = []
    vals = []
    for i in idx:
        j_start = max(0, i - n + 1)
        j_end = min(n, i + 1)
        for j in range(j_start, j_end):
            rows.append(i)
            cols.append(j)
            vals.append(a[j])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_con = M * np.ones(2*n-1)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_con, bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    
    return None

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)

    elif kind == "ramp":

        a = 0.5 + t + 0.1 * np.sin(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)

    else:

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _active_set_gd(a_init, deadline, tol_frac_start=0.01, tol_frac_end=0.00005, lr_init=0.05):
    """Active-set gradient descent with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    stall_count = 0
    max_stall = 60

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Weighted active set (inverse of slack)
        slacks = c_max - c
        slacks_active = slacks[active]
        weights = (1.0 / (slacks_active + tol * 0.1))
        weights = weights / weights.sum()
        
        w = np.zeros_like(c)
        w[active] = weights
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):

    best_a = a.copy()
