Now for the main optimization, let me think about the gradient more carefully.

The objective (in log-space with softmax parameterization) is:
`f(z) = 2n * max_k (softmax(z) * softmax(z))[k]`

The gradient of `max_k c[k]` where `c = a*a` (a = softmax(z)) is:
- If the max is at unique index k*, then `d max_c / d a[j] = a[k*-j]` (with appropriate index handling for the convolution).
- Then `d max_c / d z[j] = a[j] * (d max_c / d a[j] - sum_l a[l] * d max_c / d a[l])` (Jacobian of softmax).

For the smoothed version with temperature T:
- `p_k = exp(c[k]/T) / sum exp(c[k']/T)`
- `d (sum p_k * c_k) / d a[j] = sum_k p_k * a[k-j]`
- This can be computed as a convolution of `a` with `p` (reversed).

Let me implement the top-K weighted version:
- Select the top K indices of c.
- Assign weights w_i = decay^i for i = 0, ..., K-1.
- Normalize weights to sum to 1.
- Compute the weighted convolution gradient.

Now, for the LP-based refinement, let me think about a practical implementation.

Actually, for the LP, I think a simpler "projection" approach might work better:

**Projection approach:**
Given current `a` (sum=1, a>=0), and current peak M = max(a*a):
1. Find the "forbidden" region in the space of sequences: all sequences whose convolution exceeds M at any point.
2. Project the "ideal" sequence (which would minimize the max-convolution) onto the feasible region.

But this is hard to compute directly. Instead, let me use a simpler iterative approach:

**Iterative peak reduction:**
1. Find k* = argmax(a*a).
2. The value at k* is `(a*a)[k*] = sum_j a[j]*a[k*-j]`.
3. The "gradient" of this value w.r.t. a is the vector g where g[j] = a[k*-j] (reversed and truncated).
4. To reduce (a*a)[k*], we want to move a in the direction -g (reducing elements where g is large).
5. But we also need to maintain sum(a) = 1 and a >= 0.
6. A natural step: `a_new[j] = a[j] - eta * g[j] + eta * mean(g)` (to maintain the sum).
7. Clip to non-negative and renormalize.

This is essentially "gradient descent on the peak" with a simple projection. Let me combine this with the softmax parameterization:

In log-space: `z_new = z - lr * a * (g - mean(a*g))` (the standard softmax gradient step).

But instead of using the smoothed gradient, use the exact peak gradient (or top-K).

Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K peak gradient in log-space with adaptive temperature,
proper line search, multi-scale exploration (8192, 16384),
and LP-inspired refinement.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_gradient(a, c, top_k=10, decay=0.8):
    """Compute gradient of top-K weighted max of convolution."""
    n = len(a)
    # Get top K indices
    k_indices = np.argpartition(c, -top_k)[-top_k:]
    k_indices = k_indices[np.argsort(c[k_indices])[::-1]]
    
    # Assign weights
    weights = decay ** np.arange(top_k)
    weights = weights / weights.sum()
    
    # Build the weighted peak indicator in the full convolution space
    p = np.zeros(len(c))
    for i, idx in enumerate(k_indices):
        p[idx] = weights[i]
    
    # Gradient of (a * p_reversed) at the center
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    # The relevant part is indices [n-1, 2n-2]
    g = conv_full[n-1:2*n-1][::-1]
    return g * len(a)  # scaling factor


def _smoothed_gradient(a, c, T):
    """Compute smoothed gradient of max using softmax with temperature T."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = conv_full[n-1:2*n-1][::-1]
    return g * len(a)


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience_score = best_score
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        # Temperature schedule: hold at T_start for first 20%, then decrease
        if frac < 0.2:
            T = T_start
        else:
            frac2 = (frac - 0.2) / 0.8
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Choose gradient type

            g = _topk_gradient(a, c, top_k, decay)
        else:
            g = _smoothed_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        
        # Line search: check every 20 steps
        if t % 20 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.005:
                # Revert and reduce lr
                z_new = z
                if lr_mult > 0.1:
                    lr_mult *= 0.7
            z = z_new
        else:
            z = z_new
        
        # Evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                # Reset from best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _lp_refinement(a_init, deadline, n_iter=100):
    """
    LP-inspired refinement: find direction b that maintains (a*b)[k] <= M,
    then blend.
    Simplified version using iterative projection.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    M = float(_fft_convolve(a, a).max())
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Gradient of (a*a)[k_star] w.r.t. a
        # (a*a)[k] = sum_j a[j] * a[k-j]
        # d/d(a[j]) = a[k-j] (with a[.] = 0 outside [0,n-1])
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = a[idx]
        
        # We want to find b >= 0 such that (a*b)[k] <= M for all k
        # Simplified: b = a - alpha * g (push down where gradient is high)
        # Then renormalize
        alpha = 0.01 * a.mean() / (g.max() + 1e-15)
        b = a - alpha * g
        b = np.maximum(b, 0)
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b
        
        # Check if blending helps
        for t in [0.1, 0.2, 0.3, 0.5]:
            a_blend = (1 - t) * a + t * b
            a_blend = a_blend / a_blend.sum()
            c_blend = _fft_convolve(a_blend, a_blend)
            sc = 2.0 * n * float(c_blend.max())
            if sc < best_score * (1.0 - 1e-10):
                best_score = sc
                best_a = a_blend.copy()
                a = a_blend.copy()
                break
    
    return best_a, best_score
