def make_block_sequence(n, block_num=5):
    """Create a sequence with distinct blocks"""
    total = n
    block_len = int(total / block_num)
    avg_val = np.ones((block_len+1), dtype=np.float64)
    avg_val[0] = np.sum(avg_val) / block_len
    
    # Create sequence with distinct block values
    seq = []
    for i in range(block_num):
        start = i * block_len - 1
        end = start + block_len + 1
        
        # Avoid boundary issues
        if i == 0:
            block = np.linspace(1.5, 0.05, block_len+1)
            block[-1] = 0.9  # Force non-zero at all boundaries
            seq.append(block * (0.1 / (0.9/block_len + 1/block_len)))
            continue
            
        if end > total:
            end = total
            portion = (end - start) / (block_len)
            # Adjust block length based on remaining elements
            block = np.ones((portion+1).astype(int))
            block -= 0.4 * i  # Apply distinct penalization
            seq.append(block * (0.2 / (0.4+0.1)))
            continue
            
        # Normal block section
        block = np.linspace(0.2, i, block_len)
        block = np.ones(block_len+1)
        seq.append(block)
        
    return np.array(seq)
