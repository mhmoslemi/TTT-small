"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B with multi-temperature log-sum-exp smoothing,
targeted coordinate descent on peak-contributing coordinates,
and diverse restart strategies.

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _smoothed_objective_and_grad(x, T, n):
    """Smoothed objective using log-sum-exp, with gradient."""
    a = np.maximum(x, 0)  # enforce non-negativity (L-BFGS-B handles bounds, but just in case)
    S = float(a.sum())
    if S < 1e-15:
        return 1e10, np.zeros_like(x)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    z = (c - c_max) / T
    z_max = float(z.max())
    e = np.exp(z - z_max)
    e_sum = float(e.sum())
    if e_sum < 1e-300:
        # Fallback: just use the max
        lse = c_max
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        lse = c_max + T * (z_max + np.log(e_sum))
        p = e / e_sum
    
    score = 2.0 * n * lse / (S * S)
    
    # Gradient of lse w.r.t. a_j
    # = sum_k p_k * a_{k-j} = (convolution based)
    p_flip = p[::-1]
    N_fft = _next_pow2(n + 2 * n - 1)
    A = np.fft.rfft(a, N_fft)
    P_flip = np.fft.rfft(p_flip, N_fft)
    conv = np.fft.irfft(A * P_flip, N_fft)[:3 * n - 1]
    dlse_da = conv[n - 1:2 * n - 1][::-1]
    
    grad = (2.0 * n / (S ** 3)) * (S * dlse_da - 2.0 * lse)
    
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(x)
    
    return score, grad

def _run_lbfgs(a_init, T, deadline, max_iter=5000, ftol=1e-12, gtol=1e-10):
    """Run L-BFGS-B with smoothed objective at temperature T."""
    n = len(a_init)
    a = a_init.copy()
    
    best_a = a.copy()
    best_score = _score(a)
    
    def fun(x):
        s, _ = _smoothed_objective_and_grad(x, T, n)
        return s
    
    def jac(x):
        _, g = _smoothed_objective_and_grad(x, T, n)
        return g
    
    def fun_and_jac(x):
        s, g = _smoothed_objective_and_grad(x, T, n)
        return s, g
    
    for attempt in range(3):
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(fun_and_jac, a, method='L-BFGS-B',
                            jac=True,  # indicates fun returns (f, g)
                            bounds=[(0, None)] * n,
                            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                                    'maxfun': max_iter * 2, 'iprint': -1})
            if result.success or result.fun < best_score:
                a_trial = np.maximum(result.x, 0)
                sc = _score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
            a = result.x
        except Exception as e:
            print(f"  L-BFGS error: {e}")
            break
    
    return best_a, best_score

"""
Multi-stage L-BFGS optimizer in softmax/log-space for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- L-BFGS in z-space (unconstrained) with proper line search replaces Adam
- Multi-temperature continuation method (homotopy) for navigating non-convex landscape
- Targeted coordinate descent focusing on peak-contributing coordinate pairs
- Diverse initialization strategies and perturbation restarts
- Nesterov-style momentum as fallback when L-BFGS stalls
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_grad_z(z, T, n):
    """
    Compute smoothed objective and gradient in z-space.
    a = softmax(z), objective = 2n * L_T(a*a) where L_T is log-sum-exp smoothing.
    Since sum(a) = 1, score = 2n * max(a*a) (for T->0).
    """
    a = _softmax_stable(z)
    S = float(a.sum())
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    z_c = (c - c_max) / T
    z_c_shift = z_c - z_c.max()
    e = np.exp(z_c_shift)
    e_sum = float(e.sum())
    if e_sum < 1e-300:
        lse = c_max
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        lse = c_max + T * (z_c.max() + np.log(e_sum))
        p = e / e_sum

    score = 2.0 * n * lse / (S * S)

    # d(lse)/d(a_j) = sum_k p_k * a_{k-j}
    # = (a * flip(p))_{2n-2-j} for j = 0..n-1
    p_flip = p[::-1]
    N_fft = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N_fft)
    Pf = np.fft.rfft(p_flip, N_fft)
    conv = np.fft.irfft(A * Pf, N_fft)[:3 * n - 1]
    dlse_da = conv[n - 1:2 * n - 1][::-1]

    # d(score)/d(a_j) = (2n/S^3) * (S * dlse_da_j - 2*lse)
    g_a = (2.0 * n / (S ** 3)) * (S * dlse_da - 2.0 * lse)

    # Chain rule to z-space: d(score)/d(z_j) = a_j * (g_a_j - <a, g_a>)
    W = float(np.dot(a, g_a))
    g_z = a * (g_a - W)

    if not np.all(np.isfinite(g_z)):
        g_z = np.zeros(n)

    return score, g_z


def _lbfgs_z_space(z_init, T, deadline, max_iter=2000, max_fun=5000):
    """Run L-BFGS in z-space at given temperature."""
    n = len(z_init)
    z = z_init.copy()
    best_z = z.copy()
    best_score = _smoothed_obj_grad_z(z, T, n)[0]

    def fun_and_jac(zv):
        s, g = _smoothed_obj_grad_z(zv, T, n)
        return s, g

    try:
        result = minimize(fun_and_jac, z, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-12,
                                   'maxfun': max_fun, 'iprint': -1})
        if result.fun < best_score:
            best_score = result.fun
            best_z = result.x.copy()
        # Also try from result.x even if fun is slightly worse (numerical issues)
        z_out = result.x
    except Exception as e:
        z_out = z
        best_z = z

    # Safety: keep z bounded
    z_out = np.clip(z_out, -100.0, 100.0)
    z_out = z_out - z_out.mean()

    return z_out, best_z, best_score


def _nesterov_z_space(z_init, T, deadline, lr=0.01, momentum=0.9, max_iter=100000):
    """Nesterov-accelerated gradient descent in z-space."""
    n = len(z_init)
    z = z_init.copy()
    v = np.zeros(n)
    best_z = z.copy()
    best_score = _smoothed_obj_grad_z(z, T, n)[0]
    patience = 0

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        # Nesterov lookahead
        z_look = z - momentum * v
        _, g = _smoothed_obj_grad_z(z_look, T, n)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm * min(g_norm, 1.0)
        v = momentum * v - lr * g
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % 200 == 0:
            sc = _smoothed_obj_grad_z(z, T, n)[0]
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10:
                lr *= 0.5
                patience = 0

    return z, best_z, best_score


def _targeted_cd(a, deadline, max_rounds=100, top_peaks=8):
    """
    Coordinate descent targeting the top peaks of the convolution.
    For each peak, try adjusting the contributing coordinates.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a)
    c = _fft_convolve(best_a, best_a)

        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())

        improved = False

                continue
            # Coordinates contributing to peak k

            j_hi = min(k, n - 1)

            for j in js:
                if best_a[j] < 1e-15:

                    a_cand[j] *= factor
                    if a_cand[j] < 0:
                        continue
                    s = float(a_cand.sum())
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):

                        best_a = a_cand.copy()

                break

            # Try pair moves: equalize pairs contributing to top peak

            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            pair_improved = False
            for j1 in range(j_lo, j_hi + 1):
                j2 = k - j1
                if j2 < 0 or j2 >= n or j2 <= j1:
                    continue
                if best_a[j1] < 1e-15 or best_a[j2] < 1e-15:

                # Try to equalize

                for t_mix in [0.3, 0.5, 0.7]:
                    a_cand = best_a.copy()
                    a_cand[j1] = (1 - t_mix) * best_a[j1] + t_mix * avg
                    a_cand[j2] = (1 - t_mix) * best_a[j2] + t_mix * avg
                    s = float(a_cand.sum())
                    if s < 1e-15:

                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)

                    if sc < best_score * (1.0 - 1e-12):

                        best_a = a_cand.copy()
                        pair_improved = True
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    print(f"Initial: n={n}, score={best_score:.10f}")

    # Convert to z-space
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    # === Phase 1: Multi-stage L-BFGS from parent (0s - 400s) ===
    phase1_deadline = min(deadline, t_start_main + 400.0)
    print(f"\n=== Phase 1: Multi-stage L-BFGS (parent, {n} dims) ===")
    T_stages_1 = [0.02, 0.01, 0.005, 0.002, 0.0008, 0.0003, 0.0001]
    c_cur = _fft_convolve(best_a, best_a)
    max_c = float(c_cur.max())
    T_stages_1 = [t * max_c for t in T_stages_1]

    z_opt, a_opt, sc_opt = _multi_stage_lbfgs(z, phase1_deadline, T_stages_1)
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
    print(f"  After Phase 1: score={best_score:.10f}, t={time.monotonic()-t_start_main:.0f}s")

    # === Phase 2: Targeted CD polish (400s - 550s) ===
    phase2_deadline = min(deadline, t_start_main + 550.0)
    if time.monotonic() < phase2_deadline:
        print(f"\n=== Phase 2: Targeted CD ===")
        a_cd, sc_cd = _targeted_cd(best_a, phase2_deadline, max_rounds=50, top_peaks=6)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
        print(f"  After CD: score={best_score:.10f}, t={time.monotonic()-t_start_main:.0f}s")

    # === Phase 3: More L-BFGS at very low T (550s - 750s) ===
    phase3_deadline = min(deadline, t_start_main + 750.0)
    if time.monotonic() < phase3_deadline:
        print(f"\n=== Phase 3: Ultra-low T L-BFGS ===")
        z = np.log(np.maximum(best_a, 1e-15))
        z = z - z.mean()
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        T_stages_3 = [0.00005, 0.00002, 0.000008, 0.000003, 0.000001, 0.0000005]
        T_stages_3 = [t * max_c for t in T_stages_3]
        z_opt, a_opt, sc_opt = _multi_stage_lbfgs(z, phase3_deadline, T_stages_3)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
        print(f"  After Phase 3: score={best_score:.10f}, t={time.monotonic()-t_start_main:.0f}s")

    # === Phase 4: Aggressive CD (750s - 900s) ===
    phase4_deadline = min(deadline, t_start_main + 900.0)
    if time.monotonic() < phase4_deadline:
        print(f"\n=== Phase 4: Aggressive CD ===")
        a_cd, sc_cd = _aggressive_cd(best_a, phase4_deadline, max_rounds=80)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
        print(f"  After Phase 4: score={best_score:.10f}, t={time.monotonic()-t_start_main:.0f}s")

    # === Phase 5: Perturbation restarts (900s - 1400s) ===
    phase5_deadline = min(deadline, t_start_main + 1400.0)
    if time.monotonic() < phase5_deadline:
        print(f"\n=== Phase 5: Perturbation restarts ===")
        n_best = len(best_a)
        num_restarts = 5
        time_per_restart = max(30.0, (phase5_deadline - time.monotonic()) / num_restarts)

        for rst in range(num_restarts):
            if time.monotonic() > phase5_deadline - 30:
                break
            sigma = 0.03 + 0.02 * rst
            z_base = np.log(np.maximum(best_a, 1e-15))
            z_base = z_base - z_base.mean()

            # Different perturbation strategies
            if rst == 0:
                # Small Gaussian noise
                z_pert = z_base + sigma * rng.standard_normal(n_best)
            elif rst == 1:
                # Smooth noise (low-frequency)
                freq = np.fft.rfftfreq(n_best)
                noise = np.fft.irfft(np.exp(-20 * freq) * rng.standard_normal(len(freq)), n_best)
                z_pert = z_base + sigma * 2 * noise / (np.std(noise) + 1e-15)
            elif rst == 2:
                # Block perturbation
                block_size = n_best // 16
                z_pert = z_base.copy()
                for blk in range(16):
                    start = blk * block_size
                    end = min(start + block_size, n_best)
                    z_pert[start:end] += sigma * 1.5 * rng.standard_normal(end - start)
            elif rst == 3:
                # Reversal + small noise (exploits symmetry)
                z_pert = z_base[::-1].copy() + 0.01 * rng.standard_normal(n_best)
            else:
                # Larger perturbation
                z_pert = z_base + 0.08 * rng.standard_normal(n_best)

            z_pert = np.clip(z_pert, -80.0, 80.0)
            z_pert = z_pert - z_pert.mean()

            a_pert = _softmax_stable(z_pert)
            c_pert = _fft_convolve(a_pert, a_pert)
            max_pert = float(c_pert.max())

            # Run L-BFGS from perturbed point
            sub_deadline = min(phase5_deadline - 10, time.monotonic() + time_per_restart)
            T_stages_r = [0.003, 0.001, 0.0003, 0.0001, 0.00003, 0.00001]
            T_stages_r = [t * max_pert for t in T_stages_r]
            z_r, a_r, sc_r = _multi_stage_lbfgs(z_pert, sub_deadline, T_stages_r)
            if sc_r < best_score:
                best_score = sc_r
                best_a = a_r.copy()
            print(f"  Restart {rst} (sigma={sigma:.3f}): score={sc_r:.10f}")

    # === Phase 6: Final ultra-fine polish (1400s - deadline) ===
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 6: Final polish ===")
        z = np.log(np.maximum(best_a, 1e-15))
        z = z - z.mean()
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())

        # Very low T L-BFGS
        T_final = [0.000002, 0.000001, 0.0000005]
        T_final = [t * max_c for t in T_final]
        sub_deadline = min(deadline - 60, t_start_main + 1700.0)
        z_opt, a_opt, sc_opt = _multi_stage_lbfgs(z, sub_deadline, T_final)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()

        # Final aggressive CD
        if time.monotonic() < deadline - 10:
            a_cd, sc_cd = _aggressive_cd(best_a, deadline - 10, max_rounds=50)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cd.copy()

        # One more targeted CD
        if time.monotonic() < deadline - 5:
            a_cd2, sc_cd2 = _targeted_cd(best_a, deadline - 5, max_rounds=20, top_peaks=10)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_a = a_cd2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_main:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
