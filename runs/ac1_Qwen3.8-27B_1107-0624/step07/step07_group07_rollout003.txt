where W = max(c) (if single peak) or W = weighted average of active peaks.

Actually, for the top-k average: W = (1/k) * sum of top k c values.

So the gradient for score = 2*n*W/(sum(a))^2 is:
grad_score[j] = (2*n / (sum(a))^2) * (h[j] - W * a[j] / sum(a))

And in z-space (a = softmax(z)), the gradient transforms as:
grad_z[j] = a[j] * grad_score[j] (by the chain rule for softmax)

Wait, let me be more careful. If a = softmax(z), then da/dz[j] = a[j]*(δ_ij - a[j]).
So grad_z[j] = sum_i (da[i]/dz[j]) * grad_score[i] = a[j]*grad_score[j] - a[j]*sum_i a[i]*grad_score[i]

Let S = sum_i a[i]*grad_score[i]. Then grad_z[j] = a[j]*(grad_score[j] - S).

But actually, for the purpose of optimization, we can just use grad_z[j] = a[j]*grad_score[j] and the constant offset S doesn't matter (it just shifts all components equally, which doesn't change softmax).

So: grad_z[j] = a[j] * (2*n / (sum(a))^2) * (h[j] - W * a[j] / sum(a))

For simplicity, and since we're working with the normalized version, let me just use:
grad_z[j] = a[j] * h[j] - a[j]^2 * W / sum(a)

(up to a positive constant factor that doesn't matter for the direction)

"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
sharp top-K peak subgradient method with backtracking line search,
peak equalization, structure-aware restarts, and coordinate descent.
Key innovation: uses exact top-K subgradients instead of log-sum-exp smoothing,
enabling much sharper optimization directions and faster convergence.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_self(a):
    return _full_convolve(a, a)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, k):
    """Get weights on top-k peaks of c."""
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    return w


def _grad_z_sharp(a, c, w, n):
    """Compute z-space gradient for top-K peak objective."""
    s = a.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # grad_score[j] = (2n/s^2) * (h[j] - W*a[j]/s)
    # grad_z[j] = a[j] * grad_score[j] (up to constant)
    grad_z = a * (h - W * a / s)
    return grad_z


def _subgradient_loop(a_init, deadline, top_k=1, max_iters=50000,
                      lr_init=0.05, backtrack_alpha=0.5, backtrack_rho=0.5,
                      label=""):
    """Sharp subgradient method with backtracking line search in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15)) - np.log(np.maximum(a_init, 1e-15)).mean()
    a = _softmax(z)
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    stall = 0
    max_stall = 500
    no_improve_window = 200

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _convolve_self(a)
        w = _topk_weights(c, top_k)
        grad_z = _grad_z_sharp(a, c, w, n)

        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = lr
        improved = False
        for bt in range(10):
            z_new = z - step * grad_z
            z_new = z_new - z_new.mean()
            a_new = _softmax(z_new)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                z = z_new
                a = a_new
                best_a = a_new.copy()
                best_sc = sc_new
                stall = 0
                improved = True
                break
            step *= backtrack_rho

        if not improved:
            # Try smaller lr or restart
            stall += 1
            lr *= 0.9
            if lr < 1e-8 or stall > max_stall:
                break

        if i % no_improve_window == 0 and stall > no_improve_window // 2:
            # Gentle random perturbation to escape
            z = z + 0.01 * np.random.standard_normal(n)
            z = z - z.mean()
            a = _softmax(z)
            stall = max(0, stall - 100)

        # Slowly increase lr if making progress
        if improved and stall == 0:
            lr = min(lr * 1.01, lr_init * 2)

    return best_a, best_sc


def _peak_equalize(a_init, deadline, top_k=5, max_iters=20000):
    """Iteratively equalize the top peaks to reduce max."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _convolve_self(a)
        c_max = c.max()
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        top_mean = top_vals.mean()

        # If peaks are nearly equal, we're at a good local optimum
        if top_vals.max() - top_vals.min() < 1e-6 * c_max:
            # Still try to reduce the common level
            pass

        # Compute gradient that reduces the max (use top peak)
        w = _topk_weights(c, 1)
        grad_z = _grad_z_sharp(a, c, w, n)

        # Take a small step
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        step = 0.01
        z = z - step * grad_z
        z = z - z.mean()
        a_new = _softmax(z)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try opposite direction (sometimes helps to increase lower peaks)
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z + step * grad_z
            z = z - z.mean()
            a_try = _softmax(z)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try

    return best_a, best_sc


def _coordinate_descent_v2(a_init, deadline, max_rounds=20000):
    """Improved coordinate descent targeting peak reduction."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    c = _convolve_self(a)
    peak_idx = int(np.argmax(c))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Redetermine peak every 100 rounds
        if rnd % 100 == 0:
            c = _convolve_self(a)
            peak_idx = int(np.argmax(c))

        # Choose coordinate to modify: pick where a[peak_idx - j] is large
        # (this is where the gradient is largest)
        j_candidates = []
        for offset in range(-n + 1, n):
            k = peak_idx + offset
            if 0 <= k < n:
                j_candidates.append(k)

        if not j_candidates:
            continue

        # Pick the coordinate with largest a value at the peak-contributing position
        contrib = np.array([a[j] if 0 <= j < n else 0 for j in
                           [peak_idx - k for k in range(n) if 0 <= peak_idx - k < n]])
        valid_j = [peak_idx - k for k in range(n) if 0 <= peak_idx - k < n]
        if not valid_j:
            continue
        contrib_arr = np.array([a[j] for j in valid_j])
        # Pick a few candidates
        top_cand_idx = np.argpartition(contrib_arr, -3)[-3:]
        candidates = [valid_j[ci] for ci in top_cand_idx if a[valid_j[ci]] > 1e-10]

        if not candidates:
            candidates = [int(np.argmax(a))]

        j_mod = candidates[rnd % len(candidates)]

        # Try transferring mass from j_mod to another coordinate
        # that reduces the peak
        a_j = a[j_mod]
        if a_j < 1e-10:
            continue

        # Fraction to remove
        for frac in [0.05, 0.1, 0.2, 0.3]:
            delta = a_j * frac
            # Where to put it? Try the coordinate that least contributes to peak
            # or a random coordinate far from peak
            target = (j_mod + n // 2) % n
            if target == j_mod:
                target = (target + 1) % n

            a_try = a.copy()
            a_try[j_mod] = a_j - delta
            a_try[target] = a[target] + delta
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                c = _convolve_self(a)
                peak_idx = int(np.argmax(c))
                break

    return best_a, best_sc


def _multi_scale_search(a_init, deadline):
    """Try optimizing at different scales by interpolation."""
    n_cur = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    target_ns = []
    if n_cur > 2048:
        target_ns.append(n_cur // 2)
    if n_cur < 16384:
        target_ns.append(min(n_cur * 2, 16384))

    for tn in target_ns:
        if time.monotonic() > deadline:
            break
        if tn == n_cur:
            continue

        # Interpolate
        t_old = np.linspace(0, 1, n_cur)
        t_new = np.linspace(0, 1, tn)
        a_interp = np.interp(t_new, t_old, a_init)
        a_interp = np.maximum(a_interp, 1e-10)
        a_interp = a_interp / a_interp.sum()

        sc_interp = _score(a_interp)
        if time.monotonic() > deadline:
            break

        # Optimize at this scale
        a_opt, sc_opt = _subgradient_loop(a_interp, deadline, top_k=1,
                                          max_iters=5000, lr_init=0.03,
                                          label=f"n={tn}")
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt

    return best_a, best_sc


def _structure_restart(a, rng, mode):
    """Generate a perturbed version with specific structure."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15)) - np.log(np.maximum(a, 1e-15)).mean()

    if mode == "gauss_small":
        z = z + 0.02 * rng.standard_normal(n)
    elif mode == "gauss_med":
        z = z + 0.05 * rng.standard_normal(n)
    elif mode == "gauss_large":
        z = z + 0.10 * rng.standard_normal(n)
    elif mode == "cosine":
        t = np.arange(n) / n
        freq = int(rng.integers(1, 20))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2 * np.pi * freq * t + phase)
    elif mode == "sine_multi":
        t = np.arange(n) / n
        for _ in range(int(rng.integers(2, 6))):
            freq = int(rng.integers(1, 30))
            amp = 0.01 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.sin(2 * np.pi * freq * t + phase)
    elif mode == "block":
        n_blocks = int(rng.integers(4, 16))
        block_size = n // n_blocks
        for _ in range(n_blocks):
            start = _ * block_size
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.02, 0.02)
    elif mode == "taper":
        t = np.linspace(0, 1, n)
        alpha = rng.uniform(0.1, 0.5)
        z = z + alpha * (np.cos(np.pi * t) - 1)

    z = z - z.mean()
    return _softmax(z)


def _intensive_polish(a_init, deadline):
    """Final intensive polish combining multiple techniques."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Alternate between top-1 and top-3 subgradient
    for phase in range(4):
        if time.monotonic() > deadline:
            break

        k = 1 if phase % 2 == 0 else 3
        a_new, sc_new = _subgradient_loop(a, deadline, top_k=k,
                                          max_iters=5000, lr_init=0.02,
                                          label=f"polish_k{k}")
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    # Coordinate descent
    if time.monotonic() < deadline:
        a_cd, sc_cd = _coordinate_descent_v2(a, deadline, max_rounds=10000)
        if sc_cd < best_sc:
            best_sc = sc_cd
            best_a = a_cd.copy()

    return best_a, best_sc
