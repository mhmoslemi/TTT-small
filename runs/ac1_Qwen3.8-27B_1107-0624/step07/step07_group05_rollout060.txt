import time
from scipy.optimize import minimize, linprog

def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, 2*N)  # zero-pad to avoid circular conv
    fb = np.fft.rfft(b, 2*N)
    r = np.fft.irfft(fa * fb, 2*N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _smoobj_and_grad(z, beta, n):
    """Log-sum-exp smoothed objective and gradient in log-space."""
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:

        w = np.zeros_like(c)
        w[c.argmax()] = 1.0

        w = e / s

        k <<= 1

def _fft_conv(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _ls_obj(z, beta, n):

    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    base_beta = max(gap * 2.0, max_c * 0.05)

def _pressure_equalize_fft(a, deadline, n_iters=500, gamma_init=0.5):
    """Exact KKT-motivated pressure equalization using FFT."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    gamma = gamma_init
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_conv(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        # Adaptive active set: within 2% of max or top 10%
        n_active_target = max(10, n // 5)
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() > n_active_target * 3:
            idx = np.argsort(c)[-n_active_target:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        elif active.sum() < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        # Compute pressure via FFT
        a_rev = a[::-1]
        p = _fft_conv(active.astype(float), a_rev)[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        # Adaptive gamma: decrease as we converge
        p_variation = (p.max() - p.min()) / (p.mean() + 1e-30)
        gamma_eff = gamma * min(1.0, p_variation)
        ratio = p.mean() / p
        a = a * np.power(ratio, gamma_eff)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        if it % 25 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        # Decay gamma slowly
        gamma *= 0.995
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

def _alternating_lp(a_init, deadline, min_remaining=60):
    """Alternating LP: fix y, optimize x; fix x, optimize y; repeat."""
    n = len(a_init)
    if n > 4096:
        return a_init
    x = a_init.copy()
    y = a_init.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    n_iters_max = 200
    for it in range(n_iters_max):
        if time.monotonic() + min_remaining > deadline:
            break
        # Build constraint matrix: A[k,j] = y[k-j] if 0<=k-j<n
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                A[k, j] = y[k - j]
        # LP: min t s.t. A@x <= t*1, sum(x)=1, x>=0, t>=0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': max(5, deadline - time.monotonic() - min_remaining - 30)})
            if res.status == 0:
                x_new = res.x[:n]
                x_new = np.maximum(x_new, 0)
                s = x_new.sum()
                if s > 1e-10:
                    x_new = x_new / s
                    if np.allclose(x_new, x, atol=1e-10):
                        # Convergence check
                        sc = _score(x_new)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = x_new.copy()
                        break
                    # Swap: now optimize y given x_new
                    y = x_new.copy()
                    x = x_new.copy()
                    sc = _score(x)
                    if sc < best_sc:
                        best_a = x.copy()
                        best_sc = sc
                else:
                    break
            else:
                break
        except Exception:
            break
    return best_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(123)
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n

def _fourier_perturb(a, rng, n_modes=8, amplitude=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
    sc_final = _score(_softmax(z))
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = _softmax(z)
    return best_a, best_sc

def _multi_stage(a_init, deadline, min_remaining=60.0):
    """Full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    # Stage 1: L-BFGS
    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 60)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1
    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        pe_deadline = min(deadline - min_remaining - 10, time.monotonic() + min(90, max(10, (deadline - time.monotonic() - min_remaining) * 0.25)))
        a2, sc2 = _pressure_equalize_fft(current_a, pe_deadline, n_iters=300, gamma_init=0.5)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2
    # Stage 3: L-BFGS again
    if time.monotonic() < deadline - min_remaining - 20:
        a3 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 20)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3
    # Stage 4: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 40)
        a4, sc4 = _subgrad_polish(current_a, sg_deadline, n_max_steps=1500, lr_init=0.015)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4
    return current_a, current_sc

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # === Phase 1: Main L-BFGS + pressure on parent ===
    print(f"\n[Phase 1] Main optimization on parent n={n_parent}")
    a_main, sc_main = _multi_stage(parent, deadline, min_remaining=500.0)
    _update(a_main, sc_main)

    # === Phase 2: Alternating LP on moderate n ===
    print(f"\n[Phase 2] Alternating LP (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_best = len(best_seq)
        # Try LP at n_best if small enough, else resample
        lp_n = min(n_best, 4096)
        if lp_n != n_best:
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, lp_n)
            a_lp_init = np.interp(x_new, x_old, best_seq)
            a_lp_init = np.maximum(a_lp_init, 0)
            a_lp_init /= a_lp_init.sum()
        else:
            a_lp_init = best_seq.copy()
        lp_deadline = min(deadline - 200, time.monotonic() + 300)
        a_lp = _alternating_lp(a_lp_init, lp_deadline, min_remaining=200)
        sc_lp = _score(a_lp)
        _update(a_lp, sc_lp)
        # If LP improved, upscale back and optimize
        if sc_lp < best_score and n_best > lp_n:
            x_old2 = np.linspace(0, 1, lp_n)
            x_new2 = np.linspace(0, 1, n_best)
            a_up = np.interp(x_new2, x_old2, a_lp)
            a_up = np.maximum(a_up, 0)
            a_up /= a_up.sum()
            a_up = _run_lbfgs(a_up, deadline, min_time_remaining=min(200, deadline - time.monotonic() - 100))
            _update(a_up, _score(a_up))

    # === Phase 3: Multi-n exploration ===
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 4096, 6144, 8192, 10240, 12288, 16384]
    # Remove parent n to avoid redundant
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - 100 - time.monotonic()
        if remaining < 30:
            break
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        elif parent is not None:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(30, remaining * 0.3)
        a_opt, sc_opt = _multi_stage(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # === Phase 4: Diverse restarts from best ===
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_kinds = ["tapered", "cosine", "random"]
        for i, kind in enumerate(restart_kinds):
            if time.monotonic() > deadline - 25:
                break
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, kind)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=6, amplitude=0.04)
            min_rem = max(20, (deadline - time.monotonic()) * 0.2)
            a_r, sc_r = _multi_stage(a_pert, deadline, min_remaining=min_rem)
            _update(a_r, sc_r)

    # === Phase 5: Extended pressure equalization on best ===
    print(f"\n[Phase 5] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        pe_deadline = min(deadline - 15, time.monotonic() + min(120, (deadline - time.monotonic()) * 0.5))
        a_pe, sc_pe = _pressure_equalize_fft(best_seq, pe_deadline, n_iters=500, gamma_init=0.3)
        _update(a_pe, sc_pe)
        if sc_pe < best_score + 1e-6:
            a_follow = _run_lbfgs(a_pe, deadline, min_time_remaining=10)
            _update(a_follow, _score(a_follow))

    # === Phase 6: Final fine polish ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Fine L-BFGS
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=3)
        _update(a_fin, _score(a_fin))
        # Fine subgradient
        if time.monotonic() < deadline - 2:
            a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.005)
            _update(a_sg, sc_sg)

    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
