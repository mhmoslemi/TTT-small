"""

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hybrid_objective(z, beta, n, top_m):

    c = _full_convolve(a, a)
    c_max = float(c.max())

        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    else:
        # Only include top peaks in the smoothing
        sorted_idx = np.argsort(c)[::-1]

        s = e.sum()

            w = np.zeros_like(c)

        else:
            w = e / s
    else:
        sorted_idx = np.argsort(c)[::-1]
        top_indices = sorted_idx[:top_m]

    """Exact subgradient using only the argmax peak."""
    k_max = int(np.argmax(c))
    w = np.zeros(len(c))
    w[k_max] = 1.0
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

    # W = sum_i w[k] * c[k] = c[k_max]
    W = float(c[k_max])
    return 4.0 * n * a * (h - W)

        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)

        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _coordinate_polish(a, deadline, n_rounds=80):

    n = len(a)

        improved = False

            if old_val < 1e-15:
                continue
            c_old = _full_convolve(a, a).max()

            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tilted_uniform":
        # Slightly tilted: a_i = 1 + eps*(i/(n-1) - 0.5)
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * (t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_chirp":
        # Asymmetric chirp: real part of complex chirp, shifted positive
        t = np.linspace(0, 1, n)
        a = np.abs(np.cos(2 * np.pi * 2 * t * t)) + 0.3
        return a / a.sum()
    elif kind == "asym_chirp2":
        t = np.linspace(0, 1, n)
        a = np.abs(np.cos(np.pi * 3 * t * t)) + 0.2
        a = a * (1.0 + 0.2 * np.sin(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 4 * t) + 0.15 * np.cos(2 * np.pi * 11 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2 * 8) + 0.15
        return a / a.sum()
    elif kind == "random_positive":
        rng = np.random.default_rng(42)
        a = rng.uniform(0.5, 1.5, n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.03):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq > 0 and freq < len(z_hat):
                z_hat[-freq] = np.conj(z_hat[len(z_hat) - freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0):
    """Full optimization pipeline for a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: L-BFGS with adaptive beta
    if time.monotonic() < deadline - min_remaining - 20:
        a1 = _run_lbfgs_adaptive(current_a, deadline, min_time_remaining=min_remaining + 20)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 10:
        pe_deadline = min(deadline - min_remaining - 5, time.monotonic() + 45)
        a2, sc2_norm = _pressure_equalize(current_a, pe_deadline, n_iters=200, gamma=0.35)
        sc2 = _score(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: L-BFGS again (finer)
    if time.monotonic() < deadline - min_remaining - 5:
        a3 = _run_lbfgs_adaptive(current_a, deadline, min_time_remaining=min_remaining + 5)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 30)
        a4, sc4 = _subgrad_polish(current_a, sg_deadline, n_max_steps=2000, lr_init=0.015)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: adaptive L-BFGS + KKT pressure equalization +
    asymmetric chirp initializations + multi-n exploration + coordinate polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}, score={_score(parent):.10f}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main opt on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=600)
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure equalization + L-BFGS ---
    print(f"\n[Phase 2] Extended PE + L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        pe_dl = min(deadline - 300, time.monotonic() + 90)
        a_pe, _ = _pressure_equalize(best_seq, pe_dl, n_iters=400, gamma=0.3)
        sc_pe = _score(a_pe)
        _update(a_pe, sc_pe)
        if sc_pe < best_score + 0.001:
            a_f = _run_lbfgs_adaptive(a_pe, deadline, min_time_remaining=300)
            _update(a_f, _score(a_f))

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    # Try n values around parent and some others
    n_candidates = set()
    if n_parent > 0:
        n_candidates.add(n_parent)
        n_candidates.add(max(1024, n_parent * 3 // 4))
        n_candidates.add(n_parent + n_parent // 8)
        n_candidates.add(n_parent + n_parent // 4)
    n_candidates.update([2048, 3072, 4096, 5120, 6144, 8192, 10240])
    n_candidates = sorted([x for x in n_candidates if x >= 1024 and x <= 16384], reverse=True)

    # Prioritize: parent n first, then nearest others
    def n_priority(n_val):
        if n_val == n_parent:
            return 0
        return abs(n_val - n_parent)
    n_candidates.sort(key=n_priority)

    for n_try in n_candidates[:5]:  # Try at most 5 n values
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 40:
            break

        # Initialize from parent by interpolation or profile
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tilted_uniform")
        elif parent is not None:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tilted_uniform")

        min_rem = max(30, remaining_for_this * 0.3)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem)
        sc_opt_full = _score(a_opt)
        _update(a_opt, sc_opt_full)

    # --- Phase 4: Asymmetric chirp restarts ---
    print(f"\n[Phase 4] Asymmetric chirp restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        profiles = ["asym_chirp", "asym_chirp2", "tilted_uniform", "double_cos"]

        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 30:
                break

            # Blend best with profile
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()

            # Add Fourier perturbation
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=0.025)

            a_r, sc_r = _multi_stage_optimize(a_pert, deadline, min_remaining=30)
            _update(a_r, sc_r)

    # --- Phase 5: Fine-tuned L-BFGS on best ---
    print(f"\n[Phase 5] Fine-tuned L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_fin = _run_lbfgs_adaptive(best_seq, deadline, min_time_remaining=15)
        _update(a_fin, _score(a_fin))

    # --- Phase 6: Coordinate descent polish ---
    print(f"\n[Phase 6] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_cd, _ = _coordinate_polish(best_seq, min(deadline - 8, time.monotonic() + 60), n_rounds=60)
        _update(a_cd, _score(a_cd))

    # --- Phase 7: Final subgradient + L-BFGS ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 4, time.monotonic() + 20),
                                       n_max_steps=1000, lr_init=0.008)
        _update(a_sg, sc_sg)
        if time.monotonic() < deadline - 2:
            a_f2 = _run_lbfgs_adaptive(best_seq, deadline, min_time_remaining=2)
            _update(a_f2, _score(a_f2))

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
