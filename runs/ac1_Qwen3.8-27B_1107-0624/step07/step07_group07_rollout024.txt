def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _autocorr(a):
    return _fft_convolve(a, a)


def _make_profile_asym_exponential(n, decay=0.3):

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxcor': 100,
                                       'maxls': 60, 'disp': False})
            a_out = _softmax(result.x)
            sc = _score(a_out)
            if sc < current_sc - 1e-15:
                current_a = a_out
                current_sc = sc
                print(f"    lbfgs beta={beta:.2e}: score={sc:.10f} (iters={result.nit})")
        except Exception:
            pass
    
    return current_a


def _peak_redistribute(a_init, deadline, max_iters=10000):
    """Targeted peak reduction by redistributing mass from high-contribution positions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr(a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))
        
        # For the peak at lag k, find the pair (i, j) with i+j = k that has max product
        k = peak_idx
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        
        if i_min > i_max:
            break
        
        products = np.array([a[i] * a[k - i] for i in range(i_min, i_max + 1)])
        i_best = i_min + int(np.argmax(products))
        j_best = k - i_best
        
        # Find a position to receive mass: minimize impact on autocorrelation
        # Heuristic: position far from both i_best and j_best
        dist_to_i = np.abs(np.arange(n) - i_best)
        dist_to_j = np.abs(np.arange(n) - j_best)
        dist_score = dist_to_i + dist_to_j
        
        # Prefer positions with low current value and high distance
        candidates = np.argsort(dist_score)[::-1][:20]
        candidates = [cand for cand in candidates if cand != i_best and cand != j_best]
        
        if not candidates:
            break
        
        best_improvement = 0
        best_cand = None
        best_split = None
        
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
            delta = a[i_best] * frac
            if delta < 1e-15:
                continue
            for cand in candidates[:10]:
                a_try = a.copy()
                a_try[i_best] -= delta
                a_try[cand] += delta
                if a_try[i_best] < 0:
                    continue
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - best_improvement - 1e-14:
                    best_improvement = best_sc - sc_try
                    best_cand = cand
                    best_split = frac
                    best_a_try = a_try.copy()
        
        if best_cand is not None:
            a = best_a_try
            best_sc = best_sc - best_improvement
            best_a = a.copy()
        else:
            break
    
    return best_a, best_sc


def _asymmetric_perturbation(a, rng, strength=0.1):
    """Break symmetry of sequence by asymmetric perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    # Add asymmetric perturbation: more on one side
    t = np.arange(n)
    asym = np.sin(np.pi * t / n) * np.cos(2 * np.pi * t / n + rng.uniform(0, 2 * np.pi))
    z = z + strength * asym * rng.uniform(0.5, 1.5)
    z = z + strength * 0.3 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score_val = np.inf
    
    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_score_val = float(sc)
            best_seq = p.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")
    
    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(4096) / 4096
        print(f"No parent. Using uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init")
    
    base_n = len(best_seq) if best_seq is not None else 4096
    print(f"Base length: {base_n}, current best: {best_score_val:.10f}")
    
    # === Phase 1: Asymmetric initialization exploration ===
    print(f"\n=== Phase 1: Asymmetric initializations ===")
    t_phase = time.monotonic()
    
    asym_profiles = [
        ("exp_decay", _make_profile_asym_exponential(base_n, 0.5)),
        ("ramp_down", _make_profile_asym_ramp(base_n, 1.0)),
        ("bilateral", _make_profile_bilateral_asym(base_n, 0.4, 0.1)),
        ("cos_floor", _make_profile_floored_cos(base_n, 1.5, 0.05)),

    # Generate asymmetric sine
    t = np.linspace(0, 1, base_n)
    asym_sine = 0.5 + 0.4 * np.sin(np.pi * t) * (1 - 0.5 * t)
    asym_sine = np.maximum(asym_sine, 0.01)
    asym_sine = asym_sine / asym_sine.sum()

    for name, prof in asym_profiles:
