```python
def _solve_lp_for_g(f, M):
    """Solve: maximize sum(g) subject to (f*g)_k <= M, g >= 0"""
    import scipy.sparse as sp
    n = len(f)
    # Build the constraint matrix
    # (f*g)_k = sum_j f_{k-j} * g_j
    # F[k,j] = f_{k-j} if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
    F = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max c^T g s.t. F g <= M, g >= 0
    # scipy linprog: min c^T x s.t. A_ub x <= b_ub
    c = -np.ones(n)  # minimize -sum(g)
    A_ub = F
    b_ub = np.full(2*n-1, M)
    
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return -result.fun  # sum(g)
    return None
```

import numpy as np

def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score = 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    """Full score including normalization."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_normalized(a / s)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _active_grad(a, c, n, tol_frac=0.001):
    """Compute gradient using the exact active set (tight tolerance)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_act = active.sum()
    if n_act == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_act = 1
    w = np.zeros_like(c)
    w[active] = 1.0 / n_act
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0.0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _kkt_refine(a_init, deadline, n_iters=200):
    """
    KKT-motivated refinement: identify active set, compute minimum-norm
    subgradient, take projected gradient steps.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_normalized(a_init)
    lr = 0.01

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use tight active set
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        n_act = active.sum()
        if n_act < 1:
            idx = np.argmax(c)
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = 1

        # Compute uniform-weight gradient on active set
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Adaptive step size (backtracking)
        step = lr
        for _bt in range(20):
            z_new = z - step * grad
            z_new = np.clip(z_new, -60.0, 60.0)
            z_new = z_new - z_new.mean()
            a_try = _softmax(z_new)
            c_try = _full_convolve(a_try, a_try)
            if c_try.max() < c_max + 1e-15:
                break
            step *= 0.5

        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score_normalized(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score_normalized(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a


def _pressure_equalize(a, deadline, n_iters=300, gamma=0.3):
    """KKT-motivated multiplicative pressure equalization."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 15 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def _coordinate_sweep(a, deadline, n_rounds=30):
    """
    Coordinate descent: for coordinates contributing to the peak,
    try multiplicative adjustments.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(999)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top contributors to the peak
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1]

        # Pick top coordinates
        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = top_idx[rng.permutation(len(top_idx))[:min(len(top_idx), n_try // 3)]]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            c_old = _full_convolve(a, a).max()

            for factor in [0.6, 0.75, 0.9, 1.1, 1.25, 1.4]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-15:
                    improved = True
                    c_old = c_try
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a

    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 4 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=8, amplitude=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.015):
    """Subgradient descent with adaptive step size in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_normalized(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_grad(a, c, n, tol_frac=0.002)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -55.0, 55.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score_normalized(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score_normalized(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a


def _optimize_one(a_init, deadline, min_remaining=30.0, label=""):
    """Full optimization pipeline on a single normalized sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score_normalized(current_a)

    # Stage 1: Aggressive L-BFGS
    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _run_lbfgs_aggressive(current_a, deadline, min_remaining=min_remaining + 60)
        sc1 = _score_normalized(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1
            print(f"    [{label}] After LBFGS: {current_sc:.10f}")

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        pe_deadline = min(deadline - min_remaining - 10, time.monotonic() + 80)
        a2 = _pressure_equalize(current_a, pe_deadline, n_iters=200, gamma=0.3)
        sc2 = _score_normalized(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2
            print(f"    [{label}] After pressure: {current_sc:.10f}")

    # Stage 3: KKT refinement
    if time.monotonic() < deadline - min_remaining - 10:
        kkt_deadline = min(deadline - min_remaining - 5, time.monotonic() + 60)
        a3 = _kkt_refine(current_a, kkt_deadline, n_iters=150)
        sc3 = _score_normalized(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3
            print(f"    [{label}] After KKT: {current_sc:.10f}")

    # Stage 4: Final L-BFGS (fine)
    if time.monotonic() < deadline - min_remaining:
        a4 = _run_lbfgs_aggressive(current_a, deadline, min_remaining=min_remaining)
        sc4 = _score_normalized(a4)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4
            print(f"    [{label}] After final LBFGS: {current_sc:.10f}")

    # Stage 5: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 30)
        a5, _ = _subgrad_polish(current_a, sg_deadline, n_max_steps=1500, lr_init=0.01)
        sc5 = _score_normalized(a5)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: aggressive L-BFGS with 25-step beta continuation,
    KKT refinement, pressure equalization, multi-n exploration, and coordinate sweep.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  >>> BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent loaded: n={n_parent}, score={_score_normalized(parent):.10f}")
                _update(parent)
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent found, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main aggressive optimization on parent ---
    print(f"\n[Phase 1] Main optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _optimize_one(parent, deadline, min_remaining=600, label="main")
    _update(a_main, sc_main)

    # --- Phase 2: KKT refinement + pressure on best ---
    print(f"\n[Phase 2] Deep refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_pe = _pressure_equalize(best_seq, min(deadline - 400, time.monotonic() + 100),
                                  n_iters=400, gamma=0.25)
        _update(a_pe)
        if time.monotonic() < deadline - 400:
            a_kkt = _kkt_refine(best_seq, min(deadline - 350, time.monotonic() + 80), n_iters=300)
            _update(a_kkt)

    # --- Phase 3: Try slightly different n values ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    # Try n values near the parent
    n_options = [7680, 8448, 8704, 8960, 9216, 9472]
    n_options = [x for x in n_options if x != n_parent]

    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - 150 - time.monotonic()
        if remaining < 40:
            break

        # Interpolate parent to new length
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, parent)
        a_init = np.maximum(a_init, 0)
        s_init = a_init.sum()
        if s_init > 1e-10:
            a_init = a_init / s_init
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(25, remaining * 0.4)
        a_opt, sc_opt = _optimize_one(a_init, deadline, min_remaining=min_rem, label=f"n={n_try}")
        _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts from best ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        profiles = ["tapered", "cosine_mod", "gaussian"]
        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 40:
                break
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=6, amplitude=0.015)
            min_rem = max(20, (deadline - time.monotonic()) * 0.3)
            a_r, sc_r = _optimize_one(a_pert, deadline, min_remaining=min_rem, label=f"restart_{i}")
            _update(a_r, sc_r)

    # --- Phase 5: Coordinate sweep on best ---
    print(f"\n[Phase 5] Coordinate sweep (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        cd_deadline = min(deadline - 5, time.monotonic() + 90)
        a_cd = _coordinate_sweep(best_seq, cd_deadline, n_rounds=40)
        _update(a_cd)

    # --- Phase 6: Final polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _run_lbfgs_aggressive(best_seq, deadline - 3, min_remaining=3)
        _update(a_fin)
        if time.monotonic() < deadline - 2:
            a_sg, _ = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.004)
            _update(a_sg)

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
