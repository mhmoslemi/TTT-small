"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive active-set Newton with multi-tolerance search

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _cg_solve(matvec, rhs, max_cg=300, tol=1e-12):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _deflation_step(a, deadline, n_attempts=20):
    """Targeted deflation of top convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top contributing positions
        top_k = np.argsort(c)[::-1]
        n_top = max(3, min(len(c), int(0.05 * len(c))))
        top_positions = top_k[:n_top]

        # For each top position, identify the contributing pairs
        # and reduce the larger contributor
        a_new = a.copy()
        for k_pos in top_positions[:min(n_top, 20)]:
            i_lo = max(0, k_pos - n + 1)
            i_hi = min(k_pos, n - 1)
            # Find the pair with largest product
            max_prod = 0
            max_i = -1
            for i in range(i_lo, i_hi + 1):
                prod = a[i] * a[k_pos - i]
                if prod > max_prod:
                    max_prod = prod
                    max_i = i
            if max_i >= 0 and max_prod > 0:
                # Reduce the larger element slightly
                j = k_pos - max_i
                if a[max_i] >= a[j]:
                    a_new[max_i] *= 0.995
                else:
                    a_new[j] *= 0.995

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        else:
            # Try smaller reduction
            a_small = a.copy()
            for k_pos in top_positions[:10]:
                i_lo = max(0, k_pos - n + 1)
                i_hi = min(k_pos, n - 1)
                max_prod = 0
                max_i = -1
                for i in range(i_lo, i_hi + 1):
                    prod = a_small[i] * a_small[k_pos - i]
                    if prod > max_prod:
                        max_prod = prod
                        max_i = i
                if max_i >= 0:
                    j = k_pos - max_i
                    if a_small[max_i] >= a_small[j]:
                        a_small[max_i] *= 0.999
                    else:
                        a_small[j] *= 0.999
            a_small = np.maximum(a_small, 0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                sc_small = _score(a_small)
                if sc_small < best_sc - 1e-12:
                    best_sc = sc_small
                    best_a = a_small.copy()
                else:
                    break

def _fista_projected(a_init, deadline, n_steps=20000):
    """FISTA for min max_k c[k] s.t. a >= 0, sum(a) = 1."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(y, y)
        c_max = float(c.max())

        # Subgradient of max c[k] at y (using active set average)
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient: 2 * _corr_a_A(y, w, n)
        grad = 2.0 * _corr_a_A(y, w, n)

        # Project gradient onto simplex tangent
        g_t = grad - np.mean(grad)

        # Step size
        step_size = 0.01 / (1.0 + 0.01 * step)
        a_new = y - step_size * g_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t = t_new

        if step % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 6))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "block":
        # Two-block pattern
        a = np.zeros(n)
        block_size = n // 4
        a[:block_size] = 1.0
        a[3 * block_size:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "three_block":
        a = np.zeros(n)
        block = n // 6
        a[:block] = 1.0
        a[2 * block:3 * block] = 1.0
        a[5 * block:] = 1.0
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer with adaptive active-set Newton,
    deflation, FISTA, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p_arr = np.asarray(p, dtype=np.float64).ravel()
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        s = float(p_arr.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p_arr)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p_arr.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p_arr)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton with multi-tolerance
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        as_dl = min(deadline - 500, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=300,
                                         tol_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 450:
            as_dl2 = min(deadline - 450, time.monotonic() + 90)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=200,
                                               tol_list=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: Deflation
    print(f"\n[Phase 3] Deflation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        def_dl = min(deadline - 350, time.monotonic() + 60)
        a_def, sc_def = _deflation_step(best_seq, def_dl, n_attempts=30)
        update_best(a_def, sc_def, "deflation")
        if sc_def < best_score:
            a_def2 = _run_lbfgs(a_def, deadline, min_time_remaining=200.0)
            update_best(a_def2, _score(a_def2), "lbfgs_post_def")

    # Phase 4: FISTA
    print(f"\n[Phase 4] FISTA (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        fista_dl = min(deadline - 250, time.monotonic() + 60)
        a_fista = _fista_projected(best_seq, fista_dl, n_steps=10000)
        sc_fista = _score(a_fista)
        update_best(a_fista, sc_fista, "fista")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
            sc_init = _score(a_init)
            # Also try a different profile
            a_init2 = _make_profile(n_try, "block")
            sc_init2 = _score(a_init2)
            if sc_init2 < sc_init:
                a_init = a_init2
                sc_init = sc_init2
        else:
            a_init = _make_profile(n_try, "uniform")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 30, time.monotonic() + 40),
                                                  max_iter=100, tol_list=[0.002, 0.001, 0.0005])
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [
            ("perturb_small", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_mid", lambda: _perturb(best_seq, 0.05, rng)),
            ("perturb_large", lambda: _perturb(best_seq, 0.1, rng)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 20:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 20),
                                                  max_iter=60, tol_list=[0.001, 0.0005])
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 7: Final polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tight active-set Newton
        as_final = min(deadline - 2, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=200,
                                           tol_list=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        update_best(a_fin, sc_fin, "final_as")

        # Deflation after Newton
        if time.monotonic() < deadline - 2:
            a_def3, sc_def3 = _deflation_step(best_seq, min(deadline - 1, time.monotonic() + 20),
                                              n_attempts=20)
            update_best(a_def3, sc_def3, "final_def")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00003
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(2000, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
