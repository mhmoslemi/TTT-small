"""
Optimized step-function coefficient search using iterative linear programming (successive LP),
active-set Newton refinement, and L-BFGS with log-sum-exp smoothing.
New key idea: Successive LP that linearizes the quadratic convolution constraints and
solves a sequence of linear programs to drive toward the min-max optimum.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    4. Solve LP: min t s.t. linearized constraints, sum(a)=1, a>=0
    5. Line search / trust region step

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_cmax = float(_full_convolve(a, a).max())
    
    max_active = max(10, int(n * max_active_ratio))

        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.01 * c_max + 1e-12

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

                    break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: c_max={best_cmax:.12f}, score={2.0*n*best_cmax:.10f}, |S|={n_active}, t={time.monotonic()-t_start_global:.0f}s")

    return best_a, _score(best_a)


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    """Damped Newton on active set of max-convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:2 * n]
            active[top_idx] = True
            n_active = 2 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve
        x = np.zeros(n)
        r = g_t - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        u = x
        # Solve H w = 1
        x = np.zeros(n)
        r = np.ones(n) - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        w = x
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
        dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            res = minimize(
                _make_lsobj_func(beta, n),
                z,
                jac=_make_lsgrad_func(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_lsobj_func(beta, n):
    def func(z_local):
        return _lsobj(z_local, beta, n)
    return func


def _make_lsgrad_func(beta, n):
    def func(z_local):
        return _lsgrad(z_local, beta, n)
    return func


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.88
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * t
        return a / a.sum()
    elif kind == "asym_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (np.exp(-((t - 0.2) / 0.15) ** 2) + np.exp(-((t - 0.8) / 0.15) ** 2))
        return a / a.sum()
    else:
        return np.ones(n) / n


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: Successive LP -> L-BFGS -> Active-Set Newton -> Restarts.
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start_global:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_p = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "uniform_init")

    # Phase 1: Successive LP from parent
    print(f"\n[Phase 1] Successive LP from parent (budget ~600s)")
    lp_deadline = min(deadline - 500, time.monotonic() + 600)
    a_lp, sc_lp = _iterative_lp(parent, lp_deadline, max_iter=400, max_active=200)
    update_best(a_lp, sc_lp, "LP_parent")

    # Phase 2: L-BFGS polish on LP result
    print(f"\n[Phase 2] L-BFGS polish (budget ~300s)")
    if time.monotonic() < deadline - 250:
        a_lb, sc_lb = _run_lbfgs(best_seq, min(deadline - 250, time.monotonic() + 300), min_time_remaining=100)
        update_best(a_lb, _score(a_lb), "lbfgs_polish")

    # Phase 3: Active-set Newton refinement
    print(f"\n[Phase 3] Active-set Newton (budget ~200s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        as_dl = min(deadline - 100, time.monotonic() + 150)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 80:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 80, time.monotonic() + 60),
                                               max_iter=150, tol_active=0.0003)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 4: More LP iterations (finer)
    print(f"\n[Phase 4] Refined LP (budget ~200s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        lp_dl2 = min(deadline - 60, time.monotonic() + 200)
        a_lp2, sc_lp2 = _iterative_lp(best_seq, lp_dl2, max_iter=300, max_active=100, verbose=False)
        update_best(a_lp2, sc_lp2, "LP_refined")

        # L-BFGS + Newton after LP
        if time.monotonic() < deadline - 40:
            a_hyb, _ = _run_lbfgs(best_seq, min(deadline - 40, time.monotonic() + 80), min_time_remaining=20)
            update_best(a_hyb, _score(a_hyb), "hybrid")
            if time.monotonic() < deadline - 20:
                a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 15, time.monotonic() + 40),
                                                   max_iter=200, tol_active=0.0005)
                update_best(a_fin, sc_fin, "final_newton")

    # Phase 5: Try different n values
    print(f"\n[Phase 5] Multi-n exploration")
    n_tries = [4096, 8192, 16384]
    if n_parent not in n_tries:
        n_tries.append(n_parent)

    for n_try in n_tries:
        if time.monotonic() > deadline - 30:
            break
        if n_try == n_parent:
            continue

        if parent is not None and n_parent != n_try:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        print(f"    n={n_try}: init_score={sc_init:.10f}, t={time.monotonic()-t_start_global:.0f}s")

        remaining_for_n = min(120.0, deadline - 30 - time.monotonic())
        if remaining_for_n < 30:
            continue

        # Quick LP
        a_n, sc_n = _iterative_lp(a_init, time.monotonic() + remaining_for_n * 0.5, max_iter=100, max_active=100, verbose=False)
        update_best(a_n, sc_n, f"LP_n{n_try}")

        # Quick Newton
        if time.monotonic() < deadline - 25:
            a_n2, sc_n2 = _active_set_newton(a_n, min(deadline - 20, time.monotonic() + 50),
                                             max_iter=100, tol_active=0.001)
            update_best(a_n2, sc_n2, f"AS_n{n_try}")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts")
    if best_seq is not None and time.monotonic() < deadline - 20:
        restarts_configs = [
            ("perturb_003", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_010", lambda: _perturb(best_seq, 0.10, rng)),
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=5)),
            ("perturb_020", lambda: _perturb(best_seq, 0.20, rng)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 10:
                break
            p_start = gen_fn()
            sc_start = _score(p_start)
            # Quick LP on perturbed
            a_r, sc_r = _iterative_lp(p_start, min(deadline - 10, time.monotonic() + 40),
                                       max_iter=50, max_active=100, verbose=False)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 7: Final squeeze
    print(f"\n[Phase 7] Final squeeze")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 3, time.monotonic() + 30),
                                           max_iter=200, tol_active=0.0002)
        update_best(a_fin, sc_fin, "final_squeeze")

        # Tiny-beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0001
            remaining = max(2.0, deadline - 5 - time.monotonic())
            max_iter = min(5000, max(100, int(remaining / 0.03)))
            try:
                res = minimize(
                    _make_lsobj_func(beta_final, n),
                    z,
                    jac=_make_lsgrad_func(beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxcor': 50}
                )
                a_fin3 = _softmax(res.x)
                update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
