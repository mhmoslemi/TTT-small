def _next_pow2(n):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _armijo_backtrack(a, grad, lr_init, max_iter=80):
    """Armijo backtracking line search for active-set gradient descent."""
    n = len(a)
    s = a.sum()
    c0 = _full_convolve(a, a)
    f0 = 2.0 * n * float(c0.max())
    alpha = lr_init
    c_alpha = 1e-4
    for _ in range(max_iter):
        a_try = a - alpha * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            alpha *= 0.5
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = 2.0 * n * float(c_try.max())
        if f_try <= f0 - c_alpha * alpha * np.dot(grad, a - a):
            return a_try, f_try, True
        alpha *= 0.5
        if alpha < 1e-15:
            break
    return a, f0, False


def _active_set_polish(a_init, deadline, tol_frac_start=0.005,
                       tol_frac_end=0.00003, lr_init=0.04, topk_frac=0.1):
    """Active-set gradient descent with top-k selection and Armijo backtracking."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 500

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Top-k active set selection
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        # Also enforce minimum top-k
        min_k = max(1, int(n * topk_frac))
        if n_active < min_k:
            topk_idx = np.argsort(c)[::-1][:min_k]
            active = np.zeros_like(c, dtype=bool)
            active[topk_idx] = True
            n_active = min_k

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Armijo backtracking
        a_new, f_new, improved = _armijo_backtrack(a, grad_t, lr)

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.5
                no_improve = 0
                if tol_frac <= tol_frac_end * 1.1 and lr < lr_init * 0.1:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _adaptive_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta selection based on convergence."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # Adaptive beta schedule - start larger, decrease faster
    num_betas = 12
    betas = np.array([base_beta * (0.7 ** i) for i in range(num_betas)])
    betas = np.maximum(betas, max_c * 0.00005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = _logparam(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        class _Prob:
            def __init__(self, b, nn):
                self.b = b
                self.nn = nn
            def f(self, zz):
                return _smoobj(zz, self.b, self.nn)
            def g(self, zz):
                return _smograd(zz, self.b, self.nn)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob.f, z, jac=prob.g, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = _logparam(a_new)
                print(f"    beta={beta:.6e}, score={sc_new:.10f}, iter={result.nit}")
            elif i > 0 and sc_new < current_sc + 1e-6:
                # Even slight improvement or stagnation - update z for stability
                z = _logparam(a_new)
        except Exception as e:
            print(f"    beta={beta:.6e}, error: {e}")

    return current_a


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective n."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = _logparam(a)
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=6):
    n = len(a)
    z = _logparam(a)
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.05 * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _spectral_slide(a, deadline, step_frac=0.001):
    """Try small circular shifts in log-space to escape local minima."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for _ in range(200):
        if time.monotonic() > deadline:
            break
        # Random small perturbation in Fourier domain
        z = _logparam(a)
        freqs = np.fft.rfftfreq(n)
        mask = (freqs != 0)
        z_hat = np.fft.rfft(z)
        idx = np.where(mask)[0]
        k = rng.integers(len(idx))
        fi = idx[k]
        z_hat[fi] += rng.normal(0, 0.02 * n)
        z_new = np.fft.irfft(z_hat, n)
        a_new = _softmax(z_new)
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_sc
