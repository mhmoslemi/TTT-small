"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
active-set gradient descent, LP-based best response iteration,
fine L-BFGS beta continuation, and multi-phase polishing.
Key new idea: LP best-response step to escape local minima of gradient methods.
"""
import numpy as np
import time
from scipy import sparse
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adaptive_active_set(c, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(5, int(0.03 * len(c)))
    if n_active_est >= len(c):
        n_active_est = len(c)
    threshold_from_rank = sorted_c[min(n_active_est - 1, len(c) - 1)]
    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac * c_max
    tol = max(tol_fixed, min(tol_rank, 0.4 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _active_set_gradient(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_per_iter = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(20000, max(200, int(remaining / (est_per_iter * 25))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.3e}: score={sc:.10f} (iter={niter})")

    return current_a


def _peak_shaving(a_init, deadline, max_iters=30000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        w = _adaptive_active_set(c, 5e-5)
        g = _active_set_gradient(a, c, n, w)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        g_normalized = g / g_norm

        best_sc_try = best_sc
        best_a_try = best_a

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g_normalized
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-15:
                best_sc_try = sc_try
                best_a_try = a_try.copy()

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-9 or stall > 200:
                break

    return best_a, best_sc


def _lp_best_response(a, M, deadline):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    if time.monotonic() > deadline:
        return None

    # Build sparse constraint matrix C of size (2n-1) x n
    # C[k, i] = a[i] if max(0, k-n+1) <= i <= min(k, n-1)
    row_list = []
    col_list = []
    val_list = []

    for k in range(2 * n - 1):
        k_min = max(0, k - n + 1)
        k_max = min(k, n - 1)
        if k_min <= k_max:
            count = k_max - k_min + 1
            row_list.extend([k] * count)
            col_list.extend(range(k_min, k_max + 1))
            val_list.extend(a[k_min:k_max + 1].tolist())

    C = sparse.csr_matrix((val_list, (row_list, col_list)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': min(60, deadline - time.monotonic())})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception as e:
        print(f"    LP error: {e}")

    return None


def _lp_iteration(a_init, deadline, min_time_remaining=120.0, max_iters=200):
    """LP-based best response iteration."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    t_start = time.monotonic()
    for i in range(max_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if time.monotonic() - t_start > (deadline - min_time_remaining - t_start):
            break

        c = _full_convolve(a, a)
        M = float(c.max())
        s_cur = a.sum()
        M_abs = M * s_cur * s_cur  # M in absolute terms (since a is normalized to sum=1, M_abs = M)

        # Actually since a is normalized to sum=1, conv values are already in absolute terms
        M_use = M  # since sum(a) = 1, M is already the absolute max

        g = _lp_best_response(a, M_use, deadline - min_time_remaining)
        if g is None:
            break

        # Line search on step size
        best_t = 0.0
        best_sc_try = best_sc
        best_a_try = best_a

        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_t = t

        if best_t > 0 and best_sc_try < best_sc - 1e-14:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            if i % 10 == 0:
                elapsed = time.monotonic() - t_start
                print(f"    LP iter {i}: score={best_sc:.10f}, t={best_t}, time={elapsed:.1f}s")
        else:
            # Try smaller steps
            for t in [0.01, 0.005, 0.001]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, break
                if i > 5:
                    break

    return best_a


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_restart(a, rng, mode="gauss", scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.005 * rng.uniform(0.3, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(16, n // 12)
        for _ in range(12):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.02, 0.02)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent(a_init, deadline, max_rounds=50000):
    """Faster coordinate descent using gradient information for pair selection."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Precompute gradient for pair selection
    c = _full_convolve(a, a)
    w = _adaptive_active_set(c, 1e-4)
    g = _active_set_gradient(a, c, n, w)

    rng_local = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Select pair based on gradient (high positive gradient means should decrease,
        # high negative means should increase)
        if rnd % 500 == 0:
            c = _full_convolve(a, a)
            w = _adaptive_active_set(c, 1e-4)
            g = _active_set_gradient(a, c, n, w)

        # Pick the two most "imbalanced" coordinates
        g_sorted = np.argsort(g)
        idx = int(g_sorted[0])  # most negative gradient (should increase)
        # Pick a random partner to redistribute
        jdx = int(rng_local.integers(0, n))
        if jdx == idx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        # Try several splits
        for frac in [0.5, 0.3, 0.7, 0.2, 0.8, 0.4, 0.6]:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a[idx] = new_i
            a[jdx] = new_j
            sc_try = _score(a)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a.copy()
                improved = True
                break
            a[idx] = s_pair * (1.0 - frac)
            a[jdx] = s_pair * frac

    return best_a, best_sc

        a_cd, _ = _coordinate_descent(best_seq, deadline - 10, max_rounds=30000)
        try_update(a_cd, "coord_desc")

    # === Phase 9: Final ultra polish ===
    print(f"\n=== Phase 9: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(5):
            if time.monotonic() > deadline - 3:
                break
            a_f1, _ = _peak_shaving(best_seq, deadline - 3, max_iters=5000)
            try_update(a_f1, f"final_ps{cycle}")
            if time.monotonic() > deadline - 3:
                break
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-8
            max_iter = min(3000, max(200, int((deadline - 3 - time.monotonic()) / 0.004)))
            if max_iter > 50:
                a_f2, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                try_update(a_f2, f"final_lb{cycle}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
            if sc_eval < sc_final:
                print(f"  (evaluator gives lower score, using evaluator value)")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
