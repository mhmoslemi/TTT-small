def _next_pow2(n):

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _water_filling(a, n_active=5, gamma=0.5, n_iters=200, deadline=None):
    """
    Adaptive active-set water-filling.
    Iteratively equalizes the pressure on active convolution peaks.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        # Adaptive active set: start small, grow
        # Use top values within a fraction of the max
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        n_act = active.sum()
        
        if n_act < 2:
            idx = np.argsort(c)[-max(2, n_active):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = len(idx)
        
        if n_act > n // 4:
            # Cap the active set size
            idx = np.argsort(c)[-n//4:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = n // 4
        
        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        
        # Water-filling update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 10 == 0 or it == n_iters - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a, best_sc


def _nesterov_lse(a_init, deadline, min_remaining=20.0):
    """
    Nesterov-accelerated gradient descent on the log-sum-exp smoothed objective.
    Uses adaptive step size and momentum.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Beta schedule: start large, decrease
    c = _full_convolve(a_init, a_init)
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted)-1)] if len(c_sorted) > 5 else c_sorted[0] * 0.1
    beta_start = max(gap * 2, 1e-4)
    beta_end = max(gap * 0.01, 1e-7)
    
    n_betas = 8
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        
        # Nesterov parameters
        t = 1.0
        z_prev = z.copy()
        y = z.copy()
        
        # Estimate Lipschitz constant
        a_y = _softmax(y)
        c_y = _full_convolve(a_y, a_y)
        max_c = c_y.max()
        # Rough Lipschitz: L ~ 8n^2/beta (very conservative)
        # Better: use backtracking
        L = 4.0 * n * n / beta
        alpha = 1.0 / L
        
        n_steps = 200
        for step in range(n_steps):
            if time.monotonic() + min_remaining > deadline:
                break
            
            grad = _lse_gradient(y, beta, n)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            
            # Nesterov update
            z_new = y - alpha * grad
            y_new = z_new + (t - 1.0) / (t + 1.0) * (z_new - z_prev)
            
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            
            z_prev = z_new
            z = z_new
            y = y_new
            t = t_new
            
            # Check improvement periodically
            if step % 20 == 0:
                a_check = _softmax(z)
                sc = _score(a_check)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_check.copy()
        
        # Update z from current a
        a_cur = _softmax(z)
        z = np.log(np.maximum(a_cur, 1e-30))
        z = z - z.mean()
    
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0] * 0.1
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 8))))
        
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                jac=_lse_gradient,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _upsample(a, new_n):
    """Upsample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_window(n, kind="raised_cosine"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "raised_cosine":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "hann":
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == "hamming":
        a = 0.54 + 0.46 * np.cos(2 * np.pi * t)
    elif kind == "gaussian":
        a = np.exp(-4 * (t - 0.5) ** 2)
    elif kind == "sinc_like":
        a = np.sinc((t - 0.5) * 2)  # zero at edges
        a = np.abs(a)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        a_rev = a[::-1]
        w = np.zeros_like(c)
        w[idx] = 1.0
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
    else:
        w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _water_filling(a, deadline, n_iters=300, gamma=0.35):
    """Adaptive active-set water-filling with peak chasing."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive active set: values within tolerance of max
        # Use a small tolerance for peak chasing
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_act = active.sum()

        if n_act < 2:
            idx = np.argsort(c)[-max(2, min(n // 10, 5)):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = len(idx)

        if n_act > n // 3:
            idx = np.argsort(c)[-n // 3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_act = n // 3

        # Compute pressure: p_i = sum_{k in active} a_{k-i}
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Water-filling: reduce high-pressure elements, increase low-pressure
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 15 == 0 or it == n_iters - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _peak_chase(a, deadline, n_iters=200):
    """Directly chase and reduce the single highest peak."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find elements that contribute most to this peak
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # Contribution of a[i] = a[i] * a[peak_idx - i]
        valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        if len(valid_i) == 0:
            break
        contrib = a[valid_i] * a[peak_idx - valid_i]
        total = contrib.sum()
        if total < 1e-30:
            break

        # Reduce the top contributors
        n_reduce = max(1, len(valid_i) // 8)
        top_local = np.argsort(contrib)[-n_reduce:]
        top_global = valid_i[top_local]

        # Reduce these elements
        reduction = 0.3 + 0.2 * (it / n_iters)
        for i in top_global:
            a[i] *= (1.0 - reduction)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _nesterov_lse(a_init, deadline, min_remaining=20.0):
    """Nesterov-accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    c = _full_convolve(a_init, a_init)
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)] if len(c_sorted) > 5 else c_sorted[0] * 0.1
    beta_start = max(gap * 3, 1e-4)
    beta_end = max(gap * 0.01, 1e-8)

    n_betas = min(10, max(4, int((deadline - time.monotonic() - min_remaining) / 15)))
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break

        t = 1.0
        z_prev = z.copy()
        y = z.copy()

        # Adaptive step
        L = 6.0 * n * n / beta
        alpha = 1.0 / L

        n_steps = 150
        for step in range(n_steps):
            if time.monotonic() + min_remaining > deadline:
                break

            grad = _lse_grad(y, beta, n)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-13:
                break

            z_new = y - alpha * grad
            momentum = (t - 1.0) / (t + 1.0)
            y_new = z_new + momentum * (z_new - z_prev)

            t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

            z_prev = z_new
            z = z_new
            y = y_new

            if step % 25 == 0:
                a_check = _softmax(z)
                sc = _score(a_check)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_check.copy()

        # Reset z from current best
        a_cur = _softmax(z)
        z = np.log(np.maximum(a_cur, 1e-30))
        z = z - z.mean()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation on log-space parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else max_c * 0.05
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0004, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                jac=_lse_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _upsample(a, new_n):
    """Upsample a sequence to new length via interpolation + renormalization."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _make_window(n, kind="raised_cosine"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "raised_cosine":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == "hamming":
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * t)
    elif kind == "gaussian":
        a = np.exp(-3.0 * (t - 0.5) ** 2 * 4.0) + 0.05
    elif kind == "chirp_mag":
        a = np.abs(np.cos(np.pi * 2.0 * t * t * 3.0)) + 0.3
    elif kind == "flat_top":
        # Very flat in the middle, sharp edges
        a = 1.0 - 0.3 * (2.0 * t - 1.0) ** 8
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: Water-filling (quick, to get to a good basin)
    if time.monotonic() < deadline - min_remaining - 20:
        wf_deadline = min(deadline - min_remaining - 15, time.monotonic() + 40)
        a_wf, sc_wf = _water_filling(current_a, wf_deadline, n_iters=120, gamma=0.4)
        if sc_wf < current_sc:
            current_a = a_wf
            current_sc = sc_wf

    # Stage 2: L-BFGS (main workhorse)
    if time.monotonic() < deadline - min_remaining - 15:
        a_lb, sc_lb = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 15)
        if sc_lb < current_sc:
            current_a = a_lb
            current_sc = sc_lb

    # Stage 3: Water-filling (finer, smaller gamma)
    if time.monotonic() < deadline - min_remaining - 10:
        wf_deadline = min(deadline - min_remaining - 5, time.monotonic() + 60)
        a_wf2, sc_wf2 = _water_filling(current_a, wf_deadline, n_iters=200, gamma=0.25)
        if sc_wf2 < current_sc:
            current_a = a_wf2
            current_sc = sc_wf2

    # Stage 4: Peak chasing
    if time.monotonic() < deadline - min_remaining - 5:
        pc_deadline = min(deadline - min_remaining, time.monotonic() + 30)
        a_pc, sc_pc = _peak_chase(current_a, pc_deadline, n_iters=100)
        if sc_pc < current_sc:
            current_a = a_pc
            current_sc = sc_pc

    # Stage 5: Nesterov
    if time.monotonic() < deadline - min_remaining:
        a_nes, sc_nes = _nesterov_lse(current_a, deadline, min_remaining=min_remaining)
        if sc_nes < current_sc:
            current_a = a_nes
            current_sc = sc_nes

    # Stage 6: Final L-BFGS
    if time.monotonic() < deadline - min_remaining:
        a_lb2, sc_lb2 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining)
        if sc_lb2 < current_sc:
            current_a = a_lb2
            current_sc = sc_lb2

    return current_a, current_sc


def _try_symmetrize(a, deadline):
    """Try making the sequence symmetric and see if it helps."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return a, _score(a)
    a_sym = a_sym / s
    sc_sym = _score(a_sym)
    sc_orig = _score(a)
    if sc_sym < sc_orig:
        return a_sym, sc_sym
    return a, sc_orig
