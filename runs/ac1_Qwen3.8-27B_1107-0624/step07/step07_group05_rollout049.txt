"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Adam-based subgradient descent
in log-space with adaptive log-sum-exp smoothing, top-k active set gradients,
multi-length exploration, and L-BFGS polishing. Key innovation: fast momentum-
based optimizer replaces slow L-BFGS as primary driver, enabling millions of
steps within the time budget.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed_z(z, beta, n):
    """Gradient of 2n*(cmax + beta*log(sum(exp((c-cmax)/beta)))) w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    # Transform to z-space: dz = a * (da - a*dot(a,da))
    mean_term = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_term)
    return grad_z


def _grad_topk_z(z, k, n):
    """Gradient using uniform weights over top-k convolution values."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    if k >= len(c):
        w = np.ones_like(c) / len(c)
    else:
        top_idx = np.argpartition(c, -k)[-k:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    mean_term = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_term)
    return grad_z


def _adam_optimize(z_init, deadline, beta_init, beta_min, n_steps_max,
                   lr=0.05, b1=0.9, b2=0.999, eps=1e-8,
                   use_topk=False, topk_frac=0.1, topk_min=5,
                   beta_decay=0.95, beta_check_interval=200):
    """
    Adam optimizer in log-space with adaptive smoothing.
    Returns (best_a, best_score_normalized).
    """
    n = len(z_init)
    z = z_init.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    a = _softmax(z)
    best_a = a.copy()
    best_sc = _score_norm(a)
    beta = beta_init
    t = 0
    no_improve = 0

    for step in range(n_steps_max):
        if time.monotonic() > deadline:
            break
        t += 1

        # Choose gradient type
        if use_topk and step > beta_check_interval * 3:
            k = max(topk_min, int(len(a) * topk_frac))
            grad = _grad_topk_z(z, k, n)
        else:
            grad = _grad_smoothed_z(z, beta, n)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()
        z = np.clip(z, -50.0, 50.0)

        # Periodic evaluation and beta adaptation
        if step % beta_check_interval == 0:
            a = _softmax(z)
            sc = _score_norm(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Decay beta
            beta = max(beta_min, beta * beta_decay)
            # Reduce lr if no improvement for a while
            if no_improve > 5:
                lr *= 0.8
                no_improve = 0
            # If lr too small, reset and use topk
            if lr < 1e-4:
                lr = 0.05
                use_topk = True

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_polish(a_init, deadline, min_remaining=5.0):
    """L-BFGS polishing with log-sum-exp smoothing."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, c_max * 0.01)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score_norm(current_a)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005])
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (est_per_iter * 10))))
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                jac=_lbfgs_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-10, 'maxcor': 30, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _lbfgs_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    mean_term = float(np.dot(a, grad_a))
    return a * (grad_a - mean_term)


def _pressure_equalize(a, deadline, n_iters=100, gamma=0.3):
    """Multiplicative update to equalize pressure over active set."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-max(3, n // 100)]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        p_mean = np.mean(p)
        a = a * np.power(p_mean / p, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _make_asymmetric(n, rng, severity=0.1):
    """Create an asymmetric initialization close to uniform."""
    t = np.linspace(0, 1, n)
    f1 = rng.integers(1, 5)
    f2 = rng.integers(3, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a = 1.0 + severity * np.sin(2 * np.pi * f1 * t + p1) \
            + severity * 0.5 * np.cos(2 * np.pi * f2 * t + p2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _fourier_perturb(a, rng, n_modes=6, amp=0.03):
    """Perturb in Fourier domain for diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amp * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: Adam-based subgradient descent in log-space with
    adaptive smoothing, pressure equalization, L-BFGS polish, and multi-n
    exploration. Optimized for the 1800s budget.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        if sc is None:
            sc = _score_norm(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score_norm(parent):.10f}")
        _update(parent)
    else:
        n_parent = 4096
        parent = _make_asymmetric(n_parent, rng, severity=0.05)
        print(f"Generated parent: n={n_parent}, score={_score_norm(parent):.10f}")
        _update(parent)

    # --- Phase 1: Main Adam optimization on parent ---
    print(f"\n[Phase 1] Adam optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    n_main = n_parent
    z_main = np.log(np.maximum(parent, 1e-30))
    z_main = z_main - z_main.mean()

    c_main = _full_convolve(parent, parent)
    c_max_main = float(c_main.max())
    beta_init = c_max_main * 0.3
    beta_min = c_max_main * 0.001
    t1_deadline = min(deadline - 400, t_start + 600)
    a1, sc1 = _adam_optimize(z_main, t1_deadline, beta_init, beta_min,
                             n_steps_max=500000, lr=0.08, beta_decay=0.92,
                             beta_check_interval=100)
    _update(a1, sc1)

    # --- Phase 2: Pressure equalization ---
    print(f"\n[Phase 2] Pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        t2_deadline = min(deadline - 200, time.monotonic() + 120)
        a2, sc2 = _pressure_equalize(best_seq, t2_deadline, n_iters=200, gamma=0.35)
        sc2_full = _score_norm(a2)
        _update(a2, sc2_full)
        if sc2_full < best_score + 1e-10:
            # Follow with more Adam
            z2 = np.log(np.maximum(a2, 1e-30))
            z2 = z2 - z2.mean()
            c2 = _full_convolve(a2, a2)
            bm = float(c2.max()) * 0.1
            t2b_deadline = min(deadline - 150, time.monotonic() + 120)
            a2b, sc2b = _adam_optimize(z2, t2b_deadline, bm, bm * 0.005,
                                       n_steps_max=200000, lr=0.05,
                                       beta_decay=0.9, beta_check_interval=100)
            _update(a2b, sc2b)

    # --- Phase 3: L-BFGS polish ---
    print(f"\n[Phase 3] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        t3_deadline = min(deadline - 50, time.monotonic() + 180)
        a3 = _lbfgs_polish(best_seq, t3_deadline, min_remaining=30.0)
        sc3 = _score_norm(a3)
        _update(a3, sc3)

    # --- Phase 4: More Adam on best ---
    print(f"\n[Phase 4] Extended Adam (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        z4 = np.log(np.maximum(best_seq, 1e-30))
        z4 = z4 - z4.mean()
        c4 = _full_convolve(best_seq, best_seq)
        cmax4 = float(c4.max())
        t4_deadline = min(deadline - 40, time.monotonic() + 200)
        a4, sc4 = _adam_optimize(z4, t4_deadline, cmax4 * 0.1, cmax4 * 0.0005,
                                 n_steps_max=300000, lr=0.03, beta_decay=0.93,
                                 beta_check_interval=150, use_topk=True,
                                 topk_frac=0.05, topk_min=3)
        _update(a4, sc4)

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 2048]
    n_candidates = [x for x in n_candidates if x != len(best_seq) if best_seq is not None]
    if n_parent > 0:
        n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 30:
            break

        # Resample from best
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_asymmetric(n_try, rng, 0.05)

        # Quick Adam
        z_try = np.log(np.maximum(a_init, 1e-30))
        z_try = z_try - z_try.mean()
        c_try = _full_convolve(a_init, a_init)
        bm_try = float(c_try.max()) * 0.2
        t_try_dl = min(deadline - 30, time.monotonic() + max(20, remaining * 0.5))
        a_try, sc_try = _adam_optimize(z_try, t_try_dl, bm_try, bm_try * 0.005,
                                       n_steps_max=100000, lr=0.06,
                                       beta_decay=0.91, beta_check_interval=80)
        _update(a_try, sc_try)

        # Quick L-BFGS
        if time.monotonic() < deadline - 20:
            a_try2 = _lbfgs_polish(a_try, deadline - 10, min_remaining=5.0)
            _update(a_try2, _score_norm(a_try2))

    # --- Phase 6: Diverse restarts from best ---
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        for i in range(4):
            if time.monotonic() > deadline - 15:
                break
            # Asymmetric perturbation
            a_div = _make_asymmetric(n_best, rng, severity=0.08 * (i + 1))
            a_blend = 0.5 * best_seq + 0.5 * a_div
            a_blend = np.maximum(a_blend, 1e-10)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=4, amp=0.02)

            z_div = np.log(np.maximum(a_pert, 1e-30))
            z_div = z_div - z_div.mean()
            c_div = _full_convolve(a_pert, a_pert)
            bm_div = float(c_div.max()) * 0.15
            t_div_dl = min(deadline - 10, time.monotonic() + 60)
            a_div_opt, sc_div = _adam_optimize(z_div, t_div_dl, bm_div,
                                               bm_div * 0.003,
                                               n_steps_max=80000, lr=0.05,
                                               beta_decay=0.9,
                                               beta_check_interval=80)
            _update(a_div_opt, sc_div)
            if sc_div < best_score:
                a_div2 = _lbfgs_polish(a_div_opt, deadline - 5, min_remaining=3.0)
                _update(a_div2, _score_norm(a_div2))

    # --- Phase 7: Final polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _lbfgs_polish(best_seq, deadline - 3, min_remaining=2.0)
        _update(a_fin, _score_norm(a_fin))

        # One more Adam pass with very small beta
        if time.monotonic() < deadline - 3:
            z_fin = np.log(np.maximum(best_seq, 1e-30))
            z_fin = z_fin - z_fin.mean()
            c_fin = _full_convolve(best_seq, best_seq)
            bm_fin = float(c_fin.max()) * 0.01
            a_fin2, sc_fin2 = _adam_optimize(z_fin, deadline - 2, bm_fin,
                                             max(bm_fin * 0.1, 1e-6),
                                             n_steps_max=50000, lr=0.02,
                                             beta_decay=0.95,
                                             beta_check_interval=50,
                                             use_topk=True, topk_frac=0.02,
                                             topk_min=2)
            _update(a_fin2, sc_fin2)

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        final = final / final.sum()
        sc_final = _score_norm(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    return [float(x) for x in fallback]
