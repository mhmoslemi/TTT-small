The score is 2n * max_k c_k / (sum a)^2 where c = conv(a,a).

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = c.mean()

    # d(c_k)/d(a_i) = 2*a[k-i] for valid k-i (0 <= k-i < n)
    # grad_i = 2/(2n-1) * sum_k (c_k - mean_c) * 2*a[k-i]

    a_flip = a[::-1]
    prod = _full_convolve(centered, a_flip)
    g = 4.0 / (2 * n - 1) * prod[n - 1:2 * n - 1]
    return g

def _variance_lbfgs(a_init, deadline, min_time_remaining=60.0):

    n = len(a_init)

        return _variance_obj_val(a_local)

        g = _variance_grad(a_local, n)
        return g / a_local  # chain rule through softmax

def _coordinate_descent(a_init, deadline, max_rounds=5):
    """Coordinate descent: vary one element at a time."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for round_i in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try varying a[i]
            old_val = a[i]
            # The score depends on a[i] through:
            # c[k] for k in [i, i+n-1] (the convolution indices involving a[i])
            # c[k] = sum_j a[j]*a[k-j], and a[i] appears in c[k] as a[i]*a[k-i] + a[k-i]*a[i]
            # So c[k] = a[i]*(2*a[k-i]) + (other terms not involving a[i])
            # For the max, we want to decrease the max c[k].
            
            # Simple approach: try a few values
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 2.0]:
                a[i] = old_val * factor
                a[i] = max(0.0, a[i])
                s = a.sum()
                if s < 0.01:
                    a[i] = old_val
                    continue
                sc = _score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = old_val  # Restore
        
        if not improved:
            break
    
    return best_a, best_sc

def _pairwise_swap(a_init, deadline, n_pairs=10000):
    """Try pairwise adjustments to reduce the score."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    
    for _ in range(n_pairs):
        if time.monotonic() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Try swapping or adjusting a[i] and a[j]
        old_i, old_j = a[i], a[j]
        for frac in [0.1, 0.3, 0.5]:
            a[i] = old_i + frac * (old_j - old_i)
            a[j] = old_j + frac * (old_i - old_j)
            a[i] = max(0.0, a[i])
            a[j] = max(0.0, a[j])
            s = a.sum()
            if s < 0.01:
                a[i], a[j] = old_i, old_j
                continue
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                break
        a[i], a[j] = old_i, old_j
    
    return best_a, best_sc

def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent on the active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.95
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc
