```python
"""
Optimizer for non-negative step sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The code combines several local methods:
  - a Hankel/KKT-inspired convex subproblem (pressure equalization)
  - fast pressure equalization updates
  - log-sum-exp smoothed L-BFGS in log-space
  - coordinate polishing and multi-scale / multi-length exploration

All helpers are top-level functions; there are no lambdas or closures.
"""

import time
import numpy as np
from scipy.optimize import minimize

_TIMEOUT_DEADLINE = 0.0
_TIMEOUT_REMAINING = 0.0
_REG_Z = 1e-12


class _Timeout(Exception):
    pass


def _stop_callback(pk=None):
    if time.monotonic() >= _TIMEOUT_DEADLINE - _TIMEOUT_REMAINING:
        raise _Timeout


def _next_pow2(m):
    k = 2
    while k < max(2, int(m)):
        k <<= 1
    return k


def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    if not np.all(np.isfinite(a)):
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15 or not np.isfinite(s):
        return np.inf
    n = a.size
    c = _convolve_same(a)
    m = float(c.max()) if c.size else 0.0
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _consider(a, best, best_sc):
    if a is None:
        return best, best_sc
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return best, best_sc
    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        return _norm(a).copy(), float(sc)
    return best, best_sc


def _convolve_same(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(out, 0.0)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return z.copy()
    z = z - float(np.mean(z))
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _center_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return z.copy()
    z = z - float(np.mean(z))
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -70.0, 70.0)


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v.copy()
    if not np.all(np.isfinite(v)):
        return np.ones(n, dtype=np.float64) / float(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (css - 1.0) / rho > 0.0
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    k = int(idx[-1])
    theta = (css[k] - 1.0) / float(k + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def _resample(a, new_n):
    a = _norm(a)
    old_n = a.size
    new_n = int(max(1, new_n))
    if new_n == old_n:
        return a.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return _norm(b)


def _hankel_prepare(u, N):
    nu = np.zeros(int(N), dtype=np.float64)
    m = min(int(N), u.size)
    if m > 0:
        nu[:m] = u[:m]
    return np.fft.rfft(nu)


def _hankel_matvec_pre(v, Uhat, N):
    v = np.asarray(v, dtype=np.float64).ravel()
    nv = np.zeros(int(N), dtype=np.float64)
    m = min(int(N), v.size)
    if m > 0:
        nv[:m] = v[:m]
    ov = np.fft.rfft(nv)
    out = np.fft.irfft(np.conj(Uhat) * ov, int(N))
    return out[:v.size]


def _weights_from_conv(C, power):
    C = np.asarray(C, dtype=np.float64).ravel()
    C = np.where(np.isfinite(C), C, 0.0)
    C = np.maximum(C, 0.0)
    if C.size == 0:
        return np.array([], dtype=np.float64)
    try:
        w = np.power(C, float(power))
    except FloatingPointError:
        w = np.zeros_like(C)
    w = np.where(np.isfinite(w), w, 0.0)
    wm = float(w.max())
    if wm > 0.0:
        w = w + 1e-12 * wm
    sw = float(w.sum())
    if sw >= 1e-300 and np.isfinite(sw):
        w = w / sw
    else:
        k = max(3, C.size // 20)
        idx = np.argsort(C)[-k:]
        w = np.zeros_like(C)
        w[idx] = 1.0
        w /= float(w.size)
    return w


def _make_profile(n, kind, rng=None):
    n = int(max(1, n))
    if rng is None:
        rng = np.random.default_rng(12345)
    t = np.linspace(0.0, 1.0, n)

    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "ramp_down":
        a = 1.0 + 2.0 * (1.0 - t)
    elif kind == "ramp_up":
        a = 1.0 + 2.0 * t
    elif kind == "center":
        a = 0.15 + np.sin(np.pi * t) ** 0.7
    elif kind == "spank":
        a = 1.0 + 0.6 * np.cos(2.0 * np.pi * t)
    elif kind == "mixture":
        centers = rng.uniform(0.03, 0.97, size=5)
        amps = rng.uniform(0.25, 1.6, size=5)
        widths = rng.uniform(0.015, 0.16, size=5)
        a = np.zeros(n, dtype=np.float64)
        for c, amp, wd in zip(centers, amps, widths):
            a += amp * np.exp(-0.5 * ((t - c) / max(1e-6, wd)) ** 2)
    else:
        a = _smooth_random_profile(n, rng, 0.35)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _smooth_random_profile(n, rng, bandwidth=0.3):
    n = int(max(1, n))
    z = rng.standard_normal(n)
    f = np.fft.rfft(z)
    freq = np.fft.rfftfreq(n)
    mask = np.exp(-np.pi * (freq * n * float(bandwidth)) ** 2)
    f *= mask
    x = np.fft.irfft(f, n)
    x = np.where(np.isfinite(x), x, 0.0)
    a = np.exp(0.9 * x)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    a = _norm(a)
    n = a.size
    if n <= 1:
        return a.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - float(np.mean(z))
    zh = np.fft.rfft(z, n)
    for _ in range(int(n_modes)):
        fmax = max(2, n // 3)
        fr = int(rng.integers(1, fmax))
        if 0 <= fr < zh.size:
            zh[fr] = zh[fr] + float(amplitude) * (
                rng.standard_normal() + 1j * rng.standard_normal()
            )
    zn = np.fft.irfft(zh, n)
    zn = np.where(np.isfinite(zn), zn, 0.0)
    zn -= float(np.mean(zn))
    return _softmax(zn)


def _get_starts(n, parent, rng, count=5):
    out = []
    if parent is not None and parent.size > 1:
        base = _resample(parent, n)
        out.append(base)
        out.append(_fourier_perturb(base, rng, 4, 0.05))
    out.append(_make_profile(n, "ramp_down", rng))
    out.append(_make_profile(n, "ramp_up", rng))
    out.append(_make_profile(n, "mixture", rng))
    out.append(_smooth_random_profile(n, rng, 0.25))
    out.append(_smooth_random_profile(n, rng, 0.60))
    out.append(_fourier_perturb(out[-1], rng, 6, 0.07))
    kept = []
    for x in out[:int(max(1, count))]:
        x = _norm(x)
        if x.size == int(n):
            kept.append(x)
    return kept


def _hankel_simplex_solver(Uhat, N, n, rhs_scale, reg, max_iter, x0):
    x = _norm(x0)
    y = x.copy()
    t = 1.0
    step = 0.9 / (1.0 + float(reg))

    for _ in range(int(max_iter)):
        Hy = _hankel_matvec_pre(y, Uhat, int(N))
        if not np.all(np.isfinite(Hy)):
            break
        d = Hy - float(rhs_scale)
        g = _hankel_matvec_pre(d, Uhat, int(N)) + float(reg) * y
        if not np.all(np.isfinite(g)):
            break

        z = y - step * g
        x_new = _project_simplex(z)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _project_simplex(y)

        diff = float(np.linalg.norm(x_new - x))
        x = x_new
        t = t_new
        if diff < 1e-8:
            break

    return _norm(x)


def _kkt_refine(a_init, deadline, min_remaining, powers, inner_iters, outer_iters):
    a_init = _norm(a_init)
    if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
        return a_init.copy(), _score(a_init)

    a = a_init.copy()
    best = a.copy()
    best_sc = _score(a)
    n = a.size
    N = _next_pow2(2 * n - 1)

    for power in list(powers):
        patience = 0
        for _ in range(int(outer_iters)):
            if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
                break

            c = _convolve_same(a)
            w = _weights_from_conv(c, power)
            Uhat = _hankel_prepare(w, N)

            p = _hankel_matvec_pre(a, Uhat, N)
            if not np.all(np.isfinite(p)):
                break
            m = float(np.mean(p))
            if not np.isfinite(m) or abs(m) < 1e-30:
                m = 1.0 / float(n)

            x = _hankel_simplex_solver(Uhat, N, n, m, 0.0, int(inner_iters), a)

            sc_x = _score(x)
            improved = False

            if np.isfinite(sc_x) and sc_x < best_sc - 1e-13:
                a = x
                best = a.copy()
                best_sc = float(sc_x)
                patience = 0
                improved = True
            else:
                diff = float(np.linalg.norm(x - a))
                alpha = float(np.clip(0.45 / (1.0 + diff), 0.08, 0.75))
                cand = (1.0 - alpha) * a + alpha * x
                cand = _norm(cand)
                sc_c = _score(cand)
                if np.isfinite(sc_c) and sc_c < best_sc - 1e-13:
                    a = cand
                    best = a.copy()
                    best_sc = float(sc_c)
                    patience = 0
                    improved = True

            if not improved:
                patience += 1
                if patience >= 4:
                    break

    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = _norm(a).copy()
    return best, best_sc


def _pressure_refine(a_init, deadline, min_remaining, power=2.0, n_iters=80, gamma=0.35):
    a_init = _norm(a_init)
    if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
        return a_init.copy(), _score(a_init)

    a = a_init.copy()
    best = a.copy()
    best_sc = _score(a)
    n = a.size
    N = _next_pow2(2 * n - 1)
    patience = 0

    for _ in range(int(n_iters)):
        if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
            break

        c = _convolve_same(a)
        w = _weights_from_conv(c, power)
        Uhat = _hankel_prepare(w, N)
        p = _hankel_matvec_pre(a, Uhat, N)
        if not np.all(np.isfinite(p)):
            break

        p = np.maximum(p, 1e-30)
        m = float(np.mean(p))
        if m <= 1e-30 or not np.isfinite(m):
            break

        ratio = np.power(m / p, float(gamma))
        ratio = np.where(np.isfinite(ratio), ratio, 1.0)
        a_new = _norm(a * ratio)

        sc_new = _score(a_new)
        if np.isfinite(sc_new) and sc_new < best_sc - 1e-13:
            a = a_new
            best = a.copy()
            best_sc = float(sc_new)
            patience = 0
        else:
            blend = _norm(0.5 * a + 0.5 * a_new)
            sc_b = _score(blend)
            if np.isfinite(sc_b) and sc_b < best_sc - 1e-13:
                a = blend
                best = a.copy()
                best_sc = float(sc_b)
                patience = 0
            else:
                patience += 1
                if patience >= 10:
                    break

    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = _norm(a).copy()
    return best, best_sc


def _lsobj(z, beta, n, N):
    a = _softmax(z)
    L = 2 * n - 1
    N = int(N)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return 1e100
    m = float(c.max())
    if not np.isfinite(m):
        return 1e100

    shifted = (c - m) / float(beta)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        f = 2.0 * float(n) * m
    else:
        f = 2.0 * float(n) * (m + float(beta) * np.log(s))

    z0 = np.asarray(z, dtype=np.float64).ravel() - float(np.mean(z))
    z0 = np.where(np.isfinite(z0), z0, 0.0)
    f += _REG_Z * float(np.dot(z0, z0))
    if not np.isfinite(f):
        return 1e100
    return float(f)


def _lsgrad(z, beta, n, N):
    z = np.asarray(z, dtype=np.float64).ravel()
    z0 = z - float(np.mean(z))
    z0 = np.where(np.isfinite(z0), z0, 0.0)
    a = _softmax(z0)
    L = 2 * n - 1
    N = int(N)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.zeros_like(z)

    m = float(c.max())
    if not np.isfinite(m):
        return np.zeros_like(z)

    shifted = (c - m) / float(beta)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    w = np.where(np.isfinite(w), w, 0.0)
    sw = float(w.sum())
    if sw <= 0.0:
        return np.zeros_like(z)
    w = w / sw

    Uhat = _hankel_prepare(w, N)
    p = np.fft.irfft(np.conj(Uhat) * fa, N)[:n]
    p = np.where(np.isfinite(p), p, 0.0)

    ga = 4.0 * float(n) * p
    dot = float(np.dot(a, ga))
    if not np.isfinite(dot):
        return np.zeros_like(z)

    gz = a * (ga - dot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz += 2.0 * _REG_Z * z0
    gz = np.clip(gz, -1e6, 1e6)
    return gz


def _lbfgs_refine(a_init, deadline, min_remaining):
    a_init = _norm(a_init)
    if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
        return a_init.copy(), _score(a_init)

    global _TIMEOUT_DEADLINE, _TIMEOUT_REMAINING
    _TIMEOUT_DEADLINE = float(deadline)
    _TIMEOUT_REMAINING = max(1.0, float(min_remaining))

    a = a_init.copy()
    best = a.copy()
    best_sc = _score(a)
    n = a.size
    N = _next_pow2(2 * n - 1)

    z = _center_z(np.log(np.maximum(a, 1e-30)))

    factors = (0.35, 0.18, 0.09, 0.04, 0.02, 0.01, 0.005, 0.002)
    for fac in factors:
        if time.monotonic() >= _TIMEOUT_DEADLINE - _TIMEOUT_REMAINING:
            break

        c = _convolve_same(a)
        m = float(c.max()) if c.size else 0.0
        if m <= 1e-300 or not np.isfinite(m):
            break
        beta = float(m) * float(fac)
        if beta < 1e-300:
            break

        remaining = max(1.0, _TIMEOUT_DEADLINE - time.monotonic())
        est = 0.22 * max(1.0, float(N) / 32768.0)
        maxiter = int(min(260, max(25, remaining / (est * 4.0))))

        try:
            res = minimize(
                _lsobj,
                z,
                args=(beta, n, N),
                jac=_lsgrad,
                method="L-BFGS-B",
                options={
                    "maxiter": maxiter,
                    "ftol": 1e-16,
                    "gtol": 1e-11,
                    "maxcor": 40,
                    "maxls": 80,
                },
                callback=_stop_callback,
            )
            a_new = _norm(_softmax(res.x))
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < best_sc - 1e-14:
                best = a_new.copy()
                best_sc = float(sc_new)
                a = a_new.copy()
                z = _center_z(np.log(np.maximum(a, 1e-30)))
        except _Timeout:
            break
        except Exception:
            pass

    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = _norm(a).copy()
    return best, best_sc


def _coord_polish(a_init, deadline, min_remaining, rng=None, rounds=20):
    a_init = _norm(a_init)
    if rng is None:
        rng = np.random.default_rng(987654)
    if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
        return a_init.copy(), _score(a_init)

    a = a_init.copy()
    best = a.copy()
    best_sc = _score(a)
    n = a.size
    N = _next_pow2(2 * n - 1)

    for _ in range(int(rounds)):
        if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
            break

        c = _convolve_same(a)
        w = _weights_from_conv(c, 2.0)
        Uhat = _hankel_prepare(w, N)
        p = _hankel_matvec_pre(a, Uhat, N)
        p = np.where(np.isfinite(p), p, 0.0)
        contrib = np.maximum(p, 0.0) * a
        q = int(min(35, max(12, n // 120)))
        if contrib.size < q:
            q = contrib.size
        if q <= 0:
            break
        top = np.argsort(contrib)[-q:]
        if top.size == 0:
            break
        order = rng.permutation(top)
        improved = False

        for i in order:
            if time.monotonic() >= float(deadline) - max(0.0, float(min_remaining)):
                break
            base = a
            old = float(base[i])
            if old <= 1e-15:
                continue
            for factor in (0.55, 0.78, 1.28, 1.55):
                cand = base.copy()
                cand[i] = old * factor
                s = float(cand.sum())
                if s <= 1e-15:
                    continue
                cand = cand / s
                sc = _score(cand)
                if np.isfinite(sc) and sc < best_sc - 1e-14:
                    a = cand
                    best = a.copy()
                    best_sc = float(sc)
                    base = a
                    old = float(base[i])
                    improved = True
                    break

        if not improved:
            break

    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best = _norm(a).copy()
    return best, best_sc


def _optimize_one(a_init, phase_deadline, full=True):
    a_init = _norm(a_init)
    best = a_init.copy()
    best_sc = _score(best)

    t0 = time.monotonic()
    total = max(1.0, float(phase_deadline) - t0 - 2.0)

    if full:
        p1_stop = t0 + total * 0.16
        k1_stop = t0 + total * 0.46
        l_stop = t0 + total * 0.76
        k2_stop = t0 + total * 0.90
        p2_stop = t0 + total * 0.97
        c_stop = float(phase_deadline)

        out, sc = _pressure_refine(best, p1_stop, 1.0, power=1.7, n_iters=70, gamma=0.32)
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _kkt_refine(
            best, k1_stop, 1.0,
            powers=(1.0, 1.4, 2.0, 3.0),
            inner_iters=38,
            outer_iters=12,
        )
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _lbfgs_refine(best, l_stop, 1.0)
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _kkt_refine(
            best, k2_stop, 1.0,
            powers=(2.2, 3.5, 6.0, 9.0),
            inner_iters=48,
            outer_iters=10,
        )
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _pressure_refine(best, p2_stop, 1.0, power=3.0, n_iters=60, gamma=0.25)
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _coord_polish(best, c_stop, 1.0, rounds=14)
        best, best_sc = _consider(out, best, best_sc)
    else:
        p_stop = t0 + total * 0.36
        k_stop = t0 + total * 0.78
        l_stop = float(phase_deadline)

        out, sc = _pressure_refine(best, p_stop, 1.0, power=2.0, n_iters=50, gamma=0.32)
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _kkt_refine(
            best, k_stop, 1.0,
            powers=(1.5, 2.5, 5.0),
            inner_iters=28,
            outer_iters=8,
        )
        best, best_sc = _consider(out, best, best_sc)

        out, sc = _lbfgs_refine(best, l_stop, 1.0)
        best, best_sc = _consider(out, best, best_sc)

    sc = _score(best)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
    return best, best_sc

            sc = _score(s)
            if np.isfinite(sc) and sc < best_score:
                best_seq = _norm(s)
                best_score = float(sc)
                candidates.append(best_seq.copy())

    if best_seq is None:
        best_seq = _make_profile(1024, "ramp_down", rng)
        best_score = _score(best_seq)
        candidates.append(best_seq.copy())

    n_main = int(best_seq.size)
    if n_main < 4096:
        cand = _resample(best_seq, 8192)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = cand, float(sc)
            candidates.append(best_seq.copy())
            n_main = 8192

    long_n = int(min(32768, max(4096, 2 * n_main)))
    long_n = int(max(1024, int(round(long_n / 1024.0) * 1024)))

    # Low-resolution exploration.
    low_deadline = min(deadline - 850.0, t0 + 190.0)
    if low_deadline - time.monotonic() > 70.0:
        n_small = 512 if n_main <= 16384 else 1024
        small_starts = _get_starts(n_small, best_seq, rng, 6)
        small_best = None
        small_score = np.inf
        for s in small_starts:
            if time.monotonic() >= low_deadline - 6.0:
                break
            try:
                a_s, sc_s = _optimize_one(s, low_deadline, full=False)
                if np.isfinite(sc_s) and sc_s < small_score:
                    small_best = a_s
                    small_score = float(sc_s)
            except Exception:
                pass

        if small_best is not None:
            targets = set()
            targets.add(n_main)
            targets.add(long_n)
            for tn in targets:
                cand = _resample(small_best, tn)
                sc = _score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best_seq, best_score = cand, float(sc)
                    candidates.append(best_seq.copy())
                if best_seq.size == tn:
                    base = _resample(best_seq, tn)
                    small_same = _resample(small_best, tn)
                    blend = _norm(0.72 * base + 0.28 * small_same)
                    sc_b = _score(blend)
                    if np.isfinite(sc_b) and sc_b < best_score:
                        best_seq, best_score = blend, float(sc_b)
                        candidates.append(best_seq.copy())

    # Main optimization on the best length found so far.
    main_deadline = deadline - 520.0
    if main_deadline - time.monotonic() > 40.0:
        try:
            a_main, sc_main = _optimize_one(best_seq, main_deadline, full=True)
            best_seq, best_score = _consider(a_main, best_seq, best_score)
            candidates.append(a_main.copy())
        except Exception:
            pass

    # Try a longer length.
    long_deadline = deadline - 230.0
    if (
        long_deadline - time.monotonic() > 150.0
        and long_n != int(best_seq.size)
        and best_seq.size <= 16384
    ):
        try:
            a_long = _resample(best_seq, long_n)
            a_long, sc_long = _optimize_one(a_long, long_deadline, full=True)
            best_seq, best_score = _consider(a_long, best_seq, best_score)
            candidates.append(a_long.copy())
        except Exception:
            pass

    # Try an intermediate length if time remains.
    alt_n = int(max(4096, min(32768, int(round(best_seq.size * 1.5 / 1024.0) * 1024))))
    alt_deadline = deadline - 90.0
    if alt_deadline - time.monotonic() > 170.0 and alt_n != int(best_seq.size):
        try:
            a_alt = _resample(best_seq, alt_n)
            a_alt, sc_alt = _optimize_one(a_alt, alt_deadline, full=False)
            best_seq, best_score = _consider(a_alt, best_seq, best_score)
            candidates.append(a_alt.copy())
        except Exception:
            pass

    # Final polish.
    final_deadline = deadline - 2.0
    if final_deadline - time.monotonic() > 35.0:
        try:
            full_final = (final_deadline - time.monotonic()) > 260.0
            a_fin, sc_fin = _optimize_one(best_seq, final_deadline, full=full_final)
            best_seq, best_score = _consider(a_fin, best_seq, best_score)
            candidates.append(a_fin.copy())
        except Exception:
            pass

    # Use the actual evaluator, if available, to choose among retained candidates.
    final_seq = best_seq
    try:
        best_ev_seq = None
        best_ev_score = np.inf
        for arr in candidates:
            if arr is None or arr.size == 0:
                continue
            seq = [float(x) for x in arr]
            ev = evaluate_sequence(seq)
            if np.isfinite(ev) and ev < best_ev_score:
                best_ev_score = float(ev)
                best_ev_seq = _norm(arr)
        if best_ev_seq is not None:
            final_seq = best_ev_seq
    except Exception:
        pass

    if final_seq is None or final_seq.size == 0:
        final_seq = _make_profile(1024, "ramp_down", np.random.default_rng(1))

    final_seq = _norm(final_seq)
    return [float(x) for x in final_seq]
```
