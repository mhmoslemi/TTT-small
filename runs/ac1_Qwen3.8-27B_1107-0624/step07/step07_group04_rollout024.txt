"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: spectral initialization, aggressive L-BFGS log-sum-exp continuation,
KKT-based active set Newton with CG, dynamic active set management,
perturbation restarts, and support reduction.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _kkt_newton_polish(a_init, deadline, max_rounds=200, tight=False):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.0002 if tight else 0.001
    expand_tol = 0.005

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Dynamic active set with hysteresis
        if rnd == 0 or rnd % 10 == 0:
            # Re-compute active set with wider tolerance every 10 rounds
            active = c >= (c_max - expand_tol * c_max)
        else:
            active = c >= (c_max - tol_frac * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for numerical stability
        if n_active > 2 * n:
            threshold = np.sort(c)[::-1][min(2*n, len(c)-1)]
            active = c >= threshold
            n_active = int(active.sum())

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)

        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        # Newton step via CG
        Hg = _cg_solve(W, n, g, max_iter=400)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=400)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        c_max_old = c_max
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:

                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Sufficient decrease condition

                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Fallback: pure gradient step
            alpha = 0.005
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max_old * (1 - 1e-10):
                    a = a_try
                    step_done = True
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_extreme(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    # Very aggressive beta schedule
    num_betas = 30
    betas = base_beta * (0.6 ** np.arange(num_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

def _spectral_init(n, rng):
    """Generate a sequence with approximately flat circular autocorrelation."""
    phases = rng.uniform(0, 2 * np.pi, n)
    spectrum = np.zeros(n, dtype=complex)
    spectrum[0] = np.sqrt(n)  # DC component gives sum a = 1 after normalization
    for k in range(1, (n // 2) + 1):
        mag = 1.0 / np.sqrt(n)
        spectrum[k] = mag * np.exp(1j * phases[k % n])
        if k < n - k:
            spectrum[n - k] = np.conj(spectrum[k])
    a = np.fft.ifft(spectrum).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _tapered_init(n, kind="v_shape"):
    t = np.linspace(0, 1, n)
    if kind == "v_shape":
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 0.01)
    elif kind == "u_shape":
        a = 4.0 * (t - 0.5) ** 2 + 0.05
    elif kind == "hump":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 0.01)
    elif kind == "two_peaks":
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
    elif kind == "three_peaks":
        a = (np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.5) ** 2)
             + np.exp(-30.0 * (t - 0.8) ** 2) + 0.01)
    elif kind == "asym_seg":
        # Asymmetric: more mass on left
        a = np.exp(-5.0 * t) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return a / a.sum()


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _support_reduce(a, frac, reopt_func, deadline, min_time=30.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    idx = np.argsort(a)
    n_zero = max(1, int(n * frac))
    a_try = best_a.copy()
    a_try[idx[:n_zero]] = 0.0
    s_try = a_try.sum()
    if s_try < 1e-10:
        return best_a, best_sc
    a_try = a_try / s_try
    sc_try = _score(a_try)
    if sc_try < best_sc - 1e-12:
        best_sc = sc_try
        best_a = a_try.copy()
        if time.monotonic() + min_time < deadline:
            a_re = reopt_func(best_a, deadline, min_time)
            sc_re = _score(a_re)
            if sc_re < best_sc:
                best_a = a_re
                best_sc = sc_re
    return best_a, best_sc


def _grad_descent_smooth(a_init, deadline, min_time=30.0, max_iter=5000):
    """Smooth gradient descent with adaptive step size in log domain."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.01
    patience = 0
    beta = _full_convolve(a_init, a_init).max() * 0.01

    for i in range(max_iter):
        if time.monotonic() + min_time > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Smooth weights
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        z = z - lr * grad
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.03)
            else:
                patience += 1
                lr *= 0.7
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.005
                    beta = max(beta * 0.5, 1e-10)

    return best_a, best_sc
