def _next_pow2(n):

def _fft_convolve(a, b):

"""
Multi-start L-BFGS optimizer with adaptive temperature annealing,
asymmetric initializations, active-set balancing, and coordinate descent polish.
Targets score < 1.5027 for the self-convolution peak minimization problem.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_full(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _fft_convolve_self(a):
    """Compute conv(a,a) using FFT"""
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:out_n]

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoothed_gradient(a, c, T):
    """
    Compute gradient of smoothed objective w.r.t. z (log-space variable).
    f_T(z) = (2n/T) * log(sum_k exp(T*c_k/(2n)))
    g_j = a_j * (2n/T) * (h_j - m)
    where h = convolve(softmax(c/T), reverse(a)) and m = E_p[c]
    """
    n = len(a)
    # Softmax over c with temperature T
    c_scaled = c / T
    c_scaled = c_scaled - c_scaled.max()
    e = np.exp(c_scaled)
    s = e.sum()
    if s < 1e-300:
        # Hard max
        p = np.zeros_like(c)
        idx = c.argmax()
        p[idx] = 1.0
    else:
        p = e / s
    
    m = float(np.dot(p, c))
    # h_j = sum_k p_k * a_{k-j} (with valid indices)
    # This is equivalent to: h = convolve(p, a) evaluated at appropriate position
    # Actually: h_j = sum_k p_k * a_{k-j} = (p * a)[j] where * denotes cross-correlation
    # In terms of convolution: h_j = sum_k p_k * a_{k-j}
    # Let's compute this as: h = ifft(fft(p) * conj(fft(a))) ... no that's correlation
    # 
    # Actually: sum_k p_k * a_{k-j} = (p * a_flipped)[j + offset] 
    # Let a_flipped[k] = a_{n-1-k}, then (p * a_flipped)[j + n - 1] = sum_i p_i * a_flipped[j+n-1-i] = sum_i p_i * a_{n-1-(j+n-1-i)} = sum_i p_i * a_{i-j}
    # Hmm, that's sum_i p_i * a_{i-j}, which is what we want with i=k.
    # 
    # So h_j = sum_k p_k * a_{k-j} = (p * a_flipped)[j + n - 1]
    # where a_flipped = a[::-1]
    
    a_flip = a[::-1]
    # (p * a_flip) has length len(p) + len(a_flip) - 1 = (2n-1) + n - 1 = 3n-2
    # We want positions j + n - 1 for j = 0, ..., n-1
    pa = _fft_convolve(p, a_flip)
    h = pa[n-1 : 2*n-1]  # positions n-1, n, ..., 2n-2 in the convolution
    
    g = a * (2.0 * n / T) * (h - m)

def _active_set_gradient(a, c, top_k=5, decay=0.7):
    """Gradient using weighted active set (top-k peaks)"""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    
    m = float(np.dot(p, c))
    a_flip = a[::-1]
    pa = _fft_convolve(p, a_flip)
    h = pa[n-1 : 2*n-1]
    
    g = a * (2.0 * n) * (h - m)  # T=1 effectively, since weights are already concentrated
    return g

def _lbfgs_objective(z, n, T, c_alpha):
    """
    Objective and gradient for L-BFGS in log-space.
    z: log-space variable (length n)
    Returns (f, grad)
    """
    a = _softmax_stable(z)
    c = _fft_convolve_self(a)
    f = float((2.0 * n / T) * (np.log(np.sum(np.exp(c / T - c.max()))) + c.max() - c.max()))
    # Simplify: f = (2n/T) * (c.max() + log(sum(exp((c-c.max())/T))))
    # = (2n/T) * (c.max() + log(sum(exp((c-c.max())/T))))

    e = np.exp(c_scaled)
    f = float((2.0 * n / T) * (c.max() + np.log(e.sum())))
    g = _smoothed_gradient(a, c, T)
    return f, g

        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute search direction using two-loop recursion
        if not s_list:
            d = -g_cur
        else:
            alpha = np.zeros(len(s_list))
            q = g_cur.copy()
            for i in range(len(s_list) - 1, -1, -1):
                alpha[i] = rho_list[i] * np.dot(s_list[i], q)
                q = q - alpha[i] * y_list[i]
            
            # Estimate H_0
            if len(s_list) > 0:
                beta = (np.dot(s_list[-1], y_list[-1])) / (np.dot(y_list[-1], y_list[-1]))
            else:
                beta = 1.0
                if beta < 1e-10:
                    beta = 1e-10
            
            r = q / beta
            
            for i in range(len(s_list)):
                beta_i = rho_list[i] * np.dot(y_list[i], r + alpha[i] * s_list[i])
                r = r + (alpha[i] - beta_i) * s_list[i]
            
            d = -r
        
        # Scale direction
        d = d / (np.linalg.norm(d) + 1e-15)
        
        # Line search (backtracking)
        step = 1.0
        f_trial = f_cur
        g_trial = g_cur
        improved = False
        for ls in range(30):
            z_trial = z + step * d
            # Clip to avoid numerical issues
            z_trial = np.clip(z_trial, -100, 100)
            f_trial, g_trial = _lbfgs_objective(z_trial, n, T)
            if f_trial < f_cur - 1e-12 * step * np.dot(d, g_cur):
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try with negative direction
            step = 0.5
            for ls in range(10):
                z_trial = z - step * d
                z_trial = np.clip(z_trial, -100, 100)
                f_trial, g_trial = _lbfgs_objective(z_trial, n, T)
                if f_trial < f_cur - 1e-12 * step * (-np.dot(d, g_cur)):
                    improved = True
                    d = -d
                    break
                step *= 0.5
        
        if not improved:
            if it > 100:
                break
            continue
        
        # Update L-BFGS memory
        s_new = step * d
        y_new = g_trial - g_cur
        sy = np.dot(s_new, y_new)
        
        if sy > 1e-12:
            s_list.append(s_new)
            y_list.append(y_new)
            rho_list.append(1.0 / sy)
            if len(s_list) > m_lbfgs:
                s_list.pop(0)
                y_list.pop(0)
                rho_list.pop(0)
        
        z = z_trial
        f_cur = f_trial
        g_cur = g_trial
        
        if f_cur < f_best:
            f_best = f_cur
            z_best = z.copy()
        
        # Periodic recentering
        if it % 500 == 0 and it > 0:
            z_best = z_best - z_best.mean()
            z = z_best
    
    a_best = _softmax_stable(z_best)
    a_best = a_best - a_best.mean()  # Wait, this doesn't make sense for a
    a_best = _softmax_stable(z_best)
    return z_best, a_best, f_best
