def solve_lp(a, M):
    n = len(a)
    # Variables: b_0, ..., b_{n-1}
    # Objective: maximize sum(b) → minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: conv(a, b)[k] <= M for all k = 0, ..., 2n-2
    # conv(a,b)[k] = sum_i a[i]*b[k-i] where 0<=i<n, 0<=k-i<n
    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*b[k-i]
    
    # Build constraint matrix A (2n-1 x n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[k, k-i] = a[i]
    
    b_constraint = np.full(2*n-1, M)
    
    # Bounds: b_i >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_constraint, bounds=bounds, method='highs')

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score with sum(a)=1 assumption: 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs_continuation(a_init, deadline, min_remaining=30.0):
    """Continuation method: solve at progressively smaller beta values."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)

    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:

    base_beta = max(gap * 2.0, max_c * 0.05)

    ratios = [1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01,

        else:

        try:

                _smoobj, z, args=(beta, n),

def _coordinate_descent(a, deadline, n_rounds=200):
    """
    Targeted coordinate descent: identify bottleneck pairs and reduce them.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find which pairs (i, j) contribute most to c[peak_idx]
        # c[peak_idx] = sum_i a[i]*a[peak_idx-i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contribs = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] += a[i] * a[j]
                contribs[j] += a[i] * a[j]

        # Try reducing the top contributors
        n_try = min(n, max(5, n // 100))
        top_idx = np.argsort(contribs)[-n_try:]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-10:
                continue
            old_val = a[i]
            c_old = c.max()

            # Try reducing
            for factor in [0.7, 0.85, 0.95]:
                a[i] = old_val * factor
                # Quick score check (only recompute affected region)
                c_try = _full_convolve(a, a)
                c_try_max = float(c_try.max())
                if c_try_max < c_old - 1e-14:
                    improved = True
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, n_steps=3000, lr_init=0.03):
    """Subgradient descent in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set gradient
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Polyak step size approximation
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.01, best_lr)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = best_lr * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(4 * np.pi * t * t) + 2.0
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 7 * t) + 0.1 * np.cos(2 * np.pi * 19 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=8, amplitude=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.5) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _interpolate_sequence(a_src, n_target):
    """Interpolate a sequence to a new length."""
    n_src = len(a_src)
    if n_src == n_target:
        return a_src.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_src)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = _make_profile(n_target, "uniform")
    return a_new

    # Stage 4: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 50)
        a4, sc4 = _subgradient_polish(current_a, sg_deadline, n_steps=2000, lr_init=0.015)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Coordinate descent
    if time.monotonic() < deadline - min_remaining + 5:
        cd_deadline = min(deadline - min_remaining + 5, time.monotonic() + 60)
        a5, sc5_n = _coordinate_descent(current_a, cd_deadline, n_rounds=100)
        sc5 = _score(a5)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search: continuation L-BFGS, pressure equalization,
    coordinate descent, multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main optimization on parent (most time here) ---
    print(f"\n[Phase 1] Main optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_stage(parent, deadline, min_remaining=800, )
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure equalization on best ---
    print(f"\n[Phase 2] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        pe_deadline = min(deadline - 500, time.monotonic() + 150)
        a_pe, _ = _pressure_equalize(best_seq, pe_deadline, n_iters=400, gamma=0.3)
        sc_pe = _score(a_pe)
        _update(a_pe, sc_pe)

    # --- Phase 3: Fine L-BFGS on best ---
    print(f"\n[Phase 3] Fine L-BFGS on best (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_fb = _run_lbfgs_continuation(best_seq, deadline, min_remaining=400)
        _update(a_fb, _score(a_fb))

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 6144, 10240, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort()  # Start with smaller (faster) for exploration

    n_done = 0
    for n_try in n_candidates:
        if time.monotonic() > deadline - 250:
            break
        remaining_for_this = deadline - 250 - time.monotonic()
        if remaining_for_this < 40:
            break

        # Initialize by interpolating current best
        a_init = _interpolate_sequence(best_seq, n_try)

        # Slight perturbation for diversity
        a_init = _fourier_perturb(a_init, rng, n_modes=3, amplitude=0.01)

        min_rem = max(40, remaining_for_this * 0.4)
        a_opt, sc_opt = _multi_stage(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)
        n_done += 1

    # --- Phase 5: Diverse restarts from best ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        restart_configs = [
            ("chirp", 0.03),
            ("double_cos", 0.03),
            ("gaussian_taper", 0.02),
            ("tapered", 0.04),
        ]

        for i, (prof_kind, amp) in enumerate(restart_configs):
            if time.monotonic() > deadline - 40:
                break

            # Blend best with a new profile
            a_blend = 0.4 * best_seq + 0.6 * _make_profile(n_best, prof_kind)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=amp)

            a_r, sc_r = _multi_stage(a_pert, deadline, min_remaining=40)
            _update(a_r, sc_r)

    # --- Phase 6: Aggressive subgradient on best ---
    print(f"\n[Phase 6] Aggressive subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        sg_deadline = min(deadline - 15, time.monotonic() + 100)
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_deadline, n_steps=5000, lr_init=0.01)
        _update(a_sg, sc_sg)

    # --- Phase 7: Final L-BFGS polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _run_lbfgs_continuation(best_seq, deadline, min_remaining=3)
        _update(a_fin, _score(a_fin))

    # --- Phase 8: Coordinate descent final ---
    print(f"\n[Phase 8] Final coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        cd_deadline = min(deadline - 1, time.monotonic() + 60)
        a_cd, _ = _coordinate_descent(best_seq, cd_deadline, n_rounds=50)
        _update(a_cd, _score(a_cd))

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
