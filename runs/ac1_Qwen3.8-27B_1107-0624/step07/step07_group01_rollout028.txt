    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_self(a):
    return _full_convolve(a, a)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=120.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.0025,
                              0.001, 0.0005, 0.00025, 0.0001,
                              5e-5, 2.5e-5, 1e-5, 5e-6, 2.5e-6,
                              1e-6, 5e-7, 2.5e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.1)
        max_iter = min(15000, max(200, int(remaining / (0.003 * scale * 30))))
        max_iter = max(max_iter, 500)

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5010:
            break

    return current_a


def _gs_frequency_iterate(a_init, n_iter, deadline, gamma_init=0.3, gamma_end=0.05):
    """
    Gerchberg-Saxton style iteration in frequency domain.
    Goal: concentrate power spectrum at DC to flatten the convolution.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Anneal gamma
        frac = it / max(1, n_iter - 1)
        gamma = gamma_init + (gamma_end - gamma_init) * frac

        # Compute DFT (full, complex)
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        # Power spectrum
        P = np.abs(A) ** 2

        # Create target: reduce sidelobes
        # Keep DC, reduce others by factor gamma
        P_new = P.copy()
        # For non-DC bins, reduce power
        P_new[1:] = P[1:] * (1.0 - gamma)
        # Also apply a soft cap to prevent any single sidelobe from dominating
        P_max_sidelobe = P_new[1:].max() if len(P_new) > 1 else 0
        if P_max_sidelobe > 0:
            cap = P_max_sidelobe * 0.9
            P_new[1:] = np.minimum(P_new[1:], cap)

        # Reconstruct with new amplitude, old phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)

        # Inverse DFT
        a_new_complex = np.fft.ifft(A_new)
        a_new = a_new_complex.real[:n]

        # Project non-negative
        a_new = np.maximum(a_new, 0)

        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def _gs_with_target_shape(a_init, n_iter, deadline, shape_type='gaussian', sigma=3.0):
    """
    GS iteration with a specific target power spectrum shape.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    m = np.arange(N)
    m_centered = m - m // 2  # center for symmetric shapes

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Build target shape (normalized so sum matches)
        if shape_type == 'gaussian':
            P_target = np.exp(-m_centered ** 2 / (2 * sigma ** 2))
            P_target[0] = 1.0  # ensure DC is highest
        elif shape_type == 'rect':
            W = int(N / 4)
            P_target = np.zeros(N)
            P_target[max(0, W):N - W] = 1.0  # low-pass (keep center)
            # Actually for our purpose we want to CONCENTRATE at DC
            # So target should be high at DC and low elsewhere
            P_target = np.zeros(N)
            P_target[0] = 1.0
            P_target[1:min(2, N)] = 0.1
            P_target[max(0, N - min(2, N)):N] = 0.1
            # Spread a bit
            for k in range(1, min(10, N // 4)):
                P_target[k] = 0.01
                P_target[N - k] = 0.01
        elif shape_type == 'inverse_linear':
            P_target = 1.0 / (1.0 + np.abs(m_centered).astype(float))
            P_target[0] = 1.0
        else:
            break

        # Normalize target to match total power
        P_target = P_target / P_target.sum() * P.sum()

        # Blend: move current P toward target
        gamma = 0.2 + 0.3 * (it / max(1, n_iter))
        P_new = (1 - gamma) * P + gamma * P_target

        # Ensure non-negative
        P_new = np.maximum(P_new, 0)

        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def _peak_equalization(a_init, deadline, k_active=8, max_steps=20000, lr_init=0.03):
    """
    Iteratively equalize the top-k convolution values by adjusting a in log-space.
    Uses a "water-filling" inspired approach: push down peaks, pull up valleys.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        c_mean = c.mean()

        # Identify active set (top k) and inactive set
        idx_sorted = np.argsort(c)[::-1]
        idx_top = idx_sorted[:k_active]
        idx_bot = idx_sorted[-k_active:]

        # Gradient of each active peak w.r.t. a
        # For peak at lag k: d(conv[k])/da[j] = a[(j+k) mod ... ] + a[(j-k) mod ...]
        # More precisely for linear conv:
        # conv[k] = sum_i a[i]*a[i+k] (0<=i<n, 0<=i+k<n)
        # d(conv[k])/da[j] = a[j+k] (if j+k<n) + a[j-k] (if j-k>=0)
        #                   + a[k-j] (if k-j>=0 and k-j<n) [from the other side]
        # Wait, let me be more careful.
        # conv[k] = sum_{i: max(0,k-(n-1)) <= i <= min(n-1, k)} ... 
        # Actually: conv[k] = sum_i a[i]*a[k-i] where 0<=i<n and 0<=k-i<n
        # So i ranges from max(0, k-(n-1)) to min(n-1, k)
        # d(conv[k])/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j-k != k-j)
        # Actually: d/d a[j] sum_i a[i]*a[k-i] = a[k-j] + a[j-k] (if both valid)
        # But if k-j == j-k (i.e., k=0 or j=k/2), we get 2*a[j] for k=0.
        # Let me just use the FFT-based gradient.

        # Weighted sum of gradients for top peaks
        w_top = np.zeros(2 * n - 1)
        w_top[idx_top] = 1.0 / k_active
        # Weight for bottom peaks (negative - we want to increase these)
        w_bot = np.zeros(2 * n - 1)
        w_bot[idx_bot] = 1.0 / k_active

        # Gradient of top peaks
        a_rev = a[::-1]
        wc_top = _full_convolve(w_top, a_rev)
        h_top = wc_top[n - 1:2 * n - 1]
        W_top = float(np.dot(w_top, c))

        wc_bot = _full_convolve(w_bot, a_rev)
        h_bot = wc_bot[n - 1:2 * n - 1]
        W_bot = float(np.dot(w_bot, c))

        # Combined gradient: reduce top, increase bottom
        grad = 2.0 * n * a * (h_top - W_top - h_bot + W_bot)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step in log-space
        z = z - lr * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.2)
                stall = 0
            else:
                stall += 1
                if stall > 15:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=30000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        z = z - lr * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.03, 0.15)
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.6
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_ushaped(n, c=1.0, p=1.0):
    """U-shaped sequence: larger at edges, smaller in middle."""
    t = np.linspace(0, 1, n)
    edge_dist = np.abs(t - 0.5) * 2.0  # 0 at center, 1 at edges
    a = 1.0 + c * edge_dist ** p
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=8, reserve=60.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6

        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            n_freqs = 2 + h
            for _ in range(n_freqs):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.01 * rng.uniform(0.5, 3.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 4 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.03, 0.03)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # U-shaped perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            c = rng.uniform(0.01, 0.05)
            p = rng.uniform(0.5, 2.0)
            t = np.linspace(0, 1, n)
            edge_dist = np.abs(t - 0.5) * 2.0
            z = z + c * edge_dist ** p
            z = z - z.mean()
            a_pert = _softmax(z)

        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        max_iter = min(3000, max(200, int((hop_deadline - time.monotonic()) / 0.005)))
        if max_iter > 10:
            a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_passes=50):
    """
    Coordinate descent: for each j, optimally adjust a[j] to reduce max(conv).
    Uses the fact that conv[k] is linear in a[j] for fixed k.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for passage in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        for j in range(n):
            if time.monotonic() > deadline:
                break

            # Compute gradient of c_max w.r.t. a[j]
            # c[k] includes a[j]*a[k-j] and a[j-k]*a[j] terms
            # d(c[k])/da[j] = a[k-j] (if valid) + a[j-k] (if valid, j!=k-j)
            # For approximating c_max reduction, use top-k peaks
            idx_top = np.argsort(c)[::-1][:5]

            # Compute how a[j] contributes to each top peak
            grad_j = 0.0
            for k in idx_top:
                # a[j] appears in conv[k] as:
                # a[j] * a[k-j] if 0 <= k-j < n
                # a[j-k] * a[j] if 0 <= j-k < n
                contrib = 0.0
                if 0 <= k - j < n:
                    contrib += a[k - j]
                if 0 <= j - k < n and (j - k) != (k - j):
                    contrib += a[j - k]
                elif 0 <= j - k < n and (j - k) == (k - j):
                    contrib += a[j]  # when k=0, a[j]*a[j]
                grad_j += contrib

            # Also need the "sum" constraint gradient (derivative of S^2)
            # Score = 2n * c_max / S^2
            # d(score)/da[j] = 2n * (dc_max/da[j] * S^2 - c_max * 2*S) / S^4
            #                = 2n * (dc_max/da[j] * S - 2*c_max*a[j]) / S^3
            S = a.sum()
            if S < 1e-15:
                continue

            # Simplified: just reduce a[j] if it increases c_max
            if grad_j > 0:
                alpha = min(0.01 * a[j], 0.01 * grad_j / (c_max + 1e-30))
                a[j] = max(0, a[j] - alpha)

        # Normalize
        S = a.sum()
        if S < 1e-15:
            break
        a = a / S
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _multi_scale_refine(a_init, deadline):
    """
    Refine at a coarser scale, then interpolate back.
    This can escape local minima.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Downsample by factor 2
    n_coarse = n // 2
    if n_coarse < 64:
        return best_a, best_sc

    # Block average
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a_init[2*i:2*i+2].mean()
    a_coarse = np.maximum(a_coarse, 0)
    s = a_coarse.sum()
    if s < 1e-15:
        return best_a, best_sc
    a_coarse = a_coarse / s

    # Optimize coarse
    time_budget_coarse = min(60.0, (deadline - time.monotonic()) * 0.3)
    if time_budget_coarse > 10:
        a_coarse_opt = _adaptive_lbfgs(a_coarse, time.monotonic() + time_budget_coarse,
                                       reserve_time=5.0)
        sc_coarse = _score(a_coarse_opt)

        # Interpolate back
        a_interp = _interpolate_to_n(a_coarse_opt, n)
        sc_interp = _score(a_interp)

        # Quick refine at full scale
        if time.monotonic() < deadline - 10:
            c_i = _full_convolve(a_interp, a_interp)
            beta = c_i.max() * 0.01
            a_refined = _lbfgs_smooth(a_interp, beta, n, 2000)
            sc_refined = _score(a_refined)
            if sc_refined < best_sc:
                best_sc = sc_refined
                best_a = a_refined.copy()
        elif sc_interp < best_sc:
            best_sc = sc_interp
            best_a = a_interp.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining frequency-domain GS, L-BFGS,
    peak equalization, subgradient, and multi-scale methods.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={best_score_val:.10f}")
    else:
        # No parent, use structured inits
        n_init = 4096
        inits = [
            ("chirp_1", _make_chirp(n_init, 1.0, 0.3)),
            ("chirp_2", _make_chirp(n_init, 2.0, 0.25)),
            ("u_1_1", _make_ushaped(n_init, c=1.0, p=1.0)),
            ("u_2_1", _make_ushaped(n_init, c=2.0, p=1.0)),
            ("u_05_2", _make_ushaped(n_init, c=0.5, p=2.0)),
            ("multi_13", _make_multiband(n_init, [1, 3], 0.25)),
            ("taper", _make_tapered(n_init, 0.5)),
            ("rand", _make_random_positive(n_init, rng)),
        ]
        for name, a_i in inits:
            try_update(a_i, name)
        print(f"  Best init: {best_score_val:.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # Set working sequence
    if best_seq is None:
        best_seq = np.ones(4096) / 4096
        best_score_val = _score(best_seq)
        best_n = 4096

    # === Phase 2: Frequency-domain GS refinement ===
    print(f"\n=== Phase 2: Frequency-domain GS (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 900:
        gs_deadline = min(deadline - 900, time.monotonic() + 120)
        a_gs, sc_gs = _gs_frequency_iterate(best_seq, n_iter=500, deadline=gs_deadline,
                                            gamma_init=0.4, gamma_end=0.1)
        try_update(a_gs, "gs_basic")

    if time.monotonic() < deadline - 850:
        gs_deadline = min(deadline - 850, time.monotonic() + 80)
        a_gs2, sc_gs2 = _gs_with_target_shape(best_seq, n_iter=300, deadline=gs_deadline,
                                              shape_type='gaussian', sigma=5.0)
        try_update(a_gs2, "gs_gauss")

    # === Phase 3: Main L-BFGS ===
    print(f"\n=== Phase 3: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    reserve_p3 = 600.0
    if time.monotonic() < deadline - reserve_p3:
        a_p3 = _adaptive_lbfgs(best_seq, deadline, reserve_time=reserve_p3)
        try_update(a_p3, "lbfgs_main")

    # === Phase 4: Peak equalization ===
    print(f"\n=== Phase 4: Peak equalization (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_pe, sc_pe = _peak_equalization(best_seq, deadline - 300, k_active=8,
                                         max_steps=20000, lr_init=0.03)
        try_update(a_pe, "peak_eq_k8")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_pe2, sc_pe2 = _peak_equalization(best_seq, deadline - 200, k_active=12,
                                           max_steps=15000, lr_init=0.02)
        try_update(a_pe2, "peak_eq_k12")

    # === Phase 5: Subgradient polishing ===
    print(f"\n=== Phase 5: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 200, k_active=5,
                                      max_steps=20000, lr_init=0.03)
        try_update(a_sg, "sg_k5")

    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg2, sc_sg2 = _subgrad_polish(best_seq, deadline - 100, k_active=3,
                                        max_steps=15000, lr_init=0.02)
        try_update(a_sg2, "sg_k3")

    # === Phase 6: Second L-BFGS round ===
    print(f"\n=== Phase 6: L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_p6 = _adaptive_lbfgs(best_seq, deadline, reserve_time=80.0)
        try_update(a_p6, "lbfgs_r2")

    # === Phase 7: Basin hopping ===
    print(f"\n=== Phase 7: Basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 80, rng,
                                   n_hops=6, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Multi-scale refinement ===
    print(f"\n=== Phase 8: Multi-scale (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_ms, sc_ms = _multi_scale_refine(best_seq, deadline - 60)
        try_update(a_ms, "multi_scale")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        current_n = len(best_seq)
        lengths_to_try = [current_n * 2, current_n // 2, current_n + 1024]
        lengths_to_try = [l for l in lengths_to_try if 1024 <= l <= 32768]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 20:
                break
            if target_n == current_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            if time.monotonic() < deadline - 15:
                beta = _convolve_self(a_interp).max() * 0.01
                max_iter = min(3000, max(200, int((deadline - 15 - time.monotonic()) / 0.005)))
                a_opt = _lbfgs_smooth(a_interp, beta, target_n, max_iter)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 10: Final aggressive polishing ===
    print(f"\n=== Phase 10: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(6):
            if time.monotonic() > deadline - 5:
                break

            # Very tight L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * (1e-6 * (0.5 ** rnd))
            remaining = deadline - 5 - time.monotonic()
            max_iter = min(5000, max(500, int(remaining / 0.004)))
            if max_iter > 100:
                a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_polish(best_seq, deadline - 5,
                                         k_active=3, max_steps=8000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

            # Peak equalization
            if time.monotonic() < deadline - 5:
                a_p, _ = _peak_equalization(best_seq, deadline - 5, k_active=5,
                                            max_steps=5000, lr_init=0.01)
                try_update(a_p, f"final_pe_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
