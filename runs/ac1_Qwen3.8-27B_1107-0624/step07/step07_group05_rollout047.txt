def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_smoobj(z, beta, n, k):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if k >= len(c):
        top = c
    else:
        part_idx = np.argpartition(c, -k)[-k:]
        top = c[part_idx]
    shifted = (top - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _topk_smojac(z, beta, n, k):

def _active_set_newton(a_init, deadline, max_iters=50, tol=1e-10):
    """
    Active-set Newton method: given active set S, solve KKT linear system.
    KKT: For all i, sum_{m in S} a_{m-i} = lambda (pressure is constant).
    This gives M * a = lambda * 1 where M[i,j] = 1 if (i+j) in S.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Determine active set: top values within relative tolerance
        tol_frac = max(0.01, 0.5 / (n ** 0.4))
        active_mask = c >= (c_max - tol_frac * c_max)
        if active_mask.sum() < 2:
            idx = np.argsort(c)[-max(3, n // 100):]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True

        # Build the linear system M * a = 1 (then normalize)
        # M[i,j] = 1 if (i+j) is in active set
        # M is n x n, but we can use FFT-like structure
        # M = sum_{m in S} T_m where T_m is the "shift" matrix
        # (T_m)_{i,j} = 1 if i+j == m, i.e., j = m-i
        # So (M*a)_i = sum_j M[i,j]*a_j = sum_{m in S} a_{m-i} (where valid)
        # This is exactly the pressure computation!

        # Solve M * a = 1 via iterative method (conjugate gradient on normal equations)
        # Or: since M is structured, use the pressure computation as matrix-vector product
        a_current = a.copy()
        support = a_current > 1e-12
        if support.sum() < 2:
            break

        # Use projected gradient on the linear system
        # Minimize ||M*a - 1||^2 subject to a >= 0, sum(a) = 1
        # Use a few steps of projected gradient
        active_float = active_mask.astype(float)
        for _sub in range(30):
            if time.monotonic() > deadline:
                break
            # Compute gradient of 0.5*||M*a - 1||^2
            # = M^T * (M*a - 1)
            # M*a = pressure (computed via convolution)
            a_rev = a_current[::-1]
            p = _full_convolve(active_float, a_rev)
            p = p[n - 1:2 * n - 1]  # pressure
            # M^T * r where r = M*a - 1: (M^T r)_j = sum_i M[i,j] * r_i = sum_{m in S} r_{m-j}
            r = p - 1.0
            a_rev_r = r[::-1]
            grad = _full_convolve(active_float, a_rev_r)
            grad = grad[n - 1:2 * n - 1]

            # Projected gradient step
            step = 1.0 / active_float.sum()
            a_new = a_current - step * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            else:
                a_new = a_current

            if np.linalg.norm(a_new - a_current) < 1e-12:
                a_current = a_new
                break
            a_current = a_new

        sc = _score_normalized(a_current)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_current.copy()
            a = a_current.copy()
        else:
            # Try a convex combination
            for t in [0.3, 0.5, 0.7]:
                a_mix = (1 - t) * a + t * a_current
                a_mix = np.maximum(a_mix, 0)
                s_mix = a_mix.sum()
                if s_mix > 1e-15:
                    a_mix = a_mix / s_mix
                sc_mix = _score_normalized(a_mix)
                if sc_mix < best_sc - 1e-12:
                    best_sc = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix.copy()
                    break

    return best_a, best_sc


def _run_topk_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with top-k log-sum-exp and aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    # Beta schedule: start large, end small
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    # k schedule: start small (focus on peaks), end larger (refine)
    ks = [3, 5, 10, 20, 40, 80, 120, 200, 300, 500]
    ks = [min(k, 2 * n - 1) for k in ks]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, (beta, k) in enumerate(zip(betas, ks)):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 1024.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _topk_smoobj, z, args=(beta, n, k),
                jac=_topk_smojac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3):
    """
    Improved pressure equalization with:
    - Adaptive gamma (decrease over time for convergence)
    - Better active set (weighted by proximity to peak)
    - Early stopping on convergence
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    prev_sc = best_sc

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive tolerance: start tight, widen as we converge
        progress = it / max(1, n_iters)
        tol_frac = 0.005 + 0.02 * (1 - progress)
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Weighted active set for smoother pressure
        c_active = c[active]
        weights = np.exp(-(c_max - c_active) / (tol_frac * c_max + 1e-15))
        w = np.zeros_like(c)
        w[active] = weights
        w = w / w.sum()

        # Compute weighted pressure
        a_rev = a[::-1]
        p = _full_convolve(w, a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Adaptive gamma: start high (fast progress), end low (fine tuning)
        gamma_cur = gamma * (1.0 - 0.5 * progress)

        # Multiplicative update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma_cur)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if it % 25 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                prev_sc = sc
            elif abs(sc - prev_sc) < 1e-10:

            prev_sc = sc

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _mirror_descent(a_init, deadline, n_steps=5000, eta_init=0.1):

def _optimal_step_coord_descent(a, deadline, n_rounds=200):

        n_try = min(n, max(20, n // 30))

        for i in idx:

            a_pad = np.zeros(n_coarse * factor)
            a_pad[:n] = current_a
            a_coarse = a_pad.reshape(n_coarse, factor).mean(axis=1)

                                               n_iters=100)

    if time.monotonic() < deadline - min_remaining - 5:

def _population_search(best_a, deadline, n_pop=5, children_per_parent=3):

            break

            if time.monotonic() > deadline - 20:
                break

                z = np.log(np.maximum(population[parent_idx], 1e-30))
                z += 0.05 * rng.standard_normal(n)
                child = _softmax(z)
            elif strategy == 1:
                # Fourier perturbation
                z = np.log(np.maximum(population[parent_idx], 1e-30))
                z_hat = np.fft.rfft(z)
                n_pert = rng.integers(3, 10)
                for _ in range(n_pert):
                    freq = rng.integers(1, max(2, n // 4))
                    if freq < len(z_hat):
                        delta = 0.03 * rng.uniform(0.5, 2.0) * rng.standard_normal()
                        phase = rng.uniform(0, 2 * np.pi)
                        z_hat[freq] += delta * np.exp(1j * phase)
                        if freq < len(z_hat) - 1:
                            z_hat[-freq] = np.conj(z_hat[freq])
                z_new = np.fft.irfft(z_hat, n)
                z_new = z_new - z_new.mean()
                child = _softmax(z_new)
            elif strategy == 2:
                # Block shuffle: swap two random blocks
                child = population[parent_idx].copy()
                b1 = rng.integers(0, n // 2)
                b2 = rng.integers(n // 2, n - 10)
                length = rng.integers(10, min(100, n // 4))
                b1 = min(b1, n - length - 1)
                b2 = min(b2, n - length - 1)
                if b1 + length <= b2:
                    child[b1:b1 + length], child[b2:b2 + length] = \
                        child[b2:b2 + length].copy(), child[b1:b1 + length].copy()
                    child = child / child.sum()
            else:
                # Multiplicative perturbation
                child = population[parent_idx] * np.exp(0.02 * rng.standard_normal(n))
                child = np.maximum(child, 0)
                child = child / child.sum()

            # Brief optimization
            time_per_child = max(5, (deadline - time.monotonic() - 20) / (n_pop * children_per_parent))
            deadline_child = time.monotonic() + time_per_child

            a_opt = _run_topk_lbfgs(child, deadline_child, min_time_remaining=2)
            sc_opt = _score(a_opt)

            # Add to population if good
            if sc_opt < scores[0]:
                population[0] = a_opt
                scores[0] = sc_opt
                print(f"    Population gen {gen} child {child_i}: new best {sc_opt:.10f}")
            elif sc_opt < max(scores) if len(scores) > 1 else True:
                population.append(a_opt)
                scores.append(sc_opt)
                # Keep only top n_pop
                if len(population) > n_pop:
                    order = np.argsort(scores)
                    population = [population[i] for i in order[:n_pop]]
                    scores = [scores[i] for i in order[:n_pop]]

    return population[0], scores[0]


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    """Resample sequence to new length using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining top-k L-BFGS, active-set Newton,
    pressure equalization, multi-scale optimization, and population search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  ** BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
                _update(parent, par_sc)
    except Exception:
        pass

    if parent is None:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}, score={_score(parent):.10f}")
        _update(parent)

    # === Phase 1: Multi-scale optimization on parent ===
    print(f"\n[Phase 1] Multi-scale optimization (t={time.monotonic()-t_start:.1f}s)")
    a1, sc1 = _multi_scale_optimize(parent, deadline, min_remaining=600)
    _update(a1, sc1)

    # === Phase 2: Active-set Newton ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2, sc2_norm = _active_set_newton(best_seq.copy(),
                                          min(deadline - 500, time.monotonic() + 120),
                                          max_iters=30)
        sc2 = _score(a2)
        _update(a2, sc2)
        if sc2 < best_score:
            # Follow up with L-BFGS
            a2b = _run_topk_lbfgs(a2, deadline, min_time_remaining=500)
            _update(a2b, _score(a2b))

    # === Phase 3: Extended pressure equalization ===
    print(f"\n[Phase 3] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3, sc3_norm = _pressure_equalize_v2(best_seq.copy(),
                                             min(deadline - 400, time.monotonic() + 150),
                                             n_iters=500, gamma=0.25)
        sc3 = _score(a3)
        _update(a3, sc3)

        if sc3 < best_score:
            a3b = _run_topk_lbfgs(a3, deadline, min_time_remaining=400)
            _update(a3b, _score(a3b))

    # === Phase 4: Mirror descent ===
    print(f"\n[Phase 4] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a4, sc4 = _mirror_descent(best_seq.copy(),
                                  min(deadline - 350, time.monotonic() + 100),
                                  n_steps=3000, eta_init=0.08)
        _update(a4, sc4)

    # === Phase 5: Coordinate descent with optimal step ===
    print(f"\n[Phase 5] Optimal-step coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a5, sc5_norm = _optimal_step_coord_descent(best_seq.copy(),
                                                   min(deadline - 200, time.monotonic() + 120),
                                                   n_rounds=100)
        sc5 = _score(a5)
        _update(a5, sc5)

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 5120, 6144, 7168, 8192, 9216, 10240, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)  # Try larger first

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - 150 - time.monotonic()
        if remaining < 20:
            break

        # Initialize by resampling parent
        a_init = _resample(best_seq, n_try)

        # Brief optimization
        min_rem = max(10, remaining * 0.3)
        a_opt, sc_opt = _multi_scale_optimize(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # === Phase 7: Population-based search ===
    print(f"\n[Phase 7] Population search (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        a_pop, sc_pop = _population_search(best_seq.copy(), deadline - 80, n_pop=4, children_per_parent=2)
        _update(a_pop, sc_pop)

    # === Phase 8: Final intensive polish on the best ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Top-k L-BFGS with very fine settings
        a_fin = _run_topk_lbfgs(best_seq.copy(), deadline, min_time_remaining=3)
        _update(a_fin, _score(a_fin))

        if time.monotonic() < deadline - 3:
            a_fin2, _ = _pressure_equalize_v2(best_seq.copy(),
                                              min(deadline - 2, time.monotonic() + 20),
                                              n_iters=200, gamma=0.1)
            _update(a_fin2, _score(a_fin2))

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
