def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        return c_max
    return c_max + beta * np.log(s)

        if sc < best_sc:

            best = cand
    return best


def _newton_active(a_init, deadline, max_iter=80):
    """Newton method on active-set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() + 3 > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        tol = max(0.001 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Gradient of sum of active c_k
        grad_a = _conv_grad_a(a, active.astype(np.float64) / active.sum(), n)
        grad_t = grad_a - np.mean(grad_a)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break
        
        # Try gradient descent step (simplified Newton)
        alpha = 0.1
        improved = False
        for _ in range(20):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            f_old = float(c[active].sum())
            if f_try < f_old:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            break
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale search: L-BFGS smooth-max with beta continuation at multiple
    sequence lengths (8192, 16384, 32768), FISTA, subgradient polish, and
    Newton active-set refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        sc_par = _score(parent)
        best_seq = parent.copy()
        best_score = sc_par
        print(f"Parent: n={len(parent)}, score={sc_par:.10f}")
    else:
        n_init = 8192
        best_seq = _structured_init(n_init, seed)
        best_score = _score(best_seq)
        print(f"Init: n={n_init}, score={best_score:.10f}")
    
    # Phase 1: L-BFGS at parent length
    print(f"\n[Phase 1] L-BFGS n={len(best_seq)} t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 500:
        a1 = _lbfgs_beta_cont(best_seq, deadline, min_remaining=400)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1
            print(f"  Improved: {sc1:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (stayed at {best_score:.10f})")
    
    # Phase 2: Scale up to 16384
    print(f"\n[Phase 2] L-BFGS n=16384 t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 250:
        n2 = 16384
        a2 = _resample(best_seq, n2)
        a2 = _lbfgs_beta_cont(a2, deadline, min_remaining=150)
        sc2 = _score(a2)
        print(f"  n=16384 score={sc2:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2
            print(f"  NEW BEST: {sc2:.10f}")
    
    # Phase 3: Scale up to 32768
    print(f"\n[Phase 3] L-BFGS n=32768 t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 120:
        n3 = 32768
        a3 = _resample(best_seq, n3)
        a3 = _lbfgs_beta_cont(a3, deadline, min_remaining=60)
        sc3 = _score(a3)
        print(f"  n=32768 score={sc3:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3
            print(f"  NEW BEST: {sc3:.10f}")
    
    # Phase 4: Newton active-set polish on best
    print(f"\n[Phase 4] Newton polish t={time.monotonic()-t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a4 = _newton_active(best_seq, deadline, max_iter=60)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4
            print(f"  Improved: {sc4:.10f}")
    
    # Phase 5: FISTA on very small beta
    print(f"\n[Phase 5] FISTA t={time.monotonic()-t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        beta_f = float(c.max()) * 0.005
        a5 = _fista_smooth(best_seq, beta_f, n, deadline, min_remaining=15)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5
            print(f"  Improved: {sc5:.10f}")
    
    # Phase 6: Subgradient polish
    print(f"\n[Phase 6] Subgradient polish t={time.monotonic()-t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a6 = _subgrad_polish(best_seq, deadline, min_remaining=5)
        sc6 = _score(a6)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6
            print(f"  Improved: {sc6:.10f}")
    
    # Phase 7: Try one more L-BFGS from the best with fresh z
    print(f"\n[Phase 7] Final L-BFGS t={time.monotonic()-t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a7 = _lbfgs_beta_cont(best_seq, deadline, min_remaining=5)
        sc7 = _score(a7)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7
            print(f"  Improved: {sc7:.10f}")
    
    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256, dtype=np.float64) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
