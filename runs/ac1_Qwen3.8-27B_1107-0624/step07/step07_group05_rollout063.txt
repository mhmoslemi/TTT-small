import numpy as np
import time
from scipy.optimize import minimize

# Helper functions
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _two_level_init(n, ratio=0.25):
    """Initialize with a 2-level staircase."""
    h1 = ratio
    h2 = 1.0
    a = np.concatenate([h1 * np.ones(n//2), h2 * np.ones(n - n//2)])
    return a / a.sum()

def _multi_level_init(n, n_levels=4, base_ratio=0.1):
    """Initialize with a multi-level staircase."""
    heights = np.linspace(base_ratio, 1.0, n_levels)
    sizes = np.full(n_levels, n // n_levels)
    sizes[-1] += n - sizes.sum()
    a = np.concatenate([h * np.ones(s) for h, s in zip(heights, sizes)])
    return a / a.sum()

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: optimal sequences are anti-symmetric (increasing L-to-R) staircases
that create a flat top in the self-convolution. Uses staircase initialization,
adaptive log-sum-exp L-BFGS with aggressive beta continuation, and KKT-based
active set refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_gradient(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _staircase_init(n, n_levels=2, spread=1.0):
    if n_levels == 1:
        return np.ones(n) / n
    heights = np.linspace(1.0 - spread, 1.0, n_levels)
    heights = np.maximum(heights, 0.01)
    sizes = np.full(n_levels, n // n_levels)
    sizes[-1] += n - sizes.sum()
    a = np.concatenate([h * np.ones(s) for h, s in zip(heights, sizes)])
    return a / a.sum()


def _ramp_init(n):
    a = np.linspace(0.1, 1.0, n)
    return a / a.sum()


def _smooth_ramp_init(n, sharpness=3.0):
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-sharpness * (t - 0.5)))
    a = a - a.min() + 0.05
    return a / a.sum()


def _sigmoid_init(n, center=0.5, width=0.1):
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-(t - center) / width))
    a = a - a.min() + 0.1
    return a / a.sum()


def _fourier_init(n, rng, n_modes=3):
    z = rng.standard_normal(n) * 0.1
    z = z + np.linspace(0.5, -0.5, n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 10))
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    return _softmax(z)


def _run_lbfgs_aggressive(a_init, deadline, min_remaining=10.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 5:
        gap = c_sorted[0] - c_sorted[5]
    else:
        gap = c_sorted[0] * 0.1
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.025,
                                  0.012, 0.006, 0.003, 0.0015, 0.0007,
                                  0.0003, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score_norm(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 2048.0)
        est_time = 0.004 * scale
        max_iter = min(10000, max(30, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score_norm(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _kkt_pressure_iterate(a_init, deadline, n_iters=500, gamma=0.5):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-max(3, n // 100):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 50 == 0 or it == n_iters - 1:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_gradient(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_refine(a, deadline, n_rounds=30):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(20, n // 30))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = rng.permutation(top_idx[:min(len(top_idx), n_try // 3)])

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            for factor in [0.7, 0.85, 1.15, 1.3]:
                a[i] = old_val * factor
                c_try = _full_convolve(a, a).max()
                if c_try < c_max - 1e-14:
                    c_max = c_try
                    improved = True
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc

        if sc4 < sc:
            a, sc = a4, sc4

    return a, sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()

            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={elapsed:.1f}s")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
