"""
Multi-method optimizer combining spectral shaping (Gerchberg-Saxton),
L-BFGS with log-sum-exp, iterative peak shaving, and coarse-to-fine
refinement to minimize the peak auto-convolution ratio.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gs_spectral(a_init, deadline, n_iters=200, alpha=0.5, min_time_remaining=30.0):
    """Gerchberg-Saxton spectral shaping for flat auto-convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Forward transform
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        target = P[0] ** 2  # DC power (sum a)^2 = 1
        # Average non-DC power
        if N // 2 > 1:
            avg_non_dc = P[1:].sum() / (len(P) - 1)
        else:
            avg_non_dc = target
        
        # Gradually flatten
        # P_new[k] = P[k] * (avg / P[k])^alpha for k > 0
        P_non_dc = P[1:].copy()
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = np.maximum(P_non_dc, 1e-30) ** (-alpha)
            P_new_non_dc = P_non_dc * (avg_non_dc ** alpha) * ratio
        # Rescale to maintain total power
        total_power = P.sum()
        new_total = target + P_new_non_dc.sum()
        if new_total > 1e-30:
            P_new_non_dc = P_new_non_dc * (total_power - target) / P_new_non_dc.sum()
        
        P_new = np.concatenate([[target], P_new_non_dc])
        
        # Reconstruct A with new magnitudes, same phases
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

"""
Hybrid optimizer combining spectral shaping (Gerchberg-Saxton), adaptive L-BFGS
with log-sum-exp smoothing, direct peak shaving, and multi-scale exploration.
Key innovations: frequency-domain spectral flattening, Buterbaugh-style peak
reduction, and aggressive multi-n search to beat 1.505706.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_f(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_f(z, beta, n):
    a = _softmax_f(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_f(z, beta, n):
    a = _softmax_f(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gs_spectral(a_init, deadline, n_iters=500, min_time_remaining=30.0):
    """Gerchberg-Saxton spectral shaping: flatten |FFT(a)|^2 to reduce convolution peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    alpha = 0.3
    alpha_decay = 0.995
    patience = 0
    max_patience = 80

    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        dc_power = P[0]
        total_power = P.sum()
        non_dc_power = total_power - dc_power
        n_freqs = len(P) - 1
        if n_freqs <= 0:
            break
        avg_non_dc = non_dc_power / n_freqs

        P_non_dc = P[1:].copy()
        with np.errstate(divide='ignore', invalid='ignore'):
            P_non_dc_safe = np.maximum(P_non_dc, 1e-30)
            target_mag = np.sqrt(avg_non_dc)
            cur_mag = np.sqrt(P_non_dc_safe)
            new_mag = cur_mag * (target_mag / cur_mag) ** alpha
            P_new_non_dc = new_mag ** 2

        new_total_non_dc = P_new_non_dc.sum()
        if new_total_non_dc > 1e-30:
            P_new_non_dc = P_new_non_dc * (non_dc_power / new_total_non_dc)

        new_mags = np.zeros(len(P))
        new_mags[0] = np.sqrt(dc_power)
        new_mags[1:] = np.sqrt(np.maximum(P_new_non_dc, 0))

        A_new = new_mags * np.exp(1j * phases)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        a = a_new
        alpha *= alpha_decay
        if patience > max_patience:
            break

    return best_a


def _peak_shave(a_init, deadline, n_iters=200, min_time_remaining=30.0):
    """Buterbaugh-style direct peak shaving: iteratively reduce elements contributing to peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.02

    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Compute gradient at the peak: d c[peak] / d a[i] = 2 * sum_j a[j] * [i+j == peak or j+i == peak]
        # Actually: c[k] = sum_i a[i] * a[k-i] (for valid i, k-i)
        # d c[k] / d a[m] = a[k-m] + a[m-k] (for valid indices)
        # For k = peak: d c[peak] / d a[m] = a[peak-m] (if 0<=peak-m<n) + a[m-peak] (if 0<=m-peak<n)
        grad = np.zeros(n)
        for m in range(n):
            j1 = peak - m
            j2 = m - peak
            if 0 <= j1 < n:
                grad[m] += a[j1]
            if 0 <= j2 < n and j2 != j1:
                grad[m] += a[j2]
        grad *= 2.0

        # Reduce elements with largest gradient (they contribute most to the peak)
        # But don't reduce below zero
        reduction = step_size * grad
        a_new = a - reduction
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            step_size *= 0.5
            continue
        a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.01, 0.05)
        else:
            a = a_new
            step_size *= 0.7
            if step_size < 1e-6:
                break

    return best_a


def _lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta based on current gap."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)

    base_beta = max(gap, max_c * 0.005)

    # Adaptive schedule: start with moderate beta, decrease adaptively
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                  0.005, 0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        est_time_per_iter = 0.002 * max(1, n / 4096.0)

        def fun_z(z_local):

        try:

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":

        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_approx":
        # Approximation to the optimal continuous profile
        t = np.linspace(0, 1, n)

        # the solution of a variational problem. A good approximation:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "airy_like":
        t = np.linspace(0, 1, n)
        x = (t - 0.5) * 8.0
        a = np.abs(np.sinc(x / np.pi)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)

def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.05 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax_f(z)


def _upscale(a_coarse, n_target):
    x_old = np.linspace(0, 1, len(a_coarse))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a_coarse)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a_target / s

            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
