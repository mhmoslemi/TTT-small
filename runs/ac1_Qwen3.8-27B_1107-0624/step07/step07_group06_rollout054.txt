def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

    return e / s

def _newton_active(a_init, deadline, max_iter=200, tol_frac=0.002):
    """Newton's method on the active set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)

    best_sc = _score(a)

    for it in range(max_iter):

        c_max = float(c.max())

        # Gradient of the active set objective
        grad = _hessian_matvec_general(a, active, n)

        # Project to tangent space (sum = 0)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:

            return _hessian_matvec_general(v, active, n) + mu * v

        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        a_new = a.copy()
        alpha = 1.0
        for ls in range(60):
            a_try = a + alpha * dx

            if s_try < 1e-15:

def _frank_wolfe(a_init, deadline, max_iter=50000, tol=1e-12):
    """
    Frank-Wolfe (conditional gradient) method for minimizing
    max_k (a*a)_k subject to a >= 0, sum(a) = 1.
    Uses away-steps for acceleration.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Track the support for away-steps
    support = set(np.where(a > 1e-10)[0].tolist())
    t_iter = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_iter += 1

        c = _full_convolve(a, a)
        s_max = int(np.argmax(c))
        c_max = float(c[s_max])

        # Subgradient: grad[i] = 2*a[s_max - i] if 0 <= s_max-i < n
        grad = np.zeros(n)
        valid = np.arange(max(0, s_max - n + 1), min(n, s_max + 1))
        indices = s_max - valid
        grad[valid] = 2.0 * a[indices]

        # FW step: find j* = argmin grad[i]
        j_star = int(np.argmin(grad))
        lin_gap = float(np.dot(grad, a) - grad[j_star])

        if lin_gap < tol:
            break

        # Check for away-step
        if len(support) > 1:
            # Find j_away in support that maximizes grad
            support_arr = np.array(list(support))
            grad_support = grad[support_arr]
            j_away_idx = int(np.argmax(grad_support))
            j_away = int(support_arr[j_away_idx])
            if grad[j_away] > grad[j_star] + 1e-10:
                # Try away step
                a_away = a - a * np.zeros(n)
                a_away[j_away] = 0.0
                s_away = a_away.sum()
                if s_away > 1e-10:
                    a_away = a_away / s_away
                    c_away = _full_convolve(a_away, a_away)
                    if c_away.max() < c_max:
                        a = a_away
                        support.discard(j_away)
                        continue

        # Standard FW step with backtracking
        e_j = np.zeros(n)
        e_j[j_star] = 1.0
        gamma = 1.0
        f_old = c_max
        for ls in range(40):
            a_try = (1 - gamma) * a + gamma * e_j
            c_try_max = 0.0
            # Quick check: just evaluate the max
            # For efficiency, only check if it's better
            a_check = a + gamma * (e_j - a)
            # Compute convolution only for the max position check
            # Actually let's just do the full thing for correctness
            c_try = _full_convolve(a_check, a_check)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-15 * gamma * lin_gap:
                a = a_check
                support.add(j_star)
                break
            gamma *= 0.5
        else:
            if gamma < 1e-10:
                break
            a = a + gamma * (e_j - a)
            support.add(j_star)

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Periodic check
        if it % 5000 == 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on the smoothed max objective with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[19]
    else:
        gap = c_sorted[0] - c_sorted[-1] if len(c_sorted) > 1 else c_sorted[0]
    base_beta = max(gap * 0.1, max_c * 0.005)

    # Aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.004, 0.002, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration
        scale = max(1, n / 4096.0)
        est_time_per_iter = 0.002 * scale
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement, skip to next beta
                pass
        except Exception:
            pass

    return current_a


def _subgrad_descent(a_init, deadline, n_max_steps=10000, lr_init=0.01):
    """Subgradient descent on the max (active set gradient) with Adam-like momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2 = 0.9, 0.999
    t = 0
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_gradient(a, c, n, tol_frac=0.001)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + 1e-8) * m_hat

        z = z - step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    lr *= 0.8
                    if lr < 0.0005:
                        z = np.log(np.maximum(best_a, 1e-30))
                        z = z - z.mean()
                        m = np.zeros(n)
                        v = np.zeros(n)
                        patience = 0
                        lr = lr_init * 0.5
                        t = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.05) + np.exp(-((t - 0.7) ** 2) / 0.05)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where((t > 0.2) & (t < 0.8), 1.0, 0.3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_swap(a, rng, n_swaps=10):
    """Swap random blocks of the log-amplitude."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    for _ in range(n_swaps):
        block_size = rng.integers(max(1, n // 200), max(2, n // 50))
        i = rng.integers(0, max(1, n - 2 * block_size))
        j = rng.integers(0, max(1, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    z = z - z.mean()
    return _softmax(z)


def _variance_reduction(a_init, deadline, n_steps=500):
    """Minimize variance of the convolution as a proxy."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.002

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        mean_c = np.mean(c)
        centered = c - mean_c
        dcd_a = _full_convolve(a[::-1], centered)
        g = 2.0 * dcd_a[n - 1:2 * n - 1]
        g = g - np.mean(g)

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        v_new = float(np.mean((c - mean_c) ** 2))
        c_new = _full_convolve(a_new, a_new)
        v_new = float(np.mean((c_new - np.mean(c_new)) ** 2))
        if v_new < float(np.mean((c - mean_c) ** 2)):
            a = a_new
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.7
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining smoothed L-BFGS, active-set Newton,
    Frank-Wolfe polishing, variance reduction, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Variance reduction warm-start ===
    print(f"\n[Phase 1] Variance warm-start (t={time.monotonic() - t_start:.1f}s)")
    remaining_budget = deadline - time.monotonic()
    if remaining_budget > 600:
        v_deadline = min(deadline - 500, time.monotonic() + 100)
        a_v, sc_v = _variance_reduction(best_seq, v_deadline, n_steps=300)
        update_best(a_v, sc_v, "variance_ws")

    # === Phase 2: Main L-BFGS on smoothed max ===
    print(f"\n[Phase 2] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_main = _run_lbfgs_smoothed(best_seq, deadline, min_time_remaining=350)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Active-set Newton ===
    print(f"\n[Phase 3] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_deadline = min(deadline - 250, time.monotonic() + 200)
        a_newton, sc_newton = _newton_active(best_seq, n_deadline, max_iter=200)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 4: Frank-Wolfe polishing ===
    print(f"\n[Phase 4] Frank-Wolfe polishing (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        fw_deadline = min(deadline - 150, time.monotonic() + 150)
        a_fw, sc_fw = _frank_wolfe(best_seq, fw_deadline, max_iter=30000)
        update_best(a_fw, sc_fw, "frank_wolfe")

    # === Phase 5: Subgradient with Adam ===
    print(f"\n[Phase 5] Subgradient Adam (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        sg_deadline = min(deadline - 100, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_descent(best_seq, sg_deadline, n_max_steps=8000, lr_init=0.008)
        update_best(a_sg, sc_sg, "subgrad_adam")

    # === Phase 6: Newton + FW combination ===
    print(f"\n[Phase 6] Newton + FW (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        # Newton first
        n_dl = min(deadline - 60, time.monotonic() + 80)
        a_n2, sc_n2 = _newton_active(best_seq, n_dl, max_iter=100, tol_frac=0.001)
        update_best(a_n2, sc_n2, "newton2")
        # Then FW on the result
        if time.monotonic() < deadline - 30:
            a_fw2, sc_fw2 = _frank_wolfe(a_n2, deadline - 25, max_iter=20000)
            update_best(a_fw2, sc_fw2, "fw_after_newton")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = set()
    n_candidates.add(n_parent)
    n_candidates.add(n_parent // 2)
    n_candidates.add(n_parent * 2)
    n_candidates.add(4096)
    n_candidates.add(8192)
    n_candidates.add(16384)
    # Filter to reasonable range
    n_candidates = sorted([n for n in n_candidates if 1024 <= n <= 32768])

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if len(best_seq) == n_try:
            a_init = best_seq.copy()
        else:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

        # Quick L-BFGS
        a_opt = _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick FW polish
        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 30:
            a_fw3, sc_fw3 = _frank_wolfe(a_opt, deadline - 25, max_iter=10000)
            update_best(a_fw3, sc_fw3, f"n_{n_try}_fw")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb_log(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=6)
            elif i == 2:
                p_start = _perturb_swap(best_seq, rng, n_swaps=5)
            elif i == 3:
                p_start = _perturb_log(best_seq, 0.01, rng)
                p_start = _perturb_fourier(p_start, rng, n_modes=3)
            else:
                p_start = _perturb_swap(best_seq, rng, n_swaps=15)

            a_r = _run_lbfgs_smoothed(p_start, deadline, min_time_remaining=15)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _frank_wolfe(a_r, deadline - 10, max_iter=10000)
                update_best(a_r2, sc_r2, f"restart_{i}_fw")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very low beta L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.000005
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            max_iter = min(5000, max(200, int(remaining / 0.005)))
            try:
                result = minimize(
                    lambda zl: _smoobj(zl, beta_final, n), z,
                    jac=lambda zl: _smograd(zl, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 100}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra_lbfgs")
            except Exception:
                pass

        # Final FW
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _frank_wolfe(best_seq, max(deadline - 1, time.monotonic()), max_iter=20000)
            update_best(a_fin2, sc_fin2, "final_fw")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
                print(f"  (evaluate_sequence gives {sc_eval:.10f})")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
