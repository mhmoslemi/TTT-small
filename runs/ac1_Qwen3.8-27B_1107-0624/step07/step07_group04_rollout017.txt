    while k < n:

2. p-norm smoothed L-BFGS with aggressive continuation.
3. Adaptive active set with hysteresis.
4. Multi-n exploration with resampling.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[j,m] = 2*W[j+m], W has length >= 2n-1."""
    V = v[::-1]
    # We need W[j+m] for j,m in [0,n-1], so we need W[0:2n-1]
    W_use = W[:2*n-1] if len(W) >= 2*n-1 else np.pad(W, (0, 2*n-1-len(W)))
    out_len = n + len(W_use) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W_use, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n-1:2*n-1]


def _cg_hankel(W, n, rhs, max_iter=300, reg=0.0):
    """Solve (H + reg*I) x = rhs where H[j,m] = 2*W[j+m]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        if reg > 0:
            Hp += reg * p
        pHp = float(p @ Hp)
        if pHp < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 2.0:
            # Revert
            x = x - alpha * p
            r = r + alpha * Hp
            break
        rho = rs_new / rs if rs > 0 else 0.0
        p = r + rho * p
        rs = rs_new
        if rs < 1e-20 * rs0:
            break
    return x


def _solve_qp(W, n, max_cg_iter=500, reg=0.0):
    """Solve: min a^T H a s.t. sum(a)=1, a>=0 where H[j,m]=2*W[j+m].
    Returns a >= 0, sum(a)=1.
    """
    ones = np.ones(n)
    x = _cg_hankel(W, n, ones, max_iter=max_cg_iter, reg=reg)
    # Clip and normalize
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s


def _fejer_weights(active_idx, total_len):
    """Create Fejér (triangular) weights on the active set."""
    W = np.zeros(total_len)
    if len(active_idx) == 0:
        W[0] = 1.0
        return W
    # Assume active_idx is contiguous
    k0 = active_idx[0]
    k1 = active_idx[-1]
    m = k1 - k0 + 1
    if m == 1:
        W[k0] = 1.0
        return W
    # Triangular window
    center = m / 2.0
    for i in range(m):
        W[k0 + i] = 1.0 - abs(i - (center - 0.5)) / (m / 2.0)
    # Ensure non-negative
    W = np.maximum(W, 0.0)
    s = W.sum()
    if s < 1e-30:
        W[k0] = 1.0
    else:
        W /= s
    return W


def _uniform_weights(active_idx, total_len):
    """Create uniform weights on the active set."""
    W = np.zeros(total_len)
    if len(active_idx) == 0:
        W[0] = 1.0
        return W
    W[active_idx] = 1.0 / len(active_idx)
    return W


def _iterative_qp(a_init, deadline, max_rounds=500):
    """Main iterative active set QP optimizer."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    total_len = 2 * n - 1
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance: start tight, loosen if needed
        # Use percentile-based approach
        c_sorted = np.sort(c)[::-1]
        # Active set: top fraction
        frac = 0.3  # top 30% of coefficients
        n_active = max(5, int(n * frac))
        threshold = c_sorted[min(n_active, total_len) - 1]
        active = c >= threshold
        
        # Also ensure we include the absolute max
        active[int(np.argmax(c))] = True
        
        # Create Fejér weights
        active_idx = np.where(active)[0]
        # Use only contiguous portion around the peak
        peak = int(np.argmax(c))
        # Find contiguous block around peak
        left = peak
        while left > 0 and active[left-1]:
            left -= 1
        right = peak
        while right < total_len-1 and active[right+1]:
            right += 1
        # Extend to the full active set if it's all contiguous
        contig_idx = np.arange(left, right+1)
        
        W = _fejer_weights(contig_idx, total_len)
        
        # Solve QP
        a_candidate = _solve_qp(W, n, max_cg_iter=200, reg=0.0)
        
        # Line search
        best_t = 0.0
        best_c_max_try = c_max
        for t in [1.0, 0.5, 0.3, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * a_candidate
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < best_c_max_try:
                best_c_max_try = c_max_try
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * a_candidate
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            print(f"  QP round {rnd}: score={sc:.10f}, c_max={c_max:.6f}, t={best_t:.2f}")
        elif rnd % 50 == 0:
            print(f"  QP round {rnd}: score={sc:.10f}, c_max={c_max:.6f} (no improvement)")
    
    return best_a, best_sc


def _pnorm_obj(z, p, n):
    """p-norm smoothed objective: (sum c_k^p)^(1/p) * 2n."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0)  # numerical safety
    sp = np.sum(c**p)
    if sp < 1e-300:
        return 0.0
    return 2.0 * n * sp**(1.0/p)


def _pnorm_grad(z, p, n):
    """Gradient of p-norm objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0)
    cp = c**(p-1)
    sp = np.sum(c**p)
    if sp < 1e-300:
        return np.zeros_like(z)
    coef = 2.0 * n * sp**(1.0/p - 1.0)
    # Gradient of sum c_k^{p-1} * c_k with respect to a
    # = sum_k (p-1) c_k^{p-1} * grad c_k  ... wait
    # d/dz (sum c_k^p)^{1/p} = (1/p)(sum c_k^p)^{1/p - 1} * sum_k p c_k^{p-1} dc_k/dz
    # = (sum c_k^p)^{1/p - 1} * sum_k c_k^{p-1} dc_k/dz
    # dc_k/dz = dc_k/da * da/dz
    # dc_k/da[j] = 2 a[k-j] (with a[m]=0 for invalid m)
    # da/dz = a * (e_j - a^T e_j)  [softmax Jacobian]
    # So dc_k/dz[j] = 2 a[k-j] * a[j] * (1 - a[k-j]) ... 
    
    # Actually, let's compute it differently.
    # z -> a = softmax(z)
    # c_k = sum_i a[i] a[k-i]
    # dc_k/dz[j] = sum_i (da[i]/dz[j]) a[k-i] + sum_i a[i] (da[k-i]/dz[j])
    # da[i]/dz[j] = a[i](δ_{ij} - a[j])
    # dc_k/dz[j] = a[j](1-a[j]) a[k-j] + a[j](1-a[j]) a[k-j] = 2 a[j] a[k-j] (1-a[j])
    # Hmm wait, that's not right either. Let me be more careful.
    
    # Actually for the gradient of f(z) = (sum_k c_k(z)^p)^{1/p}:
    # df/dz[j] = (1/p)(sum c_k^p)^{1/p-1} * sum_k p c_k^{p-1} * dc_k/dz[j]
    # dc_k/dz[j] = sum_i [da_i/dz_j * a_{k-i} + a_i * da_{k-i}/dz_j]
    # da_i/dz_j = a_i(δ_ij - a_j)
    # dc_k/dz[j] = a_j(1-a_j) a_{k-j} + sum_i a_i * a_{k-i}(δ_{k-i,j} - a_j)
    # = a_j(1-a_j) a_{k-j} + a_{k-j}(1-a_j) a_{k-j} ... no
    
    # Let me just compute it numerically or use a different approach.
    # Actually, the simplest way: 
    # dc_k/da[j] = 2 a[k-j] (as derived earlier, with a[m]=0 for m<0 or m>=n)
    # da[j]/dz[l] = a[j](δ_jl - a[l])
    # dc_k/dz[l] = sum_j dc_k/da[j] * da[j]/dz[l] = sum_j 2 a[k-j] a[j](δ_jl - a[l])
    # = 2 a[k-l] a[l] - 2 a[l] sum_j a[k-j] a[j]
    # = 2 a[l] a[k-l] - 2 a[l] c_k
    # = 2 a[l] (a[k-l] - c_k)
    
    # So: df/dz[l] = (sum c_k^p)^{1/p-1} * sum_k c_k^{p-1} * 2 a[l] (a[k-l] - c_k)
    # = 2 a[l] * (sum c_k^p)^{1/p-1} * [sum_k c_k^{p-1} a[k-l] - c_k weighted sum...]
    # = 2 a[l] * (sum c_k^p)^{1/p-1} * [ (cp * a_rev)[l shifted] - a[l] * sum_k c_k^{p-1} c_k ]
    
    # Let me define:
    # u_k = c_k^{p-1}
    # v[l] = sum_k u_k * a[k-l] (cross-correlation of u and a)
    # W = sum_k u_k * c_k
    # Then df/dz[l] = 2 a[l] * (sum c_k^p)^{1/p-1} * (v[l] - W)
    
    # v = cross-correlation of u and a
    # v[l] = sum_k u[k] * a[k-l]
    # This is the same as the hankel matvec!
    
    u = cp  # u[k] = c_k^{p-1}
    # v[l] = sum_k u[k] a[k-l] = (u * a_rev)[l + n - 1] ... 
    # Actually, v[l] = sum_k u[k] a[k-l] is a cross-correlation.
    # v = _full_convolve(u, a[::-1]) then extract the right part
    a_rev = a[::-1]
    ua = _full_convolve(u, a_rev)
    v = ua[n-1:2*n-1]
    
    W_total = float(np.dot(u, c))
    
    grad = 2.0 * a * (v - W_total) * (sp ** (1.0/p - 1.0))
    
    return grad


def _lbfgs_pnorm(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with p-norm continuation."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    p_values = [4, 8, 16, 32, 64, 128, 256, 512, 1024]
    
    for p in p_values:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(50, int(remaining / (est_time_per_iter * 20))))
        
        try:
            def obj_p(z_local):
                return _pnorm_obj(z_local, p, n)
            
            def grad_p(z_local):
                return _pnorm_grad(z_local, p, n)
            
            result = minimize(
                obj_p,
                z,
                jac=grad_p,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"  p-norm p={p}: score={sc_new:.10f}")
            else:
                print(f"  p-norm p={p}: no improvement (score={sc_new:.10f})")
        except Exception as e:
            print(f"  p-norm p={p}: error {e}")
    
    return current_a


def _newton_polish(a_init, deadline, max_rounds=200):
    """CG-Newton polish on active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: very tight
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Uniform weights on active set
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()
        
        # Newton step: solve H x = grad
        # grad of sum_k W[k] c_k = H * a
        g = _hankel_matvec(a, W, n)
        # Remove mean (for sum constraint)
        g = g - np.mean(g)
        
        # Solve H dx = -g (Newton direction)
        # But we need to handle the sum constraint: 1^T dx = 0
        # Solve H y = g, H z = 1
        y = _cg_hankel(W, n, g, max_iter=200)
        z = _cg_hankel(W, n, np.ones(n), max_iter=200)
        
        # dx = -(y - (1^T y / 1^T z) * z)
        lam = float(np.ones(n) @ y) / max(float(np.ones(n) @ z), 1e-30)
        dx = -(y - lam * z)
        dx = dx - np.mean(dx)
        
        # Line search
        step_done = False
        alpha = 1.0
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_conv
