    # If solution isn't found, return a bimodal-like candidate
    s = np.sum(final_candidate)
    if s == 0:
        final_candidate = np.ones(n) / n
    
    return final_candidate

def hybrid_candidate(small_n=512, medium_n=1024, large_n=8192):
    """
    Combine optimization on scaled down sequences to fix tokens.
    Generate candidate sequence through gradient search on phases.
    """
    n = large_n  # 8192
    if n == 0:
        n = 4  # fallback to minimal sequence

    # Use gradient descent from Nelder-Mead
    first_candidate = simplex_optimization(np.ones(n) / n, n, max_iter=500)
    if not np.all(np.isfinite(first_candidate)):
        # If invalid, reconstruct from valid parts
        valid_predecessor = np.ones(n) / n
        found = False
        while time.time() < 10 and not found:
            try:
                improvement = simplex_optimization(valid_predecessor, n, attempts=3)
                first_candidate = improvement
                first_score = evaluate_sequence(first_candidate)
                found = first_score < kmo_edge_score_threshold
            except:
                valid_predecessor = np.random.rand(n) / np.sum(np.random.rand(n))
                # try other scaling
    
    # Special construction: create a pattern that is more peaky but then re-normalize
    half_candidate = np.append(first_candidate[:medium_n//2], first_candidate[medium_n//2:medium_n])
    pattern_candidate = np.abs(np.fft.rfft(half_candidate))**0.5
    pattern_candidate = pattern_candidate / np.sum(pattern_candidate)
    # Now combine the pattern_candidate of length medium_n with the original
    pattern_candidate += 0.1 * (np.linalg.norm(first_candidate) - np.linalg.norm(pattern_candidate))
    pattern_candidate = np.maximum(pattern_candidate, 1e-15)
    pattern_candidate_scaled = pattern_candidate * (small_n / large_n)  # alpha correction
    
    final_candidate = np.concatenate([pattern_candidate_scaled[:small_n], pattern_candidate_scaled[small_n:medium_n], pattern_candidate_scaled[-large_n//2:]])
    final_candidate[final_candidate < 0] = 1e-15
    final_candidate = final_candidate / np.sum(final_candidate)
    
    return final_candidate

def main_optimization(seed=42, budget=1800):
    """
    Main optimization function with many restarts until the score drops below threshold.
    """
    t_start = time.time()
    deadline = t_start + budget
    
    best_seq = None
    best_score_val = np.inf
    
    # Try generating candidate from three different initializations
    for _ in range(5):  # at most seven candidates, but at least three
        candidate = hybrid_candidate()
        if evaluate_sequence(candidate) < best_score_val:
            best_score_val = evaluate_sequence(candidate)
            best_seq = candidate.copy()
            print(f"NEWBEST:score={best_score_val:.10f}, candidate sum={np.sum(candidate):.10f}")
        # If no progress, break early
        if t_start + 300 < time.time():
            break

    # If best_seq is still None, try generating a uniform default candidate
    if best_seq is None:
        best_seq = np.ones(8192) / 8192

    # Final polish
    # We now perform a rigorous simplex optimization in the normalized space
    final_result_series = []
    for iter_count in range(10):
        if time.time() >= deadline:
            break
        # Restart the simplex by introducing random perturbations
        new_candidate = best_seq.copy()
        pert = 0.1 * (np.random.standard_normal(len(new_candidate)) - 0.5)
        pert = pert / (np.std(pert) + 1e-10)
        new_candidate += pert
        new_candidate = np.maximum(new_candidate, 0)
        s = np.sum(new_candidate)
        if s < 1e-10:  # We must maintain positive sum above tolerance
            new_candidate = np.ones(len(new_candidate))
            s = 1.0
        new_candidate = new_candidate / s
        
        sc_new = evaluate_sequence(new_candidate)
        if sc_new < best_score_val:
            best_score_val = sc_new
            best_seq = new_candidate.copy()
            print(f"REFINEMENT:score={best_score_val:.10f} after {iter_count} restarts")
        final_result_series.append(best_score_val)
    
    # Choose the candidate that achieved the lowest score
    # Use the last evaluated candidate or an improved one
    if len(final_result_series) > 0:
        best_score_val = min(final_result_series)
        final_index = np.argmin(final_result_series)
        best_seq = final_result_series[final_index]
    else:
        # Fallback to the initial candidate
        best_seq = best_seq
    
    # Check for timeouts or invalid operations
    if time.time() > deadline and best_seq is not None:
        # Return last valid candidate
        return best_seq
    
    return best_seq
