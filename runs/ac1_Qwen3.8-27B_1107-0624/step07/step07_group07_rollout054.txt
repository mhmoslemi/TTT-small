    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones_like(v) * target_sum / len(v)
    return v * (target_sum / s)


def _kkt_residual(a, c, w, n):
    """Compute how non-constant the smeared reverse is on the support."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    support = a > 1e-10
    if support.sum() < 2:
        return 0.0
    h_sup = h[support]
    h_mean = h_sup.mean()
    if abs(h_mean) < 1e-30:
        return float(np.std(h_sup))
    return float(np.std(h_sup) / abs(h_mean))


def _adam_z_optimize(a_init, deadline, min_time=50.0):
    """Adam optimizer in z-space with adaptive log-sum-exp temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    c_cur = _full_convolve(a_init, a_init)
    c_max_cur = float(c_cur.max())
    
    beta = c_max_cur * 0.1
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a_init.copy()
    best_sc = _score(a_init)
    stall = 0
    stall_threshold = 200
    lr_decay = 0.995
    
    while time.monotonic() < deadline - min_time:
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Log-sum-exp smoothing
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s_exp = e.sum()
        if s_exp < 1e-300 or not np.isfinite(s_exp):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_exp
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        # Adaptive learning rate
        if t % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.02, 0.02)
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= lr_decay
                    if lr < 1e-7:
                        break
                
                # Adaptive temperature: reduce beta when KKT residual is large
                if t % 500 == 0:
                    kkt = _kkt_residual(a_check, _full_convolve(a_check, a_check), w, n)
                    if kkt > 0.1:
                        beta *= 0.5
                    elif kkt < 0.01:
                        beta *= 0.8
        
        # Periodic beta reduction
        if t % 500 == 0:
            beta *= 0.7
            if beta < c_max_cur * 1e-8:
                beta = c_max_cur * 1e-8
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _proximal_gradient(a_init, deadline, min_time=30.0):
    """Direct-space proximal gradient with adaptive active set and line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.01
    stall = 0
    stall_threshold = 50
    iters_since_best = 0
    
    for i in range(100000):
        if time.monotonic() > deadline - min_time:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Adaptive active set with hysteresis
        sorted_c = np.sort(c)[::-1]
        n_active = max(5, int(0.05 * len(c)))
        n_active = min(n_active, len(c))
        threshold = sorted_c[n_active - 1]
        tol = max(0.01 * c_max, c_max - threshold)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Backtracking line search
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_try = a - step * frac * g
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                iters_since_best = 0
                break
        
        if improved:
            step = min(step * 1.005, 0.05)
            stall = 0
        else:
            step *= 0.9
            stall += 1
            iters_since_best += 1
            if step < 1e-10 or stall > stall_threshold:
                break
    
    return best_a, best_sc


def _lbfgs_polish(a_init, deadline, min_time=20.0):
    """Fine L-BFGS with very small temperature for final polishing."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    beta = c_max * 1e-6
    
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for attempt in range(3):
        if time.monotonic() > deadline - min_time:
            break
        
        max_iter = min(5000, max(500, int((deadline - min_time - time.monotonic()) / 0.005)))
        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxcor': 100,
                                       'maxls': 50, 'disp': False})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                z0 = result.x.copy()
        except Exception:
            break
    
    return best_a, best_sc


def _perturb(a, rng, mode="gauss", scale=0.05):
    """Perturb a sequence for restart."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 8)))
            freq2 = int(rng.integers(1, max(2, n // 8)))
            amp = scale * 0.3 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            phase2 = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z + amp * 0.5 * np.sin(2.0 * np.pi * freq2 * t / n + phase2)
    elif mode == "block":
        n_blocks = 16
        block_size = n // n_blocks
        for _ in range(n_blocks):
            start = int(_ * block_size)
            end = min(n, start + block_size)
            z[start:end] += scale * 0.2 * rng.standard_normal()
    elif mode == "sparse":
        n_spikes = max(10, n // 100)
        for _ in range(n_spikes):
            idx = int(rng.integers(0, n))
            window = max(1, n // 1000)
            lo = max(0, idx - window)
            hi = min(n, idx + window)
            z[lo:hi] += scale * rng.uniform(-1, 1)
    
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    return a_interp / a_interp.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: Adam z-space, proximal gradient, L-BFGS polish,
    multi-restart with diverse perturbations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score_val = np.inf
    
    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s (was {prev:.10f})")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(8192) / 8192.0
        print(f"No parent. Using uniform n=8192, score={_score(a0):.10f}")
        try_update(a0, "init")
    
    n_best = len(best_seq)
    
    # === Phase 1: Adam in z-space (main workhorse) ===
    print(f"\n=== Phase 1: Adam z-space (n={n_best}) ===")
    t_phase = time.monotonic()
    a1, sc1 = _adam_z_optimize(best_seq, deadline - 700.0, min_time=200.0)
    try_update(a1, "adam1")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a1):.10f}")
    
    # === Phase 2: Proximal gradient (peak shaving) ===
    print(f"\n=== Phase 2: Proximal gradient ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 500.0:
        a2, sc2 = _proximal_gradient(best_seq, deadline - 500.0, min_time=150.0)
        try_update(a2, "prox_grad")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")
    
    # === Phase 3: Second Adam pass on best ===
    print(f"\n=== Phase 3: Adam z-space pass 2 ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400.0:
        a3, sc3 = _adam_z_optimize(best_seq, deadline - 400.0, min_time=100.0)
        try_update(a3, "adam2")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")
    
    # === Phase 4: L-BFGS polishing ===
    print(f"\n=== Phase 4: L-BFGS polish ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 200.0:
        a4, sc4 = _lbfgs_polish(best_seq, deadline - 200.0, min_time=50.0)
        try_update(a4, "lbfgs")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")
    
    # === Phase 5: Multi-restart ===
    print(f"\n=== Phase 5: Multi-restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        ("gauss_s3", "gauss", 0.03),
        ("gauss_s7", "gauss", 0.07),
        ("fourier", "fourier", 0.05),
        ("block", "block", 0.05),
        ("sparse", "sparse", 0.05),
        ("gauss_s15", "gauss", 0.15),
        ("fourier2", "fourier", 0.10),
    ]
    for name, mode, scale in restart_configs:
        if time.monotonic() > deadline - 30.0:
            break
        p_start = _perturb(best_seq, rng, mode, scale)
        sc_start = _score(p_start)
        
        # Adam on perturbed
        a_r, sc_r = _adam_z_optimize(p_start, min(deadline - 20.0, time.monotonic() + 30.0), min_time=5.0)
        try_update(a_r, f"restart_{name}")
        
        # Quick prox on best from restart
        if time.monotonic() < deadline - 15.0:
            a_r2, _ = _proximal_gradient(a_r, min(deadline - 15.0, time.monotonic() + 15.0), min_time=3.0)
            try_update(a_r2, f"restart_{name}_prox")
    
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")
    
    # === Phase 6: Try different length ===
    print(f"\n=== Phase 6: Length exploration ===")
    t_phase = time.monotonic()
    for target_n in [n_best * 2, n_best // 2]:
        if target_n < 512 or target_n > 32768:
            continue
        if time.monotonic() > deadline - 30.0:
            break
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"    Interpolated: score={sc_interp:.10f}")
        try_update(a_interp, f"interp_n{target_n}")
        
        if time.monotonic() < deadline - 20.0:
            a_opt, _ = _adam_z_optimize(a_interp, min(deadline - 20.0, time.monotonic() + 25.0), min_time=5.0)
            try_update(a_opt, f"adam_n{target_n}")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish ===")
    t_phase = time.monotonic()
    for cycle in range(5):
        if time.monotonic() > deadline - 5.0:
            break
        
        a_f1, _ = _proximal_gradient(best_seq, min(deadline - 5.0, time.monotonic() + 15.0), min_time=3.0)
        try_update(a_f1, f"final_prox_{cycle}")
        
        if time.monotonic() > deadline - 5.0:
            break
        a_f2, _ = _adam_z_optimize(best_seq, min(deadline - 5.0, time.monotonic() + 10.0), min_time=2.0)
        try_update(a_f2, f"final_adam_{cycle}")
    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = np.ones(8192) / 8192.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
