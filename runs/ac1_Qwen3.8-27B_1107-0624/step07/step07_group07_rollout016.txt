   Optimizer using Newton-based active set method to minimize
   2*n*max(conv(a,a))/sum(a)^2.
   
   Key idea: At the optimum, the cross-correlation of a with the
   active set indicator must be constant (KKT condition). We solve
   this linear system using CG with FFT-based mat-vec products.
   """
   import numpy as np
   import time
   from scipy.optimize import minimize
   from scipy.sparse.linalg import cg, LinearOperator

   def _next_pow2(n):
       k = 1
       while k < n:
           k <<= 1
       return max(2, k)

   def _full_convolve(a, b):
       n, m = len(a), len(b)
       out_n = n + m - 1
       N = _next_pow2(out_n)
       fa = np.fft.rfft(a, N)
       fb = np.fft.rfft(b, N)
       r = np.fft.irfft(fa * fb, N)[:out_n]
       return r

   def _score(a):
       n = len(a)
       s = float(a.sum())
       if s < 1e-30:
           return np.inf
       c = _full_convolve(a, a)
       return float(2.0 * n * c.max() / (s * s))

   def _normalize(a):
       s = a.sum()
       if s < 1e-30:
           return a
       return a / s

   def _matvec_M(x, v, n):
       """Compute M*x where M[j,m] = v[m+j] for j,m in [0,n-1], v has length 2n-1."""
       n_v = len(v)
       N = _next_pow2(n + n_v - 1)
       f_x = np.fft.rfft(x, N)
       f_v = np.fft.rfft(v[::-1], N)
       conv = np.fft.irfft(f_x * f_v, N)[:n + n_v - 1]
       # c[j] = conv[n_v-1-j] for j=0..n-1
       # n_v-1 = 2n-2
       # So c[j] = conv[2n-2-j]
       return conv[2*n-2 : n-2 : -1]

   def _matvec_MT(x, v, n):
       """M^T = M (since M is symmetric), so same as _matvec_M."""
       return _matvec_M(x, v, n)

   def _solve_newton_step(a, n, active_tol=1e-4, max_cg_iter=200, cg_tol=1e-8, reg=1e-10):
       """
       Perform one Newton step: find a_new such that cross-correlation
       with active set indicator is constant, and sum(a_new) = 1.
       """
       b = _full_convolve(a, a)
       max_b = float(b.max())
       
       if max_b < 1e-30:
           return a, max_b, 0
       
       # Identify active set
       threshold = max_b * (1 - active_tol)
       active = b >= threshold
       n_active = active.sum()
       
       if n_active < 2:
           active = np.zeros_like(b, dtype=bool)
           active[int(np.argmax(b))] = True
           n_active = 1
       
       v = active.astype(np.float64)
       
       # Define the matrix A = M^T * P * M + reg * I
       # where P = I - (1/n)*1*1^T
       ones = np.ones(n)
       
       def matvec_A(x):
           y = _matvec_M(x, v, n)
           y = y - y.mean()  # P * y
           z = _matvec_M(y, v, n)  # M^T * (P*y), M^T = M
           return z + reg * x
       
       # Solve A * x = ones using CG
       try:
           x, info = cg(matvec_A, ones, tol=cg_tol, maxiter=max_cg_iter)
       except Exception:
           x = None
           info = -1
       
       if x is None or info < 0 or not np.all(np.isfinite(x)):
           return a, max_b, n_active
       
       # x should be the solution to M*P*M*x = 1 (with regularization)
       # The new a is proportional to x
       # But we need to ensure sum(x) > 0 and x is reasonable
       if x.sum() < 1e-10:
           return a, max_b, n_active
       
       a_new = np.maximum(x, 0.0)
       a_new = _normalize(a_new)
       
       return a_new, max_b, n_active

   def _newton_loop(a_init, max_rounds=100, active_tol_start=0.1, active_tol_end=0.01, deadline=None, reg=1e-10):
       """
       Iterate Newton steps with decreasing active set tolerance.
       """
       n = len(a_init)
       a = a_init.copy()
       best_a = a.copy()
       best_score = _score(a)
       prev_score = best_score
       stall = 0
       
       for rnd in range(max_rounds):
           if deadline is not None and time.monotonic() > deadline:
               break
           
           # Interpolate active tolerance
           frac = rnd / max(1, max_rounds - 1)
           active_tol = active_tol_start * (active_tol_end / active_tol_start) ** frac
           
           a_new, max_b, n_active = _solve_newton_step(a, n, active_tol, reg=reg)
           
           # Check improvement
           s_new = _score(a_new)
           
           if s_new < best_score - 1e-15:
               best_a = a_new.copy()
               best_score = s_new
               a = a_new
               stall = 0
           else:
               # Try damped update
               improved = False
               for t in [0.5, 0.25, 0.1, 0.05]:
                   a_d = (1-t)*a + t*a_new
                   a_d = np.maximum(a_d, 0.0)
                   a_d = _normalize(a_d)
                   s_d = _score(a_d)
                   if s_d < best_score - 1e-15:
                       best_a = a_d.copy()
                       best_score = s_d
                       a = a_d
                       stall = 0
                       improved = True
                       break
