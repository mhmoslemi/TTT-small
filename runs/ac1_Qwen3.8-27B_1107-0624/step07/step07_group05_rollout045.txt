import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score when sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * beta * a * (h - W)


def _grad_subgrad(a, c, n, top_k=5):
    """Subgradient: average over top-k peaks."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    sorted_idx = np.argsort(c)[::-1]
    top = sorted_idx[:top_k]
    # Use weights proportional to how close to max
    vals = c[top]
    weights = np.exp(-(c_max - vals) / (c_max * 0.05 + 1e-30))
    weights /= weights.sum()
    w = np.zeros(len(c))
    w[top] = weights
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * a * (h - W)


def _peak_equalize(a, K=5, alpha=1.0, max_iter=3, verbose=False):
    """
    Iterative peak equalization via minimum-norm linear algebra.
    Tries to make the top K peaks of conv(a,a) equal.
    Returns (a_new, improved_score).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score_norm(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(max_iter):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:K]
        peak_vals = c[top_k]
        pmin = float(peak_vals.min())
        pmax = float(peak_vals.max())

        if pmax - pmin < 1e-12 * pmax:
            break  # Already equalized

        target = pmin + alpha * (pmax - pmin)

        # Build A matrix: (K+1) x n
        # Row j: A[j, i] = a[k_j - i] if valid
        # Row K: ones
        A = np.zeros((K + 1, n))
        for j in range(K):
            k_j = int(top_k[j])
            # a_{k_j - i} is valid when 0 <= k_j - i < n, i.e., k_j - n + 1 <= i <= k_j
            i_start = max(0, k_j - n + 1)
            i_end = min(n - 1, k_j)
            if i_end >= i_start:
                idx = np.arange(i_start, i_end + 1)
                a_idx = k_j - idx  # indices into a
                A[j, idx] = a[a_idx]
        A[K, :] = 1.0

        # Build b vector
        b = np.zeros(K + 1)
        for j in range(K):
            b[j] = target - c[int(top_k[j])]
        b[K] = 0.0

        # Solve minimum-norm: delta = A^T (A A^T)^{-1} b
        AT = A.T
        AA_T = A @ AT  # (K+1) x (K+1)
        # Add small regularization for numerical stability
        AA_T += 1e-14 * np.eye(K + 1)
        try:
            sol = np.linalg.solve(AA_T, b)
            delta = AT @ sol
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(AA_T, b, rcond=None)[0]
                delta = AT @ sol
            except Exception:
                break

        # Compute max step to maintain non-negativity
        neg_mask = delta < -1e-15
        if np.any(neg_mask):
            step_max = np.min(-a[neg_mask] / delta[neg_mask])
            step = min(1.0, 0.95 * step_max)
        else:
            step = 1.0

        if step < 1e-15:
            break

        a_new = a + step * delta
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score_norm(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try = a + frac * step * delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score_norm(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break

    return best_a, best_sc

    base_beta = max(gap * 0.5, max_c * 0.05)

    betas = base_beta * max_beta_mult * np.array(

            break
        remaining = deadline - min_time_remaining - time.monotonic()

                current_sc = sc_new

def _subgrad_descent(a_init, deadline, n_steps=3000, lr=0.01, top_k=3):
    """Subgradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

        a = _softmax(z)

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)

            a_check = _softmax(z)
            sc = _score_norm(a_check)

    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 4 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

                a_init = a_blend / a_blend.sum()
                a_init = _fourier_perturb(a_init, rng, n_modes=5, amplitude=0.02)
            else:
                # Blend with uniform
                a_blend = 0.8 * best_seq + 0.2 * _make_profile(n_best, "uniform")
                a_init = a_blend / a_blend.sum()

            restart_deadline = time.monotonic() + max(20, min(60, (deadline - 60 - time.monotonic()) / max(1, 4 - i)))
            a_r, sc_r_norm = _optimize_cycle(a_init, restart_deadline, min_remaining=10)
            _update(a_r, _score(a_r))

    # === PHASE 5: Final intensive peak equalization ===
    print(f"\n[Phase 5] Final intensive peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Try various K values
        for K in [3, 4, 5, 6, 7, 8]:
            if time.monotonic() > deadline - 5:
                break
            a_eq, sc_eq_norm = _peak_equalize(best_seq, K=K, alpha=1.0, max_iter=10)
            sc_eq = _score(a_eq)
            _update(a_eq, sc_eq)

        # Final L-BFGS with very small beta
        if time.monotonic() < deadline - 5:
            a_fin = _run_lbfgs_scaled(best_seq, deadline - 3, min_time_remaining=2,
                                       max_beta_mult=0.1)
            _update(a_fin, _score(a_fin))

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_sg, sc_sg = _subgrad_descent(best_seq, deadline - 2, n_steps=500, lr=0.005, top_k=3)
            _update(a_sg, _score(a_sg))

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
