import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s


def _smoothed_grad(a, c, T):
    """Gradient of T*log(sum(exp(c/T))) w.r.t. a, via FFT."""

    return g

def _optimize_simplex(a_init, deadline, T_start, T_end, lr_start, lr_end,
                       max_iter=200000, top_k=3, decay=0.7,
                       eval_interval=100, use_projection=True):

    n = len(a_init)

            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

            lr_eff = lr

            if sc_cand > best_score * 1.02:

def _cd_polish(a, deadline, max_rounds=100, top_k=5):
    """Coordinate descent polishing on peak-contributing coordinates."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for k in active:
            if k < n:
                js = range(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = range(k - (n - 1), n)
            js_list = list(js)
            if len(js_list) == 0:
                continue
            for j in js_list:
                for scale in [1.2, 1.4, 0.8, 0.6, 0.4, 1.6, 0.2]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_a, _score_norm(best_a)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        # Where ag > M, we need to reduce g
        # Multiplicative update: g[i] *= exp(-eta * (ag - M)_+ related)
        excess = np.maximum(ag - M, 0)
        # Map excess back to g coordinates
        excess_rev = excess[::-1]
        # For each i, the "excess pressure" is sum over k where a[k-i]*g[i] contributes
        # Approximate: g[i] should decrease where it contributes most to excess
        pressure = np.zeros(n)
        # pressure[i] = sum_k excess[k] * a[k-i] / (g[i] + eps)
        # Simplified: use FFT
        a_rev = a[::-1]
        excess_contrib = _fft_convolve(excess_rev, a_rev)
        pressure = excess_contrib[n - 1:2 * n - 1][::-1]
        
        eta = 0.1
        g *= np.exp(-eta * pressure / (g + 1e-10))
        s = g.sum()
        if s < 1e-15:
            g = np.ones(n) / n
        else:
            g /= s
    
    return g


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: simplex projection, Adam log-space,
    peak-targeted gradients, LP-inspired directions, multi-scale search,
    and intensive coordinate descent polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_parent = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a_parent = np.ones(n) / n
        print("Using default uniform initialization")
    
    n_parent = len(a_parent)
    best_a = a_parent.copy()
    best_score = _score(a_parent)
    print(f"Initial: n={n_parent}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")
    
    # Create diverse initializations
    inits = [a_parent.copy()]
    
    # Exponential-ish
    a_exp = np.exp(-np.linspace(0, 3, n_parent) * rng.uniform(0.5, 2.0))
    a_exp += rng.uniform(0, 0.1, n_parent)
    inits.append(a_exp / a_exp.sum())
    
    # Anti-symmetric comb
    a_comb = np.ones(n_parent)
    a_comb[::2] *= 3.0
    a_comb += rng.uniform(0, 0.3, n_parent)
    inits.append(a_comb / a_comb.sum())
    
    # Gaussian bump
    x = np.linspace(0, 1, n_parent)
    a_gauss = np.exp(-50 * (x - 0.5) ** 2) + 0.01
    inits.append(a_gauss / a_gauss.sum())
    
    # Random
    a_rand = rng.uniform(0.1, 1.0, n_parent)
    inits.append(a_rand / a_rand.sum())
    
    best_init_score = np.inf
    best_init = a_parent.copy()
    for i, a_i in enumerate(inits):
        sc = _score(a_i)
        print(f"  Init[{i}]: n={len(a_i)}, score={sc:.10f}")
        if sc < best_init_score:
            best_init_score = sc
            best_init = a_i.copy()
    
    # Phase 1: Optimize best init with simplex projection (0-100s)
    p1_end = min(deadline, t_start + 100.0)
    c_cur = _fft_convolve(best_init, best_init)
    max_c = float(c_cur.max())
    
    a_opt1, sc1 = _optimize_simplex(best_init, p1_end,
                                     T_start=0.01 * max_c, T_end=0.0001 * max_c,
                                     lr_start=0.3, lr_end=0.01,
                                     max_iter=100000, top_k=3)
    full_sc1 = _score(a_opt1)
    print(f"Phase 1 (simplex, n={n_parent}): score={full_sc1:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc1) and full_sc1 < best_score:
        best_score = full_sc1
        best_a = a_opt1.copy()
    
    # Phase 2: Adam log-space (100-300s)
    p2_end = min(deadline, t_start + 300.0)
    if time.monotonic() < p2_end:
        c_cur2 = _fft_convolve(best_a, best_a)
        max_c2 = float(c_cur2.max())
        a_opt2, sc2 = _optimize_adam_z(best_a, p2_end,
                                        T_start=0.005 * max_c2, T_end=0.00005 * max_c2,
                                        lr_start=0.04, lr_end=0.003,
                                        max_iter=200000, top_k=3, decay=0.6,
                                        eval_interval=150)
        full_sc2 = _score(a_opt2)
        print(f"Phase 2 (Adam, n={n_parent}): score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()
    
    # Phase 3: CD polish (300-400s)
    p3_end = min(deadline, t_start + 400.0)
    if time.monotonic() < p3_end:
        a_cd, _ = _cd_polish(best_a, p3_end, max_rounds=80, top_k=5)
        full_sc3 = _score(a_cd)
        print(f"Phase 3 (CD, n={n_parent}): score={full_sc3:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc3) and full_sc3 < best_score:
            best_score = full_sc3
            best_a = a_cd.copy()
    
    # Phase 4: Multi-scale (400-900s)
    scales = [4096, 8192, 16384]
    for target_n in scales:
        if time.monotonic() > deadline - 300:
            break
        a_scaled = _interp_to_length(best_a, target_n)
        c_s = _fft_convolve(a_scaled, a_scaled)
        max_s = float(c_s.max())
        
        # Adam at this scale
        sub_end = min(deadline - 300, time.monotonic() + 120.0)
        a_s_opt, _ = _optimize_adam_z(a_scaled, sub_end,
                                       T_start=0.003 * max_s, T_end=0.00003 * max_s,
                                       lr_start=0.03, lr_end=0.002,
                                       max_iter=100000, top_k=3, decay=0.6,
                                       eval_interval=200)
        full_s = _score(a_s_opt)
        print(f"Phase 4 (n={target_n}): score={full_s:.10f}, t={time.monotonic()-t_start:.0f}s")
        
        if np.isfinite(full_s) and full_s < best_score:
            best_score = full_s
            best_a = a_s_opt.copy()
        
        # CD at this scale
        if time.monotonic() < deadline - 280:
            sub_end2 = min(deadline - 280, time.monotonic() + 60.0)
            a_s_cd, _ = _cd_polish(best_a, sub_end2, max_rounds=40, top_k=5)
            full_s_cd = _score(a_s_cd)
            if np.isfinite(full_s_cd) and full_s_cd < best_score:
                best_score = full_s_cd
                best_a = a_s_cd.copy()
            print(f"  CD(n={target_n}): score={full_s_cd:.10f}")
    
    # Phase 5: LP-inspired direction + simplex (900-1200s)
    p5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p5_end:
        n_cur = len(best_a)
        c_cur5 = _fft_convolve(best_a, best_a)
        M_cur = float(c_cur5.max())
        
        # Get LP direction
        g_dir = _lp_direction(best_a, M_cur, min(p5_end, time.monotonic() + 30.0))
        
        # Blend with current
        for t_blend in [0.3, 0.5, 0.7]:
            a_blend = (1 - t_blend) * best_a + t_blend * g_dir
            a_blend = np.maximum(a_blend, 0.0)
            a_blend = a_blend / a_blend.sum()
            sc_b = _score(a_blend)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_a = a_blend.copy()
                print(f"  LP blend t={t_blend}: score={sc_b:.10f}")
        
        # Further optimize the blended result
        if time.monotonic() < p5_end - 30:
            c_b = _fft_convolve(best_a, best_a)
            max_b = float(c_b.max())
            a_p5, _ = _optimize_simplex(best_a, p5_end - 5,
                                         T_start=0.005 * max_b, T_end=0.00005 * max_b,
                                         lr_start=0.2, lr_end=0.01,
                                         max_iter=80000, top_k=3)
            full_p5 = _score(a_p5)
            print(f"Phase 5 (LP+simplex): score={full_p5:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_p5) and full_p5 < best_score:
                best_score = full_p5
                best_a = a_p5.copy()
    
    # Phase 6: Intensive polish (1200-1600s)
    p6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < p6_end:
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        
        # Very low temperature Adam
        mid6 = min(p6_end, time.monotonic() + 150.0)
        a_pol, _ = _optimize_adam_z(best_a, mid6,
                                     T_start=0.001 * max_cb, T_end=0.00001 * max_cb,
                                     lr_start=0.01, lr_end=0.001,
                                     max_iter=200000, top_k=3, decay=0.7,
                                     eval_interval=100, clip_norm=0.4)
        full_pol = _score(a_pol)
        print(f"Phase 6a (low-T Adam): score={full_pol:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_pol) and full_pol < best_score:
            best_score = full_pol
            best_a = a_pol.copy()
        
        # CD polish
        if time.monotonic() < p6_end - 30:
            a_cd6, _ = _cd_polish(best_a, p6_end - 5, max_rounds=60, top_k=8)
            full_cd6 = _score(a_cd6)
            print(f"Phase 6b (CD): score={full_cd6:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_cd6) and full_cd6 < best_score:
                best_score = full_cd6
                best_a = a_cd6.copy()
    
    # Phase 7: Perturbation restarts (1600-1750s)
    p7_end = min(deadline, t_start + 1750.0)
    n_fin = len(best_a)
    if time.monotonic() < p7_end:
        for attempt in range(3):
            if time.monotonic() > p7_end - 40:
                break
            sigma = 0.015 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(p7_end - 10, time.monotonic() + 40.0)
            a_pert_opt, _ = _optimize_adam_z(a_pert, sub_dl,
                                              T_start=0.001 * max_cp, T_end=0.00001 * max_cp,
                                              lr_start=0.02, lr_end=0.001,
                                              max_iter=50000, top_k=3, decay=0.6,
                                              eval_interval=80)
            full_pert = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_pert:.10f}")
            if np.isfinite(full_pert) and full_pert < best_score:
                best_score = full_pert
                best_a = a_pert_opt.copy()
    
    # Phase 8: Final ultra-fine polish (1750-deadline)
    if time.monotonic() < deadline - 5:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _optimize_adam_z(best_a, deadline - 4,
                                       T_start=0.0003 * max_cf, T_end=0.000003 * max_cf,
                                       lr_start=0.005, lr_end=0.0003,
                                       max_iter=100000, top_k=3, decay=0.8,
                                       eval_interval=50, clip_norm=0.3)
        full_final = _score(a_final)
        print(f"Phase 8 (final): score={full_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_final) and full_final < best_score:
            best_score = full_final
            best_a = a_final.copy()
        
        # Quick CD
        if time.monotonic() < deadline - 2:
            a_cd_f, _ = _cd_polish(best_a, deadline - 1, max_rounds=15, top_k=10)
            sc_cd_f = _score(a_cd_f)
            if sc_cd_f < best_score:
                best_score = sc_cd_f
                best_a = a_cd_f.copy()
    
    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
