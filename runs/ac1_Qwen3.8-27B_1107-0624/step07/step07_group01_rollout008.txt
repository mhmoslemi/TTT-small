"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with aggressive log-sum-exp smoothing (beta down to 1e-8)
2. Precise subgradient method with backtracking line search and adaptive active set
3. Newton-like active set equalization
4. Multi-length exploration (n=2048 to 32768)
5. Basin hopping for diversification
6. Starting from parent construction (n=16384)
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv(a):
    return _full_convolve(a, a)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    def fun(z):
        return _smoobj(z, beta, n)
    def jac(z):
        return _smograd(z, beta, n)
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    current_a = a_start.copy()
    current_sc = _score(current_a)
    # More aggressive beta schedule
    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              0.00005, 0.00002, 0.00001, 0.000005,
                              0.000002, 0.000001, 0.0000005, 0.0000002,
                              0.0000001])
    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break
        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(20000, max(200, int(remaining / (0.005 * scale * 20))))
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
        if current_sc < 1.5005:
            break
    return current_a

def _precise_subgrad(a_init, deadline, k_active=5, max_steps=100000):
    """
    Precise subgradient method with backtracking line search.
    Works in log-space (softmax parameterization).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.05
    stall_count = 0
    max_stall = 50
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Adaptive active set: top-k indices
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        
        # Compute gradient in a-space
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * n * (h - W)  # gradient of 2n*max(c) w.r.t. a (unnormalized)
        
        # Convert to z-space gradient (for softmax)
        # d/da = dx/dz * dz/da, so dz_grad = grad_a / a (approximately, for softmax)
        # More precisely: if a = softmax(z), then da_i/dz_j = a_i(δ_ij - a_j)
        # So ∂f/∂z_j = sum_i ∂f/∂a_i * da_i/∂z_j = sum_i grad_i * a_i(δ_ij - a_j)
        # = a_j * (sum_i grad_i * a_i - sum_i grad_i * a_i * a_j)
        # = a_j * (G - G * a_j) where G = sum_i grad_i * a_i
        # Actually: ∂f/∂z_j = a_j * grad_j - a_j * sum_i a_i * grad_i
        grad_z = a * grad_a - a * np.dot(a, grad_a)
        
        # Backtracking line search
        sc_cur = _score(a)
        improved = False
        alpha = lr
        for _ls in range(15):
            z_try = z - alpha * grad_z
            z_try = z_try - z_try.mean()
            z_try = np.clip(z_try, -50.0, 50.0)
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                stall_count = 0
                alpha = min(alpha * 1.05, 0.5)
                lr = alpha
                improved = True
                break
            elif sc_try < sc_cur:
                # Accept even if not globally best (for progress)
                z = z_try
                improved = True
                break
            else:
                alpha *= 0.5
        
        if not improved:
            stall_count += 1
            lr *= 0.7
            if lr < 1e-10:
                break
            if stall_count > max_stall:
                break
        else:
            if improved and not (sc_try < best_sc - 1e-15):
                # We made progress but not to new best
                stall_count += 1
                if stall_count > max_stall:
                    lr *= 0.8
                    if lr < 1e-10:
                        break
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _newton_active_set(a_init, deadline, k_active=5, max_iter=100):
    """
    Try to equalize the top-k convolution values using a Newton-like approach.
    At each step, compute the direction that most reduces the spread of active values.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        spread = active_vals.max() - active_vals.min()
        
        if spread < 1e-15 * c_max:
            break
        
        # Compute gradient direction to equalize active values
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()  # Remove mean to maintain sum constraint
        
        # Line search
        improved = False
        alpha = 1.0 / n
        for _ in range(15):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6
        
        if not improved:
            break
    
    return best_a, best_sc

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

def _basin_hop(a_current, deadline, rng, n_hops=5, reserve=10.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))
    
    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.8
        
        pert_type = h % 3
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.01 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 5 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.03, 0.03)
            z = z - z.mean()
            a_pert = _softmax(z)
        
        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        a_opt = _lbfgs_smooth(a_pert, beta, n, 2000)
        sc_opt = _score(a_opt)
        
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
    
    return best_a, best_sc
