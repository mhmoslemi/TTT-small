from scipy.optimize import minimize
from scipy.linalg import solve

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _pressure_shave(a, deadline, n_peaks=20, n_iters=500, gamma=0.5):
    """
    Adaptive-weight pressure equalization.
    Focuses on the top n_peaks of the convolution.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = np.max(c)
        if c_max < 1e-30:
            break
        
        # Select active set: top n_peaks or within tolerance
        tol = 0.005 * c_max
        active_thresh = c_max - tol
        active = c >= active_thresh
        if active.sum() < n_peaks:
            # Expand to include top n_peaks
            idx = np.argsort(c)[-n_peaks:]
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
        elif active.sum() > 10 * n_peaks:
            # Narrow to top n_peaks
            idx = np.argsort(c)[-n_peaks:]
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
        
        # Compute weights: higher weight for higher peaks
        c_active = c[active]
        w_raw = c_active - c_active.min() + 1e-10
        w = w_raw / w_raw.sum()
        
        w_full = np.zeros(len(c))
        w_full[active] = w
        
        # Compute pressure: p[i] = 2 * sum_k w_k * a[k-i] (if valid)
        a_rev = a[::-1]
        p = _full_convolve(w_full, a_rev)
        p = p[n-1:2*n-1]
        p = np.maximum(p, 1e-30)
        
        # Update: a[i] *= (mean(p) / p[i])^gamma
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if it % 25 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, _score(best_a)

def _lsmobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsmgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Adaptive beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 8))))
        
        try:
            result = minimize(
                _lsmobj, z, args=(beta, n),
                jac=_lsmgrad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _coord_descent(a, deadline, n_rounds=200):
    """
    Coordinate descent: adjust individual elements to reduce the max peak.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(42)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find coordinates that most affect the peaks
        # Top peaks
        n_peaks = min(10, len(c))
        peak_idx = np.argsort(c)[-n_peaks:]
        
        # For each peak, the coordinates that contribute most are those i
        # where a[i] * a[peak_i - i] is large
        contrib = np.zeros(n)
        for k in peak_idx:
            valid_i = range(max(0, k-n+1), min(k+1, n))
            for i in valid_i:
                contrib[i] += a[i] * a[k-i]
        
        # Try modifying the top contributing coordinates
        n_try = min(n, 50)
        top_contrib = np.argsort(contrib)[-n_try:]
        order = rng.permutation(top_contrib)
        
        improved = False
        for i in order[:min(len(order), 20)]:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            c_before = c.max()
            
            # Try reducing
            for factor in [0.7, 0.85, 0.95]:
                a[i] = old_val * factor
                c_try = _full_convolve(a, a).max()
                if c_try < c_before - 1e-15:
                    improved = True
                else:
                    a[i] = old_val
                    break
        
        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a, _score(best_a)

def _chirp_init(n, freq=1.0):
    """Quadratic chirp, shifted to be non-negative."""
    t = np.arange(n) / n
    a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _gaussian_taper(n, width=0.3):
    t = np.arange(n) / n
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result
