def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)

def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

        # Select pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n

        s_pair = a[i] + a[j]
        if s_pair < 1e-15:
            continue

        # Try different splits
        improved = False
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7, 0.45, 0.55]:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[i] = new_i
            a_try[j] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

    return best_a, best_sc


def _diversified_restart(a_init, rng, mode, scale=0.03):
    """Create a perturbed version for restart."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 20)))
            amp = scale * 0.5 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "highfreq":
        t = np.arange(n)
        for _ in range(5):
            freq = int(rng.integers(n // 4, n // 2))
            amp = scale * 0.3 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(16, n // 32)
        for _ in range(12):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-scale * 0.5, scale * 0.5)

    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = max(1, n // 20)
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n - 1 - i] *= (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses adaptive continuation L-BFGS, peak equalization, momentum descent,
    Newton peak shaving, and diversified multi-restart strategy.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s (improved from {prev:.10f})")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n0 = 8192
        a0 = _make_profile(n0, "uniform")
        print(f"No parent. Using uniform n={n0}, score={_score(a0):.10f}")
        try_update(a0, "init")

    n = len(best_seq)
    print(f"\nStarting optimization with n={n}, score={best_score_val:.10f}")

    # === Phase 1: Adaptive L-BFGS with continuation ===
    print(f"\n=== Phase 1: Adaptive L-BFGS ===")
    t_phase = time.monotonic()
    p1_deadline = deadline - 400.0
    a_p1, sc_p1 = _adaptive_lbfgs(best_seq, p1_deadline, reserve_time=80.0)
    try_update(a_p1, "lbfgs_adaptive")
    print(f"  Phase 1 score: {sc_p1:.10f} (time={time.monotonic()-t_phase:.1f}s)")

    # === Phase 2: Peak equalization ===
    print(f"\n=== Phase 2: Peak Equalization ===")
    t_phase = time.monotonic()
    p2_deadline = deadline - 250.0
    if time.monotonic() < p2_deadline:
        a_p2, sc_p2 = _peak_equalization(best_seq, p2_deadline, max_rounds=2000)
        try_update(a_p2, "peak_eq")
    print(f"  Phase 2 score: {sc_p2 if 'sc_p2' in dir() else best_score_val:.10f} "
          f"(time={time.monotonic()-t_phase:.1f}s)")

    # === Phase 3: Newton peak shaving ===
    print(f"\n=== Phase 3: Newton Peak Shaving ===")
    t_phase = time.monotonic()
    p3_deadline = deadline - 150.0
    if time.monotonic() < p3_deadline:
        a_p3, sc_p3 = _newton_peak_shave(best_seq, p3_deadline, max_iters=3000)
        try_update(a_p3, "newton_ps")
    print(f"  Phase 3 score: {sc_p3 if 'sc_p3' in dir() else best_score_val:.10f} "
          f"(time={time.monotonic()-t_phase:.1f}s)")

    # === Phase 4: Momentum descent ===
    print(f"\n=== Phase 4: Momentum Descent ===")
    t_phase = time.monotonic()
    p4_deadline = deadline - 80.0
    if time.monotonic() < p4_deadline:
        a_p4, sc_p4 = _momentum_descent(best_seq, p4_deadline, max_iters=30000)
        try_update(a_p4, "momentum")
    print(f"  Phase 4 score: {sc_p4 if 'sc_p4' in dir() else best_score_val:.10f} "
          f"(time={time.monotonic()-t_phase:.1f}s)")

    # === Phase 5: Multi-restart with polish ===
    print(f"\n=== Phase 5: Multi-Restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        ("gauss_002", "gauss", 0.002),
        ("gauss_005", "gauss", 0.005),
        ("gauss_01", "gauss", 0.01),
        ("gauss_02", "gauss", 0.02),
        ("fourier", "fourier", 0.02),
        ("highfreq", "highfreq", 0.01),
        ("block", "block", 0.01),
        ("gauss_05", "gauss", 0.05),
    ]

    for name, mode, scale in restart_configs:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        p_start = _diversified_restart(best_seq, rng, mode, scale)
        sc_start = _score(p_start)

        # Quick L-BFGS
        alloc = min(remaining * 0.4, 60.0)
        sub_deadline = time.monotonic() + max(5.0, alloc)
        a_r, sc_r = _adaptive_lbfgs(p_start, sub_deadline, reserve_time=5.0)
        try_update(a_r, f"restart_{name}")

        # Quick peak shave
        if time.monotonic() < deadline - 10:
            a_r2, sc_r2 = _momentum_descent(a_r, min(time.monotonic() + 10, deadline - 5),
                                            max_iters=5000)
            try_update(a_r2, f"restart_{name}_md")

    print(f"  Phase 5 done (time={time.monotonic()-t_phase:.1f}s)")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        for cycle in range(5):
            if time.monotonic() > deadline - 3:
                break

            # Momentum descent
            a_f1, sc_f1 = _momentum_descent(best_seq, min(time.monotonic() + 8, deadline - 3),
                                            max_iters=10000)
            try_update(a_f1, f"polish_md_{cycle}")

            # Peak equalization
            if time.monotonic() < deadline - 3:
                a_f2, sc_f2 = _peak_equalization(best_seq, min(time.monotonic() + 5, deadline - 3),
                                                  max_rounds=500)
                try_update(a_f2, f"polish_pe_{cycle}")

            # Coordinate descent
            if time.monotonic() < deadline - 3:
                a_f3, sc_f3 = _coordinate_descent(best_seq, min(time.monotonic() + 3, deadline - 2),
                                                   max_rounds=5000)
                try_update(a_f3, f"polish_cd_{cycle}")

    print(f"  Phase 6 done (time={time.monotonic()-t_phase:.1f}s)")

    # === Phase 7: Ultra-fine L-BFGS squeeze ===
    print(f"\n=== Phase 7: Ultra-Fine Squeeze ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 2:
        n_best = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        # Try very small betas
        for beta_frac in [1e-7, 5e-8, 2e-8, 1e-8]:
            if time.monotonic() > deadline - 2:
                break
            beta = mb * beta_frac
            remaining = deadline - 2 - time.monotonic()
            if remaining < 3:
                break
            est_per_iter = 0.0015 * (n_best / 4096.0) ** 1.3
            max_iter = min(5000, max(200, int(remaining * 0.7 / est_per_iter)))

            a_ultra, _, niter = _lbfgs_smo(z, beta, n_best, max_iter, deadline - 1)
            try_update(a_ultra, f"ultra_{beta_frac:.0e}")

    print(f"  Phase 7 done (time={time.monotonic()-t_phase:.1f}s)")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    sc_fallback = _score(fallback)
    print(f"FALLBACK: score={sc_fallback:.10f}")
    return [float(x) for x in fallback]
