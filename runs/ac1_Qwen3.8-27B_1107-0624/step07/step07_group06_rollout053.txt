Optimized step function sequence generator targeting score < 1.5027.
Uses multi-stage optimization: power-sum smoothing with adaptive parameters,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# Power-sum smoothing: approximates max(c) using (sum c^p)^(1/p)
def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    # c is always non-negative for non-negative a
    c_p = np.power(c, p)
    val = np.sum(c_p) ** (1.0 / p)
    return 2.0 * n * val

def _power_sum_grad(z, p, n):

    c = _full_convolve(a, a)
    c_safe = np.maximum(c, 1e-30)
    c_p = np.power(c_safe, p)
    sum_cp = np.sum(c_p)
    if sum_cp < 1e-300:

# Active set gradient
def _active_grad(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _hess_vec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result

# CG solver for Newton step
def _cg_solve(hess_func, b, tol=1e-10, max_iter=200):
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_func(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new

# Fourier domain perturbation
def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.3):
    n = len(a)
    A = np.fft.rfft(a)
    # Perturb the phase (preserves amplitude)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    # Only perturb non-DC, non-Nyquist
    phase_pert[0] = 0
    if n % 2 == 0:
        phase_pert[-1] = 0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

# Block swap perturbation
def _perturb_block_swap(a, rng, n_swaps=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_swaps * 2))
    for _ in range(n_swaps):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        step = block_size
        for s in range(min(step, (n - i) // step)):
            ii = i + s * step
            jj = j + s * step
            if ii + step > n or jj + step > n:
                break
            tmp = z[ii:ii + step].copy()
            z[ii:ii + step] = z[jj:jj + step]
            z[jj:jj + step] = tmp
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

# Perturbation methods
# ============================================================

def _perturb_fourier_phase(a, rng, n_modes=10, max_amp=0.5):
    n = len(a)
    A = np.fft.rfft(a)
    phase_pert = rng.uniform(-max_amp, max_amp, len(A))
    phase_pert[0] = 0.0
    if n % 2 == 0:
        phase_pert[-1] = 0.0
    A_new = A * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_amplitude(a, rng, sigma=0.1):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    amp_pert = amp * (1.0 + sigma * rng.standard_normal(len(amp)))
    amp_pert = np.maximum(amp_pert, 0)
    phase = np.angle(A)
    A_new = amp_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_swaps=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(4, n // (4 * n_swaps))
    for _ in range(n_swaps):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_mix(a, rng):
    a1 = _perturb_log(a, 0.02, rng)
    a2 = _perturb_fourier_phase(a1, rng, n_modes=5, max_amp=0.2)
    return a2

# Profile generators

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    else:
        return np.ones(n) / n


# ============================================================
# Variance minimization (secondary objective)
# ============================================================

def _variance_obj(a):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    return g - np.mean(g)


def _variance_warm_start(a_init, deadline, n_steps=200):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.003

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _variance_grad(a, n)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        v_new = _variance_obj(a_new)
        v_old = _variance_obj(a)
        if v_new < v_old:
            a = a_new
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.85
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _resize_sequence(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a)
    a_target = np.maximum(a_target, 0)
    s = a_target.sum()
    if s < 1e-10:
        return np.ones(n_target) / n_target
    return a_target / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp and power-sum
    smoothing, Newton active-set refinement, subgradient polish, Fourier-domain
    perturbations, and multi-n exploration.
    """
    global _best_seq, _best_score
    _best_seq = None
    _best_score = np.inf

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    def upd(p, sc=None, label=""):
        global _best_seq, _best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < _best_score:
            _best_seq = p.copy()
            _best_score = float(sc)
            print(f"  NEW BEST: {_best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        upd(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        upd(parent, _score(parent), "init_uniform")

    # === Phase 1: Quick diverse exploration (60s) ===
    print(f"\n[Phase 1] Diverse exploration")
    expl_deadline = min(deadline, t_start + 60)
    n_expl = max(1024, n_parent // 8) if n_parent > 2048 else 1024
    for kind in ["uniform", "tapered", "cos2", "wheatstone", "gauss", "bimodal", "ramp"]:
        if time.monotonic() > expl_deadline:
            break
        a_try = _make_profile(n_expl, kind)
        sc_try = _score(a_try)
        upd(a_try, sc_try, f"expl_{kind}")

    # Upscale best to parent size and optimize
    if _best_seq is not None and n_parent > 0 and time.monotonic() < deadline - 600:
        a_up = _resize_sequence(_best_seq, n_parent) if len(_best_seq) != n_parent else _best_seq.copy()
        a_opt, sc_opt = _lbfgs_lse(a_up, deadline, min_remaining=500)
        upd(a_opt, sc_opt, "expl_lbfgs")

    # === Phase 2: Main L-BFGS with LSE (500s) ===
    print(f"\n[Phase 2] Main L-BFGS LSE")
    if time.monotonic() < deadline - 500:
        a_main, sc_main = _lbfgs_lse(_best_seq.copy(), deadline, min_remaining=450)
        upd(a_main, sc_main, "lbfgs_lse_main")

    # === Phase 3: Power-sum L-BFGS (200s) ===
    print(f"\n[Phase 3] Power-sum L-BFGS")
    if time.monotonic() < deadline - 200:
        a_ps, sc_ps = _lbfgs_power_sum(_best_seq.copy(), deadline, min_remaining=180)
        upd(a_ps, sc_ps, "lbfgs_power_sum")

    # === Phase 4: Newton active-set (200s) ===
    print(f"\n[Phase 4] Newton active-set")
    if time.monotonic() < deadline - 150:
        n_dead = min(deadline - 130, time.monotonic() + 200)
        a_nw, sc_nw = _newton_active_set(_best_seq, n_dead, max_iter=300, tol_frac=0.001)
        upd(a_nw, sc_nw, "newton_main")

    # === Phase 5: Subgradient polish (150s) ===
    print(f"\n[Phase 5] Subgradient polish")
    if time.monotonic() < deadline - 100:
        sg_dead = min(deadline - 80, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(_best_seq, sg_dead, n_steps=5000, lr_init=0.008, tol_frac=0.001)
        upd(a_sg, sc_sg, "sg_polish")

    # === Phase 6: Newton + SG交替 (150s) ===
    print(f"\n[Phase 6] Newton + SG alternating")
    if time.monotonic() < deadline - 120:
        a_cur = _best_seq.copy()
        for round_i in range(3):
            if time.monotonic() > deadline - 100:
                break
            # Newton step
            nw_dead = min(deadline - 100, time.monotonic() + 40)
            a_nw2, sc_nw2 = _newton_active_set(a_cur, nw_dead, max_iter=100, tol_frac=0.002)
            if sc_nw2 < _score(a_cur):
                a_cur = a_nw2
            upd(a_cur, _score(a_cur), f"alt_newton_{round_i}")
            # SG step
            sg_dead2 = min(deadline - 100, time.monotonic() + 30)
            a_sg2, sc_sg2 = _subgrad_polish(a_cur, sg_dead2, n_steps=2000, lr_init=0.005, tol_frac=0.001)
            if sc_sg2 < _score(a_cur):
                a_cur = a_sg2
            upd(a_cur, _score(a_cur), f"alt_sg_{round_i}")

    # === Phase 7: Multi-n exploration (300s) ===
    print(f"\n[Phase 7] Multi-n exploration")
    n_candidates = [2048, 4096, 8192, 12288, 16384]
    if n_parent not in n_candidates and n_parent >= 1024:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if len(_best_seq) == n_try:
            a_init = _best_seq.copy()
        else:
            a_init = _resize_sequence(_best_seq, n_try)

        # LSE optimization
        a_opt_n, sc_opt_n = _lbfgs_lse(a_init, deadline, min_remaining=60)
        upd(a_opt_n, sc_opt_n, f"n_{n_try}_lse")

        # Newton polish
        if _score(a_opt_n) < _best_score * 1.005 and time.monotonic() < deadline - 50:
            nw_dead = min(deadline - 40, time.monotonic() + 60)
            a_opt_n2, sc_opt_n2 = _newton_active_set(a_opt_n, nw_dead, max_iter=80, tol_frac=0.002)
            upd(a_opt_n2, sc_opt_n2, f"n_{n_try}_newton")

    # === Phase 8: Diverse restarts (200s) ===
    print(f"\n[Phase 8] Diverse restarts")
    if time.monotonic() < deadline - 60:
        n_restarts = 6
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            if i == 0:
                p_start = _perturb_log(_best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier_phase(_best_seq, rng, n_modes=8, max_amp=0.3)
            elif i == 2:
                p_start = _perturb_block_swap(_best_seq, rng, n_swaps=4)
            elif i == 3:
                p_start = _perturb_amplitude(_best_seq, rng, sigma=0.08)
            elif i == 4:
                p_start = _perturb_mix(_best_seq, rng)
            else:
                p_start = _perturb_log(_best_seq, 0.05, rng)

            a_r, sc_r = _lbfgs_lse(p_start, deadline, min_remaining=25)
            upd(a_r, sc_r, f"restart_{i}")

            if sc_r < _best_score * 1.003 and time.monotonic() < deadline - 20:
                nw_dead = min(deadline - 15, time.monotonic() + 25)
                a_r2, sc_r2 = _newton_active_set(a_r, nw_dead, max_iter=50, tol_frac=0.003)
                upd(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 9: Final aggressive polish (60s) ===
    print(f"\n[Phase 9] Final polish")
    if _best_seq is not None and time.monotonic() < deadline - 5:
        # Very low beta LSE
        n_fin = len(_best_seq)
        c_fin = _full_convolve(_best_seq, _best_seq)
        mb = float(c_fin.max())
        z_fin = np.log(np.maximum(_best_seq, 1e-30))
        z_fin = z_fin - z_fin.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            beta_final = mb * 0.00005
            max_iter = min(2000, max(50, int(remaining / 0.005)))
            try:
                result = minimize(
                    _lse_objective, z_fin, args=(beta_final, n_fin),
                    jac=_lse_gradient, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-14, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                upd(a_fin, sc_fin, "final_ultra_lse")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(_best_seq, deadline - 2, max_iter=50, tol_frac=0.001)
            upd(a_fin2, sc_fin2, "final_newton")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(_best_seq, deadline - 1, n_steps=3000, lr_init=0.005, tol_frac=0.0005)
            upd(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if _best_seq is not None:
        final = np.clip(_best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform").tolist()
            print(f"FALLBACK uniform n=256")
            return [float(x) for x in final]
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Ultimate fallback
    fallback = _make_profile(256, "uniform")
    print(f"ULTIMATE FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
