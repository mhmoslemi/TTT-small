```
def peak_deflation(a, c, n, target=None, max_rounds=1000):
    if target is None:
        target = c.mean()  # try to bring peaks down to the mean
    
    a = a.copy()
    for _ in range(max_rounds):
        c = full_convolve(a, a)
        k_star = argmax(c)
        c_max = c[k_star]
        if c_max - target < 1e-10:
            break
        
        # Amount to reduce: we want c[k_star] to decrease by (c_max - target)
        # c[k_star] = sum_i a[i] * a[k_star - i]
        # If we reduce a[i] by eps_i for all i, the change in c[k_star] is:
        #   delta_c = -sum_i eps_i * a[k_star-i] (first order)
        # We want delta_c = -(c_max - target)
        # Simplest: reduce all a[i] that contribute to c[k_star] proportionally
        #   eps_i = alpha * a[i] for i in support
        # Then delta_c = -alpha * sum_i a[i]^2 * ... 
        # Actually, if we reduce a[i] by eps_i = alpha * a[i], then:
        # new c[k_star] = sum_i (a[i] - alpha*a[i]) * (a[k_star-i] - alpha*a[k_star-i])
        # ≈ c[k_star] - alpha * sum_i (a[i]*a[k_star-i] + a[k_star-i]*a[i])  (first order in alpha)
        # Wait, that's not right either since both terms change.
        # First order: delta_c[k_star] ≈ -alpha * 2 * sum_i a[i] * a[k_star-i] ... no
        # delta_c[k_star] = sum_i (a[i]-eps_i)(a[k_star-i]-eps_{k_star-i}) - sum_i a[i]a[k_star-i]
        # ≈ -sum_i (eps_i * a[k_star-i] + a[i] * eps_{k_star-i})  (first order)
        # If eps_i = alpha * a[i] for all i:
        # ≈ -alpha * sum_i (a[i]*a[k_star-i] + a[i]*a[k_star-i]) = -2*alpha*c[k_star]
        # So alpha = (c_max - target) / (2 * c_max)
        
        # But this changes all other convolution values too, so it's not perfect.
        # Still, it's a good heuristic.
        
        alpha = (c_max - target) / (2.0 * c_max)
        alpha = min(alpha, 0.1)  # cap the step
        a = a * (1.0 - alpha)  # uniform reduction... but this just scales!
        
        # That's not helpful. We need to reduce specifically the elements that
        # contribute to the peak.
        
        # Better approach: for the peak at k_star, the "pressure" on a[i] is a[k_star-i].
        # Reduce a[i] more where pressure is high.
        pressure = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pressure[i] = a[j]
        
        # Reduce a[i] proportional to pressure[i]
        max_pressure = pressure.max()
        if max_pressure > 1e-15:
            step = alpha * 2.0 * pressure / max_pressure
            a = np.maximum(a - step, 0.0)
            a = a / a.sum()
    
    return a
```

```python
        violations = np.where(fb > M + 1e-8)[0]
        if len(violations) == 0:

        # Add the most violated constraint
        worst = violations[np.argmax(fb[violations])]
        if worst not in active_lags:
```

```python
"""
Optimizes non-negative step sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based best response with constraint generation, fine L-BFGS with
log-sum-exp smoothing, adaptive peak shaving, multi-length exploration,
and multi-restart with diverse perturbations.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adaptive_active_set(c, tol_frac_base=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(3, int(0.02 * len(c)))
    if n_active_est >= len(c):
        n_active_est = len(c)
    threshold_from_rank = sorted_c[min(n_active_est - 1, len(c) - 1)]
    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac_base * c_max
    tol = max(tol_fixed, min(tol_rank, 0.5 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    try:
        result = minimize(_smoobj, z0, jac=_smograd, args=(beta, n),
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, float(result.fun), int(result.nit)
    except Exception:
        return a_init.copy(), _score(a_init), 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
                              1e-6, 5e-7, 2e-7, 1e-7, 5e-8,
                              2e-8, 1e-8])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        est_per_iter = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(20000, max(300, int(remaining / (est_per_iter * 15))))
        a_new, _, _ = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.2e}: score={sc:.10f}")

    return current_a


def _peak_shaving(a_init, deadline, max_iters=80000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.015
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = c.max()
        w = _adaptive_active_set(c, 1e-5)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g_norm_dir = g / g_norm
        best_sc_try = best_sc
        best_a_try = a.copy()
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g_norm_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-15:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.08)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-10:
                break
            if stall > 200:
                break
    return best_a, best_sc


def _gradient_momentum(a_init, deadline, max_steps=50000, lr_init=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)
    mu = 0.9
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    beta = _full_convolve(a_init, a_init).max() * 0.001

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        beta = max(beta * (1.0 - 0.001), _full_convolve(a_init, a_init).max() * 1e-9)
        g = _smograd(z, beta, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        step_size = lr / (1.0 + 0.0001 * i)
        v = mu * v - step_size * g
        z_new = z + v
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            z = z_new
            lr *= 1.02
            stall = 0
        else:
            z = z + 0.1 * v
            v *= 0.5
            stall += 1
            if stall > 10:
                lr *= 0.8
                if lr < 1e-8:
                    break
    return best_a, best_sc


def _lp_best_response(f, M, n, deadline, max_cg_rounds=100):
    """LP-based best response with constraint generation."""
    b = cp.Variable(n)
    # Start with top lags
    c = _full_convolve(f, f)
    top_k = int(min(50, 2 * n - 1))
    active_lags = np.argsort(c)[::-1][:top_k].tolist()
    b_val = np.ones(n) / n

    for cg_round in range(max_cg_rounds):
        if time.monotonic() > deadline:
            break
        constraints = [b >= 0]
        for k in active_lags:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                coeffs = f[k - np.arange(j_start, j_end + 1)]
                constraints.append(coeffs @ b[j_start:j_end + 1] <= M)

        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b.value is None:
            break
        b_val = np.maximum(b.value, 0.0)
        # Check for violated constraints
        fb = _full_convolve(f, b_val)
        violations = np.where(fb > M * (1 + 1e-6) + 1e-12)[0]
        if len(violations) == 0:
            break
        worst_idx = violations[np.argmax(fb[violations])]
        if worst_idx not in active_lags:
            active_lags.append(int(worst_idx))

    return b_val


def _lp_iterations(a_init, deadline, n_lp_iters=30):
    """Run LP-based best response iterations."""
    n = len(a_init)
    f = a_init.copy()
    s = f.sum()
    f = f / s
    best_a = f.copy()
    best_sc = _score(f)
    t_step = 0.3

    for iter_i in range(n_lp_iters):
        if time.monotonic() > deadline:
            break
        M = _full_convolve(f, f).max()
        b_val = _lp_best_response(f, M, n, deadline, max_cg_rounds=50)
        if b_val is None or b_val.sum() < 1e-10:
            break
        b_val = b_val / b_val.sum()
        t = t_step / (1.0 + 0.1 * iter_i)
        f_new = (1.0 - t) * f + t * b_val
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = f_new.copy()
            f = f_new
            print(f"    LP iter {iter_i}: score={sc:.10f}")
        else:
            f = f_new
    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * 3 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0, 1, n)
        a = (t * (1 - t)) ** 0.2 + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_for_restart(a, rng, mode="gauss", scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.002 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(8, n // 12)
        for _ in range(12):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.015, 0.015)
    elif mode == "sparse":
        n_sparse = max(10, n // 50)
        idx = rng.choice(n, size=n_sparse, replace=False)
        z[idx] += rng.uniform(-0.05, 0.05, size=n_sparse)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent(a_init, deadline, max_rounds=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n
        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue
        improved = False
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        if not improved and rnd > 200:
            # Try triple
            idx2 = int(rng.integers(0, n))
            jdx2 = int(rng.integers(0, n))
            kdx2 = int(rng.integers(0, n))
            if len(set([idx2, jdx2, kdx2])) < 3:
                continue
            s_tri = a[idx2] + a[jdx2] + a[kdx2]
            if s_tri < 1e-15:
                continue
            fracs = rng.dirichlet(np.ones(3))
            a_try = a.copy()
            a_try[idx2] = s_tri * fracs[0]
            a_try[jdx2] = s_tri * fracs[1]
            a_try[kdx2] = s_tri * fracs[2]
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
    return best_a, best_sc


def _balancing_iteration(a, n_rounds=5000, deadline=None):
    """Explicit balancing: try to equalize convolution values."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.005
    stall = 0

    for i in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_mean = c.mean()
        # For each lag k above mean, push down; for each below, push up
        # Gradient of (c[k] - c_mean)^2 w.r.t. a[i]:
        # d(c[k])/da[i] = a[k-i] (if valid)
        # So the "pressure" on a[i] from lag k is a[k-i]
        # Build pressure vector
        excess = np.maximum(c - c_mean, 0)
        deficit = np.maximum(c_mean - c, 0)
        # Total pressure on a[i] = sum_k excess[k] * a[k-i] - sum_k deficit[k] * a[k-i]
        excess_rev = excess[::-1]
        deficit_rev = deficit[::-1]
        press_excess = _full_convolve(excess_rev, a)[n-1:2*n-1]
        press_deficit = _full_convolve(deficit_rev, a)[n-1:2*n-1]
        pressure = press_excess - press_deficit
        pressure = pressure - pressure.mean()
        p_norm = np.linalg.norm(pressure)
        if p_norm < 1e-15:
            break
        a_new = a - lr * pressure / p_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.05)
            stall = 0
        else:
            stall += 1
            lr *= 0.95
            if lr < 1e-10 or stall > 100:
                break
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining LP best-response, L-BFGS, peak shaving,
    balancing, multi-length exploration, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")
            return True
        return False

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        try_update(a0, "init")

    # === Phase 1: Fine L-BFGS (most aggressive beta schedule) ===
    print(f"\n=== Phase 1: Fine L-BFGS on n={len(best_seq)} ===")
    t_phase = time.monotonic()
    a_p1 = _lbfgs_schedule(best_seq, deadline - 700, min_time_remaining=600.0)
    try_update(a_p1, "lbfgs_fine")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: LP-based best response ===
    print(f"\n=== Phase 2: LP best response ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 500:
        a_lp, sc_lp = _lp_iterations(best_seq, deadline - 500, n_lp_iters=40)
        try_update(a_lp, "lp_best_resp")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Balancing iterations ===
    print(f"\n=== Phase 3: Balancing ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400:
        a_bal, sc_bal = _balancing_iteration(best_seq, n_rounds=20000,
                                             deadline=deadline - 400)
        try_update(a_bal, "balancing")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Momentum gradient descent ===
    print(f"\n=== Phase 4: Momentum gradient ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 350:
        a_mom, sc_mom = _gradient_momentum(best_seq, deadline - 350,
                                           max_steps=50000, lr_init=0.003)
        try_update(a_mom, "momentum")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    t_phase = time.monotonic()
    start_n = len(best_seq)
    target_lengths = []
    if start_n < 16384:
        target_lengths.append(16384)
    if start_n < 8192:
        target_lengths.append(8192)

    for target_n in target_lengths:
        if time.monotonic() > deadline - 200:
            break
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"    Interpolated: score={sc_interp:.10f}")
        try_update(a_interp, f"interp_n{target_n}")

        a_opt = _lbfgs_schedule(a_interp, deadline - 150, min_time_remaining=120.0)
        try_update(a_opt, f"n{target_n}_lbfgs")

        if time.monotonic() < deadline - 100:
            a_ps, _ = _peak_shaving(a_opt, deadline - 100, max_iters=15000)
            try_update(a_ps, f"n{target_n}_ps")

    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Peak shaving on best ===
    print(f"\n=== Phase 6: Intensive peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 200:
        a_ps, sc_ps = _peak_shaving(best_seq, deadline - 200, max_iters=100000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 150:
        restart_configs = [
            ("gauss_s1", "gauss", 0.01),
            ("gauss_s3", "gauss", 0.03),
            ("gauss_s8", "gauss", 0.08),
            ("fourier", "fourier", 0.0),
            ("block", "block", 0.0),
            ("sparse", "sparse", 0.0),
        ]
        for name, mode, scale in restart_configs:
            if time.monotonic() > deadline - 100:
                break
            p_start = _perturb_for_restart(best_seq, rng, mode, scale)
            a_r = _lbfgs_schedule(p_start, deadline - 80, min_time_remaining=60.0)
            try_update(a_r, f"restart_{name}")
            if time.monotonic() < deadline - 60:
                a_r2, _ = _peak_shaving(a_r, deadline - 60, max_iters=10000)
                try_update(a_r2, f"restart_{name}_ps")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Final L-BFGS + LP polish ===
    print(f"\n=== Phase 8: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Ultra-fine L-BFGS
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        ultra_betas = mb * np.array([5e-8, 2e-8, 1e-8, 5e-9, 2e-9])
        for i, beta in enumerate(ultra_betas):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - 20 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(15000, max(500, int(remaining / 0.004)))
            a_ultra, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_ultra, f"ultra_{i}")

        # Quick LP
        if time.monotonic() < deadline - 15:
            a_lp2, _ = _lp_iterations(best_seq, deadline - 10, n_lp_iters=10)
            try_update(a_lp2, "final_lp")

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Coordinate descent ===
    print(f"\n=== Phase 9: Coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=30000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 10: One more full cycle ===
    print(f"\n=== Phase 10: Final cycle ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_f1 = _lbfgs_schedule(best_seq, deadline - 3, min_time_remaining=2.0)
        try_update(a_f1, "final_lbfgs")
        if time.monotonic() < deadline - 2:
            a_f2, _ = _peak_shaving(a_f1, deadline - 2, max_iters=5000)
            try_update(a_f2, "final_ps")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
