def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _fourier_sim_anneal(a, deadline, n_restarts=5, t_init=0.1, t_min=0.01, rng=None):

                    best_a = a_try.copy()
            
            temp *= 0.97
        temp = t_init * 0.5 ** restart
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "linear_tilt":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "inverse_tilt":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (1 - t)
        return a / a.sum()
    elif kind == "double_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(7)
        a = np.abs(np.fft.irfft(rng.standard_normal(n//2 + 1) * np.exp(-4 * np.linspace(0,1,n//2+1)**2)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _multi_n_explore(parent, deadline, min_remaining, rng):
    """Explore different sequence lengths."""
    best_a = parent.copy()
    best_sc = _score(parent)
    
    # FFT-friendly lengths
    n_candidates = [6144, 7680, 8192, 9216, 10240, 11520, 12288, 13824]
    n_parent = len(parent)
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates[:4]:
        if time.monotonic() > deadline - min_remaining:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 30:
            break

        # Interpolate parent to new length
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, parent)
        a_init = np.maximum(a_init, 0)
        s = a_init.sum()
        if s < 1e-10:
            a_init = _make_profile(n_try, "tapered")
        else:
            a_init = a_init / s

        # Quick optimization
        min_rem_local = max(5, remaining * 0.3)
        a_opt = _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=min_rem_local)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"  Multi-n: n={n_try}, score={sc_opt:.10f}")

    return best_a, best_sc


def _coordinate_polish_v2(a, deadline, n_rounds=80):
    """Improved coordinate descent using analytical optimal step."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find coordinates with highest "peak contribution"
        peak_idx = int(np.argmax(c))
        # Contribution to peak: a[i] * a[peak - i]
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        
        # Also check second peak
        c_copy = c.copy()
        c_copy[peak_idx] = -np.inf
        peak_idx2 = int(np.argmax(c_copy))
        
        candidates = set()
        for pk in [peak_idx, peak_idx2]:
            lo2 = max(0, pk - (n - 1))
            hi2 = min(n - 1, pk)
            if lo2 <= hi2:
                contribs = a[lo2:hi2+1] * a[pk-hi2:pk-lo2+1][::-1]
                top_n = min(5, len(contribs))
                for idx in np.argsort(contribs)[::-1][:top_n]:
                    candidates.add(lo2 + idx)
        
        # Add some random candidates
        for _ in range(5):
            candidates.add(rng.integers(0, n))
        
        improved = False
        for i in list(candidates):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try binary search for optimal multiplier
            lo_f, hi_f = 0.0, 2.0
            c_before = _full_convolve(a, a).max()
            
            best_factor = 1.0
            best_c = c_before
            for _ in range(8):
                mid = (lo_f + hi_f) / 2
                a[i] = old_val * mid
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    break
                a_norm = a / s
                c_try = float(_full_convolve(a_norm, a_norm).max())
                if c_try < best_c:
                    best_c = c_try
                    best_factor = mid
                    lo_f = mid
                else:
                    hi_f = mid
                a[i] = old_val
            
            if best_factor != 1.0:
                a[i] = old_val * best_factor
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    improved = True

        if improved:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer: aggressive L-BFGS with fine beta schedule,
    KKT-aware pressure equalization, subgradient polish, peak shaving,
    coordinate descent, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        _update(parent, parent_sc)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        parent_sc = _score(parent)
        print(f"No parent, using tapered n={n_parent}, score={parent_sc:.10f}")
        _update(parent, parent_sc)

    # === PHASE 1: Aggressive L-BFGS on parent (main optimization) ===
    print(f"\n[Phase 1] Aggressive L-BFGS on n={n_parent}")
    phase1_deadline = deadline - (budget_s * 0.35)
    a1 = _run_lbfgs_aggressive(parent, phase1_deadline, min_time_remaining=10.0)
    sc1 = _score(a1)
    _update(a1, sc1)
    print(f"  Phase 1 done: score={sc1:.10f}")

    # === PHASE 2: Pressure equalization + re-optimize ===
    print(f"\n[Phase 2] Pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a2, sc2_norm = _pressure_equalize_v2(a1 if sc1 < best_score else best_seq,
                                              min(deadline - 250, time.monotonic() + 120),
                                              n_iters=200, gamma=0.35)
        sc2 = _score(a2)
        _update(a2, sc2)
        
        if sc2 < best_score:
            a2b = _run_lbfgs_aggressive(a2, deadline - 200, min_time_remaining=10.0)
            _update(a2b, _score(a2b))
    print(f"  Phase 2 done: best={best_score:.10f}")

    # === PHASE 3: Subgradient polish ===
    print(f"\n[Phase 3] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a3, sc3 = _subgrad_polish(best_seq, min(deadline - 150, time.monotonic() + 80),
                                   n_max_steps=3000, lr_init=0.025)
        _update(a3, sc3)
    print(f"  Phase 3 done: best={best_score:.10f}")

    # === PHASE 4: Peak shaving ===
    print(f"\n[Phase 4] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a4, sc4_norm = _peak_shave(best_seq, min(deadline - 100, time.monotonic() + 60), n_rounds=40)
        sc4 = _score(a4)
        _update(a4, sc4)
        if sc4 < best_score + 0.001:
            a4b = _run_lbfgs_aggressive(a4, deadline - 80, min_time_remaining=5.0)
            _update(a4b, _score(a4b))
    print(f"  Phase 4 done: best={best_score:.10f}")

    # === PHASE 5: Coordinate descent ===
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a5, sc5_norm = _coordinate_polish_v2(best_seq, min(deadline - 60, time.monotonic() + 50), n_rounds=60)
        sc5 = _score(a5)
        _update(a5, sc5)
    print(f"  Phase 5 done: best={best_score:.10f}")

    # === PHASE 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a6, sc6 = _multi_n_explore(best_seq, deadline, 50, rng)
        _update(a6, sc6)
    print(f"  Phase 6 done: best={best_score:.10f}")

    # === PHASE 7: Fourier simulated annealing + re-optimize ===
    print(f"\n[Phase 7] Fourier annealing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        # Try a few diverse initializations
        for prof_kind in ["raised", "linear_tilt", "double_hump"]:
            if time.monotonic() > deadline - 40:
                break
            a_blend = 0.7 * best_seq + 0.3 * _make_profile(n_best, prof_kind)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            
            a7 = _run_lbfgs_aggressive(a_blend, min(deadline - 30, time.monotonic() + 60),
                                        min_time_remaining=5.0)
            sc7 = _score(a7)
            _update(a7, sc7)
    print(f"  Phase 7 done: best={best_score:.10f}")

    # === PHASE 8: Final aggressive L-BFGS + polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a8 = _run_lbfgs_aggressive(best_seq, deadline - 3, min_time_remaining=2.0)
        _update(a8, _score(a8))
        
        if time.monotonic() < deadline - 2:
            a8b, sc8b = _subgrad_polish(best_seq, deadline - 1, n_max_steps=500, lr_init=0.01)
            _update(a8b, sc8b)

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
