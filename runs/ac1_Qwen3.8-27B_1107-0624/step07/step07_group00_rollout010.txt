import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _combined_score_norm(a, var_weight=0.05):
    """Combined objective: peak + variance penalty"""
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    if c_mean < 1e-30:
        return c_max
    c_var = float(((c - c_mean)**2).mean())
    return c_max + var_weight * c_var / (c_mean * c_mean)

def _softmax_stable(z):

    e = np.exp(z)
    s = e.sum()

    return e / s

"""
Improved multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
- Combined peak+variance objective for smoother landscape
- Spectral equalization to flatten power spectrum periodically
- Adaptive temperature with geometric decay and smart restarts
- Better coordinate descent using analytical optimal steps
- Multi-scale exploration (n=4096, 8192, 16384)
- Stochastic block moves for final polishing

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _combined_score_norm(a, var_w=0.03):
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    if c_mean < 1e-30:
        return c_max
    c_var = float(np.mean((c - c_mean) ** 2))
    return c_max + var_w * c_var / (c_mean * c_mean)


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_peak(a, c, T):
    """Compute smoothed gradient of peak correlation."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _gradient_var(a, c):
    """Compute gradient of variance of correlation (smooth, always defined)."""
    n = len(a)
    c_mean = c.mean()
    w = (c - c_mean) / (len(c) * c_mean ** 2 + 1e-30)
    w_rev = w[::-1]
    conv_full = _fft_convolve(a, w_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _combined_gradient(a, c, T, var_w=0.03):
    """Combined gradient of peak + variance objective."""
    g_peak = _gradient_peak(a, c, T)
    g_var = _gradient_var(a, c)
    return g_peak + var_w * g_var


def _spectral_equalize(a, alpha=0.6):
    """Flatten the power spectrum partially."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    S = np.abs(A) ** 2 + 1e-30
    S_mean = S.mean()
    S_new = S_mean + alpha * (S - S_mean)
    S_new = np.maximum(S_new, 1e-30)
    S_new = S_new * (S.sum() / S_new.sum())
    A_new = A * np.sqrt(S_new / S)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, var_w=0.03, eval_interval=200,
                   clip_norm=0.8, spec_interval=5000, spec_alpha=0.5,
                   patience_limit=8, restart_sigma=0.03):
    """Adam optimizer in log-space with combined objective and spectral equalization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _combined_gradient(a, c, T, var_w)
        if not np.all(np.isfinite(g)):
            g = _gradient_peak(a, c, max(T, c.max() * 0.001))

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update *= clip_norm / update_norm
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Periodic spectral equalization
        if spec_interval > 0 and t % spec_interval == 0:
            a_eq = _softmax_stable(z)
            a_eq_new = _spectral_equalize(a_eq, spec_alpha)
            z = np.log(np.maximum(a_eq_new, 1e-15))
            z = z - z.mean()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)

        # Evaluate and track best
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            # Adaptive restart
            if patience >= patience_limit:
                if current_lr_mult > 0.1:
                    current_lr_mult *= 0.5
                else:
                    # Perturb and restart
                    rng = np.random.default_rng(t)
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z + restart_sigma * rng.standard_normal(n)
                    z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fast_cd_polish(a, deadline, max_rounds=200, top_k=8):
    """Fast coordinate descent: only modify coordinates affecting peak lags."""
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for k in peak_indices:
            # Find which j values affect c[k]
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                other = k - j
                if other < 0 or other >= n:
                    continue
                # Try scaling a[j]
                current_val = best_a[j]
                if current_val < 1e-15:
                    continue
                # Optimal scale: we want to reduce c[k] which contains a[j]*a[other]
                # Reducing a[j] by factor s reduces c[k] by roughly (1-s)*a[j]*a[other]
                # but increases other c values. Try a few scales.
                best_local = best_score
                best_scale = 1.0
                for scale in [0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                    a_cand = best_a.copy()
                    a_cand[j] = current_val * scale
                    s_sum = a_cand.sum()
                    if s_sum < 1e-15:
                        continue
                    a_cand /= s_sum
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc_cand = 2.0 * n * float(c_cand.max())
                    if sc_cand < best_local * (1.0 - 1e-12):
                        best_local = sc_cand
                        best_scale = scale
                if best_scale != 1.0:
                    best_a[j] = current_val * best_scale
                    best_a /= best_a.sum()
                    best_score = best_local
                    improved = True

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _ternary_cd_polish(a, deadline, max_rounds=100, top_k=6):
    """Coordinate descent with ternary search for optimal scale per coordinate."""
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        c = _fft_convolve(best_a, best_a)
        peak_indices = np.argsort(c)[::-1][:top_k]
        improved = False

        for k in peak_indices:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                other = k - j
                if other < 0 or other >= n:
                    continue
                current_val = best_a[j]
                if current_val < 1e-12:
                    continue
                # Ternary search for optimal scale in [0.3, 2.0]
                lo, hi = 0.3, 2.0
                best_local = best_score
                best_scale = 1.0
                for _ in range(12):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    a_cand1 = best_a.copy()
                    a_cand1[j] = current_val * m1
                    a_cand1 /= a_cand1.sum()
                    sc1 = 2.0 * n * float(_fft_convolve(a_cand1, a_cand1).max())

                    a_cand2 = best_a.copy()
                    a_cand2[j] = current_val * m2
                    a_cand2 /= a_cand2.sum()
                    sc2 = 2.0 * n * float(_fft_convolve(a_cand2, a_cand2).max())

                    if sc1 < sc2:
                        hi = m2
                    else:
                        lo = m1
                    # Track best
                    for sc, sc_val in [(sc1, m1), (sc2, m2)]:
                        if sc < best_local:
                            best_local = sc
                            best_scale = sc_val

                if best_scale != 1.0 and best_local < best_score * (1 - 1e-10):
                    best_a[j] = current_val * best_scale
                    best_a /= best_a.sum()
                    best_score = best_local
                    improved = True

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _block_stochastic_search(a, deadline, max_iters=5000):
    """Stochastic search: randomly select blocks and rescale."""
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)
    rng = np.random.default_rng(12345)
    T = 0.002 * float(_fft_convolve(best_a, best_a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        T *= 0.9995
        # Pick a random block
        block_size = rng.integers(max(4, n // 64), max(8, n // 8))
        start = rng.integers(0, n - block_size)
        idx = np.arange(start, start + block_size)

        # Random perturbation of the block
        a_cand = best_a.copy()
        perturbation = 1.0 + 0.1 * rng.standard_normal(block_size)
        a_cand[idx] *= perturbation
        a_cand = np.maximum(a_cand, 0.0)
        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand /= s
        sc_cand = _score_norm(a_cand)

        # Metropolis accept
        delta = sc_cand - best_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            best_a = a_cand
            if sc_cand < best_score:
                best_score = sc_cand

    return best_a, _score_norm(best_a)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Adam log-space on parent with combined objective (0-180s)
    Phase 2: Spectral equalization + refinement (180-350s)
    Phase 3: Multi-scale n=4096 (350-650s)
    Phase 4: Multi-scale n=8192 (650-1050s)
    Phase 5: Try n=16384 (1050-1350s)
    Phase 6: Intensive polish on best (1350-1600s)
    Phase 7: CD + stochastic (1600-1750s)
    Phase 8: Final micro-optimization (1750-1790s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n_parent}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Adam on parent (0-180s)
    phase1_end = min(deadline, t_start + 180.0)
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    a_opt, sc = _adam_logspace(a, phase1_end,
                               T_start=0.004 * max_c, T_end=0.00003 * max_c,
                               lr_start=0.035, lr_end=0.002,
                               max_iter=300000, var_w=0.03,
                               eval_interval=150, clip_norm=0.7,
                               spec_interval=8000, spec_alpha=0.5,
                               patience_limit=10)
    full_sc = _score(a_opt)
    print(f"Phase 1: score={full_sc:.10f}, n={len(a_opt)}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 2: Spectral equalization cycle (180-350s)
    phase2_end = min(deadline, t_start + 350.0)
    if time.monotonic() < phase2_end:
        a_work = best_a.copy()
        for eq_iter in range(50):
            if time.monotonic() > phase2_end - 5:
                break
            # Spectral equalization
            a_eq = _spectral_equalize(a_work, alpha=0.4 + 0.3 * eq_iter / 50)
            # Short Adam refinement
            c_eq = _fft_convolve(a_eq, a_eq)
            max_eq = float(c_eq.max())
            sub_dl = min(phase2_end - 3, time.monotonic() + 25.0)
            a_ref, sc_ref = _adam_logspace(a_eq, sub_dl,
                                           T_start=0.002 * max_eq, T_end=0.00002 * max_eq,
                                           lr_start=0.02, lr_end=0.001,
                                           max_iter=50000, var_w=0.02,
                                           eval_interval=200, clip_norm=0.5,
                                           spec_interval=0, patience_limit=6)
            sc_full = _score(a_ref)
            if sc_full < best_score:
                best_score = sc_full
                best_a = a_ref.copy()
                a_work = a_ref.copy()
                print(f"  Phase 2 eq[{eq_iter}]: score={sc_full:.10f}")
            else:
                a_work = a_ref.copy()  # still use it as next starting point

        print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: n=4096 (350-650s)
    phase3_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase3_end and len(best_a) != 4096:
        a_4k = _interp_to_length(best_a, 4096)
        c_4k = _fft_convolve(a_4k, a_4k)
        max_4k = float(c_4k.max())
        mid3 = min(phase3_end - 60, t_start + 550.0)
        a_4k_opt, sc_4k = _adam_logspace(a_4k, mid3,
                                         T_start=0.003 * max_4k, T_end=0.00002 * max_4k,
                                         lr_start=0.03, lr_end=0.0015,
                                         max_iter=200000, var_w=0.03,
                                         eval_interval=200, clip_norm=0.6,
                                         spec_interval=10000, spec_alpha=0.5,
                                         patience_limit=8)
        sc_4k_full = _score(a_4k_opt)
        print(f"Phase 3 n=4096: score={sc_4k_full:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_4k_full < best_score:
            best_score = sc_4k_full
            best_a = a_4k_opt.copy()

        # CD on 4k result
        if time.monotonic() < phase3_end - 20:
            a_4k_cd, _ = _fast_cd_polish(best_a, phase3_end - 5, max_rounds=50, top_k=8)
            sc_cd = _score(a_4k_cd)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_4k_cd.copy()
                print(f"  Phase 3 CD: score={sc_cd:.10f}")

    # Phase 4: n=8192 (650-1050s)
    phase4_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase4_end and len(best_a) != 8192:
        a_8k = _interp_to_length(best_a, 8192)
        c_8k = _fft_convolve(a_8k, a_8k)
        max_8k = float(c_8k.max())
        mid4 = min(phase4_end - 80, t_start + 950.0)
        a_8k_opt, sc_8k = _adam_logspace(a_8k, mid4,
                                         T_start=0.0025 * max_8k, T_end=0.000015 * max_8k,
                                         lr_start=0.025, lr_end=0.001,
                                         max_iter=200000, var_w=0.03,
                                         eval_interval=200, clip_norm=0.6,
                                         spec_interval=8000, spec_alpha=0.4,
                                         patience_limit=8)
        sc_8k_full = _score(a_8k_opt)
        print(f"Phase 4 n=8192: score={sc_8k_full:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_8k_full < best_score:
            best_score = sc_8k_full
            best_a = a_8k_opt.copy()

        # CD on 8k
        if time.monotonic() < phase4_end - 30:
            a_8k_cd, _ = _fast_cd_polish(best_a, phase4_end - 5, max_rounds=40, top_k=10)
            sc_cd8 = _score(a_8k_cd)
            if sc_cd8 < best_score:
                best_score = sc_cd8
                best_a = a_8k_cd.copy()
                print(f"  Phase 4 CD: score={sc_cd8:.10f}")

    # Also try 8192 from the parent directly if we haven't
    if len(best_a) != 8192 and time.monotonic() < phase4_end:
        a_8k_parent = _interp_to_length(a, 8192)
        c_8kp = _fft_convolve(a_8k_parent, a_8k_parent)
        max_8kp = float(c_8kp.max())
        a_8kp_opt, _ = _adam_logspace(a_8k_parent, min(deadline, time.monotonic() + 100),
                                       T_start=0.003 * max_8kp, T_end=0.00002 * max_8kp,
                                       lr_start=0.03, lr_end=0.001,
                                       max_iter=80000, var_w=0.03,
                                       eval_interval=200, clip_norm=0.6,
                                       spec_interval=0, patience_limit=6)
        sc_8kp = _score(a_8kp_opt)
        if sc_8kp < best_score:
            best_score = sc_8kp
            best_a = a_8kp_opt.copy()
            print(f"  Phase 4 from parent n=8192: score={sc_8kp:.10f}")

    # Phase 5: Try n=16384 (1050-1350s)
    phase5_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase5_end and 16384 != len(best_a):
        a_16k = _interp_to_length(best_a, 16384)
        c_16k = _fft_convolve(a_16k, a_16k)
        max_16k = float(c_16k.max())
        a_16k_opt, sc_16k = _adam_logspace(a_16k, min(phase5_end - 10, time.monotonic() + 250),
                                            T_start=0.002 * max_16k, T_end=0.00001 * max_16k,
                                            lr_start=0.02, lr_end=0.0008,
                                            max_iter=100000, var_w=0.03,
                                            eval_interval=300, clip_norm=0.5,
                                            spec_interval=10000, spec_alpha=0.4,
                                            patience_limit=6)
        sc_16k_full = _score(a_16k_opt)
        print(f"Phase 5 n=16384: score={sc_16k_full:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_16k_full < best_score:
            best_score = sc_16k_full
            best_a = a_16k_opt.copy()
    else:
        print(f"Phase 5 skipped, t={time.monotonic()-t_start:.0f}s")

    # Phase 6: Intensive polish on best (1350-1600s)
    phase6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase6_end:
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        # Very low temperature Adam
        mid6 = min(phase6_end - 30, t_start + 1550.0)
        a_pol, sc_pol = _adam_logspace(best_a, mid6,
                                       T_start=0.0005 * max_cb, T_end=0.000003 * max_cb,
                                       lr_start=0.01, lr_end=0.0004,
                                       max_iter=200000, var_w=0.02,
                                       eval_interval=100, clip_norm=0.4,
                                       spec_interval=0, patience_limit=12,
                                       restart_sigma=0.015)
        sc_pol_full = _score(a_pol)
        print(f"Phase 6 polish: score={sc_pol_full:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_a = a_pol.copy()

    # Phase 7: CD + stochastic (1600-1750s)
    phase7_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase7_end:
        # Ternary CD
        a_cd, sc_cd = _ternary_cd_polish(best_a, min(phase7_end - 60, time.monotonic() + 120),
                                         max_rounds=30, top_k=8)
        sc_cd_full = _score(a_cd)
        print(f"Phase 7 ternary CD: score={sc_cd_full:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_a = a_cd.copy()

        # Fast CD
        if time.monotonic() < phase7_end - 30:
            a_fc, sc_fc = _fast_cd_polish(best_a, min(phase7_end - 10, time.monotonic() + 60),
                                          max_rounds=30, top_k=10)
            sc_fc_full = _score(a_fc)
            if sc_fc_full < best_score:
                best_score = sc_fc_full
                best_a = a_fc.copy()
                print(f"  Phase 7 fast CD: score={sc_fc_full:.10f}")

        # Stochastic block search
        if time.monotonic() < phase7_end - 5:
            a_stoch, sc_stoch = _block_stochastic_search(
                best_a, min(phase7_end, time.monotonic() + 40), max_iters=3000)
            sc_stoch_full = _score(a_stoch)
            if sc_stoch_full < best_score:
                best_score = sc_stoch_full
                best_a = a_stoch.copy()
                print(f"  Phase 7 stochastic: score={sc_stoch_full:.10f}")

    # Phase 8: Final micro-optimization (1750-deadline)
    if time.monotonic() < deadline - 15:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _adam_logspace(best_a, deadline - 12,
                                     T_start=0.0002 * max_cf, T_end=0.000001 * max_cf,
                                     lr_start=0.004, lr_end=0.0002,
                                     max_iter=100000, var_w=0.01,
                                     eval_interval=80, clip_norm=0.3,
                                     spec_interval=0, patience_limit=15,
                                     restart_sigma=0.01)
        sc_final = _score(a_final)
        print(f"Phase 8 final: score={sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

        # One last CD
        if time.monotonic() < deadline - 5:
            a_last, _ = _fast_cd_polish(best_a, deadline - 3, max_rounds=15, top_k=12)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_a = a_last.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
