    """Alternative: compute H @ x where H[j,k] = w[j+k], using direct FFT.
    H is (n x n), w has length 2n-1.
    (Hx)[j] = sum_k w[j+k]*x[k]
    """
    # Use the property: H = T * C(2n-1) * T^T where T is a (n x (2n-1)) matrix
    # But simpler: just compute via correlation
    # (Hx)[j] = sum_{k=0}^{n-1} w[j+k] * x[k]
    # This is a cross-correlation between w (length 2n-1) and x (length n).
    nW = len(w)  # should be 2n-1
    N = _next_pow2(nW + n - 1)
    Fw = np.fft.rfft(w, N)
    Fx = np.fft.rfft(x, N)
    # cross-correlation: ifft(conj(Fw) * Fx) gives sum_k w[k+j]*x[k]... 
    # Actually let me just use the conv approach:
    x_rev = x[::-1]
    fx = np.fft.rfft(x_rev, N)
    res = np.fft.irfft(Fw * fx, N)
    # res[m] = sum_i w[i] * x_rev[m-i] = sum_i w[i] * x[n-1-m+i]
    # We want res[j + n - 1] = sum_i w[i] * x[n-1-j-n+1+i] = sum_i w[i]*x[i-j]
    # = sum_{k=0}^{n-1} w[j+k]*x[k] ✓
    return res[n-1:2*n-1].copy()


def _lp_solve_hankel(w, rhs, n, tol=1e-10, max_iter=500):
    """Solve H @ x = rhs where H[j,k] = 2*w[j+k] (the Hessian structure).
    Uses conjugate gradient with FFT-based mat-vec.
    """
    def matvec(v):
        return 2.0 * _hankel_matvec(w, v, n)
    
    # Preconditioner: diagonal of H is 2*w[j+j] = 2*w[2j]
    diag = 2.0 * w[0:n:2]  # w[0], w[2], w[4], ..., w[2n-2]
    diag = np.maximum(diag, 1e-15)
    
    x0 = np.zeros(n)
    try:
        x, info = cg(matvec, rhs, x0=x0, tol=tol, maxiter=max_iter,
                     rtol=tol, atol=1e-15)
        if info > 0:
            # CG didn't converge, try with more iterations or different approach
            x, info = cg(matvec, rhs, x0=x0, tol=1e-8, maxiter=2000,
                         rtol=1e-8)
    except Exception:
        x = np.zeros(n)
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = _grad_conv_w(a, w, n)
    return 4.0 * n * a * g - 2.0 * n * (a.sum()) * 2.0 * a * (w @ c) / (a.sum())
    # Simplified: since sum(a)=1 after softmax, the gradient is just 4*n*a*g_normalized
    # Actually let me redo this properly.
    # f(z) = 2*n*(c_max + beta*log(sum exp((c-c_max)/beta)))
    # df/dz_j = 2*n * sum_k w[k] * dc[k]/dz_j
    # where w[k] = exp((c[k]-c_max)/beta) / sum exp((c-c_max)/beta)
    # dc[k]/dz_j = dc[k]/da_j * da_j/dz_j
    # da_j/dz_j = a_j*(1-delta_jk)... for softmax: da_j/dz_k = a_j*(delta_jk - a_k)
    # So dc[k]/dz_m = sum_j (dc[k]/da_j) * a_j*(delta_jm - a_m)
    # = (dc[k]/da_m)*a_m - a_m * sum_j (dc[k]/da_j)*a_j
    
    # This is getting complex. Let me just use the simpler formula:
    # Since softmax(a) = softmax(z), and we're just parameterizing a = softmax(z),
    # the gradient w.r.t. z is: grad_z = a_elementwise * (grad_a - a*sum(a*grad_a))
    # where grad_a is the gradient w.r.t. a of the objective.
    # Objective (simplified, sum(a)=1): 2*n*max(c) ≈ 2*n*(c_max + beta*log(sum exp((c-c_max)/beta)))
    # grad_a = 2*n * sum_k w[k] * dc[k]/da
    # = 2*n * _grad_conv_w(a, w, n)
    # = 2*n * 2 * (conv(w, a_rev))[n-1:2n-1]
    # Then grad_z = a * (grad_a - a * sum(a * grad_a))
    
    # Let me redo:
    g_a = 2.0 * n * _grad_conv_w(a, w, n)
    # Project onto tangent space of simplex (sum a = 1):
    mean_term = np.dot(a, g_a)
    g_z = a * (g_a - mean_term)
    return g_z


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=200.0, beta_mults=None):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    if beta_mults is None:
        beta_mults = np.array([0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                                0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
                                5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8])
    betas = max_c * beta_mults
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        est_per_iter = 0.003 * (n / 4096.0) ** 1.3
        max_iter = min(20000, max(300, int(remaining / (est_per_iter * 30))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.3e}: score={sc:.10f} (iter={niter})")

    return current_a


def _active_set_newton(a_init, deadline, max_iters=200, tol=1e-12):
    """Newton-based active set solver for the KKT conditions.
    
    At the optimum with active set S, the KKT conditions are:
    - g[j] = sum_{s in S} w[s] * dc[s]/da[j] = 2*lambda for j in F (free set)
    - g[j] >= 2*lambda for j not in F
    - sum(a) = 1, a >= 0
    
    We solve this iteratively using Newton's method on the linearized system.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Identify active set (top fraction of c values)
        n_active = max(10, int(0.01 * len(c)))
        sorted_idx = np.argsort(c)[::-1]
        active_indices = sorted_idx[:n_active]
        
        # Use a tolerance-based active set
        tol_active = c_max - 1e-4 * c_max
        active_mask = c >= tol_active
        if active_mask.sum() < 5:
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[active_indices[:5]] = True
        
        w = np.zeros(len(c))
        w[active_mask] = 1.0 / active_mask.sum()
        
        # Compute gradient
        g = _grad_conv_w(a, w, n)  # this is 2 * sum_s w[s] * a[s-j]
        
        # KKT: on the free set (a[j] > 0), g[j] should be constant (= 2*lambda)
        free = a > 1e-10
        if free.sum() < 3:
            break
        
        # Newton step: we want to find delta such that g(a+delta) ≈ constant on free set
        # Linearize: g(a+delta) ≈ g + H @ delta
        # H is the Hessian of sum_s w[s]*c[s], which is a Hankel matrix
        # H[j,k] = 2*w[j+k]
        
        # Target: g + H@delta = lambda * 1 (on free set)
        # Choose lambda = mean of g on free set
        lambda_target = np.mean(g[free])
        rhs = lambda_target * np.ones(n) - g  # we want H@delta = rhs (on free set)
        
        # Set rhs = 0 on constrained set (a[j] = 0) to not push them negative
        rhs[~free] = 0
        
        # Normalize rhs so that sum(delta) = 0 (maintain sum(a) = 1)
        rhs = rhs - np.mean(rhs)
        
        # Solve H @ delta = rhs using CG with Hankel mat-vec
        # H[j,k] = 2*w[j+k]
        w_full = w  # length 2n-1
        
        def h_matvec(v):
            return 2.0 * _hankel_matvec(w_full, v, n)
        
        # Preconditioner: use diagonal
        diag_H = 2.0 * w_full[0:n:2]  # H[j,j] = 2*w[2j]
        diag_H = np.maximum(diag_H, 1e-15)
        
        try:
            delta, info = cg(h_matvec, rhs, x0=np.zeros(n),
                             tol=1e-10, maxiter=200, rtol=1e-10)
        except Exception:
            delta = np.zeros(n)
        
        if np.linalg.norm(delta) < 1e-15:
            break
        
        # Line search
        best_step = 0.0
        best_a_try = a
        best_sc_try = _score(a)
        
        for step_frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_try = a + step_frac * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-15:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_step = step_frac
        
        if best_step > 0:
            a = best_a_try
            if best_sc_try < best_sc - 1e-15:
                best_sc = best_sc_try
                best_a = a.copy()
                print(f"    AS Newton iter {it}: score={best_sc:.10f}")
        else:
            # Try smaller active set
            n_active = max(5, n_active // 2)
            if n_active < 5:
                break
    
    return best_a, best_sc


def _power_sum_obj(z, p, n):
    """Power sum approximation: (sum c^p)^(1/p) as smooth max."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0)
    cp = c ** p
    s_cp = cp.sum()
    if s_cp < 1e-300:
        return 2.0 * n * c.max()
    max_approx = s_cp ** (1.0 / p)
    return 2.0 * n * max_approx


def _coordinate_descent(a_init, deadline, max_rounds=20000):
    """Randomized coordinate descent with pair updates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Pick two random indices
        idx = rnd % n
        jdx = (rnd * 7 + 3) % n
        if idx == jdx:
            jdx = (jdx + 1) % n
        
        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue
        
        # Try a few fractions
        for frac in [0.5, 0.3, 0.7, 0.1, 0.9, 0.4, 0.6]:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a[idx] = new_i
            a[jdx] = new_j
            sc_try = _score(a)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a.copy()
                break
            else:
                a[idx] = s_pair * (1.0 - frac) if frac != 0.5 else s_pair * 0.5
                # Restore
                a[idx] = a[idx]  # already set
                a[jdx] = s_pair - a[idx]
    
    return best_a, best_sc


def _perturb_restart(a, rng, scale=0.02):
    """Perturb a sequence for restart diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS log-sum-exp continuation,
    active-set Newton solving with Hankel-structured Hessian, and
    coordinate descent polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s (improvement={prev-sc:.2e})")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(8192) / 8192
        print(f"No parent. Using uniform n=8192, score={_score(a0):.10f}")
        try_update(a0, "init")

    n = len(best_seq)
    print(f"Starting optimization for n={n}")

    # === Phase 1: L-BFGS coarse (beta from 0.3*M to 0.01*M) ===
    print(f"\n=== Phase 1: L-BFGS coarse ===")
    t_phase = time.monotonic()
    beta_mults_coarse = np.array([0.3, 0.15, 0.08, 0.04, 0.02, 0.01])
    a_p1 = _lbfgs_schedule(best_seq, deadline - 900, min_time_remaining=800.0,
                           beta_mults=beta_mults_coarse)
    try_update(a_p1, "lbfgs_coarse")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: L-BFGS medium (beta from 0.005*M to 1e-4*M) ===
    print(f"\n=== Phase 2: L-BFGS medium ===")
    t_phase = time.monotonic()
    beta_mults_med = np.array([0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5])
    a_p2 = _lbfgs_schedule(best_seq, deadline - 600, min_time_remaining=500.0,
                           beta_mults=beta_mults_med)
    try_update(a_p2, "lbfgs_med")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p2):.10f}")

    # === Phase 3: Active set Newton ===
    print(f"\n=== Phase 3: Active set Newton ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_as, sc_as = _active_set_newton(best_seq, deadline - 300, max_iters=50)
        try_update(a_as, "active_set_newton")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: L-BFGS fine (beta from 1e-5*M to 1e-8*M) ===
    print(f"\n=== Phase 4: L-BFGS fine ===")
    t_phase = time.monotonic()
    beta_mults_fine = np.array([1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8])
    a_p4 = _lbfgs_schedule(best_seq, deadline - 200, min_time_remaining=150.0,
                           beta_mults=beta_mults_fine)
    try_update(a_p4, "lbfgs_fine")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p4):.10f}")

    # === Phase 5: Active set Newton round 2 ===
    print(f"\n=== Phase 5: Active set Newton (round 2) ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 150:
        a_as2, sc_as2 = _active_set_newton(best_seq, deadline - 150, max_iters=80)
        try_update(a_as2, "active_set_newton_2")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: L-BFGS ultra-fine ===
    print(f"\n=== Phase 6: L-BFGS ultra-fine ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 100:
        beta_mults_ultra = np.array([5e-8, 2e-8, 1e-8, 5e-9, 2e-9])
        a_p6 = _lbfgs_schedule(best_seq, deadline - 100, min_time_remaining=80.0,
                               beta_mults=beta_mults_ultra)
        try_update(a_p6, "lbfgs_ultra")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 50, max_rounds=50000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Multi-restart with quick L-BFGS ===
    print(f"\n=== Phase 8: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("gauss_001", 0.001),
            ("gauss_005", 0.005),
            ("gauss_01", 0.01),
            ("gauss_03", 0.03),
            ("gauss_05", 0.05),
        ]
        for name, scale in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb_restart(best_seq, rng, scale)
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")
            
            # Quick L-BFGS with small betas
            beta_mults_r = np.array([1e-5, 1e-6, 1e-7])
            a_r = _lbfgs_schedule(p_start, deadline - 25, min_time_remaining=20.0,
                                  beta_mults=beta_mults_r)
            try_update(a_r, f"restart_{name}")
            
            # Quick active set
            if time.monotonic() < deadline - 20:
                a_r2, _ = _active_set_newton(a_r, deadline - 15, max_iters=20)
                try_update(a_r2, f"restart_{name}_as")

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(4):
            if time.monotonic() > deadline - 5:
                break
            # Active set
            if time.monotonic() < deadline - 5:
                a_f1, _ = _active_set_newton(best_seq, deadline - 5, max_iters=30)
                try_update(a_f1, f"final_as_{cycle}")
            # Ultra L-BFGS
            if time.monotonic() < deadline - 5 and best_seq is not None:
                beta_mults_f = np.array([1e-8, 5e-9, 2e-9])
                a_f2 = _lbfgs_schedule(best_seq, deadline - 5, min_time_remaining=3.0,
                                       beta_mults=beta_mults_f)
                try_update(a_f2, f"final_lbfgs_{cycle}")
            # Coord descent
            if time.monotonic() < deadline - 3:
                a_f3, _ = _coordinate_descent(best_seq, deadline - 3, max_rounds=10000)
                try_update(a_f3, f"final_cd_{cycle}")

    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
