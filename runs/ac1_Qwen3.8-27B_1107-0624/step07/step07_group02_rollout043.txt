```python
"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. U-shaped (edge-weighted) initializations based on polynomial coefficient analysis
2. Coordinate descent with O(n) per-coordinate 1D golden section search
   exploiting in-place convolution updates (O(n^2) per full sweep)
3. Tighter log-sum-exp L-BFGS with adaptive beta schedule
4. Symmetric coordinate descent variant for faster convergence
5. More aggressive multi-n exploration with diverse profiles
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(20, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coord_descent(a_init, n_sweeps=20, deadline=None, symmetric=False):
    """Coordinate descent with O(n) per-coordinate 1D golden section search.
    Uses in-place convolution updates for efficiency.
    """

    a = a_init.copy()
    c = _full_convolve(a, a)
    S = float(a.sum())
    c_max = float(c.max())
    best_score = 2.0 * n * c_max / (S * S)
    best_a = a.copy()

    half_n = (n + 1) // 2
    coords = list(range(n)) if not symmetric else list(range(half_n))

    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        t_sweep_start = time.monotonic()

        for idx, i in enumerate(coords):
            if deadline and time.monotonic() > deadline:
                break

            ai = a[i]
            S_rest = S - ai
            if S_rest < 1e-15 and ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n, 2 * n - 1)
            ks = np.arange(k_start, k_end)
            a_seg = a[ks - i].copy()
            c_seg = c[ks].copy()

            # Find position of k=2i within ks
            pos_2i = None
            if 2 * i >= k_start and 2 * i < k_end:
                pos_2i = 2 * i - k_start

            # Object maximum outside affected range
            c_out_max = 0.0
            if k_start > 0:
                c_out_max = max(c_out_max, float(c[:k_start].max()))
            if k_end < 2 * n - 1:
                c_out_max = max(c_out_max, float(c[k_end:].max()))

            # Build 1D objective
            # c'_k(t) for affected k:
            #   if k == 2i: c[k] + t^2 - ai^2
            #   else: c[k] + 2*(t-ai)*a[k-i]
            # c'_k(t) for unaffected k: c[k] (constant)

            gr = (np.sqrt(5.0) - 1.0) / 2.0

            def _obj_1d(t):
                delta = t - ai
                c_new_seg = c_seg + 2.0 * delta * a_seg
                if pos_2i is not None:
                    c_new_seg[pos_2i] = c_seg[pos_2i] + t * t - ai * ai
                c_max_val = c_out_max
                seg_max = float(c_new_seg.max())
                if seg_max > c_max_val:
                    c_max_val = seg_max
                s_val = t + S_rest
                if s_val < 1e-15:
                    return 1e30
                return 2.0 * n * c_max_val / (s_val * s_val)

            # Golden section search
            t_lo = 0.0
            t_hi = max(3.0 * ai, 20.0 / n)

            f_lo = _obj_1d(t_lo)
            f_hi = _obj_1d(t_hi)
            expansions = 0
            while f_hi < f_lo and expansions < 8:
                t_lo = t_hi
                f_lo = f_hi
                t_hi *= 2.0
                f_hi = _obj_1d(t_hi)
                expansions += 1

            if t_hi <= t_lo:
                t_opt = t_hi
            else:
                t1 = t_hi - gr * (t_hi - t_lo)
                t2 = t_lo + gr * (t_hi - t_lo)
                f1 = _obj_1d(t1)
                f2 = _obj_1d(t2)

                for _ in range(50):
                    if f1 < f2:
                        t_hi = t2
                        t2 = t1
                        f2 = f1
                        t1 = t_hi - gr * (t_hi - t_lo)
                        f1 = _obj_1d(t1)
                    else:
                        t_lo = t1
                        t1 = t2
                        f1 = f2
                        t2 = t_lo + gr * (t_hi - t_lo)
                        f2 = _obj_1d(t2)
                    if abs(t_hi - t_lo) < 1e-12 * max(1.0, abs(t1)):
                        break
                t_opt = (t_lo + t_hi) / 2.0

            f_opt = _obj_1d(t_opt)
            f_cur = 2.0 * n * c_max / (S * S)

            if f_opt < f_cur - 1e-15 and abs(t_opt - ai) > 1e-15:
                delta = t_opt - ai
                # Update c in place
                c[ks] += 2.0 * delta * a_seg
                if pos_2i is not None:
                    c[2 * i] += delta * delta
                a[i] = t_opt
                S = t_opt + S_rest
                c_max = float(c.max())

                if symmetric and i < n - 1 - i:
                    j = n - 1 - i
                    a[j] = t_opt
                    k_start_j = j
                    k_end_j = min(j + n, 2 * n - 1)
                    ks_j = np.arange(k_start_j, k_end_j)
                    a_seg_j = a[ks_j - j].copy()
                    a_seg_j[j - j] = t_opt  # a[0] in the segment isn't right
                    # Actually for symmetric update, we need to redo
                    # This is complex; skip symmetric update for now
                    pass

        # End of coordinate
        sc_sweep = 2.0 * n * float(c.max()) / (S * S)
        if sc_sweep < best_score - 1e-12:
            best_score = sc_sweep
            best_a = a.copy()

        # Check convergence
        if sweep > 2:
            improvement = best_score - sc_sweep
            if best_score > 0 and abs(improvement) / best_score < 1e-8:
                break

    # Verify with direct computation
    final_score = _score(best_a)
    return best_a, final_score


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_u_shape(n, alpha=1.0):
    """U-shaped profile: higher at edges, lower in middle."""
    j = np.arange(n)
    u = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.array([0.0])
    a = 1.0 + alpha * u ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quartic_u(n, alpha=1.0):
    j = np.arange(n)
    u = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.array([0.0])
    a = 1.0 + alpha * u ** 4
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        return _make_u_shape(n, 1.5)
    elif kind == "u_shape_strong":
        return _make_u_shape(n, 3.0)
    elif kind == "quartic_u":
        return _make_quartic_u(n, 2.0)
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t + 0.1 * np.sin(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search with coordinate descent as key innovation."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_u_shape(n_parent, 1.5)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, U-shape n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1000, t_start + 400)
    a_main = _run_lbfgs(best_seq, p1_deadline, min_time_remaining=50)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Coordinate descent ===
    print(f"\n[P2] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 800, t_start + 550)
    a_cd, sc_cd = _coord_descent(best_seq, n_sweeps=30, deadline=p2_deadline)
    if sc_cd < best_score:
        best_seq = a_cd.copy()
        best_score = sc_cd
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [coord_descent] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS refinement after CD ===
    print(f"\n[P3] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        p3_deadline = min(deadline - 600, t_start + 750)
        a_ref = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=100)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_seq = a_ref.copy()
            best_score = sc_ref
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_refine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Coordinate descent again ===
    print(f"\n[P4] CD round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p4_deadline = min(deadline - 400, t_start + 900)
        a_cd2, sc_cd2 = _coord_descent(best_seq, n_sweeps=20, deadline=p4_deadline)
        if sc_cd2 < best_score:
            best_seq = a_cd2.copy()
            best_score = sc_cd2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [cd_round2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Active-set polish ===
    print(f"\n[P5] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p5_deadline = min(deadline - 300, t_start + 1100)
        a_pol, sc_pol = _active_set_polish(best_seq, p5_deadline,
                                           tol_frac_start=0.005, tol_frac_end=0.0001,
                                           lr_init=0.03)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 200)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 150:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=100)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")
                if time.monotonic() < deadline - 80:
                    a_cd3, sc_cd3 = _coord_descent(best_seq, n_sweeps=10, deadline=min(deadline - 10, t_start + 1350))
                    if sc_cd3 < best_score:
                        best_seq = a_cd3.copy()
                        best_score = sc_cd3
                        print(f"  NEW BEST: {best_score:.10f} [cd_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192, 10240, 12288]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates, key=lambda x: abs(x - len(best_seq))):
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        # Also try U-shaped at this n
        a_u = _make_u_shape(n_try, 2.0)
        # Pick the better initialization
        sc_init1 = _score(a_init)
        sc_init2 = _score(a_u)
        a_better = a_init if sc_init1 < sc_init2 else a_u

        a_opt = _run_lbfgs(a_better, deadline, min_time_remaining=50)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.02 and time.monotonic() < deadline - 40:
            a_cd_n, sc_cd_n = _coord_descent(a_opt, n_sweeps=5, deadline=min(deadline - 10, time.monotonic() + 30))
            if sc_cd_n < best_score:
                best_seq = a_cd_n.copy()
                best_score = sc_cd_n
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_cd] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Diverse restarts ===
    print(f"\n[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_starts = [
            ("pert3", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=3)),
            ("u2", _make_u_shape(n_best, 2.0)),
            ("u3", _make_u_shape(n_best, 3.0)),
            ("quartic", _make_quartic_u(n_best, 2.0)),
            ("bimo", _make_profile(n_best, "bimodal")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 15:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=10)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final tight polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    # Coordinate descent (fast)
    if time.monotonic() < deadline - 5:
        a_fin_cd, sc_fin_cd = _coord_descent(best_seq, n_sweeps=15, deadline=deadline - 3)
        if sc_fin_cd
