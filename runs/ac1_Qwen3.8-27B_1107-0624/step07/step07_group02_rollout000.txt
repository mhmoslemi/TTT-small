"""
Refined optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses improved L-BFGS with log-sum-exp, momentum-based weighted active-set
gradient descent, tight hard active-set polish, and multi-n exploration.
Key improvements: momentum smoothing for active-set oscillations, adaptive
beta stopping, and systematic n-value search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # More aggressive beta cooling schedule
    betas = base_beta * np.array([1.0, 0.5, 0.3, 0.15, 0.08, 0.04,
                                  0.02, 0.01, 0.005, 0.003, 0.0015,
                                  0.0008, 0.0004, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations based on remaining time
        est_time_per_iter = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))

        try:
            result = minimize(
                lambda zz: _smoobj(zz, beta, n),
                z,
                jac=lambda zz: _smograd(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _momentum_active(a_init, deadline, tol_start=0.01, tol_end=1e-5, lr_init=0.04):
    """Active-set gradient descent with momentum for smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    velocity = np.zeros(n)
    lr = lr_init
    momentum = 0.85
    tol = tol_start
    no_improve = 0
    stall_count = 0
    max_stall = 300

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol_val = tol * c_max
        active = c >= (c_max - tol_val)
        n_active = max(1, int(active.sum()))

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Update velocity with momentum
        velocity = momentum * velocity + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(velocity)
        if vnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (vnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 40 == 0:
                tol = max(tol_end, tol * 0.8)
                lr *= 0.85
                momentum = max(0.5, momentum * 0.98)
            if no_improve >= max_stall:
                a = best_a.copy()
                velocity = np.zeros(n)
                lr = lr_init * 0.4
                tol = max(tol_end, tol * 0.5)
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            no_improve = 0
            lr = min(lr_init, lr * 1.01)
            if tol > tol_end:
                tol = max(tol_end, tol * 0.995)

        if tol <= tol_end * 1.01 and no_improve > 50:
            break

    return best_a, _score(best_a)


def _tight_active(a_init, deadline, tol_start=0.001, tol_end=1e-6, lr_init=0.01):
    """Very tight active-set optimization for final polish."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_init
    tol = tol_start
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)

        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:

            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol_val)
        n_active = max(1, int(active.sum()))

        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h - 2.0 * np.mean(h)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad

            if s_try < 1e-15:

            if sc_try < current_sc - 1e-14:

                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1

            if no_improve > 200:
                break
        else:
            no_improve = 0
            if tol > tol_end:
                tol = max(tol_end, tol * 0.99)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _fourier_perturb(a, rng, n_modes=4, amp=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 10))
        a_m = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + a_m * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _soft_active_optimize(a_init, deadline, beta_start=None, beta_end=None):
    """Optimize using soft active-set (exponential weighting) gradient."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    if beta_start is None:
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
        beta_start = max(gap, c_max * 0.01)
    if beta_end is None:
        beta_end = c_max * 0.0002

    beta = beta_start
    lr = 0.08
    momentum = 0.8
    velocity = np.zeros(n)
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Soft weighting
        shifted = (c - c_max) / beta
        w = np.exp(np.clip(shifted, -400, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        velocity = momentum * velocity + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(velocity)

        alpha = min(lr, 0.5 / (vnorm + 1e-15))
        improved = False
        for _ in range(35):
            a_try = a - alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            beta *= 0.97
            lr *= 0.97
            if no_improve > 300:
                a = best_a.copy()
                velocity = np.zeros(n)
                lr = 0.05
                no_improve = 0
                if beta < beta_end * 1.1:
                    break
        else:
            no_improve = 0
            if beta > beta_end:
                beta = max(beta_end, beta * 0.998)
            lr = min(0.08, lr * 1.005)

    return best_a, _score(best_a)
