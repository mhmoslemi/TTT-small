Multi-scale step function optimizer using L-BFGS-B with log-sum-exp smoothing,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_val_and_weights(c, beta):

    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse = c_max
    else:
        w = e / s
        lse = c_max + beta * np.log(s)
    return lse, w


def _z_objective(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    lse, _ = _lse_val_and_weights(c, beta)
    return 2.0 * n * lse


def _z_gradient(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    lse, w = _lse_val_and_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _z_objective_gradient(z, beta, n):
    f = _z_objective(z, beta, n)
    g = _z_gradient(z, beta, n)
    return f, g


def _lbfgs_z(a_init, beta, n, max_iter, deadline):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _z_objective(z, beta, n)

    def jac(z):
        return _z_gradient(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxcor': 80,
                                   'maxls': 50, 'disp': False})
        return _softmax(result.x), int(result.nit)
    except Exception:
        return a_init, 0


def _multi_beta_lbfgs(a_start, deadline, min_time=60.0, n_betas=None):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    if n_betas is None:
        n_betas = 12
    betas = max_c * np.array([0.3, 0.15, 0.075, 0.0375, 0.019, 0.0095,
                              0.005, 0.0025, 0.00125, 0.0006,
                              0.0003, 0.00015, 0.00008, 0.00004,
                              0.00002, 0.00001])[:n_betas]

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for beta in betas:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.003 * (n / 4096.0) ** 1.3
        max_iter = min(8000, max(100, int(remaining / (est_time * 15))))
        if max_iter < 50:
            max_iter = 50

        a_new, niter = _lbfgs_z(current_a, beta, n, max_iter, deadline)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _peak_equalize(a_init, deadline, max_iters=20000, n_peaks=5):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        c_avg = c.mean()
        if c_max < 1e-15:
            break

        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        n_active = min(n_peaks, len(c))
        active = np.zeros_like(c)
        for k in range(n_active):
            active[sorted_idx[k]] = 1.0
        w = active / active.sum()

        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        # Backtracking line search
        best_step = 0
        best_sc_try = best_sc
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.075, 0.04]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_step = step

        if best_step > 0:
            a_try = a - best_step * g
            a_try = np.maximum(a_try, 0.0)
            a = a_try / a_try.sum()
            if best_sc_try < best_sc - 1e-15:
                best_sc = best_sc_try
                best_a = a.copy()
                eta = min(eta * 1.02, 0.05)
                stall = 0
            else:
                stall += 1
                eta *= 0.9
        else:
            eta *= 0.85
            stall += 1
            if eta < 1e-9 or stall > 200:
                break

    return best_a, best_sc


def _direct_gradient_backtrack(a_init, deadline, max_iters=30000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.005
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Use top few peaks for gradient
        sorted_idx = np.argsort(c)[::-1]
        n_top = min(8, len(c))
        w = np.zeros_like(c)
        for k in range(n_top):
            w[sorted_idx[k]] = 1.0 / n_top

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        improved = False
        for frac in [1.0, 0.6, 0.4, 0.2, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            eta = min(eta * 1.01, 0.03)
            stall = 0
        else:
            eta *= 0.92
            stall += 1
            if eta < 1e-10 or stall > 300:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    if target_n < n_cur:
        target_n = n_cur
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 2.0 * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, rng, scale=0.03):
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(len(a))
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.005 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer: coarse-to-fine L-BFGS with log-sum-exp,
    peak equalization, direct gradient with backtracking, and
    multi-restart with Fourier perturbations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    target_n = 8192

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(target_n, "uniform")
        print(f"No parent. Init uniform n={target_n}, score={_score(a0):.10f}")
        try_update(a0, "init")

    # === Phase 1: Multi-scale coarse-to-fine ===
    print(f"\n=== Phase 1: Multi-scale optimization ===")
    t_phase = time.monotonic()

    if parent is not None:
        # Start from parent at its native resolution, then go to target
        scales = []
        pn = len(parent)
        if pn < target_n:
            cur = pn
            while cur < target_n:
                cur = min(cur * 2, target_n)
                scales.append(cur)
        a_work = parent.copy()
    else:
        a_work = _make_profile(128, "uniform")
        scales = [256, 512, 1024, 2048, 4096, 8192]

    for scale_n in scales:
        if time.monotonic() > deadline - 300:
            break
        print(f"  Scale n={scale_n}...")

        # Interpolate to this scale
        a_work = _interpolate_to_n(a_work, scale_n)
        sc_init = _score(a_work)
        print(f"    Interpolated score: {sc_init:.10f}")

        # L-BFGS with beta continuation
        a_opt = _multi_beta_lbfgs(a_work, deadline - 250, min_time=100.0, n_betas=10)
        sc_opt = _score(a_opt)
        print(f"    After L-BFGS: {sc_opt:.10f}")
        try_update(a_opt, f"scale_{scale_n}")

        # Peak equalization
        if time.monotonic() < deadline - 200:
            a_pe, sc_pe = _peak_equalize(a_opt, deadline - 200, max_iters=10000)
            try_update(a_pe, f"scale_{scale_n}_peaq")

        a_work = best_seq if best_seq is not None and len(best_seq) == scale_n else a_opt

    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 2: Intensive L-BFGS at target scale ===
    print(f"\n=== Phase 2: Intensive L-BFGS ===")
    t_phase = time.monotonic()

    if best_seq is not None and time.monotonic() < deadline - 200:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())

        # Fine beta schedule
        fine_betas = mb * np.array([2e-4, 1e-4, 5e-5, 2.5e-5, 1e-5,
                                    5e-6, 2e-6, 1e-6, 5e-7, 2e-7,
                                    1e-7, 5e-8, 2e-8, 1e-8])
        a_cur = best_seq.copy()
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - 150 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(6000, max(200, int(remaining / 0.008)))
            a_new, niter = _lbfgs_z(a_cur, beta, n, max_iter, deadline)
            sc = _score(a_new)
            if sc < best_score_val - 1e-15:
                a_cur = a_new
                try_update(a_cur, f"fine_{i}")

    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 3: Peak equalization (aggressive) ===
    print(f"\n=== Phase 3: Peak equalization ===")
    t_phase = time.monotonic()

    if best_seq is not None and time.monotonic() < deadline - 150:
        a_pe2, sc_pe2 = _peak_equalize(best_seq, deadline - 150, max_iters=30000, n_peaks=8)
        try_update(a_pe2, "peak_eq_agg")

    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 4: Direct gradient with backtracking ===
    print(f"\n=== Phase 4: Direct gradient backtracking ===")
    t_phase = time.monotonic()

    if best_seq is not None and time.monotonic() < deadline - 100:
        a_dg, sc_dg = _direct_gradient_backtrack(best_seq, deadline - 100, max_iters=50000)
        try_update(a_dg, "direct_grad")

    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 5: Multi-restart with perturbation ===
    print(f"\n=== Phase 5: Multi-restart ===")
    t_phase = time.monotonic()

    if best_seq is not None and time.monotonic() < deadline - 60:
        restarts = [
            ("gauss_05", "gauss", 0.005),
            ("gauss_01", "gauss", 0.01),
            ("gauss_02", "gauss", 0.02),
            ("fourier", "fourier", 0.0),
            ("fourier2", "fourier", 0.0),
        ]
        for name, mode, scale in restarts:
            if time.monotonic() > deadline - 40:
                break
            if mode == "gauss":
                p_start = _perturb(best_seq, rng, scale)
            else:
                p_start = _perturb_fourier(best_seq, rng)
            sc_start = _score(p_start)

            # Quick L-BFGS
            if time.monotonic() < deadline - 30:
                n = len(p_start)
                c_s = _full_convolve(p_start, p_start)
                mb_s = float(c_s.max())
                betas_r = mb_s * np.array([1e-4, 5e-5, 1e-5, 1e-6])
                a_r = p_start
                for beta in betas_r:
                    if time.monotonic() > deadline - 25:
                        break
                    max_iter = min(2000, max(100, int((deadline - 25 - time.monotonic()) / 0.005)))
                    a_r, _ = _lbfgs_z(a_r, beta, n, max_iter, deadline)

                try_update(a_r, f"restart_{name}")

    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 6: Ultra-fine final polish ===
    print(f"\n=== Phase 6: Ultra-fine polish ===")
    t_phase = time.monotonic()

    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())

        # Very fine L-BFGS
        ultra_betas = mb * np.array([1e-8, 5e-9, 2e-9, 1e-9])
        a_u = best_seq
        for beta in ultra_betas:
            if time.monotonic() > deadline - 5:
                break
            max_iter = min(3000, max(100, int((deadline - 5 - time.monotonic()) / 0.003)))
            a_u, _ = _lbfgs_z(a_u, beta, n, max_iter, deadline)
            try_update(a_u, "ultra")

        # Final peak equalization
        if time.monotonic() < deadline - 5:
            a_f, _ = _peak_equalize(best_seq, deadline - 5, max_iters=5000, n_peaks=10)
            try_update(a_f, "final_peaq")

        # Final direct gradient
        if time.monotonic() < deadline - 3:
            a_fg, _ = _direct_gradient_backtrack(best_seq, deadline - 3, max_iters=10000)
            try_update(a_fg, "final_dg")

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
