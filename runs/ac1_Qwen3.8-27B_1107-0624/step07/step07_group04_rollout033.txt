    # w has length 2n-1, a has length n
    # result has length n

```python
def project_simplex(v, s=1.0):
    """Project v onto {a : a >= 0, sum(a) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    cond = u * np.arange(1, n+1) - cssv > 0
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] / (rho + 1.0))
    return np.maximum(v - theta, 0)
```

"""
Asymmetric profile optimizer: exploits the insight that ramp-like (asymmetric)
profiles achieve score ~4/3 vs ~2 for symmetric profiles.
Uses projected gradient with adaptive active set (top-K lags), followed by
L-BFGS polish with log-sum-exp smoothing.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    cond = u * np.arange(1, n + 1) - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _grad_conv(a, w, n):
    """grad of sum_k w[k]*(a*a)[k] w.r.t. a. Result has length n."""
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W * A, N)
    return 2.0 * conv[n - 1:2 * n - 1]

def _hankel_mv(v, w, n):
    """(Hv)_j = sum_m w[j+m]*v_m, H is Hankel with first row w[0:n]."""
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    v_rev = v[::-1]
    V = np.fft.rfft(v_rev, N)
    conv = np.fft.irfft(W * V, N)
    return conv[n - 1:2 * n - 1]

def _pg_optimize(a_init, deadline, max_iter=50000, k_init=10, k_max_frac=0.3):
    """Projected gradient with adaptive top-K active set."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    alpha = 0.1
    total_lags = 2 * n - 1
    no_improve = 0
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > (200 if i < 5000 else 500):
            break
        
        # Adaptive K: start small, grow over time
        progress = min(1.0, i / (max_iter * 0.5))
        k = max(k_init, int(total_lags * k_max_frac * progress))
        k = min(k, total_lags)
        
        # Top-K lags
        if k < total_lags:
            thresholds = np.partition(c, total_lags - k)[total_lags - k]
            active = c >= thresholds
        else:
            active = np.ones(total_lags, dtype=bool)
        
        k_eff = max(1, int(active.sum()))
        w = np.zeros(total_lags)
        w[active] = 1.0 / k_eff
        
        obj_cur = float(w @ c)
        
        # Gradient
        g = _grad_conv(a, w, n)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-18:
            break
        
        # Normalized gradient direction
        d = -g / g_norm
        
        # Backtracking line search
        armijo = 1e-4
        c1 = armijo
        step_accepted = False
        for ls in range(20):
            a_try = _project_simplex(a + alpha * d * g_norm)
            # Equivalent to: a_try = _project_simplex(a - alpha * g)
            c_try = _full_convolve(a_try, a_try)
            obj_try = float(w @ c_try)
            if obj_try <= obj_cur - c1 * alpha * g_norm * g_norm:
                a = a_try
                step_accepted = True
                alpha = min(alpha * 1.05, 0.5)
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break
        
        if not step_accepted:
            alpha = 0.01
            # Accept anyway to make progress
            a = _project_simplex(a - 0.001 * g / (g_norm + 1e-30))
    
    # Final score
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc

Wait, I think the line search condition is wrong. Let me reconsider.

The objective is $f(a) = \sum_k w_k c_k$ (for fixed $w$). The gradient is $g = \nabla f(a)$.
The direction is $d = -g$ (steepest descent).
Step: $a_{new} = \text{Proj}(a + \alpha d) = \text{Proj}(a - \alpha g)$.

Armijo condition: $f(a_{new}) \leq f(a) + c_1 \alpha \nabla f(a)^T d = f(a) - c_1 \alpha \|g\|^2$.

So the condition is: $f(a_{new}) \leq f(a) - c_1 \alpha \|g\|^2$.

But the issue is that after projection, the step might not be in the direction $-g$. The projection can change the direction. This makes the Armijo condition harder to satisfy.

A simpler approach: just check if $f(a_{new}) < f(a)$ (no descent condition, just improvement).

Let me also think about the step size. For a quadratic $f(x) = x^T W x$ with Hessian $2W$, the Lipschitz constant of the gradient is $2\|W\|_2$. For our Hankel matrix with entries $w_{j+m}$ where $w$ is uniform on $K$ lags, $\|W\|_2 \leq \|w\|_1 = 1$. So the Lipschitz constant is at most 2.

The theoretical optimal step size for gradient descent is $\alpha = 1/L = 0.5$. But with projection, we need a smaller step. Let me start with $\alpha = 0.1$ and adapt.
