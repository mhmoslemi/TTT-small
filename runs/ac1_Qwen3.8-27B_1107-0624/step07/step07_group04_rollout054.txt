def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs using CG with restart and regularization."""
    x = np.zeros(n)
    r = rhs - _hankel_matvec(x, W, n)
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if pHp < 1e-30:
            # Add regularization
            Hp = Hp + 1e-10 * p
            pHp = float(p @ Hp)
            if pHp < 1e-30:
                break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            # Restart CG
            p = r.copy()
            rs = rs_new
            continue
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    if s < 1e-300 or not np.isfinite(s):

    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)

    a = _softmax(z)

    if s < 1e-300 or not np.isfinite(s):
        var_term = np.var(c)
        return 2.0 * n * c_max + 0.01 * var_term
    smooth_max = c_max + beta * np.log(s)
    var_term = np.var(c[c > 0.01 * c_max])
    return 2.0 * n * smooth_max + 0.01 * var_term


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, use_var=False):
    """L-BFGS with adaptive beta continuation based on spectral gap."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive base beta from spectral gap
    c_sorted = np.sort(c_cur)[::-1]
    n_check = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[n_check]
    base_beta = max(gap * 0.3, max_c * 0.003)

    # More aggressive schedule with more stages
    ratios = np.array([1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02,
                       0.01, 0.006, 0.003, 0.002, 0.001, 0.0006, 0.0003,
                       0.0002, 0.0001, 0.00006, 0.00003, 0.00002,
                       0.00001, 0.000006, 0.000003, 0.000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    obj_func = _varobj if use_var else _smoobj

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 20))))

        try:
            result = minimize(
                obj_func, z,
                jac=_smograd if not use_var else None,
                args=(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i < len(betas) - 5:
                # If no improvement, try with slightly different beta
                pass
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, max_steps=8000):
    """Adaptive subgradient with momentum and dynamic active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015
    momentum = np.zeros(n)
    beta_m = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Dynamic active set
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Momentum update
        lr_t = lr / (1.0 + 0.0002 * i)
        momentum = beta_m * momentum - lr_t * grad
        z_new = z + momentum
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 50 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = 0.01
            z = z_new
        else:
            z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.08, 0.12, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.3) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(12345)
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=max(2, n // 100))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining L-BFGS with adaptive beta, active set Newton,
    subgradient with momentum, shrinkage, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS from parent ===
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # === Phase 2: Newton polish ===
    print(f"\n[Phase 2] Newton polish 1 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 500, time.monotonic() + 400))
        update_best(a2, sc2, "newton_1")

    # === Phase 3: Subgradient with momentum ===
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _subgrad_polish(best_seq, min(deadline - 400, time.monotonic() + 150))
        update_best(a3, sc3, "sg_1")

    # === Phase 4: Newton after subgradient ===
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 150))
        update_best(a4, sc4, "newton_2")

    # === Phase 5: Shrinkage ===
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=120.0)
        update_best(a5, sc5, "shrinkage")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 4096, 2048]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 60:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 50, time.monotonic() + 80))
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        restart_kinds = [
            ("perturb_small", _perturb(best_seq, 0.02, rng)),
            ("perturb_med", _perturb(best_seq, 0.08, rng)),
            ("gaussian", _make_profile(n_best, "gaussian")),
            ("bimodal", _make_profile(n_best, "bimodal")),
            ("random_smooth", _make_profile(n_best, "random_smooth")),
        ]
        for label, p_start in restart_kinds:
            if time.monotonic() > deadline - 30:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 10, time.monotonic() + 40), max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{label}_np")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 3, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 4:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 4 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.005)))
                try:
                    result = minimize(
                        _smoobj, z,
                        jac=_smograd,
                        args=(beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # One more Newton pass
        if time.monotonic() < deadline - 2 and best_seq is not None:
            a_fn2, sc_fn2 = _newton_active_polish(best_seq, deadline - 1, max_rounds=100)
            update_best(a_fn2, sc_fn2, "final_newton_2")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
