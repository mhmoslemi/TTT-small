"""
Precision active-set optimizer for flat autocorrelation sequences.
Uses tight-tolerance Newton on the active constraint set, FFT-accelerated

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, active, n):
    """Compute H*v where H[j,l] = 2*active[j+l]. Result has length n."""
    v_rev = v[::-1]
    N = _next_pow2(4 * n)
    fv = np.fft.rfft(v_rev, N)
    fa = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fv, N)
    return 2.0 * r[n - 1:2 * n - 1]

def _newton_precise(a_init, deadline, max_iter=1000, tol_frac=1e-5):

    High-precision active-set Newton method.

        c_max = float(c.max())

        grad_a = _hankel_matvec(a, active_float, n)

            dx = -grad_t
            dx = dx - np.mean(dx)

        # Line search with non-negativity
        f_old = f_active
        alpha = 1.0
        a_new = a.copy()
        found = False
        for ls in range(60):
            a_try = a + alpha * dx
            if (a_try < 0).any():
                # Find max alpha that keeps non-negativity
                neg_mask = dx < 0
                if neg_mask.any():
                    alpha_max = (a[neg_mask] / (-dx[neg_mask])).max()
                    alpha = min(alpha, alpha_max * 0.99)
                else:
                    alpha *= 0.5
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-10 * alpha * gnorm:
                a_new = a_try
                found = True
                break
            alpha *= 0.5

        if not found:
            # Try smaller step
            alpha = max(alpha * 0.5, 1e-10)
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_new = a_try / s_try
            else:
                break
            # Verify it's not terrible
            c_try = _full_convolve(a_new, a_new)
            if float(c_try[active].sum()) > f_old * 1.1:
                break

        a = a_new
        best_f_active = min(best_f_active, f_active)

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Adaptive: if stuck, reduce tol_frac to tighten active set
        if it > 50 and float(c[active].sum()) > best_f_active * 1.001:
            tol_frac = max(tol_frac * 0.5, 1e-8)

    # Final check
    sc_final = _score(best_a)
    if sc_final < best_sc:
        best_sc = sc_final
    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS on the smooth max with adaptive beta schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # Adaptive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                  0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.002 * max(1, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd_correct(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 80, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgradient_polish(a_init, deadline, max_steps=5000, lr_init=0.01):
    """Subgradient method on the max with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_float = active.astype(np.float64)

        grad = _hankel_matvec(a, active_float, n)
        grad = grad - np.mean(grad)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * t + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((t - 0.7) ** 2) / (2 * 0.1 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_swap(a, rng, n_swaps=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_swaps * 3))
    for _ in range(n_swaps):
        i = rng.integers(0, max(1, n - 2 * block_size))
        j = rng.integers(0, max(1, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Precision optimizer combining tight-tolerance active-set Newton,
    adaptive smooth-max L-BFGS, subgradient polish, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Quick L-BFGS to refine starting point ===
    print(f"\n[Phase 1] Quick L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        a1 = _lbfgs_optimize(best_seq, min(deadline - 800, time.monotonic() + 200), min_time_remaining=30)
        update_best(a1, _score(a1), "lbfgs_quick")

    # === Phase 2: High-precision active-set Newton (main workhorse) ===
    print(f"\n[Phase 2] Active-set Newton t=1e-5 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        n_deadline = min(deadline - 400, time.monotonic() + 600)
        a2, sc2 = _newton_precise(best_seq, n_deadline, max_iter=2000, tol_frac=1e-5)
        update_best(a2, sc2, "newton_1e-5")

    # === Phase 3: Even tighter Newton ===
    print(f"\n[Phase 3] Active-set Newton t=1e-7 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n_deadline = min(deadline - 200, time.monotonic() + 300)
        a3, sc3 = _newton_precise(best_seq, n_deadline, max_iter=2000, tol_frac=1e-7)
        update_best(a3, sc3, "newton_1e-7")

    # === Phase 4: L-BFGS with very small beta ===
    print(f"\n[Phase 4] Ultra-fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a4 = _lbfgs_optimize(best_seq, min(deadline - 150, time.monotonic() + 200), min_time_remaining=20)
        update_best(a4, _score(a4), "ultra_lbfgs")

    # === Phase 5: Subgradient polish ===
    print(f"\n[Phase 5] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        sg_deadline = min(deadline - 100, time.monotonic() + 120)
        a5, sc5 = _subgradient_polish(best_seq, sg_deadline, max_steps=5000, lr_init=0.005)
        update_best(a5, sc5, "subgrad")

    # === Phase 6: Newton after subgradient ===
    print(f"\n[Phase 6] Newton after subgrad (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_deadline = min(deadline - 60, time.monotonic() + 80)
        a6, sc6 = _newton_precise(best_seq, n_deadline, max_iter=500, tol_frac=1e-8)
        update_best(a6, sc6, "newton_final")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_restarts = 6
        perturb_methods = [
            lambda: _perturb_log(best_seq, 0.02, rng),
            lambda: _perturb_fourier(best_seq, rng, n_modes=4),
            lambda: _perturb_swap(best_seq, rng, n_swaps=4),
            lambda: _perturb_log(_perturb_fourier(best_seq, rng, n_modes=2), 0.01, rng),
            lambda: _perturb_swap(_perturb_log(best_seq, 0.01, rng), rng, n_swaps=3),
            lambda: _perturb_fourier(_perturb_swap(best_seq, rng, n_swaps=2), rng, n_modes=3),
        ]
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            try:
                p_start = perturb_methods[i]()
            except Exception:
                p_start = _perturb_log(best_seq, 0.02, rng)

            # Quick Newton on perturbed
            if time.monotonic() < deadline - 15:
                a_r, sc_r = _newton_precise(p_start, min(deadline - 15, time.monotonic() + 40),
                                            max_iter=200, tol_frac=1e-5)
                update_best(a_r, sc_r, f"restart_{i}_newton")

            # Quick L-BFGS
            if sc_r < best_score * 1.005 and time.monotonic() < deadline - 10:
                a_r2 = _lbfgs_optimize(a_r, min(deadline - 10, time.monotonic() + 25), min_time_remaining=5)
                update_best(a_r2, _score(a_r2), f"restart_{i}_lbfgs")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[Phase 8] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else 8192
    n_candidates = [n_best]
    for n_try in [4096, 8192, 12288, 16384]:
        if n_try != n_best and n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 25:
            break
        if n_try == n_best:
            a_init = best_seq.copy()
        else:
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

        if time.monotonic() < deadline - 20:
            a_opt, sc_opt = _newton_precise(a_init, min(deadline - 20, time.monotonic() + 40),
                                            max_iter=300, tol_frac=1e-5)
            update_best(a_opt, sc_opt, f"n_{n_try}")

    # === Phase 9: Final ultra-precise Newton ===
    print(f"\n[Phase 9] Final ultra-precise Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_final, sc_final = _newton_precise(best_seq, deadline - 1, max_iter=1000, tol_frac=1e-9)
        update_best(a_final, sc_final, "ultra_final")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence confirms: {sc_eval:.10f}")
                if sc_eval < sc_final:
                    sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
