"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive L-BFGS with progressive log-sum-exp smoothing
2. Active-set subgradient with backtracking line search
3. Mirror descent with adaptive step size
4. Multi-length exploration (n=4096 to 65536)
5. Diverse random restarts for escaping local minima
6. Final ultra-tight polishing
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=30.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Progressive tightening of beta
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        # Scale max_iter based on n and remaining time
        scale = max(1.0, (n / 4096.0) ** 1.1)
        max_iter = min(8000, max(200, int(remaining / (0.003 * scale * 15))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _subgrad_active(a_init, deadline, k_list=(5, 10, 20), max_steps=50000, lr_init=0.03):
    """Active-set subgradient with backtracking line search."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for k_active in k_list:
        a = best_a.copy()
        lr = lr_init
        stall = 0

        for i in range(max_steps):
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            c_max = c.max()

            # Get active set
            idx = np.argsort(c)[::-1][:k_active]
            w = np.zeros_like(c)
            w[idx] = 1.0 / k_active

            # Compute gradient: 4*n*a*(h - W) where h = conv(w, a_rev)[n-1:2n-1]
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                break

            # Try step with backtracking
            step = lr * a.sum() / n
            improved = False
            for back in range(10):
                a_try = a - step * grad / (a * n + 1e-30)
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    break
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-16:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
                step *= 0.7

            if improved:
                stall = 0
                lr = min(lr * 1.01, 0.1)
            else:
                stall += 1
                lr *= 0.85
                if stall > 200 or lr < 1e-10:
                    break

    return best_a, best_sc

    current_sc = _score(current_a)

                              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7, 2e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, (n / 4096.0) ** 1.1)
        # Spend more iterations at tight betas
        if i >= len(betas) // 2:
            max_iter = min(10000, max(500, int(remaining / (0.002 * scale * 10))))
        else:
            max_iter = min(5000, max(200, int(remaining / (0.004 * scale * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _subgrad_threshold(a_init, deadline, delta=0.01, max_steps=30000, lr_init=0.02):
    """Subgradient with threshold-based active set and armijo backtracking."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Threshold-based active set: all peaks within delta of max
        threshold = c_max * (1.0 - delta)
        active_mask = c >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            break

        w = active_mask.astype(np.float64) / n_active

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Armijo backtracking
        step = lr * a.sum() / n
        sc_cur = _score(a)
        improved = False
        for _ in range(15):
            a_try = a - step * grad / (a * n + 1e-30)
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                break
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            if sc_try < sc_cur + 1e-16:
                # Acceptable improvement over current (not necessarily best)
                a = a_try
                improved = True
                break
            step *= 0.6

        if improved:
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if stall > 300 or lr < 1e-12:
                break

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_steps=50000, lr_init=2.0):
    """Mirror descent in the simplex (exponential weights)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Use top-5 active set for gradient
        idx = np.argsort(c)[::-1][:5]
        w = np.zeros_like(c)
        w[idx] = 1.0 / 5.0

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W  # This is the "direction" in the simplex
        g = g - g.mean()  # Center for the simplex constraint

        z = z - lr * g
        z = z - z.mean()

        # Periodic check
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.7
                    if lr < 1e-6:
                        break
                else:
                    lr *= 0.95

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_best(a_best, rng, n, mode=0):
    """Create a perturbation of the best sequence."""
    if mode == 0:
        # Log-space Gaussian
        sigma = 0.03
        z = np.log(np.maximum(a_best, 1e-18))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        return _softmax(z)
    elif mode == 1:
        # Fourier perturbation
        z = np.log(np.maximum(a_best, 1e-18))
        n_freqs = 3
        t = np.arange(n)
        for _ in range(n_freqs):
            freq = int(rng.integers(1, max(2, n // 32)))
            amp = rng.uniform(0.01, 0.05)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        z = z - z.mean()
        return _softmax(z)
    else:
        # Block perturbation
        z = np.log(np.maximum(a_best, 1e-18))
        n_blocks = 5
        block_sz = n // n_blocks
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_sz)))
            end = min(n, start + block_sz)
            z[start:end] += rng.uniform(-0.03, 0.03)
        z = z - z.mean()
        return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):

    Multi-phase optimizer combining L-BFGS, subgradient, mirror descent,

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 16384

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev-best_score_val:.2e} t={elapsed:.1f}s")

    print("=== Loading parent ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

    except Exception:
