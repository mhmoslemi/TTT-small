**Frequency-domain alternation (POCS-like):**

The idea: alternate between:
- Time domain: enforce `b >= 0` and `sum(b) = 1` (project onto simplex)
- Frequency domain: enforce `|FFT(b)|` to be "flat" (i.e., minimize the peak of the autocorrelation)

The frequency-domain "projection" is: given `B = FFT(b)`, find `B'` that minimizes `max_m IFFT(|B'|²)[m]` subject to `|B'| = |B'|` (same magnitude? no, that doesn't make sense).

Actually, the correct POCS formulation would be:
- Set 1: `b >= 0`, `sum(b) = 1` (simplex)
- Set 2: `max_m conv(b,b)[m] <= T` for some threshold T

But Set 2 is not convex (as discussed), so POCS doesn't directly apply.

An alternative: instead of projecting onto Set 2, we "soft" enforce it by modifying the spectrum:
- `B' = B * sqrt(T / max(|B|², T))` (shrink the spectrum where it exceeds the threshold)
- This is a "spectral flattening" step

Then alternate:
1. `b = IFFT(B)` → project onto simplex → `b = max(b, 0) / sum(b)`
2. `B = FFT(b)` → flatten: `B = B * sqrt(T / max(|B|², T))`
3. Decrease T over iterations

This is similar to the "successive projection" approach used in waveform design (e.g., for radar pulses with peak-to-average power ratio constraints).

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unit(b):
    """Score assuming sum(b) = 1, b >= 0"""
    n = len(b)
    c = _fft_convolve(b, b)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _subgradient(b, c, top_k=5):
    """Compute subgradient of max(conv(b,b)) at b, using top_k active lags."""
    n = len(b)
    k_indices = np.argsort(c)[::-1][:top_k]
    # Weight by proximity to max (exponential decay)
    c_max = c[k_indices[0]]
    weights = np.ones(len(k_indices))
    for i in range(1, len(k_indices)):
        weights[i] = np.exp(-5.0 * (c_max - c[k_indices[i]]) / max(c_max, 1e-15))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, k_indices):
        # grad_k[j] = 2*b[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:

            # For j in [lo, hi]: k-j ranges from k-lo to k-hi
            idx = k - np.arange(lo, hi + 1)  # indices into b
            g[lo:hi+1] += w * 2.0 * b[idx]
    return g

`c[k] = sum_{i: 0<=i<n, 0<=k-i<n} b[i] * b[k-i]`

For a fixed k, the valid range of i is: `max(0, k-(n-1)) <= i <= min(k, n-1)`.

Let `lo_k = max(0, k-n+1)` and `hi_k = min(k, n-1)`.

Then `c[k] = sum_{i=lo_k}^{hi_k} b[i] * b[k-i]`.

The gradient w.r.t. `b[j]`:
- If j is not in [lo_k, hi_k], the gradient is 0.
- If j is in [lo_k, hi_k]:
  - b[j] appears in the term `b[j] * b[k-j]` (as the first factor)
  - b[j] also appears in the term `b[k-j] * b[j]` (as the second factor, when i = k-j)
  - These are the same term if j = k-j, i.e., 2j = k.
  - If j ≠ k-j (and k-j is also in [lo_k, hi_k]): gradient contribution is `b[k-j] + b[k-j] = 2*b[k-j]`
  - If j = k-j (i.e., 2j = k): gradient contribution is `2*b[j]` (derivative of b[j]²)

But note: k-j is in [lo_k, hi_k] if and only if j is in [lo_k, hi_k] (by symmetry of the range).

So: `∂c[k]/∂b[j] = 2*b[k-j]` if `lo_k <= j <= hi_k`, else 0.

And since `k-j` is always in range when `j` is in range (by the definition of lo_k and hi_k), this is correct.

```python
"""
Combined optimizer for minimizing 2*n*max(conv(b,b)) subject to sum(b)=1, b>=0.
Uses: (1) exact subgradient descent on simplex, (2) frequency-domain perturbation,
(3) pairwise coordinate descent, (4) simulated annealing local search.
Differs from prior approach by working directly on the simplex (no softmax parametrization),
using exact (unsmoothed) subgradients, and employing frequency-domain operations.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_unit(b):
    """Compute 2*n*max(conv(b,b)) assuming sum(b)=1."""
    n = len(b)
    c = _fft_convolve(b, b)
    return float(2.0 * n * c.max())


def _score_uneval(b):
    """Compute the full score for arbitrary b (handles normalization)."""
    n = len(b)
    s = float(b.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(b, b)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project onto probability simplex {x: sum(x)=1, x>=0} using sorting."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = int(np.max(ind[cond]))
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _compute_subgradient(b, c, top_k=5, sharpness=8.0):
    """
    Compute subgradient of max(conv(b,b)) w.r.t. b.
    Uses weighted average over top_k active lags with exponential weighting.
    """
    n = len(b)
    k_all = min(top_k, len(c))
    k_indices = np.argsort(c)[::-1][:k_all]
    c_max = float(c[k_indices[0]])
    spread = max(c_max - c[k_indices[-1]], 1e-15)
    weights = np.exp(-sharpness * (c_max - c[k_indices]) / spread)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, k_indices):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            js = np.arange(lo, hi + 1)
            idx = k - js
            g[lo:hi + 1] += w * 2.0 * b[idx]
    return g


def _subgradient_optimize(b_init, n, deadline, top_k=5, init_lr=0.005):
    """
    Subgradient descent on the simplex with adaptive step size and restarts.
    Minimizes max(conv(b,b)) subject to sum(b)=1, b>=0.
    """
    b = np.maximum(b_init, 0).astype(np.float64)
    b = b / b.sum()

    best_b = b.copy()
    c = _fft_convolve(b, b)
    best_val = float(c.max())

    lr = init_lr
    patience = 0
    max_patience = 3000
    momentum = 0.0
    v = np.zeros(n)
    beta = 0.85

    for it in range(5000000):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(b, b)
        c_max = float(c.max())

        g = _compute_subgradient(b, c, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        # Heavy ball momentum
        v = beta * v - lr * g / g_norm
        b_new = b + v
        b_new = _project_simplex(b_new)

        # Check if we improved
        if it % 50 == 0:
            val = float(_fft_convolve(b_new, b_new).max())
            if val < best_val:
                best_val = val
                best_b = b_new.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    b = best_b.copy()
                    v *= 0.5
                    lr *= 0.75
                    patience = 0
                    if lr < 1e-7:
                        break
        else:
            b = b_new

    val = float(_fft_convolve(b, b).max())
    if val < best_val:
        best_val = val
        best_b = b.copy()

    return best_b, best_val


def _freq_perturb(b_init, n, deadline, rng):
    """
    Frequency-domain perturbation: iteratively modify the phase/magnitude
    of the FFT to reduce autocorrelation peak, then project to simplex.
    """
    b = np.maximum(b_init, 0).astype(np.float64)
    b = b / b.sum()
    N = _next_pow2(2 * n - 1)

    best_b = b.copy()
    c = _fft_convolve(b, b)
    best_val = float(c.max())

    # Current spectrum
    B = np.fft.rfft(b, N)
    mag = np.abs(B)
    phase = np.angle(B)

    patience = 0
    max_patience = 2000
    gamma = 0.3  # perturbation strength

    for it in range(2000000):
        if time.monotonic() > deadline:
            break

        # Compute current autocorrelation
        c = _fft_convolve(b, b)
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Find peak lags
        peak_lags = np.argsort(c)[::-1][:3]

        # In frequency domain, to reduce c[m], we want to reduce |B[k]|
        # where the cosine component at lag m is positive.
        # c[m] = (1/N) * sum_k |B[k]|^2 * cos(2*pi*k*m/N) (approximately for real signals)
        # Actually for real signals: c[m] = (1/N) * sum_k |B[k]|^2 * cos(2*pi*k*m/N)

        # Compute gradient of c[peak] w.r.t. |B[k]|^2
        # dc[m]/d|B[k]|^2 = (1/N) * cos(2*pi*k*m/N)
        # So to reduce c[m], decrease |B[k]|^2 where cos > 0

        k_vals = np.arange(len(B))
        for m in peak_lags:
            cos_comp = np.cos(2.0 * np.pi * k_vals * m / N)
            # Only reduce where cos is positive (contributes to increasing c[m])
            reduce_mask = cos_comp > 0
            # Scale: reduce proportionally to the cosine value
            mag[reduce_mask] *= (1.0 - gamma * cos_comp[reduce_mask] * 0.1)

        # Reconstruct and project
        B_new = mag * np.exp(1j * phase)
        b_new = np.fft.irfft(B_new, N)[:n].real
        b_new = np.maximum(b_new, 0)
        s = b_new.sum()
        if s < 1e-15:
            b_new = np.ones(n) / n
        else:
            b_new = b_new / s

        val = float(_fft_convolve(b_new, b_new).max())
        if val < best_val:
            best_val = val
            best_b = b_new.copy()
            patience = 0
            b = b_new
            # Update spectrum
            B = np.fft.rfft(b, N)
            mag = np.abs(B)
            phase = np.angle(B)
        else:
            patience += 1
            if patience > max_patience:
                # Reset spectrum from best
                B = np.fft.rfft(best_b, N)
                mag = np.abs(B)
                phase = np.angle(B)
                b = best_b.copy()
                patience = 0
                gamma *= 0.8
                if gamma < 0.01:
                    break

    return best_b
