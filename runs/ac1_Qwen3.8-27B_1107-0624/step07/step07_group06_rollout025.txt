Focused optimizer using aggressive L-BFGS with fine beta continuation,
variance pre-optimization, multi-n exploration, and post-optimization
local search (element zeroing and perturbation-based refinement).

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    g = g - np.mean(g)
    return g

def _active_grad(z, n, tol_frac=0.002):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(z_init, beta, n, deadline, max_iter=None):
    """Run L-BFGS with given beta and return final z."""
    remaining = deadline - time.monotonic()
    if remaining < 2:
        return z_init, 0
    
    if max_iter is None:
        est_time_per_iter = 0.002 * max(1, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 10))))
    
    def fun_local(z_local):
        return _smoobj(z_local, beta, n)
    
    def jac_local(z_local):
        return _smograd(z_local, beta, n)
    
    try:
        result = minimize(
            fun_local, z_init, jac=jac_local, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxcor': 60, 'maxls': 80}
        )
        return result.x, result.nit
    except Exception:
        return z_init, 0
