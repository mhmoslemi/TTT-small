def _next_pow2(n):

def _make_good_init(n, kind="optimal_taper"):
    """Create initial sequences with good autocorrelation properties."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_sq":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_taper":
        # Slightly sharper than sin, designed to reduce convolution peak
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-t**2 / 2)
        return a / a.sum()
    elif kind == "watt":
        # Waterfall taper
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.5, t*4, (1-t)*4)
        a = a ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        t = np.arange(n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t / (n - 1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.arange(n)
        a = 0.5 * (1 - np.cos(2 * np.pi * t / (n - 1)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "kaiser":
        t = np.linspace(0, 1, n)
        from scipy.special import i0
        beta_k = 5.0
        a = i0(beta_k * np.sqrt(1 - (2*t - 1)**2)) / i0(beta_k)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "binary他们所":
        # Binary pattern based init
        rng = np.random.default_rng(42)
        a = (rng.random(n) > 0.5).astype(float)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _lbfgs_smoothed(z, beta, n):
    """Smoothed objective for L-BFGS."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lbfgs_smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1 : 2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS smoothing, Nesterov-accelerated gradient descent, active-set

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_fn(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad_fn(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoothedProblem:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def fun(self, z):
        return _smoothed_obj_fn(z, self.beta, self.n)

    def jac(self, z):
        return _smoothed_grad_fn(z, self.beta, self.n)


def _lbfgs_optimize(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive beta schedule on log-softmax parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(10, int(remaining / (est_time_per_iter * 20))))

        prob = _SmoothedProblem(beta, n)
        try:
            result = minimize(
                prob.fun, z, jac=prob.jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _accelerated_gd(a_init, deadline):
    """Nesterov-accelerated gradient descent on smoothed max."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    v = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001
    base_beta = max(gap, c_max * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.008, 0.004, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001])

    t_history = [time.monotonic()]
    lr_base = 0.08

    for beta in betas:
        if time.monotonic() > deadline:
            break

        elapsed = time.monotonic() - t_history[-1]
        if elapsed < 0.005:
            elapsed = 0.005
        iters_per_sec = 1.0 / elapsed
        remaining = deadline - time.monotonic()
        max_iter = max(100, int(remaining * iters_per_sec * 0.6))
        max_iter = min(max_iter, 80000)

        t_start = time.monotonic()
        prev_momentum = 0.0

        for i in range(max_iter):
            if time.monotonic() > deadline - 0.05:
                break

            # Nesterov momentum
            gamma = (1.0 + np.sqrt(1.0 + 4.0 * (i + 1) * (i + 1))) / 2.0
            momentum = i / gamma if i > 0 else 0.0
            prev_a = a.copy()
            v = _normalize(v + momentum * (v - prev_a))

            # Compute smoothed gradient at v
            c_v = _full_convolve(v, v)
            c_max_v = float(c_v.max())
            shifted = (c_v - c_max_v) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c_v)
                w[int(np.argmax(c_v))] = 1.0
            else:
                w = e / s_e

            grad = _gradient_w_active(v, w)
            grad_t = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_t)

            if gnorm < 1e-16:
                break

            lr = lr_base / (1.0 + 0.005 * i)
            lr = min(lr, 0.3 / (gnorm + 1e-15))

            a_new = v - lr * grad_t
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                lr *= 0.5
                a_new = v - lr * grad_t
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a = v
                    continue
            a_new = a_new / s_new
            a = a_new

            c_new = _full_convolve(a, a)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()

        t_history.append(time.monotonic())

    return best_a


def _active_set_gd(a_init, deadline, tol_start=0.01, tol_end=0.00002,
                   lr_start=0.06, n_phases=3):
    """Active-set gradient descent with exponential weighting and adaptive tolerance.
    Uses wider line search and tighter convergence criteria."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    tolerances = np.linspace(tol_start, tol_end, n_phases)
    lrs = np.linspace(lr_start, lr_start * 0.2, n_phases)

    for phase in range(n_phases):
        tol_frac = tolerances[phase]
        lr = lrs[phase]
        no_improve = 0
        max_no_improve = 200

        while time.monotonic() < deadline:
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max

            if current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > max_no_improve:
                    break

            # Exponential weighting on active peaks
            tol = tol_frac * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active == 0:
                idx = int(np.argmax(c))
                active = np.zeros_like(c, dtype=bool)
                active[idx] = True
                n_active = 1

            # Exponential weights: peaks closer to max get more weight
            active_vals = c[active]
            weights_active = np.exp(-(c_max - active_vals) / (tol + 1e-30))
            weights_active = weights_active / weights_active.sum()
            w = np.zeros_like(c)
            w[active] = weights_active

            grad = _gradient_w_active(a, w)
            grad_t = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_t)

            if gnorm < 1e-16:
                break

            # Wide line search
            alpha = min(lr, 0.5 / (gnorm + 1e-15))
            improved = False
            for ls in range(80):
                a_try = a - alpha * grad_t
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.4
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                if no_improve % 50 == 0:
                    tol_frac = max(tol_end, tol_frac * 0.7)
                    lr *= 0.6
                if no_improve >= max_no_improve:
                    a = best_a.copy()
                    tol_frac = max(tol_end, tol_frac * 0.5)
                    lr = lrs[phase] * 0.5
                    no_improve = 0
                    if tol_frac <= tol_end * 1.1 and lr < lrs[phase] * 0.1:
                        break
            else:
                if no_improve == 0 and tol_frac > tol_end:
                    tol_frac = max(tol_end, tol_frac * 0.95)
                lr = min(lrs[0], lr * 1.01)

    return best_a, best_sc


def _local_search(a_init, deadline, n_trials=10000):
    """Local search with diverse perturbation strategies."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        mod_type = trial % 5
        a_try = a.copy()

        if mod_type == 0:
            # Single element perturbation
            i = rng.integers(0, n)
            a_try[i] = max(0.0, a_try[i] + rng.normal(0, 0.008))
        elif mod_type == 1:
            # Block perturbation
            i = rng.integers(0, max(1, n - 20))
            block_len = rng.integers(5, 25)
            a_try[i:i + block_len] += rng.normal(0, 0.004, block_len)
        elif mod_type == 2:
            # Multiplicative perturbation on random subset
            idx = rng.choice(n, size=rng.integers(3, 15), replace=False)
            a_try[idx] *= (1.0 + rng.normal(0, 0.06, len(idx)))
            a_try = np.maximum(a_try, 0.0)
        elif mod_type == 3:
            # Fourier mode perturbation
            freq = rng.integers(1, max(2, n // 10))
            amp = rng.uniform(0.002, 0.015)
            phase = rng.uniform(0, 2.0 * np.pi)
            t = np.arange(n)
            a_try = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase))
            a_try = np.maximum(a_try, 0.0)
        else:
            # Swap-based: interchange two random blocks
            i = rng.integers(0, n - 2)
            j = rng.integers(i + 2, n)
            blk = rng.integers(1, min(5, (j - i) // 2 + 1))
            a_try[i:i + blk], a_try[j:j + blk] = a_try[j:j + blk].copy(), a_try[i:i + blk].copy()

        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try.copy()

    return best_a, best_sc


def _shrinkage(a_init, deadline):
    """Try zeroing out small elements to reduce effective length."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            sorted_idx = np.argsort(a)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_init(n, kind="taper"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_sq":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-3.5, 3.5, n)
        a = np.exp(-t ** 2 / 2)
        return a / a.sum()
    elif kind == "hamming":
        t = np.arange(n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t / max(1, n - 1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "watt":
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.5, t * 4, (1 - t) * 4)
        a = a ** 1.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS smoothing -> accelerated GD ->
    active-set GD (coarse to fine) -> local search -> shrinkage ->
    multi-n exploration -> final ultra-tight polish.
    Emphasizes final convergence for small improvements.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 18.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_init(n_parent, "taper")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Init taper n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS (30% budget) ===
    print(f"\n[P1] L-BFGS smoothed (t={time.monotonic()-t_start:.1f}s)")
    t0 = time.monotonic()
    p1_deadline = t_start + 0.30 * (budget_s - 18.0)
    a_main = _lbfgs_optimize(parent, min(p1_deadline, deadline - 500.0), min_remaining=60.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [P1] t={time.monotonic()-t_start:.1f}s")
    print(f"  Done in {time.monotonic()-t0:.1f}s")

    # === Phase 2: Accelerated GD (15% budget) ===
    print(f"\n[P2] Accelerated GD (t={time.monotonic()-t_start:.1f}s)")
    t0 = time.monotonic()
    p2_deadline = t_start + 0.45 * (budget_s - 18.0)

        a_acc = _accelerated_gd(best_seq, p2_deadline)

        if sc_acc < best_score:
            best_seq = a_acc.copy()
            best_score = sc_acc
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [P2] t={time.monotonic()-t_start:.1f}s")
    print(f"  Done in {time.monotonic()-t0:.1f}s")

    # === Phase 3: Active-set GD coarse (15% budget) ===
    print(f"\n[P3] Active-set GD coarse (t={time.monotonic()-t_start:.1f}s)")
    t0 = time.monotonic()
    p3_deadline = t_start + 0.60 * (budget_s - 18.0)
    if time.monotonic() < p3_deadline:
        a_ns, sc_ns = _active_set_gd(best_seq, p3_deadline,
                                     tol_start=0.008, tol_end=0.0005,
                                     lr_start=0.05, n_phases=2)
        if sc_ns < best_score:
            best_seq = a_ns.copy()
            best_score = sc_ns
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [P3] t={time.monotonic()-t_start:.1f}s")
    print(f"  Done in {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS second pass (10% budget) ===

    print(f"  Done in {time.monotonic()-t0:.1f}s")

    # === Phase 7: Shrinkage + re-optimize (remaining time split) ===

    t0 = time.monotonic()
    if time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage(best_seq, min(deadline - 60, time.monotonic() + 30))
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={np.count_nonzero(best_seq)}) [P7 shrink] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 30:
                a_r = _lbfgs_optimize(best_seq, deadline, min_remaining=10.0)
                sc_r = _score(a_r)
                if sc_r < best_score:
                    best_seq = a_r.copy()
                    best_score = sc_r
                    print(f"  NEW BEST: {best_score:.10f} [P7 reopt] t={time.monotonic()-t_start:.1f}s")

    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    t0 = time.monotonic()
    if time.monotonic() < deadline - 2:

                                       tol_start=0.0003, tol_end=0.000005,
                                       lr_start=0.008, n_phases=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P8 active] t={time.monotonic()-t_start:.1f}s")

        remaining = deadline - 1 - time.monotonic()
        if remaining > 4:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmoothedProblem(beta_fin, n)
            max_iter = min(3000, max(20, int(remaining / 0.05)))
            try:
                result = minimize(
                    prob.fun, z, jac=prob.jac, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-15, 'maxcor': 100, 'maxls': 60}
                )
                a_fin2 = _softmax_fn(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [P8 lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One more active-set pass with the tightened solution
        remaining2 = deadline - time.monotonic()
        if remaining2 > 3:
            a_fin3, sc_fin3 = _active_set_gd(best_seq, deadline,
                                             tol_start=0.0001, tol_end=0.000001,
                                             lr_start=0.004, n_phases=2)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [P8 final_active] t={time.monotonic()-t_start:.1f}s")
    print(f"  Done in {time.monotonic()-t0:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_init(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_init(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
