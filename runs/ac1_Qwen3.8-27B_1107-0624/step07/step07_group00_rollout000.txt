from scipy.optimize import linprog
from scipy import sparse

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

```python
"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions,
multi-scale optimization (n=2048, 4096, 8192), and gradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _build_conv_matrix(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, i.e., j <= k <= j+n-1
    # Column j: rows j to j+n-1, values a[0], a[1], ..., a[n-1]
    j_idx = np.repeat(np.arange(n), n)
    k_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    vals = np.tile(a, n)
    return sparse.csc_matrix((vals, (k_idx, j_idx)), shape=(2 * n - 1, n))

def _solve_lp(a):
    n = len(a)
    c_conv = _fft_convolve(a, a)
    M = float(c_conv.max())
    A = _build_conv_matrix(a)
    b = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        g = res.x
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def _line_search(a, g, n_trials=10):
    """Find best t in [0,1] to minimize max(conv((1-t)*a + t*g, (1-t)*a + t*g))"""
    best_t = 0
    best_val = _score_norm(a)
    for t in np.linspace(0.05, 1.0, n_trials):
        a_new = (1 - t) * a + t * g
        a_new = a_new / a_new.sum()
        val = _score_norm(a_new)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t, best_val

def _lp_optimize(a, deadline, max_iters=500, min_improvement=1e-7):
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        g = _solve_lp(a)
        if g is None:
            break
        
        t, val = _line_search(a, g)
        if t > 0 and val < best_score * (1 - min_improvement):
            a_new = (1 - t) * a + t * g
            a = a_new / a_new.sum()
            best_a = a.copy()
            best_score = val
            stagnant = 0
            print(f"  LP iter {it}: t={t:.3f}, score={val:.10f}")
        elif t > 0 and val < best_score:
            a_new = (1 - t) * a + t * g
            a = a_new / a_new.sum()
            best_a = a.copy()
            best_score = val
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 5:
                # Perturb and try again
                rng = np.random.default_rng(it)
                a_pert = a + 0.01 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0)
                a = a_pert / a_pert.sum()
                stagnant = 0
                if stagnant > 10:
                    break
    
    return best_a, best_score

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_smoothed_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

def _gradient_optimize(a, deadline, max_iter=100000, lr_start=0.03, lr_end=0.003):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    t = 0
    patience = 0
    
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    
    while t < max_iter:
        if time.monotonic() > deadline - 3:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac
        T = max_c * 0.003 * (0.1 ** frac)
        
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        g = _compute_smoothed_gradient(a_cur, c, max(T, max_c * 1e-6))
        
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
