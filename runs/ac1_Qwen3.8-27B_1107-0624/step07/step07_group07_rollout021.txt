"""
Step sequence optimizer using LP-based active set method with line search,
combined with log-sum-exp L-BFGS polishing and projected gradient descent.
Key innovation: solves successive LP relaxations of the max-convolution

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_direction(p, c, active_set):
    """Solve LP to find direction that reduces linearized max convolution."""
    n = len(p)
    # Variables: [p_0, ..., p_{n-1}, t]
    # Minimize t
    # Subject to: 2 * sum_m p_old[k-m] * p[m] - t <= c[k] for k in active_set
    #             sum p = 1, p >= 0, t >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    M = len(active_set)
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    
    for i, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # g[m] = p[k-m] for m in [lo, hi]
            # g[lo] = p[k-lo], g[lo+1] = p[k-lo-1], ..., g[hi] = p[k-hi]
            A_ub[i, lo:hi+1] = 2.0 * p[k - hi : k - lo + 1][::-1]
            A_ub[i, -1] = -1.0
        b_ub[i] = c[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n
    bounds.append((0, None))
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            return np.maximum(result.x[:n], 0.0)
    except Exception:
        pass
    return p.copy()

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    if time.monotonic() < deadline - 250:
        a_lbf, sc_lbf = _lbfgs_fine(best_seq, deadline - 250)
        try_update(a_lbf, "lbfgs")
    print(f"  Phase 2 done: {time.monotonic()-t2:.1f}s")
    
    # === Phase 3: Projected Gradient ===
    print(f"\n=== Phase 3: Projected Gradient ===")
    t3 = time.monotonic()
    if time.monotonic() < deadline - 150:
        a_pg, sc_pg = _projected_gradient(best_seq, deadline - 150, max_iters=15000)
        try_update(a_pg, "proj_grad")
    print(f"  Phase 3 done: {time.monotonic()-t3:.1f}s")
    
    # === Phase 4: LP on best (refined) ===
    print(f"\n=== Phase 4: LP Refinement ===")
    t4 = time.monotonic()
    if time.monotonic() < deadline - 100:
        a_lp2, sc_lp2 = _active_set_lp_method(best_seq, deadline - 100, max_outer=100)
        try_update(a_lp2, "lp_refined")
    print(f"  Phase 4 done: {time.monotonic()-t4:.1f}s")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length ===")
    t5 = time.monotonic()
    target_ns = [4096, 8192]
    if start_n < 16384:
        target_ns.append(16384)
    
    for tn in target_ns:
        if time.monotonic() > deadline - 60:
            break
        if tn == start_n:
            continue
        print(f"  Trying n={tn}...")
        a_interp = _interpolate_to_n(best_seq, tn)
        sc_i = _score(a_interp)
        print(f"    Interpolated: {sc_i:.10f}")
        
        # Quick LP on new length
        if time.monotonic() < deadline - 40:
            a_tn, _ = _active_set_lp_method(a_interp, deadline - 40, max_outer=50)
            try_update(a_tn, f"n{tn}_lp")
        
        # Quick L-BFGS
        if time.monotonic() < deadline - 30:
            a_tn2, _ = _lbfgs_fine(best_seq if len(best_seq)==tn else a_tn, deadline - 30)
            if len(a_tn2) == tn:
                try_update(a_tn2, f"n{tn}_lbf")
    
    print(f"  Phase 5 done: {time.monotonic()-t5:.1f}s")
    
    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-Restart ===")
    t6 = time.monotonic()
    restarts = [
        ("gauss_01", 0.01),
        ("gauss_03", 0.03),
        ("gauss_05", 0.05),
    ]
    for name, scale in restarts:
        if time.monotonic() > deadline - 20:
            break
        p_r = _perturb_restart(best_seq, rng, scale)
        sc_r = _score(p_r)
        print(f"  Restart [{name}]: {sc_r:.10f}")
        
        if time.monotonic() < deadline - 15:
            a_r, _ = _active_set_lp_method(p_r, deadline - 15, max_outer=40)
            try_update(a_r, f"restart_{name}")
    
    print(f"  Phase 6 done: {time.monotonic()-t6:.1f}s")
    
    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    t7 = time.monotonic()
    for cycle in range(4):
        if time.monotonic() > deadline - 5:
            break
        
        if time.monotonic() < deadline - 5:
            a_f1, _ = _active_set_lp_method(best_seq, deadline - 5, max_outer=30)
            try_update(a_f1, f"final_lp_{cycle}")
        
        if time.monotonic() < deadline - 5:
            a_f2, _ = _lbfgs_fine(best_seq, deadline - 5)
            try_update(a_f2, f"final_lbf_{cycle}")
        
        if time.monotonic() < deadline - 5:
            a_f3, _ = _projected_gradient(best_seq, deadline - 5, max_iters=5000)
            try_update(a_f3, f"final_pg_{cycle}")
    
    print(f"  Phase 7 done: {time.monotonic()-t7:.1f}s")
    
    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = np.ones(4096) / 4096.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
