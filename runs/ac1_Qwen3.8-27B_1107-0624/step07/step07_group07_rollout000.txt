"""
Improved step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated gradient descent in log-space with adaptive active set,
extended fine L-BFGS beta continuation, block coordinate descent, and
many small-perturbation restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp(c, beta):
    c_max = c.max()
    if c_max < 1e-300:
        return 0.0
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)


def _smoobj_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    val = _logsumexp(c, beta)
    return 2.0 * n * val


def _smograd_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_weights(c, top_frac=0.05, min_count=5):

    c_max = c.max()

        return w

    n_active = min(n_active, len(c))
    # Find threshold

    threshold = sorted_c[n_active - 1]
    active = c >= (c_max - 0.01 * c_max)  # within 1% of max

    if n_act == 0:

        active = np.zeros_like(c, dtype=bool)

    # Weights: proportional to (c_max - c[k] + eps) within active set, or uniform
    # Use uniform for stability
    w = np.zeros_like(c)

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _nesterov_gradient_logspace(a_init, deadline, max_steps=50000, lr_init=0.02):
    """Nesterov-accelerated gradient descent in log-space with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_vel = np.zeros_like(z)
    best_a = a_init.copy()
    best_sc = _score_fast(a_init)
    lr = lr_init
    momentum = 0.9
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        w = _active_set_weights(c, top_frac=0.03, min_count=8)
        grad = _gradient_active(a, c, n, w)
        grad = grad - grad.mean()  # Remove mean to preserve sum constraint approximately

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Nesterov step
        z_vel = momentum * z_vel - lr * grad / np.maximum(g_norm, 1e-15)
        z = z + z_vel
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        # Periodic check
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score_fast(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                lr = min(lr * 1.02, 0.08)
            else:
                stall += 1
                if stall > 6:
                    lr *= 0.8
                    z_vel *= 0.5
                    if lr < 1e-7:
                        break

    a_final = _softmax(z)
    sc_final = _score_fast(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _block_coordinate_descent(a_init, deadline, block_size=4, max_rounds=20000):

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_fast(a)

        if time.monotonic() > deadline:
            break

        # Pick a random block
        start = int(rng.integers(0, max(1, n - block_size)))
        block = np.arange(start, start + block_size)

        s_block = a[block].sum()
        if s_block < 1e-15:
            continue

        # Try rebalancing within the block
        best_frac = None
        best_sc_try = best_sc
        for _ in range(5):
            fracs = rng.dirichlet(np.ones(block_size))
            a_try = a.copy()
            a_try[block] = s_block * fracs
            sc_try = _score_fast(a_try)
            if sc_try < best_sc_try - 1e-14:
                best_sc_try = sc_try
                best_frac = fracs.copy()

        if best_frac is not None:
            a[block] = s_block * best_frac
            best_sc = best_sc_try
            best_a = a.copy()

    return best_a, best_sc


def _peak_shaving_refined(a_init, deadline, max_iters=30000):
    """Refined peak shaving with better adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_fast(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        w = _active_set_weights(c, top_frac=0.02, min_count=5)
        g = _gradient_active(a, c, n, w)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_norm = g / g_norm

        # Backtracking line search
        best_sc_try = _score_fast(a)
        best_a_try = a
        best_eta = eta
        found = False

        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * frac
            a_try = a - step * g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_fast(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_eta = step
                found = True
                break

        if found and best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.005, 0.05)
            stall = 0
        else:
            eta *= 0.92
            stall += 1
            if eta < 1e-9 or stall > 200:
                break

    return best_a, best_sc


def _small_restart_optimize(a_base, rng, n_restarts=30, deadline=None,
                            min_time_remaining=50.0):
    """Many small random restarts with quick L-BFGS polish."""
    n = len(a_base)
    best_a = a_base.copy()
    best_sc = _score_fast(a_base)

    for r in range(n_restarts):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break

        # Small Gaussian perturbation in log-space
        z = np.log(np.maximum(a_base, 1e-15))
        z = z - z.mean()
        scale = 0.003 + 0.015 * (r / max(1, n_restarts))
        z = z + scale * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS polish
        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        beta = mc * 0.001
        max_iter = min(3000, max(200, int(5.0 / (0.001 * (n/2048)**1.4))))
        a_opt, _ = _lbfgs_fine(a_pert, beta, n, max_iter)

        sc = _score_fast(a_opt)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Restart {r}: improved to {sc:.10f}")

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: fine L-BFGS, Nesterov gradient, block CD,
    small restarts, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score_fast(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except (NameError, Exception):
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score_fast(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(8192) / 8192.0
        print(f"No parent. Using uniform n=8192, score={_score_fast(a0):.10f}")
        try_update(a0, "init")

    start_n = len(best_seq)

    # === Phase 1: Extended fine L-BFGS ===
    print(f"\n=== Phase 1: Extended L-BFGS (n={start_n}) ===")
    t_p = time.monotonic()
    a_p1 = _lbfgs_extended_schedule(best_seq, deadline - 500, min_time_remaining=400.0)
    try_update(a_p1, "lbfgs_ext")
    print(f"  Phase 1: {time.monotonic()-t_p:.1f}s, score={_score_fast(a_p1):.10f}")

    # === Phase 2: Nesterov gradient in log-space ===
    print(f"\n=== Phase 2: Nesterov gradient ===")
    t_p = time.monotonic()
    if time.monotonic() < deadline - 350:
        a_n, sc_n = _nesterov_gradient_logspace(best_seq, deadline - 350,
                                                max_steps=40000, lr_init=0.015)
        try_update(a_n, "nesterov")
    print(f"  Phase 2: {time.monotonic()-t_p:.1f}s")

    # === Phase 3: Refined peak shaving ===
    print(f"\n=== Phase 3: Peak shaving ===")
    t_p = time.monotonic()
    if time.monotonic() < deadline - 250:
        a_ps, sc_ps = _peak_shaving_refined(best_seq, deadline - 250, max_iters=25000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 3: {time.monotonic()-t_p:.1f}s")

    # === Phase 4: Block coordinate descent ===
    print(f"\n=== Phase 4: Block CD ===")
    t_p = time.monotonic()
    if time.monotonic() < deadline - 200:
        a_cd, sc_cd = _block_coordinate_descent(best_seq, deadline - 200,
                                                block_size=6, max_rounds=15000)
        try_update(a_cd, "block_cd")
    print(f"  Phase 4: {time.monotonic()-t_p:.1f}s")

    # === Phase 5: Small random restarts ===
    print(f"\n=== Phase 5: Small restarts ===")
    t_p = time.monotonic()
    if time.monotonic() < deadline - 150:
        n_restarts = 20
        a_sr, sc_sr = _small_restart_optimize(best_seq, rng, n_restarts=n_restarts,
                                              deadline=deadline, min_time_remaining=100.0)
        try_update(a_sr, "small_restarts")
    print(f"  Phase 5: {time.monotonic()-t_p:.1f}s")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    t_p = time.monotonic()
    target_lengths = []
    if start_n < 12288:
        target_lengths.append(12288)
    if start_n < 10240:
        target_lengths.append(10240)

    for target_n in target_lengths:
        if time.monotonic() > deadline - 80:
            break
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score_fast(a_interp)
        print(f"    Interpolated: score={sc_interp:.10f}")
        try_update(a_interp, f"interp_n{target_n}")

        a_opt = _lbfgs_extended_schedule(a_interp, deadline - 60, min_time_remaining=50.0)
        try_update(a_opt, f"n{target_n}_lbfgs")
    print(f"  Phase 6: {time.monotonic()-t_p:.1f}s")

    # === Phase 7: Final ultra-fine L-BFGS ===
    print(f"\n=== Phase 7: Ultra-fine L-BFGS ===")
    t_p = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 15:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        ultra_betas = mb * np.array([5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8, 5e-9])
        for i, beta in enumerate(ultra_betas):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - 10 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(6000, max(300, int(remaining / 0.002)))
            a_ultra, _ = _lbfgs_fine(best_seq, beta, n, max_iter)
            try_update(a_ultra, f"ultra_{i}")
    print(f"  Phase 7: {time.monotonic()-t_p:.1f}s")

    # === Phase 8: Final peak shave + Nesterov ===
    print(f"\n=== Phase 8: Final polish ===")
    t_p = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_f1, _ = _peak_shaving_refined(best_seq, deadline - 5, max_iters=5000)
        try_update(a_f1, "final_ps")

        if time.monotonic() < deadline - 5:
            a_f2, _ = _nesterov_gradient_logspace(best_seq, deadline - 5,
                                                  max_steps=5000, lr_init=0.005)
            try_update(a_f2, "final_nest")
    print(f"  Phase 8: {time.monotonic()-t_p:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(8192) / 8192.0
        sc_final = _score_fast(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(8192) / 8192.0
    print(f"FALLBACK: score={_score_fast(fallback):.10f}")
    return [float(x) for x in fallback]
