def _cg_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H x = rhs where H[j,m] = W[j+m]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rs0 = rs
    if rs0 < 1e-300:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = np.dot(p, Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = np.dot(r, r)
        if rs_new < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x

"""
Active-set QP optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated Hankel matrix-vector products, CG for QP solves,
log-sum-exp L-BFGS warmup, and iterative active-set refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _hankel_matvec(v, W, n):
    """Compute H*v where H[j,m] = W[j+m]. W has length 2n-1, v has length n."""
    u = v[::-1]
    out_len = n + (2 * n - 1) - 1
    N = _next_pow2(out_len)
    Fu = np.fft.rfft(u, N)
    FW = np.fft.rfft(W, N)
    conv = np.fft.irfft(Fu * FW, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_hankel(W, n, rhs, free_mask=None, max_iter=3000, tol=1e-15):
    """
    Solve H_FF x = rhs_free where H[j,m] = W[j+m],
    using only the free variables (if free_mask provided).
    """
    if free_mask is not None:
        n_free = int(free_mask.sum())
        if n_free == 0:
            return np.zeros(0)
        rhs_f = rhs[free_mask]
        
        def matvec(v):
            v_full = np.zeros(n)
            v_full[free_mask] = v
            return _hankel_matvec(v_full, W, n)[free_mask]
    else:
        n_free = n
        rhs_f = rhs
        matvec = lambda v: _hankel_matvec(v, W, n)
    
    x = np.zeros(n_free)
    r = rhs_f.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    rs0 = rs
    if rs0 < 1e-300:
        return x
    for i in range(max_iter):
        Hp = matvec(p)
        pHp = float(np.dot(p, Hp))
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(np.dot(r, r))
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation on log-space."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)

    # Aggressive schedule
    num_betas = min(40, max(20, int((deadline - min_time_remaining - time.monotonic()) / 10)))
    ratios = np.exp(np.linspace(0, np.log(1e-8), num_betas))
    betas = base_beta * ratios

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 20))))

        try:
            result = minimize(
                _make_smoobj(beta, n),
                z,
                jac=_make_smograd(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            sc_cur = _score(current_a)
            if sc_new < sc_cur - 1e-15:
                current_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

```python
"""
QP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated Hankel matvecs, CG for exact QP solves,
log-sum-exp continuation for active set identification, and
iterative active set refinement with support handling.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _hankel_matvec(v, W, n):
    """Compute H*v where H[j,m] = W[j+m]. W has length 2n-1, v has length n."""
    u = v[::-1]
    out_len = n + (2 * n - 1) - 1
    N = _next_pow2(out_len)
    Fu = np.fft.rfft(u, N)
    FW = np.fft.rfft(W, N)
    conv = np.fft.irfft(Fu * FW, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, reg=0.0, max_iter=5000, tol=1e-14):
    """Solve (H + reg*I) x = rhs using CG. H[j,m] = W[j+m]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    rs0 = rs
    if rs0 < 1e-300:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(np.dot(p, Hp))
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(np.dot(r, r))
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a
