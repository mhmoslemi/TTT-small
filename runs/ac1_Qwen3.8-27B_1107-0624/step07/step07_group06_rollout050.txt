    grad_a = 4.0 * n * r[n - 1:2 * n - 1]
    
    # Chain rule: dz -> da (softmax)
    # da_m/dz_j = a_m * (delta_mj - a_j)
    # grad_z_j = sum_m grad_a_m * a_m * (delta_mj - a_j)
    # = a_j * grad_a_j - a_j * sum_m grad_a_m * a_m
    # = a_j * (grad_a_j - dot(grad_a, a))
    
    W = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W)
    return grad_z


def _lse_obj(z, beta, n):
    """Log-sum-exp smoothed max."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    """Compute H*v where (Hv)_i = 2 * sum_j v[j] * active[i+j]."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _qp_active_set(a_init, active, n, mu=1e-6, max_cg=200):
    """Solve the QP: min a^T H a s.t. sum(a)=1, a>=0, where H = sum_{k in active} T_k."""
    # For now, use Newton step with projection
    # The unconstrained solution (ignoring a>=0) is:
    # H a + lambda * 1 = 0, 1^T a = 1
    # => a = -H^{-1} 1 / (1^T H^{-1} 1)
    # With mu regularization for numerical stability
    def hess_vec(v):
        return _hessian_matvec(v, active, n) + mu * v
    
    one = np.ones(n)
    x = _cg_solve(hess_vec, -one, n, max_iter=max_cg)
    s = x.sum()
    if abs(s) < 1e-30 or s < 0:
        # Fallback: project a_init onto sum=1
        return a_init / a_init.sum()
    a_qp = x / s
    
    # Project onto non-negative orthant (simple projection)
    a_qp = np.maximum(a_qp, 0)
    s2 = a_qp.sum()
    if s2 < 1e-15:
        return a_init / a_init.sum()
    a_qp = a_qp / s2
    return a_qp


def _newton_active(a_init, deadline, max_iter=200, tol_frac=0.001):
    """Newton method on active set with QP subproblem."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Solve QP on active set
        a_qp = _qp_active_set(a, active, n)
        
        # Line search: a_new = (1-t)*a + t*a_qp
        c_new = c.copy()
        f_old = float(c[active].sum())
        best_t = 0.0
        best_f = f_old
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            a_try = (1.0 - t) * a + t * a_qp
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < best_f:
                best_f = f_try
                best_t = t
                c_new = c_try
        
        if best_t == 0.0:
            # Try the QP solution directly
            a_try = a_qp
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
            break

        a = (1.0 - best_t) * a + best_t * a_qp
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Check convergence
        if best_t < 0.01 and it > 10:
            break

    return best_a, best_sc


def _run_lbfgs_lse(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing and aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    top_k = min(20, len(c_sorted))
    gap = c_sorted[0] - c_sorted[top_k - 1] if top_k > 1 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                   0.01, 0.005, 0.002, 0.001,
                                   0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        max_iter = min(5000, max(30, int(remaining / 0.05)))

        def fun_local(z_local):
            return _lse_obj(z_local, beta, n)

        def jac_local(z_local):
            return _lse_grad(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Try power mean at this point
                pass
        except Exception:
            pass

    return current_a


def _run_power_mean(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with power mean smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    ps = [4, 8, 16, 32, 64, 128, 256, 512, 1024]

    for p in ps:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        max_iter = min(3000, max(20, int(remaining / 0.1)))

        def fun_pm(z_local):
            return _power_mean_obj(z_local, p, n)

        def jac_pm(z_local):
            return _power_mean_grad(z_local, p, n)

        try:
            result = minimize(
                fun_pm, z, jac=jac_pm, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _symmetric_optimize(a_init, deadline, min_time_remaining=30.0):
    """Optimize under symmetry constraint a[i] = a[n-1-i]."""
    n = len(a_init)
    half = n // 2
    # Extract half (including middle if n is odd)
    a_half = a_init[:half + (1 if n % 2 == 1 else 0)].copy()
    
    def expand(h):
        if n % 2 == 0:
            return np.concatenate([h, h[::-1]])
        else:
            return np.concatenate([h, h[-2::-1]])
    
    def score_sym(h):
        a_full = expand(h)
        a_full = np.maximum(a_full, 0)
        s = a_full.sum()
        if s < 1e-30:
            return np.inf
        a_full = a_full / s
        return _score(a_full)
    
    # Use L-BFGS on the half
    best_h = a_half.copy()
    best_sc = score_sym(best_h)
    
    z_half = np.log(np.maximum(best_h, 1e-30))
    
    def fun_sym(z_h):
        h = _softmax(z_h)
        return score_sym(h)
    
    def jac_sym(z_h):
        # Numerical gradient (expensive but works)
        h = _softmax(z_h)
        f0 = score_sym(h)
        grad = np.zeros_like(z_h)
        eps = 1e-7
        for j in range(len(z_h)):
            z_pert = z_h.copy()
            z_pert[j] += eps
            h_pert = _softmax(z_pert)
            f_pert = score_sym(h_pert)
            grad[j] = (f_pert - f0) / eps
        return grad
    
    # Actually numerical gradient is too expensive for half=4096.
    # Let me use a simpler approach: just do L-BFGS with analytic gradient
    # computed on the full sequence, then project to symmetric.
    
    # Alternative: just run the regular optimizer and then symmetrize
    a_opt = _run_lbfgs_lse(a_init, deadline, min_time_remaining=min_time_remaining)
    a_sym = 0.5 * (a_opt + a_opt[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    return a_sym


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "trunc_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.25 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.01):
    """Subgradient descent with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance
        c_sorted = np.sort(c)[::-1]
        tol = max(0.0005 * c_max, c_sorted[min(5, len(c_sorted)-1)] - c_max + 1e-10)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        step = lr / (1.0 + 0.005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage(a, deadline):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.05, 0.10, 0.20]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-technique optimizer: L-BFGS (LSE + power mean), Newton active-set QP,
    subgradient polish, shrinkage, multi-n exploration, and symmetric processing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS with LSE smoothing
    print(f"\n[Phase 1] L-BFGS LSE (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _run_lbfgs_lse(best_seq, min(deadline - 600, time.monotonic() + 300), min_time_remaining=600)
        update_best(a1, _score(a1), "lbfgs_lse")

    # Phase 2: Newton active-set QP
    print(f"\n[Phase 2] Newton QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _newton_active(best_seq, min(deadline - 500, time.monotonic() + 200), max_iter=300, tol_frac=0.001)
        update_best(a2, sc2, "newton_qp")
        # Try with different tolerance
        if time.monotonic() < deadline - 450:
            a2b, sc2b = _newton_active(best_seq, min(deadline - 450, time.monotonic() + 150), max_iter=200, tol_frac=0.005)
            update_best(a2b, sc2b, "newton_qp_wide")

    # Phase 3: Power mean L-BFGS
    print(f"\n[Phase 3] Power mean (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3 = _run_power_mean(best_seq, min(deadline - 400, time.monotonic() + 200), min_time_remaining=400)
        update_best(a3, _score(a3), "power_mean")

    # Phase 4: Subgradient polish
    print(f"\n[Phase 4] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _subgrad_polish(best_seq, min(deadline - 300, time.monotonic() + 150), n_max_steps=5000, lr_init=0.008)
        update_best(a4, sc4, "subgrad")

    # Phase 5: Shrinkage + re-optimize
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _shrinkage(best_seq, deadline - 250)
        update_best(a5, sc5, "shrinkage")
        if sc5 < best_score * 1.001 and time.monotonic() < deadline - 200:
            a5b = _run_lbfgs_lse(a5, min(deadline - 200, time.monotonic() + 100), min_time_remaining=200)
            update_best(a5b, _score(a5b), "lbfgs_post_shrink")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_tried = {len(best_seq)} if best_seq is not None else set()
    n_candidates = [4096, 8192, 12288, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    
    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 100:
            break
        if n_try in n_tried:
            continue
        n_tried.add(n_try)
        
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
        else:
            a_init = best_seq.copy() if best_seq is not None else _make_profile(n_try, "tapered")
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        a_opt = _run_lbfgs_lse(a_init, deadline, min_time_remaining=60)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        
        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _newton_active(a_opt, min(deadline - 50, time.monotonic() + 60), max_iter=100, tol_frac=0.002)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # Phase 7: Symmetric processing
    print(f"\n[Phase 7] Symmetric (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = np.maximum(a_sym, 0)
        a_sym = a_sym / a_sym.sum()
        sc_sym = _score(a_sym)
        update_best(a_sym, sc_sym, "symmetric_init")
        if sc_sym < best_score * 1.01 and time.monotonic() < deadline - 30:
            a_sym_opt = _run_lbfgs_lse(a_sym, min(deadline - 20, time.monotonic() + 50), min_time_remaining=20)
            update_best(a_sym_opt, _score(a_sym_opt), "symmetric_opt")

    # Phase 8: Final combined polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Newton on best
        a_fin, sc_fin = _newton_active(best_seq, max(deadline - 3, time.monotonic() + 10), max_iter=100, tol_frac=0.0005)
        update_best(a_fin, sc_fin, "final_newton")
        
        # Subgradient
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, max(deadline - 2, time.monotonic()), n_max_steps=3000, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
