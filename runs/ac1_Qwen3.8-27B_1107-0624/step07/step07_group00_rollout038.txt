def _next_pow2(n):
    k = 1
    while k < n:

def _fft_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    return e / s

            if sc_ema < ema_best_score:
                ema_best_score = sc_ema
                ema_best_a = ema_a.copy()

            # Use the better of current and EMA
            effective_best = min(best_score, ema_best_score)
            if effective_best < best_patience * (1 - 1e-6):
                best_patience = effective_best
                patience = 0
            elif patience > 20:
                # Reset from best
                src = best_a if best_score <= ema_best_score else ema_best_a
                z = np.log(np.maximum(src, 1e-15))
                z -= z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                ema_a = src.copy()
                patience = 0

    # Final evaluation
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    sc_ema_final = _score_norm(ema_a)
    if sc_ema_final < best_score:
        best_score = sc_ema_final
        best_a = ema_a.copy()

    return best_a, best_score


def _lp_polish(a, deadline, max_iters=200, K=6):
    """LP-based Frank-Wolfe polishing."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    a_cur = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a_cur, a_cur)
        c_max = float(c.max())
        M = c_max

        # Find active lags (top K)
        active_lags = np.argsort(c)[::-1][:K]

        # Solve LP for direction
        b_dir = _lp_dual_direction(a_cur, M, active_lags, max_K=K)
        if b_dir is None:
            # Fallback: use subgradient
            g = _compute_subgradient(a_cur, c, M * 0.01)
            b_dir = np.maximum(1.0 / np.maximum(g / (2*n) + 1e-10, 0.1), 0)
            b_dir = b_dir / b_dir.sum()

        # Line search / step size
        best_t = 0
        best_sc = _score_norm(a_cur)
        for t_try in [0.1, 0.2, 0.3, 0.5, 0.7]:
            cand = (1 - t_try) * a_cur + t_try * b_dir
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-15:
                continue
            cand = cand / s
            sc = _score_norm(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_try

        if best_t > 0:
            a_cur = (1 - best_t) * a_cur + best_t * b_dir
            a_cur = np.maximum(a_cur, 0)
            a_cur = a_cur / a_cur.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a_cur.copy()
        else:
            # Try smaller steps
            for t_try in [0.05, 0.02, 0.01]:
                cand = (1 - t_try) * a_cur + t_try * b_dir
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand = cand / s
                sc = _score_norm(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
                    a_cur = cand
                    break
            else:
                # No improvement, break
                if it > 10:
                    break

    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=500):
    """Fast coordinate descent on active constraints."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c >= tol)[0]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for k in active:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue
            for j in js:
                for scale in [1.1, 1.2, 0.8, 0.7, 0.5, 1.5]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    if a_cand[j] < 0:
                        continue
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c >= tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Relax tolerance and try again
            tol = c_max * 1.002
            active = np.where(c >= tol)[0]
            if len(active) == 0:
                break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, method, rng):
    """Generate structured initial sequences."""
    if method == 0:
        # Uniform
        a = np.ones(n) / n
    elif method == 1:
        # First harmonic (cosine)
        i = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * i / n)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif method == 2:
        # Edge-weighted (inverse of first harmonic)
        i = np.arange(n)
        mid = n / 2.0
        a = 1.0 + 0.3 * np.cos(np.pi * (i - mid) / mid)
        a = np.maximum(a, 0.01)
        a /= a.sum()
    elif method == 3:
        # Two-block asymmetric
        a = np.zeros(n)
        a[:n // 2] = 1.0
        a[n // 2:] = 0.3
        a = np.maximum(a, 1e-6)
        a /= a.sum()
    elif method == 4:
        # Random log-normal
        a = np.exp(rng.normal(0, 0.5, n))
        a /= a.sum()
    elif method == 5:
        # Triangular (peak at edge)
        i = np.arange(n)
        a = 1.0 - i / n
        a = np.maximum(a, 0.01)
        a /= a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimizer: Adam subgradient with EMA averaging, LP polish,
    coordinate descent, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_parent = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a_parent = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a_parent)
    best_a = a_parent.copy()
    best_score = _score(a_parent)
    print(f"Initial: n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Adam subgradient on parent (200s) ===
    phase1_end = min(deadline, t_start + 200.0)
    c_p = _fft_convolve(a_parent, a_parent)
    max_cp = float(c_p.max())
    a_opt, sc = _adam_subgradient(
        a_parent, phase1_end,
        T_start=0.005 * max_cp, T_end=0.0001 * max_cp,
        lr_start=0.03, lr_end=0.002,
        max_iter=300000, avg_decay=0.9995, eval_interval=100)
    full_sc = _score_full(a_opt)
    print(f"Phase 1 Adam (n={n_parent}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # === Phase 2: LP polish on best (200s - 400s) ===
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        a_lp, sc_lp = _lp_polish(best_a, phase2_end, max_iters=100, K=6)
        full_sc_lp = _score_full(a_lp)
        print(f"Phase 2 LP polish: score={full_sc_lp:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_lp) and full_sc_lp < best_score:
            best_score = full_sc_lp
            best_a = a_lp.copy()

    # === Phase 3: CD polish (400s - 500s) ===
    phase3_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase3_end:
        a_cd, sc_cd = _cd_polish(best_a, phase3_end, max_rounds=200)
        full_sc_cd = _score_full(a_cd)
        print(f"Phase 3 CD: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # === Phase 4: Try n=4096 (500s - 900s) ===
    phase4_end = min(deadline, t_start + 900.0)
    n4 = 4096
    if n4 != n_parent and time.monotonic() < phase4_end:
        a_4k = _interp_to_length(best_a, n4)
        c_4k = _fft_convolve(a_4k, a_4k)
        max_4k = float(c_4k.max())
        mid4 = min(phase4_end, t_start + 750.0)
        a_4k_opt, sc_4k = _adam_subgradient(
            a_4k, mid4,
            T_start=0.004 * max_4k, T_end=0.0001 * max_4k,
            lr_start=0.025, lr_end=0.002,
            max_iter=200000, avg_decay=0.9995, eval_interval=150)
        full_4k = _score_full(a_4k_opt)
        print(f"Phase 4 Adam n=4096: score={full_4k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_4k) and full_4k < best_score:
            best_score = full_4k
            best_a = a_4k_opt.copy()

        # LP polish at 4096
        if time.monotonic() < phase4_end - 30:
            a_4k_lp, _ = _lp_polish(best_a, phase4_end - 5, max_iters=50, K=6)
            sc_4k_lp = _score_full(a_4k_lp)
            print(f"Phase 4 LP n=4096: score={sc_4k_lp:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_4k_lp) and sc_4k_lp < best_score:
                best_score = sc_4k_lp
                best_a = a_4k_lp.copy()

    # === Phase 5: Try n=16384 (900s - 1200s) ===
    phase5_end = min(deadline, t_start + 1200.0)
    n5 = 16384
    if time.monotonic() < phase5_end:
        a_16k = _interp_to_length(best_a, n5)
        c_16k = _fft_convolve(a_16k, a_16k)
        max_16k = float(c_16k.max())
        a_16k_opt, sc_16k = _adam_subgradient(
            a_16k, phase5_end - 10,
            T_start=0.003 * max_16k, T_end=0.00005 * max_16k,
            lr_start=0.02, lr_end=0.001,
            max_iter=100000, avg_decay=0.9999, eval_interval=200)
        full_16k = _score_full(a_16k_opt)
        print(f"Phase 5 n=16384: score={full_16k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_16k) and full_16k < best_score:
            best_score = full_16k
            best_a = a_16k_opt.copy()

    # === Phase 6: Fine-tune best with very low temp (1200s - 1550s) ===
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end:
        n_b = len(best_a)
        c_b = _fft_convolve(best_a, best_a)
        max_cb = float(c_b.max())
        a_fine, sc_fine = _adam_subgradient(
            best_a, phase6_end,
            T_start=0.001 * max_cb, T_end=0.00001 * max_cb,
            lr_start=0.008, lr_end=0.0005,
            max_iter=400000, avg_decay=0.9999, eval_interval=80)
        full_fine = _score_full(a_fine)
        print(f"Phase 6 fine-tune: score={full_fine:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_fine) and full_fine < best_score:
            best_score = full_fine
            best_a = a_fine.copy()

        # LP polish
        if time.monotonic() < phase6_end - 40:
            a_lp2, _ = _lp_polish(best_a, phase6_end - 5, max_iters=80, K=5)
            sc_lp2 = _score_full(a_lp2)
            print(f"Phase 6 LP2: score={sc_lp2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_a = a_lp2.copy()

    # === Phase 7: Structured restarts (1550s - 1700s) ===
    phase7_end = min(deadline, t_start + 1700.0)
    n_fin = len(best_a)
    if time.monotonic() < phase7_end:
        for attempt in range(3):
            if time.monotonic() > phase7_end - 50:
                break
            # Perturb in log-space
            z = np.log(np.maximum(best_a, 1e-15))
            sigma = 0.01 + 0.005 * attempt
            z = z + sigma * rng.standard_normal(n_fin)
            z -= z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(phase7_end - 10, time.monotonic() + max(30, (phase7_end - time.monotonic()) / (3 - attempt) - 10))
            a_pert_opt, _ = _adam_subgradient(
                a_pert, sub_dl,
                T_start=0.001 * max_cp, T_end=0.00001 * max_cp,
                lr_start=0.015, lr_end=0.001,
                max_iter=80000, avg_decay=0.999, eval_interval=100)
            sc_pert = _score_full(a_pert_opt)
            print(f"  Restart[{attempt}]: score={sc_pert:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_pert) and sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert_opt.copy()

    # === Phase 8: Final ultra-fine polish (1700s - deadline) ===
    if time.monotonic() < deadline - 15:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _adam_subgradient(
            best_a, deadline - 20,
            T_start=0.0003 * max_cf, T_end=0.000005 * max_cf,
            lr_start=0.004, lr_end=0.0003,
            max_iter=200000, avg_decay=0.99995, eval_interval=50)
        sc_final_adam = _score_full(a_final)
        print(f"Phase 8 Adam final: score={sc_final_adam:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc_final_adam) and sc_final_adam < best_score:
            best_score = sc_final_adam
            best_a = a_final.copy()

        # Quick CD
        if time.monotonic() < deadline - 5:
            a_cd_f, _ = _cd_polish(best_a, deadline - 3, max_rounds=50)
            sc_cd_f = _score_full(a_cd_f)
            if sc_cd_f < best_score:
                best_score = sc_cd_f
                best_a = a_cd_f.copy()
            print(f"Phase 8 CD final: score={sc_cd_f:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_check = _score_full(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
