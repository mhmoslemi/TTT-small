"""
Optimizer combining KKT fixed-point linear solves with L-BFGS polishing.
Key innovation: directly solve the KKT linear system (Hankel matrix) for
the active set, then iterate to refine the active set, and use L-BFGS
as a polishing step. This avoids the slow convergence of pure gradient
methods by taking exact steps when the active set is known.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import coo_matrix
from scipy.sparse.linalg import spsolve


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.2)
        max_iter = min(15000, max(100, int(remaining / (0.004 * scale * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5003:
            break

    return current_a


def _kkt_solve(a_init, k_active=5, max_inner_iters=50):
    """
    Solve the KKT linear system for a given active set.
    Returns (a_kkt, is_valid) where is_valid means KKT conditions are met.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    for outer in range(max_inner_iters):
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find active set: top k lags
        active_lags = np.argsort(c)[::-1][:k_active]
        active_set = set(int(x) for x in active_lags)

        # Build sparse KKT matrix
        # System: H*a - lambda*1 = 0, sum(a) = 1
        # H[i,j] = (1/k) if (i+j) in active_set, else 0
        rows = []
        cols = []
        vals = []
        inv_k = 1.0 / len(active_lags)

        for i in range(n):
            for p in active_lags:
                j = int(p) - i
                if 0 <= j < n:
                    rows.append(i)
                    cols.append(j)
                    vals.append(inv_k)
            rows.append(i)
            cols.append(n)
            vals.append(-1.0)

        # Last row: sum(a) = 1
        for j in range(n):
            rows.append(n)
            cols.append(j)
            vals.append(1.0)

        A = coo_matrix((vals, (rows, cols)), shape=(n + 1, n + 1)).tocsc()
        b = np.zeros(n + 1)
        b[n] = 1.0

        try:
            sol = spsolve(A, b)
        except Exception:
            return a, False

        a_new = sol[:n]
        lam = sol[n]

        # Check non-negativity
        min_val = float(a_new.min())
        if min_val < -1e-10:
            # Try NNLS-style: zero out negative entries and re-solve
            # For simplicity, just clip and return (not a valid KKT point)
            a_clipped = np.maximum(a_new, 0.0)
            s = a_clipped.sum()
            if s < 1e-15:
                return a, False
            a_clipped = a_clipped / s
            return a_clipped, False

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            return a, False
        a_new = a_new / s

        # Verify KKT: check that non-active lags don't exceed active lags
        c_new = _full_convolve(a_new, a_new)
        max_active = float(c_new[active_lags].max())
        max_inactive = float(np.max(np.delete(c_new, active_lags)))

        if max_inactive <= max_active + 1e-10:
            # KKT conditions satisfied!
            return a_new, True

        # Update a and iterate (the active set might change)
        a = a_new

    # Check one more time if valid
    c = _full_convolve(a, a)
    c_max = float(c.max())
    active_lags = np.argsort(c)[::-1][:k_active]
    max_active = float(c[active_lags].max())
    max_inactive = float(np.max(np.delete(c, active_lags)))
    is_valid = max_inactive <= max_active + 1e-8
    return a, is_valid


def _kkt_iterate(a_init, deadline, k_values=(3, 5, 7, 10, 15, 20, 30)):
    """
    Try KKT solve with different active set sizes.
    Returns best (a, score) found.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for k in k_values:
        if time.monotonic() > deadline:
            break
        if k > 2 * n - 3:
            continue

        a_try, valid = _kkt_solve(a_init, k_active=k, max_inner_iters=30)
        sc = _score(a_try)

        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
            print(f"    KKT k={k}: score={sc:.10f} valid={valid}")

            # Use this as new starting point for larger k
            a_init = a_try.copy()

        if valid and sc < 1.502:
            break

    return best_a, best_sc


def _kkt_with_support(a_init, k_active=5, deadline=None):
    """
    KKT solve that handles the support (some a[i] may be zero).
    Iteratively fixes negative variables to zero.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    support = set(range(n))

    for outer in range(30):
        if deadline and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        active_lags = np.argsort(c)[::-1][:k_active]
        active_list = [int(x) for x in active_lags]
        inv_k = 1.0 / len(active_list)

        # Build system on the support
        support_list = sorted(support)
        m = len(support_list)
        if m == 0:
            break
        idx_map = {v: i for i, v in enumerate(support_list)}

        rows = []
        cols = []
        vals = []

        for ii, i in enumerate(support_list):
            for p in active_list:
                j = p - i
                if j in idx_map:
                    rows.append(ii)
                    cols.append(idx_map[j])
                    vals.append(inv_k)
            rows.append(ii)
            cols.append(m)
            vals.append(-1.0)

        for jj in range(m):
            rows.append(m)
            cols.append(jj)
            vals.append(1.0)

        A = coo_matrix((vals, (rows, cols)), shape=(m + 1, m + 1)).tocsc()
        b = np.zeros(m + 1)
        b[m] = 1.0

        try:
            sol = spsolve(A, b)
        except Exception:
            break

        sol_support = sol[:m]
        lam = sol[m]

        # Check for negative entries
        neg_indices = [support_list[i] for i in range(m) if sol_support[i] < -1e-8]

        if neg_indices:
            # Remove these from support
            for idx in neg_indices:
                support.discard(idx)
            # Update a: set removed indices to 0
            for idx in neg_indices:
                a[idx] = 0.0
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            continue

        # All non-negative: construct full solution
        a_new = np.zeros(n)
        for ii, i in enumerate(support_list):
            a_new[i] = max(0.0, sol_support[ii])
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Verify KKT
        c_new = _full_convolve(a_new, a_new)
        max_active = float(c_new[active_lags].max())
        # Check non-active lags
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[active_lags] = False
        max_inactive = float(c_new[mask].max()) if mask.any() else 0.0

        if max_inactive <= max_active + 1e-9:
            # Also check dual feasibility: for i not in support,
            # the gradient should be <= lambda
            grad = np.zeros(n)
            for ii, i in enumerate(support_list):
                for p in active_list:
                    j = p - i
                    if j in idx_map:
                        grad[i] += inv_k * a_new[j]
            # For i not in support: grad[i] <= lambda
            if all(i in support for i in range(n)):
                return a_new, True, lam
            else:
                non_support = [i for i in range(n) if i not in support]
                if non_support:
                    max_non_sup_grad = max(abs(grad[i]) for i in non_support)
                    # This is a simplified check
                return a_new, True, lam
        else:
            # Not valid: update a and try again (active set may change)
            a = a_new.copy()

    return a, False, 0.0


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=50000, lr_init=0.06):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    stall_threshold = max(10, k_active * 3)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = lr * a.sum()
        z = z - step * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= 0.6
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _direct_subgrad(a_init, deadline, k_active=3, max_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        a_try = a - step_size * g / np.sqrt(n)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        sc_cur = _score(a)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        if sc_try < sc_cur - 1e-16:
            a = a_try
            step_size = min(step_size * 1.01, 0.5)
        else:
            stall += 1
            step_size *= 0.92

        if stall > 300:
            break

    return best_a, best_sc


def _basin_hop(a_current, deadline, rng, n_hops=5, reserve=10.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(30.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.7

        pert_type = h % 3
        if pert_type == 0:
            sigma = 0.05 + 0.03 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 16)))
                amp = 0.02 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 3 + h
            block_sz = n // n_blocks
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.05, 0.05)
            z = z - z.mean()
            a_pert = _softmax(z)

        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.01
        a_opt = _lbfgs_smooth(a_pert, beta, n, 3000)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    1. KKT fixed-point linear solves (exact for given active set)
    2. Adaptive L-BFGS with log-sum-exp smoothing
    3. Subgradient polishing
    4. Multi-length exploration
    5. Basin hopping
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        best_seq = _make_chirp(4096, freq=1.0, eps=0.3)
        best_score_val = _score(best_seq)
        best_n = len(best_seq)
        print(f"  No parent, using chirp init: score={best_score_val:.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: KKT Fixed-Point Solver ===
    print(f"\n=== Phase 2: KKT Solver (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 600:
        kkt_deadline = deadline - 600
        a_kkt, sc_kkt = _kkt_iterate(best_seq, kkt_deadline,
                                     k_values=(3, 5, 7, 10, 15, 20))
        try_update(a_kkt, "kkt_iterate")

        # Also try with support handling for smaller k
        if time.monotonic() < deadline - 500:
            for k in (3, 5, 7):
                if time.monotonic() > deadline - 500:
                    break
                a_k, valid, lam = _kkt_with_support(a_kkt, k_active=k,
                                                     deadline=deadline - 500)
                try_update(a_k, f"kkt_support_k{k}_{'valid' if valid else 'invalid'}")

    # === Phase 3: L-BFGS Polish ===
    print(f"\n=== Phase 3: L-BFGS Polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_p3 = _adaptive_lbfgs(best_seq, deadline, reserve_time=300.0)
        try_update(a_p3, "lbfgs_polish")

    # === Phase 4: Subgradient ===
    print(f"\n=== Phase 4: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, _ = _subgrad_polish(best_seq, deadline - 300,
                                  k_active=5, max_steps=30000, lr_init=0.05)
        try_update(a_sg, "sg_k5")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_d, _ = _direct_subgrad(best_seq, deadline - 200,
                                 k_active=3, max_steps=15000)
        try_update(a_d, "sg_direct")

    # === Phase 5: Second KKT round (on improved solution) ===
    print(f"\n=== Phase 5: KKT Round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_kkt2, sc_kkt2 = _kkt_iterate(best_seq, deadline - 200,
                                       k_values=(5, 7, 10, 15))
        try_update(a_kkt2, "kkt_round2")

    # === Phase 6: More L-BFGS ===
    print(f"\n=== Phase 6: L-BFGS Round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_p6 = _adaptive_lbfgs(best_seq, deadline, reserve_time=100.0)
        try_update(a_p6, "lbfgs_r2")

    # === Phase 7: Basin hopping ===
    print(f"\n=== Phase 7: Basin Hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_hop, _ = _basin_hop(best_seq, deadline - 100, rng,
                              n_hops=5, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length Exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        lengths_to_try = [8192, 16384, 2048, 10240, 32768]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 50:
                break
            if target_n == best_n:
                continue
            if target_n > 32768:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")

            if time.monotonic() < deadline - 40:
                # Quick KKT at new length
                a_kkt_new, _ = _kkt_iterate(a_interp, deadline - 40,
                                            k_values=(5, 10))
                try_update(a_kkt_new, f"n{target_n}_kkt")

            if time.monotonic() < deadline - 30:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"n{target_n}_lbfgs")

    # === Phase 9: Final aggressive polishing ===
    print(f"\n=== Phase 9: Final Polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            max_iter = min(5000, max(200, int((deadline - 5 - time.monotonic()) / 0.005)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_polish(best_seq, deadline - 5,
                                         k_active=3, max_steps=10000, lr_init=0.02)
                try_update(a_s, f"final_sg_{rnd}")

            if time.monotonic() < deadline - 5:
                a_k, _, _ = _kkt_with_support(best_seq, k_active=5,
                                               deadline=deadline - 5)
                try_update(a_k, f"final_kkt_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
