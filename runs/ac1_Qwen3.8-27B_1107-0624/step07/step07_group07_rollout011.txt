"""
Step sequence optimizer exploiting antisymmetric (sine) perturbations about the midpoint
to reduce peak self-convolution. Uses reduced-parameter-space optimization (sine amplitudes),
full-space L-BFGS refinement with log-sum-exp smoothing, and peak shaving.
Key insight: f(1-t) = -f(t) perturbations reduce the center convolution peak,
while f(1-t) = f(t) (cosine) perturbations increase it.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_and_conv(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, None
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s)), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _sine_profile(n, freqs, amplitudes, cosine_amplitudes=None):
    """Generate a sine/cosine modulated profile normalized to sum=1."""
    t = np.arange(n) / n
    a = np.ones(n)
    for f, amp in zip(freqs, amplitudes):
        a += amp * np.sin(2.0 * np.pi * f * t)
    if cosine_amplitudes is not None:
        for f, amp in zip(freqs, cosine_amplitudes):
            a += amp * np.cos(2.0 * np.pi * f * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _smoobj_z(z, beta, n):
    """Smoothed objective in z-space (log-parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_z(z, beta, n):
    """Gradient of smoothed objective in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_conv(c, tol_frac=1e-4):
    """Get adaptive active set weights for the convolution peaks."""
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(5, int(0.01 * len(c)))
    n_active_est = min(n_active_est, len(c))
    threshold = sorted_c[n_active_est - 1]
    tol = max(tol_frac * c_max, c_max - threshold)
    tol = min(tol, 0.5 * c_max)
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros(len(c), dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    return w


def _grad_direct(a, c, n, w):
    """Gradient of weighted convolution in direct space."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_z(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    """Run L-BFGS in z-space with given smoothing parameter."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj_z(z, beta, n)

    def jac(z):
        return _smograd_z(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

    """Run L-BFGS with decreasing beta schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.5, 0.2, 0.1, 0.05, 0.025, 0.012,
                              0.006, 0.003, 0.0015, 0.0008,
                              0.0004, 0.0002, 0.0001, 5e-5,
                              2.5e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        est_per_iter = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(15000, max(300, int(remaining / (est_per_iter * 25))))
        a_new, success, niter = _lbfgs_z(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            if i % 3 == 0:
                print(f"    beta={beta:.3e}: score={sc:.10f} (iter={niter})")
    return current_a


def _peak_shaving(a_init, deadline, max_iters=80000):
    """Iterative peak reduction with mass redistribution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = c.max()
        w = _active_set_conv(c, 1e-4)
        g = _grad_direct(a, c, n, w)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_norm = g / g_norm

        best_sc_try = best_sc
        best_a_try = a
        for frac in [1.0, 0.5, 0.25, 0.125]:
            step = eta * frac
            a_try = a - step * g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-9 or stall > 200:
                break
    return best_a, best_sc


def _subgrad_z(a_init, deadline, max_steps=30000, lr_init=0.02):
    """Subgradient method in z-space with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        w = _active_set_conv(c, 1e-4)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + i * 0.0003)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.08
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.7
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.9
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _reduced_space_optimize(n, freqs, deadline, max_iter=5000):
    """
    Optimize sine amplitudes in reduced parameter space.
    a[i] = (1 + sum_k eps_k * sin(2*pi*freqs[k]*i/n)) / n
    """
    K = len(freqs)
    t = np.arange(n) / n
    # Precompute basis
    basis = np.zeros((K, n))
    for k, f in enumerate(freqs):
        basis[k] = np.sin(2.0 * np.pi * f * t)

    def obj_and_grad(eps):
        a = 1.0 + basis.T @ eps
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.inf, np.zeros(K)
        a = a / s
        c = _full_convolve(a, a)
        c_max = c.max()
        w = _active_set_conv(c, 1e-5)
        # Gradient: dc[j]/deps_k = 2*(u_k * a)[j] where u_k[i] = basis[k,i]/n (approx)
        # More precisely, using the active set:
        grad_c = np.zeros(K)
        for k in range(K):
            u = basis[k] / n  # da[i]/deps_k (approximately, ignoring renormalization)
            wu = _full_convolve(w, u)
            h_k = wu[n - 1:2 * n - 1]
            W_val = float(np.dot(w, c))
            # dc/da = 2*(w * a_rev) shifted - but we need dc/deps
            # dc[j]/deps_k = sum_i (da[i]/deps_k)*a[j-i] + a[i]*(da[j-i]/deps_k)
            # = 2*(u * a)[j] (by symmetry of convolution)
            ua = _full_convolve(u, a)
            grad_c[k] = 2.0 * float(np.dot(w, ua))
        score = 2.0 * n * float(np.dot(w, c))  # Use weighted version for smoothness
        return score, 2.0 * n * grad_c

    # Use L-BFGS in the reduced space
    eps0 = np.zeros(K)
    # Start with small amplitude for the first frequency
    eps0[0] = 0.1

    def fun(eps):
        sc, _ = obj_and_grad(eps)
        return sc

    def jac(eps):
        _, g = obj_and_grad(eps)
        return g

    try:
        result = minimize(fun, eps0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxcor': 50, 'maxls': 50})
        a_opt = 1.0 + basis.T @ result.x
        a_opt = np.maximum(a_opt, 0.0)
        a_opt = a_opt / a_opt.sum()
        return a_opt, result.x
    except Exception:
        a_fail = 1.0 + basis.T @ eps0
        a_fail = np.maximum(a_fail, 0.0)
        a_fail = a_fail / a_fail.sum()
        return a_fail, eps0


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, rng, scale=0.02):
    """Perturb a sequence in z-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer exploiting sine (antisymmetric) perturbations.
    Phases: reduced-space optimization -> full-space L-BFGS -> peak shaving
    -> subgradient -> multi-length -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(4096) / 4096
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init")

    start_n = len(best_seq) if best_seq is not None else 8192

    # === Phase 1: Sine-modulated profile exploration ===
    print(f"\n=== Phase 1: Sine profile exploration ===")
    t_phase = time.monotonic()

    # Try various sine-modulated profiles at different lengths
    sine_candidates = []
    for n_try in [4096, 8192, 16384]:
        if n_try > 16384 and time.monotonic() > deadline - 600:
            break
        t = np.arange(n_try) / n_try
        for m in [1, 2, 3, 4, 5, 7, 8, 11, 13, 16, 21, 32]:
            for eps in [0.3, 0.5, 0.7, 0.85, 0.95, 1.0]:
                if time.monotonic() > deadline - 500:
                    break
                a = (1.0 + eps * np.sin(2.0 * np.pi * m * t))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 0.01:
                    continue
                a = a / s
                a = a / n_try * n_try  # keep as is
                a = a / a.sum()
                sc = _score(a)
                if sc < best_score_val:
                    sine_candidates.append((sc, n_try, m, eps, a.copy()))
                    try_update(a, f"sine_n{n_try}_m{m}_e{eps:.2f}")

    # Try multi-sine profiles
    if time.monotonic() < deadline - 400:
        for n_try in [8192]:
            t = np.arange(n_try) / n_try
            multi_sines = [
                ([1, 2], [0.8, 0.3]),
                ([1, 3], [0.7, 0.4]),
                ([1, 2, 3], [0.6, 0.3, 0.2]),
                ([2, 3], [0.7, 0.5]),
                ([1, 4], [0.8, 0.2]),
                ([3, 5], [0.6, 0.4]),
            ]
            for freqs, amps in multi_sines:
                a = 1.0
                for f, amp in zip(freqs, amps):
                    a = a + amp * np.sin(2.0 * np.pi * f * t)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 0.01:
                    continue
                a = a / s
                sc = _score(a)
                if sc < best_score_val:
                    try_update(a, f"multi_sine_{freqs}")

    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 2: Reduced-space optimization on best candidate ===
    print(f"\n=== Phase 2: Reduced-space optimization ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 400:
        n_cur = len(best_seq)
        # Optimize in reduced sine space
        for freq_set in [[1], [1, 2], [1, 2, 3], [1, 3, 5], [1, 2, 3, 5]]:
            if time.monotonic() > deadline - 300:
                break
            a_red, eps_opt = _reduced_space_optimize(n_cur, freq_set, deadline - 250, max_iter=2000)
            sc_red = _score(a_red)
            print(f"    Reduced space freqs={freq_set}: score={sc_red:.10f}")
            if sc_red < best_score_val:
                try_update(a_red, f"reduced_{freq_set}")

    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Full-space L-BFGS with fine beta schedule ===
    print(f"\n=== Phase 3: L-BFGS fine schedule ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_p3 = _lbfgs_schedule(best_seq, deadline - 250, min_remaining=200.0)
        try_update(a_p3, "lbfgs_fine")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak shaving ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_ps, sc_ps = _peak_shaving(best_seq, deadline - 150, max_iters=100000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Subgradient in z-space ===
    print(f"\n=== Phase 5: Z-space subgradient ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_z(best_seq, deadline - 100, max_steps=50000, lr_init=0.015)
        try_update(a_sg, "subgrad_z")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Try larger n with sine initialization ===
    print(f"\n=== Phase 6: Larger n exploration ===")
    t_phase = time.monotonic()
    for target_n in [16384, 32768]:
        if time.monotonic() > deadline - 100:
            break
        if target_n <= start_n:
            continue
        print(f"  Trying n={target_n} with sine init...")
        t = np.arange(target_n) / target_n
        # Best sine profile at this length
        a_sine = (1.0 + 0.9 * np.sin(2.0 * np.pi * 1.0 * t))
        a_sine = np.maximum(a_sine, 0.0)
        a_sine = a_sine / a_sine.sum()
        sc_sine = _score(a_sine)
        print(f"    Sine init: score={sc_sine:.10f}")
        try_update(a_sine, f"sine_n{target_n}")

        if time.monotonic() < deadline - 80:
            a_opt = _lbfgs_schedule(a_sine, deadline - 60, min_remaining=50.0)
            try_update(a_opt, f"lbfgs_n{target_n}")

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Second round of refinement on best ===
    print(f"\n=== Phase 7: Second refinement round ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 50:
        # Peak shave again
        if time.monotonic() < deadline - 30:
            a_ps2, _ = _peak_shaving(best_seq, deadline - 25, max_iters=50000)
            try_update(a_ps2, "peak_shave_2")

        # Ultra-fine L-BFGS
        if best_seq is not None and time.monotonic() < deadline - 20:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            for beta_frac in [1e-6, 5e-7, 2e-7, 1e-7]:
                if time.monotonic() > deadline - 15:
                    break
                beta = mb * beta_frac
                max_iter = min(5000, max(200, int((deadline - 15 - time.monotonic()) / 0.003)))
                a_ultra, _, _ = _lbfgs_z(best_seq, beta, n, max_iter)
                try_update(a_ultra, f"ultra_b{beta_frac}")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Multi-restart with polish ===
    print(f"\n=== Phase 8: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            (0.01, "small"),
            (0.03, "med"),
            (0.05, "large"),
        ]
        for scale, name in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, rng, scale)
            sc_start = _score(p_start)
            a_r = _lbfgs_schedule(p_start, deadline - 15, min_remaining=10.0)
            try_update(a_r, f"restart_{name}")

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 3:
        for cycle in range(5):
            if time.monotonic() > deadline - 3:
                break
            if time.monotonic() < deadline - 3:
                a_f, _ = _peak_shaving(best_seq, deadline - 3, max_iters=20000)
                try_update(a_f, f"final_ps_{cycle}")
            if best_seq is not None and time.monotonic() < deadline - 3:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 5e-8
                max_iter = min(3000, max(100, int((deadline - 3 - time.monotonic()) / 0.003)))
                if max_iter > 50:
                    a_f2, _, _ = _lbfgs_z(best_seq, beta, n, max_iter)
                    try_update(a_f2, f"final_lbfgs_{cycle}")

    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(8192) / 8192
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
