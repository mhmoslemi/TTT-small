**For the QP solver**, I'll use a projected gradient method or a custom active-set method since the problem is:
$\min a^T H a$ s.t. $a \geq 0, \mathbf{1}^T a = 1$

The KKT conditions are:
- $2Ha + \lambda \mathbf{1} + \mu = 0$ (stationarity)
- $a \geq 0, \mu \geq 0, \mu_i a_i = 0$ (complementary slackness)
- $\mathbf{1}^T a = 1$ (feasibility)

For a given set of active indices $\mathcal{I} = \{i : a_i = 0\}$, we solve:
$2H_{\mathcal{I}^c} a_{\mathcal{I}^c} + \lambda \mathbf{1} = 0$
$\mathbf{1}^T a_{\mathcal{I}^c} = 1$

This is a linear system that can be solved directly. Then we check if the solution satisfies $a_i \geq 0$ for all $i \notin \mathcal{I}^c$... i.e., for all $i \in \mathcal{I}^c$ we need $a_i > 0$, and for $i \in \mathcal{I}$ we need $\mu_i > 0$.

This is the classic QP active set method. For large $n$, solving the linear system at each iteration might be expensive, but we can use CG or exploit the structure.

Actually, $H_{\mathcal{A}}$ has a special structure. Each $M_k$ is a "shift" matrix (anti-diagonal band). So $H_{\mathcal{A}}$ is a sum of such matrices, which means it's a banded matrix (or at least has a structure amenable to fast multiplication via FFT).

For the matrix-vector product $H_{\mathcal{A}} v = \sum_{k \in \mathcal{A}} M_k v$, we can compute this as:
$(H_{\mathcal{A}} v)_i = \sum_{k \in \mathcal{A}} v_{k-i}$ (for valid indices)

This is a cross-correlation that can be computed via FFT!

So the algorithm for the QP subproblem is:
1. Maintain the set of "free" variables (where $a_i > 0$)
2. Solve the linear system on the free variables using CG with FFT-based matvec
3. Check KKT conditions
4. Update the active set

This should be efficient and exact (up to CG tolerance).

For the KKT conditions to give a global minimum, we need $H$ to be PSD on the tangent space of the feasible set at the solution. This is a second-order condition that may or may not hold.

In practice, the "active set QP" approach (solving the equality-constrained minimization on the free set) will give a stationary point. Whether it's a minimum or saddle point depends on the curvature.

To handle this, I'll add a small regularization: $H \leftarrow H + \epsilon I$ to ensure the Hessian is PD. This slightly changes the problem but should help convergence.

"""
Multi-resolution optimizer combining power-sum L-BFGS, active-set QP refinement,
symmetry exploitation, and careful beta/p-sweep to beat 1.505706.
Key innovations: power-sum objective for better conditioning, exact QP subproblem
solves via FFT-accelerated CG, symmetric parameterization, and multi-scale strategy.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lso_obj(z, beta, n):
    """Log-sum-exp smooth max objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lso_grad(z, beta, n):
    """Gradient of log-sum-exp smooth max."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _ps_obj(z, p, n):
    """Power-sum smooth max objective: (sum c_k^p)^(1/p)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)
    c_p = c ** p
    total = float(np.sum(c_p))
    if total < 1e-300:
        return 2.0 * n * float(c.max())
    ps_val = total ** (1.0 / p)
    return 2.0 * n * ps_val / (1.0 / n)  # normalized


def _ps_grad(z, p, n):
    """Gradient of power-sum objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-300)
    c_p_minus_1 = c ** (p - 1)
    total = float(np.sum(c ** p))
    if total < 1e-300:
        return np.zeros_like(a)
    w = c_p_minus_1 / total  # normalized weights
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_matvec(v, active_lags, n):
    """Compute (sum_{k in active} M_k) @ v using FFT."""
    v_rev = v[::-1]
    N = _next_pow2(4 * n)
    fv = np.fft.rfft(v_rev, N)
    fa = np.fft.rfft(active_lags, N)
    r = np.fft.irfft(fv * fa, N)
    return r[n - 1:2 * n - 1]


def _qp_solve_active(active_lags, n, max_outer=100, tol=1e-10, reg=1e-6):
    """Solve min a^T H a s.t. a >= 0, sum(a)=1 using active set + CG.
    H = sum_{k in active} M_k + reg*I
    """
    a = np.ones(n) / n
    free = np.ones(n, dtype=bool)

    for outer in range(max_outer):
        n_free = int(free.sum())
        if n_free <= 1:
            break

        free_idx = np.where(free)[0]

        def _masked_mv(x):
            full_x = np.zeros(n)
            full_x[free_idx] = x
            Hx = _active_matvec(full_x, active_lags, n)
            Hx += reg * full_x
            return Hx[free_idx]

        # CG to solve H_ff z = 1
        z = np.zeros(n_free)
        r = np.ones(n_free)
        p = r.copy()
        rs = float(np.dot(r, r))
        max_cg = min(300, 3 * n_free)

        for _ in range(max_cg):
            Hp = _masked_mv(p)
            pH = float(np.dot(p, Hp))
            if abs(pH) < 1e-30:
                break
            alpha = rs / pH
            if not np.isfinite(alpha) or abs(alpha) > 1e10:
                break
            z = z + alpha * p
            r = r - alpha * Hp
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < tol * np.sqrt(rs):
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p = r + beta_cg * p
            rs = r_new

        denom = float(np.sum(z))
        if denom < 1e-15 or not np.isfinite(denom):
            break

        a_f = z / denom

        if np.min(a_f) >= -1e-10:
            a_new = np.zeros(n)
            a_new[free_idx] = np.maximum(a_f, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s

            # Check KKT: mu_i = -(2(Ha)_i - lambda) for inactive i
            Ha = _active_matvec(a_new, active_lags, n) + reg * a_new
            lambda_val = -2.0 * float(np.mean(Ha[free]))
            mu_inactive = -(2.0 * Ha[~free] - lambda_val)

            if np.min(mu_inactive) >= -1e-8:
                a = a_new
                break

            # Add most violated to free set
            inactive_idx = np.where(~free)[0]
            if len(inactive_idx) > 0:
                worst = inactive_idx[np.argmin(mu_inactive)]
                free[worst] = True
                a = a_new
            else:
                a = a_new
                break
        else:
            # Remove most negative
            min_local = int(np.argmin(a_f))
            a = np.zeros(n)
            a[free_idx] = np.maximum(a_f, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            free[free_idx[min_local]] = False

    return a


def _active_set_refine(a_init, n, deadline, max_rounds=30):
    """Iterative active set: identify active lags, solve QP, repeat."""
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: start loose, tighten
        tol_frac = max(0.005, 0.05 * (0.9 ** rnd))
        tol = tol_frac * c_max
        active = (c >= c_max - tol).astype(np.float64)
        if active.sum() < 2:
            active[int(np.argmax(c))] = 1.0
            # Also add second highest
            c_copy = c.copy()
            c_copy[int(np.argmax(c))] = -np.inf
            active[int(np.argmax(c_copy))] = 1.0

        a_new = _qp_solve_active(active, n, max_outer=50, tol=1e-10, reg=1e-6)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a mix
            for t in [0.5, 0.3, 0.1]:
                a_mix = (1 - t) * a + t * a_new
                a_mix = np.maximum(a_mix, 0)
                sm = a_mix.sum()
                if sm > 1e-15:
                    a_mix = a_mix / sm
                    sc_mix = _score(a_mix)
                    if sc_mix < best_sc:
                        best_sc = sc_mix
                        best_a = a_mix.copy()
                        a = a_mix
                        break
            else:
                if abs(sc_new - best_sc) < 1e-10:
                    break

    return best_a, best_sc


def _lbfgs_run(a_init, deadline, min_remaining, use_ps=False):

    max_c = float(c_cur.max())

        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    top_k = min(20, len(c_sorted))
    gap = c_sorted[0] - c_sorted[top_k - 1] if top_k > 1 else c_sorted[0] * 0.1
    base = max(gap, max_c * 0.005)

    if use_ps:
        ps_values = [5, 10, 20, 40, 80, 160, 320, 640]

            remaining = deadline - min_remaining - time.monotonic()

                break
            max_iter = min(5000, max(100, int(remaining / 0.02)))

            def _fun_ps(z_local):

            try:

                a_new = _softmax(res.x)

                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(current_a, 1e-30))

    else:

                                 0.005, 0.002, 0.001, 0.0005, 0.0002,

        current_a = a_init.copy()
        current_sc = _score(current_a)

            remaining = deadline - min_remaining - time.monotonic()

            try:

                               options={'maxiter': max_iter, 'ftol': 1e-18,
                                        'gtol': 1e-14, 'maxcor': 60, 'maxls': 100})
                a_new = _softmax(res.x)
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new

            except Exception:

def _symmetric_lbfgs(a_init, deadline, min_remaining):
    """Optimize only the first half (symmetric: a[i] = a[n-1-i])."""
    n = len(a_init)
    if n % 2 == 0:
        half = n // 2
    else:
        half = n // 2 + 1

    # Extract first half
    a_half = a_init[:half].copy()
    a_half = np.maximum(a_half, 1e-15)
    z_half = np.log(a_half)
    z_half = z_half - z_half.mean()

    c_full = _full_convolve(a_init, a_init)
    max_c = float(c_full.max())
    base = max(max_c * 0.01, 1e-4)

    betas = base * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    current_best = _score(a_init)

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        max_iter = min(4000, max(50, int(remaining / 0.02)))

        def _sym_fun(zh):
            ah = _softmax(zh)
            if n % 2 == 0:
                a_full = np.concatenate([ah, ah[::-1]])
            else:
                a_full = np.concatenate([ah, ah[-2::-1]])
            c = _full_convolve(a_full, a_full)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def _sym_jac(zh):
            ah = _softmax(zh)
            if n % 2 == 0:
                a_full = np.concatenate([ah, ah[::-1]])
            else:
                a_full = np.concatenate([ah, ah[-2::-1]])
            c = _full_convolve(a_full, a_full)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a_full[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad_full = 4.0 * n * a_full * (h - W)
            # Project to symmetric: grad for first half
            if n % 2 == 0:
                grad_half = grad_full[:half] + grad_full[half:][::-1]
            else:
                grad_half = grad_full[:half].copy()
                grad_half[half - 1] += grad_full[half - 1]
            return grad_half

        try:
            res = minimize(_sym_fun, z_half, jac=_sym_jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxcor': 40})
            ah_new = _softmax(res.x)
            if n % 2 == 0:
                a_full_new = np.concatenate([ah_new, ah_new[::-1]])
            else:
                a_full_new = np.concatenate([ah_new, ah_new[-2::-1]])
            sc = _score(a_full_new)
            if sc < current_best:
                current_best = sc
                a_init = a_full_new
                a_half = a_full_new[:half].copy()
                a_half = np.maximum(a_half, 1e-15)
                z_half = np.log(a_half)
                z_half = z_half - z_half.mean()
        except Exception:
            pass

    return a_init, current_best

    best_a = a_init.copy()

    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _lso_grad(z, 1e-4, n)
        step = lr / (1.0 + 0.005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

                best_sc = sc

            else:

                    z = z - z.mean()
                    patience = 0
                    lr *= 0.7

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_opt":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t) + 0.05 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _upsample(a_old, n_target):
    x_old = np.linspace(0, 1, len(a_old))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a_new / s


def _try_shrinkage(a, deadline):
    """Try zeroing smallest elements to improve score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[idx[:n_zero]] = 0.0
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
    return best_a, best_sc

        print(f"No parent, using cos_opt n=8192")
        update_best(parent, _score(parent), "init")

    a_main = best_seq.copy()
    n_main = len(a_main)

    # === Phase 1: L-BFGS with LSE (aggressive) ===
    print(f"\n[Phase 1] LSE L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a1 = _lbfgs_run(a_main, deadline, min_remaining=380, use_ps=False)
        update_best(a1, _score(a1), "lse_lbfgs")

    # === Phase 2: Power-sum L-BFGS ===
    print(f"\n[Phase 2] Power-sum L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a2 = _lbfgs_run(best_seq, deadline, min_remaining=280, use_ps=True)
        update_best(a2, _score(a2), "ps_lbfgs")

    # === Phase 3: Symmetric optimization ===
    print(f"\n[Phase 3] Symmetric optimization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a3, sc3 = _symmetric_lbfgs(best_seq, deadline, min_remaining=230)
        update_best(a3, sc3, "symmetric")

    # === Phase 4: Active set QP ===
    print(f"\n[Phase 4] Active set QP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a4, sc4 = _active_set_refine(best_seq, n_main, min(deadline - 180, time.monotonic() + 120), max_rounds=20)
        update_best(a4, sc4, "active_qp")

    # === Phase 5: L-BFGS after QP ===
    print(f"\n[Phase 5] L-BFGS post-QP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a5 = _lbfgs_run(best_seq, deadline, min_remaining=130, use_ps=False)
        update_best(a5, _score(a5), "lbfgs_post_qp")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [4096, 8192, 16384]
    if n_main not in n_targets:
        n_targets.append(n_main)
    n_targets = sorted(set(n_targets))

    for n_try in n_targets:
        if time.monotonic() > deadline - 80:
            break
        if len(best_seq) != n_try:
            a_init = _upsample(best_seq, n_try)
        else:
            a_init = best_seq.copy()

        a_opt = _lbfgs_run(a_init, deadline, min_remaining=60, use_ps=False)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n{n_try}")

        if sc_opt < best_score * 1.001 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_refine(a_opt, n_try, min(deadline - 30, time.monotonic() + 60), max_rounds=10)
            update_best(a_opt2, sc_opt2, f"n{n_try}_qp")

    # === Phase 7: Shrinkage ===
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        a_sh, sc_sh = _try_shrinkage(best_seq, deadline - 40)
        update_best(a_sh, sc_sh, "shrinkage")
        if sc_sh < best_score * 1.002 and time.monotonic() < deadline - 30:
            a_sh2 = _lbfgs_run(a_sh, deadline, min_remaining=20, use_ps=False)
            update_best(a_sh2, _score(a_sh2), "lbfgs_post_shrink")

    # === Phase 8: Subgradient final polish ===
    print(f"\n[Phase 8] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        a_sg, sc_sg = _subgrad_final(best_seq, deadline - 5, n_steps=5000, lr_init=0.006)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 9: Final very low beta L-BFGS ===
    print(f"\n[Phase 9] Ultra-low beta (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_ultra = mb * 0.00001
        remaining = deadline - 3 - time.monotonic()
        if remaining > 2:
            max_iter = min(2000, max(50, int(remaining / 0.005)))

            def _fun_ultra(z_local):
                return _lso_obj(z_local, beta_ultra, n)

            def _jac_ultra(z_local):
                return _lso_grad(z_local, beta_ultra, n)

            try:
                res = minimize(_fun_ultra, z, jac=_jac_ultra, method='L-BFGS-B',
                               options={'maxiter': max_iter, 'ftol': 1e-18,
                                        'gtol': 1e-14, 'maxcor': 50})
                a_ultra = _softmax(res.x)
                update_best(a_ultra, _score(a_ultra), "ultra_low_beta")
            except Exception:
                pass

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence confirms: {sc_eval:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
