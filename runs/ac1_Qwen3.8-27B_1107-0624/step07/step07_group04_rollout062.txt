"""
Multi-algorithm optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam optimizer, L-BFGS, active-set Newton, and Fourier parameterization.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max, c_max
    return c_max + beta * np.log(s), c_max


def _compute_grad(a, beta, n):
    """Compute gradient of smoothed objective w.r.t. z (log-a)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # h[i] = sum_k w[k] * (a[k-i] + a[i-k]) (valid indices)
    # This is the correlation of w with a, computed efficiently via FFT
    a_flip = a[::-1]
    wa = _full_convolve(w, a_flip)
    h = wa[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 2.0 * n * a * (h - W)
    return grad


def _adam_optimize(a_init, deadline, max_steps=50000):
    """Adam optimizer on the log-sum-exp smoothed objective."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

        c = np.maximum(c, 0)

            if sp < 1e-15:
                grad[j] = 0
            else:
                cp = cp_pos / sp
                grad[j] = (_score(cp) - sc) / eps
            cm = coeffs_wrapped.copy()
            cm[j] -= eps
            cm_pos = np.maximum(A @ cm, 0)
            sm = cm_pos.sum()
            if sm < 1e-15:
                grad[j] = 0
            else:
                cm = cm_pos / sm
                grad[j] = (grad[j] + (sc - _score(cm)) / eps) / 2
        return sc, grad
    
    # Use Nelder-Mead or L-BFGS
    best_coeffs = coeffs.copy()
    best_sc = _score(np.maximum(A @ coeffs, 0) / max(np.maximum(A @ coeffs, 0).sum(), 1e-15))

    s = final_a.sum()
    if s < 1e-15:
        final_a = a_init.copy()
    else:
        final_a = final_a / s

                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _fourier_obj(x, A, n):
    c = A @ x
    c = np.maximum(c, 0)
    s = c.sum()
    if s < 1e-15:
        return 1e10
    c = c / s
    return _score(c)


def _fourier_optimize(a_init, deadline, n_harmonics=40, min_time_remaining=30.0):
    n = len(a_init)
    t = np.arange(n) / n
    K = n_harmonics
    A = np.zeros((n, K + 1))
    for k in range(K + 1):
        A[:, k] = np.cos(2.0 * np.pi * k * t)
    # Add sine terms for asymmetry
    for k in range(1, K + 1):
        A[:, K + k] = np.sin(2.0 * np.pi * k * t)

    # Wait, that changes the width. Let me use a different approach.
    # Use a full Fourier series with K complex terms = 2K+1 real coefficients
    # Actually let's just use cosines for simplicity (symmetric basis)
    # and add a linear term for asymmetry
    A = np.zeros((n, K + 2))
    for k in range(K + 1):
        A[:, k] = np.cos(2.0 * np.pi * k * t)
    A[:, K + 1] = t  # linear term for asymmetry

    coeffs, _, _, _ = np.linalg.lstsq(A, a_init, rcond=None)

    best_coeffs = coeffs.copy()
    best_sc = _fourier_obj(coeffs, A, n)

    try:
        result = minimize(
            _fourier_obj, coeffs,
            args=(A, n),
            method='Nelder-Mead',
            options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12,
                     'adaptive': True}
        )
        try_sc = _fourier_obj(result.x, A, n)
        if try_sc < best_sc:
            best_sc = try_sc
            best_coeffs = result.x.copy()
    except Exception:
        pass

    final_a = np.maximum(A @ best_coeffs, 0)
    s = final_a.sum()
    if s < 1e-15:
        final_a = a_init.copy()
    else:
        final_a = final_a / s

    return final_a, _score(final_a)


def _spline_optimize(a_init, deadline, n_knots=60, min_time_remaining=30.0):
    """Optimize a sequence parameterized by a B-spline (via cubic interpolation of knot values)."""
    n = len(a_init)
    n_k = min(n_knots, n)
    knot_idx = np.linspace(0, n - 1, n_k_knots).astype(int) if False else np.round(np.linspace(0, n - 1, n_knodes)).astype(int)
    # Simpler: just use linear interpolation from knot values
    knot_idx = np.linspace(0, n - 1, n_knodes).astype(int)
    knot_vals = a_init[knot_idx]

    def spline_obj(kvals):
        a_eval = np.interp(np.arange(n), np.arange(n_knodes), kvals)
        a_eval = np.maximum(a_eval, 0)
        s = a_eval.sum()
        if s < 1e-15:
            return 1e10
        a_eval = a_eval / s
        return _score(a_eval)

    best_kv = knot_vals.copy()
    best_sc = spline_obj(knot_vals)

    try:
        result = minimize(
            _spline_obj_wrapper, knot_vals,
            args=(n, n_knodes),
            method='Nelder-Mead',
            options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10,
                     'adaptive': True}
        )
        try_sc = _spline_obj_wrapper(result.x, n, n_knodes)
        if try_sc < best_sc:
            best_sc = try_sc
            best_kv = result.x.copy()
    except Exception:
        pass

    final_a = np.interp(np.arange(n), np.arange(n_knodes), best_kv)
    final_a = np.maximum(final_a, 0)
    s = final_a.sum()
    if s < 1e-15:
        final_a = a_init.copy()
    else:
        final_a = final_a / s

    return final_a, _score(final_a)


def _spline_obj_wrapper(kvals, n, n_knodes):
    a_eval = np.interp(np.arange(n), np.arange(n_knodes), kvals)
    a_eval = np.maximum(a_eval, 0)
    s = a_eval.sum()
    if s < 1e-15:
        return 1e10
    a_eval = a_eval / s
    return _score(a_eval)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.2 * np.exp(-0.5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_decay":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 1.5 + 0.01
        return a / a.sum()
    elif kind == "gaussian_off":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.4) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-algorithm search: Adam + L-BFGS + Newton + Fourier + Spline + shrinkage.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS quick pass
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Adam optimizer
    print(f"\n[Phase 2] Adam (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _adam_optimize(best_seq, min(deadline - 400, time.monotonic() + 200))
        update_best(a2, sc2, "adam_1")

    # Phase 3: Newton active set polish
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a3, sc3 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 120))
        update_best(a3, sc3, "newton_1")

    # Phase 4: L-BFGS again (from improved point)
    print(f"\n[Phase 4] L-BFGS 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4 = _run_lbfgs(best_seq, deadline, min_time_remaining=150.0)
        update_best(a4, _score(a4), "lbfgs_2")

    # Phase 5: Fourier parameterization
    print(f"\n[Phase 5] Fourier (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _fourier_optimize(best_seq, min(deadline - 200, time.monotonic() + 80), n_harmonics=50)
        update_best(a5, sc5, "fourier")
        if sc5 < best_score + 0.005:
            a5n, sc5n = _newton_active_polish(a5, min(deadline - 180, time.monotonic() + 60))
            update_best(a5n, sc5n, "fourier_newton")

    # Phase 6: Spline parameterization
    print(f"\n[Phase 6] Spline (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a6, sc6 = _spline_optimize(best_seq, min(deadline - 150, time.monotonic() + 60), n_knots=80)
        update_best(a6, sc6, "spline")
        if sc6 < best_score + 0.005:
            a6n, sc6n = _newton_active_polish(a6, min(deadline - 130, time.monotonic() + 50))
            update_best(a6n, sc6n, "spline_newton")

    # Phase 7: Shrinkage
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a7, sc7 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=60.0)
        update_best(a7, sc7, "shrink")

    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 12288, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 40:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 35, time.monotonic() + 40))
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 9: Diverse restarts with varied profiles
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        profiles = ["asym_taper", "gaussian_off", "power_decay", "cos"]
        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 20:
                break
            n_r = len(best_seq)
            p_start = _make_profile(n_r, prof)
            # Blend with best
            blend = 0.5 * p_start + 0.5 * best_seq
            blend = blend / blend.sum()
            a_r = _run_lbfgs(blend, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{prof}")

    # Phase 10: Final aggressive polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-7
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(500, max(10, int(remaining / 0.02)))
                try:
                    result = minimize(
                        _lbfgs_obj, z,
                        jac=_lbfgs_jac,
                        args=(beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                 'maxcor': 50}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
