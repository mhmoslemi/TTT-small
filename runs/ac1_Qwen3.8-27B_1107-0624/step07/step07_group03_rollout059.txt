    # = sum_k w[k]*a[k-i]
    # Let u = w, v = a. Then v[i] = (u cross-correlate v at shift i)
    # In FFT: IFFT(FFT(u) * conj(FFT(v))) gives cross-correlation
    # But we need a specific convention. Let me just use the direct approach via _full_convolve:
    # sum_k w[k]*a[k-i]: let j = k-i, so k = j+i. sum_j w[j+i]*a[j]
    # = sum_j a[j]*w[j+i]
    # This is the convolution of a with w, evaluated at position... 
    # (a * w)[j+i] = sum_m a[m]*w[j+i-m]... no that's not right either.
    # 
    # Let me think differently. We want g[i] = sum_k w[k]*(a[k-i] + a[i-k])
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # 
    # First term: sum_k w[k]*a[k-i]. Let l = k-i, k = l+i: sum_l w[l+i]*a[l]
    # Second term: sum_k w[k]*a[i-k]. Let l = i-k, k = i-l: sum_l w[i-l]*a[l]
    # 
    # First term = cross_corr(w, a, shift=i) where cross_corr(u,v,shift) = sum_l u[l+shift]*v[l]
    # Second term = cross_corr(rev_w, a, shift=i) where rev_w[k] = w[i-k]... 
    # Actually: sum_l w[i-l]*a[l] = sum_l a[l]*w[i-l] = (a * w)[i] (convolution!)
    # And: sum_l w[l+i]*a[l] = sum_l a[l]*w[l+i]
    # Let m = l+i, l = m-i: sum_m a[m-i]*w[m] = (a_shifted * w) where a_shifted is...
    # 
    # Simpler: Use the identity that sum_k w[k]*a[k-i] = (flip(w) * a)[i + len(w)-1-n]... 
    # This is getting too complicated. Let me use the approach from the previous code:
    # wc = _full_convolve(w, a_rev) where a_rev = a[::-1]
    # Then wc[n-1+i] gives sum_k w[k]*a_rev[i-k] = sum_k w[k]*a[n-1-(i-k)] = sum_k w[k]*a[n-1-i+k]
    # Hmm, that's not quite what we want either.
    #
    # Let me just compute it directly using the property that:
    # grad[i] = sum_k w[k] * d(b[k])/d(a[i]) = sum_k w[k] * (a[k-i] + a[i-k])
    # where a[j] = 0 for j < 0 or j >= n.
    #
    # sum_k w[k]*a[k-i]: for fixed i, k ranges from i to i+n-1 (for a[k-i] to be valid)
    #   = sum_{k=i}^{i+n-1} w[k]*a[k-i] (if k is in [0, 2n-2])
    #   Let j = k-i: sum_{j=0}^{n-1} w[j+i]*a[j]
    #
    # sum_k w[k]*a[i-k]: for fixed i, k ranges from max(0, i-n+1) to min(i, 2n-2)
    #   Let j = i-k: sum_{j} w[i-j]*a[j] where i-j is in [0, 2n-2]
    #   = sum_{j=max(0,i-2n+2)}^{min(i,n-1)} w[i-j]*a[j]
    #
    # I'll use the FFT approach: 
    # First term: sum_j a[j]*w[j+i] for j=0..n-1 (w indexed 0..2n-2, out of range = 0)
    #   This is: for the vector a (length n) and w (length 2n-1), compute the "anti-diagonal" sum
    #   = element i of the "flip-convolution": (a * w)[?]
    #   (a * w)[k] = sum_j a[j]*w[k-j]. We want sum_j a[j]*w[j+i] = sum_j a[j]*w[i-j+2j-i+... 
    #   Actually: sum_j a[j]*w[j+i]. Let's define w_ext as w padded to length n + (2n-1) - 1 = 3n-2.
    #   Then sum_j a[j]*w[j+i] is the cross-correlation of a and w at shift -i.
    #   Using FFT: IFFT(FFT(a, N) * conj(FFT(w, N))) gives the cross-correlation.
    #
    # I'll just use a simpler approach: compute via _full_convolve with appropriate padding.

    # Alternative approach: use the fact that the gradient can be computed as
    # grad[i] = (correlation of w with a, shifted appropriately)
    # Let me use the same trick as before:
    # wc = _full_convolve(w, a[::-1]) has length (2n-1) + n - 1 = 3n-2
    # wc[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-(m-k)] = sum_k w[k]*a[n-1-m+k]
    # We want: first_term[i] = sum_k w[k]*a[k-i]
    #   = sum_k w[k]*a[k-i] ... let's set n-1-m+k = k-i => m = n-1+i
    #   So first_term[i] = wc[n-1+i]
    # second_term[i] = sum_k w[k]*a[i-k]
    #   We need sum_k w[k]*a[i-k]. Let's use w_rev = w[::-1] (length 2n-1)
    #   wc2 = _full_convolve(w_rev, a[::-1])
    #   wc2[m] = sum_k w_rev[k]*a[n-1-m+k] = sum_k w[2n-2-k]*a[n-1-m+k]
    #   Let l = 2n-2-k: sum_l w[l]*a[n-1-m+2n-2-l] = sum_l w[l]*a[3n-3-m-l]
    #   We want sum_l w[l]*a[i-l]: set 3n-3-m-l = i-l => 3n-3-m = i => m = 3n-3-i
    #   So second_term[i] = wc2[3n-3-i]
    #
    # This is getting very complex. Let me just use a direct O(n^2) approach for the gradient
    # since n is at most ~16384 and we can vectorize it.

    # Actually, the cleanest approach: 
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    #
    # For the first sum, define: for each i, sum over k of w[k]*a[k-i] where 0<=k-i<n
    # This is equivalent to: (the "upper triangular" part of a Toeplitz-like product)
    #
    # I'll use the approach: create a (2n-1) x n matrix M where M[k,i] = a[k-i] if valid, else 0
    # Then grad_first = M.T @ w
    # But this matrix is too large. Use FFT instead.
    #
    # Key insight: sum_k w[k]*a[k-i] = element i of the vector obtained by "folding" w through a
    # This IS the convolution a * w evaluated at certain positions, or the cross-correlation.
    #
    # Let me verify: (a * w)[k] = sum_j a[j] * w[k-j]
    # I want sum_k w[k] * a[k-i] = sum_k a[k-i] * w[k]
    # Let j = k-i, k = j+i: sum_j a[j] * w[j+i]
    # = (a cross-correlate w at shift i) [with appropriate definition]
    # = IFFT(FFT(a) * conj(FFT(w)))[i] (using the right convention)
    
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation: IFFT(conj(FFT(a)) * FFT(w)) gives sum_j a[j]*w[j+i]... 
    # Standard: if c[i] = sum_j a[j]*conj(w[j-i]), then FFT(c) = FFT(a)*conj(FFT(w))
    # We want: t[i] = sum_j a[j]*w[j+i]
    # Note: t[i] = sum_j a[j]*w[j+i] = sum_j w[j+i]*a[j]
    # Let u[j] = w[j+i], then t[i] = sum_j u[j]*a[j] = inner product
    # In FFT terms: this is the cross-correlation of a and w with a specific shift convention.
    # 
    # Actually, the simplest correct approach:
    # t1[i] = sum_j a[j]*w[j+i] = (w * a_padded) evaluated appropriately
    # where a is zero-padded.
    # t1[i] = sum_{j=0}^{n-1} a[j]*w[j+i]
    # The vector w has length 2n-1. j+i ranges from i to i+n-1.
    # t1[i] = dot(a, w[i:i+n]) (with w zero-padded beyond its length)
    
    # I can compute all t1[i] for i=0..2n-2 via:
    # t1 = _full_convolve(w, a[::-1]) then t1[n-1:n-1+n] gives the right values?
    # _full_convolve(w, a[::-1])[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-m+k]
    # We want: t1[i] = sum_k w[k]*a[k-i]
    # Set n-1-m+k = k-i => n-1-m = -i => m = n-1+i
    # So t1[i] = _full_convolve(w, a[::-1])[n-1+i] for i = 0..n-1
    # (m = n-1+i ranges from n-1 to 2n-2, which is valid since output has length 3n-2)
    
    a_rev = a[::-1]
    wc1 = _full_convolve(w, a_rev)
    t1 = wc1[n - 1:2 * n - 1]  # length n, t1[i] = sum_k w[k]*a[k-i]
    
    # t2[i] = sum_k w[k]*a[i-k]
    # = sum_k a[i-k]*w[k]
    # Let j = i-k, k = i-j: sum_j a[j]*w[i-j]
    # = (a * w)[i] (standard convolution!)
    # = _full_convolve(a, w)[i] for i = 0..n-1
    wc2 = _full_convolve(a, w)
    t2 = wc2[0:n]  # t2[i] = (a*w)[i] = sum_j a[j]*w[i-j] = sum_j a[j]*w[i-j]
    # But we want sum_j a[j]*w[i-j] where w has length 2n-1 and j is in [0,n-1]
    # (a*w)[i] = sum_{j=0}^{n-1} a[j]*w[i-j] where w[i-j] is 0 if i-j not in [0,2n-2]
    # This is exactly what we want!
    
    grad_inner = t1 + t2  # sum_k w[k]*(a[k-i] + a[i-k]) for i=0..n-1
    W = float(np.dot(w, c))
    return 4.0 * n * a * (grad_inner - W)


def _hess_vec(a, A_vec, n):
    """Compute Hessian-vector product for the active-set QP.
    H = (2/m) * sum_{k in active} Q_k
    (H*v)[i] = (2/m) * sum_{k in active} (v[k-i] + v[i-k])
    Same structure as the gradient computation above.
    """
    m = int(A_vec.sum())
    if m == 0:
        return np.zeros(n)
    a_rev = a[::-1]  # not used, but keeping pattern
    v = A_vec  # this is the "w" vector (indicator of active set)
    v_rev = v[::-1]
    wc1 = _full_convolve(v, a_rev)  # dummy to get structure... 
    # Actually we need: sum_k v[k]*a2[k-i] + sum_k v[k]*a2[i-k] for a2 = the vector being multiplied
    # Let me rename: we want (H*u)[i] = (2/m) * sum_k A_vec[k] * (u[k-i] + u[i-k])
    # This is the same formula as grad_inner above but with A_vec instead of w, and u instead of a.
    u_rev = u[::-1]
    wc1 = _full_convolve(A_vec, u_rev)
    t1 = wc1[n - 1:2 * n - 1]
    wc2 = _full_convolve(u, A_vec)
    t2 = wc2[0:n]
    return (2.0 / m) * (t1 + t2)


def _active_set_qp(a_init, deadline, max_iter=500, tol_active=0.001):
    """Active-set QP solver with proper non-negativity handling.
    Solves: min a^T H a / m s.t. sum(a)=1, a>=0
    where H = (1/m) sum_{k in active} Q_k (the normalized Hessian of the active objective).
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for efficiency
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n
        
        A_vec = active.astype(np.float64)
        m = n_active
        
        # Compute gradient of f(a) = (1/m) * sum_{k in active} b[k] = (1/m) * a^T (sum Q_k) a / 2
        # Actually b[k] = a^T Q_k a (Q_k symmetric), so (1/m)*sum b[k] = a^T H a where H = (1/(2m))*sum Q_k
        # Gradient = 2*H*a = (1/m)*sum_k Q_k * a
        # (Q_k * a)[i] = a[k-i] + a[i-k]... 
        # Wait: b[k] = sum_j a[j]*a[k-j]. As a quadratic form with symmetric Q_k:
        # b[k] = a^T Q_k a where Q_k[j,l] = 1 if j+l=k (and by symmetry Q_k[l,j]=1 too)
        # But sum_j a[j]*a[k-j] = a^T Q_k a only if Q_k is defined correctly.
        # a^T Q_k a = sum_{j,l} a[j]*Q_k[j,l]*a[l] = sum_j a[j]*a[k-j] (when Q_k[j,k-j]=1)
        # Yes! So b[k] = a^T Q_k a with Q_k[j,l] = 1 if j+l=k.
        # Q_k is symmetric. Hessian of b[k] = Q_k + Q_k^T = 2*Q_k.
        # So grad of (1/m)*sum b[k] = (2/m)*sum Q_k * a
        # (sum Q_k * a)[i] = sum_k sum_l Q_k[i,l]*a[l] = sum_k a[k-i] (where Q_k[i,l]=1 iff i+l=k)
        # = sum_k a[k-i] where k = i+l, l in [0,n-1], k in [0,2n-2]
        # = sum_l a[l] * A_vec[i+l] (where A_vec is indicator of active set)
        # This is a convolution!
        # (sum Q_k * a)[i] = sum_l a[l]*A_vec[i+l] = (a * A_vec_rev)[?]
        # Let A_rev = A_vec[::-1] (length 2n-1)
        # (a * A_rev)[m] = sum_j a[j]*A_rev[m-j] = sum_j a[j]*A_vec[2n-2-(m-j)] = sum_j a[j]*A_vec[2n-2-m+j]
        # We want: sum_l a[l]*A_vec[i+l]. Set 2n-2-m+l = i+l => 2n-2-m = i => m = 2n-2-i
        # So (sum Q_k * a)[i] = (a * A_rev)[2n-2-i]
        
        A_rev = A_vec[::-1]
        conv_result = _full_convolve(a, A_rev)
        grad = (2.0 / m) * conv_result[2*n-2 - np.arange(n)]
        # Project to tangent space (sum = 0)
        g_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break
        
        # Hessian-vector product: (H*v) where H = (1/m)*sum Q_k (note: Hessian is 2H = (2/m)*sum Q_k)
        # For the Newton step, we need to solve (2*H + mu*I + projection) * d = -g_t
        # But 2*H = (2/m)*sum Q_k, so (2*H * v)[i] = (2/m)*sum_l v[l]*A_vec[i+l]
        # Same as grad computation but with v instead of a!
        
        mu = 1e-5 * (c_max / max(m, 1)) + 1e-10
        
        def hvp(v):
            """Hessian-vector product: (2*H*v + mu*v) projected to sum=0 subspace."""
            conv_v = _full_convolve(v, A_rev)
            result = (2.0 / m) * conv_v[2*n-2 - np.arange(n)] + mu * v
            # Project to sum=0
            result = result - np.mean(result)
            return result
        
        # CG to solve (2H + mu*I) d = -g_t in the sum=0 subspace
        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hvp(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x_new = x + alpha * p
                r_new = rhs - hvp(x_new)
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
                    x = x_new
                    break
                p = r_new + (rs_new / rs) * p
                x = x_new
                r = r_new
                rs = rs_new
            return x
        
        d = cg_solve(-g_t)
        # Ensure sum(d) = 0 (it should be by construction, but enforce)
        d = d - np.mean(d)
        
        # Line search with non-negativity
        neg_mask = d < -1e-14
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-d[neg_mask])) * 0.999
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)
        
        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * d
            if np.min(a_try) < -1e-10:
                t *= 0.5
                if t < 1e-15:
                    break
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5
        
        if not accepted:
            # Fallback: gradient step
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        
        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-13:
            break
    
    return best_a, best_sc


def _coordinate_descent(a_init, deadline, n_passes=50):
    """Coordinate descent: for each coordinate, optimize the 1D subproblem."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Compute gradient structure
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Compute the "direction vector" s[i] = sum_k w[k]*(a[k-i] + a[i-k])
        A_rev = w[::-1]
        conv_a_rev = _full_convolve(a, A_rev)
        s_vec = conv_a_rev[2*n-2 - np.arange(n)]
        conv_a_w = _full_convolve(a, w)
        s_vec += conv_a_w[0:n]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (s_vec - W)
        
        # Try to improve by adjusting pairs of coordinates
        # For efficiency, just do a subgradient step with good step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Polyak step size: t = (f - f*) / ||g||^2, but we don't know f*
        # Use adaptive step size
        t = 0.01
        a_try = a - t * grad / gnorm
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        f_try = float(c_try.max())
        
        if f_try < c_max:
            a = a_try
            improved = True
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try smaller step
            t *= 0.5
            a_try = a - t * grad / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    improved = True
        
        if not improved and p > 5:
            break
    
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_scale=1.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0] * 0.1
    base_beta = max(gap, max_c * 0.01) * beta_scale
    
    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.6, 0.35, 0.2, 0.1, 0.06, 0.03,
                                   0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate max iterations based on remaining time
        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 10))))
        
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _make_fejer_profile(n, power=2.0):
    """Fejer-kernel-inspired profile: |sum exp(2piikj/n)|^2 type shape."""
    t = np.arange(n) / n
    # Fejer kernel shape: (sin(n*pi*t / 2) / sin(pi*t / 2))^2 approximated
    # Use a simpler taper: (1 - t^2)^power on [0,1] mapped to [0,n]
    x = np.linspace(-1, 1, n)
    a = np.maximum(1.0 - x**2, 0.0) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_windoubled(n):
    """Window that's a product of two sinc-like functions."""
    x = np.linspace(-1, 1, n)
    a = np.sinc(x / 2.0) ** 2  # sinc^2 window
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cosine_taper(n, asym=0.0):
    """Cosine taper with optional asymmetry."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    if asym != 0.0:
        a = a * (1 + asym * (2 * t - 1))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_optimized_profile(n, kind="fejer"):
    if kind == "fejer":
        return _make_fejer_profile(n)
    elif kind == "sinc2":
        return _make_windoubled(n)
    elif kind == "cos":
        return _make_cosine_taper(n)
    elif kind == "cos_asym":
        return _make_cosine_taper(n, asym=0.3)
    elif kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-t**2 / 4.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage(a, deadline):
    """Try zeroing out small elements to create a sparser sequence."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Diverse initialization and quick L-BFGS
    Phase 2: Main L-BFGS with refined beta schedule
    Phase 3: Active-set QP refinement
    Phase 4: Coordinate descent polish
    Phase 5: Multi-n exploration
    Phase 6: Final refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_optimized_profile(n_parent, "fejer")
        print(f"No parent, using Fejer profile n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # Phase 1: Quick diverse initialization
    print(f"\n[Phase 1] Diverse initialization (t={time.monotonic() - t_start:.1f}s)")
    n_base = n_parent if n_parent > 0 else 8192
    
    # Try several profiles at the parent size
    profiles = ["fejer", "sinc2", "cos", "cos_asym", "gaussian", "uniform"]
    profile_results = []
    for prof in profiles:
        if time.monotonic() > deadline - 600:
            break
        a_try = _make_optimized_profile(n_base, prof)
        sc_try = _score(a_try)
        profile_results.append((sc_try, a_try, prof))
        print(f"  Profile '{prof}': score={sc_try:.10f}")
    
    # Quick L-BFGS on the best 3 profiles
    profile_results.sort(key=lambda x: x[0])
    for sc_try, a_try, prof in profile_results[:3]:
        if time.monotonic() > deadline - 500:
            break
        a_opt = _run_lbfgs(a_try, deadline, min_time_remaining=400.0, beta_scale=0.5)
        update_best(a_opt, _score(a_opt), f"quick_{prof}")
    
    # Phase 2: Main L-BFGS from best
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a_main = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0, beta_scale=1.0)
        update_best(a_main, _score(a_main), "lbfgs_main")
    
    # Phase 3: Active-set QP
    print(f"\n[Phase 3] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        # Round 1: loose tolerance
        as_deadline = min(deadline - 150, time.monotonic() + 120)
        a_as, sc_as = _active_set_qp(best_seq, as_deadline, max_iter=300, tol_active=0.002)
        update_best(a_as, sc_as, "as_qp_1")
        
        # Round 2: tighter tolerance
        if best_seq is not None and time.monotonic() < deadline - 100:
            as_deadline2 = min(deadline - 80, time.monotonic() + 80)
            a_as2, sc_as2 = _active_set_qp(best_seq, as_deadline2, max_iter=200, tol_active=0.001)
            update_best(a_as2, sc_as2, "as_qp_2")
            
            # Round 3: very tight
            if best_seq is not None and time.monotonic() < deadline - 50:
                as_deadline3 = min(deadline - 40, time.monotonic() + 50)
                a_as3, sc_as3 = _active_set_qp(best_seq, as_deadline3, max_iter=150, tol_active=0.0005)
                update_best(a_as3, sc_as3, "as_qp_3")
    
    # Phase 4: Coordinate descent / subgradient polish
    print(f"\n[Phase 4] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        cd_deadline = min(deadline - 50, time.monotonic() + 60)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_deadline, n_passes=100)
        update_best(a_cd, sc_cd, "coord_descent")
        
        # Follow up with active-set QP on the CD result
        if sc_cd < best_score + 1e-10 and time.monotonic() < deadline - 40:
            a_cd2, sc_cd2 = _active_set_qp(a_cd, min(deadline - 30, time.monotonic() + 40),
                                           max_iter=100, tol_active=0.001)
            update_best(a_cd2, sc_cd2, "cd_then_as")
    
    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates and n_parent > 0:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue  # Already optimized at this n
        
        if parent is not None and n_parent != n_try and n_parent > 0:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_optimized_profile(n_try, "fejer")
        
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0, beta_scale=0.8)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")
        
        # Active-set QP on this
        if time.monotonic() < deadline - 30 and sc_opt < best_score + 0.05:
            a_opt2, sc_opt2 = _active_set_qp(a_opt, min(deadline - 20, time.monotonic() + 40),
                                             max_iter=100, tol_active=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")
    
    # Phase 6: Diverse restarts from best
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restarts = [
            ("perturb_005", lambda: _perturb(best_seq, 0.05, rng)),
            ("perturb_015", lambda: _perturb(best_seq, 0.15, rng)),
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=8)),
            ("fejer_blend", lambda: 0.5 * best_seq + 0.5 * _make_optimized_profile(len(best_seq), "fejer")),
            ("sinc_blend", lambda: 0.5 * best_seq + 0.5 * _make_optimized_profile(len(best_seq), "sinc2")),
        ]
        
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 15:
                break
            p_start = gen_fn()
            p_start = np.maximum(p_start, 0)
            s_p = p_start.sum()
            if s_p < 1e-10:
                continue
            p_start = p_start / s_p
            sc_start = _score(p_start)
            
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0, beta_scale=0.5)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")
            
            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 8:
                a_r2, sc_r2 = _active_set_qp(a_r, min(deadline - 5, time.monotonic() + 20),
                                             max_iter=50, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")
    
    # Phase 7: Final polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active-set QP
        as_final = min(deadline - 3, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_qp(best_seq, as_final, max_iter=200, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as")
        
        # One more L-BFGS with tiny beta
        if time.monotonic() < deadline - 4:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n_final = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n_final),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n_final),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                  'maxcor': 50, 'maxls': 50}
                    )
                    a_fin2 = _softmax(result.x)
                    sc_fin2 = _score(a_fin2)
                    update_best(a_fin2, sc_fin2, "final_tiny_beta")
                except Exception:
                    pass
        
        # Final shrinkage attempt
        if time.monotonic() < deadline - 3:
            a_shr, sc_shr = _shrinkage(best_seq, deadline - 2)
            update_best(a_shr, sc_shr, "final_shrink")
    
    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_optimized_profile(256, "fejer")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_optimized_profile(256, "fejer")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
