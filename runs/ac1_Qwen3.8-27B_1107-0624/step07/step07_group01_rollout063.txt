import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

One more thought: the "spectral flattening" operation. Here's a concrete algorithm:

1. Compute r = conv(p, p) (length 2n-1).
2. Compute the "flattened" target: r_target[k] = max(r[k] - γ, r_min) where r_min is the minimum of r, and γ is a "shrinkage" parameter. Or: r_target = r^β for β < 1 (power compression).
3. Compute R = FFT(p) (padded to length N >= 2n-1).
4. Compute |R_new|² = FFT(r_target) (padded).
5. Update: R_new = sqrt(|R_new|²) * phase(R) = sqrt(FFT(r_target)) * R / |R| (where defined).
6. p_new = IFFT(R_new)[0:n].
7. Project: p_new = max(p_new, 0), p_new = p_new / sum(p_new).

This is a form of "error reduction" in the frequency domain. The parameter γ (or β) controls how aggressively we flatten.

def _spectral_flatten(p, gamma=0.5, iters=50):
    """Apply iterative spectral flattening."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    p = p.copy().astype(np.float64)
    s = p.sum()
    p = p / s
    
    best_p = p.copy()
    best_sc = _score(p)
    
    for it in range(iters):
        # Compute autocorrelation
        c = _full_convolve(p, p)
        c_max = c.max()
        sc = 2.0 * n * c_max
        
        if sc < best_sc:
            best_sc = sc
            best_p = p.copy()
        
        # Compute FFT of p (padded)
        P = np.fft.rfft(p, N)
        
        # Compute |P|²
        P_abs2 = np.abs(P) ** 2
        
        # "Flatten" the autocorrelation in time domain
        # r = IFFT(|P|²) should be close to c
        # We want to reduce the peak
        c_flat = c.copy()
        c_peak = c_flat.max()
        c_floor = c_flat.min()
        # Compress: values above the mean get pulled down
        c_mean = c_flat.mean()
        c_flat = c_mean + (c_flat - c_mean) * (1 - gamma)
        c_flat = np.maximum(c_flat, 0)
        
        # Get new |P|² from flattened autocorrelation
        P_abs2_new = np.fft.rfft(c_flat, N)
        P_abs2_new = np.maximum(P_abs2_new, 0)  # ensure non-negative
        
        # Update P: keep phase, change amplitude
        phase_P = P / np.maximum(np.abs(P), 1e-30)
        P_new = np.sqrt(P_abs2_new) * phase_P
        
        # Time domain
        p_new = np.fft.irfft(P_new, N)[:n]
        p_new = np.maximum(p_new, 0)
        s_new = p_new.sum()
        if s_new < 1e-15:
            break
        p_new = p_new / s_new
        
        # Check improvement
        sc_new = _score(p_new)
        if sc_new < sc + 1e-10:
            p = p_new
        else:
            # If it got worse, try smaller gamma
            break
    
    return best_p, best_sc

"""
Optimizer for minimizing 2*n*max(conv(p,p))/(sum(p))^2 using:
1. Chirp-based and structured initializations with parameter sweeps
2. Adaptive L-BFGS with log-sum-exp smoothing (coarse phase)
3. Active-set gradient descent with soft active set and backtracking line search (fine phase)
4. Spectral perturbations for diversification
5. Multi-length exploration (n=8192 to 32768)
6. KKT-inspired final polishing with dynamic active set
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    try:
        result = minimize(_smoobj, z0, jac=_smograd, args=(beta, n),
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7])

    for i, beta in enumerate(betas):

            break

    return current_a


def _compute_gradient(p, active):
    """Compute aggregate gradient for active set."""
    n = len(p)
    m = len(active)
    G = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = p[lo:k - lo + 1][::-1] if k - lo < n else p[lo:min(n, k - lo + 1)][::-1]
            # p[k-j] for j in [lo, hi] = p[k-hi], p[k-hi+1], ..., p[k-lo]
            # reversed: p[k-lo], p[k-lo-1], ..., p[k-hi]
            idx_lo = k - hi
            idx_hi = k - lo
            G[lo:hi + 1] += (2.0 / m) * p[idx_lo:idx_hi + 1][::-1]
    return G


def _active_gradient_descent(p_init, deadline, soft_frac=0.97, max_iter=500000):
    """
    Active-set gradient descent with soft active set.
    The gradient of r[k] w.r.t. p[j] = 2*p[k-j] if 0<=k-j<n, else 0.
    """
    n = len(p_init)
    p = p_init.copy().astype(np.float64)
    s = p.sum()
    if s < 1e-15:
        return p_init, np.inf
    p = p / s

    best_p = p.copy()
    best_sc = _score(p)
    sc_cur = best_sc

    alpha = 1.0
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(p, p)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max

        # Soft active set
        threshold = soft_frac * c_max
        active = np.where(c >= threshold)[0]
        m = len(active)
        if m == 0:
            break

        # Compute gradient
        G = _compute_gradient(p, active)
        G = G - G.mean()

        gnorm = np.linalg.norm(G)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        a = alpha
        improved = False
        for ls in range(40):
            p_try = p - a * G
            p_try = np.maximum(p_try, 0.0)
            s_try = p_try.sum()
            if s_try < 1e-15:
                a *= 0.5
                continue
            p_try = p_try / s_try
            sc_try = _score(p_try)

            if sc_try < sc_cur - 1e-18:
                p = p_try
                sc_cur = sc_try
                improved = True
                break
            a *= 0.5

        if improved:
            if sc_cur < best_sc:
                best_sc = sc_cur
                best_p = p.copy()
                stall = 0
                alpha = min(a * 1.05, 10.0)
            else:
                stall += 1
                if stall > 100:
                    alpha *= 0.7
        else:
            alpha *= 0.4
            stall += 1

        if stall > 1000:
            # Restart with perturbation
            z = np.log(np.maximum(p, 1e-18))
            z += 0.01 * np.random.standard_normal(n)
            z = z - z.mean()
            p = _softmax(z)
            alpha = 0.5
            stall = 0

    return best_p, best_sc

        # H[i,j] = 2/m if i+j in active

def _make_chirp(n, freq=1.0, eps=0.3, asym=0.0):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    if asym != 0.0:
        a *= (1.0 + asym * np.cos(2.0 * np.pi * t / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2, phases=None):
    if bands is None:
        bands = [1, 3]
    if phases is None:
        phases = [0.0] * len(bands)
    t = np.arange(n)
    a = np.ones(n)
    for i, f in enumerate(bands):
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n + phases[i])
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5, skew=0.0):
    t = np.linspace(0, 1, n)
    if skew != 0.0:
        t = t ** (1.0 + skew)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _spectral_perturb(p, rng, n_perturbs=3, amp=0.03):
    """Apply random spectral (Fourier) perturbations."""
    n = len(p)
    z = np.log(np.maximum(p, 1e-18))
    for _ in range(n_perturbs):
        freq = int(rng.integers(1, max(2, n // 8)))
        a = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        t = np.arange(n)
        z = z + a * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
