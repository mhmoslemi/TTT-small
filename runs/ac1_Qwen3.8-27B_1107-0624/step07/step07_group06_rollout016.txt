    w = np.zeros_like(c)
    w[active] = 1.0
    # grad[i] = 2 * sum_j w[i+j] * a[j] (for valid indices)
    # = 2 * (flip(a) convolved with w)[n-1 + i]
    N = _next_pow2(4 * n)
    fa = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    r = np.fft.irfft(fa * fw, N)
    return 2.0 * r[n - 1:2 * n - 1]


def _active_hessian_matvec(v, active, n):
    """Compute H*v where (Hv)_i = 2 * sum_j v[j] * active[i+j]."""
    N = _next_pow2(4 * n)
    fv = np.fft.rfft(v[::-1], N)
    fa = np.fft.rfft(active.astype(float), N)
    r = np.fft.irfft(fv * fa, N)
    return 2.0 * r[n - 1:2 * n - 1]


def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.003):
    """Newton's method on active-set subproblem with regularization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        grad = _active_gradient(a, c, active)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        # Build regularized Hessian and solve
        mu = max(1e-10, 1e-8 * gnorm)
        active_f = active.astype(float)

        def hess_vec(v):
            return _active_hessian_matvec(v, active, n) + mu * v

        # Try direct sparse solve first (for smaller n), else CG
        if n <= 16384:
            # Build sparse Hessian
            # H[i,j] = 2 * active[i+j] if i+j <= 2n-2, else 0
            # This is a Hankel-like matrix
            try:
                rows = []
                cols = []
                vals = []
                # For each i, H[i,:] has nonzeros where active[i+j] != 0
                active_indices = np.nonzero(active_f)[0]
                for j in active_indices:
                    # H[i, j-i] = 2 for valid i, and H[i, i-(j-n)... ] 
                    # Actually H[i,j] = 2*active[i+j] for 0<=i<n, 0<=j<n, i+j<=2n-2
                    i_min = 0
                    i_max = min(n - 1, 2 * n - 2 - j)
                    if i_max > i_min:
                        rows.extend(range(i_min, i_max + 1))
                        cols.extend([j] * (i_max - i_min + 1))
                        vals.extend([2.0] * (i_max - i_min + 1))
                # Add diagonal for mu
                rows.extend(range(n))
                cols.extend(range(n))
                vals.extend([mu] * n)

                H_sp = csr_matrix((vals, (rows, cols)), shape=(n, n))
                # Make symmetric (it should already be since active[i+j] = active[j+i])
                H_sp = (H_sp + H_sp.T) * 0.5
                H_sp = H_sp + sp_diags([mu], [0], (n, n), format='csr')

                # Solve (H + mu*I) dx = -grad_t
                b_rhs = -grad_t
                dx = spsolve(H_sp, b_rhs)
                if not np.all(np.isfinite(dx)):
                    dx = np.zeros(n)
            except Exception:
                # Fallback to CG
                x = np.zeros(n)
                r = b_rhs.copy()
                p = r.copy()
                rs = float(np.dot(r, r))
                b_norm = float(np.linalg.norm(b_rhs))
                for _ in range(100):
                    Hp = hess_vec(p)
                    pH = float(np.dot(p, Hp))
                    if abs(pH) < 1e-30:
                        break
                    alpha = rs / pH
                    x = x + alpha * p
                    r = r - alpha * Hp
                    r_new = float(np.dot(r, r))
                    if np.sqrt(r_new) < 1e-10 * b_norm:
                        break
                    beta_cg = r_new / rs if rs > 1e-30 else 0.0
                    p = r + beta_cg * p
                    rs = r_new
                dx = x
        else:
            # CG for large n
            x = np.zeros(n)
            r = (-grad_t).copy()
            p = r.copy()
            rs = float(np.dot(r, r))
            b_norm = float(np.linalg.norm(grad_t))
            for _ in range(150):
                Hp = hess_vec(p)
                pH = float(np.dot(p, Hp))
                if abs(pH) < 1e-30:
                    break
                alpha = rs / pH
                x = x + alpha * p
                r = r - alpha * Hp
                r_new = float(np.dot(r, r))
                if np.sqrt(r_new) < 1e-10 * b_norm:
                    break
                beta_cg = r_new / rs if rs > 1e-30 else 0.0
                p = r + beta_cg * p
                rs = r_new
            dx = x

        dx = dx - np.mean(dx)

        # Line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _projected_gradient_momentum(a_init, deadline, n_steps=5000, lr_init=0.01, momentum=0.9):
    """Projected gradient descent with momentum on the max-conv objective."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a_init)
    v = np.zeros(n)  # momentum
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _active_gradient(a, c, active)
        grad_t = grad - np.mean(grad)

        v = momentum * v - lr * grad_t
        a_new = a + v
        a_new = _project_simplex(a_new)

        # Adaptive learning rate
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.005, lr_init * 2)
        else:
            a = a_new  # still update position
            lr *= 0.95
            if i % 100 == 0 and lr < lr_init * 0.1:
                # Reset from best
                a = best_a.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5

    return best_a, best_sc


def _peak_shaving(a_init, deadline, n_rounds=200):
    """Iteratively identify and shave the highest peaks by redistributing mass."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a_init)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find top K peaks
        K = min(10, len(c) // 4)
        top_k = np.argsort(c)[::-1][:K]

        # For each top peak, identify contributing elements and reduce them
        for k_pos in top_k:
            if time.monotonic() > deadline:
                break
            # c[k_pos] = sum_i a[i] * a[k_pos - i]
            # Elements that contribute: pairs (i, k_pos - i) both in [0, n-1]
            i_min = max(0, k_pos - (n - 1))
            i_max = min(n - 1, k_pos)
            if i_min > i_max:
                continue

            # Compute contribution of each element to this peak
            # Element i contributes a[i]*a[k_pos-i] (and by symmetry a[k_pos-i]*a[i])
            # Total contribution of element j to peak k: sum over pairs involving j
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = k_pos - i
                contrib[i] += a[j]
                contrib[j] += a[i]

            # Reduce the elements with highest contribution
            top_contrib = np.argsort(contrib)[::-1][:min(5, n // 10)]
            total_reduce = 0.0
            for idx in top_contrib:
                reduce_amt = 0.1 * a[idx]
                a[idx] -= reduce_amt
                total_reduce += reduce_amt

            # Redistribute to elements with low contribution to ALL top peaks
            # Find elements that contribute least to all top peaks
            low_contrib = np.argsort(contrib)[:min(5, n // 10)]
            per_elem = total_reduce / max(1, len(low_contrib))
            for idx in low_contrib:
                a[idx] += per_elem

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with smoothed max and aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                  0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per function eval
        est_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _freq_flatten(a_init, deadline, n_steps=2000):
    """Frequency-domain approach: minimize variance of |FFT(a)|^2 to flatten autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a_init)
    lr = 0.003

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Compute autocorrelation variance gradient
        c = _full_convolve(a, a)
        mean_c = np.mean(c)
        centered = c - mean_c
        # d(var)/da = 2 * (a_rev conv centered)[n-1:2n-1]
        dcd_a = _full_convolve(a[::-1], centered)
        g = 2.0 * dcd_a[n - 1:2 * n - 1]
        g = g - np.mean(g)

        a_new = a - lr * g
        a_new = _project_simplex(a_new)

        v_new = float(np.mean((c - mean_c) ** 2))
        # Check if score improved
        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.01)
        else:
            a = a_new
            lr *= 0.9
            if lr < 1e-6:
                a = best_a.copy()
                lr = 0.003

    return best_a, best_sc


def _perturb_random(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_structured(a, rng, n_perturb=10):
    """Perturb a small number of random positions significantly."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_perturb):
        idx = rng.integers(0, n)
        a_new[idx] *= rng.uniform(0.3, 3.0)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: L-BFGS smoothed (from parent) ===
    print(f"\n[Phase 1] L-BFGS smoothed (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _lbfgs_smoothed(best_seq, deadline, min_time_remaining=550)
        sc1 = _score(a1)
        update_best(a1, sc1, "lbfgs_p1")

    # === Phase 2: Newton active-set (wide tolerance) ===
    print(f"\n[Phase 2] Newton active-set wide (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        nd = min(deadline - 500, time.monotonic() + 200)
        a2, sc2 = _newton_active_set(best_seq, nd, max_iter=300, tol_frac=0.005)
        update_best(a2, sc2, "newton_wide")

    # === Phase 3: Projected gradient with momentum ===
    print(f"\n[Phase 3] Projected gradient momentum (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 450:
        pgd_deadline = min(deadline - 450, time.monotonic() + 150)
        a3, sc3 = _projected_gradient_momentum(best_seq, pgd_deadline, n_steps=3000, lr_init=0.008, momentum=0.9)
        update_best(a3, sc3, "proj_grad")

    # === Phase 4: Peak shaving ===
    print(f"\n[Phase 4] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        ps_deadline = min(deadline - 400, time.monotonic() + 120)
        a4, sc4 = _peak_shaving(best_seq, ps_deadline, n_rounds=100)
        update_best(a4, sc4, "peak_shave")
        if sc4 < best_score:
            # Polish with Newton
            if time.monotonic() < deadline - 350:
                a4b, sc4b = _newton_active_set(a4, min(deadline - 350, time.monotonic() + 60), max_iter=100, tol_frac=0.003)
                update_best(a4b, sc4b, "peak_shave_newton")

    # === Phase 5: Frequency flattening ===

        update_best(a5, sc5, "freq_flat")

            x_new = np.linspace(0, 1, n_try)

            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

            update_best(a_init, sc_init, f"n_{n_try}_init")

    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:

        for i in range(n_restarts):

                break
            if i == 0:
                p_start = _perturb_random(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_structured(best_seq, rng, n_perturb=20)
            elif i == 2:
                p_start = _perturb_random(best_seq, 0.05, rng)
            elif i == 3:
                p_start = _perturb_structured(best_seq, rng, n_perturb=5)
            elif i == 4:
                # Reverse + perturb
                p_start = best_seq[::-1].copy()
                p_start = _perturb_random(p_start, 0.01, rng)
            else:
                # Block shuffle
                n = len(best_seq)
                a_new = best_seq.copy()
                block = n // 8
                idx1 = rng.integers(0, n - 2 * block)
                idx2 = rng.integers(0, n - 2 * block)
                if abs(idx1 - idx2) > block:
                    a_new[idx1:idx1 + block], a_new[idx2:idx2 + block] = \
                        a_new[idx2:idx2 + block].copy(), a_new[idx1:idx1 + block].copy()
                p_start = a_new / a_new.sum()

            # Quick optimization
            a_r = _lbfgs_smoothed(p_start, deadline, min_time_remaining=10)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _newton_active_set(a_r, min(deadline - 15, time.monotonic() + 20), max_iter=40, tol_frac=0.004)
                update_best(a_r2, sc_r2, f"restart_{i}_nt")

    # === Phase 9: Final intensive polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very low beta L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            for beta_factor in [0.001, 0.0005, 0.0002, 0.0001]:
                if time.monotonic() > deadline - 5:
                    break
                beta_final = mb * beta_factor
                rem2 = deadline - 5 - time.monotonic()
                if rem2 < 5:
                    break
                max_iter = min(2000, max(50, int(rem2 / 0.005)))

                def fun_fin(z_local):
                    return _smoobj(z_local, beta_final, n)

                def jac_fin(z_local):
                    return _smograd(z_local, beta_final, n)

                try:
                    result = minimize(
                        fun_fin, z, jac=jac_fin, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 50, 'maxls': 100}
                    )
                    a_fin = _softmax(result.x)
                    sc_fin = _score(a_fin)
                    update_best(a_fin, sc_fin, f"final_beta_{beta_factor}")
                    z = np.log(np.maximum(best_seq, 1e-30))
                    z = z - z.mean()
                except Exception:
                    pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(best_seq, max(deadline - 3, time.monotonic()), max_iter=50, tol_frac=0.002)
            update_best(a_fin2, sc_fin2, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
