import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: d/dz [2n * max-sum-exp] = 4n * a * (conv(w_reversed, a)[n-1:2n-1] - w@c)
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 120,
                                   'maxls': 100, 'disp': False})
        return _softmax(result.x), result.success, int(result.nit)
    except Exception:
        return a_init.copy(), False, 0

                              1e-5, 5e-6, 2e-6, 1e-6, 5e-7,
                              2e-7, 1e-7, 5e-8, 2e-8, 1e-8])
    current_a = a_start.copy()
    current_sc = _score(current_a)

            print(f"    beta={beta:.3e}: score={sc:.12f} (iter={niter})")

    # I'll just compute it as: xcor_pos[j] = sum_{i=0}^{n-1} a[i] * v[i+j]
    # where v has length 2n-1 (indices 0 to 2n-2).
    # v[i+j] is valid when 0 <= i+j <= 2n-2, i.e., -j <= i <= 2n-2-j.
    # Since i is in [0, n-1], we need i <= min(n-1, 2n-2-j) and i >= max(0, -j) = 0 (since j>=0).
    # So: xcor_pos[j] = sum_{i=0}^{min(n-1, 2n-2-j)} a[i] * v[i+j]
    #
    # This can be computed as: take the first n elements of a, multiply element-wise
    # with v[j: j+n], and sum. For all j simultaneously, this is a "sliding dot product"
    # which can be done via FFT.
    #
    # xcor_pos[j] = <a, v[j:j+n]> = <pad(a, 2n-1), v shifted by -j>
    # = conv(a, v)[j + n - 1] ... no.
    # 
    # Let me just use: xcor_pos = np.correlate(v, a, mode='full')[n-1:n-1+n]
    # np.correlate(v, a, 'full')[k] = sum_i v[i] * a[i + k - (n-1)]
    # For k = n-1+j: sum_i v[i] * a[i+j] ... that's what we want (with i and a swapped)!
    # np.correlate(v, a, 'full')[n-1+j] = sum_i v[i] * a[i - (n-1) + (n-1+j)] = sum_i v[i] * a[i+j] ✓
    
    xcor_full = np.correlate(v := w, a, mode='full')
    # xcor_full[k] = sum_i w[i] * a[i + k - (n-1)]
    # We want sum_i a[i] * w[i+j] = sum_i w[i] * a[i+j]
    # = xcor_full[k] where i + k - (n-1) = i + j => k = n-1+j
    xcor_pos = xcor_full[n - 1:n - 1 + n]  # for j = 0, ..., n-1
    
    grad = conv_av + xcor_pos
    return grad


def _active_set_refine(a_init, deadline, max_rounds=50):
    """
    Active-set KKT refinement. At KKT point with active set S and all a>0:
    sum_{k in S} w_k * grad(c[k]) = lambda * 1
    sum a = 1
    We solve this linear system iteratively, updating S each round.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for round_i in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Identify active set: lags within relative tolerance of max
        # Use a tight tolerance
        tol = 1e-6 * c_max
        active_mask = c >= (c_max - tol)
        n_active = active_mask.sum()
        
        if n_active < 2:
            # Only one active lag - this is the uniform solution, can't improve
            break
        
        if n_active > 2 * n:
            # Too many active, tighten tolerance
            tol = 1e-7 * c_max
            active_mask = c >= (c_max - tol)
            n_active = active_mask.sum()
            if n_active < 2:
                break
        
        # Compute gradient of sum_{k in S} c[k] / |S|
        w = np.zeros(2 * n - 1)
        w[active_mask] = 1.0 / n_active
        
        grad = _gradient_of_max(a, c, n, w)
        
        # Center the gradient (remove mean to project onto sum=0 subspace)
        grad_centered = grad - grad.mean()
        
        # Check if we're at a critical point
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-12 * np.linalg.norm(a) + 1e-15:
            print(f"    Active set: converged (gnorm={gnorm:.2e}) at round {round_i}")
            break
        
        # Solve: find delta such that a + delta is in direction that reduces max(c)
        # Use Newton-like step: we want (B + delta_B) * (a + delta) ~ const
        # Linearize: B * delta = -grad_centered (projected)
        # But B = 2 * sum w_k Q_k is expensive to form.
        # Instead, use a steepest descent step with line search in the projected space.
        
        # Normalized descent direction
        direction = -grad_centered / gnorm
        
        # Line search: try multiple step sizes
        best_step = 0.0
        best_sc_try = _score(a)
        
        # Initial step size estimate
        step0 = 1e-4
        for frac in [4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005]:
            step = step0 * frac
            a_try = a + step * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-18:
                best_sc_try = sc_try
                best_step = step
        
        if best_step > 0:
            a = (a + best_step * direction)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_sc_try < best_sc - 1e-18:
                best_sc = best_sc_try
                best_a = a.copy()
                step0 = min(step0 * 1.1, 1e-2)
            print(f"    Active set round {round_i}: n_active={n_active}, "
                  f"step={best_step:.2e}, score={best_sc:.12f}")
        else:
            # Try smaller steps
            improved = False
            for frac in [0.001, 0.0005, 0.0002, 0.0001]:
                step = step0 * frac
                a_try = a + step * direction
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-18:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    improved = True
                    print(f"    Active set round {round_i}: small step, score={best_sc:.12f}")
                    break
            if not improved:
                break
    
    return best_a


def _peak_shaving_refined(a_init, deadline, max_iters=30000):
    """Direct-space peak shaving with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 5e-5
    stall = 0
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Adaptive active set
        tol = max(1e-5 * c_max, (c_max - np.sort(c)[::-1][min(50, len(c)-1)]))
        active = c >= (c_max - min(tol, 0.3 * c_max))
        if active.sum() < 3:
            active = np.zeros_like(c, dtype=bool)
            idx = np.argsort(c)[::-1][:5]
            active[idx] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        grad = _gradient_of_max(a, c, n, w)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        
        if gnorm < 1e-14:
            break
        
        direction = -grad_centered / gnorm
        
        best_step = 0.0
        best_sc_try = _score(a)
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a + step * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-18:
                best_sc_try = sc_try
                best_step = step
        
        if best_step > 0:
            a = np.maximum(a + best_step * direction, 0.0)
            a = a / a.sum()
            if best_sc_try < best_sc - 1e-18:
                best_sc = best_sc_try
                best_a = a.copy()
                eta = min(eta * 1.02, 5e-4)
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            eta *= 0.85
            if eta < 1e-10 or stall > 200:
                break
        
        if i % 5000 == 0 and i > 0:
            print(f"    Peak shave iter {i}: score={best_sc:.12f}, eta={eta:.2e}")
    
    return best_a


def _kkt_linear_solve(a_init, active_mask, n):
    """
    Try to solve the KKT linear system:
    2 * sum_{k in S} w_k Q_k a = lambda * 1,  sum a = 1
    by finding a in the null space direction.
    Returns the solution a or None if it fails.
    """
    # At KKT: B * a = lambda * 1 where B = 2*sum w_k Q_k
    # This means B*a is constant, i.e., (B - mean(B)*J/n) * a = 0 where J is all-ones
    # Plus sum a = 1.
    # 
    # Alternative: minimize ||B*a - mean(B*a)||^2 subject to sum a = 1, a >= 0
    # This is a QP. For now, use a simpler approach:
    # If B is a sum of shifts, B*a at position i is sum of shifted a's with weights.
    # At the solution, this must be constant.
    # 
    # For a practical approach: use the gradient direction to find a correction.
    # The idea: if c[k] for k in S are all equal to t*, and the gradient condition holds,
    # we're at KKT. Otherwise, we need to adjust a.
    #
    # Simple approach: solve the normal equations approximately using a few
    # conjugate gradient steps on the projected problem.
    
    c = _full_convolve(a_init, a_init)
    w = np.zeros_like(c)
    w[active_mask] = 1.0 / active_mask.sum()
    
    # We want to find delta (with sum delta = 0) that minimizes
    # ||B*delta + (B*a - t* * 1)||^2 where t* = mean of active c values
    # This is a least squares problem: minimize ||B*delta + r||^2 s.t. 1^T delta = 0
    # where r = B*a - t* * 1
    
    # For efficiency, use a few gradient steps on the projected space.
    t_star = c[active_mask].mean()
    
    a_curr = a_init.copy()
    for _ in range(20):
        c_curr = _full_convolve(a_curr, a_curr)
        w = np.zeros_like(c_curr)
        active_new = c_curr >= (c_curr.max() - 1e-5 * c_curr.max())
        if active_new.sum() < 3:
            break
        w[active_new] = 1.0 / active_new.sum()
        
        grad = _gradient_of_max(a_curr, c_curr, n, w)
        grad_proj = grad - grad.mean()
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-14:
            break
        
        # Newton step: step = -grad_proj / gnorm * alpha
        # Choose alpha to reduce the objective
        obj_val = (c_curr[active_new] - t_star) ** 2).sum()
        # Actually just do a line search
        direction = -grad_proj / gnorm
        best_alpha = 0.0
        best_obj = obj_val
        
        for alpha in [1e-4, 5e-5, 2e-5, 1e-5, 5e-6]:
            a_try = a_curr + alpha * direction
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            active_try = c_try >= (c_try.max() - 1e-5 * c_try.max())
            obj_try = float((c_try[active_try] - c_try.max()) ** 2).sum() + 0.5 * (c_try.max() - c_try[active_try].mean()) ** 2 * active_try.sum()
            # Simpler: just use the score
            sc_try = _score(a_try)
            if sc_try < _score(a_curr):
                best_obj = sc_try
                best_alpha = alpha
                a_curr = a_try
        
        if best_alpha == 0.0:
            break
    
    return a_curr


def _spectral_polish(a_init, deadline, n_modes_max=8):
    """
    Parameterize a[i] = base[i] * (1 + sum_j eps_j * cos(2*pi*f_j*i/n + phi_j))
    and optimize the parameters.
    """
    n = len(a_init)
    base = a_init.copy()
    base = base / base.sum()
    
    best_a = base.copy()
    best_sc = _score(base)
    
    # Try adding low-frequency cosine modulations
    for n_modes in range(1, n_modes_max + 1):
        if time.monotonic() > deadline:
            break
        
        for trial in range(3):
            if time.monotonic() > deadline:
                break
            
            # Random frequencies (low frequency to keep the sequence smooth)
            freqs = np.sort(rng_global.integers(1, max(2, n // (50 * n_modes) + 1), size=n_modes))
            phases = rng_global.uniform(0, 2 * np.pi, size=n_modes)
            amplitudes_init = np.full(n_modes, 0.01)
            
            # Optimize amplitudes (and maybe phases) using coordinate descent
            amps = amplitudes_init.copy()
            phis = phases.copy()
            
            for coord_iter in range(200):
                if time.monotonic() > deadline:
                    break
                
                improved = False
                for j in range(n_modes):
                    # Optimize amplitude j
                    a_mod = base * (1.0 + sum(amps[k] * np.cos(2.0 * np.pi * freqs[k] * np.arange(n) / n + phis[k]) for k in range(n_modes)))
                    # Actually, let's just evaluate with different amplitude for j
                    current_amp = amps[j]
                    for test_amp in np.linspace(-0.1, 0.1, 21):
                        if abs(test_amp - current_amp) < 0.001:
                            continue
                        mod = np.ones(n)
                        for k in range(n_modes):
                            amp_k = test_amp if k == j else amps[k]
                            mod = mod + amp_k * np.cos(2.0 * np.pi * freqs[k] * np.arange(n) / n + phis[k])
                        a_try = base * mod
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        sc_try = _score(a_try)
                        if sc_try < best_sc - 1e-15:
                            best_sc = sc_try
                            best_a = a_try.copy()
                            amps[j] = test_amp
                            improved = True
                
                if not improved:
                    break
            
            if best_sc < _score(base) * (1 - 1e-8):
                base = best_a.copy()
    
    return best_a


def _try_different_n(a_best, target_n, deadline):
    """Interpolate to different length and optimize."""
    a_interp = np.interp(
        np.linspace(0, 1, target_n),
        np.linspace(0, 1, len(a_best)),
        a_best
    )
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. L-BFGS with log-sum-exp (fine beta continuation)
    2. Active-set KKT refinement
    3. Peak shaving in direct space
    4. Spectral polishing (low-freq cosine modulation)
    5. Multi-restart with perturbation
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    global rng_global
    rng_global = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.12f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.12f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(8192) / 8192.0
        print(f"No parent. Using uniform n=8192, score={_score(a0):.12f}")
        try_update(a0, "init")

    if best_seq is None:
        best_seq = np.ones(4096) / 4096.0
        best_score_val = _score(best_seq)

    n = len(best_seq)
    print(f"Starting: n={n}, score={best_score_val:.12f}")

    # === Phase 1: L-BFGS with fine continuation ===
    print(f"\n=== Phase 1: L-BFGS continuation (n={n}) ===")
    t_phase = time.monotonic()
    a_p1 = _lbfgs_continuation(best_seq, deadline - 800, reserve_s=700.0)
    try_update(a_p1, "lbfgs_p1")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.12f}")

    # === Phase 2: Active-set refinement ===
    print(f"\n=== Phase 2: Active-set KKT refinement ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 600:
        a_p2 = _active_set_refine(best_seq, deadline - 600, max_rounds=200)
        try_update(a_p2, "active_set")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400:
        a_p3 = _peak_shaving_refined(best_seq, deadline - 400, max_iters=50000)
        try_update(a_p3, "peak_shave")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: More L-BFGS from current best ===
    print(f"\n=== Phase 4: Second L-BFGS pass ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_p4 = _lbfgs_continuation(best_seq, deadline - 300, reserve_s=250.0)
        try_update(a_p4, "lbfgs_p4")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Peak shaving again ===
    print(f"\n=== Phase 5: Second peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 150:
        a_p5 = _peak_shaving_refined(best_seq, deadline - 150, max_iters=30000)
        try_update(a_p5, "peak_shave_2")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Spectral polish ===
    print(f"\n=== Phase 6: Spectral polish ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 80:
        a_p6 = _spectral_polish(best_seq, deadline - 80, n_modes_max=5)
        try_update(a_p6, "spectral")
        if time.monotonic() < deadline - 60:
            a_p6b = _peak_shaving_refined(best_seq, deadline - 60, max_iters=10000)
            try_update(a_p6b, "post_spectral_ps")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Try interpolated larger n ===
    print(f"\n=== Phase 7: Larger n exploration ===")
    t_phase = time.monotonic()
    if n < 12288 and time.monotonic() < deadline - 200:
        target_n = 12288

    # === Phase 8: Final intensive polish ===

    t_phase = time.monotonic()
    for cycle in range(5):
        if time.monotonic() > deadline - 5:

        # Peak shave
        if time.monotonic() < deadline - 5:
            a_f = _peak_shaving_refined(best_seq, deadline - 5, max_iters=10000)
            try_update(a_f, f"final_ps_{cycle}")
        
        # Active set
        if time.monotonic() < deadline - 5:
            a_f2 = _active_set_refine(best_seq, deadline - 5, max_rounds=50)
            try_update(a_f2, f"final_as_{cycle}")
    
    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.12f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096.0
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
