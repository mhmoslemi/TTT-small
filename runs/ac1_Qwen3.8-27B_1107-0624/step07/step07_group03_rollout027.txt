Active-set KKT optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: L-BFGS log-sum-exp init, adaptive active-set Newton with sparse Hessian CG,
KKT fixed-point iteration, diverse restarts, and shrink-and-reoptimize.
Key novelty: direct exploitation of Hessian sparsity structure H[i,j] = (1/|S|)(1_{i+j in S} + 1_{i-j in S})
and a KKT-based fixed-point iteration that drives the cross-correlation toward constancy.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _hess_matvec_active(v, active, n, m, mu):
    """Hessian-vector product for the active-set subproblem.
    H[i,j] = (1/m) * (1_{i+j in active} + 1_{i-j in active})
    (H*v)[i] = (1/m) * sum_j (1_{i+j in active} + 1_{i-j in active}) * v[j]
    = (1/m) * (sum_{j: i+j in active} v[j] + sum_{j: i-j in active} v[j])
    """
    # sum_j 1_{i+j in active} v[j] = cross-correlation of v with active indicator
    # active has length 2n-1, v has length n
    # (v truncated cross-correlation with active)[i] = sum_j v[j] * active[i+j]
    N = _next_pow2(2 * n)
    fv = np.fft.rfft(v, N)
    fa = np.fft.rfft(active.astype(np.float64), N)
    # Cross-correlation: ifft(conj(FV) * FA) gives sum_j v[j]*a[i+j]
    corr1 = np.fft.irfft(np.conj(fv) * fa, N)[:n]
    # sum_j 1_{i-j in active} v[j] = sum_j v[j] * active[i-j]
    # This is: ifft(FV * conj(FA))[i] = sum_j v[j] * active[-(j-i)]... 
    # Actually: correlation at lag i of v and active = sum_j v[j] * active[j-i]
    # We want sum_j v[j] * active[i-j]. Let's use: ifft(FV * FA_conj) won't work directly.
    # sum_j v[j] * active[i-j] = (v * active)[i] (convolution) but truncated.
    # Actually, let's use: sum_j v[j] * active[i-j] for j=0..n-1
    # = ifft(conj(FV) * FA)[i] ... let me think.
    # ifft(FV * FA)[k] = sum_j v[j] * active[k-j] (circular, with zero-padding to N)
    # So ifft(FV * FA)[i] = sum_j v[j] * active[i-j] for i = 0..n-1 (with active having length 2n-1 < N)
    corr2 = np.fft.irfft(fv * fa, N)[:n]
    return (corr1 + corr2) / m + mu * v

def _run_lbfgs_v2(a_init, deadline, min_time_remaining=30.0):

    base_beta = max(gap * 0.5, max_c * 0.05)

def _shrinkage_refine(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.40, 0.50]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
