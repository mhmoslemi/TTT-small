def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _project_simplex(v):
    """Project v onto the simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _next_pow2(n):

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _active_subgrad(a, c, K):
    n = len(a)
    c_max = float(c.max())
    top_idx = np.argsort(c)[::-1][:K]
    w = np.zeros(2 * n - 1)
    for idx in top_idx:
        w[idx] = 1.0
    w_rev = w[::-1]
    conv_full = _fft_convolve(w_rev, a)
    h = conv_full[n - 1:2 * n - 1][::-1]
    return 2.0 * h / K


def _soft_subgrad(a, c, T):
    n = len(a)
    c_max = float(c.max())
    w = np.exp((c - c_max) / T)
    w_sum = float(w.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    w_rev = w[::-1]
    conv_full = _fft_convolve(w_rev, a)
    h = conv_full[n - 1:2 * n - 1][::-1]
    return 2.0 * h


def _adam_active(a_init, deadline, K_start=1, K_max=10,
                 lr_start=0.5, lr_end=0.05,
                 max_iter=500000, eval_interval=50,
                 momentum_start=0.9, momentum_end=0.95,
                 use_nesterov=True, patience_limit=20):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    K = K_start
    lr = lr_start
    momentum = momentum_start
    v = np.zeros(n)  # Nesterov velocity
    y = a.copy()  # Nesterov lookahead
    t = 0
    patience = 0
    best_patience = best_score

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac
        momentum = momentum_start + (momentumption_end - momentum_start) * frac

        # Compute subgradient from active set
        g = _active_subgrad(y, c, K)
        if not np.all(np.isfinite(g)):
            g = _soft_subgrad(y, c, float(c.max()) * 0.01)

        # Nesterov step
        if use_nesterov:
            y_new = a + momentum * (a - y)
        else:
            y_new = a.copy()

        # Compute subgradient at y_new
        c_y = _fft_convolve(y_new, y_new)
        g_y = _active_subgrad(y_new, c_y, K)
        if not np.all(np.isfinite(g_y)):
            g_y = g

        # Projected gradient step with line search
        lr_eff = lr
        a_cand = None
        for ls in range(8):
            a_raw = y_new - lr_eff * g_y
            a_cand = _project_simplex(a_raw)
            sc_cand = _score_norm(a_cand)
            if sc_cand < _score_norm(y_new) * (1.0 - 1e-12) or ls == 7:
                break
            lr_eff *= 0.5

        a = a_cand
        if not np.all(np.isfinite(a)) or a.sum() < 1e-15:
            a = best_a.copy()
            continue

        # Update best
        if t % eval_interval == 0 or ls < 3:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

        # Refresh active set periodically
        if t % 200 == 0:
            c = _fft_convolve(a, a)

        # Adaptive K: increase if stuck
        if patience > patience_limit:
            if K < K_max:
                K = min(K * 2, K_max)
            patience = 0
            lr = min(lr * 1.5, lr_start)

        # Reset Nesterov momentum on large improvements
        if best_score < best_patience * 0.999:
            best_patience = best_score
            v = np.zeros(n)
            y = a.copy()

    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=100):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        # Focus on top convolution entries
        top_k = np.argsort(c)[::-1][:5]
        improved = False

        for k in top_k:
            # Determine which a indices affect c[k]
            k0 = max(0, k - n + 1)
            k1 = min(k, n - 1)
            for j in range(k0, k1 + 1):
                partner = k - j
                if partner < 0 or partner >= n:
                    continue
                # Try scaling a[j] down
                for scale in [0.9, 0.8, 0.7, 1.1, 1.2]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _nesterov_simplex(a_init, deadline, max_iter=200000, lr_base=0.1,
                      K=1, eval_interval=100):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    y = a.copy()
    t_k = 1.0
    t = 0
    c = _fft_convolve(y, y)

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        c = _fft_convolve(y, y)
        g = _active_subgrad(y, c, K)
        if not np.all(np.isfinite(g)):
            break

        # Step
        step_size = lr_base / (t ** 0.5)
        a_new = y - step_size * g
        a_new = _project_simplex(a_new)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new
        y = a_new + beta * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new

        if t % eval_interval == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _gaussian_init(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _triangle_init(n):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.maximum(0, 1.0 - np.abs(x - center) / center)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)

    # Load parent

            a_init = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a_init = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a_init)
    best_a = a_init.copy()
    best_score = _score(a_init)
    print(f"Initial: n={n_parent}, score={best_score:.10f}")

    # Phase 1: Nesterov momentum active-set (0 - 600s)
    phase1_end = min(deadline, t_start + 600.0)
    print(f"Phase 1: Adam active-set, n={n_parent}, deadline={phase1_end - t_start:.0f}s")
    a_opt, sc = _adam_active(a_init, phase1_end,
                              K_start=1, K_max=8,

                              momentum_start=0.85, momentum_end=0.95,
                              use_nesterov=True, patience_limit=30)
    full_sc = _score(a_opt)
    print(f"Phase 1 done: score={full_sc:.10f}, n={len(a_opt)}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 2: Nesterov simplex (focused, K=1) (600s - 900s)
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end:

        a_nes, sc_nes = _nesterov_simplex(best_a, phase2_end,

            best_a = a_nes.copy()

    # Phase 3: CD polish (900s - 1050s)
    phase3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: CD polish")
        a_cd, sc_cd = _cd_polish(best_a, phase3_end, max_rounds=50)
        full_sc = _score(a_cd)
        print(f"Phase 3 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # Phase 4: Multi-K Adam from best (1050s - 1300s)
    phase4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase4_end:
        n_best = len(best_a)
        time_per = (phase4_end - time.monotonic()) / 3

            full_sc = _score(a_m)
            print(f"  K={K_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:

                                                max_iter=50000, eval_interval=30,
                                                momentum_start=0.9, momentum_end=0.95,
                                                use_nesterov=True, patience_limit=15)

        if np.isfinite(full_sc) and full_sc < best_score:

            best_a = a_fin.copy()

        # Quick CD
        if time.monotonic() < deadline - 10:

            full_sc2 = _score(a_cd2)
            print(f"Phase 7 CD: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
