def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_b = _convolve_grad(w, a, n)
    # dL/da = 2n * grad_b / S^2, but S = sum(a) = 1 for softmax
    # In z-space: dL/dz[j] = a[j] * (dL/da @ a) - a[j]^2 * sum(dL/da)
    # But since S=1 is maintained by softmax, we can simplify
    # Actually for the softmax parameterization with the constraint that sum(a)=1 is automatic:
    # dL/dz = a * (dL/da - mean(dL/da) * ... )
    # Simpler: dL/dz[j] = a[j] * (sum_j' dL/da[j'] - sum_j' dL/da[j'] * a[j]... 
    # No, let me just do it directly:
    # a[j] = exp(z[j]) / sum(exp(z))
    # da[j]/dz[k] = a[j] * (delta[j,k] - a[k])
    # dL/dz[k] = sum_j dL/da[j] * a[j] * (delta[j,k] - a[k])
    #           = a[k] * sum_j dL/da[j]*a[j] - a[k] * sum_j dL/da[j] * a[j]
    # Wait that's wrong. Let me redo:
    # dL/dz[k] = sum_j dL/da[j] * da[j]/dz[k]
    #          = sum_j dL/da[j] * a[j] * (delta[j,k] - a[k])
    #          = a[k]*dL/da[k] - a[k]*sum_j dL/da[j]*a[j]
    # Hmm, that's also not right. delta[j,k] means j=k:
    #          = dL/da[k]*a[k] - a[k]*sum_j dL/da[j]*a[j]
    # Wait no: sum_j dL/da[j] * a[j] * delta[j,k] = dL/da[k] * a[k]
    # sum_j dL/da[j] * a[j] * a[k] = a[k] * sum_j dL/da[j]*a[j]
    # So dL/dz[k] = a[k] * dL/da[k] - a[k] * (dL/da . a)
    #             = a[k] * (dL/da[k] - (dL/da . a))
    
    dL_da = 2.0 * n * grad_b  # since S=1
    dot = float(np.dot(dL_da, a))
    return a * (dL_da - dot)


class _AdaptiveLbfgs:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_adaptive(a_init, deadline, min_remaining=60.0, max_beta_mult=2.0, min_beta_mult=0.005):
    """L-BFGS with adaptive beta: starts with estimated good beta, decreases adaptively."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Estimate initial beta from peak spread
    c_sorted = np.sort(c_cur)[::-1]
    n_peaks = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[n_peaks - 1]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.002)

    # Adaptive beta sequence: start a bit larger, decrease more aggressively
    initial_beta = base_beta * max_beta_mult
    beta = initial_beta
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    no_improve_count = 0
    min_beta = base_beta * min_beta_mult

    while time.monotonic() < deadline - min_remaining:
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / (est_iter_time * 15))))

        prob = _AdaptiveLbfgs(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 30,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                no_improve_count = 0
                # Decrease beta more aggressively when making progress
                beta = max(min_beta, beta * 0.4)
            else:
                no_improve_count += 1
                # If no improvement, try a slightly different beta
                if no_improve_count < 3:
                    beta = max(min_beta, beta * 0.3)
                else:
                    # Reset and try larger beta
                    beta = base_beta * 0.1
                    no_improve_count = 0
                    if beta < min_beta:
                        break
        except Exception:
            beta = max(min_beta, beta * 0.5)

        if beta < min_beta * 2 and no_improve_count > 5:
            break

    return current_a


def _weighted_active_gd(a_init, deadline, top_k_init=80, top_k_min=3):
    """Gradient descent with exponentially weighted top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    top_k = top_k_init
    lr = 0.3
    no_improve = 0
    stall_limit = 200

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= stall_limit:
            # Reduce top_k to focus more
            top_k = max(top_k_min, top_k // 2)
            lr = 0.5
            no_improve = 0
            if top_k == top_k_min and no_improve > 0:
                break

        # Build weighted active set
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:top_k]
        active_vals = c[active_idx]
        
        # Exponential decay weights
        val_range = active_vals[0] - active_vals[-1]
        if val_range < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            decay = (active_vals[0] - active_vals) / val_range
            weights = np.exp(3.0 * decay)
            weights = weights / weights.sum()

        w = np.zeros_like(c)
        w[active_idx] = weights

        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * n * h  # d(2n*max(b))/da (approximate)
        
        # Center the gradient (optimization on simplex)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-16:
            break

        # Line search
        alpha = lr / (gnorm + 1e-15)
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc * (1 - 1e-12):
                a = a_try
                improved = True
                break
            alpha *= 0.4

        if not improved:
            # Try a smaller top_k
            top_k = max(top_k_min, int(top_k * 0.7))
            if no_improve > 30:
                lr *= 0.5
            if lr < 0.001:
                break

        # Slowly reduce top_k
        top_k = max(top_k_min, int(top_k * 0.995))
        lr = min(0.5, lr * 1.01)

    return best_a, _score(best_a)


def _local_exchange(a_init, deadline, n_trials=2000):
    """Try pairwise mass transfer moves to improve the score."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(12345)

    # Precompute initial convolution
    c = _full_convolve(best_a, best_a)

    # Sort indices by value (larger values first for exchange targets)
    n_trials_done = 0
    try_count = 0

    while time.monotonic() < deadline and n_trials_done < n_trials:
        # Pick a source (larger value) and target (smaller value)
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if best_a[i] <= best_a[j]:
            i, j = j, i
        if best_a[i] < 1e-15:
            continue

        # Try transferring a small fraction
        for frac in [0.01, 0.005, 0.001, 0.05]:
            delta = frac * best_a[i]
            a_try = best_a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0:
                continue
            s_try = a_try.sum()
            a_try = a_try / s_try
            
            # Quick score check
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            
            if sc_try < best_sc - 1e-12:
                best_a = a_try
                best_sc = sc_try
                c = c_try
                break
            else:
                break  # If smallest transfer didn't help, larger won't either
        
        n_trials_done += 1
        try_count += 1

    return best_a, best_sc


def _shrinkage_reopt(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    a_sorted = np.sort(a)
    # Try different fractions to zero out
    for frac in [0.02, 0.05, 0.10, 0.15, 0.20]:
        if time.monotonic() > deadline:
            break
        n_zero = int(n * frac)
        threshold = a_sorted[n_zero]
        a_try = a.copy()
        a_try[a_try <= threshold] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        if sc_opt_n < best_score:
            best_seq = a_opt_n.copy()
            best_score = sc_opt_n
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  n={n_try}: best={sc_opt_n:.10f} (no improvement)")

    # === Phase 6: Local exchange moves ===
    print(f"\n[P6] Local exchange (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = deadline - 60.0
    if time.monotonic() < p6_deadline:
        a_ex, sc_ex = _local_exchange(best_seq, p6_deadline, n_trials=5000)
        if sc_ex < best_score:
            best_seq = a_ex.copy()
            best_score = sc_ex
            print(f"  NEW BEST: {best_score:.10f} [exchange] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final fine polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _weighted_active_gd(best_seq, deadline - 3, top_k_init=20, top_k_min=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = float(final.sum())
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Verify with evaluate_sequence if available
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"Warning: could not verify with evaluate_sequence: {e}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
