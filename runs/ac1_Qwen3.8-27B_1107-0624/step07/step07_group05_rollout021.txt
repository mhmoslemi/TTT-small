"""
Active-set gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. Efficient O(n log n) gradient via FFT-based correlation
2. Adaptive active-set identification with smooth transitions
3. Multi-phase: CG descent -> L-BFGS smoothing -> coordinate polish -> n exploration
4. Barzilai-Borwein adaptive step sizes for subgradient phase
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    """Smoothed objective in log-space: 2*n*beta*log(sum exp(c/beta))."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    """Gradient of smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # h[j] = sum_k w[k] * a[k-j] for valid k-j
    # This is cross-correlation of w and a
    N = _next_pow2(2 * n)
    wa = np.fft.rfft(a, N)
    ww = np.fft.rfft(w, N)
    h_full = np.fft.irfft(wa * np.conj(ww), N)
    # h_full[j] = sum_m a[m] * w[m+j] (correlation)
    # We need h[j] = sum_k w[k]*a[k-j] = sum_m a[m]*w[m+j]
    # h_full has length N, we want elements [0, n-1]
    h = h_full[:n]
    # Also need to handle the wrap-around: for k-j valid means k in [j, j+n-1]
    # The correlation gives us h[j] = sum_{m=0}^{n-1} a[m]*w[m+j]
    # which is exactly what we want (w is defined on [0, 2n-2], a on [0, n-1])
    # Gradient of objective w.r.t. a: 4*n*h
    # Gradient w.r.t. z (through softmax): 4*n*(a*h - a*sum(a*h))
    W = float(np.dot(a, h))
    grad_a = 4.0 * n * (a * h - a * W)
    return grad_a


def _active_set_grad(a, active_lags, n):
    """Gradient of max over active lags w.r.t. a (for direct a-optimization)."""
    c = _full_convolve(a, a)
    # Get weights from active lags (softmax over active values)
    vals = c[active_lags]
    v_max = vals.max()
    w_active = np.exp(vals - v_max)
    w_active = w_active / w_active.sum()
    
    # h[j] = sum_{k in active} w_k * a[k-j] (valid)
    # Compute via direct sum for small active set
    h = np.zeros(n)
    for wi, k in zip(w_active, active_lags):
        # a[k-j] valid when 0 <= k-j < n, i.e., j in [max(0,k-n+1), min(k, n-1)]
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            h[j_min:j_max+1] += wi * a[k - np.arange(j_min, j_max + 1)]
    return 4.0 * n * h, c


def _cg_descent(a_init, deadline, n_obj_evals=50000, ls_method='backtrack'):
    """Conjugate gradient on log-space with non-monotone line search."""
    n = len(a_init)

                if g_norm_sq < 1e-30:
                    break

    n = len(a_init)
    if betas is None:

        c_sorted = np.sort(c)[::-1]

        base = max(gap * 0.5, c_sorted[0] * 0.05)

    z = z - z.mean()

def _pressure_descent(a_init, deadline, n_iters=3000, gamma=0.3):
    """Multiplicative pressure equalization with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        # Adaptive active set
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
        
        # Compute pressure via FFT correlation
        N = _next_pow2(2 * n)
        wa = np.fft.rfft(a, N)
        wact = np.fft.rfft(active.astype(float), N)
        p_full = np.fft.irfft(wa * np.conj(wact), N)
        p = p_full[:n]
        p = np.maximum(p, 1e-30)
        
        # Multiplicative update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

def _subgrad_bb(a_init, deadline, n_steps=10000, lr_init=0.05):
    """Subgradient descent with Barzilai-Borwein step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    y_prev = None
    f_prev = None
    g_prev = None

            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.02 * c_max

        if len(active_lags) < 3:
            active_lags = np.argsort(c)[-5:]
        
        # Compute gradient
        grad, _ = _active_set_grad(a, active_lags, n)
        
        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = np.dot(s_prev, y_prev)
            if sy > 1e-16:
                ss = np.dot(s_prev, s_prev)
                yy = np.dot(y_prev, y_prev)
                alpha = ss / sy
                beta = sy / yy
                lr = 0.5 * (alpha + beta)
                lr = np.clip(lr, lr_init * 0.01, lr_init * 10)
            else:
                lr = lr_init / (1 + 0.001 * it)
        else:
            lr = lr_init / (1 + 0.001 * it)
        
        z_new = z - lr * grad
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()
        
        a_new = _softmax(z_new)
        f_new = _score_norm(a_new)
        
        y = grad - g_prev if g_prev is not None else grad
        s = z_new - z
        
        z = z_new
        s_prev = s
        y_prev = y
        g_prev = grad
        f_prev = f_new
        
        if it % 200 == 0:
            sc = _score(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
    
    sc_final = _score(_softmax(z))
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = _softmax(z).copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 7 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.3
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=8, amplitude=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    max_freq = min(n // 2, len(z_hat) - 1)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, max_freq))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq % len(z_hat)] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

                break
            a_pe, _ = _pressure_descent(best_seq, time.monotonic() + 80, n_iters=3000, gamma=gamma)
            sc_pe = _score(a_pe)
            _update(a_pe, sc_pe)
            if sc_pe < best_score:
                # Follow up with L-BFGS
                a_f = _lbfgs_smooth(a_pe, min(phase1_deadline - 5, time.monotonic() + 60), min_remaining=5)
                _update(a_f, _score(a_f))
    
    # === PHASE 2: Try different lengths (400s) ===
    print(f"\n{'='*50}")
    print(f"PHASE 2: Length exploration (t={time.monotonic()-t_start:.1f}s, best={best_score:.8f})")
    print(f"{'='*50}")
    
    phase2_deadline = t_start + 1100
    n_candidates = [6144, 7168, 9216, 10240, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort()  # smaller first (faster)
    
    for n_try in n_candidates:
        if time.monotonic() > phase2_deadline - 20:
            break
        rem_for_this = phase2_deadline - 20 - time.monotonic()
        if rem_for_this < 40:
            break
        
        print(f"  Trying n={n_try} (budget={rem_for_this:.0f}s)")
        
        # Initialize from best by resampling
        if best_seq is not None:
            a_init = _resample_sequence(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")
        
        # Quick optimization (limited time)
        min_rem = max(10, rem_for_this * 0.2)
        a_opt, sc_opt = _full_optimize(a_init, min(phase2_deadline - 10, time.monotonic() + rem_for_this * 0.8),
                                        min_remaining=min_rem, label=f"n{n_try}")
        _update(a_opt, sc_opt)
    
    # === PHASE 3: Diverse restarts (250s) ===
    print(f"\n{'='*50}")
    print(f"PHASE 3: Diverse restarts (t={time.monotonic()-t_start:.1f}s, best={best_score:.8f})")
    print(f"{'='*50}")
    
    phase3_deadline = t_start + 1350
    if best_seq is not None:
        n_best = len(best_seq)
        profiles = ["chirp", "asym_taper", "tapered"]
        
        for i, prof in enumerate(profiles):
            if time.monotonic() > phase3_deadline - 20:
                break
            rem = phase3_deadline - 20 - time.monotonic()
            if rem < 30:
                break
            
            # Create blend
            a_prof = _make_profile(n_best, prof)
            blend_weight = 0.3 + 0.2 * i
            a_blend = (1 - blend_weight) * best_seq + blend_weight * a_prof
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            
            # Fourier perturb
            a_pert = _fourier_perturb(a_blend, rng, n_modes=6, amplitude=0.015)
            
            a_r, sc_r = _full_optimize(a_pert, min(phase3_deadline - 10, time.monotonic() + rem * 0.7),
                                        min_remaining=10, label=f"restart{i}")
            _update(a_r, sc_r)
        
        # Also try random restarts
        for i in range(3):
            if time.monotonic() > phase3_deadline - 20:
                break
            rem = phase3_deadline - 20 - time.monotonic()
            if rem < 30:
                break
            
            z_rand = rng.standard_normal(n_best) * 0.5
            a_rand = _softmax(z_rand)
            # Blend with best
            a_blend = 0.7 * best_seq + 0.3 * a_rand
            a_blend = a_blend / a_blend.sum()
            
            a_r, sc_r = _full_optimize(a_blend, min(phase3_deadline - 10, time.monotonic() + rem * 0.6),
                                        min_remaining=10, label=f"rand{i}")
            _update(a_r, sc_r)
    
    # === PHASE 4: Final polish (250s) ===
    print(f"\n{'='*50}")
    print(f"PHASE 4: Final polish (t={time.monotonic()-t_start:.1f}s, best={best_score:.8f})")
    print(f"{'='*50}")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Long pressure equalization with very small gamma
        a_pe, _ = _pressure_descent(best_seq, min(deadline - 30, time.monotonic() + 100),
                                     n_iters=5000, gamma=0.05)
        _update(a_pe, _score(a_pe))
        
        # L-BFGS with very fine beta
        if time.monotonic() < deadline - 20:
            c_cur = _full_convolve(best_seq, best_seq)
            c_sorted = np.sort(c_cur)[::-1]
            gap = c_sorted[0] - c_sorted[min(50, len(c_sorted)-1)] if len(c_sorted) > 50 else c_sorted[0]
            base_beta = max(gap * 0.1, c_sorted[0] * 0.005)
            fine_betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015])
            a_fin = _lbfgs_smooth(best_seq, min(deadline - 10, time.monotonic() + 80),
                                    betas=fine_betas, min_remaining=5)
            _update(a_fin, _score(a_fin))
        
        # Subgradient with very small steps
        if time.monotonic() < deadline - 10:
            a_sg, sc_sg = _subgrad_bb(best_seq, deadline - 5, n_steps=8000, lr_init=0.01)
            _update(a_sg, sc_sg)
        
        # Coordinate polish
        if time.monotonic() < deadline - 5:
            a_cd, sc_cd = _coordinate_polish(best_seq, deadline - 3, n_rounds=150)
            _update(a_cd, sc_cd)
    
    # Final validation and return
    print(f"\n{'='*50}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
