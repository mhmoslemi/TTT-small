Let me think more carefully about the KKT solver.

The problem: minimize `max_k b_k` subject to `sum(a) = 1`, `a >= 0`, where `b = conv(a, a)`.

At a KKT point with active set S and dual weights w (sum w = 1, w >= 0):
- `2 * sum_{k in S} w_k * a[k-j] = lambda` for all j with a[j] > 0
- `2 * sum_{k in S} w_k * a[k-j] <= lambda` for all j with a[j] = 0
- `sum(a) = 1`, `a >= 0`

The key equation is: `g(j) := 2 * sum_{k in S} w_k * a[k-j] = lambda` for j in support.

Let me denote `S = {s_1, s_2, ..., s_m}` (m = |S|).

`g(j) = 2 * sum_{l=1}^{m} w_{s_l} * a[s_l - j]`

For this to equal lambda for all j in support, we need the "smoothed" version of a (smoothed by the kernel `2*sum w_k * delta(k)`) to be constant on the support.

If support = {0, ..., n-1} (all positions), then we need:
`sum_{l=1}^{m} w_{s_l} * a[s_l - j] = lambda/2` for all j = 0, ..., n-1.

This is a linear system. Let me write it as `K * a = c * 1` where:
- `K[j, m] = 2 * w_{j+m}` if `j+m in S`, else 0
- `c = lambda/2`

And `sum(a) = 1`.

The total system (n+1 equations, n+1 unknowns: a_0, ..., a_{n-1}, lambda):
`K * a - lambda * 1 = 0`
`1^T * a = 1`
`a >= 0`

Since we have n+1 equations and n+1 unknowns, if the system is non-singular, there's a unique solution. We need to check that a >= 0.

If the solution has a[j] < 0 for some j, we set a[j] = 0 and remove j from the support, then re-solve (with n' < n equations and unknowns).

This is essentially the active-set method for LPs!

The matrix K is sparse: each row has at most |S| non-zeros. For |S| = 20 and n = 8192, each row has at most 20 non-zeros, so the matrix has at most 20*8192 ≈ 164K non-zeros. This is very sparse and can be solved quickly with sparse linear solvers.

    return k

Active-set KKT optimizer for autocorrelation minimization.
Combines direct sparse linear solves for KKT conditions with
L-BFGS polishing and multi-length exploration.
"""
import numpy as np
import time
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import spsolve
from scipy.optimize import minimize

def _fft_len(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _conv(a, b):
    out_n = len(a) + len(b) - 1
    N = _fft_len(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]

def _conv_sq(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _fft_len(out_n)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    b = _conv_sq(a)
    return float(2.0 * n * b.max() / (s * s))

def _kkt_solve(S_list, n, support=None, max_iter=50):
    """
    Solve KKT system for active set S_list.
    Returns (a, lambda, feasible) where a sums to 1.
    """
    S_set = set(S_list)
    m = len(S_list)
    w_val = 1.0 / m
    
    if support is None:
        support = set(range(n))

        support_list = sorted(support)
        n_sup = len(support_list)
        if n_sup == 0:
            return None, None, False
        if n_sup == 1:
            a = np.zeros(n)
            a[support_list[0]] = 1.0
            return a, None, True
        
        idx_map = {v: i for i, v in enumerate(support_list)}
        
        # Build K matrix (sparse)
        rows = []
        cols = []
        data = []
        
        for j_idx, j in enumerate(support_list):

                if m_val in idx_map:
                    rows.append(j_idx)
                    cols.append(idx_map[m_val])
                    data.append(2.0 * w_val)

"""
KKT active-set optimizer with L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses direct sparse linear system solves at KKT points,
adaptive active set management, and multi-length exploration.

import numpy as np
import time
from scipy.sparse import lil_matrix, csr_matrix
from scipy.sparse.linalg import spsolve
from scipy.optimize import minimize


def _fft_len(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv_sq(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _fft_len(out_n)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:out_n]


def _conv(a, b):
    out_n = len(a) + len(b) - 1
    N = _fft_len(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    b = _conv_sq(a)
    return float(2.0 * n * b.max() / (s * s))


def _score_from_b(a, b):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * b.max() / (s * s))


def _kkt_solve(S_list, n, support=None):
    """
    Solve KKT linear system for given active set.
    Returns (a, lambda, feasible, neg_indices).
    """
    S_set = set(S_list)
    m = len(S_list)
    if m == 0:
        return None, None, False, set()
    w_val = 1.0 / m

    if support is None:
        support = set(range(n))
    
    MAX_ITER = 100
    for _ in range(MAX_ITER):
        support_list = sorted(support)
        n_sup = len(support_list)
        if n_sup == 0:
            return None, None, False, set()
        if n_sup == 1:
            a = np.zeros(n)
            a[support_list[0]] = 1.0
            return a, None, True, set()

        idx_map = {v: i for i, v in enumerate(support_list)}

        # Build K matrix
        rows = []
        cols = []
        data = []
        for j_idx, j in enumerate(support_list):
            for s in S_list:
                m_val = s - j
                if m_val in idx_map:
                    rows.append(j_idx)
                    cols.append(idx_map[m_val])
                    data.append(2.0 * w_val)

        if len(rows) == 0:
            return None, None, False, set()

        K = lil_matrix((n_sup, n_sup))
        for r, c, v in zip(rows, cols, data):
            K[r, c] = v
        K_csr = K.tocsr()

        # Augmented system: [K | -1; 1^T | 0] [a; lam] = [0; 1]
        A = lil_matrix((n_sup + 1, n_sup + 1))
        A[:n_sup, :n_sup] = K_csr
        A[:n_sup, n_sup] = -1.0
        for i in range(n_sup):
            A[n_sup, i] = 1.0
        A_csr = A.tocsr()

        b_vec = np.zeros(n_sup + 1)
        b_vec[n_sup] = 1.0

        try:
            sol = spsolve(A_csr, b_vec)
        except Exception:
            return None, None, False, set()

        if not np.all(np.isfinite(sol)):
            return None, None, False, set()

        a_vals = sol[:n_sup]
        lam = sol[n_sup]

        neg_mask = a_vals < -1e-10
        if neg_mask.any():
            neg_idx = set()
            for i in range(n_sup):
                if neg_mask[i]:
                    neg_idx.add(support_list[i])
                    support.discard(support_list[i])
            continue

        a_vals = np.maximum(a_vals, 0.0)
        a_full = np.zeros(n)
        a_full[support_list] = a_vals
        s = a_full.sum()
        if s < 1e-15:
            return None, None, False, set()
        a_full /= s

        return a_full, lam, True, set()

    return None, None, False, set()


def _kkt_polish(a_init, deadline):
    """Polish using KKT active-set method with multiple active set sizes."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a /= s

    best_a = a.copy()
    best_sc = _score(a)

    K_candidates = [5, 10, 15, 20, 30, 50, 80, 120, 200]

    for K in K_candidates:
        if time.monotonic() > deadline:
            break

        b = _conv_sq(a)
        top_idx = np.argsort(b)[-K:]
        S_list = sorted(set(top_idx.tolist()))

        a_new, lam, feasible, _ = _kkt_solve(S_list, n)

        if feasible and a_new is not None:
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new.copy()

            # Try expanding S
            if len(S_list) < 300:
                b_new = _conv_sq(a_new)
                max_S = b_new[S_list].max()
                viol = np.where(b_new > max_S + 1e-12)[0]
                if len(viol) > 0:
                    S_exp = sorted(set(S_list) | set(viol.tolist()))
                    if len(S_exp) <= 500:
                        a_exp, lam_exp, feas_exp, _ = _kkt_solve(S_exp, n)
                        if feas_exp and a_exp is not None:
                            sc_exp = _score(a_exp)
                            if sc_exp < best_sc:
                                best_sc = sc_exp
                                best_a = a_exp.copy()
                                a = a_exp.copy()

    return best_a, best_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _conv_sq(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _conv_sq(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=20.0):
    n = len(a_init)
    c_cur = _conv_sq(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005])

    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            sc_cur = _score(current_a)
            if sc_new < sc_cur:
                current_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):

        tol = 0.005 * c_max

            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad = 4.0 * n * a * (h - W)

        if grad_norm < 1e-15:

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)

                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 3)))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if n - freq < len(z_hat):
                z_hat[n - freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)
