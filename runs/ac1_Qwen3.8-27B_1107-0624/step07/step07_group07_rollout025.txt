The key steps in each iteration:

2. P → modify → P_new

The modification in step 2 is crucial. Let me think about what modification to use.

- This preserves the mean while compressing the dynamic range
- For α < 1, peaks are reduced and valleys are raised

Option C: P_new = P / (1 + P/c) for some c > 0

- For P >> c: P_new ≈ c (capped)

I think Option B is the most natural. Let me use it.

"""
Step sequence optimizer using iterative flat-spectrum design (IFT) combined
with gradient-based refinement. Key innovation: work in the frequency domain to
directly flatten the power spectrum (equivalent to flattening the self-convolution),
then project back to the non-negative cone.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _ift_iteration(a, N, alpha, phase=None):
    """Single IFT iteration: flatten the power spectrum."""
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    
    # Modify P to be flatter
    P_mean = np.mean(P)
    if P_mean < 1e-30:
        return a.copy()
    P_new = P_mean * (P / P_mean) ** alpha
    
    # Recover with phase
    if phase is None:
        phase = np.angle(A)
    magnitude_new = np.sqrt(P_new)
    A_new = magnitude_new * np.exp(1j * phase)
    
    # IFFT
    a_new_padded = np.fft.irfft(A_new, N)
    a_new = a_new_padded[:len(a)]
    
    # Project to non-negative
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new

def _ift_optimize(a_init, deadline, alphas=None, max_iters=5000):
    """Run IFT optimization with a schedule of alpha values."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    stall = 0
    stall_limit = 200
    
    if alphas is None:
        alphas = [0.95, 0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6, 0.55, 0.5,
                  0.45, 0.4, 0.35, 0.3]
    
    for alpha in alphas:
        if time.monotonic() > deadline:
            break
        for i in range(max_iters):
            if time.monotonic() > deadline:
                break
            a_new = _ift_iteration(a, N, alpha)
            sc = _score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                a = a_new
                stall += 1
                if stall > stall_limit:
                    break
    
    return best_a, best_score

def _subgradient_refine(a_init, deadline, max_iters=10000):
    """Subgradient method in the original space to reduce peaks."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    lr = 0.01
    stall = 0
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = c.max()
        
        # Find active set (peaks)
        active = c >= c_max * 0.999
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient of max(c) w.r.t. a
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[j] = a[k-j] + a[j-k] (for valid indices)
        # Using FFT: if c = conv(a,a), then dc/da[j] = (a * w_reversed)[n-1:2n-1] type thing
        # More precisely: dc/da[j] = sum_k w[k] * (dc[k]/da[j]) = sum_k w[k] * (a[k-j] + a[j-k])
        # This is conv(w, a) and conv(w_rev, a) evaluated at appropriate offsets
        
        # Using the identity: if c = a * a (convolution), then
        # d(c*p)/da = p * a + a * p (where * is convolution)
        # restricted to the first n elements
        w_pad = np.zeros(N)
        w_pad[:len(w)] = w
        w_fft = np.fft.rfft(w_pad, N)
        a_fft = np.fft.rfft(a, N)
        grad_fft = 2 * a_fft * w_fft
        grad = np.fft.irfft(grad_fft, N)[:n]
        
        # Normalize gradient
        grad = grad - np.mean(grad)  # remove component along all-ones (preserves sum)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = lr
        a_new = a - step * grad / grad_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            if lr < 1e-8:
                break
            continue
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.1)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if lr < 1e-8 or stall > 200:
                break
    
    return best_a, best_score

def _peak_shave(a_init, deadline, max_iters=20000):
    """Directly shave peaks in the convolution by redistributing mass."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        # Find which elements of a contribute to the peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # Reduce the contributing elements and distribute their mass elsewhere
        
        # Find the peak contributors
        j_range = max(0, min(peak_idx, n-1))
        j_start = max(0, peak_idx - (n-1))
        j_end = min(n-1, peak_idx)
        
        if j_start > j_end:
            break
        
        # Get the contribution of each a[j] to c[peak_idx]
        partners = peak_idx - np.arange(j_start, j_end+1)
        partners_valid = (partners >= 0) & (partners < n)
        
        # Try reducing the element with the largest contribution
        contributions = np.zeros(n)
        for j in range(j_start, j_end+1):
            p = peak_idx - j
            if 0 <= p < n:
                contributions[j] = a[j] * a[p]
        
        # Find the element to reduce
        max_j = int(np.argmax(contributions))
        if contributions[max_j] < 1e-15:
            break
        
        # Reduce a[max_j] and distribute to a random other element
        reduce_frac = 0.1
        mass = a[max_j] * reduce_frac
        a[max_j] -= mass
        
        # Distribute to a random position
        target_j = int(np.random.randint(0, n))
        if target_j == max_j:
            target_j = (target_j + 1) % n
        a[target_j] += mass
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()
    
    return best_a, best_score
