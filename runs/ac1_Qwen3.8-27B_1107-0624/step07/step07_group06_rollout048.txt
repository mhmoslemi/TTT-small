"""
Asymmetric step function optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing, L-BFGS, active-set Newton, and boundary exploration.
Key insight: optimal sequences are asymmetric (non-palindromic), which allows
the center convolution value to be below (sum/n)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform", **kwargs):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.arange(n) / max(1, n - 1)
        alpha = kwargs.get('alpha', 2.0)
        a = 1.0 + alpha * t
        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.arange(n) / max(1, n - 1)
        alpha = kwargs.get('alpha', 3.0)
        a = np.exp(alpha * t)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_sine":
        t = np.linspace(0, 1, n)
        p = kwargs.get('p', 1.5)
        a = np.sin(np.pi * t) ** p
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_shift":
        t = np.linspace(0, 1, n)
        shift = kwargs.get('shift', 0.3)
        a = 1.0 + shift * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "piecewise":
        # Front-loaded with smooth transition
        t = np.linspace(0, 1, n)
        transition = kwargs.get('transition', 0.3)
        a = np.where(t < transition, 
                     1.0 - transition + (1-t)/transition * (transition - 0.5),
                     0.5 * (1 + np.cos(np.pi * (t - transition) / (1 - transition))))
        # Simpler: just a power law
        a = (1 - 0.7*t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.arange(n) / max(1, n - 1)
        p = kwargs.get('p', 1.0)
        a = (1 - t + 1e-3) ** p
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_run(a_init, deadline, min_remaining=30.0, beta_schedule=None):
    """Run L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    if beta_schedule is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
        base_beta = max(gap, max_c * 0.01)
        beta_schedule = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                              0.005, 0.002, 0.001, 0.0005, 0.0001, 0.00005, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 5))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-10 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p = r + beta_cg * p
            rs = r_new
        
        dx = x - np.mean(x)
        
        # Line search
        f_old = float(c[active].sum())
        a_new = a.copy()
        alpha = 1.0
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
        
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _boundary_explore(a_init, deadline):
    """Try zeroing out small elements to explore the boundary."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    idx = np.argsort(a_init)
    
    for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a_init.copy()
        a_try[idx[:n_zero]] = 0.0
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
    
    return best_a, best_sc


def _fourier_perturb(a, rng, n_modes=5, amp_base=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = amp_base * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _asym_perturb(a, rng, strength=0.1):
    """Add asymmetric perturbation (breaks palindromic symmetry)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    # Add a ramp component
    t = np.arange(n) / n
    z = z + strength * rng.uniform(-1, 1) * (2*t - 1)
    # Add quadratic asymmetry
    z = z + strength * 0.5 * rng.uniform(-1, 1) * (4*t*t - 4*t + 1)
    z = z - z.mean()
    return _softmax(z)
