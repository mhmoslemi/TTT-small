"""
Improved optimizer using adaptive-temperature smoothed gradients,
peak-targeting coordinate descent, and multi-start with diverse
initializations. Builds on previous best with focused refinements.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, c, T, top_k=5):
    """Compute smoothed gradient using softmax over top-k peaks."""
    n = len(a)
    len_c = len(c)
    
    # Find top-k positions
    if top_k >= len_c:
        top_idx = np.argsort(c)[::-1]
    else:
        # Use partition for efficiency
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    
    # Compute softmax weights over top-k
    c_top = c[top_idx]
    c_max = c_top[0]
    weights = np.exp((c_top - c_max) / T)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        p = np.zeros(len_c)
        p[top_idx[0]] = 1.0
    else:
        p = np.zeros(len_c)
        p[top_idx] = weights / w_sum
    
    # Gradient: 4n * conv(a, p_reversed)[n-1 : 2n-1]
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n-1 : 2*n-1][::-1]
    return g

def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, top_k=3, eval_interval=200,

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience_score = best_score
    lr_mult = 1.0
    adapt_top_k = True
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Adaptive top-k

            # Count peaks within 1% of max

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast single-coordinate descent with O(n) updates, adaptive-temperature
Adam in log-space, diverse multi-start, and intensive final polishing.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, T, top_k=5):
    n = len(a)
    len_c = len(c)
    if top_k >= len_c:
        top_idx = np.argsort(c)[::-1]
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    c_top = c[top_idx]
    c_max = c_top[0]
    weights = np.exp((c_top - c_max) / T)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        p = np.zeros(len_c)
        p[top_idx[0]] = 1.0
    else:
        p = np.zeros(len_c)
        p[top_idx] = weights / w_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,

                   clip_norm=0.8, use_backtrack=True, adapt_k=True):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if adapt_k:
            c_max_val = c.max()
            close = int(np.sum(c > 0.995 * c_max_val))
            k = min(max(2, close), 8)

        if update_norm > clip_norm:

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast single-coordinate descent with O(n) updates, adaptive-temperature
Adam in log-space, diverse multi-start, and intensive final polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, T, top_k=5):
    n = len(a)
    len_c = len(c)
    if top_k >= len_c:
        top_idx = np.argsort(c)[::-1]
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    c_top = c[top_idx]
    c_max = c_top[0]
    weights = np.exp((c_top - c_max) / T)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        p = np.zeros(len_c)
        p[top_idx[0]] = 1.0
    else:
        p = np.zeros(len_c)
        p[top_idx] = weights / w_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

                   max_iter=500000, top_k=3, eval_interval=200,
                   clip_norm=0.8, use_backtrack=True, adapt_k=True):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if adapt_k:
            c_max_val = c.max()
            close = int(np.sum(c > 0.995 * c_max_val))
            k = min(max(2, close), 8)
        else:
            k = top_k

        g = _compute_gradient(a, c, T, top_k=k)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 100 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)

                z_new = z

                m_buf = np.zeros(n)

        best_a = a_final.copy()

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0, endpoint=False)
    x1 = np.linspace(0, 1, n_new, endpoint=False)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _gaussian_init(n, sigma_frac=0.2):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def _cosine_init(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _bump_init(n, width_frac=0.5):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = width_frac * n
    a = np.exp(-((x - center) / width) ** 2)
    return a / a.sum()


def _stochastic_subgradient(a_init, deadline, max_iter=200000,
                            lr_start=0.01, lr_end=0.001, top_k=3):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    momentum = np.zeros(n)
    beta = 0.9
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    rng = np.random.default_rng(12345)

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        lr = lr_start * (lr_end / lr_start) ** (t / max_iter)
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if top_k < len(c):
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        else:
            top_idx = np.arange(len(c))

        count = min(top_k, len(top_idx))
        chosen = top_idx[rng.integers(0, count)]

        p = np.zeros(len(c))
        p[chosen] = 1.0
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        momentum = beta * momentum + (1.0 - beta) * z_grad
        z = z - lr * momentum
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % 500 == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score
