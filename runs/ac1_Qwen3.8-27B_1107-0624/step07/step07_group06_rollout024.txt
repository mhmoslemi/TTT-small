def _next_pow2(n):
    k = 1

    return max(2, k)


def _full_convolve(a, b):

    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)

    return r

        def hess_vec(v):
            return _hess_matvec(v, active_float, n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=300, tol=1e-12)
        dx = dx - np.mean(dx)

        # Line search with Armijo condition
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        success = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-10 * alpha * gnorm:
                a_new = a_try
                success = True
                break
            alpha *= 0.5
        if not success:
            if alpha < 1e-15:
                # Try a small step in negative gradient direction
                alpha = 1e-6
                a_try = a + alpha * (-grad_t / max(gnorm, 1e-30))
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        a = a_try
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _kkt_guided_step(a, deadline, n_steps=200):
    """
    Use KKT conditions to guide optimization.
    Solve LP to find optimal weights for active set, then take steps.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active_idx = np.where(c >= (c_max - tol))[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        m = len(active_idx)

        # Solve LP: minimize range of weighted sum
        # f_i = sum_k lambda_k * 2 * a[k-i] for valid k-i
        # Minimize max_i f_i - min_i f_i
        # = minimize (u - v) s.t. f_i <= u, f_i >= v, sum lambda = 1, lambda >= 0

        # For efficiency, only solve for first K active lags if m is large
        m_eff = min(m, 20)
        active_subset = active_idx[:m_eff]

        # Compute the matrix: M[i, k] = 2 * a[active_subset[k] - i] if valid, else 0
        # This is n x m_eff
        M = np.zeros((n, m_eff))
        for k_idx, k in enumerate(active_subset):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    M[i, k_idx] = 2.0 * a[j]

        # LP: min u - v s.t. M @ lambda <= u, M @ lambda >= v, sum(lambda)=1, lambda>=0
        lam = cp.Variable(m_eff, nonneg=True)
        u = cp.Variable()
        v = cp.Variable()
        constraints = [
            cp.sum(lam) == 1,
            M @ lam <= u,
            M @ lam >= v
        ]
        objective = cp.Minimize(u - v)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-8, reltol=1e-8)
            lam_val = lam.value
            if lam_val is None or not np.all(np.isfinite(lam_val)):
                lam_val = np.ones(m_eff) / m_eff
            lam_val = np.maximum(lam_val, 0)
            lam_sum = lam_val.sum()
            if lam_sum < 1e-10:
                lam_val = np.ones(m_eff) / m_eff
            else:
                lam_val = lam_val / lam_sum
        except Exception:
            lam_val = np.ones(m_eff) / m_eff

        # Compute weighted gradient
        g = np.zeros(n)
        for k_idx, k in enumerate(active_subset):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += lam_val[k_idx] * 2.0 * a[j]

        # Project to tangent space
        g_t = g - np.mean(g)
        gnorm = float(np.linalg.norm(g_t))
        if gnorm < 1e-14:
            break

        # Take step
        eta = 0.01 / (1.0 + 0.01 * step)
        a_try = a - eta * g_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try2 = a - eta * factor * g_t
                a_try2 = np.maximum(a_try2, 0)
                s_try2 = a_try2.sum()
                if s_try2 < 1e-15:
                    continue
                a_try2 = a_try2 / s_try2
                sc_try2 = _score(a_try2)
                if sc_try2 < best_sc - 1e-15:
                    best_sc = sc_try2
                    best_a = a_try2.copy()
                    a = a_try2
                    break

    return best_a, best_sc


def _peak_shaving(a, deadline, n_iters=100):
    """Focus on reducing the single worst autocorrelation lag."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Gradient of c_{k_star} w.r.t. a
        # d c_k / d a_i = 2 * a[k-i] for valid k-i
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]

        # We want to reduce c_{k_star}, so move a in direction -g
        # But project to maintain sum(a) = 1 and a >= 0
        g_t = g - np.mean(g)
        gnorm = float(np.linalg.norm(g_t))
        if gnorm < 1e-14:
            break

        # Try multiple step sizes
        found = False
        for eta in [0.001, 0.0005, 0.0001, 0.00005, 0.00001]:
            a_try = a - eta * g_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            # Check that we didn't increase the max too much
            if c_try.max() < c_max + 0.001 * c_max:
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-15:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    found = True
                    break
        if not found:
            # Try a random perturbation to escape
            rng_step = np.random.default_rng(it)
            a_try = a.copy()
            a_try += 0.001 * rng_step.standard_normal(n)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            else:
                break

    return best_a, best_sc


def _block_local_opt(a, deadline, n_blocks=50, block_size=30):
    """Local optimization within random windows."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for bi in range(n_blocks):
        if time.monotonic() > deadline:
            break
        # Choose a random block
        start = rng.integers(0, max(1, n - block_size))
        end = start + block_size
        indices = np.arange(start, end)

        # Fix all elements outside the block
        a_fixed = best_a.copy()
        a_block = a_fixed[indices].copy().astype(np.float64)
        a_rest = a_fixed.sum() - a_block.sum()

        # Optimize the block: minimize max autocorrelation
        # Use L-BFGS on the block variables
        def block_obj(x):
            a_try = a_fixed.copy()
            a_try[indices] = np.maximum(x, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                return 1e10
            a_try = a_try / s_try
            c = _full_convolve(a_try, a_try)
            return c.max()

        def block_grad(x):
            a_try = a_fixed.copy()
            a_try[indices] = np.maximum(x, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                return np.zeros_like(x)
            a_try = a_try / s_try
            c = _full_convolve(a_try, a_try)
            c_max = float(c.max())
            tol = 0.005 * c_max
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()
            a_rev = a_try[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            g_full = 4.0 * n * a_try * h / s_try**2
            return g_full[indices]

        try:
            result = minimize(
                block_obj, a_block, jac=block_grad,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            a_new_block = np.maximum(result.x, 0)
            a_try = best_a.copy()
            a_try[indices] = a_new_block
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
        except Exception:
            pass

    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_time_remaining=30.0, n_betas=10):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # Create beta schedule
    betas = base_beta * np.array([1.0, 0.4, 0.16, 0.064, 0.0256,
                                   0.01, 0.004, 0.0016, 0.00064, 0.000256,
                                   0.0001, 0.00004, 0.000016, 0.0000064])[:n_betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 50, 'maxfun': max_iter * 2}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _random_phase_init(n, rng):
    """Initialize using random phases in frequency domain."""
    N = 2 * n - 1
    # Random phases in frequency domain
    phases = rng.uniform(0, 2 * np.pi, N)
    # Create a "flat spectrum" signal
    freq = np.ones(N // 2 + 1)
    # Apply random phases
    spectrum = freq * np.exp(1j * phases[:N // 2 + 1])
    # Make it Hermitian for real signal
    if N % 2 == 0:
        spectrum_full = np.concatenate([spectrum, np.conj(spectrum[1:N//2][::-1])])
    else:
        spectrum_full = np.concatenate([spectrum, np.conj(spectrum[1:N//2+1][::-1])])

    # IFFT to get time domain
    signal = np.fft.ifft(spectrum_full).real[:n]
    # Take absolute value to make non-negative
    signal = np.abs(signal)
    signal = np.maximum(signal, 1e-10)
    return signal / signal.sum()


def _sinusoidal_init(n, freq, rng):
    """Initialize with a sinusoidal profile."""
    t = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _cosine_taper_init(n):
    """Cosine tapered initialization."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.7
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.01):
    """Subgradient descent on the max (active set gradient)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        # Use adaptive tolerance
        c_max = float(c.max())
        tol = 0.001 * c_max if i > 100 else 0.002 * c_max
        grad = _gradient_active(a, c, n, tol_frac=tol / c_max if c_max > 0 else 0.002)
        step = lr / (1.0 + 0.002 * i)
        z = z - step * grad / max(np.linalg.norm(grad), 1e-30)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _try_different_n(a_source, n_target, deadline, min_time_remaining=60.0):
    """Resample to target n and optimize."""
    x_old = np.linspace(0, 1, len(a_source))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a_source)
    a_target = np.maximum(a_target, 0)
    a_target = a_target / a_target.sum()
    a_opt = _lbfgs_optimize(a_target, deadline, min_time_remaining=min_time_remaining)
    return a_opt


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: combines L-BFGS, Newton active-set, KKT-guided
    refinement, peak shaving, block local optimization, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
                update_best(parent, sc_parent, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _cosine_taper_init(n_parent)
        print(f"No parent, using cosine taper n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Multi-n exploration with diverse inits ===
    print(f"\n[Phase 1] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 600:
            break
        print(f"  Trying n={n_try}...")

        # Multiple initializations for this n
        inits = []

        # From parent
        if len(parent) != n_try:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_try)
            a_from_parent = np.interp(x_new, x_old, parent)
            a_from_parent = np.maximum(a_from_parent, 0)
            a_from_parent = a_from_parent / a_from_parent.sum()
        else:
            a_from_parent = parent.copy()
        inits.append(("parent", a_from_parent))

        # Cosine taper
        inits.append(("cosine", _cosine_taper_init(n_try)))

        # Random phase
        inits.append(("randphase", _random_phase_init(n_try, rng)))

        # Sinusoidal
        inits.append(("sinusoidal", _sinusoidal_init(n_try, rng.uniform(2, 8), rng)))

        # Uniform with noise
        a_noise = np.ones(n_try) + 0.1 * rng.standard_normal(n_try)
        a_noise = np.maximum(a_noise, 0.01)
        inits.append(("noise", a_noise / a_noise.sum()))

        for init_name, a_init in inits:
            if time.monotonic() > deadline - 300:
                break
            # Quick L-BFGS
            a_opt = _lbfgs_optimize(a_init, deadline, min_time_remaining=200, n_betas=6)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n{n_try}_{init_name}_lbfgs")

    # === Phase 2: Main aggressive L-BFGS on best ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_main = _lbfgs_optimize(best_seq, deadline, min_time_remaining=350, n_betas=14)
        update_best(a_main, _score(a_main), "main_lbfgs")

    # === Phase 3: Newton active-set ===
    print(f"\n[Phase 3] Newton active-set (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_deadline = min(deadline - 250, time.monotonic() + 250)
        a_newton, sc_newton = _newton_active_set(best_seq, n_deadline, max_iter=300)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 4: KKT-guided refinement ===
    print(f"\n[Phase 4] KKT-guided (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        kkt_deadline = min(deadline - 150, time.monotonic() + 150)
        a_kkt, sc_kkt = _kkt_guided_step(best_seq, kkt_deadline, n_steps=50)
        update_best(a_kkt, sc_kkt, "kkt")

    # === Phase 5: Peak shaving ===
    print(f"\n[Phase 5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        ps_deadline = min(deadline - 100, time.monotonic() + 100)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_deadline, n_iters=200)
        update_best(a_ps, sc_ps, "peak_shave")

    # === Phase 6: Block local optimization ===
    print(f"\n[Phase 6] Block local opt (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        bl_deadline = min(deadline - 60, time.monotonic() + 80)
        a_bl, sc_bl = _block_local_opt(best_seq, bl_deadline, n_blocks=30, block_size=25)
        update_best(a_bl, sc_bl, "block_local")

    # === Phase 7: Newton on best after all phases ===
    print(f"\n[Phase 7] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        nt_deadline = min(deadline - 50, time.monotonic() + 60)
        a_nt, sc_nt = _newton_active_set(best_seq, nt_deadline, max_iter=200, tol_frac=0.0005)
        update_best(a_nt, sc_nt, "newton_polish")

    # === Phase 8: Shrinkage + re-optimize ===
    print(f"\n[Phase 8] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 50)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score * 1.002 and time.monotonic() < deadline - 40:
            a_shr2 = _lbfgs_optimize(a_shr, deadline, min_time_remaining=20, n_betas=4)
            update_best(a_shr2, _score(a_shr2), "post_shrink_lbfgs")

    # === Phase 9: Subgradient polish ===
    print(f"\n[Phase 9] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        sg_deadline = min(deadline - 10, time.monotonic() + 25)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=2000, lr_init=0.008)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 10: Final Newton ===
    print(f"\n[Phase 10] Final Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 2, max_iter=100, tol_frac=0.0003)
        update_best(a_fin, sc_fin, "final_newton")

    # === Phase 11: Try a few more n values if time remains ===
    print(f"\n[Phase 11] Extra n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        extra_ns = [12288, 2048]
        for n_ex in extra_ns:
            if time.monotonic() > deadline - 20:
                break
            if len(best_seq) != n_ex:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_ex)
                a_ex = np.interp(x_new, x_old, best_seq)
            else:
                a_ex = best_seq.copy()
            a_ex = np.maximum(a_ex, 0)
            a_ex = a_ex / a_ex.sum()
            a_ex_opt = _lbfgs_optimize(a_ex, deadline, min_time_remaining=10, n_betas=4)
            sc_ex = _score(a_ex_opt)
            update_best(a_ex_opt, sc_ex, f"extra_n{n_ex}")
            if sc_ex < best_score * 1.003 and time.monotonic() < deadline - 10:
                a_ex_nt, sc_ex_nt = _newton_active_set(a_ex_opt, deadline - 3, max_iter=50)
                update_best(a_ex_nt, sc_ex_nt, f"extra_n{n_ex}_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256 * np.sqrt(512)
        # Verify with evaluate_sequence
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                if sc_eval < sc_final:
                    sc_final = sc_eval
                print(f"  evaluate_sequence gives: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(1024) / 1024 * np.sqrt(2048)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
