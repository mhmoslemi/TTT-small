from scipy.optimize import linprog

def find_direction(a, active_set):
    n = len(a)
    # Compute v_k[i] = d(conv[k])/da[i] for k in active_set
    # v_k[i] = 2*a[k-i] (if 0<=k-i<n and k!=2i) or a[i] (if k=2i) or 0
    
    # LP: minimize t
    # subject to:
    #   v_k · d <= t for k in active_set  ->  v_k · d - t <= 0
    #   sum(d) = 0
    #   d[i] >= -a[i]
    
    # Variables: d[0], ..., d[n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    
    # A_ub * [d, t] <= b_ub
    # Row for each k in active_set: [v_k[0], ..., v_k[n-1], -1] * [d, t] <= 0
    # A_eq * [d, t] = b_eq
    # Row: [1, ..., 1, 0] * [d, t] = 0
    # bounds: d[i] in [-a[i], inf], t in [-inf, inf]
    
    m = len(active_set)
    # A_ub is m x (n+1)
    # b_ub is m
    # A_eq is 1 x (n+1)
    # b_eq is 1
    
    # This is a large LP. For n=8192 and m~100, it's 100 constraints on 8193 variables.
    # scipy's linprog with method='highs' should handle this.
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    for idx, k in enumerate(active_set):
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    A_ub[idx, i] = a[ki]
                else:
                    A_ub[idx, i] = 2 * a[ki]
        A_ub[idx, n] = -1
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.zeros(1)
    
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]  # the direction d
    
    return None

One more idea: **use the structure of the problem to reduce the dimensionality**.

The convolution has 2n-1 = 16383 values. At the optimum, many of them are at the same level (the active set). If the active set has size m, then we have m equations: conv[k] = M for k in S. Plus the normalization: sum(a) = 1. Plus the KKT conditions for the free variables.

If the active set is, say, 50 peaks, and most variables are free (a[i] > 0), then we have:
- 50 equations: conv[k] = M
- 1 equation: sum(a) = 1
- n-1 KKT equations: gradient condition

Total: 52 + (n-1) = n+51 equations in n+1 unknowns (a[0..n-1] and M).

This is an overdetermined system in general, but at the optimum it should be consistent. We can solve it using a least-squares approach or a Newton method.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Log-sum-exp smoothed L-BFGS
2. LP-based steepest descent (new)
3. Active-set Newton refinement
4. Multi-strategy restarts
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute g[i] = sum_j a[j] * A_vec[i+j] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _compute_active_set(a, tol_frac=0.001, max_active=500):
    """Identify the active set of convolution peaks."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    threshold = c_max * (1 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > max_active:
        # Keep only the top max_active
        order = np.argsort(c[active])[::-1]
        active = active[order[:max_active]]
    return active, c, c_max


def _lp_steepest_descent(a, active_set, c, c_max):
    """
    Find the steepest descent direction using LP.
    Minimize t subject to:
      sum_i v_k[i] * d[i] <= t for k in active_set
      sum_i d[i] = 0
      d[i] >= -a[i]
    """
    n = len(a)
    m = len(active_set)
    
    # Build sparse A_ub matrix
    # Each row k: [v_k[0], ..., v_k[n-1], -1] * [d, t] <= 0
    # v_k[i] = d(conv[k])/da[i]
    
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_set):
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    v = a[ki]
                else:
                    v = 2.0 * a[ki]
                if v > 1e-15:
                    rows.append(idx)
                    cols.append(i)
                    vals.append(v)
        # Add -1 for the t variable (column n)
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n + 1))
    b_ub = np.zeros(m)
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: d[i] >= -a[i], t unbounded
    lb = -a.copy()
    lb = np.maximum(lb, -1.0)  # Cap to avoid numerical issues
    ub = np.zeros(n + 1)
    ub[n] = np.inf  # t is unbounded above
    # Actually, we want t to be as small as possible, and d[i] >= -a[i]
    # scipy linprog bounds are (lower, upper) pairs
    bounds = list(zip(lb, np.full(n, np.inf))) + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n].copy()
        # Ensure sum(d) = 0 (should be from constraint, but double-check)
        d = d - np.mean(d)
        return d
    return None


def _lp_steepest_descent_direct(a, active_set, c, c_max):
    """
    More efficient LP formulation using the structure of the problem.
    We only need the top active set members.
    """
    n = len(a)
    m = len(active_set)
    
    # Instead of building the full LP matrix element by element,
    # use the FFT-based correlation to compute v_k[i] on the fly.
    
    # Actually, for the LP we need the full matrix, so let's build it efficiently.
    # v_k[i] = d(conv[k])/da[i]
    # For a given k, v_k[i] = 2*a[k-i] (if 0<=k-i<n and k≠2i), a[i] (if k=2i), 0 otherwise
    # This is essentially a (possibly truncated and scaled) shifted version of a.
    
    # Let's build the A_ub matrix row by row using the structure.
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_set):
        # v_k[i] is nonzero for i in [max(0, k-n+1), min(n-1, k)]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            ki = k - i
            if k == 2 * i:
                v = a[ki]
            else:
                v = 2.0 * a[ki]
            if v > 1e-15:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
        # t column
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n + 1))
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lb = -a.copy()
    bounds = list(zip(lb, np.full(n, np.inf))) + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n].copy()
        d = d - np.mean(d)
        return d, float(result.x[n])
    return None, np.inf


def _lp_descent_loop(a_init, deadline, max_iter=200, tol=0.0005):
    """Iterative LP-based steepest descent."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify active set
        active_set, c, c_max = _compute_active_set(a, tol_frac=tol, max_active=300)
        m = len(active_set)
        
        if m == 0:
            break
        
        # Solve LP for steepest descent direction
        d, t_val = _lp_steepest_descent_direct(a, active_set, c, c_max)
        
        if d is None:
            # Fallback: subgradient step
            w = np.zeros(2*n-1)
            for k in active_set:
                w[k] = 1.0 / m
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            grad = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            d = -grad / gnorm
            t_val = None
        
        # Line search
        f_old = c_max
        t_max = 1.0
        neg_mask = d < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-d[neg_mask])) * 0.99
            t_max = max(t_max, 1e-10)
        
        accepted = False
        alpha = min(1.0, t_max)
        for ls in range(30):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 or ls == 29:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            alpha *= 0.5
        
        if not accepted:
            break
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        
        # Check convergence
        if abs(float(_full_convolve(a, a).max()) - c_max) < 1e-10:
            break
        
        if it % 10 == 0:
            print(f"    LP iter {it}: c_max={c_max:.10f}, score={sc:.10f}, "
                  f"active_set={m}, t_val={t_val if t_val is not None else 'N/A'}")
    
    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    """Damped Newton method on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:2*n]
            active[top_idx] = True
            n_active = 2 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve
        x = np.zeros(n)
        r = g_t - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            x += alpha_cg * p
            r_new = r - alpha_cg * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        u = x
        w_sol = np.zeros(n)
        r = np.ones(n) - hess_matvec(w_sol)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            w_sol += alpha_cg * p
            r_new = r - alpha_cg * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        sum_w = float(w_sol.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_sol
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-10:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)

        return a_init

    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007,
