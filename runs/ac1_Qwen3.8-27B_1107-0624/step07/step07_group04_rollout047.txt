import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[j,m] = 2*W[j+m] (Hankel matrix)."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=200, tol=1e-12):
    """Solve H*x = rhs using CG with regularization."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp < 0:
            # Not positive definite, add regularization
            pHp = abs(pHp) + 1e-10
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj_factory(beta, n):
    """Create smoothed objective function."""
    def smoobj(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    return smoobj


def _smograd_factory(beta, n):
    """Create smoothed gradient function."""
    def smograd(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    return smograd


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Compute beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    # Smooth geometric progression
    n_betas = 25
    ratios = np.geomspace(1.0, 1e-8, n_betas)
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        # Allocate more iterations for larger beta (coarser), fewer for smaller
        frac = (n_betas - i) / n_betas
        max_iter = min(2000, max(30, int(remaining * 0.6 / (est_iter_time * 20))))

        try:
            result = minimize(
                _smoobj_factory(beta, n),
                z,
                jac=_smograd_factory(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _newton_active_polish(a_init, deadline, max_rounds=100):
    """CG-Newton polish on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with adaptive tolerance
        tol = max(1e-6 * c_max, c_max - c_max * 0.001)
        active = c >= (c_max - 0.001 * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Gradient
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        # Newton step via CG
        Hg = _cg_solve(W, n, g, max_iter=200)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=200)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Backtracking line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.1 * alpha * gn * 0.01:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            alpha = 0.01
            for ls in range(30):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _mirror_descent(a_init, deadline, max_iter=10000):
    """Mirror descent in log-space for non-negative optimization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Use diminishing step size
    lr = 0.1
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient in log-space: dF/dz_i = a_i * dF/da_i
        g = _hankel_matvec(a, w, n)
        g = g - g.mean()  # Center for sum constraint
        
        # Mirror descent update
        step = lr / (1.0 + 0.01 * i)
        z = z - step * g
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        # Check improvement every 50 steps
        if i % 50 == 0:
            a_cur = _softmax(z)
            sc_cur = _score(a_cur)
            if sc_cur < best_sc - 1e-13:
                best_sc = sc_cur
                best_a = a_cur.copy()
        
    final_a = _softmax(z)
    final_sc = _score(final_a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = final_a.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_positive":
        rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
