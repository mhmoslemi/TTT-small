Adam-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing with adaptive temperature, simplex projection,
and active-set polishing. Explores multiple initializations and sequence lengths.
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_c(a):
    return _fft_convolve(a, a)


def _lse_obj_and_grad(z, beta):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z."""
    a = _softmax(z)
    n = len(a)
    c = _compute_c(a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))

"""
Adam-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- Exploits the fact that non-symmetric (non-palindromic) sequences can beat score=2
- Uses Adam optimizer with momentum for better landscape traversal
- Non-symmetric initializations (ramp, geometric) to find score < 2 basins
- Adaptive log-sum-exp temperature with restart from best on stagnation
- Active-set polishing with adaptive threshold
- Multi-n exploration with resampling
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    return e / s if s > 1e-300 else np.ones_like(z) / len(z)


def _grad_lse(z, beta):
    """
    Gradient of the log-sum-exp smoothed objective w.r.t. z (log-space).
    Objective: 2*n * sum_k w_k * c_k where w = softmax(beta * c), c = conv(a,a)
    Gradient w.r.t. a: 2 * correlate(w, a)[n-1:2n-1]
    Chain rule through softmax: grad_z = a * (grad_a - <a, grad_a>)
    """
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (valid k-j) = 2 * correlate(w,a)[n-1+j]
    conv = _fft_convolve(w, a[::-1])
    grad_a = 2.0 * conv[n - 1:2 * n - 1]
    # Softmax chain rule
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return grad_z

                break
            if stall % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75

        # Active set: positions where c is within tol of max
        threshold = c_max * (1.0 - tol_frac)
        active = c >= threshold
        if not active.any():
            active[np.argmax(c)] = True
        n_active = int(active.sum())

        # Weight
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Gradient: 2 * correlate(w, a)[n-1:2n-1]
        conv = _fft_convolve(w, a[::-1])
        grad_a = 2.0 * conv[n - 1:2 * n - 1]

        # Center for simplex (maintain sum=1)
        grad_c = grad_a - np.mean(grad_a)

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_c
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-12:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller active set
            if n_active > 3:
                top_idx = np.argsort(c)[::-1][:max(2, n_active // 3)]
                active = np.zeros_like(c, dtype=bool)
                active[top_idx] = True
                n_active = len(top_idx)
                w = np.zeros_like(c)
                w[active] = 1.0 / n_active
                conv = _fft_convolve(w, a[::-1])
                grad_a = 2.0 * conv[n - 1:2 * n - 1]
                grad_c = grad_a - np.mean(grad_a)
            lr = max(0.001, lr * 0.8)

    return best_a, _score(best_a)


def _resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_ramp(n, decreasing=True):
    """Create a ramp initialization."""
    t = np.arange(n)
    if decreasing:
        a = (n - t).astype(float)
    else:
        a = (t + 1).astype(float)
    a = a / a.sum()
    return a


def _make_geometric(n, r=0.97):
    """Create a geometric initialization."""
    t = np.arange(n)
    a = r ** t
    a = a / a.sum()
    return a


def _make_two_level(n, ratio=0.6):
    """Two-level: first 'ratio' fraction higher."""
    a = np.ones(n)
    split = int(n * ratio)
    a[:split] = 1.5
    a[split:] = 0.5
    a = a / a.sum()
    return a


def _minimax_equalize(a, deadline, n_top=3):
    """
    Try to equalize the top n_top lag values of the convolution.
    Uses iterative water-filling: reduce the max by redistributing.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for iteration in range(500):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = c.max()
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()

        # Find top n_top positions
        top_idx = np.argsort(c)[::-1][:n_top]
        top_vals = c[top_idx]
        top_mean = top_vals.mean()
        top_spread = top_vals.max() - top_vals.min()

        if top_spread < 1e-8 * c_max:
            break  # Already equalized

        # For each top position, compute gradient and apply opposite
        # to the one with highest value
        k_max = top_idx[0]
        # grad_a for c[k_max]: 2 * a[k_max - j] (zero-padded)
        grad_max = np.zeros(n)
        valid = (k_max - np.arange(n)) >= 0
        valid &= (k_max - np.arange(n)) < n
        grad_max[valid] = 2.0 * a[k_max - np.arange(n)[valid]]

        # Subtract a small amount along this gradient (reduces c[k_max])
        # But also add to reduce other top positions
        alpha = 0.01 * c_max / (np.linalg.norm(grad_max) + 1e-15)
        a_try = a - alpha * (grad_max - grad_max.mean())
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        c_try = _fft_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())
        if sc_try < sc - 1e-14:
            a = a_try
        else:
            alpha *= 0.5
            a_try = a - alpha * (grad_max - grad_max.mean())
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                c_try = _fft_convolve(a_try, a_try)
                if 2.0 * n * float(c_try.max()) < sc - 1e-14:
                    a = a_try
            else:
                break

    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Adam optimization with non-symmetric initializations,
    active-set polishing, and multi-n exploration.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent sequence
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_ramp(n_parent, decreasing=True)
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent. Ramp n={n_parent}, score={best_score:.10f}")

    n_base = len(best_seq)

    # === Phase 1: Adam optimization from multiple initializations ===
    print(f"\n[Phase 1] Adam optimization (t={time.monotonic()-t0:.1f}s)")

    # Build list of initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("ramp_dec", _make_ramp(n_base, decreasing=True)))
    inits.append(("ramp_inc", _make_ramp(n_base, decreasing=False)))
    inits.append(("geo", _make_geometric(n_base, r=0.97)))
    inits.append(("two_level", _make_two_level(n_base, ratio=0.55)))

    # Shuffled to try diverse ones first
    inits_shuffled = inits[:2] + list(reversed(inits[2:]))

    time_per_init = min(250.0, (deadline - 500.0) / len(inits_shuffled))
    for label, a_init in inits_shuffled:
        if time.monotonic() > deadline - 450:
            break
        init_deadline = time.monotonic() + time_per_init
        if init_deadline > deadline - 400:
            init_deadline = deadline - 400

        z = np.log(np.maximum(a_init, 1e-30))
        z = z - z.mean()

        a_opt, sc_opt = _adam_optimize(z, init_deadline, lr_init=0.08, label=label)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  [{label}] score={sc_opt:.10f} n={n_base} t={time.monotonic()-t0:.1f}s")
        else:
            print(f"  [{label}] score={sc_opt:.10f} (not better) t={time.monotonic()-t0:.1f}s")

    # === Phase 2: Second round of Adam from best (with perturbation) ===
    print(f"\n[Phase 2] Refine best (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() < deadline - 350:
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        # Add small perturbation to escape potential local min
        z = z + 0.01 * rng.standard_normal(n_base)
        z = z - z.mean()
        refine_deadline = min(deadline - 300, time.monotonic() + 200)
        a_opt, sc_opt = _adam_optimize(z, refine_deadline, lr_init=0.04)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  [refine] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")

    # === Phase 3: Minimax equalization ===
    print(f"\n[Phase 3] Minimax equalize (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() < deadline - 250:
        eq_deadline = min(deadline - 200, time.monotonic() + 100)
        a_eq, sc_eq = _minimax_equalize(best_seq, eq_deadline, n_top=3)
        if sc_eq < best_score:
            best_score = sc_eq
            best_seq = a_eq.copy()
            print(f"  [equalize] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")

    # === Phase 4: Active-set polish (coarse) ===
    print(f"\n[Phase 4] Active-set polish coarse (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() < deadline - 200:
        pol_deadline = min(deadline - 180, time.monotonic() + 150)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.01, tol_frac_end=0.001)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"  [polish_coarse] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")

    # === Phase 5: Active-set polish (fine) ===
    print(f"\n[Phase 5] Active-set polish fine (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() < deadline - 100:
        pol_deadline = min(deadline - 80, time.monotonic() + 120)
        a_pol2, sc_pol2 = _active_set_polish(best_seq, pol_deadline,
                                             tol_frac_start=0.003, tol_frac_end=0.0002)
        if sc_pol2 < best_score:
            best_score = sc_pol2
            best_seq = a_pol2.copy()
            print(f"  [polish_fine] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t0:.1f}s)")
    n_candidates = [2048, 4096, 6144, 8192, 12288]
    if n_base not in n_candidates:
        n_candidates.insert(0, n_base)
    # Sort by proximity to current best n first
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_base))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_base:
            continue
        a_rs = _resample(best_seq, n_try)
        z = np.log(np.maximum(a_rs, 1e-30))
        z = z - z.mean()
        n_deadline = min(deadline - 40, time.monotonic() + 180)
        a_opt, sc_opt = _adam_optimize(z, n_deadline, lr_init=0.05)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  [n={n_try}] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")
        elif sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            # Quick polish
            a_p, sc_p = _active_set_polish(a_opt, min(deadline - 25, time.monotonic() + 60),
                                           tol_frac_start=0.005, tol_frac_end=0.0005)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"  [n={n_try}_polish] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t0:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 2,
                                           tol_frac_start=0.001, tol_frac_end=0.00005)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  [final_polish] score={best_score:.10f} t={time.monotonic()-t0:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_ramp(256, decreasing=True)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
        return [float(x) for x in final]

    fallback = _make_ramp(256, decreasing=True)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
