"""
Optimized solver for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: Efficient Hankel-matrix Newton via FFT cross-correlation,
KKT-based active set management, structured frequency-domain perturbations,
and adaptive multi-scale refinement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[i,j] = 2*W[i+j], W has length 2n-1."""
    N = _next_pow2(3 * n - 2)
    F_W = np.fft.rfft(W, N)
    F_v = np.fft.rfft(v, N)
    result = np.fft.irfft(F_W * np.conj(F_v), N)
    return 2.0 * result[:n]


def _cg_solve(W, n, rhs, max_iter=200, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hankel matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-40:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-40 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            # Regularize: add small diagonal
            r_new = r
            rs_new = rs
            p = r.copy()
            x_new = x
        else:
            x = x_new
            r = r_new
            rs = rs_new
        if rs < tol * tol * rs0 or rs < 1e-30:
            break
        beta_cg = rs_new / rs if rs > 1e-40 else 0.0
        if not np.isfinite(beta_cg) or beta_cg > 10:
            beta_cg = 0.0
        p = r + beta_cg * p
    return x


def _newton_polish(a_init, deadline, max_rounds=200, active_tol_frac=0.0002):
    """Newton active-set polish with CG solver and KKT verification."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    ones = np.ones(n)

    # Cache H^{-1}*1 if active set doesn't change
    cached_W = None
    cached_v = None

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Active set with hysteresis
        tol = active_tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for efficiency
        if n_active > 2 * n:
            # Only keep the top 2n
            thresholds = np.sort(c)[::-1][min(2*n, len(c)-1)]
            active = c >= (c_max - max(tol, c_max - thresholds))
            n_active = int(active.sum())
            if n_active == 0:
                active[int(np.argmax(c))] = True
                n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Check if we can reuse cached H^{-1}*1
        W_changed = (cached_W is None or not np.array_equal(W, cached_W))
        if W_changed:
            cached_W = W.copy()
            cached_v = _cg_solve(W, n, ones, max_iter=150)
            v_norm_sq = float(ones @ cached_v)
        else:
            v_norm_sq = float(ones @ cached_v)

        if v_norm_sq < 1e-30:
            cached_v = _cg_solve(W, n, ones, max_iter=150)
            v_norm_sq = float(ones @ cached_v)
            if v_norm_sq < 1e-30:
                break

        # Compute gradient of quadratic form: g = H*a
        g = _hankel_matvec(a, W, n)
        # Residual: r = g - mu*ones where mu = (ones@g)/(ones@v) * ... 
        # Actually: mu = (ones @ g) / (ones @ v)
        mu = float(ones @ g) / v_norm_sq
        r = g - mu * ones

        # Check KKT: if ||r|| is small, we're at the optimum
        r_norm = float(np.linalg.norm(r))
        if r_norm < 1e-12 * c_max:
            break

        # Newton step: dx = -H^{-1}*r + v * (ones^T * H^{-1}*r) / (ones^T * v)
        u = _cg_solve(W, n, r, max_iter=150)
        ones_u = float(ones @ u)
        dx = -u + cached_v * (ones_u / v_norm_sq)

        # Ensure sum(dx) = 0 (numerical cleanup)
        dx = dx - np.mean(dx)

        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            break

        # Line search with backtracking
        alpha = 1.0
        step_done = False
        f_cur = float(np.dot(W, c))  # current weighted objective
        for ls in range(40):
            a_try = a + alpha * dx
            # Enforce non-negativity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(W, c_try))
            # Armijo condition
            if f_try < f_cur - 0.01 * alpha * r_norm * dx_norm:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Try smaller fixed step
            alpha = 0.01
            for ls in range(20):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(np.dot(W, c_try)) < f_cur:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Tighten active set tolerance as we converge
        if rnd > 20:
            active_tol_frac = max(1e-6, active_tol_frac * 0.95)

    return best_a, best_sc


def _lbfgs_phase(a_init, deadline, min_remaining=60.0):
    """L-BFGS with log-sum-exp beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)

    ratios = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        rem = deadline - min_remaining - time.monotonic()
        if rem < 3:
            break
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(10, int(rem / (est_time * 10))))

        try:
            res = minimize(
                _lsmoobj,
                z,
                args=(beta, n),
                jac=_lsmograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(res.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _lsmoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsmograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _structured_perturb(a, freq, amp, phase, rng):
    """Add structured frequency-domain perturbation."""
    n = len(a)
    t = np.arange(n) / n
    mod = 1.0 + amp * np.cos(2 * np.pi * freq * t + phase)
    mod = np.maximum(mod, 0.01)
    a_new = a * mod
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _random_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

    if n_parent not in [4096, 16384, 8192]:
        n_tries.insert(0, n_parent)
    for n_try in n_tries:
        rem = deadline - 50 - time.monotonic()
        if rem < 60:
            break
        a_init_n = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt, sc_opt = _try_n_value(a_init_n, n_try, deadline,
                                      min_remaining=40.0, label=f"n{n_try}")
        update_best(a_opt, sc_opt, f"multin_{n_try}")

    # === Phase 6: Final ultra-fine Newton ===
    print(f"\n[Phase 6] Final ultra-fine (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        rem = deadline - 5 - time.monotonic()
        if rem > 3:
            a_fin, sc_fin = _newton_polish(best_seq, deadline - 2, max_rounds=300,
                                           active_tol_frac=1e-6)
            update_best(a_fin, sc_fin, "final_ultra")

        # One more pass with extremely tight tolerance
        rem = deadline - 3 - time.monotonic()
        if rem > 2:
            a_fin2, sc_fin2 = _newton_polish(best_seq, deadline - 1, max_rounds=300,
                                             active_tol_frac=1e-7)
            update_best(a_fin2, sc_fin2, "final_ultra2")

    # === Final output ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(4096)
    print(f"FALLBACK: score={_score(fallback)}")
    return [float(x) for x in fallback]
