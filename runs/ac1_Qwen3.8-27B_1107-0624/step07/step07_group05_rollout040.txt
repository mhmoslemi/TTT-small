"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Spectral shaping (flat spectrum method) for convolution flattening
2. Adaptive active-set pressure equalization with multiple gamma/tol sweeps
3. Aggressive L-BFGS beta continuation with wider range
4. Multi-n exploration including larger sizes (16384, 32768)
5. Alternating spectral/pressure iterations for faster convergence
6. LP-free KKT solving via FFT-based matvec and conjugate gradient
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _spectral_shape(a, n_iters=50, deadline=None):
    """
    Flat spectrum method: iteratively flatten |FFT(a)|^2 to flatten convolution.
    This is the key new technique - directly shapes the spectrum for flat autocorrelation.
    """
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Pad and FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # Target: flat magnitude
        target_mag = 1.0 / np.sqrt(N)

        # Interpolate between current and target (soft update to maintain non-negativity)
        blend = 0.3 + 0.5 * (it / max(1, n_iters))  # Start gentle, become more aggressive
        new_mag = (1 - blend) * np.abs(A) + blend * target_mag
        new_mag = np.maximum(new_mag, 1e-10)

        # Reconstruct with current phase, target magnitude
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_new_pad = np.fft.irfft(A_new, N)
        a_new = a_new_pad[:n]

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_normalized(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def _alternating_spectral_pressure(a, deadline, n_iters=100, gamma=0.4):
    """
    Alternate between spectral shaping and pressure equalization.
    Spectral shaping flattens the overall structure;
    pressure equalization fine-tunes the active set.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Every 3 iterations, do spectral shaping
        if it % 3 == 0:
            a, sc_sp = _spectral_shape(a, n_iters=5, deadline=min(deadline - 1, time.monotonic() + 5))
            if sc_sp < best_sc:
                best_sc = sc_sp
                best_a = a.copy()
            continue

        # Pressure equalization step
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 5 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    # Extended beta schedule with more aggressive lowering
    multipliers = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                   0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    betas = base_beta * np.array(multipliers)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations based on n and remaining time
        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient descent in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 40 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
                best_patience_sc = best_sc
            else:
                patience += 1
                lr *= 0.9
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_real":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 3 * t * t) + 1.5
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 13 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 * 10) + 0.1
        return a / a.sum()
    elif kind == "wavelet_like":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(4 * np.pi * t) * np.exp(-5 * (t - 0.3) ** 2)
        a = a + 0.1 * np.sin(8 * np.pi * t) * np.exp(-5 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wechat":
        # Weaith-like: sum of shifted copies
        a = np.zeros(n)
        for shift in range(0, n, max(1, n // 7)):
            a[shift:min(n, shift + n // 3)] += 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

def _kkt_solve(a, S, n, n_cg_iters=200):
    """
    Solve the KKT system G @ a = c * 1, sum(a) = 1 using CG with FFT matvec.
    G[m,j] = 1 if m+j in S.
    """
    a = a.copy().astype(np.float64)
    s = np.zeros(2 * n - 1)
    for k in S:
        if 0 <= k < 2 * n - 1:
            s[k] = 1.0

    N = _next_pow2(2 * n - 1)
    s_pad = np.zeros(N)
    s_pad[:2 * n - 1] = s

    def _matvec(x):
        """Compute G @ x efficiently using FFT."""
        x_pad = np.zeros(N)
        x_pad[:n] = x
        # G[m,j] = s[m+j], so (G@x)_m = sum_j s[m+j]*x_j
        # This is cross-correlation of s and x
        sx = np.fft.rfft(s_pad) * np.fft.rfft(x_pad)
        result = np.fft.irfft(sx, N)
        # (G@x)_m = sum_j s[m+j] * x_j = result[m] (cross-correlation at lag m)
        return result[:n]

    # Solve G @ a = c * 1 with sum(a) = 1
    # Augmented system: [G, -1; 1', 0] [a; c] = [0; 1]
    # Use CG on the G part with constraint handled separately
    
    # First, find c: sum(a) = 1, so c = sum(G@a)/n (at the solution)
    # Actually, from G@a = c*1, summing over m: 1'G@a = c*n, so c = (1'G@a)/n
    # At the solution: c = sum(G@a)/n
    
    # Use iterative method: start with current a, iterate
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()

    for it in range(n_cg_iters):
        p = _matvec(a_curr)
        c_est = np.mean(p)
        # Residual: we want p to be flat, i.e., p - mean(p) = 0
        # But also want sum(a) = 1
        # Use Jacobi-preconditioned CG-like update
        r = p - c_est
        # Gradient of (1/2)*||G@a - c*1||^2 + (1/2)*(sum(a)-1)^2 
        # = G'(G*a - c*1) + 1*(sum(a)-1)
        # G' = G (symmetric)
        # So grad = G*(G*a - c*1) + (sum(a)-1)*1
        # = G*r + (sum(a)-1)*1
        
        # Simple update: a_new = a - lr * (G*r + (sum(a)-1)*1)
        Gr = _matvec(r)
        sum_residual = a_curr.sum() - 1.0
        grad = Gr + sum_residual
        lr = 0.5 / (np.linalg.norm(grad) + 1e-15)
        a_curr = a_curr - lr * grad
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()

        # Check convergence
        p_new = _matvec(a_curr)
        flatness = np.std(p_new) / (np.mean(p_new) + 1e-15)
        if flatness < 1e-6:
            break

    return a_curr


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search combining spectral shaping, pressure equalization,
    L-BFGS, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Spectral shaping + L-BFGS on parent ---
    print(f"\n[Phase 1] Spectral + L-BFGS on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        # Quick spectral shaping
        a_sp, sc_sp = _spectral_shape(parent, n_iters=40, deadline=min(deadline - 550, time.monotonic() + 30))
        _update(a_sp, _score(a_sp))

        # L-BFGS on the best
        start_a = best_seq.copy() if best_seq is not None else parent
        a1 = _run_lbfgs(start_a, deadline, min_time_remaining=550)
        _update(a1, _score(a1))

    # --- Phase 2: Alternating spectral + pressure on best ---
    print(f"\n[Phase 2] Alternating optimization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_alt, sc_alt_norm = _alternating_spectral_pressure(
            best_seq, min(deadline - 450, time.monotonic() + 120),
            n_iters=150, gamma=0.3)
        sc_alt = _score(a_alt)
        _update(a_alt, sc_alt)

        # Follow up with L-BFGS
        if time.monotonic() < deadline - 400:
            a_f = _run_lbfgs(best_seq, deadline, min_time_remaining=400)
            _update(a_f, _score(a_f))

    # --- Phase 3: Multi-gamma pressure equalization sweep ---
    print(f"\n[Phase 3] Gamma sweep (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        for gamma in [0.2, 0.3, 0.4, 0.5, 0.7]:
            if time.monotonic() > deadline - 350:
                break
            a_g, sc_g_norm = _alternating_spectral_pressure(
                best_seq, min(deadline - 300, time.monotonic() + 40),
                n_iters=60, gamma=gamma)
            _update(a_g, _score(a_g))

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 30:
            break

        print(f"  Trying n={n_try}, remaining={remaining_for_this:.0f}s")

        # Initialize: interpolate from best or use profile
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        else:
            a_init = _make_profile(n_try, "tapered")

        # Quick spectral + L-BFGS
        if time.monotonic() < deadline - 100:
            a_sp, _ = _spectral_shape(a_init, n_iters=20,
                                      deadline=min(deadline - 90, time.monotonic() + 20))
            min_rem = max(30, (deadline - time.monotonic()) * 0.3)
            a_opt, sc_opt = _multi_stage_optimize(a_sp, deadline, min_remaining=min_rem, label=f"n_{n_try}")
            _update(a_opt, sc_opt)

    # --- Phase 5: Diverse restarts ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restart_profiles = ["chirp_real", "double_cos", "wavelet_like", "exponential_taper", "wechat"]

        for i, prof in enumerate(restart_profiles):
            if time.monotonic() > deadline - 30:
                break

            a_blend = 0.5 * best_seq + 0.5 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=4, amplitude=0.02)

            a_sp, _ = _spectral_shape(a_pert, n_iters=10,
                                      deadline=min(deadline - 25, time.monotonic() + 10))
            a_r, sc_r = _multi_stage_optimize(a_sp, deadline, min_remaining=25, label=f"restart_{i}")
            _update(a_r, sc_r)

    # --- Phase 6: Final deep L-BFGS on absolute best ---
    print(f"\n[Phase 6] Final deep L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=5)
        _update(a_fin, _score(a_fin))

    # --- Phase 7: Final spectral + subgradient polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_sp, _ = _spectral_shape(best_seq, n_iters=20,
                                  deadline=min(deadline - 2, time.monotonic() + 15))
        _update(a_sp, _score(a_sp))

        if time.monotonic() < deadline - 2:
            a_pol, sc_pol = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.005)
            _update(a_pol, sc_pol)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
