"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Strategy: L-BFGS with log-sum-exp continuation, Newton active-set polish,
diverse initializations (including 1/sqrt(j+1) profile), multi-length exploration,
and aggressive polishing. Targets improvement below 1.505706.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _sliding_corr(v, a, n):
    """Compute result[i] = sum_{j=0}^{n-1} v[i+j] * a[j] for i=0,...,n-1."""
    N = _next_pow2(4 * n)
    v_padded = np.zeros(N)
    v_padded[:len(v)] = v
    a_padded = np.zeros(N)
    a_padded[:n] = a
    # result[i] = sum_j v[i+j] * a[j] = (flip(a) * v)[n-1+i]
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, N)
    fv = np.fft.rfft(v_padded, N)
    r = np.fft.irfft(fa * fv, N)
    return r[n - 1:2 * n - 1]


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = 4.0 * n * _sliding_corr(w, a, n)
    return a * (g - np.dot(g, a))


def _gradient_exact(a, c, n, tol_frac=0.001):
    """Active-set gradient: gradient of mean of top c values."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    g = 4.0 * n * _sliding_corr(w, a, n)
    return a * (g - np.dot(g, a))


def _hessian_matvec(v, active, n):
    """Compute H*v where H_{ij} = 4n/|A| * [i+j in active]."""
    N = _next_pow2(4 * n)
    v_padded = np.zeros(N)
    v_padded[:len(v)] = v
    a_rev = v[::-1]
    fa = np.fft.rfft(a_rev, N)
    fa_active = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fa_active, N)
    result = r[n - 1:2 * n - 1]
    return (4.0 * n / active.sum()) * result


def _cg_solve(hess_vec_fn, b, n, max_iter=200, tol=1e-12):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec_fn(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300, tol_frac=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient in a-space
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        g = 4.0 * n * _sliding_corr(w, a, n)

        # Project to tangent space (sum a = 1)
        g_t = g - np.mean(g)
        gnorm = float(np.linalg.norm(g_t))
        if gnorm < 1e-13:
            break

        # Newton step with damping
        mu = 1e-6
        def hess_fn(v):
            return _hessian_matvec(v, active, n) + mu * v

        dx = _cg_solve(hess_fn, -g_t, n, max_iter=100, tol=1e-12)
        dx = dx - np.mean(dx)

        # Wolfe line search
        f_old = float(np.dot(w, c))
        alpha = 1.0
        a_new = a.copy()
        cgp = float(np.dot(g_t, dx))
        c1 = 1e-4
        c2 = 0.9
        for ls in range(60):
            a_try = a + alpha * dx
            if np.any(a_try < -1e-10):
                alpha *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(w, c_try))
            if f_try < f_old + c1 * alpha * cgp:
                # Check curvature condition
                a_try2 = a + 0.5 * alpha * dx
                a_try2 = np.maximum(a_try2, 0)
                s_try2 = a_try2.sum()
                if s_try2 > 1e-15:
                    a_try2 = a_try2 / s_try2
                    c_try2 = _full_convolve(a_try2, a_try2)
                    g_try2 = 4.0 * n * _sliding_corr(w, a_try2, n)
                    g_try2 = g_try2 - np.mean(g_try2)
                    curv = float(np.dot(g_try2, dx))
                    if curv > c2 * cgp:
                        a_new = a_try
                        break
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=None):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    if n_betas is None:
        n_betas = 18

    beta_ratios = np.array([1.0, 0.6, 0.36, 0.216, 0.13, 0.078, 0.047,
                            0.028, 0.017, 0.010, 0.006, 0.0036, 0.0022,
                            0.0013, 0.0008, 0.00048, 0.00029, 0.00017])[:n_betas]
    betas = base_beta * beta_ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1, n / 4096.0)
        max_iter = min(12000, max(100, int(remaining / (est_time * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            print(f"    beta={beta:.2e}, iter={i}, score={current_sc:.10f}, time={time.monotonic()-t_global_start:.0f}s")
        except Exception as ex:
            print(f"    L-BFGS error at beta={beta:.2e}: {ex}")
            break

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "inv_sqrt":
        j = np.arange(n)
        a = 1.0 / np.sqrt(j + 1.0)
        return a / a.sum()
    elif kind == "inv_sqrt_asc":
        j = np.arange(n)
        a = 1.0 / np.sqrt(n - j)
        return a / a.sum()
    elif kind == "power_alpha":
        j = np.arange(n)
        a = (j + 1.0) ** (-0.3)
        return a / a.sum()
    elif kind == "power_alpha2":
        j = np.arange(n)
        a = (j + 1.0) ** (-0.7)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_exact(a, c, n, tol_frac=0.001)
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * (0.3 + 0.7 * lr / lr_init)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_descent(a_init, deadline, n_passes=50):
    """Simple coordinate descent: optimize one element at a time."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, max(1, n // 2000)):
            if time.monotonic() > deadline:
                break
            # Try small perturbations of a[i]
            orig = a[i]
            for frac in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0]:
                a[i] = orig * frac
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[i] = orig
                # Try setting to 0
                a[i] = 0.0
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = orig

        a = best_a.copy()
        if not improved:
            break

    return best_a, best_sc


def t_global_start():
    global _t0
    return _t0


_t0 = time.monotonic()


def propose_candidate(seed=42, budget_s=1800):
    global _t0
    _t0 = time.monotonic()
    deadline = _t0 + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - _t0:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(8192, "inv_sqrt")
        sc_parent = _score(parent)
        print(f"No parent, using inv_sqrt n=8192, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    # === Phase 1: Main L-BFGS on parent ===
    print(f"\n[Phase 1] Main L-BFGS on parent (t={time.monotonic() - _t0:.1f}s)")
    if time.monotonic() < deadline - 700:
        a1 = _run_lbfgs(best_seq, deadline, min_time_remaining=700, n_betas=18)
        sc1 = _score(a1)
        update_best(a1, sc1, "lbfgs_main")

    # === Phase 2: Newton active-set polish ===
    print(f"\n[Phase 2] Newton active-set (t={time.monotonic() - _t0:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        nd = min(deadline - 500, _t0 + 250)
        a2, sc2 = _newton_active_set(best_seq, nd, max_iter=200, tol_frac=0.0005)
        update_best(a2, sc2, "newton1")
        # Second Newton pass with tighter tolerance
        if time.monotonic() < deadline - 450:
            a2b, sc2b = _newton_active_set(a2, min(deadline - 450, _t0 + 350), max_iter=100, tol_frac=0.0002)
            update_best(a2b, sc2b, "newton2")

    # === Phase 3: Subgradient polish ===
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - _t0:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, _t0 + 200)
        a3, sc3 = _subgrad_polish(best_seq, sg_deadline, n_steps=3000, lr_init=0.01)
        update_best(a3, sc3, "subgrad")
        # Newton after subgradient
        if sc3 < best_score * 1.001 and time.monotonic() < deadline - 350:
            a3b, sc3b = _newton_active_set(a3, min(deadline - 350, _t0 + 300), max_iter=100, tol_frac=0.0005)
            update_best(a3b, sc3b, "subgrad+newton")

    # === Phase 4: Alternative profiles at same n ===
    print(f"\n[Phase 4] Alternative profiles (t={time.monotonic() - _t0:.1f}s)")
    n_parent = len(best_seq) if best_seq is not None else 8192
    profiles = ["inv_sqrt", "inv_sqrt_asc", "power_alpha", "power_alpha2"]
    for prof in profiles:
        if time.monotonic() > deadline - 200:
            break
        a_init = _make_profile(n_parent, prof)
        # Cross with best: interpolate in log-space
        z_best = np.log(np.maximum(best_seq, 1e-30))
        z_prof = np.log(np.maximum(a_init, 1e-30))
        z_mix = 0.3 * z_best + 0.7 * z_prof
        z_mix = z_mix - z_mix.mean()
        a_mix = _softmax(z_mix)

        a4 = _run_lbfgs(a_mix, deadline, min_time_remaining=120, n_betas=12)
        sc4 = _score(a4)
        update_best(a4, sc4, f"profile_{prof}")

    # === Phase 5: Different n values ===
    print(f"\n[Phase 5] Different n values (t={time.monotonic() - _t0:.1f}s)")
    for n_try in [4096, 16384]:
        if time.monotonic() > deadline - 150:
            break
        if n_try == n_parent:
            continue
        # Resize best to n_try
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 1e-15)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_profile(n_try, "inv_sqrt")

        a5 = _run_lbfgs(a_init, deadline, min_time_remaining=80, n_betas=14)
        sc5 = _score(a5)
        update_best(a5, sc5, f"n_{n_try}")

        if sc5 < best_score * 1.005 and time.monotonic() < deadline - 60:
            a5b, sc5b = _newton_active_set(a5, min(deadline - 60, _t0 + 100), max_iter=80, tol_frac=0.0005)
            update_best(a5b, sc5b, f"n_{n_try}_newton")

    # === Phase 6: Final aggressive L-BFGS with very low beta ===
    print(f"\n[Phase 6] Final ultra-low beta L-BFGS (t={time.monotonic() - _t0:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00005

        remaining = deadline - 20 - time.monotonic()
        if remaining > 5:
            est_time = 0.004 * max(1, n / 4096.0)
            max_iter = min(10000, max(200, int(remaining / (est_time * 10))))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 80, 'maxls': 100}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                update_best(a_fin, sc_fin, "ultra_lbfgs")

                # Newton polish on the result
                if time.monotonic() < deadline - 5:
                    a_fin2, sc_fin2 = _newton_active_set(a_fin, deadline - 5, max_iter=50, tol_frac=0.0003)
                    update_best(a_fin2, sc_fin2, "ultra+newton")
            except Exception as ex:
                print(f"    Final L-BFGS error: {ex}")

    # === Phase 7: Very final Newton + subgradient ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - _t0:.1f}s)")
    if best_seq is not None and deadline - 3 - time.monotonic() > 2:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 3, max_iter=30, tol_frac=0.0002)
        update_best(a_fin, sc_fin, "final_newton")

        if deadline - 2 - time.monotonic() > 1:
            a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_steps=500, lr_init=0.005)
            update_best(a_sg, sc_sg, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - _t0:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
