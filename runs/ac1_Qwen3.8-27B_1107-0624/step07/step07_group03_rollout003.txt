"""

4. Multi-n exploration with interpolation and re-optimization

import numpy as np
import time
from scipy.optimize import minimize


def _npo2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _npo2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_with_active(a, active, n):
    """Compute g[i] = sum_{k active} a[k-i] (correlation of a with active set)."""
    flip_a = a[::-1]
    out_len = n + len(active) - 1
    N = _npo2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(active.astype(np.float64), N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_newton(a_init, deadline, max_iter=500, tol_active=0.001, tol_kkt=1e-10):
    """Active-set Newton with dynamic set growth and KKT verification."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c_max_prev = float(_conv(a, a).max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _conv(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for efficiency
        max_active = min(4 * n, len(c))
        if n_active > max_active:
            idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = max_active

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of f(a) = mean of active convolution values
        g = (2.0 / m) * _corr_with_active(a, A_vec, n)
        # Project to tangent space (zero mean)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)

        # KKT check: if projected gradient is small, we're at (approximate) optimum
        if gnorm < tol_kkt:
            break

        # Newton step via CG on the Hessian-restricted system
        mu = 1e-8 * (c_max / max(m, 1)) + 1e-10

        def _hess_matvec(v):
            return (2.0 / m) * _corr_with_active(v, A_vec, n) + mu * v

        # Solve (H + mu*I) u = g_t via CG
        x = np.zeros(n)
        r = g_t - _hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(200):
            Ap = _hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r, rs = r_new, rs_new
        u = x

        # Enforce sum constraint
        w = np.zeros(n)
        r = np.ones(n) - _hess_matvec(w)
        p = r.copy()
        rs = np.dot(r, r)
        for _ in range(200):
            Ap = _hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            w += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r, rs = r_new, rs_new
        w_sum = float(w.sum())
        if abs(w_sum) < 1e-15:
            dx = -u - np.mean(u)
        else:
            lam = float(u.sum()) / w_sum
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search with non-negativity
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99)
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for _ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _conv(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-13:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Fallback: projected gradient step
            t = min(0.1, t_max)
            for _ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                f_try = float(_conv(a_try, a_try).max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new_max = float(_conv(a, a).max())
        if abs(c_new_max - c_max) < 1e-13 * max(1, c_max):
            break

    return best_a, best_sc


def _subgrad_optimize(a_init, deadline, max_steps=10000, lr_init=0.02, tol=0.002):
    """Subgradient method in log-space with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30)).copy()
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_f = best_sc

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _conv(a, a)
        c_max = float(c.max())
        # Active set with tolerance
        act_tol = tol * c_max
        active = c >= (c_max - act_tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30)).copy()
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.6

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _conv(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = max_c * 0.01
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                  0.015, 0.0075, 0.0035, 0.0015, 0.0007, 0.0003])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30)).copy()
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 10))))
        try:
            result = minimize(
                _ls_obj_factory(z, beta, n),
                z,
                jac=_ls_grad_factory(z, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30)).copy()
                z = z - z.mean()
        except Exception:
            pass
    return current_a

def _lbfgs_direct(z, beta, n, deadline, min_remaining=30.0):
    """Direct L-BFGS call with shared state (no closures)."""
    remaining = deadline - min_remaining - time.monotonic()
    if remaining < 3:
        return z
    est_time = 0.004 * max(1.0, n / 2048.0)
    max_iter = min(6000, max(30, int(remaining / (est_time * 10))))
    try:
        result = minimize(
            lambda z_local: _ls_obj(z_local, beta, n),
            z,
            jac=lambda z_local: _ls_grad(z_local, beta, n),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50}
        )
        return result.x
    except Exception:
        return z


def _frank_wolfe(a_init, deadline, max_iter=5000):
    """Frank-Wolfe (conditional gradient) on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _conv(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Projected gradient direction (Frank-Wolfe: move toward best vertex)
        # s = argmin <grad, e_i> = e_{argmin grad}
        i_best = int(np.argmin(grad))
        direction = np.zeros(n)
        direction[i_best] = 1.0
        direction -= a  # a - s where s is the vertex

        # Step size: gamma = 2/(it+2) for FW, or line search
        gamma = 2.0 / (it + 2)
        # Also try line search
        best_t = gamma
        best_f = c_max
        for t_try in [gamma, gamma * 0.5, gamma * 0.25, 1.0 / (it + 2)]:
            a_try = a + t_try * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            f_try = float(_conv(a_try, a_try).max())
            if f_try < best_f:
                best_f = f_try
                best_t = t_try

        a = a + best_t * direction
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if it % 100 == 0 and best_sc > best_sc + 1e-12:
            pass  # No stagnation handling needed for FW

    return best_a


def _coord_descent(a_init, deadline, n_passes=5):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try multiple values for a[i]
            a_save = a[i]
            if a_save < 1e-15:
                continue
            candidates = [a_save * 0.5, a_save * 0.75, a_save * 1.25, a_save * 1.5,
                          a_save * 0.25, a_save * 2.0]
            best_local = a_save
            best_local_sc = best_sc
            for t in candidates:
                a_try = a.copy()
                a_try[i] = t
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_local_sc - 1e-14:
                    best_local_sc = sc_try
                    best_local = t
            if best_local != a_save:
                a[i] = best_local
                s = a.sum()
                a = a / s
                if best_local_sc < best_sc - 1e-14:
                    best_sc = best_local_sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        a[:n // 4] *= 0.5
        a[-n // 4:] *= 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Active-set Newton from parent + subgradient
    Phase 2: L-BFGS with log-sum-exp
    Phase 3: Frank-Wolfe refinement
    Phase 4: Multi-n exploration
    Phase 5: Coordinate descent + final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        sc_parent = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    # Phase 1: Active-set Newton (main workhorse)
    print(f"\n[Phase 1] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    as_deadline = min(deadline - 600, time.monotonic() + 300)
    a_as, sc_as = _active_newton(parent, as_deadline, max_iter=400, tol_active=0.001)
    update_best(a_as, sc_as, "as_newton_1")

    # Subgradient from the Newton result
    if best_seq is not None and time.monotonic() < deadline - 500:
        sg_deadline = min(deadline - 500, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_optimize(best_seq, sg_deadline, max_steps=8000, lr_init=0.015, tol=0.002)
        update_best(a_sg, sc_sg, "subgrad_1")

    # Second Newton round
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_deadline2 = min(deadline - 400, time.monotonic() + 120)
        a_as2, sc_as2 = _active_newton(best_seq, as_deadline2, max_iter=300, tol_active=0.0005)
        update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 2: L-BFGS
    print(f"\n[Phase 2] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30)).copy()
        z = z - z.mean()
        c_cur = _conv(best_seq, best_seq)
        max_c = float(c_cur.max())
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
        base_beta = max(gap, max_c * 0.005)

        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075])
        for i, beta in enumerate(betas):
            if time.monotonic() > deadline - 250:
                break
            z = _lbfgs_direct(z, beta, n, deadline, min_remaining=200 - i * 20)
            a_new = _softmax(z)
            sc_new = _score(a_new)
            update_best(a_new, sc_new, f"lbfgs_beta_{i}")

        # Final Newton after L-BFGS
        if time.monotonic() < deadline - 200:
            as_deadline3 = min(deadline - 200, time.monotonic() + 60)
            a_as3, sc_as3 = _active_newton(best_seq, as_deadline3, max_iter=200, tol_active=0.0003)
            update_best(a_as3, sc_as3, "as_newton_3")

    # Phase 3: Frank-Wolfe
    print(f"\n[Phase 3] Frank-Wolfe (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        fw_deadline = min(deadline - 150, time.monotonic() + 120)
        a_fw = _frank_wolfe(best_seq, fw_deadline, max_iter=3000)
        update_best(a_fw, _score(a_fw), "frank_wolfe")

        # Newton after Frank-Wolfe
        if time.monotonic() < deadline - 100:
            a_fw2, sc_fw2 = _active_newton(a_fw, min(deadline - 100, time.monotonic() + 40),
                                            max_iter=100, tol_active=0.0005)
            update_best(a_fw2, sc_fw2, "fw_then_as")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_parent]
    if n_parent >= 4096:
        n_candidates.append(n_parent // 2)
    if n_parent <= 8192:
        n_candidates.append(n_parent * 2)
    n_candidates.append(4096)
    n_candidates.append(8192)
    # Remove duplicates and sort
    n_candidates = sorted(set([n for n in n_candidates if 1024 <= n <= 16384]))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == n_parent:
            continue  # Already optimized

        if best_seq is not None:
            a_init = _interpolate(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        # Quick Newton
        if time.monotonic() < deadline - 30:
            a_opt, sc_opt = _active_newton(a_init, min(deadline - 30, time.monotonic() + 50),
                                            max_iter=100, tol_active=0.002)
            update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restarts_params = [
            (0.02, "perturb_small"),
            (0.05, "perturb_mid"),
            (0.10, "perturb_large"),
        ]
        for sigma, label in restarts_params:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score:
                update_best(p_start, sc_start, label)
            # Quick Newton from perturbed
            if time.monotonic() < deadline - 10:
                a_r, sc_r = _active_newton(p_start, min(deadline - 10, time.monotonic() + 30),
                                            max_iter=80, tol_active=0.001)
                update_best(a_r, sc_r, f"{label}_as")

    # Phase 6: Coordinate descent polish
    print(f"\n[Phase 6] Coord descent + final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        cd_deadline = min(deadline - 3, time.monotonic() + 40)
        a_cd = _coord_descent(best_seq, cd_deadline, n_passes=3)
        update_best(a_cd, _score(a_cd), "coord_descent")

        # Final tight Newton
        if time.monotonic() < deadline - 2:
            a_fin, sc_fin = _active_newton(best_seq, deadline - 1, max_iter=200, tol_active=0.0002)
            update_best(a_fin, sc_fin, "final_as")

        # Final subgradient with tiny LR
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _subgrad_optimize(best_seq, deadline - 0.5, max_steps=500, lr_init=0.003, tol=0.001)
            update_best(a_fin2, sc_fin2, "final_sg")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
