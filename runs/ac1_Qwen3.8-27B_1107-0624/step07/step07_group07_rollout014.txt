"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-method approach: fine L-BFGS with log-sum-exp, top-k smoothing,
iterative reweighting, structured perturbations, and multi-restart polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_conv_w_a(w, a, n):
    """Compute gradient contribution: for each j, sum_i w[i]*a[i-j] (a[m]=0 outside [0,n-1])."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _grad_conv_w_a(w, a, n)
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * (h - W)
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return grad_z


def _topk_obj(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    idx = np.argpartition(c, -k)[-k:]
    return 2.0 * n * c[idx].mean()


def _topk_grad(z, k, n, smooth_eps=1e-8):
    a = _softmax(z)
    c = _full_convolve(a, a)
    thresh = np.partition(c, -k)[-k]
    w = np.zeros_like(c)
    active = c >= (thresh - smooth_eps)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    w[active] = 1.0 / active.sum()
    h = _grad_conv_w_a(w, a, n)
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * (h - W)
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return grad_z


def _lbfgs_run(a_init, fun, jac, max_iter, ftol=1e-18, gtol=1e-13):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _lbfgs_ls_schedule(a_start, deadline, min_time_remaining=200.0):
    """L-BFGS with log-sum-exp beta continuation."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                              0.0025, 0.0012, 0.0006, 0.0003,
                              0.00015, 0.00008, 0.00004, 0.00002,
                              0.00001, 0.000005, 0.000002, 0.000001,
                              0.0000005, 0.0000002, 0.0000001, 0.00000005])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_per_iter = 0.0015 * (n / 4096.0) ** 1.3
        max_iter = min(15000, max(300, int(remaining / (est_per_iter * 15))))

        def fun(z):
            return _ls_obj(z, beta, n)

        def jac(z):
            return _ls_grad(z, beta, n)

        a_new = _lbfgs_run(current_a, fun, jac, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            if i % 4 == 0:
                print(f"    LS beta={beta:.2e}: score={sc:.10f}")

    return current_a

    """L-BFGS with top-k smoothing, decreasing k."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    c_len = 2 * n - 1
    k_values = [max(1, int(c_len * f)) for f in [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 1]]
    current_a = a_start.copy()
    current_sc = _score(current_a)

def _iterative_reweight(a_init, deadline, max_iters=5000, p_init=4.0, p_growth=1.15):
    """Iterative reweighting: minimize weighted sum of c, updating weights to concentrate on peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    p = p_init

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break

        # Compute weights: w ~ c^p (normalized)
        w = np.power(c / c_max, p)
        w = w / w.sum()

        # Gradient of weighted sum: sum_k w[k] * c[k]
        # d(sum w*c)/d(a[j]) = 2 * sum_k w[k] * a[k-j] = 2 * (w * a_rev)[j+n-1]
        h = _grad_conv_w_a(w, a, n)
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)  # We multiply by n for scaling

        # Projected gradient step in direct space (maintain non-negativity and sum)
        a = a / s  # normalize
        grad = grad - np.dot(grad, a)  # project onto tangent space of simplex
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Backtracking line search
        step = 0.05
        improved = False
        for _ in range(10):
            a_try = a - step * grad / g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                step = min(step * 1.1, 0.2)
                break
            step *= 0.5
            if step < 1e-10:
                break

        if not improved:
            p *= p_growth
            if p > 100:
                break
        else:
            p = max(2.0, p * 0.98)

    return best_a, best_sc

def _peak_shaving_direct(a_init, deadline, max_iters=30000):
    """Iterative peak shaving in direct space with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Adaptive active set: top few peaks
        sorted_c = np.sort(c)[::-1]
        tol = max(1e-6 * c_max, c_max - sorted_c[min(50, len(c) - 1)])
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        h = _grad_conv_w_a(w, a, n)
        W = float(np.dot(w, c))
        g = 4.0 * n * (h - W)
        g = g - np.dot(g, a) * a  # project to maintain normalization approximately
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        best_step = eta
        best_a_try = a.copy()
        best_sc_try = _score(a)

        for frac in [1.0, 0.5, 0.25, 0.1]:
            step = eta * frac
            a_try = a - step * g / g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_step = step

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.005, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.95
            if eta < 1e-10 or stall > 200:
                break

    return best_a, best_sc

        elif mode == 2:

    print(f"\n=== Final polish cycle ===")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_f1, _ = _peak_shaving_direct(best_seq, deadline - 3, max_iters=5000)
        try_update(a_f1, "final_ps")
        if time.monotonic() < deadline - 3:
            a_f2 = _ultra_fine_polish(best_seq, deadline - 2)
            try_update(a_f2, "final_ultra")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
