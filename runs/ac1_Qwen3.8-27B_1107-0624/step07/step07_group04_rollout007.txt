_g_beta = 1.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:

def _cg_solve(W, n, rhs, max_iter=300, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

d(a*a)_k / da_j = a_{k-j} (for valid indices)

So the gradient vector for constraint k is essentially a shifted and reversed version of a.

If we have active constraints at positions k_1, ..., k_m, we need:

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous: adaptive aggressive beta continuation for L-BFGS,
tighter CG-Newton with non-uniform active set weighting, KKT-based peak equalization,

import numpy as np

from scipy.optimize import minimize

_g_beta = 1.0
_g_n = 0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z):
    a = _softmax(z)
    n = _g_n
    beta = _g_beta
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z):
    a = _softmax(z)
    n = _g_n
    beta = _g_beta
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-15):

    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)

        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _kkt_equalize(a_init, deadline, max_rounds=50):
    """Direct KKT-based peak equalization using top-K active constraints."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top-K peaks with adaptive K
        K = min(20, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:K]
        # Only include peaks that are significantly above the rest
        threshold = c_max - 0.01 * c_max
        active_idx = top_k[c[top_k] >= threshold]
        if len(active_idx) == 0:
            active_idx = top_k[:1]
        n_act = len(active_idx)

        # Build the gradient matrix: each column is grad of (a*a)_{k_i}
        # grad_j of (a*a)_k = a_{k-j} (reversed and shifted)
        # We solve: find direction d such that sum_i w_i * (grad_i @ d) < 0
        # and (a a)_k_i are equalized

        # Compute the "excess" of each active peak
        c_act = c[active_idx]
        c_mean = c_act.mean()
        excess = c_act - c_mean

        # Build gradient vectors for each active peak
        # grad_k[j] = a[k-j] for valid j (this is the reversed a, shifted)
        a_rev = a[::-1]
        G = np.zeros((len(active_idx), n))
        for i, k in enumerate(active_idx):
            # (a*a)_k = sum_j a[j] * a[k-j], d/d a[j] = a[k-j]
            # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                G[i, lo:hi + 1] = a_rev[k - lo:k - hi + 1]

        # Weighted combination of gradients
        w = np.abs(excess) / (np.abs(excess).sum() + 1e-30)
        g_combined = G.T @ w  # shape (n,)

        # Project to zero-sum (to preserve normalization)
        g_combined = g_combined - np.mean(g_combined)
        g_norm = float(np.linalg.norm(g_combined))
        if g_norm < 1e-15:
            break

        # Newton direction: solve H*d = -g_combined where H is the Hessian
        # H_{j,m} = 2 * W[j+m] where W has the active weights
        W_act = np.zeros(2 * n - 1)
        W_act[active_idx] = w
        Hg = _cg_solve(W_act, n, g_combined, max_iter=600)
        H1 = _cg_solve(W_act, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_combined / g_norm
        else:
            lam = float(np.dot(np.ones(n), Hg)) / float(np.dot(np.ones(n), H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)

        # Line search
        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1 - 1e-10):
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Smaller gradient step
            for trial_step in [0.01, 0.005, 0.001, 0.0005]:
                a_try = a - trial_step * g_combined / (g_norm + 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s_val = float(p.sum())
        if s_val < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_val = float(parent_raw.sum())
            if s_val > 0.01:
                parent = parent_raw / s_val
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                _update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a1 = _run_lbfgs(parent, deadline, min_time_remaining=400.0, aggressive=True)
        _update_best(a1, _score(a1), "lbfgs_aggr")

    # Phase 2: CG-Newton polish (uniform weights)
    print(f"\n[Phase 2] Newton uniform (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        dl = min(deadline - 250, time.monotonic() + 120)
        a2, sc2 = _newton_polish(best_seq, dl, max_rounds=60, active_tol_frac=5e-5, weight_mode="uniform")
        _update_best(a2, sc2, "newton_uni")

    # Phase 3: CG-Newton polish (exponential weights)
    print(f"\n[Phase 3] Newton exp (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        dl = min(deadline - 200, time.monotonic() + 80)
        a3, sc3 = _newton_polish(best_seq, dl, max_rounds=50, active_tol_frac=2e-5, weight_mode="exp")
        _update_best(a3, sc3, "newton_exp")

    # Phase 4: KKT-based peak equalization
    print(f"\n[Phase 4] KKT equalize (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        dl = min(deadline - 150, time.monotonic() + 80)
        a4, sc4 = _kkt_equalize(best_seq, dl, max_rounds=60)
        _update_best(a4, sc4, "kkt_eq")

    # Phase 5: Another Newton pass (tighter)
    print(f"\n[Phase 5] Newton tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        dl = min(deadline - 100, time.monotonic() + 60)
        a5, sc5 = _newton_polish(best_seq, dl, max_rounds=80, active_tol_frac=1e-5, weight_mode="exp")
        _update_best(a5, sc5, "newton_tight")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [6144, 10240, 12288, 4096, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0, aggressive=False)
        sc_opt = _score(a_opt)
        _update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 60:
            dl = min(deadline - 50, time.monotonic() + 50)
            a_np, sc_np = _newton_polish(a_opt, dl, max_rounds=40, active_tol_frac=1e-4)
            _update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 7: Structured restarts from best
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            (0.02, "tapered"),
            (0.05, "cos"),
            (0.1, "gauss"),
            (0.03, "bimodal"),
        ]
        for i, (sigma, prof) in enumerate(restart_configs):
            if time.monotonic() > deadline - 40:
                break
            if i < 2:
                p_start = _perturb(best_seq, sigma, rng)
            else:
                p_start = _make_profile(len(best_seq), prof)
                # Blend with best
                p_start = 0.5 * p_start + 0.5 * best_seq
                p_start = p_start / p_start.sum()

            if time.monotonic() > deadline - 35:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0, aggressive=False)
            sc_r = _score(a_r)
            _update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 20:
                dl = min(deadline - 15, time.monotonic() + 25)
                a_rp, sc_rp = _newton_polish(a_r, dl, max_rounds=30, active_tol_frac=1e-4)
                _update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight Newton
        dl = deadline - 3
        a_fin, sc_fin = _newton_polish(best_seq, dl, max_rounds=100, active_tol_frac=5e-6, weight_mode="exp")
        _update_best(a_fin, sc_fin, "final_newton")

        # Final KKT
        if time.monotonic() < deadline - 3:
            a_kkt, sc_kkt = _kkt_equalize(best_seq, deadline - 2, max_rounds=50)
            _update_best(a_kkt, sc_kkt, "final_kkt")

        # Ultra-fine L-BFGS with tiny beta
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            _g_n = n
            _g_beta = mb * 5e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _smoobj, z.copy(),
                        jac=_smograd,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    _update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass
            _g_beta = 0.0
            _g_n = 0

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
