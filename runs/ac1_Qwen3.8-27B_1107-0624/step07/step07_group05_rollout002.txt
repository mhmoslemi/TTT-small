"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Soft active-set pressure equalization with Gaussian smoothing
2. Adaptive momentum in multiplicative updates
3. More aggressive L-BFGS beta continuation with wider range
4. Multi-n exploration with interpolation-based resampling
5. Spectral-domain initialization for better starting points
6. Dual-variable-based coordinate updates for final polishing

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() < 2:
        idx = np.argsort(c)[-5:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _soft_pressure_equalize(a, deadline, n_iters=300, gamma_init=0.5, gamma_end=0.1,
                            sigma_init=5.0, sigma_end=1.0, momentum=0.7):
    """
    Improved pressure equalization using Gaussian-smoothed active set.
    The 'soft' active set uses a Gaussian kernel centered at the peak,
    giving smoother gradients than the hard active set.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    v = np.zeros(n)  # momentum vector

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters - 1)
        gamma = gamma_init + (gamma_end - gamma_init) * progress
        sigma = sigma_init + (sigma_end - sigma_init) * progress

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Soft active set: Gaussian centered at argmax
        idx_max = float(np.argmax(c))
        k = np.arange(len(c))
        soft_active = np.exp(-0.5 * ((k - idx_max) / sigma) ** 2)
        soft_active = soft_active / soft_active.sum()

        # Compute pressure using soft active set
        a_rev = a[::-1]
        p = _full_convolve(soft_active, a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Multiplicative update with momentum
        p_mean = np.mean(p)
        ratio = p_mean / p
        update = np.power(ratio, gamma)
        v = momentum * v + (1 - momentum) * (update - 1.0)
        a = a * np.exp(v)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 10 == 0 or it == n_iters - 1:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Estimate active set width for initial beta
    sorted_c = np.sort(c_cur)[::-1]
    n_conv = len(sorted_c)
    # Find where convolution drops to 50% of max
    below = np.where(sorted_c < 0.5 * max_c)[0]
    if len(below) > 0:
        active_width = float(below[0])
    else:
        active_width = n_conv * 0.1
    base_beta = max(max_c / max(1, active_width), max_c * 0.01)

    # Extended beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    start_time = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations based on n and remaining time
        est_time_per_iter = 0.008 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining * 0.8 / max(est_time_per_iter, 0.001))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Subgradient descent with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.003)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_polish(a, deadline, n_rounds=80):
    """Coordinate descent focusing on peak-contributing coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(20, n // 20))
        top_idx = np.argsort(contrib)[-n_try:]
        np.random.shuffle(top_idx)

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            c_old = c_max
            for factor in [0.6, 0.8, 0.9, 1.1, 1.4]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-14:
                    c_old = c_try
                    c_max = c_try
                    improved = True
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t) \
            - 0.1 * np.cos(6 * np.pi * t) + 0.05 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_seq(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = _make_profile(new_n, "uniform")
    return a_new


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0):
    """Full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: Soft pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        t_pe = min(deadline - min_remaining - 10, time.monotonic() + 40)
        a1, sc1_norm = _soft_pressure_equalize(current_a, t_pe, n_iters=150,
                                                gamma_init=0.4, gamma_end=0.1,
                                                sigma_init=8.0, sigma_end=2.0,
                                                momentum=0.5)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: L-BFGS
    if time.monotonic() < deadline - min_remaining - 10:
        a2 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 10)
        sc2 = _score(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: Hard pressure equalization
    if time.monotonic() < deadline - min_remaining - 5:
        t_pe2 = min(deadline - min_remaining - 2, time.monotonic() + 30)
        a3, sc3_norm = _hard_pressure_equalize(current_a, t_pe2, n_iters=100, gamma=0.3,
                                                tol_frac=0.005)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: L-BFGS again (finer)
    if time.monotonic() < deadline - min_remaining - 3:
        a4 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 3)
        sc4 = _score(a4)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        a5, sc5 = _subgrad_polish(current_a, min(deadline - min_remaining,
                                                  time.monotonic() + 20),
                                  n_max_steps=1500, lr_init=0.015)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: soft/hard pressure equalization, L-BFGS with
    adaptive beta, multi-n exploration via interpolation, and coordinate polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    np.random.seed(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=800, label="main")
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure equalization ---
    print(f"\n[Phase 2] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        # Soft pressure equalization with longer run
        t_pe = min(deadline - 500, time.monotonic() + 100)
        a_pe, sc_pe_norm = _soft_pressure_equalize(best_seq, t_pe, n_iters=400,
                                                    gamma_init=0.3, gamma_end=0.05,
                                                    sigma_init=10.0, sigma_end=1.5,
                                                    momentum=0.6)
        sc_pe = _score(a_pe)
        _update(a_pe, sc_pe)

        # Follow up with L-BFGS
        if sc_pe < best_score + 0.001 and time.monotonic() < deadline - 400:
            a_follow = _run_lbfgs(a_pe, deadline, min_time_remaining=400)
            _update(a_follow, _score(a_follow))

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = []
    # Generate candidates around parent size
    for factor in [0.75, 1.0, 1.25, 1.5, 2.0]:
        n_try = int(n_parent * factor)
        # Round to nice number
        n_try = max(1024, (n_try // 128) * 128)
        if n_try not in n_candidates and n_try != n_parent:
            n_candidates.append(n_try)

    # Also add some standard values
    for n_std in [6144, 8192, 10240, 12288, 16384]:
        if n_std not in n_candidates and n_std != n_parent:
            n_candidates.append(n_std)

    # Sort by size (try larger first for potential improvement)
    n_candidates.sort(reverse=True)
    # Limit to top 4 to respect time budget
    n_candidates = n_candidates[:4]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining_for_this = deadline - 200 - time.monotonic()
        if remaining_for_this < 60:
            break

        print(f"  Trying n={n_try} (remaining: {remaining_for_this:.0f}s)")
        a_init = _resample_seq(best_seq if best_seq is not None else parent, n_try)
        min_rem = max(30, remaining_for_this * 0.25)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 4: Resample best back to various sizes and reoptimize ---
    print(f"\n[Phase 4] Resample and reoptimize (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        # Try resampling to slightly different sizes
        for n_try in [n_best + 1024, n_best + 2048, max(1024, n_best - 1024)]:
            if n_try <= 0 or time.monotonic() > deadline - 60:
                continue
            if n_try == n_best:
                continue
            remaining_here = deadline - 60 - time.monotonic()
            if remaining_here < 40:
                break
            a_res = _resample_seq(best_seq, n_try)
            a_opt, sc_opt = _multi_stage_optimize(a_res, deadline, min_remaining=30)
            _update(a_opt, sc_opt)

    # --- Phase 5: Coordinate descent polish ---
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd, sc_cd_norm = _coordinate_polish(best_seq, min(deadline - 5, time.monotonic() + 80),
                                              n_rounds=60)
        sc_cd = _score(a_cd)
        _update(a_cd, sc_cd)

    # --- Phase 6: Final L-BFGS + subgradient ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=3)
        _update(a_fin, _score(a_fin))

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 1, n_max_steps=500, lr_init=0.005)
            _update(a_fin2, sc_fin2)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
