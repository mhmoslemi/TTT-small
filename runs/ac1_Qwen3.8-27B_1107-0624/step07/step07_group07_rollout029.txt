"""
Step sequence optimizer using minimax-gradient direction finding with
active-set QP, L-BFGS continuation, and coordinate descent polishing.
Novel approach: solves a small QP to find the direction that most evenly
reduces all active autocorrelation peaks simultaneously.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _minimax_direction(a, c, n, n_active=20):
    """
    Find the direction that most evenly reduces all active autocorrelation peaks.
    Returns direction d such that a - t*d reduces the max convolution.
    """
    c_max = c.max()
    # Find active lags
    sorted_idx = np.argsort(c)[::-1]
    active_idx = sorted_idx[:min(n_active, len(c))]
    # Filter to those close to max
    tol = 0.01 * c_max
    active_idx = active_idx[c[active_idx] >= c_max - tol]
    if len(active_idx) == 0:
        active_idx = sorted_idx[:1]
    
    m = len(active_idx)
    # Build A matrix: A[i, j] = a[k_i - j] if 0 <= k_i - j <= n-1, else 0
    # We need A^T w for w in R^m
    # A^T w = sum_i w_i * A[i,:]
    # A[i, j] = a[k_i - j] (if valid)
    # So (A^T w)[j] = sum_i w_i * a[k_i - j] (if k_i - j in [0, n-1])
    
    def compute_At_w(w):
        """Compute A^T w efficiently without forming A"""
        result = np.zeros(n)
        for i in range(m):
            k = active_idx[i]
            wi = w[i]
            if wi < 1e-15:
                continue
            # A[i, j] = a[k - j] for j in [max(0, k-n+1), min(k, n-1)]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                result[j] += wi * a[k - j]
        return result
    
    # Solve: min_{w in simplex} ||A^T w||_2^2
    # = min_{w >= 0, sum(w)=1} w^T (A A^T) w
    # C = A A^T, C[i,j] = A[i,:] . A[j,:]
    
    def compute_C(i, j):
        """C[i,j] = sum_k A[i,k] * A[j,k] = sum_t a[k_i - t] * a[k_j - t] (over valid t)"""
        ki = active_idx[i]
        kj = active_idx[j]
        # Valid t: 0 <= ki-t <= n-1 and 0 <= kj-t <= n-1
        t_start = max(0, ki - n + 1, kj - n + 1)
        t_end = min(ki, kj, n - 1)
        if t_start > t_end:
            return 0.0
        # sum_{t=t_start}^{t_end} a[ki-t] * a[kj-t]
        seg_i = a[ki - t_end:ki - t_start + 1][::-1]
        seg_j = a[kj - t_end:kj - t_start + 1][::-1]
        return float(np.dot(seg_i, seg_j))
    
    # For small m, compute full C matrix
    C = np.zeros((m, m))
    for i in range(m):
        for j in range(i, m):
            C[i, j] = compute_C(i, j)
            C[j, i] = C[i, j]

    # Use scipy
    if m == 1:
        w_opt = np.array([1.0])
    else:
        def qp_obj(w):
            return float(w @ C @ w)
        
        def qp_grad(w):
            return 2.0 * C @ w
        
        constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1.0,
                       'jac': lambda w: np.ones(m)}]

"""
Step sequence optimizer using minimax-gradient direction finding via active-set QP,
L-BFGS with log-sum-exp continuation, and coordinate descent polishing.
Key novel component: solves a small QP to find the direction that most evenly
reduces all active autocorrelation peaks simultaneously (minimax descent).
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _autocorr_from_conv(a):
    """Compute autocorrelation r[lag] = sum_t a[t+lag]*a[t] from conv(a, a_rev)."""
    a_rev = a[::-1]
    c = _full_convolve(a, a_rev)
    n = len(a)
    # c[n-1+lag] = r[lag] for lag in [-(n-1), ..., n-1]
    return c


def _minimax_direction(a, c, n, n_active=30):
    """
    Find direction d such that a - t*d reduces max convolution.
    Uses QP: min_{w in simplex} ||A^T w||_2^2
    Returns (direction, expected_rate) or (None, 0) if no direction found.
    """
    c_max = c.max()
    if c_max < 1e-30:
        return None, 0.0

    sorted_idx = np.argsort(c)[::-1]
    active_idx = sorted_idx[:n_active]
    tol = 0.05 * c_max
    mask = c[active_idx] >= c_max - tol
    active_idx = active_idx[mask]
    if len(active_idx) < 2:
        active_idx = sorted_idx[:min(5, len(sorted_idx))]

    m = len(active_idx)

    # Compute autocorrelation for C matrix: C[i,j] = r[k_i - k_j]
    a_rev = a[::-1]
    ac = _full_convolve(a, a_rev)
    # ac[n-1+lag] = sum_t a[t+lag]*a[t] = r[lag]

    C = np.zeros((m, m))
    for i in range(m):
        for j in range(i, m):
            lag = active_idx[i] - active_idx[j]
            if -n + 1 <= lag <= n - 1:
                C[i, j] = ac[n - 1 + lag]
            C[j, i] = C[i, j]

    # Solve QP: min w^T C w s.t. sum(w)=1, w>=0
    if m == 1:
        w_opt = np.array([1.0])
    else:
        def _qp_obj(w):
            return float(w @ C @ w)

        def _qp_grad(w):
            return 2.0 * C @ w

        def _qp_con_fun(w):
            return np.sum(w) - 1.0

        def _qp_con_jac(w):
            return np.ones(m)

        constraints = [{'type': 'eq', 'fun': _qp_con_fun, 'jac': _qp_con_jac}]
        bounds = [(0.0, 1.0)] * m
        w0 = np.ones(m) / m
        try:
            res = minimize(_qp_obj, w0, jac=_qp_grad, method='SLSQP',
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 1000, 'ftol': 1e-14})
            w_opt = np.clip(res.x, 0, None)
            ws = w_opt.sum()
            if ws < 1e-15:
                w_opt = np.ones(m) / m
            else:
                w_opt = w_opt / ws
        except Exception:
            w_opt = np.ones(m) / m

    # Compute direction d = A^T w via FFT
    # d[j] = sum_i w_i * a[k_i - j]
    # Using: d[j] = conv(h, a_rev)[n-1+j] where h[k_i] = w_i
    h = np.zeros(2 * n - 1)
    for i in range(m):
        h[active_idx[i]] = w_opt[i]

    d_full = _full_convolve(h, a_rev)
    d = d_full[n - 1:2 * n - 1].copy()

    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return None, 0.0

    d = d / d_norm
    # Expected reduction rate (per unit step)
    rate = float(np.min(C @ w_opt)) / d_norm
    return d, rate


def _minimax_iterate(a, deadline, max_iters=5000, n_active=30):
    """Iteratively apply minimax direction steps."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    t = 0.05
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        d, rate = _minimax_direction(a, c, n, n_active)
        if d is None:
            break

        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = t * frac
            a_try = a - step * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                t = min(t * 1.03, 0.2)
                improved = True
                stall = 0
                break

        if not improved:
            t *= 0.7
            stall += 1
            if stall > 100:
                break

    return best_a, best_sc


def _lbfgs_continuation(a_start, deadline, min_remaining=60.0):
    """L-BFGS with log-sum-exp beta continuation."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)

    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def _fun(z):
            return _smoobj(z, beta, n)

        def _jac(z):
            return _smograd(z, beta, n)

        est_iter_cost = 0.003 * (n / 4096.0) ** 1.3
        max_iter = min(20000, max(500, int(remaining / (est_iter_cost * 15))))

        try:

                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxcor': 100,
                                       'maxls': 100, 'disp': False})
            a_new = _softmax(result.x)

            if sc < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc
                print(f"    beta={beta:.2e}: score={sc:.10f} (nit={result.nit})")
        except Exception:
            pass

def _coordinate_descent(a_init, deadline, max_rounds=50000):
    """Coordinate descent: optimize pairs of elements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng_local = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = int(rng_local.integers(0, n))
        jdx = int(rng_local.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7, 0.45, 0.55]:
            a_try = a.copy()
            a_try[idx] = s_pair * frac
            a_try[jdx] = s_pair * (1.0 - frac)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    if time.monotonic() < deadline - 250:
        a_mm1, sc_mm1 = _minimax_iterate(best_seq, deadline - 250, max_iters=3000, n_active=25)
        try_update(a_mm1, "minimax_p2")
    print(f"  Phase 2 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 3: Fine L-BFGS ===
    print(f"\n=== Phase 3: Fine L-BFGS ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 150:
        a_p3 = _lbfgs_continuation(best_seq, deadline - 150, min_remaining=100.0)
        try_update(a_p3, "lbfgs_p3")
    print(f"  Phase 3 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 4: Alternate minimax + L-BFGS ===
    print(f"\n=== Phase 4: Alternate minimax + L-BFGS ===")
    t_ph = time.monotonic()
    for cycle in range(6):
        if time.monotonic() > deadline - 100:
            break
        # Minimax steps
        if time.monotonic() < deadline - 80:
            a_mm, _ = _minimax_iterate(best_seq, deadline - 80, max_iters=1000, n_active=40)
            try_update(a_mm, f"mm_alt_{cycle}")
        # Fine L-BFGS
        if time.monotonic() < deadline - 60:
            a_lb = _lbfgs_continuation(best_seq, deadline - 60, min_remaining=50.0)
            try_update(a_lb, f"lbfgs_alt_{cycle}")
    print(f"  Phase 4 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    t_ph = time.monotonic()
    for target_n in [4096, 16384]:
        if time.monotonic() > deadline - 80:
            break
        if target_n == start_n:
            continue
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_i = _score(a_interp)
        print(f"    Interpolated: score={sc_i:.10f}")
        try_update(a_interp, f"interp_{target_n}")

        a_opt = _lbfgs_continuation(a_interp, deadline - 60, min_remaining=50.0)
        try_update(a_opt, f"lbfgs_{target_n}")

        if time.monotonic() < deadline - 40:
            a_mm, _ = _minimax_iterate(a_opt, deadline - 40, max_iters=500, n_active=25)
            try_update(a_mm, f"mm_{target_n}")
    print(f"  Phase 5 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-restart ===")
    t_ph = time.monotonic()
    for name, scale in [("s1", 0.01), ("s2", 0.03), ("s5", 0.05), ("s10", 0.1)]:
        if time.monotonic() > deadline - 40:
            break
        a_pert = _perturb(best_seq, rng, scale)
        a_r = _lbfgs_continuation(a_pert, deadline - 30, min_remaining=25.0)
        try_update(a_r, f"restart_{name}")
        if time.monotonic() < deadline - 25:
            a_r2, _ = _minimax_iterate(a_r, deadline - 25, max_iters=500, n_active=30)
            try_update(a_r2, f"restart_{name}_mm")
    print(f"  Phase 6 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coordinate descent ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 10:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 10, max_rounds=20000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 7 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 8: Final intensive cycle ===
    print(f"\n=== Phase 8: Final polish ===")
    t_ph = time.monotonic()
    for cyc in range(4):
        if time.monotonic() > deadline - 5:
            break
        if time.monotonic() < deadline - 5:
            a_f1, _ = _minimax_iterate(best_seq, deadline - 5, max_iters=2000, n_active=50)
            try_update(a_f1, f"final_mm_{cyc}")
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            z0 = np.log(np.maximum(best_seq, 1e-15))
            z0 = z0 - z0.mean()
            beta = mb * 1e-8
            def _fin_fun(z):
                return _smoobj(z, beta, n)
            def _fin_jac(z):
                return _smograd(z, beta, n)
            rem = deadline - 5 - time.monotonic()
            if rem > 5:
                max_iter = min(10000, max(500, int(rem / 0.004)))
                try:
                    res = minimize(_fin_fun, z0, jac=_fin_jac, method='L-BFGS-B',
                                   options={'maxiter': max_iter, 'ftol': 1e-16,
                                            'gtol': 1e-13, 'maxcor': 100,
                                            'maxls': 100, 'disp': False})
                    a_f2 = _softmax(res.x)
                    try_update(a_f2, f"final_lbfgs_{cyc}")
                except Exception:
                    pass
    print(f"  Phase 8 done: {time.monotonic()-t_ph:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    return [float(x) for x in fallback]
