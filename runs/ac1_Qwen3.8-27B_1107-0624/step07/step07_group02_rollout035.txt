Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous SOTA:

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c_max = float(c.max())

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_levels=16):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.6, 0.37, 0.22, 0.13, 0.08,
                                  0.05, 0.03, 0.018, 0.011, 0.007,
                                  0.004, 0.0025, 0.0015, 0.0008, 0.0004])[:n_levels]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 4096.0)
        est_time_per_iter = 0.004 * scale
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 15))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Reset z from current best
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-16))
        a_old_sc = current_sc
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.65
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.05)

    return best_a, _score(best_a)


def _coordinate_scaling(a_init, deadline):
    """Try scaling individual coordinates to reduce the peak."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Iteratively try to reduce the peak by scaling coordinates
    for iteration in range(200):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find the peak position
        k_peak = int(np.argmax(c))

        # Find which coordinates contribute to the peak
        # c[k_peak] = sum_i a[i]*a[k_peak-i]
        i_min = max(0, k_peak - n + 1)
        i_max = min(n - 1, k_peak)

        improved = False
        for i in range(i_min, i_max + 1):
            j = k_peak - i
            if i == j:
                continue  # Skip diagonal terms (less effective)
            # Try reducing a[i] and a[j] proportionally
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            # Try small decrease
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                a_try = a.copy()
                dv_i = frac * a[i]
                dv_j = frac * a[j]
                a_try[i] -= dv_i
                a_try[j] -= dv_j
                # Redistribute to a random other element
                if not improved:
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-13:
                        a = a_try
                        current_sc = sc_try
                        improved = True
                        break
        if not improved:
            # Try increasing off-peak elements
            peak_mask = np.zeros(2*n-1, dtype=bool)
            peak_mask[np.argsort(c)[-max(1, n//10):]] = True
            for _ in range(min(50, n // 100)):
                i = np.random.randint(0, n)
                if a[i] > 0.1 * a.mean():
                    continue
                a_try = a.copy()
                a_try[i] += 0.001 * a.mean()
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-13:
                    a = a_try
                    current_sc = sc_try
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_sc


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

    print(f"  P8 done: elapsed={time.monotonic()-t8:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
