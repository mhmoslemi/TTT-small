Convex optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive L-BFGS with log-sum-exp continuation, Adam subgradient,
and pressure equalization for KKT-based refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj_and_grad(z, beta, n):
    """Returns (objective, gradient) of the smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        # Fallback: use active set
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad

def _lbfgs_continue(a_init, deadline, min_remaining=20.0):
    """Adaptive L-BFGS with continuation on beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Estimate gap
    sorted_c = np.sort(c_cur)[::-1]

    base_beta = max(gap * 1.5, max_c * 0.05)
    
    # Fine continuation schedule
    betas = base_beta * np.array([1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

        if time.monotonic() + min_remaining > deadline:

            break

def _adam_subgrad(a_init, deadline, n_steps=3000, lr=0.01):
    """Adam optimizer for the subgradient of the max-norm objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: coefficients within 1% of max
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        if t % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

"""
Convex optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive L-BFGS with log-sum-exp continuation, Nesterov-accelerated
subgradient descent, KKT pressure equalization, and multi-n exploration.
Exploits the convexity of the problem (max of convex quadratics over simplex).
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_continue(a_init, deadline, min_remaining=20.0):
    """Adaptive L-BFGS with fine continuation schedule on beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    sorted_c = np.sort(c_cur)[::-1]
    if len(sorted_c) > 30:
        gap = sorted_c[0] - sorted_c[30]
    elif len(sorted_c) > 2:
        gap = sorted_c[0] - sorted_c[-1]
    else:
        gap = max_c
    base_beta = max(gap * 2.0, max_c * 0.05)

    betas = base_beta * np.array([
        1.0, 0.5, 0.3, 0.18, 0.1, 0.06, 0.035, 0.02, 0.012,
        0.007, 0.004, 0.002, 0.001, 0.0006, 0.0003, 0.0001
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    scale = max(1.0, n / 2048.0)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * scale
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_subgrad(a_init, deadline, n_steps=3000, lr_init=0.015):
    """Nesterov-accelerated subgradient on the active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    t_val = 1.0

    best_a = a_init.copy()
    best_sc = _score(a_init)

    for it in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.008 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        lr = lr_init / (1.0 + it * 0.0002)
        z = z - lr * grad / (grad_norm + 1e-12)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        theta = (t_val - 1.0) / t_new
        z_prev = z + theta * (z - z_prev)
        z_prev = np.clip(z_prev, -50.0, 50.0)
        z_prev = z_prev - z_prev.mean()
        t_val = t_new

        if it % 80 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _adam_subgrad(a_init, deadline, n_steps=3000, lr=0.01):
    """Adam optimizer for active-set subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_a = a_init.copy()
    best_sc = _score(a_init)

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -45.0, 45.0)
        z = z - z.mean()

        if t % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _pressure_eq(a, deadline, n_iters=200, gamma=0.3):
    """KKT-motivated multiplicative pressure equalization."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.004 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-6:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def _block_cd_polish(a, deadline, block_size=128, n_rounds=30):
    """Block coordinate descent: optimize each block while fixing others."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = (n + block_size - 1) // block_size

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False

        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)
            block = a[start:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-15:
                continue

            # For this block, try scaling to reduce max convolution
            # The block contributes to conv[k] = sum_i a_i * a_{k-i}
            # For indices in the block, changing a_i affects conv[k] for
            # k in [i, i + n - 1]
            # Simple heuristic: try uniform redistribution within block
            c_old = _full_convolve(a, a).max()

            # Try: make block more uniform
            block_uniform = np.full_like(block, block_sum / len(block))
            a[start:end] = block_uniform
            c_try = _full_convolve(a, a).max()
            if c_try < c_old * (1.0 - 1e-8):
                improved = True
            else:
                a[start:end] = block

            # Try: reduce peak-contributing elements
            c_cur = _full_convolve(a, a)
            c_max = c_cur.max()
            # Contribution of block to peak
            peak_idx = int(np.argmax(c_cur))
            contrib = np.zeros(end - start)
            for i in range(start, end):
                j = peak_idx - i
                if 0 <= j < n:
                    contrib[i - start] = a[i] * a[j]
            # Scale down top contributors
            top_frac = max(3, len(block) // 4)
            top_mask = np.zeros(len(block), dtype=bool)
            top_indices = np.argsort(contrib)[-top_frac:]
            top_mask[top_indices] = True
            factor = 0.7
            a[start:end][top_mask] *= factor
            # Renormalize
            a[start:end] = a[start:end] * (block_sum / a[start:end].sum())
            c_try2 = _full_convolve(a, a).max()
            if c_try2 < c_try * (1.0 - 1e-8):
                improved = True

        if improved:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.35) ** 2) + 0.3 * np.exp(-6.0 * (t - 0.65) ** 2) + 0.05
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2 * t - 1) ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[len(z_hat) - 1 - freq + freq] = z_hat[freq]
    # Reconstruct (real signal)
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)
