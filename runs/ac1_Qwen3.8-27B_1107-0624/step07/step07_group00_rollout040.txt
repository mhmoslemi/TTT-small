LP-guided and gradient-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Linear programming to find optimal descent directions,
combined with soft-peak multiplicative updates and coordinate descent polishing.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_lp_matrix(a):
    """Build the constraint matrix for the LP: C[k,j] = a[j] if 0 <= k-j <= n-1 else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    # C[k,j] = a[j] * 1[0 <= k-j <= n-1]
    # = a[j] * 1[j <= k <= j+n-1]
    # For each j, C[j:j+n, j] = a[j]
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a[j]
    return C

def _lp_direction(a, M):
    """Solve LP: max sum(g) s.t. C@ g <= M, g >= 0. Returns g or None."""
    n = len(a)
    C = _build_lp_matrix(a)

    b_ub = np.full(2*n-1, M)

            return res.x
        else:
            return None

- i = j: the term is a[j] * a[k-j] (valid if 0 <= k-j <= n-1)
- k-i = j, i.e., i = k-j: the term is a[k-j] * a[j] (valid if 0 <= k-j <= n-1)

If j != k-j (i.e., j != k/2), these are two different terms, and both equal a[j] * a[k-j].
If j = k/2 (k = 2j), it's a single term a[j]^2.

So:
- If k = 2j: contribution to conv(k) from a[j] is a[j]^2
- If k != 2j and 0 <= k-j <= n-1: contribution is 2*a[j]*a[k-j]
- Otherwise: no contribution

So for fixed j:
conv(k) = {
  a[j]^2 + R_2j(k) if k = 2j
  2*a[j]*a[k-j] + R_j(k) if 0 <= k-j <= n-1 and k != 2j
  R_j(k) otherwise
}

where R is the part not involving a[j].

To minimize max_k conv(k), I need to find a[j] >= 0 such that the maximum over k of the above expressions is minimized.

The expressions that involve a[j] are:
- f_0(a[j]) = a[j]^2 + R_2j (at k = 2j, if 2j < 2n-1)
- f_k(a[j]) = 2*a[j]*a[k-j] + R_k (at various k)

All f_k are increasing in a[j] (since a[k-j] >= 0). And f_0 is also increasing (for a[j] >= 0).

So the max of all f's is maximized by increasing a[j], and minimized by setting a[j] = 0!

But wait, we also need sum(a) = 1. If we set a[j] = 0, we need to increase other elements to maintain the sum. So the constraint sum(a) = 1 couples all coordinates.

In coordinate descent with the simplex constraint, when we change a[j], we need to adjust the sum. There are different ways to handle this:
1. Change a[j] and redistribute the difference proportionally to other elements
2. Change a[j] and all other elements simultaneously (not really coordinate descent)
3. Work in the "log-space" or "simplex" directly

For simplicity, let me use approach 1: when I change a[j] by delta, I subtract delta/(n-1) from each other element. This maintains the sum.

But this is getting complicated. Let me just use a simpler CD: for each j, try a few values of a[j] (e.g., 0.5x, 0.7x, 1x, 1.3x, 2x of current value), renormalize, and pick the best.

This is what the previous code does in `_fast_cd_polish`. Let me keep that approach but make it faster.

"""
LP-guided and gradient-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated soft-peak multip梯度 updates, LP-based direction finding
(via scipy linprog on downsampled problems), coordinate descent polishing,
and multi-scale exploration (n=2048/4096/8192/16384).
Key innovation: LP directions exploit the banded structure of the convolution
constraint to find principled descent directions beyond simple gradients.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_peak_grad(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / max(T, 1e-15)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _active_peak_grad(a, c, top_k=5, decay=0.6):
    n = len(a)
    idx = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, ix in enumerate(idx):
        p[ix] = decay ** i
    ps = p.sum()
    if ps < 1e-300:
        return np.zeros(n)
    p = p / ps
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _mult_step(a, g, lr):
    z = np.log(np.maximum(a, 1e-30))
    z = z - lr * g
    z = np.clip(z, -100.0, 100.0)
    a_new = _softmax_stable(z)
    return a_new


def _grad_optimize(a_init, deadline, T_start=0.003, T_end=0.0001,
                   lr_start=0.05, lr_end=0.005, max_iter=200000,
                   eval_interval=200, active_k=3, patience_limit=6):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sn = _score_norm(a_init)
    t = 0
    patience = 0
    lr_mult = 1.0
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    plateau_frac = 0.25

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / max(1e-10, 1.0 - plateau_frac)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if T < T_start * 0.05:
            g = _active_peak_grad(a, c, active_k, decay=0.5)
        else:
            g = _soft_peak_grad(a, c, T)

        W = float(np.dot(a, g))
        g_centered = g - W
        if not np.all(np.isfinite(g_centered)):
            g_centered = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * g_centered
        v_buf = b2 * v_buf + (1 - b2) * (g_centered * g_centered)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 0.5:
            update = update * (0.5 / un)

        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if t % 100 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_sn * 1.02:
                z_new = z
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_sn:
                best_sn = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_limit and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sn:
        best_sn = sc_final
        best_a = a_final.copy()
    return best_a, best_sn


def _build_sparse_conv_matrix(a):
    n = len(a)
    row_idx = []
    col_idx = []
    val_idx = []
    for j in range(n):
        if a[j] > 1e-15:
            row_idx.extend(range(j, j + n))
            col_idx.extend([j] * n)
            val_idx.extend([a[j]] * n)
    C = coo_matrix((val_idx, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return C.tocsr()


def _lp_direction(a, M, max_lp_n=2048):
    n = len(a)
    if n <= max_lp_n:
        a_work = a
        work_n = n
    else:
        x0 = np.linspace(0, 1, n)
        x1 = np.linspace(0, 1, max_lp_n)
        a_work = np.interp(x1, x0, a)
        a_work = np.maximum(a_work, 0)
        s = a_work.sum()
        if s < 1e-15:
            return None
        a_work /= s
        work_n = max_lp_n

        c_work = _fft_convolve(a_work, a_work)
        M = float(c_work.max())

    C = _build_sparse_conv_matrix(a_work)
    c_obj = -np.ones(work_n)
    b_ub = np.full(2 * work_n - 1, M + 1e-10)
    bounds = [(0, None)] * work_n

    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60})
        if res.success:
            g = res.x
            if g is not None and g.sum() > 0:
                if work_n != n:
                    x0 = np.linspace(0, 1, work_n)
                    x1 = np.linspace(0, 1, n)
                    g = np.interp(x1, x0, g)
                    g = np.maximum(g, 0)
                    gs = g.sum()
                    if gs > 1e-15:
                        g /= gs
                    else:
                        return None
                return g
    except Exception:
        pass
    return None


def _lp_optimize(a_init, deadline, max_iter=20, step_decay=0.8, lp_n=2048):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sn = _score_norm(a)
    step = 0.3

    for it in range(max_iter):
        if time.monotonic() > deadline - 10:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        t0 = time.monotonic()
        g = _lp_direction(a, M, max_lp_n=lp_n)
        t_lp = time.monotonic() - t0
        if g is None:
            break
        g = g / g.sum()
        a_new = (1 - step) * a + step * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_sn:
            best_sn = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            step *= step_decay
            if step < 0.01:
                break
        print(f"  LP iter {it}: score={sc:.10f}, best={best_sn:.10f}, "
              f"step={step:.3f}, lp_time={t_lp:.1f}s")

    return best_a, best_sn


def _cd_polish(a_init, deadline, max_rounds=100):
    n = len(a_init)
    a = a_init.copy()
    best_sn = _score_norm(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for k in active:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue
            for j in js:
                for scale in [1.2, 1.5, 0.8, 0.6, 0.4]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(2.0 * n * c_cand.max())
                    if sc < best_sn * (1 - 1e-12):
                        best_sn = sc
                        a = a_cand
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return a, best_sn


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=3, decay=0.6,
                  eval_interval=200, clip_norm=0.5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    plateau_frac = 0.2

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / max(1e-10, 1.0 - plateau_frac)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if T < T_start * 0.02:
            g = _active_peak_grad(a, c, active_k, decay)
        else:
            g = _soft_peak_grad(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        if t % 50 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.015:
                z_new = z
        z = z_new
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _random_perturb_and_optimize(best_a, deadline, rng, n_attempts=3):
    n = len(best_a)
    best_score = _score_norm(best_a)
    best_out = best_a.copy()
    for att in range(n_attempts):
        if time.monotonic() > deadline - 30:
            break
        sigma = 0.03 + 0.02 * att
        z = np.log(np.maximum(best_a, 1e-15))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax_stable(z)
        c_p = _fft_convolve(a_pert, a_pert)
        max_cp = float(c_p.max())
        time_per = max(15.0, (deadline - 30 - time.monotonic()) / max(1, n_attempts - att))
        sub_dl = min(deadline - 30, time.monotonic() + time_per)
        a_opt, sc = _adam_z_space(a_pert, sub_dl,
                                   T_start=0.001 * max_cp,
                                   T_end=0.00001 * max_cp,
                                   lr_start=0.02, lr_end=0.002,
                                   max_iter=50000, eval_interval=100)
        full_sc = _score(a_opt)
        print(f"  Perturb[{att}] sigma={sigma:.3f}: score={full_sc:.10f}")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_out = a_opt.copy()
    return best_out, best_score
