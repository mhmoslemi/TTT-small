import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _hankel_diag(W, n):
    return W[:n].copy() + W[n:2 * n - 1] if len(W) > n else np.zeros(n)


def _cg_solve_precond(W, n, rhs, max_iter=500, tol=1e-13):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / (pHp + 1e-30)
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _active_set_expand(a_init, deadline, max_rounds=50):
    """Iteratively expand active set and re-solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Start with very tight active set
    frac = 0.001
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        n_active = max(2, int(frac * (2 * n - 1)))
        top_indices = np.argsort(c)[::-1][:n_active]
        active = np.zeros(2 * n - 1, dtype=bool)
        active[top_indices] = True

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve_precond(W, n, g, max_iter=200, tol=1e-12)
        H1 = _cg_solve_precond(W, n, np.ones(n), max_iter=200, tol=1e-12)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.dot(np.ones(n), Hg)) / float(np.dot(np.ones(n), H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            if float(np.linalg.norm(dx)) < 1e-30:
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-6:
                a = a_try
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Gradually expand active set
        if rnd < 20:
            frac = min(frac * 1.3, 0.05)

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = max(c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)], 1e-15)
    base_beta = max(gap * 0.3, max_c * 0.01)

    # 40-stage beta schedule (much more aggressive)
    ratios = []
    for i in range(40):
        ratios.append(base_beta * (0.7 ** i))
    # Ensure minimum beta
    min_beta = max_c * 1e-6
    ratios = [max(r, min_beta) for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_iter_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(30, int(remaining / (est_iter_time * 20))))

        try:
            start_i = time.monotonic()
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            elapsed = time.monotonic() - start_i
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 10 == 0:
                    print(f"    L-BFGS stage {i}/{len(ratios)}: beta={beta:.2e}, "
                          f"score={sc_new:.10f}, time={elapsed:.1f}s")
        except Exception:
            pass

    return current_a


def _fista_smooth(a_init, beta, deadline, max_iter=5000):
    """FISTA for the smoothed problem."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    t_k = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Estimate Lipschitz constant
    a_est = _softmax(z)
    g_est = _smograd(z, beta, n)
    L = 2.0 * np.linalg.norm(g_est) + n * beta * 0.1
    L = max(L, 1.0)

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        # Interpolation
        z_k = z + ((t_k - 1.0) / t_new) * (z - z_prev)

        # Gradient step
        g_k = _smograd(z_k, beta, n)
        z = z_k - g_k / L

        # Evaluate
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

        z_prev = z_k
        t_k = t_new

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.03
        return a / a.sum()
    elif kind == "parabolic":
        t = np.linspace(0, 1, n)
        a = t * (1 - t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * ((t - 0.5) ** 4))
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wiener":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 5.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == 1:
        return np.ones(n_new) / n_new
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _prune_and_reopt(a, deadline, min_time=60):
    """Zero out smallest entries and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.05, 0.1, 0.2, 0.3]:
        if time.monotonic() + min_time > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            # Re-optimize
            a_reopt = _run_lbfgs(best_a, min(deadline - min_time, time.monotonic() + 60),
                                 min_time_remaining=min_time)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient with Polyak step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    f_star = best_sc
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with tight tolerance
        tol = 0.0005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Polyak step size
        sc_cur = _score(a)
        gap = sc_cur - f_star
        if gap > 1e-12:
            step = gap / (grad_norm * grad_norm + 1e-30)
        else:
            step = 0.01 / (1.0 + 0.001 * i)
        step = min(step, 0.1)

        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            sc_check = _score(_softmax(z))
            if sc_check < f_star - 1e-13:
                f_star = sc_check
                best_sc = sc_check
                best_a = _softmax(z).copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining L-BFGS smoothing, active set Newton,
    subgradient, and pruning to minimize the convolution peak ratio.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Phase 0: Load parent or initialize
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Try multiple analytical profiles for better initialization
    print(f"\n[Phase 1] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["tapered", "cos", "parabolic", "super_gauss", "raised_cos2", "wiener", "bimodal"]
    best_profile = parent.copy()
    best_profile_sc = best_score
    for prof_kind in profiles:
        if time.monotonic() > deadline - 600:
            break
        a_p = _make_profile(n_parent, prof_kind)
        sc_p = _score(a_p)
        if sc_p < best_profile_sc:
            best_profile_sc = sc_p
            best_profile = a_p
            print(f"  Profile '{prof_kind}': score={sc_p:.10f}")
    update_best(best_profile, best_profile_sc, "best_profile")

    # Phase 2: Main L-BFGS optimization from best starting point
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    start_point = best_seq.copy() if best_seq is not None else parent
    a2 = _run_lbfgs(start_point, deadline, min_time_remaining=400.0)
    update_best(a2, _score(a2), "lbfgs_main")

    # Phase 3: Active set Newton polish (tight)
    print(f"\n[Phase 3] Newton tight (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a3, sc3 = _active_set_newton(best_seq, min(deadline - 300, time.monotonic() + 200),
                                     max_rounds=80, n_active_frac=0.003)
        update_best(a3, sc3, "newton_tight")

    # Phase 4: Active set with expansion
    print(f"\n[Phase 4] Newton expand (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _active_set_expand(best_seq, min(deadline - 250, time.monotonic() + 150),
                                     max_rounds=60)
        update_best(a4, sc4, "newton_expand")

    # Phase 5: Subgradient polish
    print(f"\n[Phase 5] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _subgrad_polish(best_seq, min(deadline - 200, time.monotonic() + 100),
                                  max_steps=2000)
        update_best(a5, sc5, "subgrad")

    # Phase 6: Another Newton after subgradient
    print(f"\n[Phase 6] Newton after SG (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a6, sc6 = _active_set_newton(best_seq, min(deadline - 150, time.monotonic() + 100),
                                     max_rounds=60, n_active_frac=0.002)
        update_best(a6, sc6, "newton_after_sg")

    # Phase 7: Pruning
    print(f"\n[Phase 7] Pruning (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a7, sc7 = _prune_and_reopt(best_seq, deadline, min_time=80)
        update_best(a7, sc7, "pruned")

    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [16384, 4096, 2048]:
        if time.monotonic() > deadline - 100:
            break
        attempt = _resample(best_seq if best_seq is not None else parent, n_try)
        a_n = _run_lbfgs(attempt, deadline, min_time_remaining=60)
        sc_n = _score(a_n)
        update_best(a_n, sc_n, f"n_{n_try}")

        if sc_n < best_score + 0.001 and time.monotonic() < deadline - 50:
            a_np, sc_np = _active_set_newton(a_n, min(deadline - 40, time.monotonic() + 40),
                                             max_rounds=40, n_active_frac=0.003)
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 9: Diverse restarts
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [
            ("perturb_01", _perturb(best_seq, 0.02, rng)),
            ("perturb_05", _perturb(best_seq, 0.05, rng)),
            ("perturb_10", _perturb(best_seq, 0.1, rng)),
        ]
        for label, start in restarts:
            if time.monotonic() > deadline - 25:
                break
            a_r = _run_lbfgs(start, deadline, min_time_remaining=15)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, label)
            if sc_r < best_score + 0.0005 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 30),
                                                 max_rounds=40, n_active_frac=0.002)
                update_best(a_rp, sc_rp, label + "_np")

    # Phase 10: Final ultra-fine polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1,
                                            max_rounds=80, n_active_frac=0.001)
        update_best(a_fin, sc_fin, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
