Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using exact subgradients
in log-space with Adam, adaptive peak selection, and coordinate descent polishing.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _exact_subgradient(a, c, top_k=10):
    """Compute exact subgradient of max(c) w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    # Find top-k peak positions
    k_indices = np.argsort(c)[::-1][:top_k]
    # Weighted: use softmax weighting over the top peaks
    c_top = c[k_indices]
    c_max = c_top[0]
    # Soft weighting: peaks closer to max get more weight
    weights = np.exp((c_top - c_max) / (c_max * 0.01 + 1e-15))
    weights /= weights.sum()

        m_vals = np.arange(m_start, m_end + 1)
        a_vals = a[k - m_vals]
        g[m_vals] += weights[i] * a_vals
    
    return g * 2.0  # Factor of 2 from the derivative

    W = float(np.dot(a, g_a))
    return a * (g_a - W)

        z_new = z - update
        z_new = np.clip(z_new, -100.0, 100.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        # Periodic evaluation and patience
        if t % 100 == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > 10 and lr_mult > 0.1:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _stochastic_subgradient(a, c, top_k=10, sample_frac=0.5, rng=None):
    """Stochastic version: randomly select a subset of top peaks."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    c_max = c[k_indices[0]]
    weights = np.exp((c[k_indices] - c_max) / (c_max * 0.01 + 1e-15))
    weights /= weights.sum()
    
    if sample_frac < 1.0 and rng is not None:
        n_sample = max(1, int(len(k_indices) * sample_frac))
        selected = rng.choice(len(k_indices), size=n_sample, replace=False)
    else:
        selected = np.arange(len(k_indices))
    
    g = np.zeros(n)
    w_sum = 0
    for i in selected:
        w = weights[i]
        k = k_indices[i]
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start > m_end:
            continue
        m_vals = np.arange(m_start, m_end + 1)
        g[m_vals] += w * a[k - m_vals]
        w_sum += w
    
    if w_sum < 1e-15:
        return g
    return g * 2.0 / w_sum

def _stochastic_adam(a_init, deadline, top_k=10, sample_frac=0.5,
                     lr_start=0.03, lr_end=0.001, max_iter=300000, seed=0):
    """Adam with stochastic subgradient for exploration."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        g_a = _stochastic_subgradient(a, c, top_k, sample_frac, rng)
        z_grad = _z_gradient(a, g_a)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if t % 200 == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _coord_descent_polish(a, deadline, max_rounds=500):
    """Coordinate descent: try scaling each element."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    # Only try coordinates that are near the peaks
    # Find which coordinates affect the peak most
    peak_positions = np.argsort(c)[::-1][:5]
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        # Focus on coordinates that contribute to the peaks
        candidate_coords = set()
        for k in peak_positions:
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            # Add a subset of coordinates
            if m_end - m_start < 50:
                candidate_coords.update(range(m_start, m_end + 1))
            else:
                # Sample
                candidate_coords.update(np.random.default_rng(round_num * 1000 + k).choice(
                    range(m_start, m_end + 1), size=50, replace=False))
        
        for j in sorted(candidate_coords):
            old_val = best_a[j]
            for scale in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5, 0.0]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale if scale > 0 else 0.0
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = 2.0 * n * float(c_cand.max())
                if sc < best_score * (1.0 - 1e-10):
                    best_score = sc
                    best_a = a_cand.copy()
                    improved = True
                    c = c_cand
                    break
            if improved:
                peak_positions = np.argsort(c)[::-1][:5]
                break
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)

def _freq_flatten(a, iterations=200, alpha_start=10.0, alpha_end=1.5):
    """Frequency domain flattening: compress dynamic range of |FFT(a)|^2."""
    n = len(a)
    a = a.copy() / a.sum()
    N = _next_pow2(2 * n - 1)
    
    for i in range(iterations):
        alpha = alpha_start + (alpha_end - alpha_start) * (i / max(1, iterations - 1))
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        # Compress: mag_new = mag^(1/alpha) * mag.mean()^(1-1/alpha)
        mag_mean = mag.mean()
        mag_new = (mag / mag_mean) ** (1.0 / alpha) * mag_mean
        # Keep the phase
        A_new = mag_new * (A / (mag + 1e-300))
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a_new / s
    
    return a

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
