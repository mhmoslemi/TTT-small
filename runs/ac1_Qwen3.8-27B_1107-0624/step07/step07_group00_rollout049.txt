            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 4 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _subgradient_descent(a_init, deadline, max_iter=50000, top_k=5, lr_init=0.05):
    """
    Subgradient descent using the exact structure ∂c[k]/∂a[i] = 2*a[k-i].
    Projects onto simplex after each step.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = lr_init
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        g = _compute_subgradient(a, c, top_k)
        
        # Line search: try a few step sizes
        best_local_lr = lr
        best_local_a = a
        best_local_sc = best_score
        for lr_try in [lr, lr * 0.7, lr * 0.5, lr * 0.3]:
            a_try = a - lr_try * g
            a_try = _project_simplex(a_try)
            sc_try = _score_norm(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_local_a = a_try
                best_local_lr = lr_try
                break
        
        if best_local_sc < best_score:
            best_score = best_local_sc
            best_a = best_local_a.copy()
            a = best_local_a
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-5:
                break
        
        if t % 100 == 0:
            # Periodic evaluation
            sc_cur = _score_norm(a)
            if sc_cur < best_score:
                best_score = sc_cur
                best_a = a.copy()
    
    return best_a, best_score

def _fienup_symmetric(a_init, deadline, max_iter=20000):
    """
    Fienup/IAA with symmetry constraint.
    """
    n = len(a_init)
    half = (n + 1) // 2
    
    # Make initial symmetric
    a_sym = (np.asarray(a_init, dtype=np.float64) + np.asarray(a_init, dtype=np.float64)[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a_init, _score_norm(a_init)
    a_sym = a_sym / s
    
    best_a = a_sym.copy()
    best_score = _score_norm(a_sym)
    n_fft = _next_pow2(2 * n)
    
    # Current power spectrum
    A = np.fft.fft(a_sym, n_fft)
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        power = np.abs(A) ** 2
        mean_nc = power[1:].mean()
        
        # Annealing parameter
        frac = min(1.0, t / max_iter)
        alpha = 0.05 + 0.9 * frac * frac  # quadratic annealing
        
        target_power = np.full(n_fft, mean_nc)
        target_power[0] = 1.0
        
        new_power = alpha * target_power + (1.0 - alpha) * power
        new_power = np.maximum(new_power, 1e-30)
        
        # Reconstruct with new amplitude, old phase
        A_new = np.sqrt(new_power) * np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.ifft(A_new, n_fft))[:n]
        
        # Enforce symmetry
        a_new = (a_new + a_new[::-1]) / 2.0
        
        # Clip and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a_sym = a_new
        A = np.fft.fft(a_sym, n_fft)
    
    return best_a, best_score
