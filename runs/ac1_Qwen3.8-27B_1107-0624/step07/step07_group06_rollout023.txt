import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_eval(z, beta, n, topk=None):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if topk is not None and topk < len(c):
        sorted_idx = np.argpartition(-c, topk)[:topk]
        c_use = c[sorted_idx]
    else:
        c_use = c
    shifted = (c_use - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max, c, c_max
    return 2.0 * n * (c_max + beta * np.log(s)), c, c_max


def _smograd_eval(z, beta, n, topk=None):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if topk is not None and topk < len(c):
        sorted_idx = np.argpartition(-c, topk)[:topk]
        c_use = c[sorted_idx]
        w = np.zeros_like(c)
    else:
        c_use = c
        w = np.zeros_like(c)
    shifted = (c_use - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w[int(np.argmax(c))] = 1.0
    else:
        if topk is not None and topk < len(c):
            w[sorted_idx] = e / s
        else:
            w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad


def _active_gradient(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / max(pH, 1e-30)
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active(a_init, deadline, max_iter=200, tol_adapt=True):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.005 if tol_adapt else 0.002

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: tighten as we converge
        if tol_adapt:
            tol_frac_cur = max(0.001, tol_frac * (0.95 ** it))
        else:
            tol_frac_cur = tol_frac
        tol = tol_frac_cur * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hess_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        # Newton with adaptive regularization
        mu = 1e-6 * c_max
        if mu < 1e-12:
            mu = 1e-12

        def hess_vec(v):
            return _hess_matvec(v, active, n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=300, tol=1e-12)
        dx = dx - np.mean(dx)

        # Damped Newton with backtracking
        f_old = float(c[active].sum())
        alpha = 1.0
        accepted = False
        a_new = a.copy()
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            if alpha < 1e-12:
                # Try subgradient step instead

                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
            break

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "center_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "oscillatory":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_init(n, rng, n_tries=20):
    """Try to find a sequence with flatter autocorrelation via spectral methods."""
    best_a = None
    best_sc = np.inf
    for _ in range(n_tries):
        # Random phase in frequency domain
        N_fft = _next_pow2(2 * n)
        freq = np.zeros(N_fft // 2 + 1)
        # Flat magnitude with random phase
        freq[1:-1] = rng.uniform(0.5, 1.5, N_fft // 2 - 1) * np.exp(1j * rng.uniform(0, 2 * np.pi, N_fft // 2 - 1))
        freq[0] = rng.uniform(0.5, 1.5)
        if N_fft % 2 == 0:
            freq[-1] = freq[1].real  # Real signal constraint
        a = np.fft.irfft(freq, N_fft)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _perturb(a, rng, mode=0):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    if mode == 0:
        z = z + 0.02 * rng.standard_normal(n)
    elif mode == 1:
        t = np.arange(n)
        for _ in range(3):
            freq = rng.integers(1, max(2, n // 4))
            amp = 0.02 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == 2:
        # Block swap
        bs = max(1, n // 8)
        i = rng.integers(0, n - 2 * bs)
        j = rng.integers(0, n - 2 * bs)
        if abs(i - j) > bs:
            tmp = z[i:i + bs].copy()
            z[i:i + bs] = z[j:j + bs]
            z[j:j + bs] = tmp
    z = np.clip(z, -50, 50)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try zeroing small elements, then re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15, 0.2, 0.3]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc

    if best_seq is not None and time.monotonic() < deadline - 500:
        a_newton, sc_newton = _newton_active(best_seq, min(deadline - 500, time.monotonic() + 300),
                                             max_iter=300, tol_adapt=True)
        update_best(a_newton, sc_newton, "newton_tight")

    # === Phase 3: Ultra-low beta L-BFGS ===
    print(f"\n[Phase 3] Ultra-low beta L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        # Compute very small beta based on current gap
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        c_max = float(c.max())
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[5] if len(c_sorted) > 5 else c_max * 0.001
        beta_ultra = max(gap * 0.001, c_max * 1e-6)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        remaining = deadline - 400 - time.monotonic()
        if remaining > 5:
            max_iter = min(5000, max(200, int(remaining / 0.01)))
            try:
                def fun_ultra(z_local):
                    val, _, _ = _smoobj_eval(z_local, beta_ultra, n)
                    return val

                def jac_ultra(z_local):
                    return _smograd_eval(z_local, beta_ultra, n)

                result = minimize(
                    fun_ultra, z, jac=jac_ultra, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 100, 'maxls': 300, 'maxfun': max_iter * 3}
                )
                a_ultra = _softmax(result.x)
                update_best(a_ultra, _score(a_ultra), "ultra_lbfgs")
            except Exception:
                pass

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 300, time.monotonic() + 200),
                                      n_max_steps=3000, lr_init=0.005)
        update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 5: Newton again on best ===
    print(f"\n[Phase 5] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_n2, sc_n2 = _newton_active(best_seq, min(deadline - 250, time.monotonic() + 150),
                                     max_iter=200, tol_adapt=False)
        update_best(a_n2, sc_n2, "newton_polish")

    # === Phase 6: Shrinkage ===
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 200)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score + 1e-6:
            # Polish the shrunk version
            a_shr2 = _run_lbfgs_adaptive(a_shr, deadline, min_time_remaining=150)
            update_best(a_shr2, _score(a_shr2), "shrink_lbfgs")
            if time.monotonic() < deadline - 100:
                a_shr3, sc_shr3 = _newton_active(a_shr2, min(deadline - 100, time.monotonic() + 80),
                                                  max_iter=100, tol_adapt=True)
                update_best(a_shr3, sc_shr3, "shrink_newton")

    # === Phase 7: Multi-n exploration (try larger n) ===
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 12288, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - len(best_seq)))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        else:
            a_init = best_seq.copy()

        # Quick L-BFGS
        time_alloc = min(120, deadline - 80 - time.monotonic())
        if time_alloc > 10:
            a_opt = _run_lbfgs_adaptive(a_init, time.monotonic() + time_alloc, min_time_remaining=10)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")

            # Newton if competitive
            if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 60:
                a_opt2, sc_opt2 = _newton_active(a_opt, min(deadline - 60, time.monotonic() + 50),
                                                 max_iter=80, tol_adapt=True)
                update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_restarts = 6
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, rng, mode=i % 3)
            time_alloc = min(40, deadline - 25 - time.monotonic())
            if time_alloc > 5:
                a_r = _run_lbfgs_adaptive(p_start, time.monotonic() + time_alloc, min_time_remaining=5)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{i}")
                if sc_r < best_score + 0.01 and time.monotonic() < deadline - 20:
                    a_r2, sc_r2 = _newton_active(a_r, min(deadline - 20, time.monotonic() + 15),
                                                 max_iter=50, tol_adapt=True)
                    update_best(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 9: Final combined polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Subgradient with very small steps
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 5, n_max_steps=3000, lr_init=0.003)
        update_best(a_fin, sc_fin, "final_sg")

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active(best_seq, max(deadline - 2, time.monotonic()),
                                             max_iter=100, tol_adapt=True)
            update_best(a_fin2, sc_fin2, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence confirms: {sc_eval:.10f}")
                if sc_eval < sc_final:
                    sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
