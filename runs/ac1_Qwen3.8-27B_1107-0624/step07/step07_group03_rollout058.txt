"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with log-sum-exp smoothing (adaptive beta)
2. Active-set Newton with KKT-based constraint management
3. Multi-n exploration with interpolation
4. Diverse restarts with penalty oscillation
5. Final tight polish with very small tolerances

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hess_matvec(v, A_vec, m, mu):
    """Compute (H + mu*I) * v where H = (2/m) * sum_{k in S} M_k."""
    n = len(v)
    flip_v = v[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(flip_v, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fv * fA, N)[:out_len]
    h = result[n - 1:2 * n - 1]
    return (2.0 / m) * h + mu * v


def _grad_active(a, A_vec, m, n):
    """Gradient of (1/m) * sum_{k in S} conv[k] with respect to a."""
    a_rev = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a_rev, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    h = result[n - 1:2 * n - 1]
    W = float(np.dot(A_vec.astype(np.float64) / m, _full_convolve(a, a)))
    return (2.0 / m) * h - (2.0 / m) * W * a


def _active_set_newton(a_init, deadline, max_iter=500, tol_start=0.005, tol_end=0.0001):
    """Active-set Newton with KKT-based constraint management."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    mu_base = 1e-8
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive tolerance
        tol_f = tol_start + (tol_end - tol_start) * min(1.0, it / (max_iter * 0.7))
        tol_add = tol_f
        tol_remove = tol_f * 0.3

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify active set
        active = c >= (c_max - tol_add * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n

        A_vec = active.astype(np.float64)
        m = n_active

        # Compute gradient of active-set objective
        g = _grad_active(a, A_vec, m, n)
        # Project to sum=0 subspace
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Regularization for CG
        mu = mu_base * (c_max / max(m, 1)) + 1e-10

        # CG solve for H*u = g_t and H*w = 1
        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - _hess_matvec(x, A_vec, m, mu)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(max_cg):
                Ap = _hess_matvec(p, A_vec, m, mu)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(rhs)):
                    break
                if rs < 1e-30:
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u - mu * np.zeros(n)
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Remove mu contribution from the direction (it was just for CG stability)
        dx = dx  # The mu*v term in hess_matvec acts as regularization; direction is fine

        # Line search with non-negativity
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)

        for ls in range(50):
            a_try = a + t * dx
            if np.any(a_try < -1e-10):
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent step
            t = min(0.5, t_max)
            for ls in range(30):
                a_try = a - t * g_t / (gnorm + 1e-30)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # KKT convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13 * c_max:
            break

        # Check for stagnation
        if it > 50 and abs(float(c_new.max()) - c_max) < 1e-15 * c_max:
            break

    return best_a, best_sc


def _lsobj_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                  0.015, 0.0075, 0.0037, 0.00185, 0.0009,
                                  0.00045, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time_per_iter * 5))))

        try:
            result = minimize(
                _lsobj_fn, z, args=(beta, n),
                jac=_lsgrad_fn,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=10000, lr_init=0.01):
    """Subgradient method with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))

    best_a = a_init.copy()

    patience = 0
    best_patience_lr = lr

    for i in range(n_max_steps):

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        m = active.sum()
        w = np.zeros_like(c)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + 0.0001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)

        if i % 50 == 0:

                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2) + 1e-10
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage(a, deadline):
    """Try zeroing out small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.02, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc

                a_opt, min(deadline - 50, time.monotonic() + 80),
                max_iter=300, tol_start=0.002, tol_end=0.0002)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 4: Diverse restarts
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_configs = [
            ("small_pert", 0.02),
            ("med_pert", 0.05),
            ("large_pert", 0.10),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _active_set_newton(
                    a_r, min(deadline - 10, time.monotonic() + 40),
                    max_iter=200, tol_start=0.001, tol_end=0.0001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

        # Fourier perturb
        if time.monotonic() > deadline - 80:
            pass
        else:
            p_fourier = _perturb_fourier(best_seq, rng, n_modes=4)
            a_f = _run_lbfgs(p_fourier, deadline, min_time_remaining=20.0)
            update_best(a_f, _score(a_f), "restart_fourier")

    # Phase 5: Final polish
    print(f"\n[Phase 5] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        as_deadline_final = min(deadline - 3, time.monotonic() + 120)
        a_fin, sc_fin = _active_set_newton(
            best_seq, as_deadline_final,
            max_iter=500, tol_start=0.0005, tol_end=0.00005)
        update_best(a_fin, sc_fin, "final_as_tight")

        # Tiny-beta L-BFGS
        if time.monotonic() < deadline - 5:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj_fn, z, args=(beta_final, n),
                        jac=_lsgrad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 70, 'maxls': 300}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # Final subgradient with tiny lr
        if time.monotonic() < deadline - 3:
            a_fin4, sc_fin4 = _subgrad_polish(
                best_seq, deadline - 2, n_max_steps=2000, lr_init=0.003)
            update_best(a_fin4, sc_fin4, "final_sg_tiny")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(1024, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
