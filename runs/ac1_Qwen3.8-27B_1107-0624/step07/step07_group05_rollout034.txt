Corrected multi-stage optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key fix: correct gradient computation (includes both f1 and f2 convolution terms).
Uses L-BFGS with log-sum-exp, FISTA with active-set subgradient, symmetrization,
and coordinate descent polish. Multi-length exploration with multiple restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _npow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _npow2(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * _conv(a, a).max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / z.size
    return e / s


def _grad_a(a, w, n):
    """Compute ga[i] = sum_k w[k]*(a[k-i] + a[i-k]) for i=0..n-1 (correct gradient)."""
    w_r = w[::-1]
    f1 = _conv(w_r, a)[n - 1:2 * n - 1]
    f2 = _conv(w, a)[0:n]
    return f1 + f2


def _lss_obj(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _lss_jac(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    ga = _grad_a(a, w, n)
    return 2.0 * n * a * (ga - np.dot(a, ga))


def _active_grad_a(a, c, n, tol_frac=0.005, max_active=20):
    """Compute gradient using top-K active lags (equal weight)."""
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * cm
    active = c >= (cm - tol)
    n_act = active.sum()
    if n_act > max_active:
        idx = np.argsort(c)[-max_active:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_act = max_active
    if n_act < 1:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_act = 1
    w = np.zeros_like(c)
    w[active] = 1.0 / n_act
    return _grad_a(a, w, n)


def _lbfgs_stage(a_init, deadline, min_remaining=30.0):

def _fista_stage(a_init, deadline, n_iters=3000, tol_frac=0.003):
    """FISTA with active-set subgradient and simplex projection."""
    n = len(a_init)
    x = a_init.copy()
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-15:
        return a_init
    x = x / s
    y = x.copy()
    t = 1.0
    best_a = x.copy()
    best_sc = _score(x)
    step_base = 1.0 / (4.0 * n * max(1e-10, x.max()) * (2 * n))

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _conv(y, y)
        ga = _active_grad_a(y, c, n, tol_frac=tol_frac, max_active=15)
        g = 2.0 * n * ga
        step = step_base / (1.0 + 0.001 * it)
        z = y - step * g
        # Project onto simplex
        z = z - z.max()
        # Soft threshold for projection to {sum=1, a>=0}
        lam = np.sort(z)[::-1]
        nnz = np.arange(1, z.size + 1)
        rho = np.where(lam - (z.sum() - np.cumsum(lam)) / nnz > 0)[0]
        if len(rho) == 0:
            theta = 1.0 / n
        else:
            theta = (z.sum() - np.cumsum(lam)[rho[-1]]) / (rho[-1] + 1)
        x_new = np.maximum(z - theta, 0)
        # Ensure sum = 1
        ss = x_new.sum()
        if ss < 1e-15:
            x_new = np.ones(n) / n
        else:
            x_new = x_new / ss

        sc = _score(x_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = x_new.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = best_a.copy()
        else:
            y = y / ys
        x = x_new
        t = t_new

    return best_a
