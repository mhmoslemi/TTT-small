    N = 1
    while N < n:
        N <<= 1

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: Adaptive mirror descent with progressive active set,
aggressive L-BFGS beta continuation, Fourier-domain simulated annealing,
and n-adaptive search with smart time allocation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N <<= 1
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _sls_obj(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _sls_grad(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _compute_active_grad(c, n, tol_frac=0.01):
    """Compute subgradient using active set (top elements within tolerance)."""
    a = None  # will be passed
    c_max = c.max()
    if c_max < 1e-30:
        return None
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() < 2:
        idx = np.argsort(c)[-3:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    return active


def _mirror_descent(a_init, deadline, n_iters=500, alpha=1.0, tol_frac=0.01):
    """Adaptive mirror descent with progressive active set."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_conv(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break

        # Adaptive active set: expand as we converge
        effective_tol = tol_frac * max(1.0, c_max / (1.0 / n))
        # Start strict, loosen over time
        cur_tol = (tol_frac / (1.0 + 0.01 * it)) * c_max
        active = c >= (c_max - cur_tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-max(3, int(0.01 * len(c))):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Gradient: g_i = 2 * sum_{k in active} a[k-i]
        S = active.astype(np.float64)
        a_rev = a[::-1]
        # h[i] = sum_k S[k] * a[k-i]
        wc = _fft_conv(S, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h

        # Mirror descent update with adaptive step size
        g_range = g.max() - g.min()
        if g_range < 1e-15:
            break
        eta = alpha / g_range
        # Clamp to avoid overflow
        log_update = -eta * (g - g.mean())
        log_update = np.clip(log_update, -20, 20)
        a = a * np.exp(log_update)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_remaining=30.0):
    """L-BFGS with aggressive beta continuation on softmax parameterization."""
    n = len(a_init)
    c_cur = _fft_conv(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Estimate initial beta from the gap
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 1.0, max_c * 0.05)

    # Very aggressive beta schedule with many small values
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.001953125,
        0.0009765625, 0.00048828125, 0.000244140625,
        0.0001220703125, 0.00006103515625, 0.000030517578125
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break

        # Estimate time per iteration
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _sls_obj, z, args=(beta, n),
                jac=_sls_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _fourier_anneal(a_init, deadline, n_steps=2000, T_init=0.005, T_min=0.0001):
    """Simulated annealing in Fourier domain."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Temperature schedule
        frac = step / max(1, n_steps)
        T = T_init * (T_min / T_init) ** frac

        # Fourier-domain perturbation
        z_hat = np.fft.rfft(z)
        n_perturb = max(1, int(n * 0.01 * (1 + frac)))  # more high-freq as T decreases
        for _ in range(min(n_perturb, n // 2)):
            if np.random.rand() < 0.5:
                freq = rng.integers(1, max(2, n // 4))  # low freq
            else:
                freq = rng.integers(n // 4, max(n // 2 + 1, n // 4 + 1))  # high freq
            if freq < len(z_hat):
                amp = T * 0.5 * rng.uniform(0.5, 2.0)
                delta = amp * rng.standard_normal()
                phase = rng.uniform(0, 2 * np.pi)
                z_hat[freq] += delta * np.exp(1j * phase)
                if freq < len(z_hat) - 1:
                    z_hat[-freq] = np.conj(z_hat[freq])
                elif freq > 0:
                    z_hat[freq] = np.conj(z_hat[freq])

        z_new = np.fft.irfft(z_hat, n)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)

        sc_new = _score(a_new)

        # Metropolis acceptance
        delta = sc_new - best_sc
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            z = z_new
            a = a_new
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()

    return best_a, best_sc


def _coord_descent(a_init, deadline, n_rounds=200):
    """Coordinate descent trying to reduce each coordinate."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_conv(a, a)
        c_max = c.max()
        # Focus on coordinates that contribute most to peak
        active = c >= (c_max - 0.05 * c_max)
        S = active.astype(np.float64)
        a_rev = a[::-1]
        wc = _fft_conv(S, a_rev)
        contrib = wc[n - 1:2 * n - 1] * a

        n_try = min(n, max(5, n // 100))
        sorted_idx = np.argsort(contrib)[::-1]
        # Mix top contributors with random
        top = sorted_idx[:n_try // 2]
        rand_idx = rng.integers(0, n, n_try - len(top))
        candidates = np.concatenate([top, rand_idx])
        rng.shuffle(candidates)

        improved = False
        for i in candidates:
            if time.monotonic() > deadline:
                break
            old = a[i]
            if old < 1e-15:
                # Try setting to small positive value
                for val in [1.0 / n, 0.5 / n]:
                    a[i] = val
                    s = a.sum()
                    if s > 0:
                        new_a = a / s
                        if _score_norm(new_a) < _score_norm(a / a.sum()):
                            improved = True
                            break
                if not improved:
                    a[i] = old
                continue

            # Try various multiplicative factors
            for factor in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
                a[i] = old * factor
                if a[i] < 0:
                    a[i] = old
                    break
                s = a.sum()
                if s < 1e-15:
                    a[i] = old
                    break
                a_norm = a / s
                if _score_norm(a_norm) < _score_norm(a_init if False else a / s):
                    improved = True
                    break
                a[i] = old

            if improved:
                s = a.sum()
                if s > 0:
                    a = a / s
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 - 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        t = np.linspace(0, 1, n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = 0.1 * n
        for i in range(int(taper)):
            a[i] *= 0.5 * (1 + np.cos(np.pi * i / taper))
            a[n-1-i] *= 0.5 * (1 + np.cos(np.pi * i / taper))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_sequence(a_src, n_dst):
    """Interpolate a sequence to a new length."""
    n_src = len(a_src)
    if n_src == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = a_dst.sum()
    if s > 1e-15:
        a_dst = a_dst / s
    else:
        a_dst = np.ones(n_dst) / n_dst
    return a_dst


def _multi_stage(a_init, deadline, min_remaining=30.0):
    """Run full optimization pipeline on a single sequence."""
    n = len(a_init)
    current = a_init.copy()
    current_sc = _score(current)

    # Stage 1: Mirror descent for good starting point
    t_mid = time.monotonic() + (deadline - time.monotonic() - min_remaining) * 0.2
    a1, sc1 = _mirror_descent(current, min(t_mid, time.monotonic() + 60),
                              n_iters=300, alpha=1.0)
    sc1_full = _score(a1)
    if sc1_full < current_sc:
        current = a1
        current_sc = sc1_full

    # Stage 2: L-BFGS (main workhorse)
    t_end_lbfgs = min(deadline - min_remaining, time.monotonic() + 300)
    a2 = _lbfgs_optimize(current, t_end_lbfgs + 60, min_remaining=min_remaining)
    sc2 = _score(a2)
    if sc2 < current_sc:
        current = a2
        current_sc = sc2

    # Stage 3: Mirror descent again (to escape potential L-BFGS local min)
    if time.monotonic() < deadline - min_remaining - 30:
        t_mid2 = time.monotonic() + 30
        a3, sc3 = _mirror_descent(current, t_mid2, n_iters=100, alpha=0.5)
        sc3_full = _score(a3)
        if sc3_full < current_sc:
            current = a3
            current_sc = sc3_full
            # Follow up with L-BFGS
            if time.monotonic() < deadline - min_remaining:
                a3b = _lbfgs_optimize(current, deadline - min_remaining,
                                      min_remaining=10)
                sc3b = _score(a3b)
                if sc3b < current_sc:
                    current = a3b
                    current_sc = sc3b

    return current, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining mirror descent, L-BFGS with aggressive
    beta continuation, Fourier annealing, and n-adaptive exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # ============ Phase 1: Intensive optimization on parent ============
    print(f"\n[Phase 1] Intensive optimization on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    min_rem_p1 = max(600, (deadline - t_start) * 0.45)
    t_phase1_end = min(deadline - min_rem_p1, time.monotonic() + 600)
    a_p1, sc_p1 = _multi_stage(parent, t_phase1_end, min_remaining=30)
    _update(a_p1, sc_p1)

    # Additional L-BFGS pass with tighter tolerances
    if best_seq is not None and time.monotonic() < deadline - 500:
        n_b = len(best_seq)
        print(f"  Additional L-BFGS pass (t={time.monotonic()-t_start:.1f}s)")
        a_extra = _lbfgs_optimize(best_seq, deadline - 500, min_remaining=400)
        _update(a_extra, _score(a_extra))

    # ============ Phase 2: n-exploration ============
    print(f"\n[Phase 2] n-exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 12288, 16384, 20480, 24576]
    n_candidates = [x for x in n_candidates if x != n_parent]

    # Sort by proximity to parent n (likely similar quality)
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - 200 - time.monotonic()
        if remaining < 60:
            break

        # Interpolate parent to new n
        a_init = _interpolate_sequence(parent, n_try)

        # Quick optimization
        min_rem = max(30, remaining * 0.2)
        t_end = min(deadline - 100, time.monotonic() + remaining * 0.4)
        a_opt, sc_opt = _multi_stage(a_init, t_end, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # ============ Phase 3: Fourier annealing for global exploration ============
    print(f"\n[Phase 3] Fourier annealing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        # Multiple annealing runs with different temperatures
        for temp_mult in [1.0, 0.5, 2.0, 0.25]:
            if time.monotonic() > deadline - 50:
                break
            t_end = min(deadline - 40, time.monotonic() + 120)
            a_ann, sc_ann = _fourier_anneal(best_seq, t_end,
                                            n_steps=1000,
                                            T_init=0.003 * temp_mult,
                                            T_min=0.0001 * temp_mult)
            _update(a_ann, sc_ann)
            # Follow up with L-BFGS if improved
            if sc_ann < best_score + 1e-8:
                if time.monotonic() < deadline - 30:
                    a_f = _lbfgs_optimize(a_ann, deadline - 20, min_remaining=10)
                    _update(a_f, _score(a_f))

    # ============ Phase 4: Diverse restarts ============
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_b = len(best_seq)
        profiles = ["hann", "hamming", "chirp", "flat_taper", "raised_cos"]
        for prof in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_prof = _make_profile(n_b, prof)
            # Blend with best
            for blend in [0.3, 0.5, 0.7]:
                if time.monotonic() > deadline - 20:
                    break
                a_blend = blend * best_seq + (1 - blend) * a_prof
                a_blend = np.maximum(a_blend, 0)
                s = a_blend.sum()
                if s > 0:
                    a_blend = a_blend / s
                # Quick optimize
                t_end = min(deadline - 15, time.monotonic() + 40)
                a_r, sc_r = _multi_stage(a_blend, t_end, min_remaining=5)
                _update(a_r, sc_r)

    # ============ Phase 5: Coordinate descent polish ============
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_end = min(deadline - 10, time.monotonic() + 120)
        a_cd, sc_cd = _coord_descent(best_seq, t_end, n_rounds=300)
        _update(a_cd, sc_cd)
        if sc_cd < best_score:
            a_f = _lbfgs_optimize(a_cd, deadline - 5, min_remaining=2)
            _update(a_f, _score(a_f))

    # ============ Phase 6: Final intensive L-BFGS ============
    print(f"\n[Phase 6] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _lbfgs_optimize(best_seq, deadline - 1, min_remaining=1)
        _update(a_fin, _score(a_fin))

        if time.monotonic() < deadline - 1:
            # One more mirror descent + L-BFGS cycle
            a_md, _ = _mirror_descent(best_seq, min(deadline - 1, time.monotonic() + 20),
                                      n_iters=50, alpha=0.3)
            sc_md = _score(a_md)
            if sc_md < best_score:
                a_fin2 = _lbfgs_optimize(a_md, deadline - 0.5, min_remaining=0.5)
                _update(a_fin2, _score(a_fin2))

    # ============ Final validation ============
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        final_sc = _score(final)
        print(f"FINAL: score={final_sc:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
