"""
Multi-strategy optimizer combining multiplicative mirror descent, top-K peak shaving,
spectral shaping, and aggressive L-BFGS with diverse initializations to minimize
the peak autocorrelation of non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_z(z, beta, n):
    """Smoothed max objective in log-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_z(z, beta, n):
    """Gradient of smoothed max in log-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    r = np.fft.irfft(fa * fw, N)
    h = r[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    return grad_a / np.maximum(a, 1e-15)


def _topk_weights(c, K, beta_soft):
    """Compute soft weights for top-K elements of c."""
    if K >= len(c):
        c_max = c.max()
        shifted = (c - c_max) / beta_soft
        e = np.exp(np.clip(shifted, -700, 0))
        return e / e.sum()
    idx = np.argpartition(c, -K)[-K:]
    max_c = float(c.max())
    c_top = c[idx]
    shifted = (c_top - max_c) / beta_soft
    e_top = np.exp(np.clip(shifted, -700, 0))
    w = np.zeros_like(c)
    w[idx] = e_top / e_top.sum()
    return w


def _grad_topk(a, w, n):
    """Gradient of sum w[m]*c[m] w.r.t. a, then projected to tangent space."""
    a_rev = a[::-1]
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    r = np.fft.irfft(fa * fw, N)
    h = r[n - 1:2 * n - 1]
    c = _full_convolve(a, a)
    W = float(np.dot(w, c))
    grad_a = 2.0 * a * (h - W) * 2.0  # factor 2 for derivative of c=a*a, factor 2 for scaling
    # Actually let me be precise: F = sum_m w[m] * c[m], c[m] = sum_j a[j]*a[m-j]
    # dF/da[i] = 2 * sum_m w[m] * a[m-i] * I(valid) = 2 * h[i] where h[i] = sum_m w[m]*a[m-i]
    # But we also need to account for the constraint sum(a)=1, so we subtract the mean.
    # Let me recompute: h[i] = sum_j a[j] * w[i+j] = (a_rev * w)[n-1+i]
    # So dF/da[i] = 2 * h[i]
    # Projected: 2*h[i] - mean(2*h)
    return 2.0 * (h - np.mean(h))


def _mirror_descent(a_init, deadline, n_K_schedule, beta_start, beta_end, lr_init, n_max_iter):
    """Multiplicative mirror descent with top-K peak shaving."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive K: start small, grow
        frac = n_K_schedule[min(it, n_K_schedule.size - 1)]
        K = max(1, int(n * frac))
        # Adaptive beta_soft for the top-K weighting
        beta_soft = max(c_max * 0.01, 1e-10)
        
        w = _topk_weights(c, K, beta_soft)
        grad = _grad_topk(a, w, n)
        
        # Nesterov-like momentum
        momentum = mu * momentum + (1 - mu) * grad
        momentum = momentum - np.mean(momentum)
        
        # Multiplicative update
        step = lr / (1.0 + 0.001 * it)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - step * momentum
        a_new = _softmax(log_a_new)
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, lr_init)
        else:
            # Backtrack
            lr *= 0.85
            a = a  # keep old
            if lr < lr_init * 0.01:
                # Reset from best
                a = best_a.copy()
                lr = lr_init * 0.5
                momentum = np.zeros(n)

        if it % 100 == 0:
            print(f"    MD iter {it}: score={_score(a):.10f}, best={best_sc:.10f}, lr={lr:.6f}")

    return best_a, best_sc


def _lbfgs_phase(a_init, deadline, min_time_remaining, betas):
    """L-BFGS with given beta schedule in log-space."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 6))))

        def fun_local(z_local):
            return _smoobj_z(z_local, beta, n)

        def jac_local(z_local):
            return _smograd_z(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxcor': 50, 'maxls': 80, 'maxfun': max_iter * 2}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception as e:
            print(f"    L-BFGS beta={beta:.2e} failed: {e}")

    return current_a, current_sc

def _make_init(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian_abs":
        a = np.abs(rng.standard_normal(n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.7 * np.sin(2.0 * np.pi * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        a = rng.standard_normal(n)
        # Smooth with a moving average
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

            update_best(a_md2, sc_md2, "md_then_lbfgs")

    # === Phase 3: Variance minimization ===
    print(f"\n[Phase 3] Variance minimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        v_deadline = min(deadline - 350, time.monotonic() + 120)
        a_var, sc_var = _variance_minimize(best_seq, v_deadline, n_steps=300)
        update_best(a_var, sc_var, "variance_min")
        if sc_var < best_score * 1.001:
            a_var2, sc_var2 = _lbfgs_phase(a_var, deadline, min_time_remaining=250,
                                            betas=np.array([0.001, 0.0001, 0.00005]))
            update_best(a_var2, sc_var2, "var_then_lbfgs")

    # === Phase 4: Newton polish ===
    print(f"\n[Phase 4] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        nw_deadline = min(deadline - 200, time.monotonic() + 150)
        a_nw, sc_nw = _newton_polish(best_seq, nw_deadline, max_iter=200)
        update_best(a_nw, sc_nw, "newton")

    # === Phase 5: Diverse restarts ===
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else 8192
    restart_kinds = ["gaussian_abs", "random_smooth", "wheatstone", "tapered"]
    for i, kind in enumerate(restart_kinds):
        if time.monotonic() > deadline - 100:
            break
        a_init_r = _make_init(n_best, kind, rng)
        # Blend with best
        a_init_r = 0.5 * a_init_r + 0.5 * best_seq
        a_init_r = a_init_r / a_init_r.sum()
        
        betas_r = np.array([0.005, 0.002, 0.001, 0.0005])
        a_r, sc_r = _lbfgs_phase(a_init_r, deadline, min_time_remaining=60, betas=betas_r)
        update_best(a_r, sc_r, f"restart_{kind}")

    # === Phase 6: Try different n ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_options = sorted(set([n_best // 2, n_best, n_best * 2]), key=lambda x: abs(x - n_best))
    for n_try in n_options:
        if time.monotonic() > deadline - 80:
            break
        if n_try < 512 or n_try > 32768:
            continue
        if len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init_n = np.interp(x_new, x_old, best_seq)
            a_init_n = np.maximum(a_init_n, 0)
            a_init_n = a_init_n / a_init_n.sum()
        else:
            a_init_n = best_seq.copy()
        
        betas_n = np.array([0.003, 0.001, 0.0005, 0.0001])
        a_n, sc_n = _lbfgs_phase(a_init_n, deadline, min_time_remaining=40, betas=betas_n)
        update_best(a_n, sc_n, f"n_{n_try}")

    # === Phase 7: Aggressive final polish ===
    print(f"\n[Phase 7] Final aggressive polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)
        # Very low beta L-BFGS
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00001
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(50, int(remaining / 0.01)))
            try:
                result = minimize(
                    lambda z_l: _smoobj_z(z_l, beta_final, n), z,
                    jac=lambda z_l: _smograd_z(z_l, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                             'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _newton_polish(best_seq, deadline - 5, max_iter=50)
            update_best(a_fin2, sc_fin2, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence confirms: {sc_eval:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
