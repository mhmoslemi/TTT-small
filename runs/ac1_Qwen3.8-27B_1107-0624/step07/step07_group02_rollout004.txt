import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    # Clean up numerical noise
    r[np.abs(r) < 1e-18] = 0.0
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        if momentum:
            vel = mu * vel + (1 - mu) * grad_t
            step_dir = vel
            dir_norm = np.linalg.norm(step_dir)
            if dir_norm < 1e-16:
                step_dir = grad_t
                dir_norm = gnorm
        else:
            step_dir = grad_t
            dir_norm = gnorm

        alpha = min(lr, 0.5 / (dir_norm + 1e-15))
        improved = False

        # Line search with backtracking
        for _ in range(80):
            a_try = a - alpha * step_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.45

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
                if momentum:
                    vel *= 0.5
            if no_improve >= max_stall:
                a = best_a.copy()
                c = _full_convolve(a, a)
                c_max = float(c.max())
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.4
                if momentum:
                    vel = np.zeros(n)
                    mu = 0.5
                no_improve = 0
                stall_count += 1
                if stall_count > 12:
                    break
                # Try a slightly different active set width
                tol_frac = min(0.02, tol_frac * 2.0)
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

        iter_count += 1

    return best_a, _score(best_a)


def _coordinate_descent_polish(a_init, deadline, n_trials=2000):
    """Try small perturbations on individual coordinates to reduce max conv."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Identify important elements: those with larger values
    # and those near the "hot" lags
    hot_lags = np.argsort(c)[::-1][:50]

    # Elements that contribute to hot lags
    important_indices = set()
    for k in hot_lags[:20]:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                important_indices.add(i)

    important_indices = list(important_indices)
    if not important_indices:
        important_indices = list(range(n))
    rng = np.random.default_rng(12345)

    trials_per_elem = max(1, n_trials // max(1, len(important_indices)))

    for idx in important_indices:
        if time.monotonic() > deadline:
            break
        original_val = a[idx]
        if original_val < 1e-10:
            continue

        for trial in range(min(trials_per_elem, 30)):
            if time.monotonic() > deadline:
                break
            # Try relative perturbations
            delta = original_val * rng.uniform(-0.05, 0.05)
            a_try = a.copy()
            a_try[idx] = max(0.0, original_val + delta)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                c = c_try
                c_max = float(c.max())

    return best_a, best_sc


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 6)))
        amp = 0.02 * rng.uniform(0.5, 2.5)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _targeted_lag_reduction(a, deadline, n_lags=10, n_steps=200):
    """Directly target the hottest lags for reduction."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find top n_lags hottest positions
        top_idx = np.argsort(c)[::-1][:n_lags]
        # For each hot lag, compute which a[i] to reduce
        # C[k] = sum_i a[i]*a[k-i], dC[k]/da[i] = a[k-i] (if valid) + a[i] (if k-i valid)
        # To reduce C[k], decrease a[i] where a[k-i] is large, or a[k-i] where a[i] is large

        # Build a combined "pressure" vector: for each element, sum of contributions to hot lags
        pressure = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Step: reduce elements with high pressure
        # But ensure we don't make elements negative
        step_size = 0.01 * (c_max / a.max() if a.max() > 0 else 0.01)
        a_new = a - step_size * pressure * a
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * len(a_new) * float(c_new.max())

        if sc_new < current_sc - 1e-15:
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - 0.3 * step_size * pressure * a
            a_new2 = np.maximum(a_new2, 0.0)
            s_new2 = a_new2.sum()
            if s_new2 < 1e-15:
                break
            a_new2 = a_new2 / s_new2
            c_new2 = _full_convolve(a_new2, a_new2)
            sc_new2 = 2.0 * len(a_new2) * float(c_new2.max())
            if sc_new2 < current_sc - 1e-15:
                a = a_new2
            else:
                # Decrease step size
                step_size *= 0.5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase refined search for minimizing the convolution peak ratio."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main optimization (50% of budget) ===
    p1_deadline = t_start + (deadline - t_start) * 0.45
    print(f"\n[P1] L-BFGS main (budget ends at t={p1_deadline - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, p1_deadline, min_time_remaining=20.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  L-BFGS: {sc_main:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish coarse (15% of budget) ===
    p2_deadline = t_start + (deadline - t_start) * 0.60
    print(f"\n[P2] Active-set coarse (budget ends at t={p2_deadline - t_start:.1f}s)")
    if time.monotonic() < p2_deadline - 10:
        a_pol, sc_pol = _active_set_polish(best_seq, p2_deadline,
                                           tol_frac_start=0.006, tol_frac_end=0.0001,
                                           lr_init=0.03)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish_coarse] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Coarse polish: {sc_pol:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set polish fine (15% of budget) ===
    p3_deadline = t_start + (deadline - t_start) * 0.75
    print(f"\n[P3] Active-set fine (budget ends at t={p3_deadline - t_start:.1f}s)")
    if time.monotonic() < p3_deadline - 10:
        a_pol2, sc_pol2 = _active_set_polish(best_seq, p3_deadline,
                                             tol_frac_start=0.002, tol_frac_end=0.00002,
                                             lr_init=0.012)
        if sc_pol2 < best_score:
            best_seq = a_pol2.copy()
            best_score = sc_pol2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish_fine] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Fine polish: {sc_pol2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Targeted lag reduction (10% of budget) ===
    p4_deadline = t_start + (deadline - t_start) * 0.85
    print(f"\n[P4] Targeted lag reduction (budget ends at t={p4_deadline - t_start:.1f}s)")
    if time.monotonic() < p4_deadline - 5:
        a_tgt, sc_tgt = _targeted_lag_reduction(best_seq, p4_deadline, n_lags=15, n_steps=150)
        if sc_tgt < best_score:
            best_seq = a_tgt.copy()
            best_score = sc_tgt
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [targeted] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Targeted: {sc_tgt:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(1024, n_best - 1024),
        max(1024, n_best - 512),
        n_best,
        n_best + 512,
        n_best + 1024,
        n_best + 2048,
    ]))

    # Filter out n candidates that are too large (would be slow)
    n_candidates = [n for n in n_candidates if 1024 <= n <= 20000]
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_best:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick L-BFGS
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts (if time) ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier3", _fourier_perturb(best_seq, rng, n_modes=3)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("peri", _make_profile(n_best, "periodic")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 30:
                break
            sc_r = _score(a_r)
            if sc_r < best_score + 0.005:  # Only optimize promising restarts
                a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=10.0)
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Ultra-tight active-set
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           tol_frac_start=0.0005, tol_frac_end=0.000005,
                                           lr_init=0.004)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS with very small beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One more active-set pass after L-BFGS
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _active_set_polish(best_seq, deadline - 1,
                                                 tol_frac_start=0.0003, tol_frac_end=0.000002,
                                                 lr_init=0.003)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [final_polish2] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
