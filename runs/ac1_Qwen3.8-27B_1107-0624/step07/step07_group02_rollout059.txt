```python
Multi-phase optimizer with adaptive smoothing, diverse initialization,
and direct peak-targeting for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_max(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _lse_max(c, beta)

def _smograd(z, beta, n):

    c_max = float(c.max())
```

Let me also think about the "peak-targeting" step more carefully.

When the peak of the convolution is at lag k*, the subgradient of the score with respect to a is:
2*n * (a * e_{k*}) / S^2 (where e_{k*} is the unit vector at position k*)
= 2*n * a_rev[k*] / S^2 (shifted to the right range)

More precisely, d(max_c)/d(a_i) = a[k* - i] (if k* - i is in [0, n-1], else 0).

So the subgradient direction (for decreasing the peak) is:
dg_i = a[k* - i] (for valid i)

We want to decrease a in the direction of dg, but keep a >= 0 and sum(a) = 1.

A simple approach: a_i <- max(0, a_i - eta * dg_i), then normalize.

But a better approach that maintains positivity: a_i <- a_i * (1 - eta * dg_i / max(dg)) or a_i <- a_i * exp(-eta * dg_i).

Let me implement both and see which works better.

Actually, I realize there's a subtlety. The peak might not be unique. If multiple lags have the same maximum, the subgradient is the average of the unit vectors at those peak locations. This is automatically handled by the log-sum-exp smoothing.

For the direct subgradient step (non-smooth), I'll use the average of the peak locations.

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Diverse initialization (tilted, cosine, bimodal, Gaussian)
- Adaptive L-BFGS with longer beta schedule
- Direct peak-targeting subgradient steps
- Multi-n exploration with more values
- Perturbation-based escape from local minima
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=False):
    """Run adaptive L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Determine initial beta from the gap
    c_sorted = np.sort(c_cur)[::-1]
    n_top = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[n_top]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Extended beta schedule
    betas = base_beta * np.array([
        1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.06, 0.04, 0.03, 0.02, 0.015, 0.01, 0.007,
        0.005, 0.003, 0.002, 0.001, 0.0007, 0.0005, 0.0003
    ])

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break

        est_time_per_iter = 0.0015 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 20))))

        prob = _SmoProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"      beta={beta:.2e} ({i+1}/{len(betas)}): score={sc_new:.10f} iters={result.nit}")
        except Exception:
            pass

    return current_a


def _peak_target_steps(a, max_steps=200, deadline=None):
    """Direct subgradient steps targeting the peak of the convolution."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = 0.01
    patience = 0
    
    for step in range(max_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 50:
                break
        
        # Find peak locations (within 0.1% of max)
        tol = 0.001 * c_max
        peak_mask = c >= (c_max - tol)
        peak_indices = np.where(peak_mask)[0]
        
        # Subgradient: for each peak at lag k, the gradient w.r.t. a_i is a[k-i]
        # Average over all peak locations
        grad = np.zeros(n)
        for k in peak_indices:
            # c[k] = sum_i a[i]*a[k-i]
            # d(c[k])/d(a_i) = a[k-i] (if 0 <= k-i < n)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad /= len(peak_indices)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Multiplicative update to maintain positivity
        # a_i *= exp(-lr * grad_i / max(grad))
        grad_norm = grad / (grad.max() + 1e-30)
        a = a * np.exp(-lr * grad_norm)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Decay learning rate
        lr *= 0.995
    
    return best_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.005, tol_frac_end=0.00003, lr_init=0.03):
    """Active-set gradient descent with line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())
    
    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set: peaks within tol_frac of max
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Weight: uniform over active set (or softmax)
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        
        # Gradient: 2 * (a * w) restricted to [n-1, 2n-2]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init * 2, lr * 1.03)
    
    return best_a, _score(best_a)


def _make_init(n, kind, rng):
    """Create diverse initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tilted":
        alpha = rng.uniform(0.05, 0.4)
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 + alpha * (2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tilted2":
        alpha = rng.uniform(0.1, 0.6)
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 + alpha * (t - 0.5) + 0.1 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        freq = rng.integers(1, max(2, n // 10))
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        c1 = rng.uniform(0.15, 0.35)
        c2 = rng.uniform(0.65, 0.85)
        w = rng.uniform(0.01, 0.05)
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2) + 0.01
        return a / a.sum()
    elif kind == "gaussian":
        sigma = rng.uniform(0.1, 0.3)
        center = rng.uniform(0.3, 0.7)
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - center) / sigma) ** 2) + 0.01
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp_cos":
        t = np.arange(n) / max(1, n - 1)
        a = (0.5 + 0.5 * t) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    
    return best_a, best_sc
```
