where p_k depends on a (through c[k]), so this isn't a simple convex combination. The smoothed max is the log-sum-exp, which is convex in c, and c is quadratic in a. So f_T is not necessarily convex in a.

However, for small T (close to the true max), it's approximately the max of a finite set of quadratic forms, which is convex but non-smooth. The smoothing makes it smooth but not necessarily convex.

Despite this, L-BFGS often works well on non-convex problems, especially when started near a local minimum.

"""
L-BFGS + Adaptive Temperature Smoothing + Active-Set CD + Multi-scale optimizer.
Improvements over previous (1.5057):
- L-BFGS quasi-Newton with proper line search (vs Adam)
- Active-set subgradient descent for the non-smooth max
- More aggressive multi-directional coordinate descent
- n=16384 exploration via interpolation + refinement
- Better temperature annealing schedule
- Fourier-based diverse restarts
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _full_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _norm_score(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoothed_max_and_grad(a, T):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        smoothed = c_max
    else:
        p = e / s
        smoothed = c_max + T * np.log(s)
    # Gradient: d/d(a[j]) sum_k p[k]*c[k] = 2*sum_m a[m]*p[j+m]
    N = _next_pow2(2 * (2 * n))
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    result = np.fft.irfft(np.conj(fa) * fp, N)
    grad_a = 2.0 * result[:n]
    return smoothed, grad_a, p

def _obj_and_grad_z(z, T):
    """Objective and gradient in log-space z, for normalized a."""
    n = len(z)
    a = _softmax_stable(z)
    smoothed, grad_a, p = _smoothed_max_and_grad(a, T)
    score = 2.0 * n * smoothed  # sum(a)=1
    # d(score)/d(a[j]) = 2n*grad_a[j] - 2*score*a[j] (from 1/sum^2 term, but sum=1 so just the first)
    # Actually score = 2n*smoothed(a) where a=sum-1 normalized.
    # In z-space: da[j]/dz[k] = a[j]*(delta[jk] - a[k])
    # d(score)/dz[k] = sum_j dscore/da[j] * a[j]*(delta[jk]-a[k])
    # dscore/da[j] = 2n*grad_a[j] (since sum=1 is maintained by the normalization)
    # Wait, score = 2n*smoothed, and smoothed depends on a which depends on z.
    # d(smoothed)/dz[k] = sum_j (d smoothed/da[j]) * (da[j]/dz[k])
    # d(smoothed)/da[j] = grad_a[j]
    # da[j]/dz[k] = a[j]*(delta[jk] - a[k])
    # So d(smoothed)/dz[k] = grad_a[k]*a[k] - a[k]*sum_j grad_a[j]*a[j]
    # = a[k]*(grad_a[k] - W) where W = dot(grad_a, a)
    # d(score)/dz[k] = 2n*a[k]*(grad_a[k] - W)
    W = float(np.dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - W)
    return float(score), grad_z

def _active_set_subgrad(a, top_k=5, decay=0.8):
    """Compute weighted active-set subgradient."""
    n = len(a)
    c = _fft_convolve(a, a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    N = _next_pow2(2 * (2 * n))
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    result = np.fft.irfft(np.conj(fa) * fp, N)
    return 2.0 * result[:n]

def _lbfgs_optimize(a_init, deadline, T_start=None, T_end=None, max_iter=2000, n_phase=1):
    """L-BFGS optimization in log-space with annealed temperature."""
    n = len(a_init)
    a = a_init / a_init.sum()
    if T_start is None:
        c = _fft_convolve(a, a)
        T_start = 0.005 * c.max()
    if T_end is None:
        T_end = 0.0001 * T_start

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _norm_score(a)
    n_iter_done = 0

    # Anneal in phases
    n_blocks = max(1, n_phase)
    iters_per_block = max(100, max_iter // n_blocks)

    for block in range(n_blocks):
        if time.monotonic() > deadline:
            break
        frac = (block + 0.5) / n_blocks
        T = T_start * (T_end / T_start) ** frac

        block_start = time.monotonic()
        # Limit time per block
        block_deadline = min(deadline, block_start + max(5.0, (deadline - block_start) / (n_blocks - block)))

        local_count = [0]

        def obj_grad(z_vec):
            local_count[0] += 1
            val, g = _obj_and_grad_z(z_vec, T)
            if not np.all(np.isfinite(g)):
                g = np.zeros_like(z_vec)
            return val, g

        try:
            res = minimize(obj_grad, z, method='L-BFGS-B',
                          jac=True, options={'maxiter': iters_per_block,
                                            'ftol': 1e-12, 'gtol': 1e-8,
                                            'maxls': 20})
            z = res.x
        except Exception:
            pass

        a_cur = _softmax_stable(z)
        sc = _norm_score(a_cur)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        n_iter_done += local_count[0]

        # Also try active-set subgradient step
        if time.monotonic() < block_deadline - 1:
            for _ in range(50):
                if time.monotonic() > block_deadline - 1:
                    break
                a_cur = _softmax_stable(z)
                g = _active_set_subgrad(a_cur, top_k=3, decay=0.7)
                W = float(np.dot(g, a_cur))
                g_z = a_cur * (g - W)
                g_norm = np.linalg.norm(g_z)
                if g_norm < 1e-10:
                    break
                # Backtracking line search
                lr = 0.5
                improved = False
                while lr > 1e-6:
                    z_try = z - lr * g_z / g_norm
                    a_try = _softmax_stable(z_try)
                    sc_try = _norm_score(a_try)
                    sc_cur = _norm_score(a_cur)
                    if sc_try < sc_cur * (1 - 1e-10):
                        z = z_try
                        if sc_try < best_score:
                            best_score = sc_try
                            best_a = a_try.copy()
                        improved = True
                        break
                    lr *= 0.5
                if not improved:
                    break

    a_final = _softmax_stable(z)
    sc_final = _norm_score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _cd_polish(a, deadline, max_rounds=200, aggressive=True):
    """Coordinate descent polishing - multi-direction, active-lag focused."""
    n = len(a)
    best_a = a.copy()
    best_a = best_a / best_a.sum()
    best_score = _norm_score(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c >= tol)[0]

    # Precompute which j's affect which active k's
    scales = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0] if aggressive else [0.7, 0.85, 0.5, 1.3, 1.5]

    round_num = 0
    stall = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        # Process active lags
        for k in active:
            if time.monotonic() > deadline:
                break
            # Find j range that affects conv[k]
            if k < n:
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
            else:
                j_lo = k - (n - 1)
                j_hi = n - 1

            for j in range(j_lo, j_hi + 1):
                if time.monotonic() > deadline:
                    break
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in scales:
                    new_val = old_val * scale
                    if new_val < 0:
                        continue
                    best_a[j] = new_val
                    s = best_a.sum()
                    if s < 1e-15:
                        best_a[j] = old_val
                        continue
                    best_a = best_a / s
                    c_new = _fft_convolve(best_a, best_a)
                    sc = 2.0 * n * float(c_new.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        c = c_new
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c >= tol)[0]
                        improved = True
                        break
                if not improved:
                    best_a[j] = old_val
                    best_a = best_a / best_a.sum()
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

    return best_a, _norm_score(best_a)

def _fast_cd_polish(a, deadline, max_rounds=100):
    """Faster CD: only try a few promising j values per active lag."""
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _norm_score(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        tol = c_max * 1.001
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            break
        improved = False

        for k in active[:20]:  # limit to top 20 active lags
            if time.monotonic() > deadline:
                break
            if k < n:
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
            else:
                j_lo = k - (n - 1)
                j_hi = n - 1

            # Try a subset of j values (every other one for speed)
            js = range(j_lo, j_hi + 1, max(1, (j_hi - j_lo) // 10))
            for j in js:
                if time.monotonic() > deadline:
                    break
                old_val = best_a[j]
                if old_val < 1e-10:
                    continue
                for scale in [0.5, 0.8, 1.3, 2.0, 0.0]:
                    best_a[j] = old_val * scale
                    s = best_a.sum()
                    if s < 1e-15:
                        best_a[j] = old_val
                        continue
                    best_a = best_a / s
                    c_new = _fft_convolve(best_a, best_a)
                    sc = 2.0 * n * float(c_new.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        c = c_new
                        c_max = float(c.max())
                        improved = True
                        break
                if not improved:
                    best_a[j] = old_val
                    best_a = best_a / best_a.sum()
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, _norm_score(best_a)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, n0 - 1, n0)
    x1 = np.linspace(0, n_new - 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _fourier_init(n, n_harmonics, rng):
    """Generate a random Fourier-based initial sequence."""
    i = np.arange(n)
    log_a = np.zeros(n)
    for j in range(n_harmonics):
        amp = rng.uniform(0.1, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        log_a += amp * np.cos(2 * np.pi * (j + 1) * i / n + phase)
    a = _softmax_stable(log_a)
    return a

def _adam_quick(a_init, deadline, lr=0.01, max_iter=5000):
    """Quick Adam optimization in log-space for short bursts."""
    n = len(a_init)
    c0 = _fft_convolve(a_init, a_init)
    T = 0.005 * c0.max()
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _norm_score(a_init)

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        T_cur = T * (0.5 ** (t / 2000.0))
        a = _softmax_stable(z)
        smoothed, grad_a, p = _smoothed_max_and_grad(a, T_cur)
        W = float(np.dot(grad_a, a))
        g = a * (grad_a - W)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        m_buf = b1 * m_buf + (1 - b1) * g
        v_buf = b2 * v_buf + (1 - b2) * (g * g)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 100 == 0:
            a_cur = _softmax_stable(z)
            sc = _norm_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()

    return best_a, best_score
