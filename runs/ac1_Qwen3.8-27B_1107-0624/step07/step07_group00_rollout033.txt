"""
Multi-phase optimizer combining L-BFGS with annealed log-sum-exp smoothing
and intensive coordinate descent using incremental convolution updates.
Focuses on precise local optimization to beat the previous best of 1.505688.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is normalized (sum=1). Returns 2n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_stable(c, T):
    """Compute T * log(sum(exp(c/T))) stably."""
    c_max = c.max()
    shifted = (c - c_max) / T
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + T * np.log(s)


def _smoothed_obj_and_grad(a, T):
    """
    Compute the smoothed objective value and gradient.
    f_T(a) = 2n * T*log(sum exp(c/T)) / S^2
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n)
    c = _fft_convolve(a, a)
    F = _logsumexp_stable(c, T)
    # Softmax weights
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    p = e / e.sum()
    # p_rev for convolution
    p_rev = p[::-1]
    conv = _fft_convolve(a, p_rev)
    # Gradient: 2n/S^2 * 2*conv[n-1-j] - 4nF/S^3 for each j
    # conv[n-1-j] for j=0..n-1 is conv[0:n] reversed
    main_term = 4.0 * n * conv[0:n][::-1] / (S * S)
    shift_term = 4.0 * n * F / (S * S * S)
    grad = main_term - shift_term
    obj_val = float(2.0 * n * F / (S * S))
    return obj_val, grad


def _lbgfs_smoothed(a_init, deadline, T_start, T_end, max_iter=50000, m=20):
    """
    Run L-BFGS on the smoothed objective in log-space (softmax parameterization).
    """
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    def obj_z(z):
        a = _softmax_stable(z)
        val, _ = _smoothed_obj_and_grad(a, T_current[0])
        return val

    def grad_z(z):
        a = _softmax_stable(z)
        _, grad_a = _smoothed_obj_and_grad(a, T_current[0])
        # Log-space gradient: a[j] * (grad_a[j] - sum(a*grad_a))
        W = float(np.dot(a, grad_a))
        return a * (grad_a - W)

    # Anneal T over iterations
    T_current = [T_start]
    n_steps = max_iter
    t0 = time.monotonic()

        frac = step / max(n_steps - 1, 1)
        T_current[0] = T_start * (T_end / T_start) ** frac

        res = minimize(obj_z, z0, jac=grad_z, method='L-BFGS-B',
                       options={'maxiter': 5, 'ftol': 1e-12, 'gtol': 1e-8,

        if res.x is not None and np.all(np.isfinite(res.x)):

                if 2 * j >= k_start and 2 * j <= k_end:
                    affected[2 * j - k_start] += delta * delta
                scale_sq = scale * scale
                max_affected = float(affected.max()) * scale_sq
                # Also check unaffected max
                # Build a mask or just check a few top unaffected values
                # For speed, check the top few overall
                if max_affected < c_max * 0.5:
                    # Unaffected max dominates
                    # Need to find max of unaffected c values
                    # This is expensive; use a precomputed value
                    # For now, just use c_max as upper bound (conservative)
                    # Actually the unaffected values are scaled too
                    max_unaffected = c_max * scale_sq  # conservative (uses global max)
                    return 2.0 * n * max(max_affected, max_unaffected)
                else:
                    # Affected max dominates, but we still need to check unaffected
                    # Quick check: if c_max * scale_sq < max_affected, we're fine
                    if c_max * scale_sq <= max_affected:
                        return 2.0 * n * max_affected
                    else:
                        # Need full computation (rare)
                        a_try = best_a.copy()
                        a_try[j] += delta
                        a_try = a_try / a_try.sum()
                        c_try = _fft_convolve(a_try, a_try)
                        return 2.0 * n * float(c_try.max())

            # Do a coarse grid search first
            n_grid = 16
            deltas = np.linspace(lo, hi, n_grid)
            best_d = 0.0
            best_v = eval_delta(0.0)
            for d in deltas:
                vd = eval_delta(d)
                if vd < best_v:
                    best_v = vd
                    best_d = d

            # Refine with ternary search around the best grid point
            i_best = int(np.argmin([eval_delta(d) for d in [lo, hi]])) if n_grid < 2 else None
            # Find the two best grid points for bounding
            vals = np.array([eval_delta(d) for d in deltas])
            i_min = int(np.argmin(vals))
            if i_min > 0:
                lo_t = deltas[i_min - 1]
            else:
                lo_t = deltas[i_min]
            if i_min < n_grid - 1:
                hi_t = deltas[i_min + 1]
            else:
                hi_t = deltas[i_min]

            # Ternary search
            for _ in range(30):
                if hi_t - lo_t < 1e-14 * v:
                    break
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = eval_delta(m1)
                f2 = eval_delta(m2)
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1

            d_opt = (lo_t + hi_t) / 2.0
            f_opt = eval_delta(d_opt)

            if f_opt < best_score * (1 - 1e-12):
                # Apply the update
                best_a[j] += d_opt
                best_a = best_a / best_a.sum()
                best_score = f_opt
                c = _fft_convolve(best_a, best_a)
                c_max = float(c.max())
                improved = True
                n_improved_this_round += 1

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _cd_fast(a, deadline, max_rounds=500, focus_top=50):
    """
    Faster CD that focuses on coordinates with high impact.
    Uses the fact that only coordinates near active peaks matter most.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a).copy()
    c_max = float(c.max())

    # Identify "important" coordinates: those that contribute to high c values
    # A coordinate j contributes to c[k] via a[j]*a[k-j], so it affects c values
    # at k = j + i for all i. The "impact" of coordinate j is related to
    # how much it contributes to the peak c values.

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        # Determine priority order: coordinates with highest "sensitivity"
        # Sensitivity of c_max to a[j] is approximately -2*a[j] * (sum of weights at active peaks)
        # For simplicity, prioritize coordinates proportional to a[j]
        # (large coordinates have more impact on convolution)
        order = np.argsort(-best_a)[:min(200, n)]  # top 200 largest
        # Also include some random coordinates for exploration
        if round_num % 5 == 0:
            random_idx = np.random.choice(n, size=min(50, n), replace=False)
            order = np.concatenate([order, random_idx])

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            v = best_a[j]
            if v < 1e-14:
                continue

            # Binary/ternary search for optimal a[j]
            # Try a range of multipliers
            best_local = best_score
            best_mult = 1.0
            # Coarse search
            for mult in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a_try = best_a.copy()
                a_try[j] = v * mult
                a_try = a_try / a_try.sum()
                c_try = _fft_convolve(a_try, a_try)
                sc = 2.0 * n * float(c_try.max())
                if sc < best_local:
                    best_local = sc
                    best_mult = mult

            if best_mult != 1.0:
                # Refine around best_mult
                lo_m = best_mult * 0.95 if best_mult < 1.0 else best_mult * 1.05
                hi_m = best_mult * 1.05 if best_mult < 1.0 else best_mult * 0.95
                for _ in range(20):
                    m1 = lo_m + (hi_m - lo_m) / 3.0
                    m2 = hi_m - (hi_m - lo_m) / 3.0
                    a1 = best_a.copy()
                    a1[j] = v * m1
                    a1 = a1 / a1.sum()
                    f1 = 2.0 * n * float(_fft_convolve(a1, a1).max())
                    a2 = best_a.copy()
                    a2[j] = v * m2
                    a2 = a2 / a2.sum()
                    f2 = 2.0 * n * float(_fft_convolve(a2, a2).max())
                    if f1 < f2:
                        hi_m = m2
                    else:
                        lo_m = m1
                    if hi_m - lo_m < 1e-8:
                        break
                opt_mult = (lo_m + hi_m) / 2.0
                a_try = best_a.copy()
                a_try[j] = v * opt_mult
                a_try = a_try / a_try.sum()
                sc = 2.0 * n * float(_fft_convolve(a_try, a_try).max())
                if sc < best_score * (1 - 1e-12):
                    best_a = a_try
                    best_score = sc
                    c = _fft_convolve(best_a, best_a)
                    c_max = float(c.max())
                    improved = True

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=200000, active_k=3, decay=0.6,
                  eval_interval=100, clip_norm=0.5, use_backtrack=True):
    """Adam in log-space with adaptive temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < 0.2:
            T = T_start
        else:
            frac2 = (frac - 0.2) / 0.8
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if T < T_start * 0.01:
            # Weighted active gradient
            k_indices = np.argsort(c)[::-1][:active_k]
            p = np.zeros_like(c)
            for i, idx in enumerate(k_indices):
                p[idx] = decay ** i
            p_sum = p.sum()
            if p_sum < 1e-300:
                g = np.zeros(n)
            else:
                p = p / p_sum
                p_rev = p[::-1]
                conv_full = _fft_convolve(a, p_rev)
                g = 4.0 * n * conv_full[0:n][::-1]
        else:
            # Smoothed gradient
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s_e = e.sum()
            if s_e < 1e-300 or not np.isfinite(s_e):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s_e
            p_rev = p[::-1]
            conv_full = _fft_convolve(a, p_rev)
            g = 4.0 * n * conv_full[0:n][::-1]

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 30 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.005:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Load starting point
    2. Quick Adam warmup
    3. L-BFGS with annealed smoothing
    4. Intensive coordinate descent
    5. Second L-BFGS pass
    6. More CD
    7. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Quick Adam warmup (60s)
    phase1_end = min(deadline, t_start + 60.0)
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    a_opt, sc = _adam_z_space(a, phase1_end,
                              T_start=0.003 * max_c, T_end=0.00003 * max_c,
                              lr_start=0.03, lr_end=0.002,
                              max_iter=50000, active_k=3,
                              eval_interval=100, use_backtrack=True)
    full_sc = _score(a_opt)
    print(f"Phase 1 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 2: L-BFGS with annealing (60s - 400s)
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        c_2 = _fft_convolve(best_a, best_a)
        max_c2 = float(c_2.max())
        a_lbgfs, sc_lbgfs = _lbgfs_smoothed(best_a, phase2_end,
                                             T_start=0.002 * max_c2,
                                             T_end=0.00001 * max_c2,
                                             max_iter=2000, m=20)
        full_sc_lbgfs = _score(a_lbgfs)
        print(f"Phase 2 L-BFGS: score={full_sc_lbgfs:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_lbgfs) and full_sc_lbgfs < best_score:
            best_score = full_sc_lbgfs
            best_a = a_lbgfs.copy()

    # Phase 3: Intensive CD (400s - 800s)
    phase3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase3_end:
        a_cd, sc_cd = _cd_polish(best_a, phase3_end, max_rounds=500)
        full_sc_cd = _score(a_cd)
        print(f"Phase 3 CD: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 4: Second L-BFGS (800s - 1100s)
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end:
        c_4 = _fft_convolve(best_a, best_a)
        max_c4 = float(c_4.max())
        a_lbgfs2, sc_lbgfs2 = _lbgfs_smoothed(best_a, phase4_end,
                                                T_start=0.0005 * max_c4,
                                                T_end=0.000005 * max_c4,
                                                max_iter=3000, m=20)
        full_sc2 = _score(a_lbgfs2)
        print(f"Phase 4 L-BFGS2: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_lbgfs2.copy()

    # Phase 5: More CD (1100s - 1500s)
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end:
        a_cd2, sc_cd2 = _cd_polish(best_a, phase5_end, max_rounds=1000)
        full_sc_cd2 = _score(a_cd2)
        print(f"Phase 5 CD2: score={full_sc_cd2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd2) and full_sc_cd2 < best_score:
            best_score = full_sc_cd2
            best_a = a_cd2.copy()

    # Phase 6: Ultra-fine Adam + CD (1500s - 1700s)
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end:
        c_6 = _fft_convolve(best_a, best_a)
        max_c6 = float(c_6.max())
        mid6 = min(phase6_end, t_start + 1600.0)
        a_fin, sc_fin = _adam_z_space(best_a, mid6,
                                       T_start=0.0001 * max_c6,
                                       T_end=0.000001 * max_c6,
                                       lr_start=0.005, lr_end=0.0001,
                                       max_iter=100000, active_k=2,
                                       eval_interval=50, use_backtrack=True)
        full_sc_fin = _score(a_fin)
        print(f"Phase 6 Adam fin: score={full_sc_fin:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_fin) and full_sc_fin < best_score:
            best_score = full_sc_fin
            best_a = a_fin.copy()

        if time.monotonic() < phase6_end - 20:
            a_cd3, _ = _cd_fast(best_a, phase6_end - 5, max_rounds=200)
            sc_cd3 = _score(a_cd3)
            print(f"Phase 6 CD fast: score={sc_cd3:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_cd3) and sc_cd3 < best_score:
                best_score = sc_cd3
                best_a = a_cd3.copy()

    # Phase 7: Final polish (1700s - deadline)
    if time.monotonic() < deadline - 5:
        a_final_cd, _ = _cd_polish(best_a, deadline - 3, max_rounds=50)
        sc_final_cd = _score(a_final_cd)
        if np.isfinite(sc_final_cd) and sc_final_cd < best_score:
            best_score = sc_final_cd
            best_a = a_final_cd.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
