KKT active-set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direct KKT system solving with FFT-accelerated iterations,
momentum gradient descent for warm-starting, and multi-n exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _b_full_matvec(x, s_ind, n, s_count):
    """Compute (B*x)[j] = (1/s_count) * sum_i x[i] * s_ind[i+j] for j=0..n-1."""
    N = _next_pow2(3 * n)
    x_p = np.zeros(N)
    x_p[:n] = x
    s_rev = s_ind[::-1]  # reverse of s_ind (length 2n-1)
    s_rev_p = np.zeros(N)
    s_rev_p[:2*n-1] = s_rev
    conv = np.fft.irfft(np.fft.rfft(x_p, N) * np.fft.rfft(s_rev_p, N), N)
    # result[j] = conv[2n-2-j] for j=0..n-1
    # That's conv[2n-2 : n-1 : -1] = conv[n-1 : 2n-1][::-1]
    result = conv[n-1:2*n-1][::-1]
    return result / s_count

"""
KKT active-set optimizer with FFT-accelerated linear solves, momentum GD diversification,
and multi-n exploration. Key innovation: directly solves the KKT linear system
B_T * a_T = lambda*1 (Hankel structure) using iterative methods with FFT mat-vec products,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _b_full_matvec(x, s_ind, n, s_count):
    """Compute (B*x)[j] = (1/s_count) * sum_i x[i]*s_ind[i+j] for j=0..n-1."""
    N = _next_pow2(3 * n)
    x_p = np.zeros(N)
    x_p[:n] = x
    s_rev = s_ind[::-1]
    s_rev_p = np.zeros(N)
    s_rev_p[:2*n-1] = s_rev
    conv = np.fft.irfft(np.fft.rfft(x_p, N) * np.fft.rfft(s_rev_p, N), N)
    result = conv[n-1:2*n-1][::-1]
    return result / s_count


def _kkt_solve(T_arr, s_ind, n, s_count):
    """Solve B_T * y = 1 where B_T[p,q] = s_ind[T[p]+T[q]]/s_count.
    Returns (y, success) where y has length len(T_arr)."""
    m = len(T_arr)
    if m == 0:
        return None, False

    if m <= 5000:
        t = T_arr.astype(np.int64)
        idx = t[:, None] + t[None, :]
        B = np.zeros((m, m))
        valid = idx < 2*n - 1
        B[valid] = s_ind[idx[valid]] / s_count
        B = 0.5 * (B + B.T)
        try:
            y = np.linalg.solve(B + 1e-12 * np.eye(m), np.ones(m))
            return y, True
        except np.linalg.LinAlgError:
            try:
                y, _, _, _ = np.linalg.lstsq(B, np.ones(m), rcond=1e-10)
                return y, True
            except Exception:
                return None, False
    else:
        from scipy.sparse.linalg import LinearOperator, cg, gmres
        # Preconditioner: diagonal of B_T
        ones_full = np.zeros(n)
        ones_full[T_arr] = 1.0
        diag = _b_full_matvec(ones_full, s_ind, n, s_count)[T_arr]
        diag = np.maximum(np.abs(diag), 1e-14)

        def _make_matvec(T_arr, s_ind, n, s_count, reg=0.0):
            def matvec(x):
                x_full = np.zeros(n)
                x_full[T_arr] = x
                res = _b_full_matvec(x_full, s_ind, n, s_count)[T_arr]
                if reg > 0:
                    res = res + reg * x
                return res
            return matvec

        # Try CG first (fastest for SPD)
        try:
            mv = _make_matvec(T_arr, s_ind, n, s_count)
            op = LinearOperator((m, m), matvec=mv)
            y, info = cg(op, np.ones(m), rtol=1e-8, maxiter=300,

        except Exception:

        return None, False

def _momentum_gd(a_init, deadline, lr_init=0.03, momentum=0.9):
    """Momentum gradient descent with adaptive active set."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    conv = _fft_conv(a, a)
    best_score = 2.0 * n * float(conv.max())
    lr = lr_init
    tol_frac = 0.01
    no_improve = 0

    while time.monotonic() < deadline:
        conv = _fft_conv(a, a)
        c_max = float(conv.max())
        sc = 2.0 * n * c_max

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set
        tol = tol_frac * c_max
        active = conv >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        w = np.zeros(2*n-1)
        w[active] = 1.0 / n_active

        # Gradient: grad[j] = (1/|S|) * sum_{k in S} a[k-j] (valid)
        # = _b_full_matvec with w as s_ind
        a_rev = a[::-1]
        wc = _fft_conv(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h
        grad_t = grad - grad.mean()

        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))

        # Momentum update
        v = momentum * v - alpha * grad_t
        a_try = a + v
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            v *= 0.5
            continue
        a_try = a_try / s_try

        sc_try = 2.0 * n * float(_fft_conv(a_try, a_try).max())
        if sc_try < sc + 1e-14:
            a = a_try
            if no_improve % 20 == 0 and no_improve > 0:
                tol_frac = max(0.0001, tol_frac * 0.9)
                lr *= 0.95
            if no_improve > 200:
                # Reset momentum, try different tol
                v = np.zeros(n)
                tol_frac = min(0.05, tol_frac * 2.0)
                lr = lr_init * 0.5
                no_improve = 0
        else:
            v *= 0.5
            lr *= 0.8

    return best_a, _score(best_a)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _shrinkage(a, deadline):
    """Try zeroing out small elements to see if score improves."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted = np.argsort(a)
    for frac in [0.002, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted = np.argsort(a)
    return best_a, best_sc

    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 80)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # KKT after shrinkage
            if time.monotonic() < deadline - 50:
                p5b_deadline = min(deadline - 40, time.monotonic() + 50)
                a_kkt5, sc_kkt5 = _kkt_optimize(a_shr, p5b_deadline)
                if sc_kkt5 < best_score:
                    best_seq = a_kkt5.copy()
                    best_score = sc_kkt5
                    print(f"  NEW BEST: {best_score:.10f} [kkt_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final KKT polish (tight) ===
    print(f"\n[P6] Final KKT polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        p6_deadline = deadline - 3
        a_fin, sc_fin = _kkt_optimize(best_seq, p6_deadline, verbose=True)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_kkt] t={time.monotonic()-t_start:.1f}s")

        # One more momentum GD pass with very small lr
        remaining = p6_deadline - time.monotonic()
        if remaining > 15:
            a_fin2, sc_fin2 = _momentum_gd(best_seq, p6_deadline - 2, lr_init=0.005, momentum=0.95)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_gd] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
