"""
Improved multi-phase optimizer: aggressive L-BFGS continuation, adaptive active-set Newton
with CG solve, n=16384 exploration, and diverse restarts. Targets < 1.5027.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=300, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if pH <= 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=500, tol_start=0.003, tol_end=0.0005):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        # Adaptive tolerance: start wide, tighten over iterations
        frac = it / max(1, max_iter)
        tol = tol_start + (tol_end - tol_start) * frac
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient of sum_{k in active} c_k
        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break

        # Newton step with CG
        mu = 1e-6 + 1e-8 * it  # Slightly increasing regularization
        active_f = active.astype(np.float64)

        def hess_vec(v):
            return _hessian_matvec(v, active_f, n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=250, tol=1e-10)
        dx = dx - np.mean(dx)

        # Line search (backtracking with Armijo)
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            # Recompute active set for the line search evaluation
            c_try_max = float(c_try.max())
            active_try = c_try >= (c_try_max - tol * c_try_max)

            f_try = float(c_try[active_try].sum())
            if f_try < f_old - 1e-4 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:

            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:

            best_a = a.copy()

        np.ones(3) * 1.0,

    betas = np.maximum(betas, max_c * final_beta_frac)
    # Ensure monotonically decreasing

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        grad_z = _smograd(z, beta, n)
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-14:
            break
        z = z - lr * grad_z
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
                lr = min(lr * 1.05, lr_init)
            else:
                no_improve += 1
                lr *= 0.92
                if no_improve > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    no_improve = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.04 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS on parent size ===
    print(f"\n[Phase 1] Aggressive L-BFGS n={len(best_seq)} (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_main = _run_lbfgs(best_seq, deadline, min_time_remaining=400, n_stages=20, final_beta_frac=5e-6)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 2: Active-set Newton (multiple passes with tightening) ===
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        # First pass: wider tolerance
        n_deadline = min(deadline - 300, time.monotonic() + 200)
        a_n1, sc_n1 = _newton_active_set(best_seq, n_deadline, max_iter=200, tol_start=0.005, tol_end=0.001)
        update_best(a_n1, sc_n1, "newton_pass1")

        # Second pass: tighter tolerance
        if time.monotonic() < deadline - 150:
            n_deadline2 = min(deadline - 150, time.monotonic() + 120)
            a_n2, sc_n2 = _newton_active_set(best_seq, n_deadline2, max_iter=200, tol_start=0.001, tol_end=0.0003)
            update_best(a_n2, sc_n2, "newton_pass2")

    # === Phase 3: Direct gradient polish ===
    print(f"\n[Phase 3] Direct gradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        dg_deadline = min(deadline - 200, time.monotonic() + 100)
        a_dg, sc_dg = _run_direct_gradient(best_seq, dg_deadline, n_steps=3000, lr_init=0.008)
        update_best(a_dg, sc_dg, "direct_grad")

    # === Phase 4: Try larger n (16384) ===
    print(f"\n[Phase 4] Try n=16384 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_large = 16384
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_large)
        a_large = np.interp(x_new, x_old, best_seq)
        a_large = np.maximum(a_large, 1e-10)
        a_large = a_large / a_large.sum()
        a_large_opt = _run_lbfgs(a_large, deadline, min_time_remaining=60, n_stages=12, final_beta_frac=1e-5)
        update_best(a_large_opt, _score(a_large_opt), "n16384_lbfgs")

        if time.monotonic() < deadline - 50:
            a_large_n, sc_ln = _newton_active_set(a_large_opt, min(deadline - 50, time.monotonic() + 60),
                                                  max_iter=100, tol_start=0.003, tol_end=0.0008)
            update_best(a_large_n, sc_ln, "n16384_newton")

    # === Phase 5: Diverse restarts ===
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb_log(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=6)
            elif i == 2:
                p_start = _perturb_block(best_seq, rng, n_blocks=5)
            elif i == 3:
                p_start = _perturb_log(best_seq, 0.05, rng)
            else:
                p_start = _perturb_fourier(p_start if i == 2 else best_seq, rng, n_modes=3)

            remaining = deadline - 25 - time.monotonic()
            if remaining < 10:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10, n_stages=10, final_beta_frac=1e-4)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _newton_active_set(a_r, min(deadline - 15, time.monotonic() + 30),
                                                 max_iter=60, tol_start=0.002, tol_end=0.0005)
                update_best(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 6: Final aggressive Newton ===
    print(f"\n[Phase 6] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_set(best_seq, deadline - 3, max_iter=300,
                                           tol_start=0.001, tol_end=0.0002)
        update_best(a_fin, sc_fin, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
