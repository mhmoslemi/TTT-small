        if rs > rs0 * 100:
            break
        rhat_new = _hankel_matvec(rhat, W, n)
        rhatr_new = float(rhat_new @ r)
        if abs(rhatr_new) < 1e-30:
            break
        beta = rhatr_new / rhatr
        p = r + beta * p
        rhat = rhat_new
        rhatr = rhatr_new
    return x


def _newton_active_polish(a_init, deadline, max_rounds=80):
    """Active-set Newton polish with adaptive tolerance and GMRES fallback."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set with expanding tolerance
        base_tol = 0.0003 * c_max
        tol = base_tol * (1.0 + 0.5 * np.exp(-rnd / 10.0))
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Build weight vector
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute gradient of weighted sum of active c_k
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-15 * n:
            break

        # Newton step via CG (with fallback to bi-CG)
        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))

        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                # Try bi-CG
                Hg2 = _bisolve(W, n, g, max_iter=200)
                H12 = _bisolve(W, n, np.ones(n), max_iter=200)
                H12_norm = float(np.linalg.norm(H12))
                if H12_norm > 1e-20:
                    lam2 = float(np.ones(n) @ Hg2) / float(np.ones(n) @ H12)
                    dx = -Hg2 + lam2 * H12
                    dx = dx - np.mean(dx)
                    dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / (gn + 1e-30)

        # Damped Newton with backtracking
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-10 * alpha * gn:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Compute initial beta from the spread of convolution values
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    # More aggressive schedule with smaller ratios
    ratios = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008,
              0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
              0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
              0.0000005, 0.0000002, 0.0000001]
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations based on remaining time and dimension
        est_iter_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))

        try:
            result = minimize(
                _smoobj,
                z,
                args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                if beta < base_beta * 0.1:
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
            elif i < len(betas) - 5:
                # If no improvement, try slightly different z
                z = z + 0.01 * np.random.standard_normal(n)
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _adam_polish(a_init, deadline, max_steps=20000):
    """Adam optimizer in log-space with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.05
    t = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        t += 1
        if i % 50 == 0 and (i // 50) > 5:
            lr = 0.05 / (1.0 + 0.01 * (i // 50))

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use smoothed gradient with moderate beta
        beta = max(c_max * 0.01, c_max - float(np.sort(c)[::-1][min(20, len(c)-1)]))
        shifted = (c - c_max) / (beta + 1e-30)
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    patience = 0
                    lr = max(lr * 0.7, 0.005)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        # Flat top with raised cosine edges
        t = np.linspace(0, 1, n)
        edge = n // 8
        a = np.ones(n)
        for i in range(edge):
            a[i] = 0.5 + 0.5 * np.cos(np.pi * i / edge)
            a[n - 1 - i] = 0.5 + 0.5 * np.cos(np.pi * i / edge)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try
            a_sorted_idx = np.argsort(best_a)
            # Re-optimize from improved point
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs_adaptive(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: adaptive L-BFGS + active-set Newton + Adam +
    shrinkage + multi-n + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main adaptive L-BFGS from parent
    print(f"\n[Phase 1] Main adaptive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs_adaptive(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Active-set Newton polish
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 500, time.monotonic() + 300))
        update_best(a2, sc2, "newton_1")

    # Phase 3: Adam polish
    print(f"\n[Phase 3] Adam polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _adam_polish(best_seq, min(deadline - 400, time.monotonic() + 150))
        update_best(a3, sc3, "adam_1")

    # Phase 4: Second Newton polish
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 350, time.monotonic() + 120))
        update_best(a4, sc4, "newton_2")

    # Phase 5: Another L-BFGS round with the improved point
    print(f"\n[Phase 5] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5 = _run_lbfgs_adaptive(best_seq, deadline, min_time_remaining=100.0)
        update_best(a5, _score(a5), "lbfgs_2")

    # Phase 6: Shrinkage + re-optimize
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a6, sc6 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=100.0)
        update_best(a6, sc6, "shrinkage")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384, 2048]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.002 and time.monotonic() < deadline - 80:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 60, time.monotonic() + 60))
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 8: Diverse restarts with different profiles
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["flat_top", "gaussian", "super_tapered", "cos"]
    for pi, prof in enumerate(profiles):
        if time.monotonic() > deadline - 60:
            break
        # Start from best_seq perturbed with profile blend
        n_cur = len(best_seq)
        p_profile = _make_profile(n_cur, prof)
        p_start = 0.5 * best_seq + 0.5 * p_profile
        p_start = np.maximum(p_start, 1e-10)
        p_start = p_start / p_start.sum()

        a_r = _run_lbfgs_adaptive(p_start, deadline, min_time_remaining=30.0)
        sc_r = _score(a_r)
        update_best(a_r, sc_r, f"profile_{prof}")

        if sc_r < best_score + 0.001 and time.monotonic() < deadline - 20:
            a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 10, time.monotonic() + 30), max_rounds=30)
            update_best(a_rp, sc_rp, f"profile_{prof}_np")

    # Phase 9: Small perturbations + Newton
    print(f"\n[Phase 9] Perturbation search (t={time.monotonic() - t_start:.1f}s)")
    for i in range(6):
        if time.monotonic() > deadline - 15:
            break
        sigma = 0.02 * (i + 1)
        p_start = _perturb(best_seq, sigma, rng)
        a_r = _run_lbfgs_adaptive(p_start, deadline, min_time_remaining=15.0)
        sc_r = _score(a_r)
        update_best(a_r, sc_r, f"perturb_{i}")
        if sc_r < best_score + 0.0005 and time.monotonic() < deadline - 10:
            a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 5, time.monotonic() + 20), max_rounds=20)
            update_best(a_rp, sc_rp, f"perturb_{i}_np")

    # Phase 10: Final ultra-fine polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            n_fin = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _smoobj,
                        z,
                        args=(beta_fin, n_fin),
                        jac=_smograd,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 60, 'maxls': 200}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
