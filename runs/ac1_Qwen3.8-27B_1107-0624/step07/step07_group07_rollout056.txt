def _project_simplex(x, target_sum=1.0):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    n = len(x)
    # We need sum(max(x_i - tau, 0)) = target_sum
    # Sort x in decreasing order
    u = np.sort(x)[::-1]
    # Find the right threshold
    cssv = np.cumsum(u)
    rho = np.max(np.argwhere(u * np.arange(1, n+1) > (cssv - target_sum)))
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(x - theta, 0)

def _peak_targeting(a_init, deadline, max_iters=5000):
    """Targeted peak reduction: identify worst peaks and redistribute mass."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

        peak_vals = c[top_K]
        
        # Weight: focus on the highest peaks
        w_shifted = peak_vals - peak_vals.min()
        if w_shifted.max() < 1e-15:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[top_K] = (w_shifted + 0.1)
            w /= w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]

        grad = grad - grad.mean()  # Preserve sum

        if g_norm < 1e-15:
            break
        
        g_dir = -grad / g_norm  # Descent direction

        best_sc_try = _score(a)

"""
Step sequence optimizer v2: Minimizes 2*n*max(conv(a,a))/sum(a)^2 using
deep L-BFGS beta continuation (to 1e-9), block coordinate descent,
peak-targeted gradient descent, pairwise coordinate descent, and
multi-restart exploration. Key improvements: much deeper beta schedule,
novel block-wise refinement, and better time allocation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_z(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_z(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_z(z0, beta, n, max_iter):
    def _fun(z):
        return _smoobj_z(z, beta, n)

    def _jac(z):
        return _smograd_z(z, beta, n)

    try:
        result = minimize(_fun, z0, jac=_jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        return result.x, bool(result.success)
    except Exception:
        return z0, False


def _lbfgs_deep_schedule(a_start, deadline, min_time_remaining=100.0):

    c_cur = _full_convolve(a_start, a_start)

        2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8,
        5e-9, 2e-9
    ])

    z = np.log(np.maximum(a_start, 1e-15))

def _block_fun(z_block, a_full, start, end, n, beta):
    a_try = a_full.copy()
    a_try[start:end] = _softmax(z_block) * a_full[start:end].sum()
    c = _full_convolve(a_try, a_try)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s_e = float(e.sum())
    if s_e < 1e-300 or not np.isfinite(s_e):
        return 2.0 * n * c_max / (a_try.sum() ** 2)
    return 2.0 * n * (c_max + beta * np.log(s_e)) / (a_try.sum() ** 2)


def _block_jac(z_block, a_full, start, end, n, beta):
    a_try = a_full.copy()
    a_try[start:end] = _softmax(z_block) * a_full[start:end].sum()
    c = _full_convolve(a_try, a_try)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s_e = float(e.sum())
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    a_rev = a_try[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    a_b = a_try[start:end]
    s_b = a_b.sum()
    if s_b < 1e-15:
        return np.zeros_like(a_b)
    return 4.0 * n * a_b * (h[start:end] - W * a_b / s_b)


def _block_coordinate_descent(a_init, deadline, block_size=128, outer_iters=30):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = (n + block_size - 1) // block_size
    beta = 1e-4

    for outer in range(outer_iters):
        if time.monotonic() > deadline:
            break
        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size
            end = min(n, start + block_size)
            a_b = np.maximum(a[start:end], 1e-15)
            z0 = np.log(a_b / a_b.sum())
            try:
                result = minimize(_block_fun, z0, jac=_block_jac,
                                  args=(a, start, end, n, beta),
                                  method='L-BFGS-B',
                                  options={'maxiter': 150, 'ftol': 1e-18,
                                           'gtol': 1e-12, 'maxcor': 20,
                                           'disp': False})
                a_block_new = _softmax(result.x) * a[start:end].sum()
                a[start:end] = a_block_new
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
            except Exception:
                pass
    return best_a, best_sc


def _peak_targeting(a_init, deadline, max_iters=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        K = min(8, len(c))
        top_K = np.argsort(c)[::-1][:K]
        peak_vals = c[top_K]
        w = np.zeros_like(c)
        w[top_K] = (peak_vals - peak_vals.min() + 0.05)
        w /= w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        g_dir = -grad / g_norm

        best_step = 0.0
        best_sc_try = _score(a)
        for step in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            a_try = a + step * g_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-15:
                best_step = step
                best_sc_try = sc_try

        if best_step > 0:
            a = a + best_step * g_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_sc_try < best_sc - 1e-15:
                best_sc = best_sc_try
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 200:
            break

    return best_a, best_sc


def _pair_cd(a_init, deadline, max_rounds=100000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        s_pair = a[i] + a[j]
        if s_pair < 1e-12:
            continue
        improved = False
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            a_try = a.copy()
            a_try[i] = s_pair * frac
            a_try[j] = s_pair * (1.0 - frac)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
        if no_improve > 5000:
            break
    return best_a, best_sc


def _perturb(a_base, rng, mode, scale=0.03):
    n = len(a_base)
    z = np.log(np.maximum(a_base, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.002 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        bs = max(16, n // 32)
        for _ in range(12):
            st = int(rng.integers(0, max(1, n - bs)))
            z[st:st + bs] += rng.uniform(-0.008, 0.008)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()

            print(f"  NEW BEST final: {best_score_val:.10f}")
    print(f"  Phase 8 done: {time.monotonic()-t8:.1f}s")

    # Final
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval err: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    return [float(x) for x in fallback]
