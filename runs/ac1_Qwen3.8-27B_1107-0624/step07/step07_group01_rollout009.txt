"""
Optimizer using active-set Newton with FFT-accelerated Jacobian operations.
Key insight: The Jacobian of the autocorrelation peaks has a shift structure,
making JJ^T computable in O(K^2) from the autocorrelation, and J^T*y computable
in O(n log n) via FFT. This enables fast Newton iterations on the active peak set.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_JJT_from_corr(c, top_idx):
    """Compute JJ^T matrix using the autocorrelation.
    (JJ^T)[m,m'] = 4 * c[k_m - k_m'] (approximately)
    """
    K = len(top_idx)
    JJT = np.zeros((K, K))
    for m in range(K):
        for p in range(K):
            lag = int(top_idx[m]) - int(top_idx[p])
            # c is defined for indices 0..2n-2
            # c[lag] if 0 <= lag < len(c), else c[lag + len(c)] (circular approx)
            if 0 <= lag < len(c):
                JJT[m, p] = 4.0 * c[lag]
            elif -len(c) < lag < 0:
                JJT[m, p] = 4.0 * c[len(c) + lag]
            else:
                JJT[m, p] = 0.0

"""
Optimizer using active-set Newton with FFT-accelerated Jacobian operations.
Key innovations over previous approach:
1. Active-set Newton: exploits the shift structure of the autocorrelation Jacobian
   to compute JJ^T in O(K^2) and J^T*y in O(K*n), enabling fast Newton iterations
   directly on the active peak set.
2. Adaptive peak tracking: dynamically identifies and tracks the active set.
3. Hybrid polishing: combines Newton with L-BFGS for robust convergence.
4. Multi-length and multi-start strategy for global exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _grad_c_k(a, k):
    """Compute gradient of c_k = (a*a)_k w.r.t. a.
    Result[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    (The special case k=2i gives 2*a[i] still, since d(a_i*a_i)/d(a_i) = 2*a_i)
    """
    n = len(a)
    grad = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo <= i_hi:
        grad[i_lo:i_hi + 1] = 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    return grad


def _JT_y(a, top_idx, y):
    """Compute J^T * y where J[m,i] = grad_c_k(top_idx[m])[i].
    Result[i] = sum_m y[m] * 2 * a[top_idx[m] - i] (for valid indices).
    """
    n = len(a)
    K = len(top_idx)
    result = np.zeros(n)
    for m in range(K):
        km = int(top_idx[m])
        ym = y[m]
        i_lo = max(0, km - n + 1)
        i_hi = min(n - 1, km)
        if i_lo <= i_hi:
            result[i_lo:i_hi + 1] += ym * a[km - i_hi:km - i_lo + 1][::-1]
    return 2.0 * result


def _JJT_from_corr(c, top_idx):
    """Compute JJ^T using autocorrelation: (JJ^T)[m,p] = 4*c[k_m - k_p] approx."""
    K = len(top_idx)
    n_c = len(c)
    JJT = np.zeros((K, K))
    for m in range(K):
        for p in range(K):
            lag = int(top_idx[m]) - int(top_idx[p])
            if 0 <= lag < n_c:
                JJT[m, p] = 4.0 * c[lag]
            elif -n_c < lag < 0:
                JJT[m, p] = 4.0 * c[n_c + lag]
    return JJT


def _active_set_newton(a_init, deadline, K_schedule=None, max_inner=30, lambda_scale=1e-8):
    """Active-set Newton method for minimizing max autocorrelation.
    Directly equalizes the top-K peaks using Newton's method."""
    if K_schedule is None:
        K_schedule = [3, 5, 8, 12, 20, 35, 50]

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for K in K_schedule:
        if time.monotonic() > deadline:
            break
        K = min(K, 2 * n - 1)

        for it in range(max_inner):
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            top_idx = np.argsort(c)[::-1][:K]
            active_vals = c[top_idx]
            mean_val = active_vals.mean()
            residuals = active_vals - mean_val

            # Check convergence
            if np.max(np.abs(residuals)) < 1e-12 * mean_val:
                break

            # Build JJ^T from autocorrelation
            JJT = _JJT_from_corr(c, top_idx)

            # Regularize and solve
            reg = lambda_scale * np.trace(JJT) / K
            JJT_reg = JJT + reg * np.eye(K)

            try:
                y = np.linalg.solve(JJT_reg, -residuals)
            except np.linalg.LinAlgError:
                try:
                    y = np.linalg.lstsq(JJT_reg, -residuals, rcond=None)[0]
                except Exception:
                    break

            # Compute Newton direction: da = J^T * y
            da = _JT_y(a, top_idx, y)

            # Center the direction (remove mean to preserve normalization)
            da = da - da.mean()

            # Line search
            sc_cur = _score(a)
            alpha = 1.0
            improved = False
            for _ in range(25):
                a_try = a + alpha * da
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                if sc_try < sc_cur - 1e-16:
                    a = a_try
                    sc_cur = sc_try
                    improved = True
                    break
                alpha *= 0.6

            if not improved:
                # Try a smaller fixed step
                alpha = 0.1
                a_try = a + alpha * da
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    sc_try = _score(a_try)
                    if sc_try < sc_cur:
                        a = a_try
                    elif sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                break

        # Update a to best found so far for next K
        a = best_a.copy()

    return best_a, best_sc


def _active_set_subgrad(a_init, deadline, K=5, max_steps=50000, lr_init=0.03):

        c = _full_convolve(a, a)
        top_idx = np.argsort(c)[::-1][:K]

        # grad = sum_m (1/K) * grad_c_{top_idx[m]}
        g = np.zeros(n)
        for m in range(K):
            g += _grad_c_k(a, int(top_idx[m]))
        g /= K

        # Project gradient: work in log-space
        # d/dz (sum top-K c_k) = a * (sum top-K grad_c_k)
        grad_z = a * g
        grad_z = grad_z - grad_z.mean()

        gnorm = np.linalg.norm(grad_z)

        z = z - step
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

            else:

                if stall > stall_threshold:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95

    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter=5000):
    if beta <= 0:
        return a_init.copy()
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

            break

        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(15000, max(200, int(remaining / (0.003 * scale * 25))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5010:
            break

    return current_a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric(n, gamma=1.5):
    """Asymmetric ramp - helps reduce central peak."""
    t = np.linspace(0, 1, n)
    a = t ** gamma + 0.1
    return a / a.sum()


def _make_bimodal(n, sep=0.3, width=0.1):
    """Two peaks away from center."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5 - sep) / width) ** 2) + np.exp(-((t - 0.5 + sep) / width) ** 2)
    a += 0.05
    return a / a.sum()


def _make_fourier_restricted(n, n_freqs=5, rng=None):
    """Sequence with restricted Fourier support."""
    if rng is None:
        rng = np.random.default_rng(42)
    absorb = np.zeros(n)
    for _ in range(n_freqs):
        f = int(rng.integers(1, max(2, n // 4)))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        t = np.arange(n)
        absorb += amp * np.cos(2.0 * np.pi * f * t / n + phase)
    a = 1.0 + absorb
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _poly_phase_optimize(a_init, deadline, n_basis=8):
    """Optimize using a polynomial parameterization of the log-sequence.
    z[i] = sum_j p_j * (i/n)^j, optimize p."""
    n = len(a_init)
    t = np.linspace(0, 1, n)
    T = np.vander(t, N=n_basis, increasing=True).T  # n x n_basis

    # Initial parameters from current sequence
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    p0, _, _, _ = np.linalg.lstsq(T, z0, rcond=None)

    c = _full_convolve(a_init, a_init)
    max_c = c.max()
    beta = max_c * 0.005

    def fun_p(p):
        z = T @ p
        return _smoobj(z, beta, n)

    def jac_p(p):
        z = T @ p
        gz = _smograd(z, beta, n)
        return T.T @ gz

    try:
        result = minimize(fun_p, p0, jac=jac_p, method='L-BFGS-B',
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'disp': False})
        p_opt = result.x
        z_opt = T @ p_opt
        a_opt = _softmax(z_opt)
        return a_opt, result.x
    except Exception:
        return a_init.copy(), p0


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=60.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(30.0, (deadline - reserve - time.monotonic()) / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + min(time_per_hop * 0.6, 120.0)

        # Create perturbation
        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(3):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.01 * rng.uniform(0.5, 3.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Asymmetric perturbation
            t = np.arange(n) / n
            asym = rng.uniform(-0.05, 0.05)
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + asym * t
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Local block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 5
            block_sz = n // n_blocks
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.03, 0.03)
            z = z - z.mean()
            a_pert = _softmax(z)

        # Quick Newton from perturbation
        a_opt, _ = _active_set_newton(a_pert, hop_deadline,
                                      K_schedule=[5, 10, 20], max_inner=15)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Initialization (parent + diverse starts)
    Phase 2: Active-set Newton (main optimization)
    Phase 3: L-BFGS with adaptive smoothing
    Phase 4: Poly-phase optimization (reduced dimensionality)
    Phase 5: Subgradient polishing
    Phase 6: Basin hopping
    Phase 7: Length exploration
    Phase 8: Final aggressive polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        parent = _make_chirp(4096, freq=1.0, eps=0.3)
        try_update(parent, "default_chirp")
        print(f"  No parent, using chirp: n={len(parent)}, score={_score(parent):.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: Active-set Newton (main) ===
    print(f"\n=== Phase 2: Active-set Newton (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        a_ns, sc_ns = _active_set_newton(
            parent, deadline - 300,
            K_schedule=[3, 5, 8, 12, 20, 35, 50, 80, 120],
            max_inner=40, lambda_scale=5e-9)
        try_update(a_ns, f"newton_K_schedule")

    # === Phase 3: L-BFGS refinement ===
    print(f"\n=== Phase 3: L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_l, sc_l = _adaptive_lbfgs(best_seq, deadline, reserve_time=250.0)
        try_update(a_l, "lbfgs_refine")

    # === Phase 4: More Newton from L-BFGS result ===
    print(f"\n=== Phase 4: Newton after L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_n2, sc_n2 = _active_set_newton(
            best_seq, deadline - 250,
            K_schedule=[5, 10, 20, 50, 100, 200],
            max_inner=50, lambda_scale=1e-9)
        try_update(a_n2, "newton_round2")

    # === Phase 5: Poly-phase optimization ===
    print(f"\n=== Phase 5: Poly-phase (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        for n_basis in [6, 8, 10, 12]:
            if time.monotonic() > deadline - 100:
                break
            a_poly, _ = _poly_phase_optimize(best_seq, deadline - 100, n_basis=n_basis)
            try_update(a_poly, f"poly_{n_basis}")
            # Newton polish from poly result
            if time.monotonic() < deadline - 80:
                a_np, _ = _active_set_newton(a_poly, deadline - 80,
                                             K_schedule=[10, 20, 50], max_inner=20)
                try_update(a_np, f"poly{ n_basis}_newton")

    # === Phase 6: Subgradient polishing ===
    print(f"\n=== Phase 6: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _active_set_subgrad(best_seq, deadline - 150,
                                          K=5, max_steps=30000, lr_init=0.04)
        try_update(a_sg, "subgrad_K5")

    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg2, sc_sg2 = _active_set_subgrad(best_seq, deadline - 100,
                                            K=10, max_steps=20000, lr_init=0.02)
        try_update(a_sg2, "subgrad_K10")

    # === Phase 7: Basin hopping ===
    print(f"\n=== Phase 7: Basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 80, rng,
                                   n_hops=5, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        current_n = len(best_seq)
        # Try nearby lengths
        offsets = [512, -512, 1024, -1024, 2048, -2048]
        for off in offsets:
            target_n = current_n + off
            if target_n < 1024 or target_n > 65536:
                continue
            if time.monotonic() > deadline - 40:
                break
            print(f"  Trying n={target_n} (offset {off})...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_i = _score(a_interp)
            print(f"    Interpolated: {sc_i:.10f}")

            if time.monotonic() < deadline - 30:
                a_opt, _ = _active_set_newton(a_interp, deadline - 20,
                                              K_schedule=[5, 10, 20, 50],
                                              max_inner=20)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 9: Final aggressive polishing ===
    print(f"\n=== Phase 9: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        for rnd in range(10):
            if time.monotonic() > deadline - 2:
                break

            # Newton with very fine K
            a_f, sc_f = _active_set_newton(
                best_seq, deadline - 1,
                K_schedule=[10, 20, 50, 100, 200, 500],
                max_inner=30, lambda_scale=1e-10)
            try_update(a_f, f"final_newton_{rnd}")

            # Tiny L-BFGS
            if time.monotonic() < deadline - 1:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-8
                a_lb = _lbfgs_smooth(best_seq, beta, n, max_iter=2000)
                try_update(a_lb, f"final_lbfgs_{rnd}")

            # Subgradient
            if time.monotonic() < deadline - 1:
                a_s, _ = _active_set_subgrad(best_seq, deadline - 1,
                                             K=5, max_steps=5000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
