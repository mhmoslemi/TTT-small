"""

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _convolve(a):
    return _full_convolve(a, a)

def _proj_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_of_C_k(a, k):
    """Compute gradient of C[k] = conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    # C[k] = sum_i a[i]*a[k-i] for valid i, k-i
    # dC[k]/da[j] = a[k-j] + (if k-j != j) a[k-j]
    # = 2*a[k-j] if 0<=k-j<n and j != k-j
    # = a[j] if k=2j
    # = 0 if k-j < 0 or k-j >= n
    for j in range(n):
        kj = k - j
        if 0 <= kj < n:
            if kj == j:
                g[j] = a[j]
            else:
                g[j] = 2.0 * a[kj]
    return g

def _grad_of_C_k_fast(a, k):
    """Fast computation of gradient of C[k] using the shift structure."""
    n = len(a)
    # dC[k]/da[j] = 2*a[k-j] for valid k-j, except when k-j==j where it's a[j]
    # This is essentially 2 * (a shifted and reversed)
    g = np.zeros(n)
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    indices = np.arange(j_start, j_end + 1)
    k_minus_j = k - indices
    g[indices] = 2.0 * a[k_minus_j]
    # Fix the center point if k is even
    if k % 2 == 0:
        center = k // 2
        if 0 <= center < n:
            g[center] = a[center]
    return g

def _min_norm_subgradient(a, C, k_active=10, tol=1e-10):
    """
    Compute the minimum-norm subgradient of max(C) at the current point.
    Returns (g, active_set) where g is the subgradient and active_set is the list of active indices.
    """
    n = len(a)
    c_max = C.max()
    
    # Find active set: indices where C[k] >= c_max - tol * c_max
    threshold = c_max - tol * c_max
    active = np.where(C >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(C)])
    if len(active) == 1:
        return _grad_of_C_k_fast(a, active[0]), active
    
    # Limit to top k_active
    if len(active) > k_active:
        top_indices = np.argsort(C[active])[::-1][:k_active]
        active = active[top_indices]
    
    m = len(active)
    if m == 1:
        return _grad_of_C_k_fast(a, active[0]), active
    
    # Compute the m x m Gram matrix G where G[i,j] = grad_ki . grad_kj
    # grad_ki . grad_kj = sum_l grad_ki[l] * grad_kj[l]
    # = sum_l (2*a[ki-l]) * (2*a[kj-l]) (approximately, ignoring center corrections)
    # For large n, this is approximately 4 * sum_l a[ki-l] * a[kj-l]
    
    # Compute gradients (m vectors of length n)
    grads = np.zeros((m, n))
    for i in range(m):
        grads[i] = _grad_of_C_k_fast(a, active[i])
    
    # Gram matrix
    G = grads @ grads.T  # m x m
    
    # Solve: min w^T G w s.t. w >= 0, sum(w) = 1
    # This is a QP. For small m, use scipy.
    from scipy.optimize import minimize as sp_min
    
    w0 = np.ones(m) / m
    
    def obj(w):
        return w @ G @ w
    
    def grad_obj(w):
        return 2 * G @ w
    
    constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1}]
    bounds = [(0, 1)] * m
    
    try:
        result = sp_min(obj, w0, jac=grad_obj, method='SLSQP',
                        bounds=bounds, constraints=constraints,
                        options={'maxiter': 100, 'ftol': 1e-15})
        w = np.maximum(result.x, 0)
        w = w / w.sum()
    except:
        w = w0
    
    g = w @ grads
    return g, active

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    """L-BFGS with log-sum-exp smoothing."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    
    def fun(z):
        a = _softmax(z)
        c = _convolve(a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def jac(z):
        a = _softmax(z)
        c = _convolve(a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    
    try:

                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _coordinate_descent(a_init, deadline, max_steps=500000):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce the max convolution.
    """
    n = len(a_init)

            break
        
        C = _convolve(a)
        k_max = int(np.argmax(C))

        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        indices = np.arange(j_start, j_end + 1)
        products = a[indices] * a[k_max - indices]
        j_best = indices[np.argmax(products)]
        j_other = k_max - j_best
        
        if j_best == j_other:

        # Try moving mass from the larger of a[j_best], a[j_other] to a random other position
        if a[j_best] > a[j_other]:
            src = j_best
        else:
            src = j_other
        
        # Try a few destination positions

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
