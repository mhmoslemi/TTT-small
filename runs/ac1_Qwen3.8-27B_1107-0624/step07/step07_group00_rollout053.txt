"""
LP-based active-set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a cutting-plane LP approach to find optimal perturbation directions,
combined with structured initializations (quadratic residues, m-sequences),
multi-scale exploration, and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import scipy.sparse as sp

def _next_pow2(n):
    return 1 << max(1, (n - 1).bit_length())

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _quadratic_residue_init(n, p=None):
    """Initialize using quadratic residues modulo a prime."""
    if p is None:
        # Find a prime near n
        p = n
        while True:
            if p > 2 and all(p % i for i in range(2, int(p**0.5) + 1)):
                break
            p += 1
    # Quadratic residues mod p
    residues = set((x * x) % p for x in range(p))
    a = np.zeros(n)
    for i in range(min(n, p)):
        if i in residues:
            a[i] = 1.0
    if a.sum() < 1e-10:
        a = np.ones(n)
    return _normalize(np.maximum(a, 0.0))

def _solve_lp_direction(f, M, active_indices=None, n=None):
    """
    Solve the LP: maximize sum(g) subject to g>=0, (f*g)_k <= M for k in active_indices.
    Returns the optimal g (not normalized).
    """
    if n is None:
        n = len(f)
    
    if active_indices is None:
        active_indices = list(range(2 * n - 1))
    
    n_constraints = len(active_indices)
    
    # Build constraint matrix A (sparse)
    # Row k, col j: f[j] if 0 <= k-j <= n-1
    
    rows = []
    cols = []
    vals = []
    
    for row_idx, k in enumerate(active_indices):
        # (f*g)_k = sum_j f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(f[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    b = np.full(n_constraints, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'dual_feasibility_tolerance': 1e-8})
    
    if res.status == 0:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n

    """
    Run LP-based iterations to reduce the peak of conv(a,a).

    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        score = 2.0 * n * M
        
        # Find active constraints (top peaks)
        top_k = min(n_max_constraints, 2 * n - 1)
        active_indices = np.argsort(c)[::-1][:top_k]
        
        # Solve LP
        g = _solve_lp_direction(a, M, active_indices.tolist(), n)
        g = np.maximum(g, 0.0)
        s_g = g.sum()
        if s_g < 1e-10:
            break
        g = g / s_g
        
        # Check if g actually improves
        g_score = _score_norm(g)
        
        # Blend
        for t in [1.0, 0.5, 0.3, 0.1, 0.05]:
            a_new = _normalize((1 - t) * a + t * g)
            s_new = _score_norm(a_new)
            if s_new < score:
                a = a_new
                score = s_new
                break
        
        if _score_norm(a) < best_score:
            best_score = _score_norm(a)
            best_a = a.copy()
        
        # Convergence check
        if it > 0 and abs(score - best_score) < 1e-8 * best_score:
            break
    
    return best_a, best_score

def _smoothed_gradient(a, c, T):
    """Compute smoothed gradient of max(conv(a,a)) with temperature T."""
    n = len(a)
    c_shifted = (c - c.max()) / (T + 1e-30)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    # Gradient of 2n*max(conv) w.r.t. a_j is 4n * (a * p_rev)[n-1+j]
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

def _adam_logspace(a_init, deadline, n_iters=200000, T_start=None, T_end=None,
                   lr_start=0.03, lr_end=0.002, active_k=5, decay=0.7):

    n = len(a_init)

    if T_end is None:
        T_end = T_start * 0.01
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_a = a.copy()
    best_score = _score(a)
    patience = 0

    while t < n_iters:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / n_iters)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _fft_convolve(a, a)

            p = p / p.sum()
            p_rev = p[::-1]
            conv_full = _fft_convolve(a, p_rev)
            g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        else:
            g = _smoothed_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip
        upd_norm = np.linalg.norm(update)
        if upd_norm > 1.0:
            update = update * (1.0 / upd_norm)
        
        z = z - update
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        # Regular evaluation
        if t % 100 == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
                if patience > 5 and lr_mult > 0.05:
                    lr_mult *= 0.5
                    patience = 0
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)

        for k in peaks:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):

                    a_cand = a.copy()

                    if s < 1e-15:

        if not improved:

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            break

        weight = np.zeros(n)
        for k in top_k:

            j_hi = min(n - 1, k)

"""
Multi-strategy optimizer combining LP-based active-set directions, iterative
de-correlation, Adam in log-space with adaptive temperature, multi-scale
exploration, and coordinate descent polishing. Key innovation over previous
approach: (1) LP gives provably non-increasing directions for the convex
envelope of the peak, (2) iterative de-correlation directly attacks peak
alignment, (3) more aggressive multi-restart with structured initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    return 1 << max(1, (n - 1).bit_length())


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _smoothed_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / (T + 1e-30)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _active_gradient(a, c, top_k=5, decay=0.7):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_logspace(a_init, deadline, n_iters=200000, T_start=None, T_end=None,
                   lr_start=0.03, lr_end=0.002, active_k=5, decay=0.7,
                   eval_interval=100, clip_norm=1.0, plateau_frac=0.2):
    n = len(a_init)
    a = _normalize(a_init)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    if T_start is None:
        c = _fft_convolve(a, a)
        T_start = 0.005 * float(c.max())
    if T_end is None:
        T_end = T_start * 0.01

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = _score(a)
    patience = 0
    lr_mult = 1.0

    while t < n_iters:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / n_iters)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / max(1.0 - plateau_frac, 0.01)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if T < T_start * 0.03:
            g = _active_gradient(a, c, active_k, decay)
        else:
            g = _smoothed_gradient(a, c, T)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        upd_norm = np.linalg.norm(update)
        if upd_norm > clip_norm:
            update = update * (clip_norm / upd_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
                if patience > 4 and lr_mult > 0.05:
                    lr_mult *= 0.5
                    patience = 0
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)

    return best_a, best_score


def _decorr_iteration(a, deadline, n_iters=20000, alpha_start=0.3, alpha_end=0.01,
                      top_k=3, eval_interval=200):
    """Iterative de-correlation: reduce alignment of a with its shifts at peak positions."""
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(n_iters, 1))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        c = _fft_convolve(a, a)
        M = float(c.max())
        peaks = np.argsort(c)[::-1][:top_k]

        # Compute weight[j] = sum_{k in peaks} a[j] * a[k-j]
        weight = np.zeros(n)
        for k in peaks:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                for j in range(j_lo, j_hi + 1):
                    weight[j] += a[j] * a[k - j]

        w_max = weight.max()
        if w_max > 1e-20:
            a_new = a / (1.0 + alpha * weight / w_max)
            a_new = np.maximum(a_new, 0.0)
            a = _normalize(a_new)

        if it % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _solve_lp_direction(f, M, active_indices, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for k in active_indices."""
    n_constraints = len(active_indices)
    rows = []
    cols = []
    vals = []

    for row_idx, k in enumerate(active_indices):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(f[j])

    if len(vals) == 0:
        return np.ones(n) / n

    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    b = np.full(n_constraints, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 15, 'dual_feasibility_tolerance': 1e-7})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n


def _lp_optimize(a_init, deadline, max_iters=30, max_constraints=80, step_fracs=None):
    """LP-based active-set optimization."""
    if step_fracs is None:
        step_fracs = [1.0, 0.5, 0.3, 0.2, 0.1]
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        M = float(c.max())
        score = 2.0 * n * M

        top_k = min(max_constraints, 2 * n - 1)
        active = np.argsort(c)[::-1][:top_k]

        g = _solve_lp_direction(a, M, active.tolist(), n)
        g = np.maximum(g, 0.0)
        s_g = g.sum()
        if s_g < 1e-10:
            break
        g = g / s_g

        # Try different step sizes
        improved = False
        for t_frac in step_fracs:
            a_new = _normalize((1 - t_frac) * a + t_frac * g)
            sc_new = _score_norm(a_new)
            if sc_new < score * (1 - 1e-12):
                a = a_new
                score = sc_new
                improved = True
                break

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=40, top_k=5):
    """Coordinate descent polishing on active peak indices."""
    n = len(a)
    a = _normalize(a.copy())
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = _fft_convolve(a, a)
        peaks = np.argsort(c)[::-1][:top_k]

        for k in peaks:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                for scale in [0.85, 0.9, 1.1, 1.15, 0.75, 1.25]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1 - 1e-10):
                        best_score = sc
                        a = a_cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return a, best_score


def _structured_init(n, kind="quadratic", seed=0):
    """Generate structured initial sequences."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "quadratic":
        # Use quadratic residues for a structured pattern
        rng = np.random.default_rng(seed)
        # Create a pattern based on quadratic residues mod a prime
        p = n
        while p > 2:
            is_prime = all(p % i != 0 for i in range(2, int(p ** 0.5) + 1))
            if is_prime:
                break
            p += 1
        residues = set((x * x) % p for x in range(p))
        a = np.zeros(n)
        for i in range(min(n, p)):
            a[i] = 1.0 if i in residues else 0.1
        a += rng.uniform(0, 0.1, size=n)
        return _normalize(np.maximum(a, 0.0))
    elif kind == "banded":
        # Concentrate mass in a band
        center = n // 2
        width = n // 4
        a = np.zeros(n)
        a[center - width:center + width] = 1.0
        a += np.random.default_rng(seed).uniform(0, 0.05, size=n)
        return _normalize(a)
    elif kind == "exponential":
        rng = np.random.default_rng(seed)
        a = rng.exponential(2.0, size=n)
        return _normalize(a)
    elif kind == "bimodal":
        c1, c2 = n // 3, 2 * n // 3
        a = np.zeros(n)
        w = n // 8
        a[c1 - w:c1 + w] = 1.0
        a[c2 - w:c2 + w] = 1.0
        a += np.random.default_rng(seed).uniform(0, 0.05, size=n)
        return _normalize(a)
    else:
        rng = np.random.default_rng(seed)
        a = rng.uniform(0.1, 1.0, size=n)
        return _normalize(a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. LP-based active-set optimization (principled direction finding)
    2. Iterative de-correlation (fast peak alignment reduction)
    3. Adam in log-space with adaptive temperature
    4. Multi-scale exploration with structured initializations
    5. Coordinate descent polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        a_parent = _normalize(parent)
    except Exception:
        a_parent = np.ones(2048) / 2048
        print("No parent available, using uniform init")

    best_a = a_parent.copy()
    best_score = _score(a_parent)
    n_parent = len(a_parent)
    print(f"Parent: n={n_parent}, score={best_score:.10f}")

    # === Phase 1: LP-based optimization (0-200s) ===
    p1_end = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: LP optimization ---")

    # Work at n=4096 for LP (manageable size)
    n_lp = min(n_parent, 4096)
    a_lp_init = _interp_to_length(a_parent, n_lp) if n_lp != n_parent else a_parent.copy()
    sc_lp_init = _score(a_lp_init)
    print(f"  LP init (n={n_lp}): score={sc_lp_init:.10f}")

    a_lp, sc_lp_norm = _lp_optimize(a_lp_init, p1_end, max_iters=25, max_constraints=60,
                                     step_fracs=[1.0, 0.5, 0.25, 0.1, 0.05])
    sc_lp = _score(a_lp)
    print(f"  LP result (n={n_lp}): score={sc_lp:.10f}, time={time.monotonic()-t_start:.1f}s")
    if sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
    else:
        print(f"  LP did not improve over parent")

    # === Phase 2: De-correlation on best (200-400s) ===
    p2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < p2_end:
        print(f"\n--- Phase 2: De-correlation (n={len(best_a)}) ---")
        a_dc, sc_dc_norm = _decorr_iteration(best_a, p2_end, n_iters=30000,
                                              alpha_start=0.5, alpha_end=0.005,
                                              top_k=5, eval_interval=300)
        sc_dc = _score(a_dc)
        print(f"  De-corr result: score={sc_dc:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_dc < best_score:
            best_score = sc_dc
            best_a = a_dc.copy()

    # === Phase 3: Adam polish (400-700s) ===
    p3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < p3_end:
        print(f"\n--- Phase 3: Adam log-space (n={len(best_a)}) ---")
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_adam, sc_adam = _adam_logspace(best_a, p3_end, n_iters=200000,
                                         T_start=0.003 * max_c, T_end=0.00003 * max_c,
                                         lr_start=0.025, lr_end=0.001,
                                         active_k=5, decay=0.7,
                                         eval_interval=100, plateau_frac=0.25)
        sc_adam_full = _score(a_adam)
        print(f"  Adam result: score={sc_adam_full:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_adam_full < best_score:
            best_score = sc_adam_full
            best_a = a_adam.copy()

    # === Phase 4: Multi-scale structured exploration (700-1100s) ===
    p4_end = min(deadline, t_start + 1100.0)
    print(f"\n--- Phase 4: Multi-scale exploration ---")

    configs = [
        (4096, "quadratic", 7),
        (4096, "bimodal", 13),
        (4096, "banded", 23),
        (8192, "exponential", 37),
        (2048, "quadratic", 43),
    ]

    for n_try, kind, sd in configs:
        if time.monotonic() > p4_end - 60:
            break
        a_init = _structured_init(n_try, kind=kind, seed=sd)
        # Also try interpolating current best
        a_interp = _interp_to_length(best_a, n_try)
        a_init = 0.3 * a_init + 0.7 * a_interp
        a_init = _normalize(a_init)
        sc_init = _score(a_init)

        time_per = max(30.0, (p4_end - time.monotonic()) / max(1, len(configs)))
        sub_dl = min(p4_end - 5, time.monotonic() + time_per)

        # Quick Adam
        c_i = _fft_convolve(a_init, a_init)
        mi = float(c_i.max())
        a_opt, sc = _adam_logspace(a_init, sub_dl, n_iters=50000,
                                    T_start=0.004 * mi, T_end=0.00004 * mi,
                                    lr_start=0.03, lr_end=0.002,
                                    active_k=4, decay=0.7)
        sc_full = _score(a_opt)
        print(f"  n={n_try} {kind}: init={sc_init:.8f} -> opt={sc_full:.8f} (t={time.monotonic()-t_start:.0f}s)")
        if sc_full < best_score:
            best_score = sc_full
            best_a = a_opt.copy()

    # === Phase 5: LP + de-corr cycle on best (1100-1400s) ===
    p5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < p5_end:
        print(f"\n--- Phase 5: LP + De-corr cycle (n={len(best_a)}) ---")
        n_cur = len(best_a)
        if n_cur <= 4096:
            a_lp2, _ = _lp_optimize(best_a, min(p5_end, time.monotonic() + 100),
                                     max_iters=15, max_constraints=50)
            sc = _score(a_lp2)
            print(f"  LP2: score={sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_a = a_lp2.copy()

        # De-corr refinement
        sub_dl = min(p5_end - 10, time.monotonic() + 150)
        a_dc2, _ = _decorr_iteration(best_a, sub_dl, n_iters=20000,
                                      alpha_start=0.2, alpha_end=0.003,
                                      top_k=5, eval_interval=200)
        sc_dc2 = _score(a_dc2)
        print(f"  De-corr2: score={sc_dc2:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_dc2 < best_score:
            best_score = sc_dc2
            best_a = a_dc2.copy()

    # === Phase 6: Fine Adam (1400-1650s) ===
    p6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < p6_end:
        print(f"\n--- Phase 6: Fine Adam (n={len(best_a)}) ---")
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_fine, sc_fine = _adam_logspace(best_a, p6_end, n_iters=150000,
                                         T_start=0.001 * max_cf, T_end=0.000005 * max_cf,
                                         lr_start=0.01, lr_end=0.0003,
                                         active_k=7, decay=0.8,
                                         eval_interval=80, plateau_frac=0.1)
        sc_fine_full = _score(a_fine)
        print(f"  Fine Adam: score={sc_fine_full:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_fine_full < best_score:
            best_score = sc_fine_full
            best_a = a_fine.copy()

    # === Phase 7: Perturbation restarts (1650-1770s) ===
    if time.monotonic() < deadline - 30:
        print(f"\n--- Phase 7: Perturbation restarts ---")
        n_fin = len(best_a)
        for attempt in range(3):
            if time.monotonic() > deadline - 35:
                break
            sigma = 0.015 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            time_per = max(15.0, (deadline - 30 - time.monotonic()) / (3 - attempt))
            sub_dl = min(deadline - 32, time.monotonic() + time_per)
            a_popt, _ = _adam_logspace(a_pert, sub_dl, n_iters=50000,
                                        T_start=0.001 * max_cp, T_end=0.000005 * max_cp,
                                        lr_start=0.015, lr_end=0.0005,
                                        active_k=5, decay=0.75)
            sc_p = _score(a_popt)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={sc_p:.10f}")
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_popt.copy()

    # === Phase 8: Final CD polish (last 30s) ===
    if time.monotonic() < deadline - 5:
        a_cd, _ = _cd_polish(best_a, deadline - 3, max_rounds=20, top_k=5)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
        print(f"\nCD polish: score={sc_cd:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
