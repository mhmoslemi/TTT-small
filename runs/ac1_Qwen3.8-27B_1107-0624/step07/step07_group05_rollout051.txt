"""
Asymmetric-sequence optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: monotone/asymmetric initializations exploit the fact that
for non-symmetric sequences, the convolution peak is off-center with fewer
overlapping terms, enabling scores below 2. Combined with log-sum-exp L-BFGS,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0] - c_sorted[-1] if len(c_sorted) > 1 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 4096.0)
        est_time = 0.01 * scale
        max_iter = min(6000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_project(a, deadline, n_iters=200, clip_frac=0.5):
    """
    Fienup-style spectral projection:
    1. Compute autocorrelation
    2. Clip/reduce the peak
    3. Reconstruct via phase preservation
    4. Rectify and normalize
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    N = 2 * n
    A = np.fft.rfft(a, N)
    phase = np.angle(A)
    mag = np.abs(A)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Create modified autocorrelation: reduce the peak
        c_mod = c.copy()
        # Clip all values above threshold
        threshold = clip_frac * c_max
        c_mod[c_mod > threshold] = threshold

        # Ensure non-negative (autocorrelation of real signal can have negative values
        # in the modified version, but we need |A|^2 >= 0)
        c_mod = np.maximum(c_mod, 0)

        # Get new magnitude spectrum
        C_mod = np.fft.rfft(c_mod, N)
        new_mag2 = np.real(C_mod)
        new_mag2 = np.maximum(new_mag2, 0)
        new_mag = np.sqrt(new_mag2)

        # Reconstruct with preserved phase
        a_new = np.fft.irfft(new_mag * np.exp(1j * phase), N)[:n]

        # Rectify
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        # Update phase for next iteration
        a = a_new
        A = np.fft.rfft(a, N)
        phase = np.angle(A)

    return best_a, best_sc


def _spectral_project_v2(a, deadline, n_iters=100):
    """
    Alternative spectral projection: iteratively flatten the autocorrelation
    by adjusting the magnitude spectrum.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    N = 2 * n

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Current spectrum
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Autocorrelation
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Target: make autocorrelation as flat as possible
        # Reduce values above the mean
        c_mean = c.mean()
        c_target = np.where(c > c_mean, c_mean + 0.3 * (c - c_mean), c)
        c_target = np.maximum(c_target, 0)

        # New magnitude from target autocorrelation
        C_t = np.fft.rfft(c_target, N)
        new_mag2 = np.maximum(np.real(C_t), 0)
        new_mag = np.sqrt(new_mag2)

        # Blend old and new magnitude (to preserve structure)
        alpha = 0.7
        blended_mag = alpha * mag + (1 - alpha) * new_mag

        # Reconstruct
        a_new = np.fft.irfft(blended_mag * np.exp(1j * phase), N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


def _pressure_equalize(a, deadline, n_iters=200, gamma=0.3):
    """
    KKT-motivated multiplicative update to equalize pressure across
    active convolution positions.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 20 == 0 or it == n_iters - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set gradient
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0001 * i)
        z = z - step * grad
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_ramp(n, exponent=1.0, reverse=False):
    """Create an asymmetric (ramp) initialization."""
    t = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        t = t[::-1]
    a = np.power(t, exponent)
    a = a / a.sum()
    return a


def _make_exp_ramp(n, alpha=1.0, reverse=False):
    """Exponential ramp initialization."""
    t = np.linspace(0, 1, n)
    if reverse:
        t = t[::-1]
    a = np.exp(alpha * t)
    a = a / a.sum()
    return a


def _make_blended(n, alpha=0.5):
    """Blend of ramp and uniform."""
    ramp = _make_ramp(n)
    uniform = np.ones(n) / n
    a = alpha * ramp + (1 - alpha) * uniform
    return a / a.sum()


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: Quick L-BFGS
    if time.monotonic() < deadline - min_remaining - 20:
        a1 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 20)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Spectral projection
    if time.monotonic() < deadline - min_remaining - 10:
        t_sp = min(deadline - min_remaining - 5, time.monotonic() + 30)
        a2, sc2 = _spectral_project(current_a, t_sp, n_iters=100, clip_frac=0.4)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: L-BFGS refinement
    if time.monotonic() < deadline - min_remaining - 10:
        a3 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 10)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 5:
        t_pe = min(deadline - min_remaining - 3, time.monotonic() + 30)
        a4, sc4 = _pressure_equalize(current_a, t_pe, n_iters=150, gamma=0.25)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Final L-BFGS
    if time.monotonic() < deadline - min_remaining:
        a5 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining)
        sc5 = _score(a5)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses asymmetric (ramp) initializations as the key differentiator,
    combined with L-BFGS, spectral projection, and pressure equalization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent)
    else:
        n_parent = 4096
        parent = _make_ramp(n_parent, exponent=1.0)
        sc_parent = _score(parent)
        print(f"No parent, using ramp n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent)

    # --- Phase 1: Optimize parent with L-BFGS ---
    print(f"\n[Phase 1] L-BFGS on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_opt = _run_lbfgs(parent, deadline, min_time_remaining=600)
        sc_opt = _score(a_opt)
        _update(a_opt, sc_opt)
        if sc_opt < best_score:
            parent = a_opt

    # --- Phase 2: Try asymmetric initializations at various n ---
    print(f"\n[Phase 2] Asymmetric initializations (t={time.monotonic()-t_start:.1f}s)")

    n_list = [4096, 8192, 6144, 5120]
    # Ensure we try the parent's n
    if n_parent not in n_list:
        n_list.insert(0, n_parent)

    for n_try in n_list:
        if time.monotonic() > deadline - 300:
            break

        # Generate multiple asymmetric initializations
        inits = []

        # Linear ramp
        inits.append(_make_ramp(n_try, exponent=1.0))
        # Quadratic ramp
        inits.append(_make_ramp(n_try, exponent=2.0))
        # Exponential ramps with different alphas
        for alpha in [0.5, 1.0, 2.0, 3.0]:
            inits.append(_make_exp_ramp(n_try, alpha=alpha))
        # Reverse ramps
        inits.append(_make_ramp(n_try, exponent=1.0, reverse=True))
        inits.append(_make_exp_ramp(n_try, alpha=2.0, reverse=True))
        # Blended
        inits.append(_make_blended(n_try, alpha=0.3))
        inits.append(_make_blended(n_try, alpha=0.7))

        # Also try resampled parent
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_resample = np.interp(x_new, x_old, parent)
            a_resample = np.maximum(a_resample, 0)
            if a_resample.sum() > 1e-10:
                inits.append(a_resample / a_resample.sum())

        # Pick top 3 by score and optimize each
        scored_inits = []
        for a_init in inits:
            sc = _score(a_init)
            scored_inits.append((sc, a_init))
        scored_inits.sort(key=lambda x: x[0])

        n_to_optimize = min(3, len(scored_inits))
        time_per_init = max(30, (deadline - 300 - time.monotonic()) / max(1, n_to_optimize))

        for sc_init, a_init in scored_inits[:n_to_optimize]:
            if time.monotonic() > deadline - 100:
                break
            if sc_init >= best_score:
                continue

            print(f"    n={n_try}, init_score={sc_init:.8f}")
            t_budget = min(time.monotonic() + time_per_init, deadline - 50)
            a_res, sc_res = _multi_stage_optimize(a_init, t_budget, min_remaining=10)
            _update(a_res, sc_res)

    # --- Phase 3: Spectral projection on best ---
    print(f"\n[Phase 3] Spectral projection (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        t_sp = min(deadline - 80, time.monotonic() + 60)
        a_sp, sc_sp = _spectral_project(best_seq, t_sp, n_iters=200, clip_frac=0.5)
        _update(a_sp, sc_sp)

        if sc_sp < best_score:
            a_sp2 = _run_lbfgs(a_sp, deadline, min_time_remaining=50)
            _update(a_sp2, _score(a_sp2))

        # Also try v2
        if time.monotonic() < deadline - 50:
            t_sp2 = min(deadline - 40, time.monotonic() + 30)
            a_sp2b, sc_sp2 = _spectral_project_v2(best_seq, t_sp2, n_iters=80)
            _update(a_sp2b, sc_sp2)

    # --- Phase 4: Pressure equalization + L-BFGS combo ---
    print(f"\n[Phase 4] Pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        t_pe = min(deadline - 30, time.monotonic() + 60)
        a_pe, sc_pe = _pressure_equalize(best_seq, t_pe, n_iters=300, gamma=0.2)
        _update(a_pe, sc_pe)

        if sc_pe < best_score and time.monotonic() < deadline - 20:
            a_pe2 = _run_lbfgs(a_pe, deadline, min_time_remaining=10)
            _update(a_pe2, _score(a_pe2))

    # --- Phase 5: Try different n values with best shape ---
    print(f"\n[Phase 5] Multi-n with best shape (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        n_explore = [n_best // 2, n_best * 2, n_best + 1024, n_best - 1024]
        n_explore = [x for x in n_explore if 128 <= x <= 32768]
        # Remove duplicates
        seen = set()
        n_explore_unique = []
        for x in n_explore:
            if x not in seen:
                seen.add(x)
                n_explore_unique.append(x)

        for n_e in n_explore_unique[:2]:
            if time.monotonic() > deadline - 30:
                break
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_e)
            a_e = np.interp(x_new, x_old, best_seq)
            a_e = np.maximum(a_e, 0)
            s_e = a_e.sum()
            if s_e < 1e-10:
                continue
            a_e = a_e / s_e
            sc_e = _score(a_e)
            print(f"    n={n_e}, resample score={sc_e:.8f}")
            if sc_e < best_score:
                _update(a_e, sc_e)
                # Quick L-BFGS
                if time.monotonic() < deadline - 20:
                    a_e2 = _run_lbfgs(a_e, deadline, min_time_remaining=10)
                    _update(a_e2, _score(a_e2))

    # --- Phase 6: Subgradient polish ---
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_sub, sc_sub = _subgrad_polish(best_seq, min(deadline - 5, time.monotonic() + 60),
                                        n_max_steps=2000, lr_init=0.01)
        _update(a_sub, sc_sub)

    # --- Phase 7: Final L-BFGS ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=1)
        _update(a_fin, _score(a_fin))

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_ramp(256, exponent=1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_ramp(256, exponent=1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
