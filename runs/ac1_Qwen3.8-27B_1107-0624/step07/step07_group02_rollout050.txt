Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_smoothed_max(c, k, beta):
    """Approximate max using top-k log-sum-exp."""
    c_max = float(c.max())
    # Use top-k elements for a tighter approximation
    if k < len(c):
        idx = np.argpartition(c, -k)[-k:]
        top = c[idx]
    else:
        top = c
    top_max = float(top.max())
    shifted = (top - top_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return top_max
    return top_max + beta * np.log(s)


def _smoobj(z, beta, n, topk=50):

    c = _full_convolve(a, a)
    return 2.0 * n * _topk_smoothed_max(c, topk, beta)


def _smograd(z, beta, n, topk=50):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    top = c[idx]
    top_max = float(top.max())
    shifted = (top - top_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    
    w = np.zeros_like(c)

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-15:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())

                a = a_try
                improved = True
                break

                tol_frac = max(tol_frac_end, tol_frac * 0.95)
            lr = min(lr_init, lr * 1.01)
    
    return best_a, _score(best_a)


def _peak_reducing_step(a, n_peaks=10, n_steps=50, deadline=None):
    """Directly try to reduce the top-k peaks in the convolution."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for step in range(n_steps):
        if deadline and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Find top-k peaks
        k = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        # For each peak position p, the gradient of c[p] w.r.t. a[j] is a[p-j] (with bounds)
        # We want to decrease all top peaks simultaneously
        # Use a weighted combination
        weights = np.zeros(len(c))
        for i, p in enumerate(top_idx):
            weights[p] = (len(top_idx) - i) / len(top_idx)
        
        # Gradient of the weighted sum of peaks
        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(weights, c))
        grad = 2.0 * n * a * (h - W)
        grad_t = grad - np.mean(grad)
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break
        alpha = 0.02 / (gnorm * np.max(a) + 1e-15)
        alpha = min(alpha, 0.1)
        
        improved = False
        for _ in range(30):

            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:

                continue

            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            break
    
    return best_a


def _random_fourier_init(n, rng, n_modes=None):
    """Generate a random non-negative sequence via random Fourier coefficients."""
    if n_modes is None:
        n_modes = max(5, n // 16)
    
    # Random log-amplitude and phase
    amps = np.exp(rng.standard_normal(n_modes) * 0.8)
    phases = rng.uniform(0, 2 * np.pi, n_modes)
    
    # Build signal
    t = np.arange(n)
    x = np.ones(n) * 0.5
    for k in range(1, n_modes):
        x += amps[k] * np.cos(2 * np.pi * k * t / n + phases[k])
    
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s


def _chebyshev_init(n, rng):
    """Initialize using Chebyshev polynomial structure."""
    t = np.linspace(-1, 1, n)
    # Random Chebyshev series
    coeffs = rng.standard_normal(min(n, 50)) * np.exp(-np.arange(min(n, 50)) * 0.1)
    x = np.zeros(n)
    for k, c in enumerate(coeffs):
        x += c * np.cos(k * np.arccos(np.clip(t, -1, 1)))
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):

    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    
    return best_a, best_sc

Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive L-BFGS with top-k log-sum-exp smoothing, active-set gradient
descent with proximity weighting, direct peak reduction, structured random

Key improvements over previous: top-k smoothing, proximity-weighted active set,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_smoothed_max(c, k, beta):
    """Approximate max using top-k log-sum-exp for tighter focus."""
    if k < len(c):
        idx = np.argpartition(c, -k)[-k:]
        top = c[idx]
    else:
        idx = np.arange(len(c))
        top = c
    top_max = float(top.max())
    shifted = (top - top_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return top_max
    return top_max + beta * np.log(s)


def _smoobj(z, beta, n, topk):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _topk_smoothed_max(c, topk, beta)


def _smograd(z, beta, n, topk):
    a = _softmax(z)
    c = _full_convolve(a, a)

    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    top = c[idx]
    top_max = float(top.max())
    shifted = (top - top_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

    w = np.zeros_like(c)
    if s < 1e-300 or not np.isfinite(s):
        w[int(idx[np.argmax(top)])] = 1.0
    else:

            w[idx_i] = e[i] / s

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        k = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        sorted_top = c[top_idx][np.argsort(c[top_idx])[::-1]]
        ranks = np.argsort(np.argsort(c[top_idx]))[::-1]

        weights = np.zeros(len(c))
        for rank, idx_i in enumerate(top_idx):
            weights[idx_i] = (k - rank) / (k * (k + 1) / 2.0)

        a_rev = a[::-1]
        wc = _full_convolve(weights, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(weights, c))
        grad = 2.0 * n * a * (h - W)
        grad_t = grad - np.mean(grad)

        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break
        alpha = min(0.03 / (gnorm * np.max(a) + 1e-15), 0.05)

        improved = False
        for _ in range(25):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            break

    return best_a, best_sc


def _random_fourier_init(n, rng, n_modes):
    """Generate a random non-negative sequence via random Fourier coefficients."""
    t = np.arange(n)
    x = np.ones(n) * 0.5
    for k in range(1, n_modes):
        amp = np.exp(rng.standard_normal() * 0.6) * 0.3
        phase = rng.uniform(0, 2.0 * np.pi)
        x += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / s


def _chebyshev_init(n, rng):
    """Initialize using random Chebyshev polynomial structure."""
    t = np.linspace(-1, 1, n)
    n_terms = min(n, 60)
    coeffs = rng.standard_normal(n_terms) * np.exp(-np.arange(n_terms) * 0.08)
    x = np.zeros(n)
    for k in range(n_terms):
        x += coeffs[k] * np.cos(k * np.arccos(np.clip(t, -1, 1)))
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return x / n if s < 1e-10 else x / s


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to reduce effective n."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _block_refine(a, deadline, block_size=256):
    """Refine blocks of the sequence independently for local improvement."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = (n + block_size - 1) // block_size

    for bi in range(n_blocks):
        if time.monotonic() > deadline:
            break
        start = bi * block_size
        end = min(start + block_size, n)
        if end - start < 4:
            continue

        # Try small perturbations of this block
        a_block = best_a[start:end].copy()
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            c = _full_convolve(best_a, best_a)
            c_max = float(c.max())
            # Gradient for elements in this block
            w = np.zeros(2 * n - 1)
            # Focus on peaks
            k = min(30, 2 * n - 1)
            top = np.argpartition(c, -k)[-k:]
            for tp in top:
                w[tp] = 1.0 / k

            a_rev = best_a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * n * best_a * (h - W)

            # Only update this block
            g_block = grad[start:end]
            g_block = g_block - g_block.mean()
            gnorm = np.linalg.norm(g_block)
            if gnorm < 1e-15:
                break
            alpha = 0.01 / (gnorm + 1e-15)
            a_try = best_a.copy()
            a_try[start:end] = np.maximum(a_try[start:end] - alpha * g_block, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try
                best_sc = sc_try
            else:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: aggressive L-BFGS with top-k smoothing,
    proximity-weighted active-set GD, peak reduction, block refinement,
    structured inits, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Main aggressive L-BFGS ===
    print(f"\n[P1] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a_opt = _run_lbfgs_aggressive(parent, deadline, min_time_remaining=350, topk_scale=40)
    sc_opt = _score(a_opt)
    if sc_opt < best_score:
        best_seq = a_opt.copy()
        best_score = sc_opt
        print(f"  NEW BEST: {best_score:.10f} (n={n_parent}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish (proximity weighted) ===
    print(f"\n[P2] Active-set polish proximity (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 280:
        pol_dl = min(deadline - 280, time.monotonic() + 100)
        a_pol, sc_pol = _active_set_polish_v2(best_seq, pol_dl,
                                              tol_frac_start=0.005, tol_frac_end=0.00005,
                                              lr_init=0.03, weight_mode="proximity")
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish_prox] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak reduction ===
    print(f"\n[P3] Peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a_pk, sc_pk = _peak_reducing_step(best_seq, n_peaks=30, n_steps=100,
                                          deadline=min(deadline - 250, time.monotonic() + 80))
        if sc_pk < best_score:
            best_seq = a_pk.copy()
            best_score = sc_pk
            print(f"  NEW BEST: {best_score:.10f} [peak_reduce] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n[P4] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_fine = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=200, topk_scale=30)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Shrinkage ===
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 180:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 180)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            if time.monotonic() < deadline - 150:
                a_shr2 = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=150, topk_scale=30)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Block refinement ===
    print(f"\n[P6] Block refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a_blk, sc_blk = _block_refine(best_seq, min(deadline - 120, time.monotonic() + 60),
                                      block_size=512)
        if sc_blk < best_score:
            best_seq = a_blk.copy()
            best_score = sc_blk
            print(f"  NEW BEST: {best_score:.10f} [block_refine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n_parent, 4096, 6144, 8192, 10240, 12288, 16384]))
    n_candidates = [n for n in n_candidates if 2048 <= n <= 16384]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue

        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)

        a_struct = _random_fourier_init(n_try, rng, n_modes=max(10, n_try // 32))
        sc_struct = _score(a_struct)

        a_cheb = _chebyshev_init(n_try, rng)
        sc_cheb = _score(a_cheb)

        # Pick best init
        candidates = [(sc_res, a_res, "resample"), (sc_struct, a_struct, "fourier"), (sc_cheb, a_cheb, "cheb")]
        candidates.sort(key=lambda x: x[0])
        sc_init, a_init, src = candidates[0]
        print(f"  n={n_try}: best_init={src} score={sc_init:.10f}")

        a_opt_n = _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=60, topk_scale=40)
        sc_opt_n = _score(a_opt_n)
        if sc_opt_n < best_score:
            best_seq = a_opt_n.copy()
            best_score = sc_opt_n
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_{src}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Diverse restarts ===
    print(f"\n[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    if time.monotonic() < deadline - 40:
        restarts = []
        restarts.append(("pert2", _perturb_log(best_seq, 0.02, rng)))
        restarts.append(("pert5", _perturb_log(best_seq, 0.05, rng)))
        restarts.append(("fourier", _random_fourier_init(n_best, rng, n_modes=n_best // 16)))
        restarts.append(("cheb", _chebyshev_init(n_best, rng)))

        for label, a_r in restarts:
            if time.monotonic() > deadline - 15:
                break
            a_opt_r = _run_lbfgs_aggressive(a_r, deadline, min_time_remaining=15, topk_scale=40)
            sc_opt_r = _score(a_opt_r)
            if sc_opt_r < best_score:
                best_seq = a_opt_r.copy()
                best_score = sc_opt_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final ultra-fine polishing ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_polish_v2(best_seq, deadline - 2,
                                              tol_frac_start=0.0005, tol_frac_end=0.000003,
                                              lr_init=0.005, weight_mode="exponential")
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            a_fin2 = _run_lbfgs_aggressive(best_seq, deadline - 1, min_time_remaining=1, topk_scale=20)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
