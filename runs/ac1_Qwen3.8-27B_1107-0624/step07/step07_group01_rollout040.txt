"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initializations (parent, windows, chirps)
2. L-BFGS with adaptive log-sum-exp smoothing
3. LP-based active-set refinement (novel: minimize max convolution via linear programming)
4. Subgradient polishing
5. Multi-length exploration
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_indices(n):
    """Build row/col indices for convolution matrix C where C@d = conv(a,d)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        n_elem = j_end - j_start + 1
        if n_elem > 0:
            js = np.arange(j_start, j_end + 1)
            rows_list.append(np.full(n_elem, k, dtype=np.int32))
            cols_list.append(js.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 70,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

    n_rows = 2 * n - 1

    all_rows = np.concatenate([rows, np.arange(n_rows, dtype=np.int32)])
    all_cols = np.concatenate([cols, np.full(n_rows, n, dtype=np.int32)])

    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([0.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_aa = _full_convolve(a, a)
        M = float(conv_aa.max())

        # Target: reduce M by a small amount
        target_M = M * reduction_factor

        # LP: find d such that conv(a,a)[k] + 2*conv(a,d)[k] <= target_M
        # This is a feasibility problem. If infeasible, increase target_M.
        C_data = 2.0 * a[rows - cols]
        all_data = np.concatenate([C_data, -np.ones(n_rows)])
        A_ub = csr_matrix((all_data, (all_rows, all_cols)), shape=(n_rows, n + 1))
        b_ub = np.full(n_rows, target_M) - conv_aa

        lower_bounds = np.append(-a, 0.0)
        upper_bounds = np.full(n + 1, np.inf)
        bounds = list(zip(lower_bounds, upper_bounds))

        c_zero = np.zeros(n + 1)  # Feasibility: minimize 0

        result = linprog(c_zero, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')

        if result.status == 0:
            d = result.x[:n]
            d_norm = np.linalg.norm(d)
            if d_norm > 1e-14:
                # Feasible! Take the step.
                neg_mask = d < 0
                if neg_mask.any():
                    max_alpha = (-a[neg_mask] / d[neg_mask]).min()
                else:
                    max_alpha = 1.0
                max_alpha = max(0.05, min(max_alpha * 0.9, 1.0))

                a_try = np.maximum(a + max_alpha * d, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        current_M = float(_full_convolve(a, a).max())
                        if it % 30 == 0:
                            print(f"    LP-reduce iter {it}: score={best_sc:.10f}, M={current_M:.6f}")
            else:
                # d=0 is feasible, meaning current M <= target_M
                # We can directly reduce target_M more aggressively
                pass
        elif result.status == 2:  # Infeasible
            # Try a less aggressive reduction
            stalled = True
            for _ in range(5):
                target_M = target_M * 1.001 + M * 0.001
                b_ub = np.full(n_rows, target_M) - conv_aa
                result2 = linprog(c_zero, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                  bounds=bounds, method='highs')
                if result2.status == 0:
                    d = result2.x[:n]
                    d_norm = np.linalg.norm(d)
                    if d_norm > 1e-14:
                        neg_mask = d < 0
                        if neg_mask.any():
                            max_alpha = (-a[neg_mask] / d[neg_mask]).min()
                        else:
                            max_alpha = 1.0
                        max_alpha = max(0.05, min(max_alpha * 0.9, 1.0))
                        a_try = np.maximum(a + max_alpha * d, 0.0)
                        s_try = a_try.sum()
                        if s_try > 1e-15:
                            a_try = a_try / s_try
                            sc_try = _score(a_try)
                            if sc_try < best_sc - 1e-15:
                                best_sc = sc_try
                                best_a = a_try.copy()
                                a = a_try
                                current_M = float(_full_convolve(a, a).max())
                                stalled = False
                                break
            if stalled:
                break
        else:
            break

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    t = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / (n * sigma_frac)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _direct_subgrad(a_init, deadline, k_active=3, max_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.3 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        a_try = a - step_size * g / np.sqrt(n)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        sc_cur = _score(a)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        if sc_try < sc_cur - 1e-16:
            a = a_try
            step_size = min(step_size * 1.01, 0.5)
        else:
            stall += 1
            step_size *= 0.93

        if stall > 200:
            break

    return best_a, best_sc


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=10.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_budget = time.perf_counter() + time_per_hop * 0.6

        pert_type = h % 3
        z = np.log(np.maximum(best_a, 1e-18))
        if pert_type == 0:
            sigma = 0.04 + 0.02 * h
            z = z + sigma * rng.standard_normal(n)
        elif pert_type == 1:
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 16)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        else:
            n_blocks = 3 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)

        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        max_iter = min(3000, max(100, int((hop_budget - time.perf_counter()) / (0.004 * max(1, n / 4096)))))
        if max_iter > 50:
            a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, LP-based refinement, and subgradient methods.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={(prev - best_score_val) if np.isfinite(prev) else 0:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Try parent at different lengths
    parent_by_len = {}
    for target_n in [2048, 4096, 8192, 16384]:
        if time.monotonic() > deadline - 1700:
            break
        if parent is not None:
            a_interp = _interpolate_to_n(parent, target_n)
            sc = _score(a_interp)
            parent_by_len[target_n] = a_interp
            print(f"  Parent @ n={target_n}: score={sc:.10f}")
            try_update(a_interp, f"parent_n{target_n}")

    # Structured initializations at n=4096
    n_init = 4096
    structured = [
        ("chirp1", _make_chirp(n_init, freq=1.0, eps=0.3)),
        ("chirp2", _make_chirp(n_init, freq=2.0, eps=0.2)),
        ("taper05", _make_tapered(n_init, power=0.5)),
        ("taper10", _make_tapered(n_init, power=1.0)),
        ("gauss", _make_gaussian(n_init, sigma_frac=0.35)),
        ("gauss_wide", _make_gaussian(n_init, sigma_frac=0.5)),
    ]
    for name, a_s in structured:
        sc = _score(a_s)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_s, name)

    if best_seq is None:
        best_seq = _make_chirp(n_init, freq=1.0, eps=0.3)
        best_score_val = _score(best_seq)
        best_n = n_init

    print(f"  Phase 1 done. Best: {best_score_val:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 2: L-BFGS at multiple lengths ===
    print(f"\n=== Phase 2: L-BFGS optimization ===")

    lengths_to_optimize = []
    for ln in [4096, 8192, 16384]:
        if parent is not None and ln in parent_by_len:
            lengths_to_optimize.append((ln, parent_by_len[ln]))
        elif ln == n_init:
            lengths_to_optimize.append((ln, best_seq.copy() if len(best_seq) == ln else _make_chirp(ln, 1.0, 0.3)))

    # Ensure we have at least 4096
    if not any(ln == 4096 for ln, _ in lengths_to_optimize):
        lengths_to_optimize.insert(0, (4096, _make_chirp(4096, 1.0, 0.3)))

    time_per_length = min(180.0, (deadline - time.monotonic() - 1200.0) / max(1, len(lengths_to_optimize)))

    for ln, a_start in lengths_to_optimize:
        if time.monotonic() > deadline - 1000:
            break
        print(f"  Optimizing n={ln} (time budget: {time_per_length:.0f}s)...")
        phase_deadline = min(deadline - 1000, time.monotonic() + time_per_length)
        a_opt = _adaptive_lbfgs(a_start, phase_deadline, reserve_time=30.0)
        try_update(a_opt, f"lbfgs_n{ln}")

    lp_candidates = []

    if best_seq is not None and len(best_seq) <= 8192:

            remaining_lp = min(deadline - 200, time.monotonic() + time_per_lp * 0.5)
