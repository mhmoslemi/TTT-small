3. Add Nesterov momentum to subgradient polishing
4. Use KKT-based active set identification for better convergence
5. Try multiple n values including 32768
6. Implement a "peak equalization" step that directly targets the spread of top autocorrelation values

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive adaptive L-BFGS with ultra-fine log-sum-exp cooling
2. Nesterov momentum subgradient with adaptive active set
3. KKT-inspired peak equalization
4. Multi-length exploration (n=8192 to 32768)
5. Multiplicative (log-domain) gradient flow with restarts
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        from scipy.optimize import minimize
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs_v2(a_start, deadline, reserve_time=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Much finer beta schedule - go much lower
    ratios = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
              0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
              5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8, 5e-9, 2e-9,
              1e-9, 5e-10, 2e-10, 1e-10]
    betas = max_c * np.array(ratios)

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        # Scale max_iter based on remaining time and n
        scale = max(1.0, (n / 4096.0) ** 1.3)
        est_time_per_iter = 0.003 * scale
        max_iter = min(20000, max(200, int(remaining / (est_time_per_iter * 30))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.2e}: score={current_sc:.10f} (improved)")
        else:
            print(f"    beta={beta:.2e}: score={sc:.10f} (no improvement)")

        if current_sc < 1.5001:
            break

    return current_a


def _momentum_subgrad(a_init, deadline, k_active=5, max_steps=100000,
                      lr_init=0.03, momentum=0.9):
    """Nesterov momentum subgradient in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    v = np.zeros_like(z)  # momentum
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        z_look = z - momentum * v
        a = _softmax(z_look)
        c = _full_convolve(a, a)

        # Adaptive active set: top k_active peaks
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Update momentum
        v = momentum * v - lr * grad / (a * n + 1e-30)
        z = z + v
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        # Periodic evaluation
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.01
                stall = 0
            else:
                stall += 1
                if stall > k_active * 5:
                    lr *= 0.5
                    v *= 0.5
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _peak_equality_refine(a_init, deadline, max_iter=2000):
    """Iteratively equalize the top peaks by redistributing mass."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        # Find all peaks within 1% of max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break

        # Compute the "excess" at each active peak
        excess = c[active] - c[active].mean()
        if np.abs(excess).max() < 1e-15:
            break

        # Build gradient: for each active peak k, the gradient is 2*a[k-m]
        # We want to reduce the spread, so move mass from high-peak contributions
        # to low-peak contributions
        grad = np.zeros(n)
        for k in active:
            weight = (c[k] - c[active].mean()) / (np.abs(excess).max() + 1e-30)
            # Gradient of b[k] w.r.t. a[m] is 2*a[k-m] for valid k-m
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += weight * 2.0 * a[k - np.arange(lo, hi + 1)]

        # Normalize and apply step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search
        alpha = 0.1 / n
        improved = False
        for _ in range(15):
            a_try = a - alpha * grad / np.linalg.norm(grad)
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            # Try adding a small random perturbation
            rng = np.random.default_rng(it + 42)
            a_pert = a + 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            sc_pert = _score(a_pert)
            if sc_pert < best_sc:
                best_sc = sc_pert
                best_a = a_pert.copy()
                a = a_pert
            else:
                break

    return best_a, best_sc


def _kkt_check(a, k_active=5):
    """Check KKT conditions and return the residual."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    idx = np.argsort(c)[::-1][:k_active]

    # Check: sum lambda_k * a[k-m] should be constant
    # Solve least squares: A * lambda = mu * 1
    # where A[m, j] = a[idx[j] - m]
    A = np.zeros((n, k_active))
    for j, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            A[lo:hi + 1, j] = a[k - np.arange(lo, hi + 1)]

    # Want A @ lambda ≈ mu * 1
    # Eliminate mu: (A[m] - A[0]) @ lambda = 0 for m=1,...,n-1
    if k_active >= 2:
        B = A[1:] - A[0:1]
        # Find lambda that minimizes ||B @ lambda|| subject to sum lambda = 1, lambda >= 0
        # Use least squares with constraint
        from scipy.optimize import minimize
        x0 = np.ones(k_active) / k_active

        def obj(x):
            r = B @ x
            return np.dot(r, r)

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0}]
        bounds = [(0, 1)] * k_active
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 100})
            if res.success:
                residual = np.linalg.norm(B @ res.x)
                return residual, idx, res.x
        except Exception:
            pass

    # Fallback: just report the spread of active peaks
    spread = c[idx].max() - c[idx].min()
    return spread, idx, None


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _quick_optimize(a_start, deadline, n_betas=10):
    """Quick multi-beta L-BFGS for time-constrained optimization."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    current_a = a_start.copy()
    current_sc = _score(current_a)

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return current_a

    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001, 5e-5])[:n_betas]

    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        time_left = deadline - 5 - time.monotonic()
        max_iter = min(5000, max(100, int(time_left / 0.02)))
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def propose_candidate(seed=42, budget_s=1800):

    best_seq = None
    best_score_val = np.inf
    best_n = 16384

        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()

        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Load parent and quick init ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={best_score_val:.10f}")
    else:
        n_init = 16384
        a_chirp = _make_chirp(n_init, freq=1.3, eps=0.4)
        try_update(a_chirp, "chirp_fallback")
        parent = a_chirp
        print(f"  Fallback chirp: n={n_init}, score={best_score_val:.10f}")

    # === Phase 2: Aggressive L-BFGS (main workhorse) ===
    print(f"\n=== Phase 2: Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    # Reserve time for later phases
    reserve = max(300.0, 0.25 * float(budget_s))
    a_p2 = _adaptive_lbfgs_v2(best_seq, deadline, reserve_time=reserve)
    try_update(a_p2, "main_lbfgs_v2")

    # === Phase 3: Momentum subgradient ===
    print(f"\n=== Phase 3: Momentum subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - reserve * 0.6:
        sub_deadline = min(deadline - reserve * 0.5, time.monotonic() + 200)
        a_sg, _ = _momentum_subgrad(best_seq, sub_deadline, k_active=5,
                                     max_steps=50000, lr_init=0.02, momentum=0.85)
        try_update(a_sg, "mom_sg_k5")

    if time.monotonic() < deadline - reserve * 0.5:
        sub_deadline = min(deadline - reserve * 0.4, time.monotonic() + 150)
        a_sg2, _ = _momentum_subgrad(best_seq, sub_deadline, k_active=3,
                                      max_steps=40000, lr_init=0.015, momentum=0.9)
        try_update(a_sg2, "mom_sg_k3")

    # === Phase 4: Peak equality refinement ===
    print(f"\n=== Phase 4: Peak equality (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - reserve * 0.4:
        peak_deadline = min(deadline - reserve * 0.3, time.monotonic() + 120)
        a_pe, _ = _peak_equality_refine(best_seq, peak_deadline, max_iter=3000)
        try_update(a_pe, "peak_eq")

    # === Phase 5: Second L-BFGS round (finer) ===
    print(f"\n=== Phase 5: Second L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - reserve * 0.3:
        a_p5 = _adaptive_lbfgs_v2(best_seq, deadline, reserve_time=reserve * 0.25)
        try_update(a_p5, "lbfgs_round2")

    # === Phase 6: KKT diagnostic and targeted fix ===
    print(f"\n=== Phase 6: KKT diagnostic (t={time.monotonic()-t_start:.1f}s) ===")

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())

            # Very fine L-BFGS
            beta = mb * 1e-10
            time_left = deadline - 10 - time.monotonic()
            max_iter = min(8000, max(500, int(time_left / 0.01)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Quick momentum SG
            if time.monotonic() < deadline - 10:

                                            k_active=3, max_steps=20000,
                                            lr_init=0.01, momentum=0.92)
                try_update(a_s, f"final_sg_{rnd}")

            # Peak equality
            if time.monotonic() < deadline - 10:
                a_pe, _ = _peak_equality_refine(best_seq, deadline - 8, max_iter=1000)
                try_update(a_pe, f"final_pe_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:

        if s < 0.01:

        sc_final = _score(final)
