"""
Newton active-set optimizer with alternating w-update and CG-based Newton step,
using FFT-based Hankel matvec for efficiency. Multiple n sizes explored.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_stable(z, beta):
    """Compute softmax(z / beta) stably."""
    z_scaled = z / beta
    z_scaled = z_scaled - np.max(z_scaled)
    e = np.exp(np.clip(z_scaled, -700, 0))
    s = e.sum()
    if s < 1e-300:
        # All mass on the max
        result = np.zeros_like(z)
        result[np.argmax(z)] = 1.0
        return result
    return e / s

def _hankel_matvec(h, v):
    """Compute H @ v where H[i,j] = h[i+j], i,j = 0..n-1.
    h has length 2n-1."""
    n = len(v)
    v_flip = v[::-1]
    conv = _full_convolve(h, v_flip)
    # (H @ v)[i] = conv[i + n - 1] for i = 0..n-1
    return conv[n - 1:2 * n - 1]

def _newton_cg(h, b, n, mu=1e-6, max_iter=200, tol=1e-10, x0=None):
    """Solve (2*H + mu*I) x = b using CG, where H is Hankel with diag h."""
    if x0 is None:
        x = np.zeros(n)
    else:
        x = x0.copy()
    
    r = b - 2.0 * _hankel_matvec(h, x) - mu * x
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    
    for _ in range(max_iter):
        Hp = 2.0 * _hankel_matvec(h, p) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        r_norm = float(np.dot(r_new, r_new))
        if np.sqrt(r_norm) < tol * b_norm:
            x = x_new
            break
        beta_cg = r_norm / rs if rs > 1e-30 else 0.0

        rs = r_norm
        r = r_new
        x = x_new

def _newton_step(a, beta, deadline, max_iter=50):
    """One phase of Newton iterations with fixed beta."""
    n = len(a)
    a_work = a.copy()
    a_work = np.maximum(a_work, 0)
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s
    
    best_a = a_work.copy()
    best_max_c = float(_full_convolve(a_work, a_work).max())
    
    x_prev = None
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a_work, a_work)
        c_max = float(c.max())
        
        # Compute w = softmax(c / beta)
        w = _softmax_stable(c, beta)
        
        # Gradient: 2 * (H @ a) where H[i,j] = w[i+j]
        grad = 2.0 * _hankel_matvec(w, a_work)
        
        # Project to tangent space (sum = 0)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break
        
        # Newton step: solve (2H + mu*I) dx = -grad_t
        mu = 1e-6
        dx = _newton_cg(w, -grad_t, n, mu=mu, max_iter=100, tol=1e-10, x0=x_prev)
        x_prev = dx
        dx = dx - np.mean(dx)  # Ensure sum(dx) = 0
        
        # Line search
        alpha = 1.0
        a_new = a_work.copy()
        for ls in range(40):
            a_try = a_work + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-10:
                break
            a_new = a_work
        
        a_work = a_new
        curr_max = float(_full_convolve(a_work, a_work).max())
        if curr_max < best_max_c:
            best_max_c = curr_max
            best_a = a_work.copy()
    
    return best_a

def _run_newton_pipeline(a_init, deadline):

def _smoobj(z, beta, n):
    """Smoothed objective for L-BFGS."""
    a = _softmax_stable(z, 1.0)  # softmax with beta=1 for the parameterization
    # Actually, let's use a different parameterization
    # a = exp(z) / sum(exp(z))
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    a = e / e.sum()
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    ex = np.exp(np.clip(shifted, -700, 0))
    s = ex.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    """Gradient of smoothed objective."""
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    a = e / e.sum()
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    ex = np.exp(np.clip(shifted, -700, 0))
    s = ex.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_phase(a_init, deadline, min_time=60.0):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap, max_c * 0.02)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                   0.03125, 0.015625, 0.0078125,
                                   0.0039, 0.00195, 0.00098, 0.00049])
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))
        
        def fun_local(z_local):
            return _smoobj(z_local, beta, n)
        
        def jac_local(z_local):
            return _smograd(z_local, beta, n)
        
        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxcor': 50, 'maxls': 50}
            )
            z_new = result.x
            a_new = np.exp(z_new - np.max(z_new))
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = z_new - z_new.mean()
        except Exception:
            pass
    
    return current_a

def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.01):
    """Subgradient descent on the max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = np.exp(z - np.max(z))
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set gradient
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = np.exp(z - np.max(z))
            a_check = a_check / a_check.sum()
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
    
    a_final = np.exp(z - np.max(z))
    a_final = a_final / a_final.sum()
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

"""
Newton active-set optimizer using direct Hankel system solve via CG with FFT matvec.
Alternates between w-update (softmax of autocorrelation) and a-update (solve H@x=1).
Includes L-BFGS warm-start, multi-resolution exploration, and subgradient polish.

import numpy as np
import time
from scipy.optimize import minimize
from functools import partial


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_stable(z, beta):
    z_scaled = z / beta
    z_scaled = z_scaled - np.max(z_scaled)
    e = np.exp(np.clip(z_scaled, -700, 0))
    s = e.sum()
    if s < 1e-300:
        result = np.zeros_like(z)
        result[np.argmax(z)] = 1.0
        return result
    return e / s


def _hankel_matvec(h, v):
    """Compute H @ v where H[i,j] = h[i+j], i,j = 0..n-1."""
    n = len(v)
    v_flip = v[::-1]
    conv = _full_convolve(h, v_flip)
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(h, b, n, mu=1e-8, max_iter=300, tol=1e-10, x0=None):
    """Solve (2*H + mu*I) x = b where H is Hankel with h[i+j] entries."""
    if x0 is None:
        x = np.zeros(n)
    else:
        x = x0.copy()
    r = b - 2.0 * _hankel_matvec(h, x) - mu * x
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = 2.0 * _hankel_matvec(h, p) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_norm = float(np.dot(r, r))
        if np.sqrt(r_norm) < tol * b_norm:
            break
        beta_cg = r_norm / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_norm
    return x


def _cg_solve_hankel_direct(h, b, n, max_iter=300, tol=1e-10, x0=None):
    """Solve H x = b where H is Hankel with h[i+j] entries (no 2x factor)."""
    if x0 is None:
        x = np.zeros(n)
    else:
        x = x0.copy()
    r = b - _hankel_matvec(h, x)
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hankel_matvec(h, p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_norm = float(np.dot(r, r))
        if np.sqrt(r_norm) < tol * b_norm:
            break
        beta_cg = r_norm / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_norm
    return x

def _smoobj(z, beta, n):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    a = e / e.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    ex = np.exp(np.clip(shifted, -700, 0))
    s = ex.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    a = e / e.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    ex = np.exp(np.clip(shifted, -700, 0))
    s = ex.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        return z_init

    est_time = 0.005 * max(1.0, n / 4096.0)
    max_iter = min(max_iter, max(20, int((deadline - min_time - time.monotonic()) / (est_time * 12))))
    if max_iter < 10:
        return z_init

    f = partial(_smoobj, beta=beta, n=n)
    j = partial(_smograd, beta=beta, n=n)

    try:
        result = minimize(
            f, z_init, jac=j, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxcor': 50, 'maxls': 50}
        )
        return result.x
    except Exception:
        return z_init


def _lbfgs_phase(a_init, deadline, min_time=120):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                   0.03125, 0.015625, 0.0078125,
                                   0.004, 0.002, 0.001])

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for beta in betas:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break

        z = _lbfgs_run(z, beta, n, 3000, deadline, min_time=min_time)
        a_new = np.exp(z - np.max(z))
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc_new
            z = z - z.mean()

    return current_a


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01):
    """Subgradient descent on the max autocorrelation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = np.exp(z - np.max(z))
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = np.exp(z - np.max(z))
            a_check = a_check / a_check.sum()
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = np.exp(z - np.max(z))
    a_final = a_final / a_final.sum()
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc
