One more thought: the **p-norm continuation** is a key difference from the previous program's log-sum-exp approach. The p-norm has the advantage that:
- For small p, the gradient is "spread out" (all convolution values contribute).
- For large p, the gradient concentrates on the peak.
- The transition is smooth and well-behaved.

The gradient of the p-norm: d/d_a (sum_k b[k]^p)^(1/p) = (1/p) * (sum_k b[k]^p)^(1/p - 1) * p * sum_k b[k]^(p-1) * db[k]/da

= (sum_k b[k]^p)^(1/p - 1) * sum_k b[k]^(p-1) * db[k]/da

For p = 2: this is (1/sqrt(sum b^2)) * sum_k b[k] * db[k]/da = (1/||b||_2) * <b, db/da>

This is a weighted average of the gradients, with weights proportional to b[k] (for p=2) or b[k]^(p-1) (for general p).

As p → ∞, the weights concentrate on the maximum b[k], recovering the max gradient.

This is very similar to the log-sum-exp gradient (which uses softmax weights), but the p-norm has a different weighting that might work better in practice.

def _kkt_residual(a, n):

    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0
    # Active set: top 10% of convolution values
    sorted_c = np.sort(c)[::-1]
    n_active = max(3, int(0.05 * len(c)))
    threshold = sorted_c[min(n_active - 1, len(c) - 1)]
    tol = max(1e-6 * c_max, c_max - threshold)
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True

def _make_profile(n, kind="uniform"):

    elif kind == "tapered":

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":

        a = 1.0 + 0.25 * np.cos(4.0 * np.pi * t)

    elif kind == "chirp":

        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * (3 * t + 0.5 * t * t))

    elif kind == "random_sym":

    else:

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()

    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score_val = sc_parent
        start_n = len(parent)
    else:
        a0 = _make_profile(4096, "uniform")
        best_seq = a0.copy()
        best_score_val = _score(a0)
        start_n = 4096
        print(f"No parent. uniform n={start_n}, score={best_score_val:.10f}")

    # === Phase 1: Initial L-BFGS with lsexp ===
    print(f"\n=== Phase 1: L-BFGS lsexp on n={start_n} ===")
    t_p = time.monotonic()
    n = len(best_seq)
    z = np.log(np.maximum(best_seq, 1e-15))
    z = z - z.mean()
    c_init = _full_convolve(best_seq, best_seq)
    mc = float(c_init.max())
    betas = mc * np.array([0.1, 0.04, 0.015, 0.005, 0.002, 0.0008, 0.0003, 0.0001, 4e-5, 1.5e-5, 6e-6, 2e-6])
    phase_deadline = deadline - 900
    for i, beta in enumerate(betas):
        if time.monotonic() > phase_deadline:
            break
        remaining = phase_deadline - time.monotonic()
        if remaining < 20:
            break
        z, niter = _lbfgs_lsexp(z, beta, n, phase_deadline, max_iter=min(4000, max(200, int(remaining / 0.2))))
        a_new = _softmax(z)
        sc = _score(a_new)
        if sc < best_score_val - 1e-14:
            best_seq = a_new.copy()
            best_score_val = sc
            print(f"    beta={beta:.2e} (p1): score={sc:.10f} iters={niter}")
    print(f"  Phase 1 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 2: P-norm continuation ===
    print(f"\n=== Phase 2: P-norm continuation ===")
    t_p = time.monotonic()
    z = np.log(np.maximum(best_seq, 1e-15))
    z = z - z.mean()
    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512]
    phase_deadline = deadline - 600
    for p in p_schedule:
        if time.monotonic() > phase_deadline:
            break
        remaining = phase_deadline - time.monotonic()
        if remaining < 15:
            break
        z, niter = _lbfgs_pnorm(z, p, n, phase_deadline, max_iter=min(3000, max(100, int(remaining / 0.15))))
        a_new = _softmax(z)
        sc = _score(a_new)
        if sc < best_score_val - 1e-14:
            best_score_val = sc
            best_seq = a_new.copy()
            print(f"    p={p}: score={sc:.10f} iters={niter}")
    print(f"  Phase 2 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 3: Fine L-SEXP ===
    print(f"\n=== Phase 3: Fine L-SEXP ===")
    t_p = time.monotonic()
    n = len(best_seq)
    z = np.log(np.maximum(best_seq, 1e-15))
    z = z - z.mean()
    c_b = _full_convolve(best_seq, best_seq)
    mb = float(c_b.max())
    fine_betas = mb * np.array([1e-5, 4e-6, 1.5e-6, 6e-7, 2e-7, 8e-8, 3e-8, 1e-8, 4e-9])
    phase_deadline = deadline - 350
    for beta in fine_betas:
        if time.monotonic() > phase_deadline:
            break
        remaining = phase_deadline - time.monotonic()
        if remaining < 10:
            break
        z, niter = _lbfgs_lsexp(z, beta, n, phase_deadline, max_iter=min(5000, max(200, int(remaining / 0.15))))
        a_new = _softmax(z)
        sc = _score(a_new)
        if sc < best_score_val - 1e-14:
            best_score_val = sc
            best_seq = a_new.copy()
            print(f"    beta={beta:.2e}: score={sc:.10f} iters={niter}")
    print(f"  Phase 3 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 4: De-peak targeted local search ===
    print(f"\n=== Phase 4: De-peak local search ===")
    t_p = time.monotonic()
    if time.monotonic() < deadline - 250:
        a_dp, sc_dp = _de_peak_search(best_seq, deadline - 250, max_evals=80000)
        try_update(a_dp, "de_peak")
    print(f"  Phase 4 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 5: KKT diagnostics + support adjustment ===
    print(f"\n=== Phase 5: KKT + support ===")
    t_p = time.monotonic()
    n = len(best_seq)
    if time.monotonic() < deadline - 200:
        a = best_seq.copy()
        s = a.sum()
        a = a / s
        kkt_res = _kkt_residual(a, n)
        print(f"  KKT residual: {kkt_res:.6e}")

        # Try zeroing out small elements and re-optimizing
        if kkt_res > 0.01:
            a_zero = a.copy()
            threshold = np.sort(a_zero)[int(0.9 * n)]  # zero out bottom 10%
            a_zero[a_zero < threshold] = 0.0
            sz = a_zero.sum()
            if sz > 0.01:
                a_zero = a_zero / sz
                z_zero = np.log(np.maximum(a_zero, 1e-15))
                z_zero = z_zero - z_zero.mean()
                c_z = _full_convolve(a_zero, a_zero)
                mb_z = float(c_z.max())
                for beta in [mb_z * 1e-4, mb_z * 1e-5, mb_z * 1e-6]:
                    if time.monotonic() > deadline - 180:
                        break
                    z_zero, _ = _lbfgs_lsexp(z_zero, beta, n, deadline - 180, max_iter=2000)
                    a_test = _softmax(z_zero)
                    # Project back: zero out the same elements
                    a_test[a_test < threshold * 0.1] = 0
                    st = a_test.sum()
                    if st > 0.01:
                        a_test = a_test / st
                        sc_test = _score(a_test)
                        if sc_test < best_score_val - 1e-14:
                            best_score_val = sc_test
                            best_seq = a_test.copy()
                            print(f"    support-zeroed: score={sc_test:.10f}")
    print(f"  Phase 5 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    t_p = time.monotonic()
    target_ns = []
    if start_n < 16384:
        target_ns.append(16384)
    if start_n < 8192:
        target_ns.append(8192)

    for tn in sorted(target_ns, reverse=True):
        if time.monotonic() > deadline - 120:
            break
        if tn == start_n:
            continue
        print(f"  Trying n={tn}...")
        a_interp = _interpolate_to_n(best_seq, tn)
        sc_interp = _score(a_interp)
        print(f"    Interp score: {sc_interp:.10f}")
        try_update(a_interp, f"interp_n{tn}")

        n_t = tn
        z_t = np.log(np.maximum(best_seq, 1e-15))
        z_t = z_t - z_t.mean()
        # Re-interpolate in z-space for better quality
        a_t = _interpolate_to_n(best_seq, tn)
        z_t = np.log(np.maximum(a_t, 1e-15))
        z_t = z_t - z_t.mean()

        c_t = _full_convolve(a_t, a_t)
        mb_t = float(c_t.max())
        for beta in [mb_t * 0.01, mb_t * 0.001, mb_t * 0.0001, mb_t * 1e-5]:
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 10:
                break
            z_t, _ = _lbfgs_lsexp(z_t, beta, n_t, deadline - 100,
                                   max_iter=min(3000, max(100, int(remaining / 0.3))))
            a_test = _softmax(z_t)
            sc_test = _score(a_test)
            if sc_test < best_score_val - 1e-14:
                best_score_val = sc_test
                best_seq = a_test.copy()
                print(f"    n={tn} beta={beta:.2e}: score={sc_test:.10f}")
    print(f"  Phase 6 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart ===")
    t_p = time.monotonic()
    n = len(best_seq)
    restart_configs = [
        ("gauss_003", "gauss", 0.003),
        ("gauss_01", "gauss", 0.01),
        ("gauss_03", "gauss", 0.03),
        ("gauss_1", "gauss", 0.1),
        ("fourier", "fourier", 0.0),
        ("fourier_big", "fourier_big", 0.0),
    ]
    # Also try convex combinations with other profiles
    other_profiles = ["tapered", "bimodal", "chirp", "random_sym"]
    for pk in other_profiles:
        restart_configs.append((f"mix_{pk}", "mix", pk))

    for name, mode, param in restart_configs:
        if time.monotonic() > deadline - 30:
            break

        if mode == "gauss":
            p_start = _perturb(best_seq, rng, scale=param)
        elif mode == "fourier":
            p_start = _fourier_perturb(best_seq, rng, n_modes=8, amp=0.02)
        elif mode == "fourier_big":
            p_start = _fourier_perturb(best_seq, rng, n_modes=12, amp=0.05)
        elif mode == "mix":
            other = _make_profile(n, param)
            alpha = 0.3
            p_start = (1 - alpha) * best_seq + alpha * other
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            continue

        sc_start = _score(p_start)
        z_r = np.log(np.maximum(p_start, 1e-15))
        z_r = z_r - z_r.mean()
        c_r = _full_convolve(p_start, p_start)
        mb_r = float(c_r.max())

        for beta in [mb_r * 0.01, mb_r * 0.001, mb_r * 1e-4, mb_r * 1e-5]:
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - 25 - time.monotonic()
            if remaining < 8:
                break
            z_r, _ = _lbfgs_lsexp(z_r, beta, n, deadline - 25,
                                   max_iter=min(2000, max(50, int(remaining / 0.15))))
            a_r = _softmax(z_r)
            sc_r = _score(a_r)
            if sc_r < best_score_val - 1e-14:
                best_score_val = sc_r
                best_seq = a_r.copy()
                print(f"  Restart [{name}]: score={sc_r:.10f}")
                break
        # Quick de-peak on the restart result
        if time.monotonic() < deadline - 20:
            a_dp_r, sc_dp_r = _de_peak_search(best_seq, deadline - 20, max_evals=5000)
            try_update(a_dp_r, f"restart_{name}_dp")
    print(f"  Phase 7 done: {time.monotonic()-t_p:.0f}s, best={best_score_val:.10f}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n=== Phase 8: Final polish ===")
    t_p = time.monotonic()
    n = len(best_seq)
    z = np.log(np.maximum(best_seq, 1e-15))
    z = z - z.mean()
    c_b = _full_convolve(best_seq, best_seq)
    mb = float(c_b.max())

    for beta in [mb * 2e-7, mb * 8e-8, mb * 3e-8, mb * 1e-8, mb * 4e-9, mb * 1.5e-9]:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - 5 - time.monotonic()
        if remaining < 5:
            break
        z, niter = _lbfgs_lsexp(z, beta, n, deadline - 5,
                                max_iter=min(5000, max(100, int(remaining / 0.1))))
        a_final = _softmax(z)
        sc_final = _score(a_final)
        if sc_final < best_score_val - 1e-14:
            best_score_val = sc_final
            best_seq = a_final.copy()
            print(f"  Ultra beta={beta:.2e}: score={sc_final:.10f} iters={niter}")

    # One more de-peak pass
    if time.monotonic() < deadline - 3:
        a_dp_f, sc_dp_f = _de_peak_search(best_seq, deadline - 3, max_evals=10000)
        try_update(a_dp_f, "final_depeak")

    print(f"  Phase 8 done: {time.monotonic()-t_p:.0f}s")

    # === Final ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.0f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
