def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n):

    s = e.sum()

        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        # Reduce a[i] where it contributes most to the peaks
        # Contribution of a[i] to b[k] is a[i]*a[k-i]

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Adaptive beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)
    
    # More stages for finer control
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
                                   0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        # Estimate time per iteration
        est_time = 0.005 * max(1, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 10))))
        
        def fun_local(z_local):
            return _lse_obj(z_local, beta, n)
        
        def jac_local(z_local):
            return _lse_grad(z_local, beta, n)
        
        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = 0.21557895 + 0.41663158*np.cos(2*np.pi*t) + 0.277263158*np.cos(4*np.pi*t) + 0.083578947*np.cos(6*np.pi*t) + 0.006947368*np.cos(8*np.pi*t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:

    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
