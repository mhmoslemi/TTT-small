"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
parameterized initialization, tight active-set L-BFGS with Newton refinement,
multiplicative peak updates, multi-length exploration, and aggressive polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def make_profile(n, kind="uniform", param=0.0):

        return np.ones(n) / n
    elif kind == "linear_taper":
        # a_i = 1 + param * (2*i/n - 1), normalized
        t = np.linspace(0, 1, n)

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential":

        t = np.linspace(0, 1, n)
        a = np.exp(param * t)
        return a / a.sum()
    elif kind == "power_law":
        # a_i = (i+1)^(-param), normalized
        i = np.arange(1, n + 1)
        a = i ** (-param)
        return a / a.sum()
    elif kind == "quadratic_taper":

    a = _softmax(z)
    c = _full_convolve(a, a)

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _get_gradient_exact(a, c, n, active_set):
    """Compute gradient of max over active set."""
    # Gradient of c_k w.r.t. a_i is 2*a_{k-i}
    # For active set with uniform weights:

    # This is 2 * (w * reverse(a))[n-1 + i] / |A| where w is uniform on A
    w = np.zeros(len(c))
    w[active_set] = 1.0 / len(active_set)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # The gradient of the objective 2n*max(c)/sum(a)^2 w.r.t. a is:
    # 2n * (grad_max - 2*max(c)/sum(a) * 1) / sum(a)
    # For the log-sum-exp version: 2n * (h - W) (since softmax normalizes)
    return 2.0 * n * (h - W)

    current_a = a_start.copy()
    current_sc = _score(current_a)
    z0 = np.log(np.maximum(current_a, 1e-15))
    z0 = z0 - z0.mean()

    for i, beta in enumerate(betas):

        def fun(z):

        # Take top 5% or those within 0.1% of max, whichever is larger
        n_top = max(5, int(0.05 * len(c)))
        threshold_rank = sorted_c[min(n_top - 1, len(c) - 1)]
        tol_abs = 0.001 * c_max
        threshold = max(threshold_rank, c_max - tol_abs)
        active = c >= threshold

        # Compute gradient
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Multiplicative update in log-space
        log_a = np.log(np.maximum(a, 1e-15))
        update = eta * (h - W)
        log_a = log_a - update
        log_a = np.clip(log_a, -50, 50)
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.03, 0.3)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-10:
                break
            if stall > 200:
                break

    return best_a, best_sc


def _newton_refinement(a_start, deadline, max_iters=500):
    """
    Newton-like method using approximate Hessian from the log-sum-exp.
    Uses the fact that at small beta, the Hessian is dominated by the active set.
    """
    n = len(a_start)
    z = np.log(np.maximum(a_start, 1e-15))
    z = z - z.mean()
    best_a = a_start.copy()
    best_sc = _score(a_start)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Use a very small beta for near-exact max
        beta = c_max * 1e-6
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s_e = e.sum()
        if not np.isfinite(s_e):
            break
        w = e / s_e

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if np.linalg.norm(grad) < 1e-15:
            break

        # Simple steepest descent with backtracking
        step = 0.1
        for _ in range(20):
            z_new = z - step * grad
            z_new = np.clip(z_new, -100, 100)
            z_new = z_new - z_new.mean()
            a_new = _softmax(z_new)
            sc_new = _score(a_new)
            if sc_new < _score(a) - 1e-16:
                break
            step *= 0.5

        z_new = np.clip(z - step * grad, -100, 100)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            z = z_new
        else:
            break

    return best_a, best_sc


def _coordinate_descent_aggressive(a_init, deadline, max_rounds=50000):
    """Aggressive coordinate descent with pair updates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick random pair
        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        # Try several fractions
        for frac in np.linspace(0.1, 0.9, 9):
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if rnd % 10000 == 0:
            print(f"    CD round {rnd}: best={best_sc:.10f}")

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_restart(a, rng, mode="gauss", scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "smooth":
        # Add smooth perturbation using random frequencies
        t = np.arange(n)
        for _ in range(7):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.01 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "antisym":
        # Add anti-symmetric perturbation
        delta = scale * rng.standard_normal(n)
        delta = delta - delta[::-1]  # make anti-symmetric
        z = z + delta
    z = z - z.mean()
    return _softmax(z)
