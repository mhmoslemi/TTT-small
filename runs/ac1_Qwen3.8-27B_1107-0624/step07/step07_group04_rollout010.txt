        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _grad_smooth_max(a, beta, n):
    """Gradient of smooth max: beta * logsumexp(conv/beta - max)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return _hankel_matvec(a, w, n)

def _hankel_hessian_solve(W, n, rhs, max_iter=300, tol=1e-12):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        if rs_new < 1e-30:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _fista_smooth(a_init, beta, deadline, max_iter=5000):
    """FISTA (accelerated proximal gradient) on smooth max objective."""
    n = len(a_init)
    # Work in simplex: minimize 2*n*smooth_max(conv(a,a)) s.t. sum(a)=1, a>=0
    # prox onto simplex {sum=1, a>=0}
    z = np.log(np.maximum(a_init, 1e-30))
    y = a_init.copy()
    theta = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Adaptive step size
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    # Lipschitz constant estimate: 4*n*max|H_k|*max(a) ~ 4*n*c_max (rough)
    L = 4.0 * n * (2.0 * c_max + 0.01)
    lr = 1.0 / max(L, 1e-6)
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        grad = _hankel_matvec(y, _active_weights(_full_convolve(y, y), n), n)
        # Actually, for smooth version:
        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())
        shifted = (c_y - c_max_y) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / s
        grad = 4.0 * n * _hankel_matvec(y, w, n) / 2.0
        # Wait, let me recompute. The smooth objective is:
        # f(a) = 2*n*(c_max + beta*log(sum exp((c_k - c_max)/beta)))
        # df/da = 2*n * sum_k w_k * 2*H_k*a / ... 
        # Actually df/da_i = 2*n * 2 * sum_k w_k * a[k-i] (for valid indices)
        # = 4*n * (H_w * a)_i
        # where H_w is the Hankel matrix with row k = w_k * H_k
        
        # Hmm, let me just use the gradient I computed above
        # grad = 4*n * H_w * a (this is the gradient of 2*n*smooth_max)
        
        # Wait, I need to be more careful. Let me define:
        # phi(a) = 2*n * (c_max(a) + beta * log(sum_k exp((c_k(a) - c_max(a))/beta)))
        # d phi / d a_i = 2*n * (1/beta) * sum_k exp((c_k - c_max)/beta) * d c_k / d a_i
        # = 2*n * sum_k w_k * d c_k / d a_i (where w_k = exp((c_k-c_max)/beta) / sum exp(...))
        # d c_k / d a_i = 2 * a[k-i] (for valid k-i)
        # So d phi / d a_i = 4*n * sum_k w_k * a[k-i] = 4*n * (H_w * a)_i
        
        # So grad = 4*n * H_w * a, where H_w * a = _hankel_matvec(a, w, n) / 2
        # Wait, _hankel_matvec computes 2*sum_k W[k]*a[k-i], so:
        # H_w * a (in my convention) = sum_k W[k] * 2*a[k-i]
        # So grad = 2*n * sum_k w_k * 2*a[k-i] = 2*n * _hankel_matvec(a, w, n)
        
        # Let me just recompute cleanly:
        grad = 2.0 * n * _hankel_matvec(y, w, n)
        
        # FISTA step
        x_new = y - lr * grad
        # Project onto simplex {a: sum(a)=1, a>=0}
        x_new = _proj_simplex(x_new)
        
        # Update y for next iteration
        theta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * theta * theta))
        y = x_new + ((theta - 1.0) / theta_new) * (x_new - y)
        theta = theta_new
        
        # Backtracking: check if lr is too large
        if i % 10 == 0:
            sc = _score(x_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = x_new.copy()
        
        # Adaptive lr: reduce if not decreasing
        if i > 100 and i % 50 == 0:
            c_check = _full_convolve(x_new, x_new)
            c_max_check = float(c_check.max())
            if c_max_check > c_max_y * 1.01:
                lr *= 0.5
            elif c_max_check < c_max_y * 0.99:
                lr *= 1.1
    
    sc_final = _score(best_a)
    return best_a, sc_final

def _proj_simplex(v):
    """Project v onto {a: sum(a)=1, a>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _active_weights(c, n, tol_frac=0.001):
    """Get weights for active set (conv values near max)."""
    c_max = float(c.max())
    if c_max < 1e-15:
        w = np.zeros_like(c)
        w[0] = 1.0
        return w
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w

def _newton_active_polish_v2(a_init, deadline, max_rounds=100):
    """Improved CG-Newton active set polish with exact active set tracking."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    # Start with tight active set, expand if needed
    tol_fracs = [0.0001, 0.0002, 0.0005, 0.001, 0.002, 0.005]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Try multiple active set sizes
        best_step_sc = c_max
        best_new_a = None
        
        for tf in tol_fracs:
            tol = tf * c_max
            active = c >= (c_max - tol)
            n_active = int(active.sum())
            if n_active == 0:
                active[int(np.argmax(c))] = True
                n_active = 1
            
            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_active
            
            # KKT: minimize sum_k W[k] * c_k s.t. sum(a)=1, a>=0
            # grad = 2*n*H_W*a, KKT: H_W*a = mu (on support), H_W*a <= mu (off support)
            # Newton step: solve H_W * dx = -(H_W*a - mu) + lam*1
            # where mu is chosen so that sum(dx) = 0
            
            g = _hankel_matvec(a, W, n)  # This is 2*H_W*a
            g_t = g - np.mean(g)  # Centered gradient
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-14:
                continue
            
            # CG solve for Newton direction
            Hg = _hankel_hessian_solve(W, n, g, max_iter=200)
            H1 = _hankel_hessian_solve(W, n, np.ones(n), max_iter=200)
            
            # Newton step with sum=0 constraint
            a1 = float(np.ones(n) @ Hg)
            a2 = float(np.ones(n) @ H1)
            if abs(a2) < 1e-20:
                dx = -g_t / (gn + 1e-30)
            else:
                lam = a1 / a2
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
            
            # Damped step
            alpha = 1.0
            for ls in range(40):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                sc_try = _score(a_try)
                if c_max_try < c_max - 1e-10:
                    if c_max_try < best_step_sc:
                        best_step_sc = c_max_try
                        best_new_a = a_try
                    break
                alpha *= 0.5
                if alpha < 1e-12:
                    break
        
        if best_new_a is None:
            # Gradient fallback
            c = _full_convolve(a, a)
            w = _active_weights(c, n, 0.001)
            grad = 2.0 * n * _hankel_matvec(a, w, n)
            alpha = 0.01
            for ls in range(30):
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                if float(_full_convolve(a_try, a_try).max()) < c_max:
                    best_new_a = a_try
                    break
                alpha *= 0.5
        
        if best_new_a is None:
            # No improvement, try with different active set
            if rnd > 20:
                break
            continue
        
        a = best_new_a
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def _run_lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)
    
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
              0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        beta = base_beta * ratio
        
        def obj_fun(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s))
        
        def grad_fun(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = float(c_local.max())
            shifted = (c_local - c_max_local) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e / s
            # d/dz of softmax-based objective
            # f(a) = 2*n*smooth_max
            # df/da = 2*n * H_w * a (in my notation)
            # da/dz = a * (1 - a) (for softmax parameterization, this is the Jacobian)
            # Actually for softmax: a_i = exp(z_i)/sum(exp(z)), so da_i/dz_j = a_i*(delta_ij - a_j)
            # grad_z_i = sum_j (df/da_j) * (da_j/dz_i) = (df/da_i) * a_i * (1 - a_i) + a_i * sum_j (df/da_j) * (-a_j)
            # = a_i * [(df/da_i) * (1 - a_i) - a_i * sum_j (df/da_j) * a_j / a_i]
            # Hmm this is getting complicated. Let me use a simpler approach.
            # For the log-softmax parameterization, we can use:
            # grad_z = a * (grad_a - sum(a * grad_a))
            grad_a = 2.0 * n * _hankel_matvec(a_local, w, n)
            grad_z = a_local * (grad_a - float(np.dot(a_local, grad_a)))
            return grad_z
        
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(20, int(remaining / (est_iter_time * 20))))
        
        try:
            result = minimize(
                obj_fun, z,
                jac=grad_fun,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered_sharp":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "triple":
        t = np.linspace(0, 1, n)
        a = (np.exp(-12.0 * (t - 0.2) ** 2) + np.exp(-12.0 * (t - 0.5) ** 2) +
             np.exp(-12.0 * (t - 0.8) ** 2) + 0.01)
        return a / a.sum()
    elif kind == "flat_top":
        # Flat top window (like in FFT)
        t = np.linspace(0, 1, n)
        a = 0.21557895 + 0.41663158 * np.cos(2 * np.pi * t) + \
            0.277263158 * np.cos(4 * np.pi * t) + \
            0.083578947 * np.cos(6 * np.pi * t) + \
            0.006947368 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "kaiser":
        # Kaiser-Bessel window
        t = np.linspace(-1, 1, n)
        from scipy.special import i0
        alpha = 5.0
        a = i0(alpha * np.sqrt(np.maximum(1 - t**2, 0))) / i0(alpha)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _shrinkage_reopt(a, deadline, min_time_remaining=60.0):

    n = len(a)

    fractions = [0.01, 0.02, 0.05, 0.08, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:

        n_zero = max(1, int(n * frac))
        a_sorted_idx = np.argsort(best_a)
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:

            best_a = a_try.copy()

            a_np, sc_np = _newton_active_polish_v2(best_a, min(deadline - 30, time.monotonic() + 60), max_rounds=50)
