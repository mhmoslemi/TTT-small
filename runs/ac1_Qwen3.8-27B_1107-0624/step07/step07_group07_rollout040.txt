"""
Step sequence optimizer using POCS (Projection Onto Convex Sets) with 
spectral peak flattening, followed by L-BFGS fine-tuning and peak shaving.
Novel approach: Uses alternating projections between the non-negativity/simplex 
constraint and the spectral peak constraint to find well-distributed 
autocorrelation profiles.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _pocs_spectral_flatten(a_init, deadline, max_outer=50, max_inner=50):
    """
    POCS: Alternate between projecting onto {a>=0, sum=1} and {max(conv(a,a)) <= t}.
    The spectral projection reduces peaks by adjusting the power spectrum.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Start with a conservative t and decrease
    c0 = _full_convolve(a, a)
    t = float(c0.max()) * 0.95
    t_min = float(c0.max()) * 0.7  # We aim to reduce by at least 30%
    
    best_a = a.copy()
    best_sc = _score(a)
    
    outer_iter = 0
    while outer_iter < max_outer and time.monotonic() < deadline:
        # Inner iterations: POCS
        for inner in range(max_inner):
            if time.monotonic() > deadline:
                break
                
            # Project onto spectral constraint: max(conv(a,a)) <= t
            A = np.fft.rfft(a, N)
            P = A.real**2 + A.imag**2  # |A|^2
            
            # Compute current conv
            c = np.fft.irfft(A * np.conj(A), N)[:2*n-1]
            c_max = c.max()
            
            if c_max <= t * (1 + 1e-10):
                break
            
            # Reduce power spectrum to bring peak down
            # Frequency-selective reduction: reduce bins that constructively 
            # contribute to the peak lag
            k_star = int(np.argmax(c))
            # Phase of contribution at each frequency bin
            # For real sequences, the contribution of bin m to lag k is
            # proportional to cos(2*pi*m*k/N) (for the positive frequency bins)
            # But we need to be more careful with the full linear conv indexing
            
            # Simple approach: uniform scaling + gradient step
            excess = c_max - t
            scale = t / c_max
            P = P * scale
            
            # Additional gradient step on top
            alpha = 0.05
            freqs = np.fft.rfftfreq(N)
            # Contribution to peak: correlate P with the "peak pattern"
            # The peak pattern in frequency domain for lag k_star is cos(2*pi*freqs*k_star)
            peak_pattern = np.cos(2 * np.pi * freqs * k_star)
            # Reduce P where peak_pattern is positive (constructive)
            reduction = alpha * np.maximum(peak_pattern, 0) * excess / t
            P = np.maximum(P - reduction * P, P * 0.01)  # Don't reduce below 1%
            
            # Reconstruct
            mag = np.sqrt(P)
            phase = np.angle(A)
            A_new = mag * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, N)[:n]
            
            # Project onto {a >= 0, sum = 1}
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s_new
            
            a = a_new
        
        # Check improvement
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Decrease t
        c_cur = _full_convolve(a, a)
        t = max(t_min, t * 0.97)
        
        outer_iter += 1
        if outer_iter % 5 == 0:
            print(f"  POCS iter {outer_iter}: t={t:.6e}, score={best_sc:.10f}")
    
    return best_a, best_sc

def _lbfgs_logsumexp(a_init, beta, n, max_iter=5000):
    """L-BFGS with log-sum-exp smoothing of max(conv)"""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init

def _lbfgs_continue(a_start, deadline, min_time=100.0):
    """L-BFGS with decreasing beta schedule"""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 
                              0.002, 0.001, 0.0005, 0.0002, 
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break
        
        max_iter = min(5000, max(200, int(remaining / 0.005)))
        a_new = _lbfgs_logsumexp(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            print(f"  L-BFGS beta={beta:.3e}: score={sc:.10f}")
    
    return current_a

def _peak_shaving_direct(a_init, deadline, max_iters=20000):
    """Direct peak shaving: identify peaks, redistribute mass"""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Find active peaks (top few)
        sorted_idx = np.argsort(c)[::-1]
        top_k = min(10, len(c))
        active = np.zeros_like(c)
        for k in range(top_k):
            if c[sorted_idx[k]] >= c_max * 0.95:
                active[sorted_idx[k]] = 1.0
        if active.sum() == 0:
            active[int(np.argmax(c))] = 1.0
        active = active / active.sum()
        
        # Gradient of weighted conv w.r.t. a
        a_rev = a[::-1]
        wc = _full_convolve(active, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(active, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()  # Center to maintain sum
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        g = g / g_norm
        
        # Line search
        best_eta_try = eta
        best_a_try = a
        best_sc_try = best_sc
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_eta_try = step
        
        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.05, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.85
            if eta < 1e-9 or stall > 200:
                break
    
    return best_a, best_sc

def _coordinate_descent_block(a_init, deadline, block_size=4):
    """Block coordinate descent: optimize blocks of consecutive elements"""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = n // block_size
    
    for rnd in range(200000):
        if time.monotonic() > deadline:
            break
        
        # Pick a random block
        bidx = (rnd * 7919) % n_blocks
        start = bidx * block_size
        end = min(start + block_size, n)
        block_len = end - start
        s_block = a[start:end].sum()
        
        if s_block < 1e-15:
            continue
        
        # Try different distributions within the block
        # Keep total sum of block fixed
        best_frac = None
        best_sc_block = best_sc
        
        for pattern in range(3):
            if pattern == 0:  # Uniform
                new_block = np.full(block_len, s_block / block_len)
            elif pattern == 1:  # Linear ramp up
                t = np.linspace(0.1, 0.9, block_len)
                new_block = s_block * t / t.sum()
            elif pattern == 2:  # Linear ramp down
                t = np.linspace(0.9, 0.1, block_len)
                new_block = s_block * t / t.sum()
            elif pattern == 3:  # Bump
                t = np.sin(np.pi * (np.arange(block_len) + 0.5) / block_len)
                new_block = s_block * t / t.sum()
            
            a_try = a.copy()
            a_try[start:end] = new_block
            sc_try = _score(a_try)
            
            if sc_try < best_sc_block:
                best_sc_block = sc_try
                best_frac = new_block.copy()
        
        if best_frac is not None:
            best_sc = best_sc_block
            best_a = a.copy()
            a[start:end] = best_frac
    
    return best_a, best_sc

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z, T=1.0):
    z = z / T
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _power_sum_obj(z, p, n):
    """Objective using Lp-norm approximation to max of conv."""
    a = _softmax(z, 1.0)
    c = _full_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)
    ps = np.sum(c_clipped ** p) ** (1.0 / p)
    return 2.0 * n * ps


def _power_sum_grad(z, p, n):
    """Gradient of power-sum objective w.r.t. z."""
    a = _softmax(z, 1.0)
    c = _full_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)
    ps = np.sum(c_clipped ** p) ** (1.0 / p)
    # d(C_p)/dc_k = c_k^(p-1) / C_p^(p-1)
    weights = c_clipped ** (p - 1) / (ps ** (p - 1))
    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(weights, c))
    # Gradient w.r.t. a: 2*n*(h - W) * something
    # Then chain rule through softmax
    grad_a = 2.0 * n * (h - W)
    # d(a)/d(z) = a * (I - a^T) for softmax
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z


def _logsumexp_obj(z, beta, n):

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = c.max()

def _block_coordinate_descent(a_init, deadline, block_sizes=(2, 4, 8)):
    """

        start = bidx * bs

        s_block = a[start:end].sum()
        if s_block < 1e-15:
            continue
        
        blen = end - start

        # Try parametric redistributions

                t = np.cos(np.pi * np.arange(blen) / blen) ** 2 + 0.1
            else:
                t = np.exp(-((np.arange(blen) - blen / 2) ** 2) / (blen / 4))
            
            t = np.maximum(t, 0.01)
            new_block = s_block * t / t.sum()
            
            a_try = a.copy()
            a_try[start:end] = new_block
            sc_try = _score(a_try)

def _make_profile(n, kind="uniform"):
    """Generate various initial profiles."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        sigma = 0.25
        a = np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2))
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        a = np.hamming(n)
        return a / a.sum()
    elif kind == "hann":
        a = np.hanning(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        a = np.blackman(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        width = 0.4
        a = np.ones(n) * 0.1
        mask = (t > 0.5 - width / 2) & (t < 0.5 + width / 2)
        a[mask] = 1.0
        # Smooth edges
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, rng, scale=0.02):

    z = np.log(np.maximum(a, 1e-15))
    z = z + scale * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z, 1.0)

    print(f"\n=== Phase 8: Multi-restart ===")
    t_phase = time.monotonic()
    
    restart_configs = [
        (0.01, "r1_s001"),
        (0.03, "r2_s003"),
        (0.05, "r3_s005"),
        (0.08, "r4_s008"),
        (0.15, "r5_s015"),
    ]
    
    for scale, name in restart_configs:
        if time.monotonic() > deadline - 40:
            break
        a_pert = _perturb(best_seq, rng, scale)
        sc_pert = _score(a_pert)
        
        # Quick L-BFGS
        remaining = deadline - 40 - time.monotonic()
        if remaining > 30:
            a_quick = _logsumexp_schedule(a_pert, deadline - 35, min_time=20.0)
            try_update(a_quick, f"restart_{name}")
        
        # Quick peak shave
        if time.monotonic() < deadline - 30:
            a_qs, _ = _multi_peak_shaving(best_seq, deadline - 28, max_iters=5000)
            try_update(a_qs, f"restart_{name}_ps")

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n=== Phase 9: Final ultra-fine polish ===")
    t_phase = time.monotonic()
    
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Ultra-fine L-BFGS
        n = len(best_seq)
        z_final = np.log(np.maximum(best_seq, 1e-15))
        z_final = z_final - z_final.mean()
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        
        for beta_frac in [5e-7, 2e-7, 1e-7]:
            if time.monotonic() > deadline - 10:
                break
            beta = mb * beta_frac
            remaining = deadline - 10 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(3000, max(100, int(remaining / 0.002)))
            
            def fun_final(z):
                return _logsumexp_obj(z, beta, n)
            
            def jac_final(z):
                return _logsumexp_grad(z, beta, n)
            
            z_new = _lbfgs_run(z_final, fun_final, jac_final, max_iter)
            a_final = _softmax(z_new, 1.0)
            try_update(a_final, f"ultra_beta_{beta_frac}")
            z_final = z_new
        
        # Final peak shave
        if time.monotonic() < deadline - 8:
            a_fs, _ = _multi_peak_shaving(best_seq, deadline - 5, max_iters=10000)
            try_update(a_fs, "final_shave")

    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "raised_cosine")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
