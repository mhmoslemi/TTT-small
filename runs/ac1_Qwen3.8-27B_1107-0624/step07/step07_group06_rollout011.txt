Combines aggressive smooth-max L-BFGS, Newton active-set refinement,
spectral equalization, and diverse restarts to beat 1.5057.

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    """FFT-based full linear convolution."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_max_obj(z, beta, n):
    """Smooth max of convolution via log-sum-exp."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smooth_max_grad(z, beta, n):
    """Gradient of smooth max objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    h = wc[n-1:2*n-1]

def _hessian_matvec(v, active, n):

    result = r[n-1:2*n-1]

def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    """Conjugate gradient solver."""
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):

        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.01
    patience = 0
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Weight the top 1% of peaks more heavily
        threshold = np.percentile(c, 99.0)
        w = np.zeros_like(c)
        w[c >= threshold] = 1.0
        w = w / w.sum()
        
        # Gradient of weighted sum of peaks
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 2.0 * a * (h - W)
        
        # Step
        step = lr / (1.0 + 0.005 * i)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.02, 0.05)
        else:
            patience += 1
            lr *= 0.95
            if patience > 50:
                a = best_a.copy()
                patience = 0
                lr = 0.01
    
    return best_a, best_sc

def _spectral_equalize(a_init, deadline, n_iters=50):
    """Iterative spectral equalization (Gerchberg-Saxton style)."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        # FFT
        A = np.fft.rfft(a, N)
        
        # Target: flatten |A^2| in time domain means making A more uniform
        # We want the inverse of A^2 (circular) to be flat
        # equivalently, we want |A| to be more uniform
        mag = np.abs(A)
        
        # Geometric mean as target
        log_mag = np.log(mag + 1e-30)
        target_log = 0.3 * np.mean(log_mag) + 0.7 * log_mag  # partial flattening
        
        # Reconstruct
        A_new = np.exp(target_log) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * t
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    """Log-normal perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
