            # Try decreasing LM parameters (more Newton-like)
            for lam in [2.0 * h_norm_est, 1.0 * h_norm_est, 0.5 * h_norm_est,
                        0.2 * h_norm_est, 0.1 * h_norm_est, 0.05 * h_norm_est]:
                # Solve (2H + lam*I) x1 = g  (for Newton direction)
                x1 = _cg_solve_reg(W, n, g, lam, max_iter=200)
                # Solve (2H + lam*I) x2 = 1  (for constraint)
                x2 = _cg_solve_reg(W, n, np.ones(n), lam, max_iter=200)

                x1_dot_1 = float(np.sum(x1))
                x2_dot_1 = float(np.sum(x2))

                if abs(x2_dot_1) < 1e-20:
                    dx = -g_t / (gn + 1e-30)
                else:
                    dx = -x1 + (x1_dot_1 / x2_dot_1) * x2
                    dx = dx - np.mean(dx)

                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    continue

                # Backtracking line search - accept any improvement
                alpha = 1.0
                step_found = False
                for ls in range(50):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0.0)
                    s_try = float(a_try.sum())
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    c_max_try = float(c_try.max())
                    if c_max_try < c_max - 1e-15:
                        a = a_try
                        step_found = True
                        break
                    alpha *= 0.5
                    if alpha < 1e-14:
                        break

                if step_found:
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Fallback: steepest descent with backtracking
            W = np.zeros(2 * n - 1)
            c2 = _full_convolve(a, a)
            c2_max = float(c2.max())
            active = c2 >= (c2_max - 1e-2 * c2_max)
            if active.sum() == 0:
                active[int(np.argmax(c2))] = True
            W[active] = 1.0 / active.sum()
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn > 1e-14:
                alpha = 1.0 / (gn + 1e-30)
                for ls in range(30):
                    a_try = a - alpha * g_t
                    a_try = np.maximum(a_try, 0.0)
                    s_try = float(a_try.sum())
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max:
                        a = a_try
                        improved = True
                        break
                    alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient method with adaptive active set and backtracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient of weighted sum
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0005 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 10 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.01
        z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # Aggressive schedule
    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0.0)
            s = float(a_new.sum())
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.25) ** 2) + np.exp(-6.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "triple":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.2) ** 2) + np.exp(-8.0 * (t - 0.5) ** 2) + \
              np.exp(-8.0 * (t - 0.8) ** 2) + 0.02
        return a / a.sum()
    elif kind == "alternating":
        a = np.ones(n)
        a[1::2] = 0.4
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _support_reduction(a, deadline, min_time_remaining=60.0):
    """Try zeroing out small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for frac in [0.05, 0.10, 0.15, 0.20, 0.30, 0.40]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_sorted_idx = np.argsort(best_a)
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = float(a_try.sum())
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_a = a_try.copy()
            best_sc = sc_try
            # Re-optimize from improved point
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_a = a_reopt
                best_sc = sc_reopt
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: L-BFGS + LM-Newton polish + subgradient +
    support reduction + multi-n + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS from parent
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Newton polish (primary improvement)
    print(f"\n[Phase 2] Newton polish 1 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _newton_polish(best_seq, min(deadline - 300, time.monotonic() + 300), max_rounds=150)
        update_best(a2, sc2, "newton_1")

    # Phase 3: Subgradient polish
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a3, sc3 = _subgrad_polish(best_seq, min(deadline - 250, time.monotonic() + 80))
        update_best(a3, sc3, "subgrad_1")

    # Phase 4: Newton polish after subgradient
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _newton_polish(best_seq, min(deadline - 200, time.monotonic() + 150), max_rounds=100)
        update_best(a4, sc4, "newton_2")

    # Phase 5: Support reduction
    print(f"\n[Phase 5] Support reduction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a5, sc5 = _support_reduction(best_seq, deadline, min_time_remaining=100.0)
        update_best(a5, sc5, "support")
        # Newton polish after support reduction
        if time.monotonic() < deadline - 120:
            a5b, sc5b = _newton_polish(a5, min(deadline - 80, time.monotonic() + 80), max_rounds=80)
            update_best(a5b, sc5b, "support_newton")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 4096, 2048]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        if n_try > 16384:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 50:
            a_np, sc_np = _newton_polish(a_opt, min(deadline - 30, time.monotonic() + 50), max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            (0.02, "tapered"),
            (0.08, "bimodal"),
            (0.15, "triple"),
            (0.05, "alternating"),
        ]
        for i, (sigma, profile) in enumerate(restart_configs):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, sigma, rng)
            else:
                p_start = _make_profile(len(best_seq), profile)
                p_start = _perturb(p_start, 0.1, rng)

            if time.monotonic() > deadline - 15:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")
            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _newton_polish(a_r, min(deadline - 5, time.monotonic() + 25), max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 1, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 5e-7
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                 'maxcor': 60}
                    )
                    a_flb = _softmax(result.x)
                    a_flb = np.maximum(a_flb, 0.0)
                    s_flb = float(a_flb.sum())
                    if s_flb > 1e-15:
                        a_flb = a_flb / s_flb
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
