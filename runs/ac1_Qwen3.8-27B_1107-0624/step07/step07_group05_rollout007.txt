"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_active(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)

        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _pressure_equalize(a, deadline, n_iters=300, gamma=0.4, tol_frac=0.01):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 5

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:

        a = a / s

        if it % 10 == 0:

                best_sc = sc
                best_a = a.copy()

def _pressure_anneal(a, deadline, n_total=500):

    gammas = [0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08]

        if c_max < 1e-30:
            break

        top_idx = np.argsort(c)[::-1][:n_peaks]
        reduced = False

        for k in top_idx:
            contribs = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribs[i] += a[i] * a[j]
                    contribs[j] += a[i] * a[j]

            total = contribs.sum()
            if total < 1e-30:
                continue

            top_c = np.argsort(contribs)[::-1][:max(3, n // 100)]
            shave = reduce_frac / len(top_c)
            for i in top_c:
                old = a[i]
                a[i] = old * (1.0 - shave)
                if a[i] < 0:
                    a[i] = old

            s = a.sum()
            if s > 1e-15:
                a = a / s
            reduced = True

        if reduced:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=10.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                      0.007, 0.003, 0.001, 0.0005, 0.0002,
                                      0.0001, 0.00005, 0.00002, 0.00001,
                                      0.000005, 0.000002, 0.000001, 0.0000005])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                      0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        if aggressive and i >= len(betas) - 5:
            max_iter = min(20000, max(200, int(remaining / (est_time * 6))))
        else:
            max_iter = min(10000, max(100, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _grad_active(a, c, n, tol_frac=0.003)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_window":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "trapezoid":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        frac = 0.2
        a[:int(frac * n)] = np.linspace(0.3, 1.0, int(frac * n))
        a[-int(frac * n):] = np.linspace(1.0, 0.3, int(frac * n))
        return a / a.sum()
    elif kind == "gaussian_broad":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.conj(np.exp(1j * phase))
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _interpolate_sequence(a_old, n_new):
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = _make_profile(n_new, "uniform")
    return a_new


def _coordinate_polish(a, deadline, n_rounds=80):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]
        perm = rng.permutation(len(top_idx))
        top_idx = top_idx[perm[:max(5, len(top_idx) // 4)]]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            c_old = _full_convolve(a, a).max()

            for factor in [0.4, 0.6, 0.8, 1.2]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    break
                a_norm = a / s
                c_try = _full_convolve(a_norm, a_norm).max()
                if c_try < c_old / s - 1e-15:
                    improved = True
                    a = a_norm
                    break
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _multi_stage(a_init, deadline, min_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 60, aggressive=False)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    if time.monotonic() < deadline - min_remaining - 20:
        a2, sc2 = _pressure_anneal(current_a, min(deadline - min_remaining - 10,
                                                   time.monotonic() + 80))
        sc2_full = _score(a2)
        if sc2_full < current_sc:
            current_a = a2
            current_sc = sc2_full

    if time.monotonic() < deadline - min_remaining - 20:
        a3 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 20, aggressive=True)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    if time.monotonic() < deadline - min_remaining:
        a4, sc4 = _subgrad_polish(current_a, min(deadline - min_remaining,
                                                  time.monotonic() + 50),
                                   n_max_steps=2000, lr_init=0.015)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        _update(parent)
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # Phase 1: Quick optimization of parent
    print(f"\n[Phase 1] Quick optimization of parent (t={time.monotonic()-t_start:.1f}s)")
    a1, sc1 = _multi_stage(parent, deadline, min_remaining=900)
    _update(a1, sc1)

    # Phase 2: Aggressive optimization
    print(f"\n[Phase 2] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        a2 = _run_lbfgs(best_seq, deadline, min_time_remaining=500, aggressive=True)
        sc2 = _score(a2)
        _update(a2, sc2)

    # Phase 3: Peak shaving
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a3, sc3_norm = _peak_shave(best_seq, min(deadline - 400, time.monotonic() + 60),
                                    n_rounds=40, n_peaks=3, reduce_frac=0.2)
        sc3 = _score(a3)
        _update(a3, sc3)
        if sc3 < best_score + 0.01:
            a3b = _run_lbfgs(a3, deadline, min_time_remaining=400, aggressive=False)
            _update(a3b, _score(a3b))

    # Phase 4: Pressure equalization (extended)
    print(f"\n[Phase 4] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a4, sc4_norm = _pressure_anneal(best_seq, min(deadline - 300, time.monotonic() + 120))
        sc4 = _score(a4)
        _update(a4, sc4)
        if sc4 < best_score:
            a4b = _run_lbfgs(a4, deadline, min_time_remaining=300, aggressive=True)
            _update(a4b, _score(a4b))

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 9216, 10240, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(key=lambda x: -abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - 200 - time.monotonic()
        if remaining < 60:
            break

        a_init = _interpolate_sequence(parent, n_try)
        min_rem = max(30, remaining * 0.3)
        a_opt, sc_opt = _multi_stage(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        restart_kinds = ["tapered", "cosine_window", "trapezoid", "gaussian_broad"]

        for i, kind in enumerate(restart_kinds):
            if time.monotonic() > deadline - 50:
                break
            a_prof = _make_profile(n_best, kind)
            a_blend = 0.6 * best_seq + 0.4 * a_prof
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=6, amplitude=0.04)
            a_r, sc_r = _multi_stage(a_pert, deadline, min_remaining=50)
            _update(a_r, sc_r)

    # Phase 7: Coordinate descent polish
    print(f"\n[Phase 7] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cd, sc_cd_norm = _coordinate_polish(best_seq, min(deadline - 15, time.monotonic() + 90),
                                              n_rounds=60)
        sc_cd = _score(a_cd)
        _update(a_cd, sc_cd)

    # Phase 8: Final aggressive L-BFGS
    print(f"\n[Phase 8] Final L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = _run_lbfgs(best_seq, deadline - 5, min_time_remaining=5, aggressive=True)
        _update(a_fin, _score(a_fin))

    # Phase 9: Subgradient final polish
    print(f"\n[Phase 9] Final subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.008)
        _update(a_sg, sc_sg)

    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
