- Reducing a_i where a_{i+k*} is large (and vice versa).

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """Compute value and gradient of smoothed max objective.
    f(z) = 2n * (c_max + beta * log(sum_k exp((c_k - c_max)/beta)))
    grad_z = 4n * a * (h - W)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    if s < 1e-300 or not np.isfinite(s):

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)

    return best_a


def _active_set_polish(a_init, deadline, K=3, max_iter=10000, verbose=True, t0=0.0):
    """Projected gradient on the sum of top-K lag values (active set method)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.003
    stuck = 0

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        topK_idx = np.argsort(c)[::-1][:K]

        # Gradient of (1/K)*sum_{k in topK} b[k] w.r.t. a
        # = (2/K) * sum_{k in topK} a[k-j] for each j
        # Computed via FFT: h = convolve(a_rev, w)[n-1:2n-1]
        w = np.zeros(2 * n - 1)
        w[topK_idx] = 1.0 / K
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Project to tangent space (sum=0)
        grad = grad - np.mean(grad)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            stuck = 0
            lr = min(lr * 1.005, 0.015)
            if verbose and i % 2000 == 0:
                print(f"    [polish] it={i} K={K} score={best_sc:.10f} "
                      f"lr={lr:.6f} t={time.monotonic()-t0:.1f}s")
        else:
            stuck += 1
            lr *= 0.92
            if stuck > 200:
                # Restart with fresh lr
                lr = 0.003
                stuck = 0
            if lr < 1e-9:
                break

    return best_a


def _direct_pgd(a_init, deadline, n_steps=8000, verbose=True, t0=0.0):
    """Projected gradient descent on soft-min directly in a-space (no softmax)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    beta = max(float(c_sorted[0]) - float(c_sorted[min(5, len(c_sorted)-1)]),
               float(c_sorted[0]) * 0.01)

    lr = 0.001
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_exp = e.sum()
        if s_exp < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_exp

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h  # grad of soft-min w.r.t. a
        grad = grad - np.mean(grad)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.003, 0.008)
        else:
            lr *= 0.94
            if lr < 1e-10:
                break

        if i % 200 == 0:
            beta *= 0.6
            beta = max(beta, c_max * 5e-7)

        if verbose and i % 2000 == 0:
            print(f"    [pgd] it={i} beta={beta:.2e} score={best_sc:.10f} "
                  f"t={time.monotonic()-t0:.1f}s")

    return best_a


def _perturb_diverse(a, rng, n_pert=4):
    """Generate diverse perturbations of a for multi-start."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    results = []

    # Small Gaussian
    z1 = z + 0.02 * rng.standard_normal(n)
    results.append(_softmax(z1 - z1.mean()))

    # Medium Gaussian
    z2 = z + 0.05 * rng.standard_normal(n)
    results.append(_softmax(z2 - z2.mean()))

    # Fourier perturbation
    z3 = z.copy()
    t = np.arange(n)
    for _ in range(3):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.03 * rng.uniform(1.0, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z3 = z3 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    results.append(_softmax(z3 - z3.mean()))

    # Block shuffle
    z4 = z.copy()
    n_blocks = 6
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = int(rng.integers(0, n - 2 * block_size))
        j = int(rng.integers(0, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z4[i:i + block_size].copy()
        z4[i:i + block_size] = z4[j:j + block_size]
        z4[j:j + block_size] = tmp
    results.append(_softmax(z4 - z4.mean()))

    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    1. Adaptive-continuation L-BFGS (main workhorse)
    2. Active-set gradient polishing
    3. Direct projected gradient
    4. Multi-n exploration
    5. Diverse restarts
    6. Final ultra-low-beta polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw / parent_raw.sum()
                print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = _score(parent)
    else:
        n_init = 8192
        best_seq = np.ones(n_init) / n_init
        best_score = _score(best_seq)
        print(f"No parent, using uniform n={n_init}")

    print(f"Initial: score={best_score:.10f}, n={len(best_seq)}, "
          f"deadline in {deadline - t_start:.0f}s")

    # === Phase 1: Main adaptive continuation ===
    print(f"\n[Phase 1] Adaptive continuation L-BFGS")
    phase1_budget = deadline - (deadline - t_start) * 0.50
    if time.monotonic() < phase1_budget:
        a1 = _run_continuation(best_seq, phase1_budget, min_time_remaining=60,
                               verbose=True, t0=t_start)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  Phase 1 result: {best_score:.10f}")
        else:
            print(f"  Phase 1 no improvement ({sc1:.10f} >= {best_score:.10f})")

    # === Phase 2: Active-set polish (K=2, 3, 5) ===
    print(f"\n[Phase 2] Active-set polishing")
    for K in [2, 3, 5]:
        if time.monotonic() > deadline - 200:
            break
        a2 = _active_set_polish(best_seq, min(deadline - 180, time.monotonic() + 120),
                                K=K, max_iter=8000, verbose=False, t0=t_start)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  Polish K={K}: improved to {best_score:.10f}")
        else:
            print(f"  Polish K={K}: no improvement ({sc2:.10f})")

    # === Phase 3: Direct PGD ===
    print(f"\n[Phase 3] Direct projected gradient descent")
    if time.monotonic() < deadline - 120:
        a3 = _direct_pgd(best_seq, min(deadline - 100, time.monotonic() + 100),
                         n_steps=6000, verbose=False, t0=t_start)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  PGD: improved to {best_score:.10f}")
        else:
            print(f"  PGD: no improvement ({sc3:.10f})")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n exploration")
    n_candidates = [4096, 8192, 16384]
    # Avoid duplicating current n
    n_candidates = [x for x in n_candidates if x != len(best_seq)]
    n_candidates.insert(0, len(best_seq))  # Keep current n first for efficiency

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 80:
            break
        # Interpolate to target n
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, best_seq)
        a_init = np.maximum(a_init, 1e-15)
        a_init = a_init / a_init.sum()

        min_remain = 40 if n_try <= 8192 else 20
        if time.monotonic() > deadline - min_remain - 30:
            continue

        a4 = _run_continuation(a_init, min(deadline - min_remain, time.monotonic() + 120),
                               min_time_remaining=min_remain, verbose=False, t0=t_start)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  n={n_try}: improved to {best_score:.10f}")
        else:
            print(f"  n={n_try}: score={sc4:.10f} (no improvement)")

        # Quick polish
        if sc4 < best_score + 0.002 and time.monotonic() < deadline - 20:
            a4b = _active_set_polish(a4, min(deadline - 10, time.monotonic() + 60),
                                     K=3, max_iter=3000, verbose=False, t0=t_start)
            sc4b = _score(a4b)
            if sc4b < best_score:
                best_score = sc4b
                best_seq = a4b.copy()
                print(f"  n={n_try} polished: {best_score:.10f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n[Phase 5] Diverse restarts")
    if time.monotonic() < deadline - 40:
        perturbations = _perturb_diverse(best_seq, rng)
        for pi, a_pert in enumerate(perturbations[:3]):
            if time.monotonic() > deadline - 20:
                break
            min_remain = 10
            a5 = _run_continuation(a_pert, min(deadline - min_remain, time.monotonic() + 40),
                                   min_time_remaining=min_remain, verbose=False, t0=t_start)
            sc5 = _score(a5)
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()
                print(f"  restart {pi}: improved to {best_score:.10f}")
            else:
                print(f"  restart {pi}: {sc5:.10f} (no improvement)")

    # === Phase 6: Final ultra-low-beta polish ===
    print(f"\n[Phase 6] Final ultra-low-beta L-BFGS + polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        c_sorted = np.sort(c)[::-1]
        gap_final = float(c_sorted[0]) - float(c_sorted[min(2, len(c_sorted)-1)])
        beta_final = max(gap_final * 0.1, float(c_sorted[0]) * 1e-7)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            _SM_STATE['beta'] = beta_final
            _SM_STATE['n'] = n
            max_iter = min(5000, max(200, int(remaining / 0.004)))
            try:
                result = minimize(
                    _smoobj_scipy, z, jac=_smograd_scipy, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 50, 'maxls': 100}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_score = sc_fin
                    best_seq = a_fin.copy()
                    print(f"    Final L-BFGS: {best_score:.10f}")
            except Exception:
                pass

        # Last active-set polish
        if time.monotonic() < deadline - 2:
            a_last = _active_set_polish(best_seq, max(deadline - 1, time.monotonic() + 30),
                                        K=2, max_iter=5000, verbose=False, t0=t_start)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
                print(f"    Final polish: {best_score:.10f}")

    # === Final validation and return ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                sc_final = min(sc_final, sc_eval)
        except Exception:
            pass
        print(f"\n{'='*60}")
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
