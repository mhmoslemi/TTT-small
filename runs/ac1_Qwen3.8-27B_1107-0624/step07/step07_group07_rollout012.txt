def _proj_simplex(v):
    """Project v onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    I = np.where(cond)[0]
    if len(I) == 0:
        return np.ones(n) / n
    rho = I[-1]
    tau = (cssv[rho]) / (rho + 1)
    return np.maximum(v - tau, 0.0)

def _compute_Q_times_a(a, k_min, k_max):
    """
    Compute Q*a where Q[j,l] = 1[k_min <= j+l <= k_max] / m.
    Uses prefix sums for O(n) computation.
    """
    n = len(a)
    m = k_max - k_min + 1
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(a)
    
    # For each j, the valid range of l is [max(0, k_min - j), min(n-1, k_max - j)]
    j = np.arange(n)
    l_low = np.maximum(0, k_min - j)
    l_high = np.minimum(n - 1, k_max - j)
    
    # Valid if l_low <= l_high
    valid = l_low <= l_high
    l_low_clamped = np.clip(l_low, 0, n)
    l_high_clamped = np.clip(l_high + 1, 0, n + 1)  # +1 for exclusive upper in prefix
    
    result = np.zeros(n)
    result[valid] = (prefix[l_high_clamped[valid]] - prefix[l_low_clamped[valid]]) / m
    return result

    """
    Solve: minimize a^T Q a subject to a in simplex,
    where Q[j,l] = 1[k_min <= j+l <= k_max] / m.
    Using FISTA (accelerated projected gradient).
    """

    m = k_max - k_min + 1
    
    if step is None:
        # Step size = 1 / (2 * lambda_max(Q))
        # lambda_max(Q) <= max_j (sum_l Q[j,l]) = max_j (count of valid l) / m
        # The max count is at most min(n, m)
        # So lambda_max <= min(n, m) / m
        # L = 2 * min(n, m) / m
        # step = 1/L = m / (2 * min(n, m))
        step = m / (2.0 * min(n, m))
    
    x = a_init.copy()
    y = a_init.copy()
    t = 1.0
    
    best_x = x.copy()
    best_obj = float(x @ (_compute_Q_times_a(x, k_min, k_max)))
    
    for k in range(max_iter):
        # Gradient: 2 * Q * y
        Qy = _compute_Q_times_a(y, k_min, k_max)
        grad = 2.0 * Qy
        
        # Gradient step
        z = y - step * grad
        
        # Projection
        x_new = _proj_simplex(z)
        
        # Check objective
        Qx = _compute_Q_times_a(x_new, k_min, k_max)
        obj = float(x_new @ Qx)
        if obj < best_obj:
            best_obj = obj
            best_x = x_new.copy()

        # Ensure y is in simplex (it should be, but just in case)
        y = _proj_simplex(y)
        
        x = x_new
        t = t_new
    
    return best_x

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    cssv = np.cumsum(u) - 1.0

    I = np.where(cond)[0]

    rho = I[-1]

def _fista_qp(a_init, k_min, k_max, max_iter=150, step_scale=0.85):
    """Solve QP: min a^T Q a s.t. a in simplex, Q[j,l]=1[k_min<=j+l<=k_max]/m."""
    n = len(a_init)
    m = k_max - k_min + 1
    step = step_scale * m / (2.0 * n)
    step = min(step, 0.49)

    x = a_init.copy()
    y = a_init.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = float(x @ _compute_Qa_prefix(x, k_min, k_max))

    for k in range(max_iter):
        Qy = _compute_Qa_prefix(y, k_min, k_max)
        grad = 2.0 * Qy
        z = y - step * grad
        x_new = _proj_simplex(z)
        Qx = _compute_Qa_prefix(x_new, k_min, k_max)
        obj = float(x_new @ Qx)
        if obj < best_obj:
            best_obj = obj
            best_x = x_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _proj_simplex(y)
        x = x_new
        t = t_new
    return best_x


def _fista_qp_fft(a_init, w, m, max_iter=100, step_scale=0.85):
    """Solve QP with arbitrary active set w using FFT."""
    n = len(a_init)
    step = step_scale * m / (2.0 * n)
    step = min(step, 0.49)

    x = a_init.copy()
    y = a_init.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = float(x @ _compute_Qa_fft(x, w, m))

    for k in range(max_iter):
        Qy = _compute_Qa_fft(y, w, m)
        grad = 2.0 * Qy
        z = y - step * grad
        x_new = _proj_simplex(z)
        Qx = _compute_Qa_fft(x_new, w, m)
        obj = float(x_new @ Qx)
        if obj < best_obj:
            best_obj = obj
            best_x = x_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _proj_simplex(y)
        x = x_new
        t = t_new
    return best_x


def _active_set_optimize(a_init, deadline, max_outer=300, inner_iters=120):
    """Active-set method: identify near-max lags, solve QP, iterate."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    no_improve = 0

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_center = int(np.argmax(c))

        for tol_frac in [0.003, 0.01, 0.03, 0.08]:
            tol = tol_frac * c_max
            active_mask = c >= (c_max - tol)
            if active_mask.sum() < 3:
                continue

            # Find contiguous range around center
            k_left = k_center
            while k_left > 0 and active_mask[k_left - 1]:
                k_left -= 1
            k_right = k_center
            while k_right < len(c) - 1 and active_mask[k_right + 1]:
                k_right += 1

            m = k_right - k_left + 1
            if m < 3:
                continue

            a_new = _fista_qp(a, k_left, k_right, max_iter=inner_iters)
            sc = _score(a_new)

            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve >= 8:
                break

    return best_a, best_score


def _reweighting_optimize(a_init, deadline, max_outer=100, inner_iters=80):
    """Iterative reweighting: softmax-weighted QP subproblems."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    beta_rw = 0.01

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        weights = np.exp((c - c_max) / (beta_rw * c_max + 1e-30))
        total_w = weights.sum()
        if total_w < 1e-10:
            break
        w = weights / total_w
        m = float(total_w)

        a_new = _fista_qp_fft(a, w, m, max_iter=inner_iters)
        sc = _score(a_new)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            beta_rw *= 0.95
        else:
            beta_rw = max(0.005, beta_rw * 0.8)

    return best_a, best_score


def _smoobj(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter=5000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = np.exp(result.x - result.x.max())
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        return a_init


def _lbfgs_schedule(a_start, deadline, reserve=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.3, 0.15, 0.07, 0.035, 0.018, 0.009,
                              0.004, 0.002, 0.001, 0.0005,
                              0.0002, 0.0001, 5e-5, 2e-5, 1e-5])
    current_a = a_start.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve > deadline:
            break
        remaining = deadline - reserve - time.monotonic()
        if remaining < 20:
            break
        max_iter = min(8000, max(500, int(remaining / 0.004)))
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < _score(current_a) - 1e-15:
            current_a = a_new

    return current_a


def _peak_shaving(a_init, deadline, max_iters=30000):
    """Direct peak shaving with adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    eta = 0.015
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_normed = g / g_norm

        best_eta = eta
        best_sc = best_score
        best_a_try = a

        for frac in [1.0, 0.5, 0.25, 0.125]:
            step = eta * frac
            a_try = a - step * g_normed
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a_try = a_try.copy()
                best_eta = step

        if best_sc < best_score - 1e-14:
            best_score = best_sc
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.08)
            stall = 0
        else:
            stall += 1
            eta *= 0.88
            if eta < 1e-7 or stall > 150:
                break

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spectral":
        # Initialize via flat-spectrum construction
        rng_local = np.random.default_rng(12345)
        phases = rng_local.uniform(0, 2 * np.pi, n)
        phases[0] = 0.0
        spectrum = np.exp(1j * phases)
        a = np.real(np.fft.ifft(spectrum))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        return a
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    return a_interp / a_interp.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining active-set FISTA QP solving,
    L-BFGS smoothing, peak shaving, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Using uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init")

    start_n = len(best_seq) if best_seq is not None else 4096

    # === Phase 1: Active-set FISTA (primary novel method) ===
    print(f"\n=== Phase 1: Active-set FISTA on n={start_n} ===")
    t_phase = time.monotonic()
    a_as, _ = _active_set_optimize(best_seq, deadline - 500, max_outer=300, inner_iters=150)
    try_update(a_as, "active_set")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 2: Reweighting optimization ===
    print(f"\n=== Phase 2: Reweighting ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400:
        a_rw, _ = _reweighting_optimize(best_seq, deadline - 400, max_outer=80, inner_iters=80)
        try_update(a_rw, "reweighting")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: L-BFGS smoothing ===
    print(f"\n=== Phase 3: L-BFGS smoothing ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_lb = _lbfgs_schedule(best_seq, deadline - 300, reserve=200)
        try_update(a_lb, "lbfgs")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 200:
        a_ps, _ = _peak_shaving(best_seq, deadline - 200, max_iters=30000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Second round of active-set on best ===
    print(f"\n=== Phase 5: Active-set round 2 ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 150:
        a_as2, _ = _active_set_optimize(best_seq, deadline - 150, max_outer=200, inner_iters=200)
        try_update(a_as2, "active_set_2")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Try larger n ===
    print(f"\n=== Phase 6: Multi-length ===")
    t_phase = time.monotonic()
    target_lengths = []
    if start_n <= 8192:
        target_lengths.append(12288)
    if start_n <= 8192:
        target_lengths.append(16384)
    if start_n < 4096:
        target_lengths.append(8192)

    for tn in target_lengths:
        if time.monotonic() > deadline - 80:
            break
        print(f"  Trying n={tn}...")
        a_interp = _interpolate_to_n(best_seq, tn)
        sc_i = _score(a_interp)
        print(f"    Interpolated score: {sc_i:.10f}")
        try_update(a_interp, f"interp_n{tn}")

        if time.monotonic() < deadline - 60:
            a_opt = _active_set_optimize(a_interp, deadline - 40, max_outer=50, inner_iters=80)
            try_update(a_opt, f"as_n{tn}")

        if time.monotonic() < deadline - 30:
            a_lb2 = _lbfgs_schedule(a_interp, deadline - 20, reserve=10)
            try_update(a_lb2, f"lbfgs_n{tn}")

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(4):
            if time.monotonic() > deadline - 3:
                break
            a_f1, _ = _peak_shaving(best_seq, deadline - 3, max_iters=10000)
            try_update(a_f1, f"ps_{cycle}")
            if time.monotonic() > deadline - 3:
                break
            a_f2, _ = _active_set_optimize(best_seq, deadline - 3, max_outer=50, inner_iters=200)
            try_update(a_f2, f"as_{cycle}")
    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
