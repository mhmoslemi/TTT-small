from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _solve_kkt(n, k_lo, k_hi):
    """Solve A*x = 1 where A[i,j] = 1/m if k_lo <= i+j <= k_hi."""
    m = k_hi - k_lo + 1
    inv_m = 1.0 / m
    
    rows = []
    cols = []
    vals = []
    for i in range(n):
        j_lo = max(0, k_lo - i)
        j_hi = min(n - 1, k_hi - i)
        if j_lo <= j_hi:
            rows.extend([i] * (j_hi - j_lo + 1))
            cols.extend(range(j_lo, j_hi + 1))
            vals.extend([inv_m] * (j_hi - j_lo + 1))
    
    A = csr_matrix((vals, (rows, cols)), shape=(n, n))
    x = spsolve(A, np.ones(n))
    
    if not np.all(np.isfinite(x)):
        return None
    
    s = x.sum()
    if abs(s) < 1e-15:
        return None
    
    a = x / s
    return a
