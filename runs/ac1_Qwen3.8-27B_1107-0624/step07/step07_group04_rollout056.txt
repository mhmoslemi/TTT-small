    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=200, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-300:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-300 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _convolution_gradient(a, w, n):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a, using FFT."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _newton_active_polish(a_init, deadline, max_rounds=80):
    """Multi-cardinality active set Newton polish."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    active_sizes = [1, 2, 3, 5, 8, 12, 20, 30, 50]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: try multiple sizes, pick best
        best_improve = -np.inf
        best_a_new = None

        for n_active in active_sizes:
            if n_active >= len(c):
                n_active = len(c) - 1
            tol = c_max * (0.0002 + 0.01 * (n_active / 50.0))
            active = c >= (c_max - tol)
            if active.sum() < n_active:
                idx = np.argsort(c)[::-1][:n_active]
                active = np.zeros_like(c, dtype=bool)
                active[idx] = True
            elif active.sum() > n_active * 1.5:
                idx = np.argsort(c)[::-1][:n_active]
                active = np.zeros_like(c, dtype=bool)
                active[idx] = True

            n_act = int(active.sum())
            if n_act == 0:
                active[int(np.argmax(c))] = True
                n_act = 1

            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / n_act

            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-14 * n:
                continue

            Hg = _cg_solve(W, n, g, max_iter=150)
            H1 = _cg_solve(W, n, np.ones(n), max_iter=150)
            H1_norm = float(np.linalg.norm(H1))
            if H1_norm < 1e-25 or not np.isfinite(H1_norm):
                dx = -g_t / (gn + 1e-30)
            else:
                lam = float(np.ones(n) @ Hg) / (float(np.ones(n) @ H1) + 1e-30)
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / (gn + 1e-30)

            # Backtracking
            alpha = 1.0
            for ls in range(40):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max:
                    improve = c_max - c_max_try
                    if improve > best_improve:
                        best_improve = improve
                        best_a_new = a_try.copy()
                    break
                alpha *= 0.5
                if alpha < 1e-14:
                    break

        if best_a_new is not None:
            a = best_a_new
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try gradient step as fallback
            c = _full_convolve(a, a)
            c_max = float(c.max())
            tol = 0.001 * c_max
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            W = np.zeros(2 * n - 1)
            W[active] = 1.0 / active.sum()
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-14:
                break
            alpha = 0.1
            step_done = False
            for ls in range(30):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    sc = _score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    step_done = True
                    break
                alpha *= 0.5
            if not step_done:
                break

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad = _convolution_gradient(a, w, n)
    # Subtract mean to account for sum constraint in softmax parameterization
    return 2.0 * n * (grad - float(np.dot(w, c)) * 2.0 * a)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    # More aggressive: start smaller, more steps
    ratios = np.array([1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.025, 0.015,
                       0.008, 0.004, 0.002, 0.001, 0.0005, 0.0003,
                       0.00015, 0.00008, 0.00004, 0.00002, 0.00001,
                       0.000005, 0.000002, 0.000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient with Polyak stepping and dynamic active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    best_c = _full_convolve(best_a, best_a).max()
    f_star = best_sc
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = float(2.0 * n * c_max)  # normalized by sum=1

        # Adaptive active set
        tol = max(0.0005 * c_max, c_max * (i / max_steps) * 0.01)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()

        grad = _convolution_gradient(a, w, n)
        W_val = float(np.dot(w, c))
        grad = 2.0 * n * (grad - 2.0 * W_val * a)
        grad = grad - float(a @ grad)  # project out sum direction
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        polyak_step = max(1e-10, (sc - f_star) / max(grad_norm * grad_norm, 1e-30))
        step = min(0.05, 0.01 * polyak_step / (1.0 + 0.001 * i))

        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Evaluate every 10 steps
        if i % 10 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-14:
                best_sc = sc_check
                best_a = a_check.copy()
                f_star = sc_check
                patience = 0
            else:
                patience += 1
                if patience > 100:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.20]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.03
        return a / a.sum()
    elif kind == "tripod":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.15) ** 2) + np.exp(-12.0 * (t - 0.5) ** 2) + np.exp(-12.0 * (t - 0.85) ** 2)
        return a / a.sum()
    elif kind == "flat_center":
        a = np.ones(n) * 0.3
        center = n // 2
        width = n // 3
        a[center - width:center + width] = 1.0
        return a / a.sum()
    elif kind == "rand_smooth":
        rng = np.random.default_rng(12345)
        a = rng.standard_normal(n)
        # Smooth
        kernel = np.exp(-np.arange(-20, 21) ** 2 / 200.0)
        kernel = kernel / kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a) + 1e-6
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _spectral_init(n, rng):
    """Initialize using frequency domain: try to make power spectrum flat."""
    N = _next_pow2(2 * n - 1)
    # Start with random phase, flat magnitude
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = 1.0
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * 2.0 * np.pi * rng.random())
    a = np.fft.irfft(A, N)[:n]
    a = np.abs(a) + 1e-10
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with multi-resolution warm start,
    adaptive active-set Newton, aggressive beta continuation,
    subgradient with Polyak stepping, and diverse exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    # Phase 0: Multi-resolution warm start
    print(f"\n[Phase 0] Multi-resolution warm start (t={time.monotonic() - t_start:.1f}s)")
    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        update_best(parent, _score(parent), "init_tapered")

    # Try smaller n first for faster convergence, then upscale
    small_n = max(256, n_parent // 16)
    a_small = _resample(parent, small_n)
    a_small = _run_lbfgs(a_small, deadline, min_time_remaining=1200.0)
    sc_small = _score(a_small)
    print(f"  Small n={small_n}: score={sc_small:.10f}")
    update_best(a_small, sc_small, f"small_{small_n}")

    # Upscale and use as alternative initial
    a_up = _resample(a_small, n_parent)
    a_up = _run_lbfgs(a_up, deadline, min_time_remaining=1000.0)
    sc_up = _score(a_up)
    update_best(a_up, sc_up, f"upscale_{n_parent}")

    # Phase 1: Main L-BFGS from best so far
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        a1 = _run_lbfgs(best_seq, deadline, min_time_remaining=800.0)
        update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Newton active set polish
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 500, time.monotonic() + 200))
        update_best(a2, sc2, "newton_1")

    # Phase 3: Subgradient
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _subgrad_polish(best_seq, min(deadline - 400, time.monotonic() + 150))
        update_best(a3, sc3, "sg_1")

    # Phase 4: Newton again
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 120))
        update_best(a4, sc4, "newton_2")

    # Phase 5: Shrinkage
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _shrinkage_reopt(best_seq, deadline, min_time_remaining=120.0)
        update_best(a5, sc5, "shrinkage")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 4096, 8192, 2048]
    if n_parent in n_candidates:
        n_candidates.remove(n_parent)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample(best_seq if best_seq is not None else parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 80:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 60, time.monotonic() + 60), max_rounds=40)
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        profiles = ["cos", "bimodal", "tripod", "flat_center", "rand_smooth"]
        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 40:
                break
            if i < 2:
                p_start = _perturb(best_seq, 0.05 + 0.05 * i, rng)
            else:
                p_start = _make_profile(len(best_seq), prof)
                p_start = _perturb(p_start, 0.03, rng)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=30.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{prof}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 20:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 10, time.monotonic() + 40), max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{prof}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=120)
        update_best(a_fin, sc_fin, "final_newton")

        if time.monotonic() < deadline - 3:
            n_fin = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n_fin),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n_fin),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
