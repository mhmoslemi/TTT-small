"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
Newton-CG with FFT-accelerated Hessian-vector products, adaptive log-sum-exp
beta continuation, spectral shaping for local minima escape, and active set
refinement. Key innovations: second-order Newton-CG (O(n log n) per matvec),
spectral domain power flattening, and dual-tracking active set identification.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_weights(c, beta):
    c_max = c.max()
    shifted = (c - c_max) / beta
    max_shifted = shifted.max()
    e = np.exp(shifted - max_shifted)
    w = e / e.sum()
    lse_val = c_max + beta * (max_shifted + np.log(e.sum()))
    return w, lse_val


def _grad_lse(a, w, n):
    """Gradient of 2n*LSE(conv(a,a)) with respect to a (using softmax parametrization).
    Returns gradient in a-space (before softmax Jacobian)."""
    # grad_a[j] = 4n * h[j] where h[j] = sum_k w_k * a_{k-j}
    # h = cross_corr(w, a) valid part (first n values)
    N = _next_pow2(3 * n - 1)
    Wft = np.fft.fft(w, N)
    Aft = np.fft.fft(a, N)
    h = np.fft.ifft(Wft * np.conj(Aft))[:n].real
    return 4.0 * n * h


def _hess_vec_lse(a, w, v, beta, n):
    """Compute Hessian-vector product for the LSE smoothed objective.
    H = (2n/beta) * Jc^T (diag(w)-ww^T) Jc
    Returns H @ v."""
    # u = Jc @ v: u[k] = 2 * conv(a, v)[k]
    u = 2.0 * _full_convolve(a, v)
    # r = (diag(w) - ww^T) u = w * (u - dot(w, u))
    wu = float(np.dot(w, u))
    r = w * (u - wu)
    # Jc^T @ r: result[j] = 2 * sum_k a[k-j] * r[k] = 2 * cross_corr(r, a)[j]
    N = _next_pow2(3 * n - 1)
    Rft = np.fft.fft(r, N)
    Aft = np.fft.fft(a, N)
    cc = np.fft.ifft(Rft * np.conj(Aft))[:n].real
    return (4.0 * n / beta) * cc


def _newton_cg_step(a, w, beta, n, max_cg=40, tol=1e-12):
    """Compute Newton direction using CG on the projected Hessian."""
    # Gradient
    g = _grad_lse(a, w, n)
    g_proj = g - g.mean()
    g_norm = np.linalg.norm(g_proj)
    if g_norm < 1e-15:
        return np.zeros(n), g_proj

    # Regularization
    reg_test = _hess_vec_lse(a, w, g_proj / g_norm, beta, n)
    reg = max(1e-8, 1e-6 * float(np.dot(g_proj / g_norm, reg_test)))

    # CG to solve (H + reg*I) d = -g_proj
    r = -g_proj.copy()
    p = r.copy()
    r_norm_sq = float(np.dot(r, r))
    d = np.zeros(n)

    for i in range(max_cg):
        Hp = _hess_vec_lse(a, w, p, beta, n) + reg * p
        pH = float(np.dot(p, Hp))
        if pH < 1e-14:
            Hp += (1e-6 * r_norm_sq / max(1e-30, float(np.dot(p, p)))) * p
            pH = float(np.dot(p, Hp))
            if pH < 1e-14:
                break
        alpha = r_norm_sq / pH
        d += alpha * p
        r_old = r.copy()
        r -= alpha * Hp
        r_norm_sq_new = float(np.dot(r, r))
        d = d - d.mean()  # Maintain projected
        if r_norm_sq_new < tol * g_norm * g_norm:
            break
        beta_cg = r_norm_sq_new / max(1e-30, r_norm_sq)
        p = r + beta_cg * p
        p = p - p.mean()
        r_norm_sq = r_norm_sq_new

    d = d - d.mean()
    return d, g_proj

    """Newton-CG optimization in a-space (with simplex projection)."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    step_size = step_init

    for i in range(max_steps):
        if time.monotonic() > deadline - min_time:
            break

        c = _full_convolve(a, a)
        w, lse_val = _lse_weights(c, beta)
        d, g_proj = _newton_cg_step(a, w, beta, n, max_cg=30)

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break

        d = d / d_norm * step_size

        # Line search
        best_a_try = a.copy()
        best_sc_try = best_sc
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a + frac * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)

                if frac == 1.0:

            best_a = best_a_try.copy()

            if step_size < 1e-8:
                break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a
