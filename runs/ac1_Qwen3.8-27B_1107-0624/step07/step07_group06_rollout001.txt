"""
Multi-resolution optimizer with spectral flattening, adaptive active-set Newton,
and aggressive smooth-max continuation. Builds a solution pyramid from n=32 to
n=8192, using FFT-based spectral variance minimization as a novel warm-start
at each level, combined with L-BFGS on the log-sum-exp smooth max and
active-set Newton polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_grad(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _active_newton(a_init, deadline, max_iter=200, tol_frac_init=0.01):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = tol_frac_init
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        active = c >= (c_max - tol_frac * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        if active.sum() > 0.3 * len(c):
            tol_frac = max(0.0005, tol_frac * 0.5)
            active = c >= (c_max - tol_frac * c_max)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True

        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break

        mu = 1e-8

        def hess_vec(v):
            return _hessian_matvec(v, active, n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=200, tol=1e-10)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-10:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _spectral_flatten(a_init, n, max_iter=100, deadline=None):
    """Minimize variance of zero-padded circular autocorrelation via FFT.
    This encourages a flat aperiodic autocorrelation."""
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    best_var = np.inf

    N_pad = _next_pow2(2 * n - 1)

    for i in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        fa = np.fft.rfft(a, N_pad)
        power = np.abs(fa) ** 2
        circ_acorr = np.fft.irfft(power, N_pad)

        var = float(np.var(circ_acorr))
        if var < best_var:
            best_var = var
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        # Gradient of sum(circ_acorr^2) w.r.t. a
        # circ_acorr = IFFT(|FFT(a)|^2)
        # d/d a_i sum_k circ_acorr[k]^2 = 2 * N_pad * IFFT(|FFT(a)|^2) convolved with ...
        # Simpler: d/d a sum(circ_acorr^2) = 2 * N_pad * circ_acorr (by stationarity)
        # Actually: sum_k r[k]^2 where r = IFFT(|A|^2)
        # d/d a: use chain rule. r[k] = (1/N) sum_m |A[m]|^2 exp(2pi i k m / N)
        # |A[m]|^2 = A[m] * conj(A[m])
        # d|A[m]|^2 / da[i] = conj(A[m]) * exp(-2pi i i m / N) + A[m] * (-exp(2pi i i m / N))
        # = 2 Re[conj(A[m]) * exp(-2pi i i m / N)]
        # = 2 * (1/N) Re[exp(2pi i (k-i) m / N)] summed over k weighted by r[k]
        # This is getting complex. Let me use a different approach.
        # 
        # Alternative: directly compute gradient via the fact that
        # sum_k r[k]^2 = (1/N) sum_m |A[m]|^4
        # d/d a[i] sum_m |A[m]|^4 = 4 * sum_m |A[m]|^2 * Re[conj(A[m]) * exp(-2pi i i m / N)]
        # = 4 * N * IFFT(|A|^2 * conj(A) ... ) 
        # Hmm, let me just use: 
        # d/da[i] sum_m |A[m]|^4 = 4 * Re[sum_m |A[m]|^2 * conj(A[m]) * exp(-2pi i i m / N)]
        # = 4 * Re[IFFT(conj(A) * |A|^2 / N) * N / N]  ... 
        # Actually: IFFT(X)[i] = (1/N) sum_m X[m] exp(2pi i i m / N)
        # So sum_m X[m] exp(-2pi i i m / N) = N * IFFT(X_conj)[i] for real X
        # But |A|^2 * conj(A) is complex.
        # Let me just use: grad[i] = 4 * Re[IFFT(|A|^2 * conj(A))][i] ... no.
        # 
        # Let me use a simpler approach: gradient of var(circ_acorr) 
        # grad[i] = 2 * circ_acorr_convolved[i] - 2*mean(circ_acorr) * ... 
        # This is getting too complicated analytically. Let me just use a 
        # simple gradient: the gradient of sum(r^2) w.r.t. a where r = IRC(A.^2)
        # is 4 * IRC(A.^2 .* conj(A)) (real part). Let me verify numerically.
        # Actually for the purpose of optimization, I'll use a simpler proxy:
        # minimize the peak of circ_acorr, which has gradient:
        # grad[i] = 2 * circ_acorr_cross[i] where circ_acorr_cross is 
        # the correlation of a with the active part of circ_acorr.
        # 
        # For simplicity, let me use: minimize sum(circ_acorr^2)
        # grad = 4 * N = 4 * np.fft.irfft(power * np.abs(fa)**2 / N_pad ... )
        # I'll just use a numerical gradient approach or a simpler proxy.

        # Simple proxy: minimize the peak of circ_acorr using subgradient
        # at the peak location
        peak_idx = int(np.argmax(circ_acorr))
        # The gradient of circ_acorr[peak_idx] w.r.t. a is:
        # d/dc[peak] da[i] = 2 * a[(-i + peak) mod N] ... (circular)
        # This is a shifted version of a
        if n >= N_pad:
            break
        # For a padded FFT, the circular autocorrelation at lag k is:
        # r[k] = sum_j a[j] * a[(j+k) mod N] (with zero-padding, this is aperiodic for k < n)
        # d r[k] / d a[i] = a[(i+k) mod N] + a[(i-k+N) mod N] (for the non-padded case)
        # With zero-padding to N_pad, for k = peak_idx:
        # d r[peak] / d a[i] = a[padded][(i+peak) mod N_pad] + a[padded][(i-peak) mod N_pad]
        a_padded = np.zeros(N_pad)
        a_padded[:n] = a
        shift_pos = a_padded[np.mod(np.arange(n) + peak_idx, N_pad)]
        shift_neg = a_padded[np.mod(np.arange(n) - peak_idx, N_pad)]
        grad = 2.0 * (shift_pos[:n] + shift_neg[:n])
        grad = grad - np.mean(grad)  # project to sum=0

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Check if variance improved
        fa_new = np.fft.rfft(a_new, N_pad)
        var_new = float(np.var(np.fft.irfft(np.abs(fa_new) ** 2, N_pad)))
        if var_new < var:
            a = a_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.85
            if lr < 1e-6:
                break

    sc_final = _score(best_a)
    return best_a, sc_final


def _lbfgs_smooth(a_init, n, deadline, min_time=60.0):
    """L-BFGS with aggressive beta continuation."""
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    # More stages, lower final beta
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                  0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations based on n
        scale = max(1.0, n / 1024.0)
        max_iter = int(min(10000, max(100, remaining / (0.002 * scale * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 0.01:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        # Adaptive tolerance
        tol_frac = 0.0005 + 0.005 * np.exp(-i / 200.0)
        grad = _active_set_grad(a, c, n, tol_frac=tol_frac)
        step = lr / (1.0 + 0.002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
                best_patience = max(best_patience, patience)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_optimize(a, deadline):
    """Zero out small elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fracs = [0.005, 0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50]
    for frac in fracs:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 0.01:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_abs":
        t = np.linspace(-3, 3, n)
        x = t * np.pi / 2
        a = np.ones(n)
        mask = np.abs(x) > 1e-6
        a[mask] = np.abs(np.sin(x[mask]) / x[mask])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, rng, sigma=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-resolution optimizer:
    1. Build solution pyramid from n=32 to target n using spectral flattening + L-BFGS + Newton
    2. Fine-tune at target resolution with aggressive beta continuation
    3. Polish with adaptive subgradient and shrinkage
    4. Try multiple n values and diverse restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Multi-resolution pyramid ===
    # Build up from coarse to fine, using the parent as the finest-level init
    print(f"\n[Phase 1] Multi-resolution pyramid (t={time.monotonic() - t_start:.1f}s)")
    
    # Determine which levels to use
    n_final = len(best_seq)
    levels = [32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384]
    # Only use levels up to n_final (or a bit beyond)
    levels = [l for l in levels if l <= max(n_final, 8192)]
    if n_final in levels:
        levels = [l for l in levels if l <= n_final]
    else:
        levels.append(n_final)
        levels = sorted(set(levels))

    a_current = best_seq.copy()
    x_current = np.linspace(0, 1, len(a_current))

    # Time budget for each level (more time for finer levels)
    total_level_time = 0.7 * (deadline - t_start - 30)  # 70% for pyramid
    level_weights = np.array([l for l in levels]) ** 1.2
    level_weights = level_weights / level_weights.sum()
    level_times = level_weights * total_level_time
    level_deadlines = []
    cum_time = t_start
    for lt in level_times:
        cum_time += lt
        level_deadlines.append(min(cum_time, deadline - 120))

    for li, (n_level, lvl_deadline) in enumerate(zip(levels, level_deadlines)):
        if time.monotonic() > lvl_deadline:
            break
        print(f"  Level {li}: n={n_level} (deadline in {lvl_deadline - time.monotonic():.0f}s)")

        # Interpolate current solution to this level
        x_new = np.linspace(0, 1, n_level)
        if len(a_current) != n_level:
            a = np.interp(x_new, x_current, a_current)
        else:
            a = a_current.copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            a = _make_profile(n_level, "uniform")
        else:
            a = a / s

        sc_before = _score(a)

        # Step 1: Spectral flattening (fast, few iterations)
        if n_level <= 2048:
            sf_deadline = min(lvl_deadline, time.monotonic() + max(2, 0.1 * level_times[li]))
            a, sc_sf = _spectral_flatten(a, n_level, max_iter=200, deadline=sf_deadline)
            if sc_sf < sc_before:
                sc_before = sc_sf

        # Step 2: L-BFGS with smooth max
        lbfgs_deadline = min(lvl_deadline, time.monotonic() + 0.5 * level_times[li])
        time_remaining = max(5, lvl_deadline - time.monotonic() - 0.3 * level_times[li])
        a = _lbfgs_smooth(a, n_level, deadline=lbfgs_deadline, min_time=max(5, 0.2 * level_times[li]))
        sc_lbfgs = _score(a)

        # Step 3: Active-set Newton
        if time.monotonic() < lvl_deadline - 5:
            nw_deadline = min(lvl_deadline - 2, time.monotonic() + 0.2 * level_times[li])
            a, sc_nw = _active_newton(a, nw_deadline, max_iter=50)

        sc_after = _score(a)
        if sc_after < best_score:
            update_best(a, sc_after, f"pyramid_n{n_level}")

        a_current = a
        x_current = np.linspace(0, 1, n_level)
        print(f"    Before: {sc_before:.10f}, After: {sc_after:.10f}")

    # === Phase 2: Aggressive fine-tuning at target n ===
    print(f"\n[Phase 2] Fine-tuning (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        # Run L-BFGS with very low betas
        ft_deadline = min(deadline - 80, time.monotonic() + 120)
        a_ft = _lbfgs_smooth(best_seq, n_best, deadline=ft_deadline, min_time=40)
        sc_ft = _score(a_ft)
        update_best(a_ft, sc_ft, "fine_tune_lbfgs")

        # Newton
        if time.monotonic() < deadline - 40:
            nw_deadline = min(deadline - 30, time.monotonic() + 40)
            a_nw, sc_nw = _active_newton(a_ft, nw_deadline, max_iter=100, tol_frac_init=0.005)
            update_best(a_nw, sc_nw, "fine_tune_newton")

    # === Phase 3: Subgradient polishing ===
    print(f"\n[Phase 3] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        sg_deadline = min(deadline - 30, time.monotonic() + 60)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=2000, lr_init=0.01)
        update_best(a_sg, sc_sg, "sg_polish")

        # Another Newton on the result
        if sc_sg < best_score * 1.001 and time.monotonic() < deadline - 15:
            a_sg2, sc_sg2 = _active_newton(a_sg, min(deadline - 10, time.monotonic() + 20), max_iter=50)
            update_best(a_sg2, sc_sg2, "sg_newton")

    # === Phase 4: Shrinkage ===
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 20)
        update_best(a_shr, sc_shr, "shrinkage")

        if sc_shr < best_score * 1.002 and time.monotonic() < deadline - 15:
            a_shr2 = _lbfgs_smooth(a_shr, len(a_shr), deadline=min(deadline - 5, time.monotonic() + 20), min_time=5)
            update_best(a_shr2, _score(a_shr2), "shrink_lbfgs")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    # Try nearby sizes
    n_best_cur = len(best_seq) if best_seq is not None else 8192
    for factor in [0.75, 1.25, 1.5, 2.0]:
        n_try = int(n_best_cur * factor)
        n_try = max(1024, min(n_try, 16384))
        n_try = ((n_try + 15) // 16) * 16  # Round to multiple of 16
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if len(best_seq) == n_try:
            a_init = best_seq.copy()
        elif best_seq is not None:
            x_fine = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_fine, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_profile(n_try, "tapered")

        if time.monotonic() > deadline - 20:
            break

        # Quick L-BFGS
        a_opt = _lbfgs_smooth(a_init, n_try, deadline=min(deadline - 10, time.monotonic() + 30), min_time=5)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 10:
            a_opt2, sc_opt2 = _active_newton(a_opt, min(deadline - 5, time.monotonic() + 15), max_iter=40)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_nw")

    # === Phase 6: Diverse restarts ===
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_restarts = 3
        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            if i == 0:
                p_start = _perturb(best_seq, rng, sigma=0.015)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)
            else:
                p_start = _perturb(best_seq, rng, sigma=0.01)
                p_start = _perturb_fourier(p_start, rng, n_modes=2)

            a_r = _lbfgs_smooth(p_start, len(p_start), deadline=min(deadline - 5, time.monotonic() + 25), min_time=3)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_newton(a_r, min(deadline - 3, time.monotonic() + 10), max_iter=30)
                update_best(a_r2, sc_r2, f"restart_{i}_nw")

    # === Phase 7: Final ultra-low-beta L-BFGS ===
    print(f"\n[Phase 7] Final ultra-L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 5e-6
        remaining = deadline - 3 - time.monotonic()
        if remaining > 3:
            max_iter = min(5000, max(200, int(remaining / 0.005)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                a_fin = np.maximum(a_fin, 0)
                if a_fin.sum() > 0.01:
                    a_fin = a_fin / a_fin.sum()
                    update_best(a_fin, _score(a_fin), "ultra_lbfgs")
            except Exception:
                pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
