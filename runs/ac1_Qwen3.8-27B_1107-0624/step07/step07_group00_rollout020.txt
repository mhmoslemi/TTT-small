"""
Optimized multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Adaptive top-K gradient (K changes based on peak gap)
- Backtracking line search for better step sizes
- Simpler, more focused optimization loop
- Efficient O(n) coordinate descent for polishing
- Multi-n exploration (4096, 8192, 16384)

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_sum1(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _gradient_topk(a, c, K=3, T=None):
    """
    Compute gradient of the smoothed max of convolution.
    If T is None, uses exact top-K uniform weights.
    If T is given, uses softmax with temperature T.
    Returns gradient vector of length n.
    """
    n = len(a)
    if T is not None and T > 1e-10:
        # Softmax weights
        c_shifted = c - c.max()
        e = np.exp(c_shifted / T)
        w = e / e.sum()
    else:
        # Exact top-K
        w = np.zeros_like(c)
        top_k_idx = np.argsort(c)[::-1][:K]
        w[top_k_idx] = 1.0 / K
    
    # Gradient: (dF/da)_m = 4*n * sum_j w_j * a_{j-m}
    # = 4*n * (w * a) at appropriate indices
    # conv(w_rev, a) gives what we need
    w_rev = w[::-1]
    cross = _fft_convolve(w_rev, a)
    # cross[k] = sum_i w_rev[k-i] * a[i] = sum_i w[n-1-k+i] * a[i]
    # For k in [n-1, 2n-2]: cross[k] = sum_i w[i-(k-n+1)] * a[i]
    # Let m = k - (n-1), cross[n-1+m] = sum_i w[i-m] * a[i] = conv(w,a)_m
    g = 4.0 * n * cross[n-1:2*n-1]
    return g

def _adam_optimize(a_init, deadline, T_schedule, lr=0.02, K_adaptive=True,
                   backtrack_every=20, max_iter=500000):
    """
    Adam optimizer in log-space with adaptive top-K and backtracking.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score_sum1(a_init)
    a = a / s  # normalize to sum 1
    
    z = np.log(a + 1e-30)
    z = z - z.mean()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = _score_sum1(a)
    patience = 0
    best_patience = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        
        # Get current a from z
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Compute convolution
        c = _fft_convolve(a, a)
        
        # Adaptive K
        if K_adaptive:
            sorted_c = np.sort(c)[::-1]
            if len(sorted_c) >= K:
                gap = sorted_c[0] - sorted_c[1]
                if gap < 0.01 * sorted_c[0]:
                    K_cur = 5
                elif gap < 0.05 * sorted_c[0]:
                    K_cur = 3
                else:
                    K_cur = 1
            else:
                K_cur = len(sorted_c)
        else:
            K_cur = 3
        
        # Temperature
        T_cur = T_schedule(frac) * c.max()
        
        # Gradient
        if T_cur > 1e-10 * c.max():
            c_shifted = c - c.max()
            e = np.exp(c_shifted / T_cur)
            w = e / e.sum()
        else:
            w = np.zeros_like(c)
            top_k_idx = np.argsort(c)[::-1][:K_cur]
            w[top_k_idx] = 1.0 / K_cur
        
        w_rev = w[::-1]
        cross = _fft_convolve(w_rev, a)
        g = 4.0 * n * cross[n-1:2*n-1]
        
        # Project gradient for log-space (center it)
        W = np.dot(a, g)
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        lr_cur = lr * (1.0 / (1.0 + 0.1 * t))
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1**t)
        vh = v_buf / (1.0 - b2**t)
        update = lr_cur * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z_new = z - update
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        
        # Backtracking
        if t % backtrack_every == 0:
            a_cand = np.exp(z_new - z_new.max())
            a_cand = a_cand / a_cand.sum()
            sc_cand = _score_sum1(a_cand)
            if sc_cand > best_score * 1.02:
                # Revert
                z = z  # no change
            else:
                z = z_new
        else:
            z = z_new
        
        # Track best
        if t % 100 == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_sum1(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10:
                # Reset
                z = np.log(best_a + 1e-30)
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0
    
    # Final
    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_sum1(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _cd_polish(a, deadline, max_rounds=100):
    """
    Coordinate descent polish. For each coordinate, try scaling by various factors.
    Uses direct O(n) update for the convolution.
    """
    n = len(a)
    best_a = a.copy()
    s = best_a.sum()
    best_a = best_a / s
    best_score = _score_sum1(best_a)
    c = _fft_convolve(best_a, best_a)
    
    round_num = 0
    stagnation = 0
    
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved_this_round = False
        
        for j in range(n):
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            
            best_sc = best_score
            best_new_val = old_val
            
            # Try several scales
            for scale in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 0.0]:
                new_val = old_val * scale
                a_cand = best_a.copy()
                a_cand[j] = new_val
                s_cand = a_cand.sum()
                if s_cand < 1e-15:
                    continue
                a_cand = a_cand / s_cand
                
                # Compute new convolution efficiently
                # conv_new = conv_old + delta
                # The change in convolution from changing a_j:
                # old contribution: a_j_old * a (shifted)
                # new contribution: a_j_new * a (shifted)
                # But since we also renormalize, it's more complex.
                # For simplicity, just do the full FFT (it's O(n log n))
                c_cand = _fft_convolve(a_cand, a_cand)
                sc_cand = 2.0 * n * c_cand.max()
                
                if sc_cand < best_sc - 1e-12:
                    best_sc = sc_cand
                    best_new_val = new_val
                    best_a = a_cand
                    c = c_cand
                    best_score = sc_cand
                    improved_this_round = True
            
        if not improved_this_round:
            break
    
    return best_a, _score_sum1(best_a)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: adaptive top-K gradient with dynamic K selection,
simplex-projected gradient descent as alternative to log-space Adam,
efficient restricted coordinate descent, and multi-n exploration (4096/8192/16384).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_sum1(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_weights(c, T):
    c_shifted = c - c.max()
    e = np.exp(c_shifted / T)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return e / s


def _topk_weights(c, K):
    w = np.zeros_like(c)
    idx = np.argsort(c)[::-1][:K]
    w[idx] = 1.0 / K
    return w


def _compute_gradient(a, w):
    n = len(a)
    w_rev = w[::-1]
    cross = _fft_convolve(w_rev, a)
    g = 4.0 * n * cross[n - 1:2 * n - 1]
    return g


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, backtrack_every=15, eval_every=100):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score_sum1(a_init)
    a = a / s
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_sum1(a)
    patience = 0
    patience_count = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(float(max_iter), 1.0))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _fft_convolve(a, a)
        c_max = float(c.max())

        # Adaptive K
        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - sorted_c[1]
        if gap < 0.005 * c_max:
            K_cur = 5
        elif gap < 0.02 * c_max:
            K_cur = 3
        else:
            K_cur = 1

        if T > 1e-12 * c_max:
            w = _softmax_weights(c, T)
        else:
            w = _topk_weights(c, K_cur)

        g = _compute_gradient(a, w)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.8:
            update = update * (0.8 / update_norm)

        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if t % backtrack_every == 0:
            a_cand = np.exp(z_new - z_new.max())
            a_cand = a_cand / a_cand.sum()
            sc_cand = _score_sum1(a_cand)
            if sc_cand > best_score * 1.03:
                z = z
            else:
                z = z_new
        else:
            z = z_new

        if t % eval_every == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_sum1(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 12:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0
                patience_count += 1

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_sum1(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _simplex_grad_desc(a_init, deadline, T_start, T_end, lr_start, lr_end,
                       max_iter=300000, eval_every=50):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_sum1(a)
    step = lr_start
    patience = 0

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max(float(max_iter), 1.0))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        c = _fft_convolve(a, a)
        c_max = float(c.max())

        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - sorted_c[1]
        if gap < 0.005 * c_max:
            K_cur = 5
        elif gap < 0.02 * c_max:
            K_cur = 3
        else:
            K_cur = 1

        if T > 1e-12 * c_max:
            w = _softmax_weights(c, T)
        else:
            w = _topk_weights(c, K_cur)

        g = _compute_gradient(a, w)
        # Project gradient for simplex (remove mean component)
        g = g - g.mean()

        a_candidate_a = a
        sc_candidate = best_score
        found_improve = False
        for shrunk in [1.0, 0.5, 0.25, 0.125]:
            a_try = a - lr * shrunk * g
            a_try = _project_simplex(a_try)
            if np.any(a_try > 0):
                sc_try = _score_sum1(a_try)
                if sc_try < sc_candidate:
                    a_candidate_a = a_try
                    sc_candidate = sc_try
                    found_improve = True
                    break

        if found_improve:
            a = a_candidate_a
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a.copy()
                patience = 0
        else:
            step *= 0.7
            patience += 1
            if patience > 20:
                a = best_a.copy()
                patience = 0

        if t % eval_every == 0:
            sc_cur = _score_sum1(a)
            if sc_cur < best_score:
                best_score = sc_cur
                best_a = a.copy()

    return best_a, best_score

        improved = False
        # Focus on top-m coordinates by value

        for j in top_idx:
            old_val = best_a[j]

                new_val = old_val * sc

                    best_new = new_val

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

                best_score = sc_p
                best_a = a_p_opt.copy()

    # === Phase 8: Final CD polish (1720 - deadline) ===
    if time.monotonic() < deadline - 10:
        a_final_cd, sc_fcd = _cd_active_polish(best_a, deadline - 5, max_rounds=30)
        if sc_fcd < best_score:
            best_score = sc_fcd
            best_a = a_final_cd.copy()
        # Also try top-m CD
        if time.monotonic() < deadline - 5:
            a_final_cd2, sc_fcd2 = _cd_polish(best_a, deadline - 3, max_rounds=20, top_m=60)
            if sc_fcd2 < best_score:
                best_score = sc_fcd2
                best_a = a_final_cd2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_check = _score(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
