"""

import numpy as np

from scipy.sparse import lil_matrix, csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lp_direction(a, M, n_active_max=200):
    """Solve reduced LP: max sum(g) s.t. conv(a,g)_k <= M for active k, g >= 0."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 0.01 * c_max
    active = c >= (c_max - tol)
    active_idx = np.where(active)[0]
    if len(active_idx) > n_active_max:
        top = np.argsort(c)[::-1][:n_active_max]
        active_idx = top
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c))])

    # Build constraint matrix for active constraints only
    rows = []
    cols = []
    vals = []
    for k in active_idx:
        k_int = int(k)
        for j in range(n):
            idx = k_int - j
            if 0 <= idx < n:
                rows.append(int(k_int))
                cols.append(j)
                vals.append(a[idx])

    n_constraints = len(active_idx)
    A_mat = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))

    obj = -np.ones(n)
    b_ub = np.full(n_constraints, M)

    try:
        res = linprog(obj, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

    n = len(a_init)

        if g is None:
            break
        g = np.maximum(g, 0)
        g_s = g.sum()
        if g_s < 1e-15:
            break
        g = g / g_s

        # Line search for blending factor
        best_t = 0.3
        best_try_sc = _score(a)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)

            if s_try < 1e-15:

            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_try_sc:

        if grad_norm < 1e-14:

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)

                best_a = a_check.copy()

                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))

                    patience = 0

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(3.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.exp(-20.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-20.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _refined_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Enhanced L-BFGS with periodic restarts and perturbation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.015)

    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1,
                                  0.07, 0.05, 0.03, 0.02, 0.01, 0.007,
                                  0.004, 0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    rng = np.random.default_rng(12345)

    no_improve = 0
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 6))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 200,
                    'iprint': -1,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 3:
                    # Perturb and restart
                    pert = _perturb(current_a, 0.05, rng)
                    z = np.log(np.maximum(pert, 1e-30))
                    z = z - z.mean()
                    no_improve = 0
        except Exception:
            no_improve += 1

    return current_a


def _multi_start_optimize(n, deadline, min_time_remaining, rng):
    """Try multiple initializations and return the best."""
    best_a = None
    best_sc = np.inf
    profiles = ["uniform", "tapered", "cos", "asymmetric", "bump", "ramp"]
    time_per_start = max(10.0, (deadline - min_time_remaining - time.monotonic()) / max(1, len(profiles)))

    for kind in profiles:
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_init = _make_profile(n, kind)
        sc_init = _score(a_init)
        sub_deadline = min(deadline - min_time_remaining, time.monotonic() + time_per_start * 0.7)
        a_opt = _refined_lbfgs(a_init, sub_deadline, min_time_remaining=5.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc
