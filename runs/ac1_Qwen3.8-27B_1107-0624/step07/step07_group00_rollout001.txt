import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_adaptive(a, c, top_k=8, mode="adaptive"):
    """
    Compute gradient of max(conv(a,a)) using adaptive weighting of top-k entries.
    mode: 'adaptive' (gap-based weights), 'uniform' (equal top-k), 'single' (only max)
    """
    n = len(a)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    max_val = c[sorted_idx[0]]
    
    if mode == "single":
        k = sorted_idx[0]
        g = np.zeros(n)
        js = np.arange(max(0, k - n + 1), min(k + 1, n))
        g[js] = 2.0 * a[k - js]
        return g
    
    elif mode == "uniform":
        # Gather top_k indices close to max
        close = [idx for idx in sorted_idx if max_val - c[idx] < max_val * 0.005]
        if not close:
            close = [sorted_idx[0]]
        g = np.zeros(n)
        for k in close:
            js = np.arange(max(0, k - n + 1), min(k + 1, n))
            g[js] += 2.0 * a[k - js]
        g /= len(close)
        return g
    
    else:  # adaptive
        # Weight based on proximity to max (exponential decay)
        vals = c[sorted_idx]
        gaps = max_val - vals
        # Adaptive scale: median gap among top entries
        nonzero_gaps = gaps[gaps > 0]
        if len(nonzero_gaps) > 0:
            scale = max(nonzero_gaps[len(nonzero_gaps) // 2], max_val * 0.0001)
        else:
            scale = max_val * 0.001
        weights = np.exp(-gaps / scale)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(top_k)
            weights[0] = 1.0
            w_sum = 1.0
        else:
            weights /= w_sum
        
        g = np.zeros(n)
        for i, k in enumerate(sorted_idx):
            w = weights[i]
            if w < 1e-12:
                continue
            js = np.arange(max(0, k - n + 1), min(k + 1, n))
            g[js] += w * 2.0 * a[k - js]
        return g


def _variance_gradient(a):
    """Gradient of sum_k (c_k - mean(c))^2 where c = conv(a,a)."""
    n = len(a)
    c = _fft_convolve(a, a)
    mean_c = c.mean()
    c_dev = c - mean_c
    # d/d(a_j) sum_k (c_k - mean)^2 ~ 4 * (c_dev * a)_valid (simplified)
    conv_prod = _fft_convolve(c_dev, a)
    g = 4.0 * conv_prod[n - 1:2 * n - 1][::-1]
    return g


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_adaptive(a_init, deadline, T_mode="auto", top_k=8,
                   lr_start=0.04, lr_end=0.002, max_iter=400000,
                   eval_interval=100, clip_norm=0.8, use_backtrack=True,
                   grad_mode="adaptive", variance_phase_frac=0.3):
    """
    Adam optimizer in log-space with adaptive gradient and optional variance phase.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0
    variance_phase = True
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        
        # Determine if in variance phase
        if T_mode == "auto":
            in_var_phase = (frac < variance_phase_frac)
        else:
            in_var_phase = False
        
        # Temperature for adaptive gradient
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        if in_var_phase:
            g = _variance_gradient(a)
            # Normalize gradient
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                g = g / g_norm * c_max
        else:
            g = _compute_gradient_adaptive(a, c, top_k, grad_mode)
            # Normalize gradient scale
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                g = g / g_norm * c_max
        
        # Learning rate schedule
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        
        # Adam update in z-space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        
        # Backtracking
        if use_backtrack and t % 30 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.02:
                z_new = z
        z = z_new
        
        # Evaluate
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.03:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _subgradient_refine(a_init, deadline, max_iter=50000, step_init=0.01,
                        top_k=5, project=True):
    """
    Projected subgradient method with adaptive step size using exact subgradient.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    step = step_init
    patience = 0
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Compute exact subgradient (average over top-k closest to max)
        g = _compute_gradient_adaptive(a, c, top_k, "uniform")
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Take step
        a_new = a - step * g / c_max
        
        if project:
            a_new = _project_simplex(a_new)
        else:
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Increase step slightly
            step = min(step * 1.05, 0.1)
        else:
            patience += 1
            a = a_new  # Still move, but don't update best
            if patience > 10:
                step *= 0.7
                patience = 0
                if step < 1e-6:
                    break
    
    return best_a, best_score


def _coord_descent_line_search(a_init, deadline, max_rounds=200, top_k_active=10):
    """
    Coordinate descent with line search on most impactful coordinates.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        
        # Find most impactful convolution entries
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        tol = c_max * 0.999
        active_k = np.where(c > tol)[0]
        
        if len(active_k) == 0:
            break
        
        improved = False
        for k in active_k[:top_k_active]:
            # Coordinates that affect conv[a][k]
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            
            if not js:
                continue
            
            # Try adjusting each coordinate
            for j in js:
                if a[j] < 1e-10:
                    continue
                # Line search: try different scales
                best_local = best_score
                best_scale = 1.0
                for scale in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5, 0.3]:
                    a_tmp = a.copy()
                    a_tmp[j] *= scale
                    s = a_tmp.sum()
                    if s < 1e-15:
                        continue
                    a_tmp /= s
                    c_tmp = _fft_convolve(a_tmp, a_tmp)
                    sc = float(c_tmp.max()) * 2.0 * n
                    if sc < best_local:
                        best_local = sc
                        best_scale = scale
                
                if best_scale != 1.0 and best_local < best_score * (1 - 1e-10):
                    a[j] *= best_scale
                    a /= a.sum()
                    best_score = best_local
                    best_a = a.copy()
                    improved = True
                    c = _fft_convolve(a, a)
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _symmetrize(a):
    """Make sequence symmetric: a[j] = a[n-1-j]."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym / a_sym.sum()


def _fista_smooth(a_init, deadline, max_iter=50000, top_k=5, T_frac=0.003):
    """
    FISTA on smoothed max objective.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    t_k = 1.0
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        T = T_frac * c_max
        
        # Smoothed gradient
        g = _compute_gradient_adaptive(y, c, top_k, "adaptive")
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Adaptive step
        step = 0.5 / g_norm
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        
        if t % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
