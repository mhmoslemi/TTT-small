"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: aggressive cyclic L-BFGS/Newton alternation,
extended beta schedule to 40 levels, multiplicative weight diversification,
n=16384 exploration, adaptive active set sweeping, and support shape search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_polish(a_init, deadline, max_rounds=80, tight_start=False):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Adaptive tolerance sweep
    if tight_start:
        tol_schedule = [0.0001, 0.0003, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    else:
        tol_schedule = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.10]

    tol_idx = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            idxs = np.argsort(c)[::-1]
            active[idxs[0:3]] = True
            n_active = 3
        # Cap active set size for efficiency
        if n_active > 500:
            idxs = np.argsort(c)[::-1]
            active = np.zeros_like(active)
            active[idxs[:500]] = True
            n_active = 500

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)
            continue

        Hg = _cg_solve(W, n, g, max_iter=200)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=200)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.05 * alpha * gn * 0.01:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if not step_done:

            for ls in range(30):

                a_try = np.maximum(a_try, 0)

                if s_try < 1e-15:

                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        else:
            tol_idx = min(tol_idx + 1, len(tol_schedule) - 1)

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.003)

    # Extended beta schedule: 40 levels, going much lower
    ratios = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
        0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
        0.000025, 0.00001, 0.000005, 0.0000025, 0.000001,
        0.0000005, 0.00000025, 0.0000001, 0.00000005, 0.00000002,
        0.00000001, 0.000000005, 0.000000002, 0.000000001, 0.0000000005,
        0.0000000002, 0.0000000001, 5e-11, 2.5e-11, 1e-11,
        5e-12, 2e-12, 1e-12, 5e-13, 1e-13
    ])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 1:

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),

            a_new = _softmax(result.x)

            if sc_new < current_sc - 1e-16:

                z = np.log(np.maximum(a_new, 1e-30))

    n_cycles = 8

            break

        if remaining < 5:
            break

        # L-BFGS phase with 3 beta levels
        beta_start = base_beta * (0.1 ** cyc)

        z = z - z.mean()

                    z,

def _multiplicative_weights(a_init, deadline, min_time_remaining=30.0, n_steps=200):
    """Multiplicative weight updates to explore the landscape."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.5

    for i in range(n_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with moderate tolerance
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient of weighted sum
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * a * (h - W)

        # Multiplicative update
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        grad_normed = grad / (norm_grad + 1e-30)
        a_pos = np.maximum(a, 1e-15)
        a_new = a_pos * np.exp(-eta * grad_normed)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Adaptive learning rate
        if i % 20 == 19:
            if sc < best_sc + 1e-10:
                eta *= 0.7
            else:
                eta = min(eta * 1.2, 2.0)
        a = a_new

    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.25, 0.40]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.25) ** 2) + np.exp(-10.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "wide_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "trapezoid":
        a = np.ones(n)
        taper = n // 8
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n - 1 - i] *= (i + 1) / taper
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _flat_spectrum_init(n, rng):
    """Initialize using a flat-spectrum construction in frequency domain."""
    N = _next_pow2(2 * n - 1)
    # Random phase flat spectrum
    freq = np.zeros(N // 2 + 1, dtype=np.complex128)
    freq[0] = np.sqrt(n)  # DC component
    for k in range(1, N // 2 + 1):
        phase = rng.uniform(0, 2 * np.pi)
        freq[k] = np.exp(1j * phase)
    # Ensure conjugate symmetry for real signal
    signal = np.fft.irfft(freq, N)
    # Take first n elements
    a = signal[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return _make_profile(n, "tapered")
    return a / s


def _structured_init(n, rng):
    """Try various parametric forms."""
    kinds = ["tapered", "cos", "bimodal", "wide_taper", "gaussian", "trapezoid"]
    kind = kinds[rng.integers(len(kinds))]
    base = _make_profile(n, kind)
    # Add small random perturbation
    return _perturb(base, 0.05, rng)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS from parent with extended beta schedule
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Cyclic L-BFGS/Newton optimization
    print(f"\n[Phase 2] Cyclic optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        a2 = _cyclic_optimize(best_seq, deadline, min_time_remaining=500.0)
        update_best(a2, _score(a2), "cyclic_1")

    # Phase 3: Tight Newton polish
    print(f"\n[Phase 3] Tight Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 450:
        a3, sc3 = _newton_active_polish(best_seq, min(deadline - 450, time.monotonic() + 200),
                                         max_rounds=100, tight_start=True)
        update_best(a3, sc3, "newton_tight")

    # Phase 4: Shrinkage + re-optimize
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a4, sc4 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=200.0)
        update_best(a4, sc4, "shrinkage")

    # Phase 5: More cyclic optimization after shrinkage
    print(f"\n[Phase 5] Cyclic 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5 = _cyclic_optimize(best_seq, deadline, min_time_remaining=180.0)
        update_best(a5, _score(a5), "cyclic_2")

    # Phase 6: n=16384 exploration
    print(f"\n[Phase 6] n=16384 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a16 = _resample(parent, 16384)
        a16_opt = _run_lbfgs(a16, deadline, min_time_remaining=120.0)
        update_best(a16_opt, _score(a16_opt), "n16384")
        if _score(a16_opt) < best_score + 0.005 and time.monotonic() < deadline - 80:
            a16c = _cyclic_optimize(a16_opt, deadline, min_time_remaining=60.0)
            update_best(a16c, _score(a16c), "n16384_cyclic")

    # Phase 7: n=12288 exploration
    print(f"\n[Phase 7] n=12288 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a12 = _resample(parent, 12288)
        a12_opt = _run_lbfgs(a12, deadline, min_time_remaining=100.0)
        update_best(a12_opt, _score(a12_opt), "n12288")

    # Phase 8: Diversified restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            (0.03, "tapered"),
            (0.1, "bimodal"),
            (0.05, "cos"),
            (0.15, "wide_taper"),
        ]
        for i, (sigma, kind) in enumerate(restart_configs):
            if time.monotonic() > deadline - 40:
                break
            if i < 2:
                p_start = _perturb(best_seq, sigma, rng)
            else:
                p_start = _structured_init(len(best_seq), rng)
            if time.monotonic() > deadline - 30:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")
            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 20:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 15, time.monotonic() + 30),
                                                     max_rounds=40, tight_start=True)
                update_best(a_rp, sc_rp, f"restart_{i}_np")

    # Phase 9: Final ultra-fine polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 3, max_rounds=150, tight_start=True)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS with extremely small beta
        if time.monotonic() < deadline - 4:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-9
            remaining = deadline - 4 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 80, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs_ultra")
                except Exception:
                    pass

        # One more Newton pass
        if time.monotonic() < deadline - 2:
            a_fn2, sc_fn2 = _newton_active_polish(best_seq, deadline - 1, max_rounds=50, tight_start=True)
            update_best(a_fn2, sc_fn2, "final_newton_2")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
