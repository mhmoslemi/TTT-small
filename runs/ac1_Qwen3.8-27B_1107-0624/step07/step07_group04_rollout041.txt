"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) FISTA with backtracking in the simplex (no log-space),
(2) Active-set CG refinement with matrix-free Hessian,
(3) Diverse structured initializations including sparse spikes and spectral profiles,
(4) Aggressive multi-n exploration, (5) Adaptive beta continuation.
Key difference from previous: direct simplex optimization with Nesterov acceleration,
proper active-set QP via CG in the null space, and Chebyshev-based spectral init.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _cross_corr_fft(w, a):
    """Compute sum_k w[k]*a[k-j] for j=0,...,len(a)-1 using FFT."""
    n_w = len(w)
    n_a = len(a)
    N = _next_pow2(n_w + n_a - 1)
    Fw = np.fft.fft(w, N)
    Fa = np.fft.fft(a, N)
    result = np.fft.ifft(Fw * np.conj(Fa)).real
    return result[:n_a]


def _smooth_grad(a, beta, n):
    """Compute gradient of 2n*(smooth_max(c, beta)) w.r.t. a."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    cc = _cross_corr_fft(w, a)
    return 4.0 * n * cc


def _smooth_obj(a, beta, n):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))


def _fista_optimize(a_init, beta, deadline, max_iter=200000, verbose=False):
    """FISTA on smoothed objective in the simplex with backtracking."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    y = a.copy()
    t = 1.0
    alpha = 0.5
    best_a = a.copy()
    best_sc = _score(a)
    best_sm_obj = _smooth_obj(a, beta, n)
    no_improve = 0
    max_no_improve = 5000

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_y = _full_convolve(y, y)
        c_y_max = float(c_y.max())
        shifted = (c_y - c_y_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / s_e

        grad = 4.0 * n * _cross_corr_fft(w, y)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break

        # Backtracking line search
        obj_y = _smooth_obj(y, beta, n)
        step_done = False
        for ls in range(30):
            y_new = _project_simplex(y - alpha * grad)
            obj_new = _smooth_obj(y_new, beta, n)
            armijo = obj_new <= obj_y + 0.01 * alpha * g_norm * g_norm + 1e-15
            if armijo:
                step_done = True
                break
            alpha *= 0.5

        if not step_done:
            y_new = _project_simplex(y - alpha * grad)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = y_new + momentum * (y_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = y_new
        a = y_new
        t = t_new

        # Periodic score check
        if i % 200 == 0 or i < 10:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
                if verbose:
                    print(f"    FISTA iter {i}: score={sc:.10f}")
            else:
                no_improve += 200

        if no_improve > max_no_improve:
            break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def _hankel_matvec(v, W, n):
    """Compute H*v where H is the active-set Hessian matrix."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_active(W, n, rhs, max_iter=200, tol=1e-12):
    """Solve H*x = rhs using CG where H is the active-set Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _active_set_qp(a_init, deadline, max_rounds=100, verbose=False):
    """Active-set QP: minimize average of active c_k's subject to a>=0, sum=1."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Multiple active set tolerances for robustness
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active == 1:
            # Only one active - use gradient step instead
            idx = int(np.argmax(c))
            W = np.zeros(2 * n - 1)
            W[idx] = 1.0
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            g_norm = np.linalg.norm(g_t)
            if g_norm < 1e-14:
                break
            # Gradient step with backtracking
            alpha = 0.1
            for ls in range(40):
                a_try = _project_simplex(a - alpha * g_t)
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

            W[active] = 1.0 / n_active

            # Compute gradient of average active c_k
            g = _hankel_matvec(a, W, n)
            # Project gradient onto null space of ones (for sum constraint)
            g_t = g - np.mean(g)
            g_norm = float(np.linalg.norm(g_t))
            if g_norm < 1e-14 * n:
                break

            # Newton step via CG
            Hg = _cg_solve_active(W, n, g, max_iter=150)
            H1 = _cg_solve_active(W, n, np.ones(n), max_iter=150)
            H1_norm = float(np.linalg.norm(H1))

            if H1_norm > 1e-20 and np.isfinite(H1_norm):
                lam = float(np.sum(Hg)) / float(np.sum(H1))
                dx = -Hg + lam * H1
                dx = dx - np.mean(dx)
                dx_norm = float(np.linalg.norm(dx))
                if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                    dx = -g_t / (g_norm + 1e-30)
            else:
                dx = -g_t / (g_norm + 1e-30)

            # Damped step with backtracking
            alpha = 1.0
            step_done = False
            for ls in range(50):
                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max - 0.05 * alpha * g_norm * 0.01:
                    a = a_try
                    step_done = True
                    break
                alpha *= 0.5
                if alpha < 1e-12:
                    break

            if not step_done:
                # Fallback: steepest descent
                alpha = 0.01
                for ls in range(30):
                    a_try = _project_simplex(a - alpha * g_t)
                    c_try = _full_convolve(a_try, a_try)
                    if float(c_try.max()) < c_max:
                        a = a_try
                        break
                    alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            if verbose and rnd % 10 == 0:
                print(f"    QP round {rnd}: score={sc:.10f}")

def _make_tapered(n, power=2.0):
    """Create a tapered profile: a_i ∝ (1 - (2i/n - 1)^2)^power."""
    t = np.linspace(-1, 1, n)
    a = np.maximum(1.0 - t ** 2, 0.0) ** power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_gaussian(n, width=0.3):
    """Create a Gaussian profile centered at n/2."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / width) ** 2)
    return a / a.sum()


def _make_bimodal(n, sep=0.25, width=0.12):
    """Create a bimodal profile."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5 + sep) / width) ** 2) + \
        np.exp(-0.5 * ((t - 0.5 - sep) / width) ** 2)
    a += 0.01
    return a / a.sum()


def _make_sparse_spikes(n, n_spikes=16, rng=None):
    """Create a sparse spike profile."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.zeros(n)
    positions = rng.choice(n, size=n_spikes, replace=False)
    for p in positions:
        # Add a small Gaussian bump at each position
        width = max(1, n // (n_spikes * 4))
        idx = np.arange(max(0, p - width), min(n, p + width + 1))
        a[idx] += np.exp(-0.5 * ((idx - p) / (width * 0.5)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _make_chebyshev_profile(n, order=6):
    """Create profile based on Chebyshev polynomial window."""
    t = np.linspace(-1, 1, n)
    # Use |T_order(t) * (1-t^2)| as window shape
    if order == 0:
        a = 1.0 - t ** 2
    elif order == 1:
        a = np.abs(t) * (1.0 - t ** 2)
    elif order == 2:
        a = np.abs(2 * t ** 2 - 1) * (1.0 - t ** 2)
    elif order == 3:
        a = np.abs(4 * t ** 3 - 3 * t) * (1.0 - t ** 2)
    elif order == 4:
        a = np.abs(8 * t ** 4 - 8 * t ** 2 + 1) * (1.0 - t ** 2)
    else:
        # General: product of first few Chebyshev values
        a = (1.0 - t ** 2)
        for k in range(1, order + 1):
            tk = np.polynomial.chebyshev.chebval(t, np.zeros(k + 1) + (1 if k == 0 else 0))
            # Simpler: just use (1-t^2)^k
            a = a * np.abs(t ** k - (k * t ** (k - 2)))
    a = np.abs(a) + 0.001
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Adaptive subgradient method with dynamic active set and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)  # momentum
    lr = 0.1
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Dynamic active set with hysteresis
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()

        # Subgradient
        grad = 2.0 * n * _cross_corr_fft(w, a)
        grad = grad - np.mean(grad)  # project to null space of sum constraint
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break

        # Nesterov momentum
        v = 0.9 * v - lr * grad
        a_new = a + v
        a_new = _project_simplex(a_new)

        # Check improvement
        if i % 50 == 0:
            sc = _score(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
                patience = 0
                lr = min(lr * 1.05, 0.5)
            else:
                patience += 1
                if patience > 8:
                    lr *= 0.7
                    v = np.zeros(n)
                    patience = 0
                    if lr < 0.001:
                        break
        a = a_new

    return best_a, best_sc


def _beta_continuation(a_init, deadline, min_time_remaining=30.0):
    """Run FISTA with decreasing beta (continuation)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    c_max = float(c_cur.max())
    if c_max < 1e-15:
        return a_init

    # Determine initial beta from the gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, c_max * 0.01)

    # Continuation schedule
    ratios = [1.0, 0.4, 0.16, 0.064, 0.0256, 0.01, 0.004, 0.0016,
              0.00064, 0.000256, 0.0001]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations by remaining time
        max_iter = min(50000, max(1000, int(remaining * 300)))
        a_new = _fista_optimize(current_a, float(beta), time.monotonic() + remaining * 0.8,
                                max_iter=max_iter)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            current_sc = sc_new
            current_a = a_new

    return current_a


def _shrink_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out small elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.05, 0.1, 0.2, 0.3]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _beta_continuation(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)

def _multi_resolution(a_final_shape, deadline, min_time_remaining=100.0):
    """Coarse-to-fine optimization: start small, grow."""
    n_target = len(a_final_shape)
    sizes = []
    s = max(64, n_target // 64)
    while s < n_target:
        sizes.append(s)
        s *= 2
    sizes.append(n_target)
    # Remove duplicates and ensure sorted
    sizes = sorted(set(sizes))

    current_a = np.ones(64) / 64  # Start uniform at smallest size
    for sz in sizes:
        if time.monotonic() + min_time_remaining > deadline:
            break
        if sz == 64:
            continue
        # Upsample
        current_a = _resample(current_a, sz)
        # Optimize at this size
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(20000, max(500, int(remaining * 50)))
        beta = _full_convolve(current_a, current_a).max() * 0.05
        current_a = _fista_optimize(current_a, beta, time.monotonic() + remaining * 0.7,
                                    max_iter=max_iter)
        # Active set polish at this size
        if time.monotonic() + min_time_remaining < deadline:
            a_p, sc_p = _active_set_qp(current_a,
                                       min(deadline - min_time_remaining, time.monotonic() + remaining * 0.3))
            if sc_p < _score(current_a):
                current_a = a_p

    return _resample(current_a, n_target)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"delta={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 4096
        parent = _make_tapered(n_parent, power=1.5)
        update_best(parent, _score(parent), "init_tapered")
        print(f"No parent found, using tapered n={n_parent}")

    # === Phase 1: FISTA from parent with beta continuation ===
    print(f"\n[Phase 1] FISTA + beta continuation from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _beta_continuation(parent, deadline, min_time_remaining=600)
        update_best(a1, _score(a1), "fista_parent")

    # === Phase 2: Active set QP polish ===
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _active_set_qp(best_seq, min(deadline - 500, time.monotonic() + 200),
                                  max_rounds=150, verbose=True)
        update_best(a2, sc2, "qp_1")

    # === Phase 3: Diverse initializations ===
    print(f"\n[Phase 3] Diverse inits (t={time.monotonic() - t_start:.1f}s)")
    diverse_inits = []
    n_try = len(best_seq) if best_seq is not None else 4096

    if time.monotonic() < deadline - 450:
        # Gaussian
        a_g = _make_gaussian(n_try, width=0.25)
        diverse_inits.append(("gauss_w025", a_g))

    if time.monotonic() < deadline - 400:
        # Bimodal
        a_b = _make_bimodal(n_try, sep=0.2, width=0.15)
        diverse_inits.append(("bimodal", a_b))

    if time.monotonic() < deadline - 350:
        # Sparse spikes
        a_s = _make_sparse_spikes(n_try, n_spikes=32, rng=rng)
        diverse_inits.append(("spikes32", a_s))

    if time.monotonic() < deadline - 300:
        # Different taper power
        a_t = _make_tapered(n_try, power=0.8)
        diverse_inits.append(("taper_p08", a_t))

    for label, a_init in diverse_inits:
        if time.monotonic() > deadline - 250:
            break
        a_opt = _beta_continuation(a_init, deadline, min_time_remaining=200)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"div_{label}")
        if sc_opt < best_score + 0.001 and time.monotonic() < deadline - 150:
            a_q, sc_q = _active_set_qp(a_opt,
                                       min(deadline - 150, time.monotonic() + 60), max_rounds=50)
            update_best(a_q, sc_q, f"div_{label}_qp")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192, 16384]
    for n_c in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_c == n_parent and n_c < 5000:
            continue  # Already explored
        a_init_n = _resample(parent, n_c)
        remaining = deadline - 150 - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(30000, max(2000, int(remaining * 40)))
        beta_init = _full_convolve(a_init_n, a_init_n).max() * 0.05
        a_opt_n = _fista_optimize(a_init_n, beta_init,
                                   time.monotonic() + remaining * 0.6, max_iter=max_iter)
        sc_n = _score(a_opt_n)
        update_best(a_opt_n, sc_n, f"n_{n_c}")

        if sc_n < best_score + 0.002 and time.monotonic() < deadline - 100:
            a_qn, sc_qn = _active_set_qp(a_opt_n,
                                         min(deadline - 100, time.monotonic() + 50), max_rounds=40)
            update_best(a_qn, sc_qn, f"n_{n_c}_qp")

    # === Phase 5: Subgradient polish ===
    print(f"\n[Phase 5] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a5, sc5 = _subgrad_polish(best_seq, min(deadline - 100, time.monotonic() + 80),
                                   max_steps=5000)
        update_best(a5, sc5, "subgrad")

    # === Phase 6: Shrinkage ===
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a6, sc6 = _shrink_and_reopt(best_seq, deadline, min_time_remaining=50)
        update_best(a6, sc6, "shrink")

    # === Phase 7: Restart from best with perturbation ===
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        for i in range(3):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                z = np.log(np.maximum(best_seq, 1e-30))
                z += 0.05 * rng.standard_normal(len(z))
                z -= z.mean()
                p_start = np.exp(z - z.max())
                p_start /= p_start.sum()
            elif i == 1:
                p_start = _make_tapered(len(best_seq), power=rng.uniform(0.5, 3.0))
            else:
                p_start = _make_gaussian(len(best_seq), width=rng.uniform(0.15, 0.4))

            a_r = _beta_continuation(p_start, deadline, min_time_remaining=20)
            update_best(a_r, _score(a_r), f"restart_{i}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active set QP
        a_fin, sc_fin = _active_set_qp(best_seq, deadline - 2, max_rounds=200)
        update_best(a_fin, sc_fin, "final_qp")

        # One more FISTA with tiny beta
        if time.monotonic() < deadline - 3:
            n_f = len(best_seq)
            c_f = _full_convolve(best_seq, best_seq)
            beta_f = float(c_f.max()) * 1e-6
            a_fb = _fista_optimize(best_seq, beta_f, deadline - 1, max_iter=50000)
            update_best(a_fb, _score(a_fb), "final_fista")

        # Subgradient final
        if time.monotonic() < deadline - 2:
            a_sf, sc_sf = _subgrad_polish(best_seq, deadline - 1, max_steps=3000)
            update_best(a_sf, sc_sf, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(256, 1.5)
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            ev_sc = evaluate_sequence([float(x) for x in final])
            print(f"Final _score={sc_final:.10f}, evaluate_sequence={ev_sc:.10f}, "
                  f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
            if np.isfinite(ev_sc) and ev_sc < sc_final:
                sc_final = ev_sc
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_tapered(4096, 1.5)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
