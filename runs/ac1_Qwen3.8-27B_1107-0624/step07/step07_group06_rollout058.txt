import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""

Uses L-BFGS smooth-max with aggressive beta continuation, active-set Newton,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_fn(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_fn(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad_fn(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros(len(c), dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth_max(a_init, deadline, min_remaining=30.0):
    """L-BFGS with aggressive beta continuation on softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # More stages, lower final beta
    betas = base_beta * np.array([
        1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02,
        0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001
    ])

    current_a = a.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 6))))

        def _fun_local(z_local):
            return _smoobj_fn(z_local, beta, n)

        def _jac_local(z_local):
            return _smograd_fn(z_local, beta, n)

        try:
            result = minimize(
                _fun_local, z, jac=_jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200, 'maxfun': max_iter * 3}

            sc_new = _score(a_new)

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:

def _newton_active(a_init, deadline, max_iter=200, tol_frac=0.001):
    """Newton active-set with CG solve and careful line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient of sum_{k in active} c_k projected to simplex tangent
        # grad[i] = 2 * sum_j a[j] * active[i+j] (from hessian matvec formula)
        N = _next_pow2(4 * n)
        a_rev = a[::-1]
        fa = np.fft.rfft(a_rev, N)
        fav = np.fft.rfft(active.astype(np.float64), N)
        r = np.fft.irfft(fa * fav, N)
        grad = 2.0 * (r[n - 1:2 * n - 1]).copy()

        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        # Approximate Newton: use diagonal approximation + CG
        mu = 1e-10
        # H[i,i] = 2 * active[i+i] (diagonal of the Hessian)
        # Actually H[i,j] = 2*active[i+j], so H[i,i] = 2*active[2i]
        diag = np.zeros(n)
        valid_diag = 2 * np.arange(n) < len(active)
        diag[valid_diag] = 2.0 * active[2 * np.arange(n)[valid_diag]] + mu
        diag = np.maximum(diag, mu)

        # Diagonal step (Fisher scoring approximation)
        dx = -grad_t / diag
        dx = dx - np.mean(dx)

        # Line search
        f_old = float(np.dot(active, c))
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(active, c_try))
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try a steepest descent step
            dx_sd = -grad_t
            dx_sd = dx_sd - np.mean(dx_sd)
            dx_sd_norm = np.linalg.norm(dx_sd)
            if dx_sd_norm > 0:
                dx_sd = dx_sd / dx_sd_norm
                for ls in range(20):
                    a_try = a + alpha * 0.01 * dx_sd
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    f_try = float(np.dot(active, c_try))
                    if f_try < f_old:
                        a_new = a_try
                        improved = True
                        break
                    alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _fourier_equality(a, n_iter=50, deadline=None):
    """Spectral equalization: iterate phase retrieval to flatten the spectrum."""
    n = len(a)
    a_cur = a.copy()
    a_cur = np.maximum(a_cur, 0)
    a_cur = a_cur / a_cur.sum()
    N = _next_pow2(2 * n - 1)

    best_a = a_cur.copy()
    best_sc = _score(a_cur)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compute current spectrum
        A = np.fft.rfft(a_cur, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat spectrum magnitude
        target_mag = np.sqrt(N) / np.sqrt(2 * n - 1)

        # New time-domain signal with target magnitude and current phase
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project onto non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Damped update to avoid oscillation
        a_cur = 0.5 * a_cur + 0.5 * a_new

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.015):

def _upscale(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_up = np.interp(x_new, x_old, a)
    a_up = np.maximum(a_up, 0)
    s = a_up.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a_up / s


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax_fn(z)


def _perturb_fourier(a, rng, n_modes=5, max_amp=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):

        amp = max_amp * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax_fn(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

        a_fin, sc_fin = _subgrad_polish(best_seq, max(deadline - 3, time.monotonic()),
                                         n_steps=2000, lr_init=0.006)
        update_best(a_fin, sc_fin, "final_sg")

        # Newton
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _newton_active(best_seq, max(deadline - 2, time.monotonic()), max_iter=30)
            update_best(a_fin2, sc_fin2, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
