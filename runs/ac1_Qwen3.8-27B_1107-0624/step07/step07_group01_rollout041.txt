```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain active-set KKT solver (novel)
2. Adaptive L-BFGS with log-sum-exp smoothing
3. Subgradient polishing with adaptive active set
4. Multi-length exploration and interpolation
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    """Smoothed objective using log-sum-exp."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    """Gradient of smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_smooth(a_init, beta, n, max_iter):
    """L-BFGS on smoothed objective in log-space."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        return _smograd(z, beta, n)
```

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive L-BFGS with geometric log-sum-exp smoothing (very tight betas)
2. Novel KKT active-set equalization solver (iterative multiplicative updates)
3. Frequency-domain flattening (Gerchberg-Saxton inspired)
4. Subgradient polishing with adaptive learning rate
5. Interleaved multi-method refinement
6. Multi-length exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxcor': 100,
                                   'maxls': 120, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _kkt_active_solve(a, k, n_iter=300, deadline=None, eta_init=0.5):
    """
    Solve KKT conditions for active set of size k.
    Iteratively equalizes the top-k convolution values using
    multiplicative weight updates.
    """
    n = len(a)
    a_cur = a.copy()
    eta = eta_init

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a_cur, a_cur)
        idx = np.argsort(c)[::-1][:k]

        # Compute b_j = sum_{l in idx} a_{l-j} (zero-padded)
        b = np.zeros(n)
        for l in idx:
            i_start = max(0, l - n + 1)
            i_end = min(n - 1, l)
            if i_end >= i_start:
                ii = np.arange(i_start, i_end + 1)
                b[ii] += a_cur[l - ii]

        C = np.mean(b)
        spread = np.max(b) - np.min(b)
        if spread < 1e-15:
            break

        # Responsibility: d_i = sum_{l in idx} (b_{l-i} - C) for valid l-i
        d = np.zeros(n)
        for l in idx:
            i_start = max(0, l - n + 1)
            i_end = min(n - 1, l)
            if i_end >= i_start:
                ii = np.arange(i_start, i_end + 1)
                d[ii] += b[l - ii] - C

        max_d = np.max(np.abs(d))
        if max_d < 1e-16:
            break

        # Multiplicative update
        a_cur = a_cur * np.exp(-eta * d / max_d)
        a_cur = np.maximum(a_cur, 1e-18)
        a_cur = a_cur / a_cur.sum()

        # Adapt eta: reduce if not converging
        if it > 20:
            eta *= 0.999

    return a_cur


def _freq_flatten(a, n_iter=50, alpha=0.3, deadline=None):
    """
    Frequency-domain flattening: adjust the power spectrum to be more uniform.
    Gerchberg-Saxton inspired: alternate between time-domain (non-negativity)
    and frequency-domain (flat spectrum) constraints.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_cur = a.copy()

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Compute DFT of zero-padded a
        a_pad = np.zeros(N)
        a_pad[:n] = a_cur
        A = np.fft.rfft(a_pad)

        # Target: flat magnitude (|A(f)| = constant for all f)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target magnitude: uniform, preserving DC
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]

        # Blend
        new_mag = alpha * target_mag + (1.0 - alpha) * mag
        A_new = new_mag * np.exp(1j * phase)

        # Back to time domain
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_cur = a_new / s

    return a_cur


def _subgrad_polish(a_init, deadline, k=5, max_steps=20000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        z = z - lr * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -100, 100)

        if i % 2000 == 0:
            a_check = _softmax(z)

            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.02, 0.1)
                stall = 0
            else:

                if stall > 5:
                    lr *= 0.5
                    if lr < 1e-12:
                        break

                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final
    return best_a, best_sc


def _adaptive_lbfgs(a_start, deadline, reserve_time=120.0):
    """L-BFGS with geometric beta schedule, adaptive iteration count."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Geometric schedule: from 0.05*max_c to 1e-10*max_c
    n_betas = 30
    ratio = (1e-10 / 0.05) ** (1.0 / (n_betas - 1))
    betas = [0.05 * max_c * (ratio ** i) for i in range(n_betas)]
    betas = [max(b, 1e-14) for b in betas]

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        # Allocate iterations based on remaining time and problem size
        scale = max(1.0, (n / 4096.0) ** 1.1)
        # Estimate time per iteration
        time_per_iter = 0.002 * scale
        max_iter = min(10000, max(100, int(remaining / (time_per_iter * 3))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.501:
            break

    return current_a


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, rng, sigma=0.05):
    """Perturb in log-space for diversification."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    result = _softmax(z)
    return result

    print(f"\n=== Phase 1: L-BFGS (n={n}) ===")
    a_p1 = _adaptive_lbfgs(best_seq, deadline, reserve_time=400.0)
    try_update(a_p1, "lbfgs_main")

    # === Phase 2: KKT active-set solving (multiple k values) ===
    print(f"\n=== Phase 2: KKT active-set ===")
    for k in [3, 5, 7, 10, 15, 20]:
        if time.monotonic() > deadline - 200:
            break
        a_kkt = _kkt_active_solve(best_seq, k, n_iter=500,
                                  deadline=deadline - 200, eta_init=0.5)
        sc = _score(a_kkt)
        if sc < best_score_val:
            try_update(a_kkt, f"kkt_k{k}")
        else:
            # Try L-BFGS from KKT result as warm start
            if time.monotonic() < deadline - 150:
                c_k = _full_convolve(a_kkt, a_kkt)
                beta_k = float(c_k.max()) * 1e-6
                a_lb = _lbfgs_smooth(a_kkt, beta_k, n, 3000)
                try_update(a_lb, f"kkt_k{k}_refine")

    # === Phase 3: Frequency-domain flattening ===
    print(f"\n=== Phase 3: Frequency flattening ===")
    for alpha_val in [0.2, 0.4, 0.6]:
        if time.monotonic() > deadline - 150:
            break
        a_ff = _freq_flatten(best_seq, n_iter=100, alpha=alpha_val,
                             deadline=deadline - 150)
        try_update(a_ff, f"freq_a{alpha_val}")
        if time.monotonic() < deadline - 100:
            # Refine with L-BFGS
            c_ff = _full_convolve(a_ff, a_ff)
            beta_ff = float(c_ff.max()) * 1e-5
            a_ref = _lbfgs_smooth(a_ff, beta_ff, n, 2000)
            try_update(a_ref, f"freq_a{alpha_val}_ref")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient polishing ===")
    for k in [3, 5, 8]:
        if time.monotonic() > deadline - 100:
            break
        a_sg, _ = _subgrad_polish(best_seq, deadline - 100, k=k,
                                  max_steps=15000, lr_init=0.03)
        try_update(a_sg, f"sg_k{k}")

    # === Phase 5: Interleaved refinement loop ===
    print(f"\n=== Phase 5: Interleaved refinement ===")
    for rnd in range(20):
        if time.monotonic() > deadline - 60:
            break

        # KKT with small k
        a_k = _kkt_active_solve(best_seq, 3, n_iter=200,
                                deadline=deadline - 60, eta_init=0.3)
        try_update(a_k, f"int_kkt3_{rnd}")

        # KKT with medium k
        if time.monotonic() < deadline - 60:
            a_k5 = _kkt_active_solve(best_seq, 5, n_iter=200,
                                     deadline=deadline - 60, eta_init=0.3)
            try_update(a_k5, f"int_kkt5_{rnd}")

        # Tight L-BFGS
        if time.monotonic() < deadline - 60:
            c_b = _full_convolve(best_seq, best_seq)
            beta = float(c_b.max()) * (1e-8 * (0.5 ** rnd))
            beta = max(beta, 1e-14)
            max_it = min(3000, max(200, int((deadline - 60 - time.monotonic()) / 0.01)))
            a_l = _lbfgs_smooth(best_seq, beta, n, max_it)
            try_update(a_l, f"int_lbfgs_{rnd}")

        # Very tight subgradient
        if time.monotonic() < deadline - 60:
            a_s, _ = _subgrad_polish(best_seq, deadline - 60, k=3,
                                     max_steps=5000, lr_init=0.01)
            try_update(a_s, f"int_sg_{rnd}")

    # === Phase 6: Diversification (perturb + re-optimize) ===
    print(f"\n=== Phase 6: Diversification ===")
    for pert in range(5):
        if time.monotonic() > deadline - 60:
            break
        sigma = 0.02 + 0.01 * pert
        a_pert = _perturb(best_seq, rng, sigma)
        try_update(a_pert, f"perturb_{pert}")

        # Quick L-BFGS from perturbation
        if time.monotonic() < deadline - 40:
            c_p = _full_convolve(a_pert, a_pert)
            beta_p = float(c_p.max()) * 1e-5
            a_opt = _lbfgs_smooth(a_pert, beta_p, n, 2000)
            try_update(a_opt, f"perturb_opt_{pert}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration ===")
    for target_n in [8192, 32768, 16384, 4096]:
        if time.monotonic() > deadline - 40:
            break
        if target_n == n:
            continue
        if target_n > 32768 or target_n < 1024:
            continue
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        try_update(a_interp, f"interp_{target_n}")

        if time.monotonic() < deadline - 20:
            c_i = _full_convolve(a_interp, a_interp)
            beta_i = float(c_i.max()) * 0.005
            max_it = min(4000, max(500, int((deadline - 20 - time.monotonic()) / 0.01)))
            a_opt = _lbfgs_smooth(a_interp, beta_i, target_n, max_it)
            try_update(a_opt, f"opt_{target_n}")

    # === Phase 8: Final aggressive polishing ===
    print(f"\n=== Phase 8: Final polishing ===")
    final_n = len(best_seq)
    for rnd in range(20):
        if time.monotonic() > deadline - 3:
            break

        # Ultra-tight L-BFGS
        c_b = _full_convolve(best_seq, best_seq)
        beta = float(c_b.max()) * 1e-12
        beta = max(beta, 1e-16)
        max_it = min(2000, max(100, int((deadline - 3 - time.monotonic()) / 0.008)))
        a_f = _lbfgs_smooth(best_seq, beta, final_n, max_it)
        try_update(a_f, f"final_lbfgs_{rnd}")

        # KKT with very small eta
        a_k = _kkt_active_solve(best_seq, 3, n_iter=100,
                                deadline=deadline - 3, eta_init=0.1)
        try_update(a_k, f"final_kkt_{rnd}")

        # KKT with k=5
        if time.monotonic() < deadline - 3:
            a_k5 = _kkt_active_solve(best_seq, 5, n_iter=100,
                                     deadline=deadline - 3, eta_init=0.1)
            try_update(a_k5, f"final_kkt5_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
