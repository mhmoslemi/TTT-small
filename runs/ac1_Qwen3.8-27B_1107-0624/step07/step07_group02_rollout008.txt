"""
Hybrid QP + L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set QP for peak reduction, L-BFGS with log-sum-exp smoothing,
and multi-scale n exploration. Key innovation: convex QP sub-problems for
targeted peak cancellation via first-order linearization of convolution.

import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    def __init__(self, beta, n):
        self.beta = beta

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

        # We need B[k, j] = a[k-j] if k-j is in [i_lo, i_hi], i.e., j in [k-i_hi, k-i_lo]
        j_lo = k - i_hi
        j_hi = k - i_lo
        # B[k, j] = a[k-j] for j in [j_lo, j_hi]
        row = np.zeros(n)
        a_vals = a[j_lo:k - j_lo + 1][::-1]  # a[k-j] for j from j_lo to j_hi
        row[j_lo:j_hi + 1] = a_vals
        B_rows.append(row)
        rhs.append((target - c[k]) / 2.0)

    B = np.array(B_rows)  # shape (m, n)
    rhs = np.array(rhs)

    # Feasibility check: if any rhs < 0, the constraint is already violated
    # and we can't satisfy it (since conv(a,d) can be at most something)
    # Actually we CAN satisfy it by reducing d in the right direction
    # Let's just try to solve.

    if not HAS_CVXPY:
        return None, 'no_cvxpy'

    d_var = cp.Variable(n)
    # Non-negativity: only add constraints for a_i close to 0 to keep QP small
    zero_tol = 1e-6
    neg_mask = a < zero_tol * a.max()
    nneg_constraints = [d_var[neg_mask] >= -a[neg_mask]] if neg_mask.any() else []

    constraints = [cp.sum(d_var) == 0]
    constraints.append(B @ d_var <= rhs)
    constraints.extend(nneg_constraints)

    objective = cp.Minimize(cp.sum_squares(d_var))
    prob = cp.Problem(objective, constraints)

    solver_list = ['ECOS', 'SCS'] if HAS_CVXPY else []
    # Try available solvers
    solved = False
    for solver in solver_list:
        if time.monotonic() > deadline:
            break
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue

    if not solved:
        # Try default
        try:
            prob.solve()
            solved = True
        except Exception:
            return None, 'solve_failed'

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, f'status_{prob.status}'

    return d_var.value.copy(), 'ok'


def _qp_phase(a_init, deadline, min_time_remaining=60.0):
    """Main QP-based peak reduction phase."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Adaptive parameters
    active_frac = 0.02  # start with top 2%
    target_frac = 0.98  # reduce peaks to 98% of max initially
    consecutive_failures = 0
    max_failures = 10
    iteration = 0
    t_start = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            if iteration % 20 == 0:
                print(f"  QP iter {iteration}: score={best_sc:.10f}, active_frac={active_frac:.4f}, target_frac={target_frac:.4f}")
            consecutive_failures = 0
        else:
            consecutive_failures += 1

        if consecutive_failures >= max_failures:
            # Relax parameters
            active_frac = min(0.1, active_frac * 1.5)
            target_frac = min(0.999, target_frac + 0.01)
            consecutive_failures = 0
            if active_frac >= 0.1 and target_frac >= 0.999:
                break

        # Select active set
        n_active = max(5, int(n * active_frac))
        c_sorted_idx = np.argsort(c)[::-1]
        active_indices = c_sorted_idx[:n_active]
        active_vals = c[active_indices]

        # Set target: reduce peaks to target_frac of max
        target = c_max * target_frac

        # Solve QP
        d, status = _solve_qp_peak_reduction(a, active_indices, target,
                                             deadline - min_time_remaining)
        if d is None:
            consecutive_failures += 2
            continue

        # Line search
        a_old_sc = current_sc
        alpha = 1.0
        improved = False
        for ls_iter in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller target reduction
            target_frac = min(0.9999, target_frac + 0.005)
            active_frac = max(0.005, active_frac * 0.7)

        # Gradually tighten
        if time.monotonic() - t_start < (deadline - min_time_remaining - t_start) * 0.5:
            target_frac = max(0.95, target_frac * 0.995)
        else:
            target_frac = max(0.90, target_frac * 0.98)

        if time.monotonic() + min_time_remaining > deadline:
            break

    return best_a, _score(best_a)


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 400
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.75)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 8:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.02)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _peak_cancellation_1d(a, deadline, min_time_remaining=10.0):
    """1D peak cancellation: directly reduce the top peaks by adjusting
    the elements that contribute most to them."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    t_start = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()

        # Find top peaks
        top_k = int(np.argmax(c))
        peak_val = c[top_k]

        # The peak at position top_k is sum_i a[i]*a[top_k-i]
        # Find which a[i] contribute most
        i_lo = max(0, top_k - (n - 1))
        i_hi = min(n - 1, top_k)
        contributions = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = top_k - i
            contributions[i] += a[j]
            if i != j:
                contributions[j] += a[i]

        # Reduce the top contributing elements
        n_reduce = max(1, int(n * 0.005))
        top_contrib_idx = np.argsort(contributions)[::-1][:n_reduce]

        # Apply reduction proportional to contribution
        reduction = peak_val * 0.01  # reduce peak by 1%
        for idx in top_contrib_idx:
            a[idx] -= reduction / (2.0 * contributions[idx] + 1e-30) * 0.1

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if time.monotonic() - t_start > 300:
            break

    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """Hybrid QP + L-BFGS optimizer with peak cancellation and multi-n exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS warm-up ===
    print(f"\n[P1] L-BFGS warm-up (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: QP-based peak reduction ===
    print(f"\n[P2] QP peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        qp_deadline = min(deadline - 500, time.monotonic() + 600)
        a_qp, sc_qp = _qp_phase(best_seq, qp_deadline, min_time_remaining=100.0)
        if sc_qp < best_score:
            best_seq = a_qp.copy()
            best_score = sc_qp
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [qp] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  QP: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set polish ===
    print(f"\n[P3] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 200)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.005, tol_frac_end=0.0001,
                                           lr_init=0.03)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Peak cancellation 1D ===
    print(f"\n[P4] Peak cancellation 1D (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pc_deadline = min(deadline - 200, time.monotonic() + 100)
        a_pc, sc_pc = _peak_cancellation_1d(best_seq, pc_deadline, min_time_remaining=50.0)
        if sc_pc < best_score:
            best_seq = a_pc.copy()
            best_score = sc_pc
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_cancel] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 12288, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        if time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 55, time.monotonic() + 50),
                                                 tol_frac_start=0.003, tol_frac_end=0.0001,
                                                 lr_init=0.015)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 80)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 60:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=30.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: QP again with tightened parameters ===
    print(f"\n[P7] QP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        qp_deadline2 = min(deadline - 40, time.monotonic() + 100)
        a_qp2, sc_qp2 = _qp_phase(best_seq, qp_deadline2, min_time_remaining=20.0)
        if sc_qp2 < best_score:
            best_seq = a_qp2.copy()
            best_score = sc_qp2
            print(f"  NEW BEST: {best_score:.10f} [qp2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           tol_frac_start=0.001, tol_frac_end=0.00001,
                                           lr_init=0.005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(2000, max(20, int(remaining / 0.04)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 60})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
