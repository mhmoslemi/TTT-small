"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral iterations with non-negativity projection
2. Smooth gradient descent in log-space with adaptive beta annealing
3. Active-set subgradient with FFT-based gradient computation
4. Basin hopping with spectral perturbations
5. Multi-length exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve_fft(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _autocorr(a):
    n = len(a)
    c = _full_convolve_fft(a, a)
    return c


def _autocorr_fft(a):
    """Compute linear autocorrelation using FFT. Returns full (2n-1) vector."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    power = A * np.conj(A)
    c = np.fft.irfft(power, N)[:2 * n - 1]
    return c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax(z):
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    return z - z_max - np.log(s)


def _smooth_max_grad(c, beta):
    """
    Compute the log-sum-exp smooth approximation of max(c) and its
    gradient with respect to c (i.e., the softmax weights).
    Returns: (smooth_value, weights)
    """
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    w = e / s
    return c_max + beta * np.log(s), w


def _grad_smooth_max(a, beta, n):
    """
    Compute gradient of smooth_max(autocorr(a)) with respect to a.
    Uses FFT-based computation.
    
    Returns gradient vector of length n.
    """
    c = _autocorr_fft(a)
    smooth_val, w = _smooth_max_grad(c, beta)
    
    # Gradient: 2 * Re(IFFT(FFT(w) * A))[0:n]
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    grad_full = np.fft.irfft(W * A, N)
    grad = 2.0 * grad_full[:n]
    
    return smooth_val, grad


def _gs_iteration(a, mode='exp', alpha=1.0, gamma=1.0):
    """
    One Gerchberg-Saxton iteration.
    
    mode: 'scale' - divide by sqrt(max)
          'exp'   - multiply by exp(-alpha*(p - mean))
          'power' - divide by p^(1/gamma)
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    power = A * np.conj(A)
    p = np.fft.irfft(power, N)[:2 * n - 1]
    
    p_max = p.max()
    if p_max < 1e-30:
        return a.copy()
    
    if mode == 'scale':
        # Scale power spectrum to reduce peak
        new_power = power / np.sqrt(p_max)
    elif mode == 'exp':
        # Exponential weighting
        correction = np.exp(-alpha * (p - p.mean()))
        correction = correction / correction.mean()
        new_power = power * correction
    elif mode == 'power':
        # Power law
        safe_p = np.maximum(p, 1e-30)
        correction = 1.0 / np.power(safe_p, 1.0 / gamma)
        correction = correction / correction.mean()
        new_power = power * correction
    else:
        return a.copy()
    
    # Preserve phase
    new_A = np.sqrt(np.maximum(new_power, 0)) * np.exp(1j * np.angle(A + 0j))
    # Actually, for rfft, we need to be careful
    # A is real FFT, so let's work with the magnitude and phase
    mag_A = np.abs(A)
    phase_A = np.angle(A)
    new_mag = np.sqrt(np.maximum(new_power, 0))
    new_A = new_mag * np.exp(1j * phase_A)
    
    a_new = np.fft.irfft(new_A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new / s
    return a_new


def _gs_optimize(a_init, deadline, n_iters=5000, reserve=30.0):
    """
    Optimize using GS iterations with adaptive parameters.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    # Adaptive alpha: start large, decrease over time
    alpha = 2.0
    alpha_decay = 0.999
    
    for it in range(n_iters):
        if time.monotonic() > deadline - reserve:
            break
        
        # GS step
        a_new = _gs_iteration(a, mode='exp', alpha=alpha)
        
        # Also try a 'scale' mode every few iterations
        if it % 5 == 0:
            a_new2 = _gs_iteration(a, mode='scale')
            sc2 = _score(a_new2)
            if sc2 < _score(a_new):
                a_new = a_new2
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            # Slightly increase alpha for more aggressive correction
            alpha = min(alpha * 1.01, 5.0)
        else:
            a = a_new  # still update a (GS is a fixed-point iteration)
            alpha *= alpha_decay
        
        if it % 1000 == 0:
            print(f"    GS iter {it}: score={best_sc:.10f}, alpha={alpha:.4f}")
    
    return best_a, best_sc


def _smooth_grad_optimize(a_init, deadline, beta_start=None, beta_end=None,
                          n_iters=10000, reserve=30.0):
    """
    Smooth gradient descent in log-space with adaptive beta.
    """
    n = len(a_init)
    
    if beta_start is None:
        c0 = _autocorr_fft(a_init)
        beta_start = c0.max() * 0.5
    if beta_end is None:
        beta_end = c0.max() * 1e-6
    
    # Initialize in log-space
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    log_beta_start = np.log(beta_start)
    log_beta_end = np.log(beta_end)
    
    # Learning rate
    lr = 0.1
    patience = 0
    max_patience = 200
    
    for it in range(n_iters):
        if time.monotonic() > deadline - reserve:
            break
        
        # Anneal beta
        frac = it / max(1, n_iters - 1)
        beta = np.exp(log_beta_start + frac * (log_beta_end - log_beta_start))
        
        a = _softmax(z)
        
        # Compute gradient of smooth max w.r.t. a
        smooth_val, grad_a = _grad_smooth_max(a, beta, n)
        
        # Convert to z-space gradient: grad_z = a * (grad_a - dot(grad_a, a))
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        # Gradient step
        z = z - lr * grad_z
        z = z - z.mean()
        z = np.clip(z, -50.0, 50.0)
        
        # Check every 100 iterations
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
                patience = 0
                lr = min(lr * 1.02, 1.0)
            else:
                patience += 1
                if patience > max_patience:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                    patience = 0
        
        if it % 2000 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            print(f"    SMOOTH iter {it}: beta={beta:.2e}, score={best_sc:.10f}, lr={lr:.4f}")
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _active_set_optimize(a_init, deadline, k_active=5, n_iters=20000, reserve=30.0):
    """
    Active-set gradient descent: focus on the top-k peaks.
    Uses the subgradient of max at the active set.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.5
    patience = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline - reserve:
            break
        
        a = _softmax(z)
        c = _autocorr_fft(a)
        c_max = c.max()
        
        # Active set: top k indices
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active
        
        # Gradient: 2 * IFFT(FFT(w) * A)[0:n]
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        grad_full = np.fft.irfft(W * A, N)
        grad_a = 2.0 * grad_full[:n]
        
        # z-space gradient
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        # Step
        z = z - lr * grad_z
        z = z - z.mean()
        z = np.clip(z, -50.0, 50.0)
        
        if it % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
                patience = 0
                lr = min(lr * 1.01, 2.0)
            else:
                patience += 1
                if patience > 100:
                    lr *= 0.8
                    if lr < 1e-8:
                        break
                    patience = 0
        
        if it % 5000 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            print(f"    ACTIVE iter {it}: k={k_active}, score={best_sc:.10f}, lr={lr:.4f}")
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _combine_methods(a_init, deadline, reserve=60.0):
    """
    Combine GS and gradient methods in an alternating fashion.
    """
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    phase = 0  # 0=GS, 1=smooth grad, 2=active set
    phase_iters = 1000
    
    while time.monotonic() < deadline - reserve:
        if phase == 0:
            a_new, sc_new = _gs_optimize(current_a, deadline, n_iters=phase_iters, reserve=reserve)
        elif phase == 1:
            c = _autocorr_fft(current_a)
            beta = c.max() * 0.1
            a_new, sc_new = _smooth_grad_optimize(current_a, deadline, beta_start=beta,
                                                   n_iters=phase_iters, reserve=reserve)
        else:
            a_new, sc_new = _active_set_optimize(current_a, deadline, k_active=5,
                                                  n_iters=phase_iters, reserve=reserve)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            current_a = a_new.copy()
            print(f"      Phase {phase}: NEW BEST {best_sc:.10f}")
        else:
            current_a = a_new.copy()  # still update to potentially escape
        
        phase = (phase + 1) % 3
    
    return best_a, best_sc


def _basin_hop_spectral(a_current, deadline, rng, n_hops=5, reserve=60.0):
    """
    Basin hopping with spectral-domain perturbations.
    """
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)
    
    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6
        
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a_current, N)
        
        pert_type = h % 3
        if pert_type == 0:
            # Random phase perturbation in frequency domain
            noise = rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A))
            A_pert = A * (1 + 0.05 * noise / np.abs(A + 1e-30))
            a_pert = np.fft.irfft(A_pert, N)[:n]
        elif pert_type == 1:
            # Band-limited perturbation
            noise = np.zeros_like(A)
            band_start = int(n * 0.1 * rng.uniform(0, 1))
            band_end = min(len(A), band_start + int(n * 0.1))
            noise[band_start:band_end] = rng.standard_normal(band_end - band_start)
            A_pert = A + 0.1 * noise
            a_pert = np.fft.irfft(np.abs(A_pert) * np.exp(1j * np.angle(A + 0j)), N)[:n]
        else:
            # Log-space perturbation
            z = np.log(np.maximum(a_current, 1e-18))
            z = z + 0.1 * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-30:
            continue
        a_pert = a_pert / s
        
        # Quick GS optimization from perturbed point
        a_opt, sc_opt = _gs_optimize(a_pert, hop_deadline, n_iters=2000, reserve=5.0)
        
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"      HOP {h}: NEW BEST {best_sc:.10f}")
    
    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining GS, smooth gradient, and active-set methods.
    Key innovation: FFT-based gradient computation and spectral GS iterations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score_val = np.inf
    best_n = 4096
    
    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
