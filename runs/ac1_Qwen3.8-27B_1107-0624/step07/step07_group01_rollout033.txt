Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected subgradient with Nesterov acceleration and Polyak step
2. FFT-based fast convolution and subgradient computation
3. Periodic L-BFGS polishing with log-sum-exp smoothing
4. Adaptive active set management
5. Multi-phase strategy with time-budget allocation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _fft_selfconv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return r


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _fft_selfconv(a)
    return float(2.0 * n * np.max(c) / (s * s))


def _proj_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _subgradient(a, active_idx):
    """Compute subgradient of max_k q_k(a) at given active indices.
    q_k(a) = sum_j a[j]*a[k-j], grad w.r.t a is 2*a[k-j] (reversed/shifted).
    Returns gradient vector of length len(a).
    """
    n = len(a)
    # For each active lag k, the gradient contribution at position j is 2*a[k-j]
    # Sum over active lags, average
    # g[j] = (2/|A|) * sum_{k in A} a[k-j] (where k-j is in [0, n-1])
    # This is: (2/|A|) * (indicator_A correlated with a)[j]
    # Use FFT for efficiency
    indicator = np.zeros(2 * n - 1)
    for k in active_idx:
        indicator[k] = 1.0
    # correlation: (indicator * a_rev) where a_rev[i] = a[n-1-i]
    # Actually: sum_k indicator[k] * a[k-j] = convolution of indicator with a, cropped
    # conv(indicator, a)[j] = sum_k indicator[k] * a[j-k]
    # We want sum_k indicator[k] * a[k-j] which is conv(indicator, a_reversed)
    a_rev = a[::-1]
    corr = _fft_convolve(indicator, a_rev)
    # corr[j] = sum_k indicator[k] * a_rev[j-k] = sum_k indicator[k] * a[n-1-j+k]
    # We want sum_k indicator[k] * a[k-j]
    # Let's think differently: result[j] = sum_k indicator[k] * a[k-j]
    # This is the cross-correlation of indicator with a
    # = conv(indicator, a_flipped)[n-1-j] ... this is getting confusing
    # Let me just do it directly for small K, or use a different FFT trick
    # Direct: for each active k, add a[k-j] for valid j
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    g = np.zeros(n)
    for k in active_idx:
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            g[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
    g *= 2.0 / len(active_idx)
    return g


def _subgradient_fast(a, active_idx):
    """Faster subgradient using FFT when K is large."""
    n = len(a)
    K = len(active_idx)
    if K <= 20:
        return _subgradient(a, active_idx)
    # Use FFT
    NG = _next_pow2(3 * n)
    A = np.zeros(NG)
    A[:n] = a
    A[NG - n + 1:NG] = a[::-1]  # for the "wrap-around" part
    ind = np.zeros(NG)
    for k in active_idx:
        ind[k] = 1.0
    ind_fft = np.fft.rfft(ind)
    a_fft = np.fft.rfft(A)
    corr = np.fft.irfft(ind_fft * a_fft, NG)
    # corr[j] = sum_k ind[k] * A[j-k] (circular)
    # We want sum_k ind[k] * a[k-j] = sum_k ind[k] * A[n-1-(k-j)] ... 
    # Actually simpler: let b[j] = a[n-1-j] (reverse). Then a[k-j] = b[n-1-k+j]
    # sum_k ind[k]*a[k-j] = sum_k ind[k]*b[n-1-k+j]
    # Let m = n-1+j: sum_k ind[k]*b[m-k] = (ind * b)[m] (convolution)
    # So result[j] = (ind * b)[n-1+j]
    b = a[::-1].copy()
    conv_ib = _fft_convolve(ind[:2*n-1], b)  # length (2n-1)+(n-1) = 3n-2
    # result[j] = conv_ib[n-1+j] for j=0,...,n-1
    g = conv_ib[n - 1:2 * n - 1] * (2.0 / K)
    return g


def _lbfgs_smooth(a_init, beta, n, max_iter, deadline=None, reserve=10.0):
    """L-BFGS on log-sum-exp smoothed objective in log-space."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - np.mean(z0)

    def fun(z):
        p = np.exp(z - np.max(z))
        p = p / p.sum()
        c = _fft_selfconv(p)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def jac(z):
        p = np.exp(z - np.max(z))
        p = p / p.sum()
        c = _fft_selfconv(p)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = p[::-1]
        wc = _fft_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * p * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = np.exp(result.x - np.max(result.x))
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        return a_init.copy()


def _proj_subgrad_nesterov(a0, deadline, k_active=10, k_decay=0.9999):
    """Projected subgradient with Nesterov acceleration."""
    n = len(a0)
    a = a0.copy()
    y = a0.copy()  # lookahead point
    best_a = a0.copy()
    best_val = float(np.max(_fft_selfconv(a0)))
    t_mom = 1.0
    beta_nest = 0.9  # momentum
    f_lower = 1.0 / (2 * n - 1)  # lower bound on max conv value
    base_scale = 1.0 / n  # typical scale of conv values
    step_scale = base_scale
    stall = 0
    k_cur = k_active

    for it in range(2000000):
        if time.monotonic() > deadline:
            break

        # Evaluate at lookahead point
        c_y = _fft_selfconv(y)
        val_y = float(c_y.max())
        idx = np.argsort(c_y)[::-1][:k_cur]

        # Subgradient at y
        g = _subgradient_fast(y, idx)
        # Project to tangent space
        g_t = g - np.mean(g)
        g_norm_sq = float(np.dot(g_t, g_t))
        if g_norm_sq < 1e-30:
            break

        # Nesterov momentum update
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        a = a + ((t_mom - 1.0) / t_mom) * (y - a)

        # Polyak-like step size
        alpha = (val_y - f_lower) / (g_norm_sq + 1e-30)
        # Cap the step
        alpha = min(alpha, 5.0 * step_scale)
        alpha = max(alpha, 1e-15)

        # Take step from a (not y, for stability)
        a_new = _proj_simplex(a - alpha * g_t)

        # Evaluate
        c_new = _fft_selfconv(a_new)
        val_new = float(c_new.max())

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                beta_nest = max(0.5, beta_nest * 0.95)
                step_scale *= 0.7
                stall = 0
            elif stall > 200:
                step_scale *= 0.85

        # Decay K over time (focus on fewer, more important constraints)
        k_cur = max(2, int(k_active * (k_decay ** it)))

        # Update y for next iteration
        if val_new <= val_y + 1e-15:
            y = a_new + beta_nest * (a_new - a)
        else:
            y = a_new.copy()

    return best_a, best_val


def _proj_subgrad_simple(a0, deadline, k_active=5):
    """Simple projected subgradient with backtracking."""
    n = len(a0)
    a = a0.copy()
    best_a = a0.copy()
    best_val = float(np.max(_fft_selfconv(a0)))
    f_lower = 1.0 / (2 * n - 1)
    alpha = 1.0 / n
    stall = 0

    for it in range(5000000):
        if time.monotonic() > deadline:
            break

        c = _fft_selfconv(a)
        val_cur = float(c.max())
        idx = np.argsort(c)[::-1][:k_active]

        g = _subgradient_fast(a, idx)
        g_t = g - np.mean(g)
        g_norm_sq = float(np.dot(g_t, g_t))
        if g_norm_sq < 1e-30:
            break

        # Try step
        a_try = _proj_simplex(a - alpha * g_t)
        val_try = float(np.max(_fft_selfconv(a_try)))

        if val_try < best_val - 1e-15:
            best_val = val_try
            best_a = a_try.copy()
            stall = 0
            a = a_try
            alpha = min(alpha * 1.01, 10.0 / n)
        elif val_try < val_cur:
            a = a_try
            stall = 0
        else:
            stall += 1
            alpha *= 0.7
            if stall > 500:
                alpha = 1.0 / n
                stall = 0

    return best_a, best_val


def _alternate_qp(a0, deadline, k_active=5):
    """Alternating method: solve QP to reduce active values."""
    n = len(a0)
    a = a0.copy()
    best_a = a0.copy()
    best_val = float(np.max(_fft_selfconv(a0)))

    for it in range(5000):
        if time.monotonic() > deadline:
            break

        c = _fft_selfconv(a)
        val_cur = float(c.max())
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        target = val_cur * 0.999  # try to reduce by 0.1%

        # Build QP constraints: 2*(Q_k * a) . w <= target - c[k] for each active k
        # (Q_k * a)[j] = a[k-j] for valid k-j
        C = np.zeros((k_active, n))
        for i, k in enumerate(idx):
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    C[i, j] = 2.0 * a[kj]
        d = target - active_vals

        # Solve: min 0.5*||w||^2 s.t. a+w >= 0, sum(w)=0, Cw <= d
        # Use scipy or manual
        try:
            # Simple approach: projected gradient on the QP
            # Or use the fact that this is equivalent to finding
            # the point in {w: a+w>=0, sum(w)=0, Cw<=d} closest to 0
            # Use iterative method
            w = np.zeros(n)
            for _ in range(50):
                # Project onto sum(w)=0
                w -= np.mean(w)
                # Project onto a+w >= 0 (w >= -a)
                w = np.maximum(w, -a + 1e-15)
                w -= np.mean(w)
                w = np.maximum(w, -a + 1e-15)
                # Project onto Cw <= d (violation-based)
                viol = C @ w - d
                pos_viol = np.maximum(viol, 0)
                c_norms = np.maximum(np.linalg.norm(C, axis=1), 1e-15)
                w -= (C.T @ (pos_viol / c_norms**2))
                w -= np.mean(w)
                w = np.maximum(w, -a + 1e-15)
                w -= np.mean(w)

            a_try = a + w
            a_try = _proj_simplex(a_try)
            val_try = float(np.max(_fft_selfconv(a_try)))

            if val_try < best_val:
                best_val = val_try
                best_a = a_try.copy()
                a = a_try
        except Exception:
            break

    return best_a, best_val


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5, asym=0.0):
    """Tapered with optional asymmetry."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    if asym != 0:
        a *= (1.0 + asym * (2 * t - 1))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_ramp(n, strength=0.5):
    """Linearly ramped sequence."""
    t = np.arange(n) / n
    a = 1.0 + strength * (2 * t - 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic(n, curvature=1.0, shift=0.0):
    """Quadratic shape."""
    t = np.linspace(0, 1, n)
    a = 1.0 + curvature * ((t - 0.5) ** 2 - 0.25) + shift * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-15)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _adaptive_lbfgs_refine(a_start, deadline, reserve=30.0):
    """Multi-scale L-BFGS refinement."""
    n = len(a_start)
    c_cur = _fft_selfconv(a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Start with moderate beta, decrease
    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve > deadline:
            break
        remaining = deadline - reserve - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.0)
        max_iter = min(5000, max(50, int(remaining / (0.005 * scale))))
        
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter, deadline, reserve)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5001:
            break

    return current_a


def _coordinate_descent(a, deadline, max_rounds=100):
    """Coordinate descent: optimize pairs of coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_val = float(np.max(_fft_selfconv(a)))
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick random pairs to swap mass
        n_pairs = min(500, n // 4)
        for _ in range(n_pairs):
            i = int(rng.integers(0, n - 1))
            j = int(rng.integers(i + 1, n))

            # Try moving mass from i to j and vice versa
            for direction in [1, -1]:
                for frac in [0.5, 0.1, 0.01]:
                    amt = direction * frac * (a[i] + a[j]) / 2
                    if a[i] + amt < 0 or a[j] - amt < 0:
                        continue
                    a[i] += amt
                    a[j] -= amt
                    val = float(np.max(_fft_selfconv(a)))
                    if val < best_val:
                        best_val = val
                        best_a = a.copy()
                    else:
                        a[i] -= amt
                        a[j] += amt
                        break

        # Periodic L-BFGS
        if rnd % 20 == 19 and time.monotonic() < deadline - 10:
            a = _lbfgs_smooth(a, _fft_selfconv(a).max() * 0.01, n, 2000)
            val = float(np.max(_fft_selfconv(a)))
            if val < best_val:
                best_val = val
                best_a = a.copy()

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using projected subgradient with Nesterov acceleration,
    L-BFGS polishing, and adaptive strategies.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    n_init = 16384

    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 10:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        candidates.append(("parent", parent))
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Structured initializations
    structured = [
        ("ramp05", _make_ramp(n_init, 0.5)),
        ("ramp10", _make_ramp(n_init, 1.0)),
        ("ramp15", _make_ramp(n_init, 1.5)),
        ("quad1", _make_quadratic(n_init, 1.0, 0.0)),
        ("quad05s", _make_quadratic(n_init, 0.5, 0.3)),
        ("taper_asym", _make_tapered(n_init, 0.5, 0.3)),
        ("taper_strong", _make_tapered(n_init, 1.0, 0.5)),
        ("chirp1", _make_chirp(n_init, 1.0, 0.3)),
    ]

    for name, a_init in structured:
        candidates.append((name, a_init))
        sc = _score(a_init)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_init, name)

    if best_seq is None:
        best_seq = _make_ramp(n_init, 0.5)
        best_score_val = _score(best_seq)
        print(f"  Default: score={best_score_val:.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: Quick L-BFGS to get a good starting point ===
    print(f"\n=== Phase 2: Initial L-BFGS (t={elapsed:.1f}s) ===")
    p2_deadline = min(deadline, t_start + 120.0)
    a_p2 = _adaptive_lbfgs_refine(best_seq, p2_deadline, reserve=5.0)
    try_update(a_p2, "init_lbfgs")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 2 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 3: Projected subgradient with Nesterov (main workhorse) ===
    print(f"\n=== Phase 3: Projected Subgradient (t={elapsed:.1f}s) ===")
    p3_deadline = min(deadline, t_start + 700.0)
    
    # Run with different K values
    for k_try in [15, 10, 7, 5, 3]:
        if time.monotonic() > p3_deadline:
            break
        seg_deadline = min(p3_deadline, time.monotonic() + 100.0)
        a_sg, sc_sg = _proj_subgrad_nesterov(best_seq, seg_deadline, k_active=k_try)
        try_update(a_sg, f"sg_nest_k{k_try}")

        # Interleave with L-BFGS
        if time.monotonic() < p3_deadline - 30:
            a_lb = _adaptive_lbfgs_refine(best_seq, min(p3_deadline, time.monotonic() + 60.0), reserve=10.0)
            try_update(a_lb, f"lbfgs_k{k_try}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 3 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 4: Simple subgradient + more L-BFGS ===
    print(f"\n=== Phase 4: Simple Subgradient + L-BFGS (t={elapsed:.1f}s) ===")
    p4_deadline = min(deadline, t_start + 1200.0)

    for k_try in [5, 3, 2]:
        if time.monotonic() > p4_deadline:
            break
        seg_deadline = min(p4_deadline, time.monotonic() + 80.0)
        a_sg, sc_sg = _proj_subgrad_simple(best_seq, seg_deadline, k_active=k_try)
        try_update(a_sg, f"sg_simple_k{k_try}")

        if time.monotonic() < p4_deadline - 30:
            a_lb = _adaptive_lbfgs_refine(best_seq, min(p4_deadline, time.monotonic() + 60.0), reserve=10.0)
            try_update(a_lb, f"lbfgs_post_k{k_try}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 4 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 5: Aggressive L-BFGS with tiny beta ===
    print(f"\n=== Phase 5: Aggressive L-BFGS (t={elapsed:.1f}s) ===")
    p5_deadline = min(deadline, t_start + 1600.0)
    if best_seq is not None:
        n = len(best_seq)
        c_b = _fft_selfconv(best_seq)
        mb = float(c_b.max())

        # Very tight smoothing
        for beta_frac in [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]:
            if time.monotonic() > p5_deadline:
                break
            beta = mb * beta_frac
            max_iter = min(8000, max(200, int((p5_deadline - time.monotonic()) / 0.01)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"tight_lbfgs_{beta_frac}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 5 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (t={elapsed:.1f}s) ===")
    p6_deadline = min(deadline, t_start + 1750.0)
    if best_seq is not None:
        current_n = len(best_seq)
        other_lengths = [8192, 32768, 4096]
        for target_n in other_lengths:
            if time.monotonic() > p6_deadline - 30:
                break
            if target_n == current_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            try_update(a_interp, f"interp_n{target_n}")

            if time.monotonic() < p6_deadline - 30:
                a_opt = _adaptive_lbfgs_refine(a_interp, min(p6_deadline, time.monotonic() + 80.0), reserve=10.0)
                try_update(a_opt, f"opt_n{target_n}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 6 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 7: Final aggressive polishing ===
    print(f"\n=== Phase 7: Final Polish (t={elapsed:.1f}s) ===")
    if best_seq is not None:
        for rnd in range(8):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _fft_selfconv(best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            max_iter = min(3000, max(100, int((deadline - 5 - time.monotonic()) / 0.005)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_{rnd}")

            # Quick subgradient
            if time.monotonic() < deadline - 5:
                a_sg, _ = _proj_subgrad_nesterov(best_seq, deadline - 5, k_active=3)
                try_update(a_sg, f"final_sg_{rnd}")

    # === Final output ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
