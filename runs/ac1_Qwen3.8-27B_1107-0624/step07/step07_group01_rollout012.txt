def _next_pow2(n):

    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating minimax (primal L-BFGS + dual softmax update)
2. Peak equalization with balanced descent directions
3. Log-sum-exp L-BFGS polishing
4. Spectral shaping initializations
5. Multi-length exploration with interpolation
6. Micro-polishing with extremely tight tolerances

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _autocorr_gradient(a, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a, where c = convolve(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _autocorr_gradient(a, w)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,

def _alternating_minimax(a_init, deadline, reserve=30.0):
    """
    Alternating minimax:
    - Primal: L-BFGS to minimize sum_k p[k]*c[k] on simplex
    - Dual: Update p = softmax(c/tau) with decreasing tau
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    tau = c_max * 0.05
    p = np.exp((c - c_max) / tau)
    p = p / p.sum()

    stall = 0
    tau_decay = 0.97

    for iteration in range(50000):
        if time.monotonic() > deadline - reserve:
            break

        # Primal step: minimize sum_k p[k]*c[k] subject to a>=0, sum(a)=1
        # Use L-BFGS in log-space
        z0 = np.log(np.maximum(a, 1e-18))
        z0 = z0 - z0.mean()

        def primal_fun(z):
            a_loc = _softmax(z)
            c_loc = _full_convolve(a_loc, a_loc)
            return float(np.dot(p, c_loc))

        def primal_jac(z):
            a_loc = _softmax(z)
            grad = _autocorr_gradient(a_loc, p)
            # The gradient of sum_k p[k]*c[k] w.r.t. a is:
            # 2 * autocorr_gradient(a, p) (as we analyzed)
            # But in log-space: d/df = (d/dz * dz/da)^{-1} ... 
            # Actually in log-space with softmax: da/dz = a*(e_i - a)
            # So dL/dz = a * (dL/da - a^T * dL/da)
            return a_loc * (2.0 * grad - np.dot(a_loc, 2.0 * grad))

        max_iter = min(3000, max(100, int(60.0 / (0.003 * max(1, n/4096)))))
        try:
            result = minimize(primal_fun, z0, jac=primal_jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-10, 'maxcor': 50, 'disp': False})
            a_new = _softmax(result.x)
        except Exception:
            a_new = a.copy()

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        # Dual step: update p
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tau = max(c_max * 0.005, c_max * 0.05 * tau_decay ** iteration)
        p_new = p * np.exp((c - c_max) / tau)
        p_new_sum = p_new.sum()
        if p_new_sum < 1e-300 or not np.isfinite(p_new_sum):
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = p_new / p_new_sum

        # Check improvement
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

    return best_a, best_sc


def _peak_equalization(a_init, deadline, reserve=30.0, max_iter=20000):
    """
    Find the balanced descent direction that reduces all active peaks equally.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    step_size = 2.0 / n
    stall = 0

    for iteration in range(max_iter):
        if time.monotonic() > deadline - reserve:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set: peaks within tolerance of max
        tol = 0.005 * c_max
        active = np.where(c >= c_max - tol)[0]
        k_active = len(active)

        if k_active == 0:
            break

        # Compute balanced direction
        # We want d such that grad_c[k_j] . d = -1 for all j
        # This is: G d = -ones, where G[j,i] = (grad_c[k_j])[i]
        # Minimum norm solution: d = -G^T (G G^T + eps*I)^{-1} ones

        # For efficiency with large k_active, use top-k only
        if k_active > 20:
            top_idx = np.argsort(c)[::-1][:20]
        else:
            top_idx = active

        k_top = len(top_idx)

        # Build G matrix (k_top x n) - but this is expensive for large n
        # Instead, use the fact that grad_c[k] = 2*shifted_a
        # grad_c[k][i] = 2*a[k-i] if 0 <= k-i < n

        # Build G efficiently
        # G[j, i] = 2*a[top_idx[j] - i] if 0 <= top_idx[j] - i < n
        G = np.zeros((k_top, n))
        for j, k in enumerate(top_idx):
            if k < n:
                # G[j, 0:k+1] = 2*a[k:k:-1]
                G[j, 0:k + 1] = 2.0 * a[k:k:-1]
            else:
                # G[j, k-n+1:n] = 2*a[n-1:k-n:-1]
                start = k - n + 1
                G[j, start:n] = 2.0 * a[n - 1:start - 1:-1]

        # Project to tangent space of simplex: d - mean(d) = 0
        # Solve (G G^T + eps*I) d_coeffs = ones, then d = G^T d_coeffs
        GGT = G @ G.T
        k_top = GGT.shape[0]
        eps_reg = 1e-8 * np.trace(GGT) / k_top if k_top > 0 else 1e-8
        np.fill_diagonal(GGT, GGT.diagonal() + eps_reg)

        try:
            coeffs = np.linalg.solve(GGT, np.ones(k_top))
        except np.linalg.LinAlgError:
            coeffs = np.linalg.lstsq(GGT, np.ones(k_top), rcond=None)[0]

        d = G.T @ coeffs
        d = d - d.mean()  # project to tangent space

        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break
        d = d / dnorm

        # Line search
        best_alpha = step_size
        best_val = np.inf
        for trial in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
            a_try = a - trial * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            val = _score(a_try)
            if val < best_val:
                best_val = val
                best_alpha = trial

        a = a - best_alpha * d
        a = np.maximum(a, 0)
        a = a / a.sum()

        if best_val < best_sc - 1e-14:
            best_sc = best_val
            best_a = a.copy()
            stall = 0
            step_size = min(step_size * 1.02, 10.0 / n)
        else:
            stall += 1
            step_size *= 0.8
            if stall > 300:
                break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=30000, lr_init=0.05):
    """Subgradient polishing in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        h = _autocorr_gradient(a, w)
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = lr * a.sum()
        z = z - step * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall = 0
            else:
                stall += 1
                if stall > k_active * 3:
                    lr *= 0.6
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_spectral_flat(n, rng, n_harmonics=10):
    """Create a sequence with approximately flat power spectrum."""
    # Random phase in frequency domain
    spectrum = np.ones(n // 2 + 1)
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    spectrum[1:-1] = np.cos(phases[1:-1]) + 0j
    # Actually, let's use a real signal approach
    # a = IDFT of a flat-phase spectrum
    freq_s = np.zeros(n)
    freq_s[0] = 1.0
    for k in range(1, n // 2 + 1):
        freq_s[k] = np.cos(rng.uniform(0, 2 * np.pi))
        freq_s[n - k] = freq_s[k]
    a = np.fft.ifft(freq_s).real
    a = np.abs(a) + 0.01
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=5, reserve=10.0):
    """Basin hopping with small perturbations and quick re-optimization."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6

        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier perturbation (low frequency)
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2 + h // 2):
                freq = int(rng.integers(1, max(2, n // 32)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 2 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.03, 0.03)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Peak-targeted perturbation: increase values that contribute to peaks
            c = _full_convolve(best_a, best_a)
            peak_idx = int(np.argmax(c))
            # The peak at c[k] involves products a[i]*a[k-i]
            # Perturb a slightly around the indices that contribute most
            z = np.log(np.maximum(best_a, 1e-18))
            if peak_idx < n:
                affected = np.arange(max(0, peak_idx - n // 4), min(n, peak_idx + 1 + n // 4))
            else:
                affected = np.arange(max(0, peak_idx - n + 1 - n // 4), min(n, peak_idx - n + 2 + n // 4))
            z[affected] += rng.uniform(-0.02, 0.02, len(affected))
            z = z - z.mean()
            a_pert = _softmax(z)

        # Quick optimization
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.02
        if time.monotonic() < hop_deadline:
            a_opt = _lbfgs_smooth(a_pert, beta, n, min(3000, int((hop_deadline - time.monotonic()) / 0.005)))
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a, best_sc


def _micro_polish(a_init, deadline):
    """Very aggressive final polishing with tiny betas."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    c_init = _full_convolve(a_init, a_init)
    max_c = float(c_init.max())

    # Extremely tight beta schedule
    betas = [max_c * x for x in [1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(5000, max(500, int(remaining / 0.008)))
        a_new = _lbfgs_smooth(best_a, beta, n, max_iter)
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()

    # Final subgradient polishing
    if time.monotonic() < deadline - 5:
        a_sg, sc_sg = _subgrad_polish(best_a, deadline, k_active=3, max_steps=10000, lr_init=0.01)
        if sc_sg < best_sc:
            best_sc = sc_sg
            best_a = a_sg.copy()

    return best_a, best_sc


def _direct_descent(a_init, deadline, reserve=10.0):
    """Direct subgradient descent on the simplex with exact line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(50000):
        if time.monotonic() > deadline - reserve:
            break

        c = _full_convolve(a, a)
        idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        g = _autocorr_gradient(a, w)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm

        # Exact line search (a few trials)
        best_alpha = 0
        best_val = _score(a)
        for alpha in [step_size, step_size * 0.5, step_size * 0.75, step_size * 1.5]:
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            val = _score(a_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha == 0:
            step_size *= 0.5
            stall += 1
            if stall > 200:
                break
            continue

        a = a - best_alpha * g
        a = np.maximum(a, 0)
        a = a / a.sum()

        if best_val < best_sc - 1e-15:
            best_sc = best_val
            best_a = a.copy()
            stall = 0
            step_size = min(step_size * 1.01, 5.0 / np.sqrt(n))
        else:
            stall += 1
            step_size *= 0.9

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining alternating minimax, peak equalization,
    L-BFGS smoothing, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
        start_a = parent.copy()
    else:
        start_a = _make_chirp(4096, freq=1.0, eps=0.3)
        try_update(start_a, "init_chirp")
        print(f"  No parent, using chirp init")

    n_best = len(start_a)

    # === Phase 2: Alternating Minimax (NEW) ===
    print(f"\n=== Phase 2: Alternating Minimax (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        a_am, sc_am = _alternating_minimax(start_a, deadline, reserve=400)
        try_update(a_am, "alt_minimax")

    # === Phase 3: Peak Equalization (NEW) ===
    print(f"\n=== Phase 3: Peak Equalization (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_pe, sc_pe = _peak_equalization(best_seq, deadline, reserve=350)
        try_update(a_pe, "peak_eq")

    # === Phase 4: Adaptive L-BFGS ===
    print(f"\n=== Phase 4: Adaptive L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_lb = _adaptive_lbfgs(best_seq, deadline, reserve_time=300)
        try_update(a_lb, "lbfgs_adaptive")

    # === Phase 5: Direct Descent ===
    print(f"\n=== Phase 5: Direct Descent (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_dd, sc_dd = _direct_descent(best_seq, deadline - 250)
        try_update(a_dd, "direct_descent")

    # === Phase 6: Peak Equalization Round 2 ===
    print(f"\n=== Phase 6: Peak Equalization R2 (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_pe2, sc_pe2 = _peak_equalization(best_seq, deadline, reserve=200)
        try_update(a_pe2, "peak_eq_r2")

    # === Phase 7: Subgradient Polishing ===
    print(f"\n=== Phase 7: Subgradient Polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 150, k_active=5, max_steps=20000, lr_init=0.04)
        try_update(a_sg, "sg_k5")

    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg2, sc_sg2 = _subgrad_polish(best_seq, deadline - 100, k_active=3, max_steps=15000, lr_init=0.03)
        try_update(a_sg2, "sg_k3")

    # === Phase 8: Basin Hopping ===
    print(f"\n=== Phase 8: Basin Hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 100, rng, n_hops=4, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 9: Micro-polishing ===
    print(f"\n=== Phase 9: Micro-polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_mp, sc_mp = _micro_polish(best_seq, deadline)
        try_update(a_mp, "micro_polish")

    # === Phase 10: Length Exploration ===
    print(f"\n=== Phase 10: Length Exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        lengths_to_try = []
        bn = best_n
        for delta in [1024, 2048, 512, 4096, 256]:
            lengths_to_try.append(bn + delta)
            lengths_to_try.append(bn - delta)
        # Deduplicate and filter
        lengths_to_try = sorted(set(l for l in lengths_to_try if l >= 512 and l <= 32768))

        for target_n in lengths_to_try[:4]:
            if time.monotonic() > deadline - 30:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            # Quick optimization
            if time.monotonic() < deadline - 25:
                c_t = _full_convolve(a_interp, a_interp)
                mt = c_t.max()
                beta = mt * 0.02
                max_iter = min(2000, max(200, int((deadline - 25 - time.monotonic()) / 0.01)))
                a_opt = _lbfgs_smooth(a_interp, beta, target_n, max_iter)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 11: Final aggressive round ===
    print(f"\n=== Phase 11: Final Round (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # One more peak equalization + micro polish cycle
        a_final_pe, _ = _peak_equalization(best_seq, deadline, reserve=10, max_iter=5000)
        try_update(a_final_pe, "final_pe")

        if time.monotonic() < deadline - 5:
            a_final_mp, _ = _micro_polish(best_seq, deadline)
            try_update(a_final_mp, "final_mp")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
