    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_smoothed_max(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta continuation on smoothed max objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)] if len(c_sorted) > 5 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
        0.0075, 0.0037, 0.0018, 0.0009, 0.0004, 0.0002, 0.0001, 0.00005, 0.00002
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.002 * max(1, n / 4096.0)
        max_iter = int(min(10000, max(20, remaining / (est_time_per_iter * 10))))

        def fun_local(zl):
            return _smoobj(zl, beta, n)

        def jac_local(zl):
            return _smograd(zl, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 150, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _gradient_active(a, c, n, tol_frac=0.001):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.01):
    """Subgradient descent on active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.0005)
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _local_search(a_init, deadline, rng, n_trials=2000, sigma_init=0.01):
    """Direct local search: try random perturbations, keep improvements."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    sigma = sigma_init

    for i in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Random perturbation in log-space
        z = np.log(np.maximum(best_a, 1e-30))
        # Perturb a random subset
        n_perturb = max(1, int(n * 0.1))
        idx = rng.integers(0, n, size=n_perturb)
        z[idx] += sigma * rng.standard_normal(n_perturb)
        z = z - z.mean()
        a_try = _softmax(z)
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
            sigma = min(sigma * 1.1, sigma_init)
        else:
            sigma *= 0.98

        if i % 200 == 0:
            print(f"    local_search: iter {i}, score={best_sc:.10f}, sigma={sigma:.6f}")

    return best_a, best_sc


def _newton_active(a_init, deadline, max_iter=100):
    """Newton's method on the active set with CG linear solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient of sum_{k in active} c_k
        grad = _q_s_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        # Newton step: solve (H + mu*I) dx = -grad_t
        mu = 1e-8
        def hess_vec(v):
            return _q_s_matvec(v, active, n) + mu * v

        dx = _cg_solve_qs(active, n, -grad_t, mu=mu, max_iter=500, tol=1e-12)
        dx = dx - np.mean(dx)

        # Line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer combining active-set QP, smoothed max,
    Newton active-set, subgradient, and local search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(4096, "uniform")
        print(f"No parent, using uniform n={len(parent)}")
        update_best(parent, _score(parent), "init")

    n_best = len(best_seq)
    print(f"\nStarting optimization with n={n_best}, score={best_score:.10f}")

    # === Phase 1: Active Set QP (main new approach) ===
    print(f"\n[Phase 1] Active Set QP (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        p1_deadline = min(deadline - 600, time.monotonic() + 500)
        a_qp, sc_qp = _active_set_qp(best_seq, p1_deadline)
        update_best(a_qp, sc_qp, "active_qp")

    # === Phase 2: Newton Active-Set ===
    print(f"\n[Phase 2] Newton Active-Set (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p2_deadline = min(deadline - 400, time.monotonic() + 250)
        a_newt, sc_newt = _newton_active(best_seq, p2_deadline, max_iter=150)
        update_best(a_newt, sc_newt, "newton")

    # === Phase 3: Smoothed Max L-BFGS (very fine) ===
    print(f"\n[Phase 3] Smoothed Max L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_sm = _run_smoothed_max(best_seq, deadline, min_time_remaining=300)
        update_best(a_sm, _score(a_sm), "smoothed_max")

    # === Phase 4: Subgradient Polish ===
    print(f"\n[Phase 4] Subgradient Polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, p4_deadline, n_max_steps=3000, lr_init=0.008)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 5: Local Search ===
    print(f"\n[Phase 5] Local Search (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p5_deadline = min(deadline - 150, time.monotonic() + 100)
        a_ls, sc_ls = _local_search(best_seq, p5_deadline, rng, n_trials=1500, sigma_init=0.005)
        update_best(a_ls, sc_ls, "local_search")

    # === Phase 6: Second pass Active Set QP (tighter) ===
    print(f"\n[Phase 6] Active Set QP Round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        p6_deadline = min(deadline - 100, time.monotonic() + 80)
        tol_tight = [0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        a_qp2, sc_qp2 = _active_set_qp(best_seq, p6_deadline, tol_schedule=tol_tight)
        update_best(a_qp2, sc_qp2, "active_qp2")

    # === Phase 7: Newton on best ===
    print(f"\n[Phase 7] Newton Round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        p7_deadline = min(deadline - 50, time.monotonic() + 40)
        a_n2, sc_n2 = _newton_active(best_seq, p7_deadline, max_iter=80)
        update_best(a_n2, sc_n2, "newton2")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse Restarts (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        for i in range(3):
            if time.monotonic() > deadline - 10:
                break
            a_pert = _perturb(best_seq, 0.02 + 0.01 * i, rng)
            # Quick smoothed max
            a_r = _run_smoothed_max(a_pert, min(deadline - 5, time.monotonic() + 20), min_time_remaining=5)
            update_best(a_r, _score(a_r), f"restart_{i}")

    # === Phase 9: Final active set QP ===
    print(f"\n[Phase 9] Final QP (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        tol_final = [0.002, 0.001, 0.0005, 0.0001]
        a_fin, sc_fin = _active_set_qp(best_seq, deadline - 3, tol_schedule=tol_final)
        update_best(a_fin, sc_fin, "final_qp")

    # === Phase 10: Multi-n exploration ===
    print(f"\n[Phase 10] Multi-n Exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        for n_try in [4096, 16384]:
            if time.monotonic() > deadline - 5:
                break
            if len(best_seq) != n_try:
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_try)
                a_init = np.interp(x_new, x_old, best_seq)
                a_init = np.maximum(a_init, 0)
                a_init = a_init / a_init.sum()
            else:
                a_init = best_seq.copy()

            # Quick active set QP
            a_opt, sc_opt = _active_set_qp(a_init, min(deadline - 3, time.monotonic() + 30),
                                           tol_schedule=[0.05, 0.02, 0.01, 0.005])
            update_best(a_opt, sc_opt, f"n_{n_try}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
