def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

    """Iterative flattening: flatten b=a*a, take conv sqrt, project."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):

            break
        
        b = _full_convolve(a, a)
        b_mean = b.mean()
        
        # Flatten: b_target = (1-lam)*b + lam*b_mean
        b_target = (1.0 - lam) * b + lam * b_mean

        if s_new < 1e-15:
            # Failed, try smaller lambda
            lam *= 0.5

        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Update
        a = a_new
        
        # Adaptive lambda: decrease as we converge
        # Check the "flatness" of b
        b_current = _full_convolve(a, a)
        flatness = b_current.std() / (b_current.mean() + 1e-30)
        
        if it % 10 == 0:
            print(f"    iter {it}: score={best_sc:.10f}, flatness={flatness:.4f}, lam={lam:.4f}")
