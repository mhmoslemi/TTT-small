M @ v = ?
(M @ v)[i] = sum_j M[i,j] * v[j] = sum_j A[i+j] * v[j]

This is a cross-correlation: (M @ v)[i] = sum_j A[i+j] * v[j] = (corr(A, v))[i] (with appropriate definition).

We can compute this in O(n log n) using FFT!

So we can use CG to solve M @ a = b in O(n log n) per iteration, and CG typically converges in O(n) iterations (or fewer for well-conditioned systems).

The key insight for the "active set equalization" is:
- If c[k1] > c[k2] > ... > c[km] are the top m values, we want to reduce the spread.
- The gradient of c[k] with respect to a is: 2 * (a shifted by k) (as derived above).
- So to reduce c[k1] - c[k2], we want to move a in the direction that decreases c[k1] and increases c[k2].
- The direction is: -2 * (a shifted by k1) + 2 * (a shifted by k2) = 2 * ((a shifted by k2) - (a shifted by k1))

This is essentially a "balancing" operation on the convolution values.

def equalize_convolution(a, top_k=20, iterations=100):
    """Try to equalize the top convolution values."""
    n = len(a)
    for it in range(iterations):
        c = conv(a, a)
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_mean = c_top.mean()
        c_spread = c_top.std()
        if c_spread < 1e-10:
            break
        
        # Compute gradient direction: reduce variance of top values
        # grad of (1/m) * sum c[k]^2 with respect to a:
        # = (2/m) * sum_k c[k] * d(c[k])/da = (2/m) * sum_k c[k] * 2 * shift(a, k)
        # = (4/m) * sum_k c[k] * shift(a, k)
        # where shift(a, k)[i] = a[i+k] (if valid)
        
        # This is: (4/m) * cross_corr(c_top_vec, a)
        # where c_top_vec is c with zeros except at top_idx
        
        w = np.zeros(2*n-1)
        w[top_idx] = c[top_idx] - c_mean  # center to maintain sum constraint
        
        # Gradient direction: (4/m) * sum_k w[k] * shift(a, k)
        # = (4/m) * cross_corr(w, a) (with appropriate definition)
        
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] (for valid indices)
        # = sum_k w[k] * a[k-i]
        
        # Using FFT: this is the correlation of w and a.
        N = next_pow2(2*n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        # corr(w, a)[i] = sum_j w[j] * a[j-i] ... hmm need to be careful
        # Let's use: result[i] = sum_j w[j] * a[j-i]
        # = sum_j w[j] * a[j-i]
        # Let b[j] = a[j-i], then result[i] = sum_j w[j] * b[j] = dot(w, b)
        # b is a shifted version of a: b[j] = a[j-i]
        # So result[i] = dot(w, shift(a, -i))
        # = (w * a_rev)[i + n - 1] (convolution)
        
        a_rev = a[::-1]
        out_len = len(w) + len(a_rev) - 1
        Na = next_pow2(out_len)
        fw2 = np.fft.rfft(w, Na)
        fa2 = np.fft.rfft(a_rev, Na)
        result = np.fft.irfft(fw2 * fa2, Na)
        grad = result[n-1:2*n-1]  # n elements
        
        # Project to sum=0
        grad = grad - grad.mean()
        
        # Line search
        best_t = 0
        best_c_spread = c_spread
        for t in [0.01, 0.005, 0.001, 0.0005, 0.0001]:
            a_try = a + t * grad
            a_try = np.maximum(a_try, 0)
            if a_try.sum() < 1e-10:
                continue
            a_try = a_try / a_try.sum()
            c_try = conv(a_try, a_try)
            top_try = np.argsort(c_try)[::-1][:top_k]
            spread_try = c_try[top_try].std()
            if spread_try < best_c_spread:
                best_t = t
                best_c_spread = spread_try
                best_a = a_try.copy()
        
        if best_t == 0:
            break
        a = best_a
    
    return a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) diverse structural initializations, (2) L-BFGS with log-sum-exp,
(3) spectral flattening iterations, (4) active-set equalization, (5) coordinate
descent polish. Explores multiple n values and selects the best candidate.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "exp_decay":
        t = np.arange(n) / max(n - 1, 1)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    elif kind == "exp_decay_rev":
        t = np.arange(n) / max(n - 1, 1)
        a = np.exp(-2.0 * (1 - t))
        return a / a.sum()
    elif kind == "half":
        a = np.zeros(n)
        a[:n // 2] = 1.0
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / 0.25) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.35) / 0.2) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t
        return a / a.sum()
    elif kind == "inv_ramp":
        t = np.linspace(0, 1, n)
        a = 1.5 - t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
                                  0.015, 0.0075, 0.004, 0.002, 0.001,
                                  0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.008 * max(1, n / 1024.0)
        max_iter = min(15000, max(50, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_flatten(a_init, deadline, n_iters=500, min_time_remaining=20.0):
    """Spectral flattening: iteratively flatten the DFT magnitude and project to non-negative."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break

        fa = np.fft.rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Compress the magnitude spectrum
        # Target: constant magnitude (flat spectrum)
        target_mag = np.full_like(mag, mag.mean())
        # Blend: keep some of the original structure
        alpha = 0.3 + 0.5 * (it / min(n_iters, 100))
        new_mag = alpha * target_mag + (1 - alpha) * mag
        # Handle DC
        new_mag[0] = mag[0]

        new_fa = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_fa, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def _conv_equalize(a_init, deadline, min_time_remaining=20.0):
    """Directly equalize the top convolution values using gradient steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    top_k = 50
    lr = 0.01
    stall = 0

    for it in range(5000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_mean = c_top.mean()
        c_spread = c_top.max() - c_top.min()

        if c_spread < 1e-12 * max(c.max(), 1e-10):
            break

        # Gradient of (1/2) * sum_{k in top} (c[k] - c_mean)^2 w.r.t. a
        # = sum_k (c[k] - c_mean) * dc[k]/da
        # dc[k]/da[i] = 2*a[k-i] (if 0 <= k-i < n)
        # So grad[i] = 2 * sum_k (c[k] - c_mean) * a[k-i]
        # This is 2 * cross_correlation of w with a, where w[k] = c[k]-c_mean for k in top

        w = np.zeros(2 * n - 1)
        w[top_idx] = c[top_idx] - c_mean

        # grad[i] = 2 * sum_{k in top} w[k] * a[k-i] (for valid k-i)
        # = 2 * (w_conv_a)[i] where w_conv_a[i] = sum_k w[k] * a[k-i]
        # Use FFT: this is correlation of w and a
        N = _next_pow2(3 * n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        # corr(w, a)[i] = sum_j w[j] * a[j-i]
        # = ifft(fft(w) * conj(fft(a))) shifted appropriately
        # Actually: sum_j w[j] * a[j-i] = (w * a_rev)[i + n - 1]
        a_rev = np.zeros(N)
        a_rev[:n] = a[::-1]
        fa_rev = np.fft.rfft(a_rev, N)
        result = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * result[n - 1:2 * n - 1]

        # Project to sum=0
        grad = grad - grad.mean()

        # Line search
        f_old = float(c.max())
        accepted = False
        t = lr
        for ls in range(30):
            a_try = a + t * grad
            if np.min(a_try) < -1e-12:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if accepted:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                stall = 0
                lr = min(lr * 1.1, 0.05)
            else:
                stall += 1
        else:
            stall += 1

        if stall > 50:
            lr *= 0.7
            stall = 0
            if lr < 1e-6:
                break

    return best_a


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    """Damped Newton on identified active set for final precision."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:3 * n]
            active[top_idx] = True
            n_active = 3 * n

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of (1/m) * sum_{k in active} c[k]
        # grad[i] = (2/m) * sum_j a[j] * A_vec[i+j]
        # = (2/m) * (a_rev * A_vec)[i + n - 1]
        flip_a = a[::-1]
        N = _next_pow2(3 * n)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)
        out_len = n + len(A_vec) - 1
        g = (2.0 / m) * result[n - 1:2 * n - 1]

        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-8

        def hess_matvec(v):
            N2 = _next_pow2(3 * n)
            fv = np.fft.rfft(v[::-1], N2)
            fA2 = np.fft.rfft(A_vec, N2)
            res = np.fft.irfft(fv * fA2, N2)
            return (2.0 / m) * res[n - 1:2 * n - 1] + mu * v

        # CG solve
        def cg_solve(rhs, max_cg=100):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w_vec = cg_solve(np.ones(n))
        sum_w = float(w_vec.sum())
        if abs(sum_w) < 1e-15:
            dx = -u - np.mean(-u)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_vec
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            improved = False
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13:
                    a = a_try
                    improved = True
                    break
                t *= 0.5
            if not improved:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, min_time_remaining=15.0):
    """Coordinate descent: optimize one or a few coordinates at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    rng = np.random.default_rng(12345)
    c = _full_convolve(a, a)

    for it in range(10000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Pick a random coordinate (or a small block)
        if n <= 4096:
            block_size = 1
        else:
            block_size = rng.integers(1, 8)

        start = rng.integers(0, max(1, n - block_size + 1))
        idx = np.arange(start, min(n, start + block_size))

        # For each coordinate in the block, find optimal value
        # c[k] = sum_i a[i]*a[k-i], so changing a[j] affects c[k] for k = j + i for all i
        # This is complex. Instead, use a small gradient step on the block.

        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.002 * c_max
        active = c >= (c_max - tol)

        # Gradient at selected indices
        flip_a = a[::-1]
        A_vec = active.astype(np.float64)
        N = _next_pow2(3 * n)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)
        g = (2.0 / max(active.sum(), 1)) * result[n - 1:2 * n - 1]
        g_t = g - np.mean(g)

        # Small step in negative gradient direction on the block
        lr = 0.01
        for _ in range(5):
            a_try = a.copy()
            a_try[idx] = a_try[idx] - lr * g_t[idx]
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < _score(a) - 1e-14:
                a = a_try
                break
            lr *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()

    return best_a


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick exploration of different n and initializations
    print(f"\n[Phase 1] Exploration (t={time.monotonic()-t_start:.1f}s)")
    n_explore = [2048, 4096, 8192, 16384]
    initializations = ["uniform", "exp_decay", "gaussian", "asym_gauss", "cos_taper", "half"]

    candidates = []
    for n_try in n_explore:
        if time.monotonic() > deadline - 600:
            break
        for kind in initializations:
            if time.monotonic() > deadline - 500:
                break
            a_init = _make_profile(n_try, kind)
            sc_init = _score(a_init)
            # Quick L-BFGS (short budget for exploration)
            time_budget = min(30.0, (deadline - time.monotonic() - 300) / max(1, len(initializations)))
            if time_budget < 5:
                continue
            a_opt = _run_lbfgs(a_init, time.monotonic() + time_budget + 600, min_time_remaining=500)
            sc_opt = _score(a_opt)
            candidates.append((sc_opt, a_opt, f"n{n_try}_{kind}"))
            update_best(a_opt, sc_opt, f"explore_n{n_try}_{kind}")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidates: {[(f'{s:.8f}', l) for s, _, l in candidates[:5]]}")

    # Phase 2: Deep optimization on best candidates
    print(f"\n[Phase 2] Deep optimization (t={time.monotonic()-t_start:.1f}s)")

    # Use parent as primary if it's good
    primary = parent
    primary_sc = _score(primary)
    if candidates and candidates[0][0] < primary_sc:
        primary = candidates[0][1]
        primary_sc = candidates[0][0]

    # Spectral flattening
    if time.monotonic() < deadline - 200:
        a_spec = _spectral_flatten(primary, deadline, n_iters=200, min_time_remaining=200)
        update_best(a_spec, _score(a_spec), "spectral")

    # Update primary if spectral helped
    if best_seq is not None:
        primary = best_seq.copy()

    # Convolution equalization
    if time.monotonic() < deadline - 180:
        a_eq = _conv_equalize(primary, deadline, min_time_remaining=180)
        update_best(a_eq, _score(a_eq), "equalize")

    if best_seq is not None:
        primary = best_seq.copy()

    # Full L-BFGS with aggressive beta
    if time.monotonic() < deadline - 100:
        a_lbfgs = _run_lbfgs(primary, deadline, min_time_remaining=100)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_deep")

    if best_seq is not None:
        primary = best_seq.copy()

    # Phase 3: Active-set Newton refinement
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 60, time.monotonic() + 120),
                                         max_iter=300, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton")

        if time.monotonic() < deadline - 50:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 40, time.monotonic() + 60),
                                               max_iter=200, tol_active=0.0005)
            update_best(a_as2, sc_as2, "as_newton_tight")

    if best_seq is not None:
        primary = best_seq.copy()

    # Phase 4: Coordinate descent polish
    print(f"\n[Phase 4] Coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        a_cd = _coordinate_descent(primary, deadline, min_time_remaining=40)
        update_best(a_cd, _score(a_cd), "coord_desc")

    # Phase 5: Perturbation restarts
    print(f"\n[Phase 5] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 20:
                break
            p_pert = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_pert, deadline, min_time_remaining=20)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"perturb_{sigma}")
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 10, time.monotonic() + 30),
                                                  max_iter=100, tol_active=0.001)
                update_best(a_r2, sc_r2, f"perturb_{sigma}_as")

    # Phase 6: Final refinement
    print(f"\n[Phase 6] Final refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active set
        a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 2, time.monotonic() + 60),
                                           max_iter=200, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 2:
                max_iter = min(500, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 50}
                    )
                    a_fin2 = _softmax(result.x)
                    update_best(a_fin2, _score(a_fin2), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
