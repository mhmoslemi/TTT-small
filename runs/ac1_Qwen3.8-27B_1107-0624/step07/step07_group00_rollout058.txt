"""
Optimizer using exponentiated gradient with exact subgradient,
Nesterov acceleration, and exact coordinate descent polishing.
Exploits the structure of the autocorrelation peak minimization problem.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume sum(a)=1, compute 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _subgradient(a, k_indices):
    """
    Compute subgradient of max_{k in k_indices} conv(a,a)[k] w.r.t. a.
    For a single position k, the gradient at position i is:
      2*a[k-i] if 0 <= k-i <= n-1 and k != 2*i
      2*a[i]   if 2*i == k
      0        otherwise
    For multiple positions, sum the contributions.
    """
    n = len(a)
    g = np.zeros(n)
    for k in k_indices:
        # grad_k[i] = 2 * a[k-i] for valid i
        # This is a flipped and shifted copy of a
        if 0 <= k < 2 * n - 1:
            # valid range of i: max(0, k-(n-1)) <= i <= min(k, n-1)
            i_lo = max(0, k - (n - 1))
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                if 2 * i == k:
                    g[i] += 2.0 * a[i]
                else:
                    g[i] += 2.0 * a[j]
    return g

def _subgradient_fast(a, k):
    """Fast version for a single k using array operations."""
    n = len(a)
    g = np.zeros(n)
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    if i_lo > i_hi:
        return g
    # For i in [i_lo, i_hi], j = k - i is in [0, n-1]
    # grad[i] = 2*a[k-i] (with special case 2i==k giving 2*a[i])
    js = k - np.arange(i_lo, i_hi + 1)
    g[i_lo:i_hi+1] = 2.0 * a[js]
    # Fix the center if k is even
    if k % 2 == 0:
        i_center = k // 2
        if i_lo <= i_center <= i_hi:
            g[i_center] = 2.0 * a[i_center]
    return g

def _expgd_nesterov(a_init, deadline, eta_start=0.1, eta_end=0.001,
                     top_k=5, momentum_start=0.0, momentum_end=0.9,
                     eval_interval=100):
    """
    Exponentiated gradient with Nesterov-style momentum.
    Works in the simplex (sum(a)=1, a>=0).
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()  # lookahead point
    t = 1
    best_a = a.copy()
    best_score = _score_norm(a)
    best_t = 1
    
    while time.monotonic() < deadline:
        frac = min(1.0, t / 50000)
        eta = eta_start * (eta_end / eta_start) ** frac
        mu = momentum_start + (momentum_end - momentum_start) * frac
        
        # Compute convolution at y
        c_y = _fft_convolve(y, y)
        c_max = float(c_y.max())
        
        # Find active set: top-k positions
        k_indices = np.argsort(c_y)[::-1][:top_k]
        
        # Compute subgradient at y
        g = np.zeros(n)
        for k in k_indices:
            g += _subgradient_fast(y, int(k))
        
        # Normalize gradient for stability
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Nesterov update
        # a_new = y - eta * g, project to simplex
        # Then y = a_new + momentum * (a_new - a)
        a_candidate = y - eta * g
        # Project to simplex
        a_new = _project_simplex(a_candidate)
        
        # Update lookahead
        y = a_new + mu * (a_new - a)
        y = _project_simplex(y)
        
        a = a_new
        t += 1
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_t = t
    
    final_sc = _score_norm(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    
    return best_a, best_score

def _project_simplex(v):
    """Project v onto the simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _multiplicative_weight(a_init, deadline, eta_start=0.05, eta_end=0.001,
                          top_k=5, eval_interval=200):
    """
    Pure multiplicative weight (exponentiated gradient) method.
    a_j <- a_j * exp(-eta * g_j), then normalize.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30) / a.sum()
    z = np.log(a)  # work in log space
    best_a = a.copy()
    best_score = _score_norm(a)
    t = 1
    
    while time.monotonic() < deadline:
        frac = min(1.0, t / 50000)
        eta = eta_start * (eta_end / eta_start) ** frac
        
        c = _fft_convolve(a, a)
        k_indices = np.argsort(c)[::-1][:top_k]
        
        g = np.zeros(n)
        for k in k_indices:
            g += _subgradient_fast(a, int(k))
        
        # Multiplicative update in log space
        z = z - eta * g
        # Clip for numerical stability
        z = np.clip(z, -50, 50)
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        t += 1
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    final_sc = _score_norm(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    
    return best_a, best_score

    while round_num < max_rounds and time.monotonic() < deadline:
        round_num += 1
        improved = False

        # Coordinate j affects conv positions from j to j+n-1 (i.e., k where 0<=k-j<=n-1, so j<=k<=j+n-1)

            # But also conv_new[k] = conv_old[k] + (a_j' - a_j) * a_{k-j} for the symmetric term

            old_val = a[j]

            # We can find the minimum by bisection on the subgradient

            # conv[k](t) = conv_old[k] + (t - old_val) * (a_{k-j} if 0<=k-j<=n-1) + (t - old_val) * (a_{k-j} if k-j != j and 0<=k-j<=n-1)

            # conv[k] = sum_i a_i * a_{k-i}

            # conv_new[k] = conv_old[k] - old_val*a_{k-j} + t*a_{k-j} (from i=j term)
            #              + (if k-j != j and 0<=k-j<=n-1: -old_val*a_{k-j} + t*a_{k-j})

            #   i = k-j: a_{k-j} * a_j (valid if 0 <= k-j <= n-1 AND k-j != j, i.e., 2j != k)

            #   c_jk = a_{k-j} if 0<=k-j<=n-1)
            #   if k-j != j and 0<=k-j<=n-1: c_jk += a_{k-j}

            # If k-j != j, the two terms are a_j*a_{k-j} and a_{k-j}*a_j,
            # and d/d(a_j) = a_{k-j} + a_{k-j} = 2*a_{k-j}.

            # Where 0<=k-j is NOT in [0,n-1], the coefficient is 0.

                new_max = max(float(new_affected.max()), unaffected_max)

                if new_max < best_score / (2.0 * n) * (1.0 - 1e-12):
                    # Good improvement, move lo up
                    lo = mid
                else:
                    # Check slope: if increasing t increases max, move hi down
                    # Compute subgradient sign at mid
                    ks_for_grad = np.arange(j, min(j + n, 2*n-1))
                    new_vals = c[j:min(j+n, 2*n-1)] + delta * 2.0 * a[js_valid[:len(ks)]]
                    idx_max_affected = np.argmax(new_vals)
                    k_star = ks_for_grad[idx_max_affected]
                    slide = int(k_star) - j
                    if 0 <= slide < n:
                        slope = 2.0 * a[slide]
                    else:
                        slope = 0.0
                    if slope > 0:
                        hi = mid
                    else:
                        lo = mid
            
            optimal_t = 0.5 * (lo + hi)
            if optimal_t > 0 and optimal_t != old_val:
                a[j] = optimal_t
                sc = _score_norm(a)
                if sc < best_score * (1.0 - 1e-12):
                    best_score = sc
                    best_a = a.copy()
                    c = _fft_convolve(a, a)
                    c_max = float(c.max())
                    active_k = np.where(c > c_max * 0.995)[0]
                    improved = True
        
        if not improved:
            # Renormalize
            a = a / a.sum()
            break
    
    a = a / a.sum()
    return a, _score_norm(a)

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

def _spectral_init(n, rng):
    """
    Initialize with a sequence that has a flat-ish power spectrum.
    Use random phases in frequency domain, then take IFFT and clamp.
    """
    # Target: |A(w)|^2 should be as flat as possible
    # A(w) = sum_j a_j * exp(-i*j*w)
    # If we want |A(w)|^2 = const, then A(w) = sqrt(const) * exp(i*phi(w))
    # But A(w) must be a polynomial in exp(-iw) (i.e., finite Fourier series)
    # So we can't have exactly flat spectrum, but we can try.
    
    # Approach: random phases
    N = _next_pow2(n)
    phi = rng.standard_normal(N // 2 + 1) * np.pi  # random phases
    A = np.ones(N // 2 + 1) * np.exp(1j * phi)  # unit magnitude
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0)  # clamp to non-negative
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _cosine_init(n, freq, rng):
    """Initialize with a cosine-like pattern with random offset."""
    t = np.arange(n)
    a = np.cos(2 * np.pi * freq * t / n) + 1.0
    a = a + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _adam_z_space_v2(a_init, deadline, top_k=3, lr_start=0.03, lr_end=0.001,
                     eval_interval=200, use_exact_grad=True):
    """
    Improved Adam in log-space with exact (or weighted) subgradient.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    
    while time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / 100000)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Weighted active set gradient
        c_max = float(c.max())
        weights = np.zeros(len(c))
        top_indices = np.argsort(c)[::-1][:top_k]
        for i, idx in enumerate(top_indices):
            w = (c[idx] / c_max) ** 10  # sharp weight
            weights[int(idx)] = w
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights[top_indices[0]] = 1.0
            w_sum = 1.0
        weights = weights / w_sum
        
        # Compute gradient as weighted sum of subgradients
        g = np.zeros(n)
        for idx in top_indices:
            if weights[int(idx)] > 0.01:
                g += weights[int(idx)] * _subgradient_fast(a, int(idx))
        
        # Convert to z-space gradient
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

**Iterative thresholding / water-filling in the frequency domain**:
1. Compute the DFT of `a`: `A[k] = sum_j a_j * exp(-2πijk/n)`
2. The autocorrelation is the inverse DFT of `|A[k]|²`.
3. We want to minimize the max of the inverse DFT of `|A[k]|²`.

5. But `|A[k]|²` is not freely choosable - it's constrained by the structure of `a` (real, non-negative, finite support).

1. In the frequency domain, project `|A[k]|²` toward a flat spectrum.

    """
    Gerchberg-Saxton style alternating projection.
    Time domain: enforce non-negativity and sum=1.
    Frequency domain: flatten |A[k]|^2.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()

            break

        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2

        # We want power to be as close to its mean as possible
        power_mean = power.mean()
        # Compress toward mean
        alpha = 0.1  # mixing parameter

        phase = np.angle(A)

        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a = a_new
        
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam in log-space with exact weighted active-set subgradient
2. Fast coordinate descent with incremental convolution updates
3. Multiplicative weight (exponentiated gradient) with Nesterov momentum
4. Multi-scale exploration and diverse initializations
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgrad_single(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a. Returns vector of length n."""
    n = len(a)
    g = np.zeros(n)
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    if i_lo > i_hi:
        return g
    js = k - np.arange(i_lo, i_hi + 1)
    g[i_lo:i_hi + 1] = 2.0 * a[js]
    if k % 2 == 0:
        ic = k // 2
        if i_lo <= ic <= i_hi:
            g[ic] = 2.0 * a[ic]
    return g


def _weighted_subgrad(a, c, top_k=5, sharpness=8):
    """Weighted sum of subgradients at top-k positions."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    top_idx = np.argsort(c)[::-1][:top_k]
    weights = (c[top_idx] / c_max) ** sharpness
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    weights = weights / w_sum
    g = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if weights[i] > 0.01:
            g += weights[i] * _subgrad_single(a, int(idx))
    return g


def _adam_exact(a_init, deadline, top_k=3, sharpness=8,
                lr_start=0.03, lr_end=0.0005, eval_interval=100):
    """Adam in log-space with exact weighted subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / 200000)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _weighted_subgrad(a, c, top_k, sharpness)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.03:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent_fast(a_init, deadline, max_rounds=30):
    """Coordinate descent with incremental convolution updates and bisection."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    c = _fft_convolve(a, a)
    c_len = 2 * n - 1
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    round_num = 0
    while round_num < max_rounds and time.monotonic() < deadline:
        round_num += 1
        c_max = float(c.max())
        improved = False
        threshold = c_max * 0.99

        # Find coordinates that affect near-max positions
        active_k = np.where(c > threshold)[0]
        coord_set = set()
        for k in active_k:
            ki = int(k)
            j_lo = max(0, ki - (n - 1))
            j_hi = min(ki, n - 1)
            for j in range(j_lo, j_hi + 1):
                coord_set.add(j)

        coords = sorted(coord_set)
        for j in coords:
            old_val = a[j]
            if old_val < 1e-15:
                continue

            k_lo = j
            k_hi = min(j + n - 1, c_len - 1)
            coeff_len = k_hi - k_lo + 1
            if coeff_len <= 0:
                continue
            coeffs = 2.0 * a[:coeff_len]

            lo = 0.0
            hi = 3.0 * old_val

            for _ in range(12):
                mid = 0.5 * (lo + hi)
                delta = mid - old_val
                new_aff = c[k_lo:k_hi + 1] + delta * coeffs
                new_max_aff = float(new_aff.max())
                if k_lo > 0:
                    unaff_max = float(max(c[:k_lo].max(), c[k_hi + 1:].max())) if k_hi + 1 < c_len else float(c[:k_lo].max())
                elif k_hi + 1 < c_len:
                    unaff_max = float(c[k_hi + 1:].max())
                else:
                    unaff_max = 0.0
                total_max = max(new_max_aff, unaff_max)

                target = best_score / (2.0 * n)
                if total_max < target * (1.0 - 1e-10):
                    lo = mid
                else:
                    idx_m = int(np.argmax(new_aff))
                    k_star = k_lo + idx_m
                    slide = k_star - j
                    slope = 2.0 * a[slide] if 0 <= slide < n else 0.0
                    if slope > 0:
                        hi = mid
                    else:
                        lo = mid

            optimal_t = 0.5 * (lo + hi)
            if abs(optimal_t - old_val) > 1e-12 * max(old_val, 1e-15):
                delta = optimal_t - old_val
                a[j] = optimal_t
                c[k_lo:k_hi + 1] += delta * coeffs
                new_sc = float(2.0 * n * c.max())
                if new_sc < best_score * (1.0 - 1e-12):
                    best_score = new_sc
                    best_a = a.copy()
                    improved = True

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        c = _fft_convolve(a, a)

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _expgd_nesterov(a_init, deadline, eta_start=0.08, eta_end=0.001,
                    top_k=5, sharpness=6, eval_interval=200):
    """Exponentiated gradient (multiplicative weight) with Nesterov momentum."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).astype(np.float64)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    mu = 0.0
    t = 1
    best_a = a.copy()
    best_score = _score_norm(a)
    a_prev = a.copy()

    while time.monotonic() < deadline:
        frac = min(1.0, t / 50000)
        eta = eta_start * (eta_end / eta_start) ** frac
        mu_t = 0.5 * frac

        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        g = _weighted_subgrad(a_cur, c, top_k, sharpness)

        W = float(np.dot(a_cur, g))
        z_grad = g - W

        z_new = z - eta * z_grad
        z_new = np.clip(z_new, -50.0, 50.0)
        a_new = _softmax_stable(z_new)

        # Nesterov: blend
        y = a_new + mu_t * (a_new - a_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()
        z = np.log(y)
        z = z - z.mean()

        a_prev = a_cur
        a_cur = y
        t += 1

        if t % eval_interval == 0:
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()

    final_sc = _score_norm(a_cur)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a_cur.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _diverse_init(n, rng, method=0):
    if method == 0:
        return np.ones(n) / n
    elif method == 1:
        a = rng.standard_normal(n) + 2.0
        a = np.maximum(a, 0)
        return a / a.sum()
    elif method == 2:
        t = np.arange(n)
        freq = rng.uniform(0.1, 0.8)
        a = np.cos(2 * np.pi * freq * t / n) + 1.5
        a = np.maximum(a, 0)
        return a / a.sum()
    elif method == 3:
        N = _next_pow2(n)
        phi = rng.standard_normal(N // 2 + 1) * np.pi
        A = np.exp(1j * phi)
        a = np.fft.irfft(A, N)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    else:
        a = np.exp(-np.linspace(0, 3, n))
        a = a + rng.normal(0, 0.05, n)
        a = np.maximum(a, 0)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Adam with exact subgradient on parent (0-250s)
    Phase 2: Exponentiated gradient with Nesterov (250-400s)
    Phase 3: Fast coordinate descent polishing (400-550s)
    Phase 4: Multi-n exploration (550-900s)
    Phase 5: Diverse initializations (900-1100s)
    Phase 6: Intensive final polish (1100-1600s)
    Phase 7: Validation and return (1600-1800s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_parent = parent / parent.sum()
        else:
            raise ValueError("bad parent")
    except Exception:
        a_parent = np.ones(8192) / 8192
        print("Using uniform init")

    best_a = a_parent.copy()
    best_score = _score(a_parent)
    print(f"Initial: n={len(a_parent)}, score={best_score:.10f}")

    # Phase 1: Adam exact on parent
    p1_end = min(deadline, t_start + 250.0)
    if time.monotonic() < p1_end:
        a1, sc1 = _adam_exact(a_parent, p1_end, top_k=3, sharpness=8,
                              lr_start=0.03, lr_end=0.0005, eval_interval=100)
        full1 = _score(a1)
        print(f"P1 Adam: {full1:.10f} t={time.monotonic()-t_start:.0f}s")
        if full1 < best_score:
            best_score = full1
            best_a = a1.copy()

    # Phase 2: Exponentiated gradient + Nesterov
    p2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < p2_end:
        a2, sc2 = _expgd_nesterov(best_a, p2_end, eta_start=0.06, eta_end=0.001,
                                  top_k=5, sharpness=6, eval_interval=200)
        full2 = _score(a2)
        print(f"P2 ExpGD: {full2:.10f} t={time.monotonic()-t_start:.0f}s")
        if full2 < best_score:
            best_score = full2
            best_a = a2.copy()

    # Phase 3: Coordinate descent
    p3_end = min(deadline, t_start + 550.0)
    if time.monotonic() < p3_end:
        a3, sc3 = _coord_descent_fast(best_a, p3_end, max_rounds=20)
        full3 = _score(a3)
        print(f"P3 CD: {full3:.10f} t={time.monotonic()-t_start:.0f}s")
        if full3 < best_score:
            best_score = full3
            best_a = a3.copy()

    # Phase 4: Multi-n exploration
    p4_end = min(deadline, t_start + 900.0)
    n_parent = len(best_a)
    n_options = [2048, 4096, 16384]
    if n_parent in n_options:
        n_options.remove(n_parent)
    for n_try in n_options:
        if time.monotonic() > p4_end - 60:
            break
        a_interp = _interp_to_length(best_a, n_try)
        sub_dl = min(p4_end - 20, time.monotonic() + 100.0)
        a_on, sc_on = _adam_exact(a_interp, sub_dl, top_k=3, sharpness=8,
                                  lr_start=0.02, lr_end=0.001, eval_interval=150)
        full_on = _score(a_on)
        print(f"P4 n={n_try}: {full_on:.10f} t={time.monotonic()-t_start:.0f}s")
        if full_on < best_score:
            best_score = full_on
            best_a = a_on.copy()
        # Quick CD at this n
        if full_on < best_score + 0.01 and time.monotonic() < p4_end - 30:
            a_on_cd, _ = _coord_descent_fast(a_on, min(p4_end - 10, time.monotonic() + 50.0),
                                             max_rounds=5)
            full_cd = _score(a_on_cd)
            if full_cd < best_score:
                best_score = full_cd
                best_a = a_on_cd.copy()
