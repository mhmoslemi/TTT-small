"""
Optimized step function search using Gerchberg-Saxton spectral initialization,
active-set KKT analysis, antisymmetric perturbation, and L-BFGS polishing.
Key new ideas:
1. GS (Gerchberg-Saxton) algorithm for spectrally-flat initialization

4. Adaptive multi-scale L-BFGS with aggressive beta continuation
5. Population-based diversity maintenance

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _ks_init(n, n_iters=200, seed=42):
    """
    Gerchberg-Saxton style initialization.
    Alternates between:
    - Time domain: enforce non-negativity and sum=1
    - Frequency domain: enforce flat power spectrum
    """
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)
    
    # Initialize with random phases
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0  # DC = 1
    # Target magnitude: by Parseval, sum(a^2) = (1/N)*sum|A|^2
    # We want sum(a^2) ~ 1/n, so (1/N)*sum|A|^2 ~ 1/n
    # => sum|A|^2 ~ N/n
    # A[0]^2 = 1, so sum_{k>0} |A[k]|^2 ~ N/n - 1
    target_energy = N / n - 1
    C = np.sqrt(target_energy / (N - 1))
    
    for k in range(1, N):
        A[k] = C * np.exp(2j * np.pi * rng.random())
    
    best_a = None
    best_sc = np.inf
    
    for it in range(n_iters):
        # Time domain
        a_full = np.fft.ifft(A).real
        a = a_full[:n].copy()
        
        # Enforce non-negativity
        a = np.maximum(a, 0)
        
        # Enforce sum = 1
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Score check
        if it % 20 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
        # Frequency domain
        A = np.fft.fft(a, N)
        # Enforce DC
        A[0] = abs(A[0])  # Make it real and positive (= sum(a) = 1)
        # Enforce flat magnitude (soft)
        mags = np.abs(A[1:])
        mean_mag = mags.mean()
        if mean_mag < 1e-30:
            break
        # Soft enforcement: shrink towards target magnitude
        A[1:] = A[1:] * (C / mags)
        # Or: A[1:] = A[1:] * ((mags + C) / (mags + C)) ... just set to C
        # Actually, let's use a softer version:
        # A[1:] has phase preserved, magnitude set to C
        
    if best_a is None:
        best_a = np.ones(n) / n
    return best_a


def _ks_init_v2(n, n_iters=300, seed=42, relax=0.5):
    """
    Improved GS with relaxed spectrum constraint.
    Instead of setting |A[k]| = C exactly, we blend: |A[k]| = (1-relax)*|A[k]| + relax*C
    This allows the algorithm to converge to a better local optimum.
    """
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)
    
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0
    target_energy = N / n - 1
    C = np.sqrt(max(target_energy, 0.01) / (N - 1))
    
    for k in range(1, N):
        A[k] = C * np.exp(2j * np.pi * rng.random())
    
    best_a = None
    best_sc = np.inf
    
    for it in range(n_iters):
        a_full = np.fft.ifft(A).real
        a = a_full[:n].copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
        A = np.fft.fft(a, N)
        A[0] = 1.0  # Force DC = 1
        mags = np.abs(A[1:])
        safe_mags = np.maximum(mags, 1e-30)
        # Relaxed flat spectrum
        new_mags = (1 - relax) * mags + relax * C
        A[1:] = A[1:] * (new_mags / safe_mags)
    
    if best_a is None:
        best_a = np.ones(n) / n
    return best_a


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time * 10))))
        
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3):
    """Improved pressure equalization with adaptive active set."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        # Adaptive active set
        tol = max(0.005 * c_max, (c_max - c.min()) * 0.05)
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 5
        
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _antisymmetric_perturb(a, n_trials=50, seed=42):
    """
    Try adding antisymmetric perturbations to reduce the central peak.
    c[n-1] = ||s||^2 - ||d||^2 where a = s + d (s symmetric, d antisymmetric)
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Current antisymmetric component
    a_sym = (a + a[::-1]) / 2
    a_anti = (a - a[::-1]) / 2
    
    for trial in range(n_trials):
        # Random antisymmetric perturbation
        d = np.zeros(n)
        half_n = n // 2
        d[:half_n] = rng.standard_normal(half_n) * 0.1
        d[half_n:] = -d[:n-half_n][::-1] if n % 2 == 0 else -d[half_n-1::-1]
        if n % 2 == 1:
            d[half_n] = 0
        
        # Try different step sizes
        for step in [0.01, 0.05, 0.1, 0.2, 0.3]:
            a_try = a + step * d
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
    
    return best_a, best_sc


def _kkt_check_and_refine(a, n_iters=100, deadline=None):
    """
    Use KKT conditions to identify the active set and refine.
    At optimum: for active k, (grad c_k)_i = lambda_sum for all i with a_i > 0.
    grad c_k / da_i = a[i+k] + a[i-k] (where valid)
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: where c is close to max
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        
        if active.sum() == 0:
            idx = np.argmax(c)
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        
        # Compute the "pressure" for each coordinate
        # For each active k, contribution to coord i is a[i+k] + a[i-k]
        # Total pressure: sum_{k in active} (a[i+k] + a[i-k])
        a_full = np.zeros(2 * n - 1)
        a_full[n-1:n-1+n] = a  # Center a in the full convolution array
        
        # Actually, let's compute directly
        # p[i] = sum_{k in active} (a[(i+k) % n] + a[(i-k) % n]) -- no, this is circular
        # For linear: p[i] = sum_{k in active, 0<=i+k<n} a[i+k] + sum_{k in active, 0<=i-k<n} a[i-k]
        
        # Use convolution: p = conv(active, a)[n-1:2n-1] + conv(active_rev, a)[...]
        # Actually: (grad c_k)_i = a[i+k] + a[i-k] (linear, with bounds)
        # sum_k active[k] * (a[i+k] + a[i-k])
        # = (active * a_reversed)[n-1+i] ... this is getting complicated
        # Let's just use the formula directly with FFT
        
        active_f = np.fft.rfft(active.astype(float), 2*n-1)
        a_f = np.fft.rfft(a, 2*n-1)
        # conv(active, a)[m] = sum_k active[k] * a[m-k]
        ca = np.fft.irfft(active_f * a_f, 2*n-1)
        # ca[m] = sum_k active[k] * a[m-k]
        # We want p[i] = sum_k active[k] * (a[i+k] + a[i-k])
        # = sum_k active[k] * a[i+k] + sum_k active[k] * a[i-k]
        # First term: sum_k active[k] * a[i+k] = (active_rev * a)[2(n-1)-i] ... 
        # Let's just compute it directly for small n or use the correlation
        
        # Simpler: p[i] = (a * active_rev)[i + n - 1] where active_rev = active[::-1]
        # Hmm, let me just use the full convolve approach
        active_float = active.astype(float)
        p = _full_convolve(a, active_float[::-1])
        p = p[n-1:2*n-1]  # This gives sum_k a[i+k] * active_rev[k] = sum_k a[i+k] * active[n-1-k]
        
        # Actually this is getting too complicated. Let's use a simpler approach:
        # Just use the subgradient from the full weight vector
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        grad = _gradient_active(a, c, n)
        
        # KKT: at optimum, grad should be constant on the support of a (positive coords)
        # supported coords
        support = a > 1e-10
        if support.sum() < 2:
            break
        
        # Check: is grad approximately constant on support?
        g_support = grad[support]
        g_mean = g_support.mean()
        g_std = g_support.std()
        
        if g_std < 1e-6 * abs(g_mean):
            # We're at (approximately) a KKT point
            break
        
        # Newton-like step: adjust a to make grad more uniform on support
        # a_i <- a_i * exp(-eta * (grad_i - g_mean))
        eta = 0.1
        a[support] = a[support] * np.exp(-eta * (grad[support] - g_mean))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        t = np.linspace(0, 1, n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1 - np.abs(2*t - 1)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    # Stage 1: L-BFGS
    if time.monotonic() < deadline - min_remaining - 20:
        a1 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 20)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1
    
    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 10:
        a2, sc2 = _pressure_equalize_v2(current_a, min(deadline - min_remaining - 5, time.monotonic() + 60),
                                        n_iters=200, gamma=0.25)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2
    
    # Stage 3: Antisymmetric perturbation
    if time.monotonic() < deadline - min_remaining - 5:
        a3, sc3 = _antisymmetric_perturb(current_a, n_trials=30, seed=len(current_a))
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3
    
    # Stage 4: L-BFGS again
    if time.monotonic() < deadline - min_remaining:
        a4 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining)
        sc4 = _score(a4)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4
    
    # Stage 5: KKT refinement
    if time.monotonic() < deadline - min_remaining + 5:
        a5, sc5 = _kkt_check_and_refine(current_a, n_iters=50, deadline=deadline - min_remaining + 5)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5
    
    return current_a, current_sc
