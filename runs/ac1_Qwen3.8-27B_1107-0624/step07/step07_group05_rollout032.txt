import numpy as np

from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n
        = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smoobj_pm(z, p, n):
    """Power-mean objective: (sum c_k^p)^(1/p) as smooth max."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return 2.0 * n * c_max
    # Normalize to avoid overflow
    cn = c / c_max
    pm = (np.sum(np.power(np.maximum(cn, 0), p))) ** (1.0 / p)
    return 2.0 * n * c_max * pm


def _smograd_pm(z, p, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    cn_p = np.power(cn, p)
    S = np.sum(cn_p)
    if S < 1e-300:
        return np.zeros_like(a)
    # d/dz [(sum cn^p)^(1/p)] = (1/p) * S^(1/p - 1) * p * sum(cn^(p-1) * dc/dz)
    # = S^(1/p - 1) * sum(cn^(p-1) * dc_k/dz)
    # dc_k/dz_j = 4*n*a_j*(h_j) where h = conv(c_weights, a_rev)[n-1:2n-1]
    # cn^(p-1) = (c/c_max)^(p-1)
    w = np.power(np.maximum(c / c_max, 0), p - 1)
    w = w / (c_max * S ** (1 - 1.0 / p))  # proper scaling
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c) / c_max)
    return 4.0 * n * a * (h - W) * (2.0 * n) / (2.0 * n * c_max)


def _gradient_active(a, c, n, tol_frac=0.001):
    """Gradient using only near-active constraints (sharper than LSE)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = active.sum()
    if n_active < 1:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_active = 1
    # Use weighting: closer to max gets more weight
    w = np.zeros_like(c)
    w[active] = (c[active] - c_max + tol) / tol  # linear weighting in [0,1]
    w_sum = w.sum()
    if w_sum < 1e-15:
        w[active] = 1.0 / n_active
    else:
        w = w / w_sum
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.4):
    """
    Improved pressure equalization with exact KKT pressure
    including the diagonal (self-term) correction.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = float(_full_convolve(a, a).max())

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive tolerance: tighten as we iterate
        frac = max(0.005, 0.05 * (1.0 - it / n_iters))
        tol = frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-max(3, n // 1000):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = active.sum()

        # Compute pressure with exact formula:
        # p_j = sum_{k in S} lambda_k * 2 * a_{k-j} (with lambda_k = 1/n_active)
        # = (2/n_active) * sum_{k in S} a_{k-j}
        # = (2/n_active) * (conv(active, a_rev))[n-1:2n-1]_j
        a_rev = a[::-1]
        af = active.astype(np.float64)
        p = _full_convolve(af, a_rev)
        p = p[n - 1:2 * n - 1] * (2.0 / n_active)

        # Diagonal correction: for k=2j, the gradient has 2*a_j instead of a_{k-j}+a_{k-j}
        # This is already handled correctly by the convolution formula above
        # (since the convolution naturally gives the right count)

        p = np.maximum(p, 1e-30)

        # Multiplicative update: push a toward equalizing pressure
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 10 == 0:
            c_new = _full_convolve(a, a)
            sc_new = float(c_new.max())
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()

    c_final = _full_convolve(a, a)
    sc_final = float(c_final.max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs_v2(a_init, deadline, min_time_remaining=30.0, use_pm=False):
    """L-BFGS with very aggressive beta/p-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)] if len(c_sorted) > 5 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    # Much more aggressive schedule: 25 steps
    if use_pm:
        # Power mean with increasing p
        ps = np.array([4, 6, 8, 10, 12, 16, 20, 24, 28, 32, 40, 48, 56, 64, 80,
                       96, 112, 128, 160, 200, 256, 320, 400, 500, 640])
    else:
        betas = base_beta * np.array([1.0, 0.6, 0.36, 0.22, 0.13, 0.08, 0.05, 0.03,
                                      0.018, 0.011, 0.006, 0.004, 0.002, 0.0012,
                                      0.0007, 0.0004, 0.0002, 0.0001, 0.00006,
                                      0.00004, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = float(_full_convolve(current_a, current_a).max())
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    steps = ps if use_pm else betas
    for i, param in enumerate(steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 10))))

        try:
            if use_pm:
                result = minimize(
                    _smoobj_pm, z, args=(param, n),
                    jac=_smograd_pm,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxcor': 60, 'maxls': 200}
                )
            else:
                result = minimize(
                    _smoobj_lse, z, args=(param, n),
                    jac=_smograd_lse,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxcor': 60, 'maxls': 200}
                )
            a_new = _softmax(result.x)
            sc_new = float(_full_convolve(a_new, a_new).max())
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _newton_refine(a_init, deadline, n_max_iter=200, tol_frac=0.002):
    """
    Newton-like refinement: identify active set, then solve KKT system
    via iterative linearization.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = float(_full_convolve(a, a).max())

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 3

        # KKT: sum_{k in S} lambda_k * grad C_k = constant (for positive a)
        # grad C_k = 2 * shift(a, k)
        # So: sum_k lambda_k * shift(a, k) = const
        # This means: the "weighted sum of shifted a's" should be flat.
        # Compute the current "pressure" and try to flatten it via a linear step.

        a_rev = a[::-1]
        af = active.astype(np.float64) / n_active
        p = _full_convolve(af, a_rev)
        p = p[n - 1:2 * n - 1] * 2.0

        # We want p to be constant. The correction to a:
        # Delta p_j = 2/n_active * sum_{k in S} Delta a_{k-j}
        # If we set Delta a = alpha * (p_mean - p), then:
        # Delta p_j = 2/n_active * sum_{k in S} alpha * (p_mean - p_{k-j})
        # We want this to reduce the variance of p.
        # Simple steepest descent: Delta a = -alpha * grad
        # grad of (1/2)||p - p_mean||^2 w.r.t. a is related to the "double" pressure.

        # Use a simpler approach: multiplicative update with smaller step
        p = np.maximum(p, 1e-30)
        p_mean = np.mean(p)
        # Smaller gamma for stability
        gamma = 0.15 + 0.1 * (it / n_max_iter)  # gradually increase
        ratio = p_mean / p
        a_new = a * np.power(ratio, gamma)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = float(_full_convolve(a_new, a_new).max())
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            a_new = a * np.power(ratio, gamma * 0.5)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / s_new
            sc_new = float(_full_convolve(a_new, a_new).max())
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
            # Reduce tol to sharpen active set
            tol_frac = max(0.0005, tol_frac * 0.8)

    return best_a


def _subgrad_polish_v2(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Subgradient descent with active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = float(_full_convolve(a_init, a_init).max())
    lr = lr_init
    patience = 0
    best_patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        # Tighten tolerance over iterations
        tol_frac = max(0.0005, 0.01 * (1.0 - i / n_max_steps))
        grad = _gradient_active(a, c, n, tol_frac=tol_frac)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = float(_full_convolve(a_check, a_check).max())
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init * 1.5)
                best_patience = patience
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_polish_v2(a, deadline, n_rounds=80):
    """Coordinate descent: adjust top-contributing coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = float(_full_convolve(a, a).max())
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.1 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = top_idx[rng.permutation(len(top_idx))[:min(len(top_idx), n_try)]]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try turning it on slightly
                for factor in [0.01, 0.02, 0.05]:
                    a[i] = factor * np.mean(a[np.maximum(a, 1e-15)])
                    c_try = _full_convolve(a, a).max()
                    if c_try < c_max - 1e-15:
                        improved = True
                    else:
                        a[i] = old_val
                        break
            else:
                # Try scaling
                c_old = c_max
                for factor in [0.5, 0.7, 0.85, 1.15, 1.3]:
                    a[i] = old_val * factor
                    c_try = _full_convolve(a, a).max()
                    if c_try < c_old - 1e-15:
                        improved = True
                        c_old = c_try
                    else:
                        a[i] = old_val
                        break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = float(_full_convolve(a, a).max())
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 2 * t * t) + 1.5
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _interpolate_sequence(a_src, n_dst):
    """Interpolate a sequence to a different length."""
    n_src = len(a_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    a_dst = np.interp(x_dst, x_src, a_src)
    a_dst = np.maximum(a_dst, 0)
    s = a_dst.sum()
    if s > 1e-10:
        a_dst = a_dst / s
    else:
        a_dst = _make_profile(n_dst, "uniform")
    return a_dst


def _multi_stage_optimize(a_init, deadline, min_remaining=60.0):
    """Full optimization pipeline for a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = float(_full_convolve(current_a, current_a).max())

    # Stage 1: L-BFGS with LSE
    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _run_lbfgs_v2(current_a, deadline, min_time_remaining=min_remaining + 60, use_pm=False)
        sc1 = float(_full_convolve(a1, a1).max())
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        pe_deadline = min(deadline - min_remaining - 10, time.monotonic() + 80)
        a2, sc2 = _pressure_equalize_v2(current_a, pe_deadline, n_iters=200, gamma=0.35)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: Power-mean L-BFGS
    if time.monotonic() < deadline - min_remaining - 30:
        a3 = _run_lbfgs_v2(current_a, deadline, min_time_remaining=min_remaining + 30, use_pm=True)
        sc3 = float(_full_convolve(a3, a3).max())
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Newton refinement
    if time.monotonic() < deadline - min_remaining - 10:
        nw_deadline = min(deadline - min_remaining - 5, time.monotonic() + 60)
        a4 = _newton_refine(current_a, nw_deadline, n_max_iter=150, tol_frac=0.003)
        sc4 = float(_full_convolve(a4, a4).max())
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 50)
        a5, sc5 = _subgrad_polish_v2(current_a, sg_deadline, n_max_steps=3000, lr_init=0.015)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Aggressive multi-phase search with L-BFGS (LSE + power-mean),
    pressure equalization, Newton refinement, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc_normalized=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        n = len(p)
        if sc_normalized is None:
            c = _full_convolve(p, p)
            sc_normalized = float(c.max())
        sc = 2.0 * n * sc_normalized / (s * s)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={n}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=600)
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure + Newton on best ---
    print(f"\n[Phase 2] Extended refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_pe, sc_pe = _pressure_equalize_v2(best_seq, min(deadline - 450, time.monotonic() + 120),
                                            n_iters=400, gamma=0.25)
        _update(a_pe, sc_pe)
        if sc_pe < float(_full_convolve(a_main, a_main).max()):
            a_nw = _newton_refine(a_pe, min(deadline - 400, time.monotonic() + 90),
                                  n_max_iter=200, tol_frac=0.002)
            _update(a_nw, float(_full_convolve(a_nw, a_nw).max()))

    # --- Phase 3: L-BFGS with power mean on best ---
    print(f"\n[Phase 3] Power-mean L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_pm = _run_lbfgs_v2(best_seq, deadline, min_time_remaining=400, use_pm=True)
        _update(a_pm, float(_full_convolve(a_pm, a_pm).max()))

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != n_parent]
    # Sort: try sizes close to parent first, then explore
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining_for_this = deadline - 200 - time.monotonic()
        if remaining_for_this < 60:
            break

        if best_seq is not None:
            a_init = _interpolate_sequence(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(30, (deadline - 200 - time.monotonic()) * 0.4)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 5: Diverse restarts ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        profiles = ["asymmetric", "sawtooth", "gaussian", "double_cos", "chirp"]

        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 50:
                break
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=0.02)
            min_rem = max(20, deadline - 50 - time.monotonic())
            if min_rem < 30:
                break
            a_r, sc_r = _multi_stage_optimize(a_pert, deadline, min_remaining=min_rem)
            _update(a_r, sc_r)

    # --- Phase 6: Coordinate descent on best ---
    print(f"\n[Phase 6] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        cd_deadline = min(deadline - 10, time.monotonic() + 150)
        a_cd, sc_cd = _coordinate_polish_v2(best_seq, cd_deadline, n_rounds=60)
        _update(a_cd, sc_cd)

    # --- Phase 7: Final aggressive L-BFGS ---
    print(f"\n[Phase 7] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = _run_lbfgs_v2(best_seq, deadline, min_time_remaining=5, use_pm=False)
        _update(a_fin, float(_full_convolve(a_fin, a_fin).max()))

    # --- Phase 8: Final subgradient ---
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_sg, sc_sg = _subgrad_polish_v2(best_seq, deadline - 2, n_max_steps=500, lr_init=0.005)
        _update(a_sg, sc_sg)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
            if eval_score < np.inf:
                best_score = min(best_score, eval_score)
        except Exception as e:
            print(f"Verification error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(1024, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
