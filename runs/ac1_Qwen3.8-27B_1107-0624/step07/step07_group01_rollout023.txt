def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out

def _kkt_solve_active(a_init, deadline, k_active=6, tol=1e-12):
    """
    Solve the KKT conditions for a given active set.
    At optimum: for the active set S, the gradient of (1/|S|)sum_{k in S} conv[k]
    should be constant on the support of a.
    
    We iterate: 
    1. Compute active set S (top K conv values)
    2. Compute gradient g = grad of avg conv over S
    3. Set a[i] proportional to exp(-g[i]/lambda) for some lambda
    4. Repeat until convergence
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(200):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]
        
        # Check if active set is stable
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        
        # The KKT condition: a[i] > 0 implies g[i] = lambda
        # So we want to find a such that g is constant on the support.
        # Iterative approach: a[i] = exp(-g[i] / temp) / sum exp(-g[j]/temp)
        # This is a multiplicative update / entropy regularization
        
        # Use soft threshold: positions where g is close to max(g) get more mass
        g_max = g.max()
        if g_max < 1e-15:
            break
        
        # Multiplicative update: a_new[i] = a[i] * exp(-eta * (g[i] - mean(g)))
        eta = 0.1
        a_new = a * np.exp(-eta * (g - g.mean()))
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-14:
            break
        a = a_new if sc_new <= _score(a) else a
    
    return best_a, best_sc

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()

def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

def _asymmetric_perturb(a, rng, strength=0.05):
    """Create an asymmetric perturbation."""
    n = len(a)
    # Multiply by a slowly varying asymmetric factor
    t = np.arange(n, dtype=np.float64)
    # Random asymmetric polynomial
    coeffs = rng.normal(0, strength / n, size=5)
    factor = np.ones(n)
    for i, c in enumerate(coeffs):
        factor += c * (t / n) ** i
    a_new = a * np.maximum(factor, 0.1)
    a_new = a_new / a_new.sum()
    return a_new

def _freq_domain_perturb(a, rng, n_freqs=5, strength=0.02):
    """Perturb in the frequency domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    Z = np.fft.rfft(z)
    for _ in range(n_freqs):
        freq = int(rng.integers(1, len(Z)))
        Z[freq] += strength * rng.standard_normal() * (1 + 1j)
    z_new = np.fft.irfft(Z, n)
    z_new = z_new - z_new.mean()
    a_new = _softmax(z_new)
    return a_new

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Fine-grained adaptive log-sum-exp smoothing with L-BFGS,
(2) Nesterov-accelerated subgradient on the exact max,
(3) KKT active-set equalization via multiplicative updates,
(4) Asymmetric perturbations for basin hopping,
(5) Multi-length exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-12):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs_fine(a_start, deadline, reserve_time=20.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Very fine beta schedule: 50 levels, ratio 0.87
    beta0 = max_c * 0.12
    ratio = 0.87
    n_betas = 50
    betas = [beta0 * (ratio ** i) for i in range(n_betas)]

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, (n / 4096.0) ** 1.15)
        # Increase iterations for later (smaller beta) stages
        iter_factor = 1.0 + 0.8 * (i / n_betas)
        max_iter = min(30000, max(300, int(remaining / (0.002 * scale * 5) * iter_factor)))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-17:
            current_a = a_new
            current_sc = sc
            if i % 10 == 0:
                print(f"    beta[{i}]={beta:.3e} score={sc:.12f} t={time.monotonic()-deadline+deadline:.1f}s")

        if current_sc < 1.5015:
            print(f"    Early stop at beta[{i}]: score={current_sc:.12f}")
            break

    return current_a


def _nesterov_subgrad(a_init, deadline, k_active=6, lr_init=0.05, momentum_decay=0.999):
    """Nesterov-accelerated subgradient on the exact max."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    t = 1.0
    stall = 0

    for i in range(100000):
        if time.monotonic() > deadline:
            break

        # Active set: top K positions
        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        # Subgradient of average active conv values
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        a_nest = a + theta * (a - a_prev)
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest < 1e-15:
            a_nest = a.copy()
        else:
            a_nest = a_nest / s_nest

        # Gradient at Nesterov point (reuse active set for stability)
        a_rev_n = a_nest[::-1]
        wc_n = _full_convolve(w, a_rev_n)
        h_n = wc_n[n - 1:2 * n - 1]
        c_n = _full_convolve(a_nest, a_nest)
        W_n = float(np.dot(w, c_n))
        grad_n = 2.0 * (h_n - W_n)
        grad_n = grad_n - grad_n.mean()

        # Step with adaptive lr
        step = lr / (t_new * np.sqrt(n))
        a_new = a_nest - step * grad_n
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a_prev = a
        a = a_new
        t = t_new

        # Periodic check
        if i % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-16:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 80:
                    lr *= 0.6
                    if lr < 1e-12:
                        break
                    # Reset momentum
                    a = best_a.copy()
                    a_prev = a.copy()
                    t = 1.0
                    stall = 0

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _multiplicative_kkt(a_init, deadline, k_active=6, eta_init=0.15):
    """Multiplicative update to equalize active convolution values."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta_init
    stall = 0

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W

        g_mean = g.mean()
        g_dev = g - g_mean

        # If gradient is nearly constant, we're at (near) KKT
        if np.abs(g_dev).max() < 1e-16:
            break

        # Multiplicative update
        a_new = a * np.exp(-eta * g_dev)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            eta *= 0.5
            stall += 1
        else:
            a_new = a_new / s_new
            sc_new = _score(a_new)

            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                # Slowly increase eta if improving
                eta = min(eta * 1.01, 1.0)
            else:
                a = a_new  # still move (convex, so this is fine)
                stall += 1
                if stall > 200:
                    eta *= 0.5
                    if eta < 1e-10:
                        break
                    stall = 0

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric_taper(n, skew=0.3):
    """Asymmetric taper: shift the peak away from center."""
    t = np.linspace(0, 1, n)
    center = 0.5 + skew * 0.3
    width = 0.25 + 0.1 * skew
    a = np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_asym(a, rng, strength=0.04):
    """Asymmetric polynomial perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    coeffs = rng.normal(0, strength / 3.0, size=6)
    factor = np.ones(n)
    for i, c in enumerate(coeffs):
        factor += c * t ** i
    factor = np.maximum(factor, 0.05)
    a_new = a * factor
    a_new = a_new / a_new.sum()
    return a_new


def _perturb_freq(a, rng, n_freqs=4, strength=0.02):
    """Frequency domain perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    Z = np.fft.rfft(z)
    for _ in range(n_freqs):
        f = int(rng.integers(1, len(Z)))
        Z[f] += strength * rng.standard_normal() * (1.0 + 1.0j)
    z_new = np.fft.irfft(Z, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _perturb_log_gaussian(a, rng, sigma=0.05):
    """Gaussian perturbation in log space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

            else:
                # Asymmetric chirp blend
                ac = _make_asymmetric_taper(n, 0.2 * (h + 1))
                blend = 0.3 + 0.1 * h
                a_pert = (1 - blend) * best_seq + blend * ac
                a_pert = a_pert / a_pert.sum()

            # Quick L-BFGS polish
            c_p = _full_convolve(a_pert, a_pert)
            mc = float(c_p.max())
            beta = mc * 0.003
            a_opt = _lbfgs_smooth(a_pert, beta, n, 8000)
            try_update(a_opt, f"hop_{h}")

    # === Phase 5: Length exploration (5% of budget) ===
    deadline_p5 = t_start + total_budget * 0.95
    print(f"\n=== Phase 5: Length exploration (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline_p5:
        n_cur = len(best_seq)
        candidates_n = []
        for tn in [2 * n_cur, n_cur // 2, 4096, 8192, 16384, 32768, 65536]:
            if tn != n_cur and 1024 <= tn <= 65536 and tn not in candidates_n:
                candidates_n.append(tn)

        for tn in candidates_n:
            if time.monotonic() > deadline_p5 - 30:
                break
            print(f"  Exploring n={tn}...")
            a_interp = _interpolate_to_n(best_seq, tn)
            sc_int = _score(a_interp)
            print(f"    Interp score: {sc_int:.12f}")
            try_update(a_interp, f"interp_{tn}")

            if time.monotonic() < deadline_p5 - 30:
                a_opt = _adaptive_lbfgs_fine(a_interp, deadline_p5 - 20,
                                             reserve_time=10.0)
                try_update(a_opt, f"opt_n{tn}")

    # === Phase 6: Final ultra-fine polishing ===
    print(f"\n=== Phase 6: Final polish (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        for rnd in range(15):
            if time.monotonic() > deadline - 3:
                break

            # Ultra-small beta L-BFGS
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 3e-8
            remaining = deadline - 3 - time.monotonic()
            max_iter = min(20000, max(500, int(remaining / 0.003)))
            if max_iter > 200:
                a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_f, f"final_lbfgs_{rnd}")

            # Multiplicative KKT
            if time.monotonic() < deadline - 3:
                a_k, _ = _multiplicative_kkt(best_seq, deadline - 3,
                                             k_active=5, eta_init=0.05)
                try_update(a_k, f"final_kkt_{rnd}")

            # Nesterov (short)
            if time.monotonic() < deadline - 3:
                a_n, _ = _nesterov_subgrad(best_seq, deadline - 3,
                                           k_active=4, lr_init=0.02)
                try_update(a_n, f"final_nest_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.12f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
