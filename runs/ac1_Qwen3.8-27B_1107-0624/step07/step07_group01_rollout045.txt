"""

1. Projected subgradient with backtracking line search and adaptive active sets
2. Momentum-based acceleration (Nesterov)
3. L-BFGS smoothing for basin entry

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _simplex_project(x, target_sum=1.0):
    """Project x onto the simplex {y >= 0, sum(y) = target_sum}."""
    n = len(x)
    # Binary search for tau such that sum(max(x - tau, 0)) = target_sum
    # Equivalent: find tau so that sum(x_i - tau for x_i > tau) = target_sum
    # Sort x in descending order
    u = np.sort(x)[::-1]
    # Find rho: largest index where u[rho] > (sum(u[1:rho+1]) - target_sum) / rho
    cssv = np.cumsum(u)
    val = (u - target_sum / np.arange(1, n + 1)) > 0
    rho = int(np.sum(val))
    if rho == 0:
        return np.full(n, target_sum / n)
    tau = (cssv[rho - 1] - target_sum) / rho
    y = np.maximum(x - tau, 0.0)
    # Fix any numerical issues
    s = y.sum()
    if s > 0 and abs(s - target_sum) > 1e-12:
        y = y * target_sum / s
    return y


def _autocorr_gradient(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Returns array of same length as a.
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, _full_convolve(a, a)))
    return 2.0 * n * (h - W)


def _autocorr_gradient_simple(a, k_indices):
    """Simpler gradient for a few active indices."""
    n = len(a)
    c = _full_convolve(a, a)
    w = np.zeros(2 * n - 1)
    w[k_indices] = 1.0 / len(k_indices)
    return _autocorr_gradient(a, w)


def _proj_subgrad(a_init, deadline, reserve=10.0, k_init=10, k_max=50,
                  momentum=0.9, max_steps=None):
    """Projected subgradient with backtracking line search and momentum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_sc = _score(a)
    
    v = np.zeros(n)  # momentum
    lr = 0.5 / n
    ar = 0.5  # Armijo parameter
    stall = 0
    stall_threshold = 200
    
    max_steps = max_steps if max_steps else 1000000
    
    for it in range(max_steps):
        if time.monotonic() > deadline - reserve:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Adaptive active set: include positions where c > fraction of max
        # Start with top-k, expand if not improving
        frac = 0.95 - 0.05 * min(1.0, it / 5000.0)
        threshold = c_max * frac
        active = np.where(c >= threshold)[0]
        
        # Ensure we have at least k_init and at most k_max
        if len(active) < k_init:
            top_idx = np.argsort(c)[::-1][:k_init]
            active = top_idx
        elif len(active) > k_max:
            top_idx = np.argsort(c)[::-1][:k_max]
            active = top_idx
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        g = _autocorr_gradient(a, w)
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * g
        v = v - v.mean()  # remove mean (stays on simplex)
        
        # Backtracking line search
        a_new = a
        lr_try = lr
        f_cur = _score(a)
        improved = False
        
        for ls_iter in range(30):
            candidate = a - lr_try * v
            candidate = _simplex_project(candidate, 1.0)
            f_try = _score(candidate)
            
            if f_try <= f_cur - ar * lr_try * g_norm * np.max(np.abs(v)):
                a_new = candidate
                improved = True
                break
            lr_try *= 0.7
        
        if improved:
            a = a_new
            if f_try < best_sc:
                best_sc = f_try
                best_a = a.copy()
                stall = 0
            lr = min(lr * 1.05, 1.0)
        else:
            stall += 1
            lr *= 0.8
            if lr < 1e-10:
                # Try to escape: increase active set
                if len(active) < k_max:
                    frac = max(0.8, frac - 0.05)
                else:
                    break
            if stall > stall_threshold:
                # Reset momentum and try smaller lr
                v = np.zeros(n)
                lr = 1.0 / n
                stall = 0
                if it > 100000:
                    break
    
    return best_a, best_sc


def _lbfgs_smooth(a_init, beta, n, max_iter=5000, ftol=1e-18, gtol=1e-14):
    """L-BFGS on log-sum-exp smoothed objective."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    
    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _nesterov_subgrad(a_init, deadline, reserve=10.0, max_steps=200000):
    """Nesterov accelerated projected subgradient."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()  # extrapolation point
    
    best_a = a.copy()
    best_sc = _score(a)
    
    t_k = 1.0
    lr = 1.0
    momentum = 0.9
    stall = 0
    
    for it in range(max_steps):
        if time.monotonic() > deadline - reserve:
            break
        
        # Evaluate at y (Nesterov lookahead)
        c = _full_convolve(y, y)
        c_max = c.max()
        
        # Adaptive active set
        top_k = min(20, 2 * n - 1)
        active = np.argsort(c)[::-1][:top_k]
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = _autocorr_gradient(y, w)
        g = g - g.mean()
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta_k = (t_k - 1.0) / t_new
        
        a_old = a.copy()
        a = _simplex_project(y - lr * g / t_new, 1.0)
        y = a + beta_k * (a - a_old)

        if it % 100 == 0:
            sc = _score(a)

                if stall > 50:
                    lr *= 0.5
                    stall = 0
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    t = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((t - center) / (sigma_frac * n)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, reserve=20.0, n_hops=5):

def _kkt_flatness_check(a):

    if time.monotonic() < deadline - 100:
        a_p5 = _adaptive_lbfgs(best_seq, deadline, reserve_time=60.0)
        try_update(a_p5, "tight_lbfgs")
    
    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        n_hops = 5 if time.monotonic() < deadline - 200 else 3
        a_p6, sc_p6 = _basin_hop(best_seq, deadline, rng, reserve=30.0, n_hops=n_hops)
        try_update(a_p6, "basin_hop")
    
    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        lengths_to_try = []
        # Try larger and smaller lengths
        if n_parent < 16384:
            lengths_to_try = [16384, 8192, 32768]
        else:
            lengths_to_try = [32768, 8192, 2048]
        
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 40:
                break
            if target_n == n_parent:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")
            try_update(a_interp, f"interp_n{target_n}")
            
            if time.monotonic() < deadline - 35:
                # Quick optimization
                c_i = _full_convolve(a_interp, a_interp)
                mc = c_i.max()
                beta = mc * 0.01
                max_iter = min(5000, max(500, int((deadline - 35 - time.monotonic()) / 0.002)))
                a_opt = _lbfgs_smooth(a_interp, beta, target_n, max_iter)
                try_update(a_opt, f"lbfgs_n{target_n}")
    
    # === Phase 8: Final intensive polishing ===
    print(f"\n=== Phase 8: Final polishing (t={time.monotonic()-t_start:.1f}s) ===")
    n_best = len(best_seq)
    
    for rnd in range(10):
        if time.monotonic() > deadline - 3:
            break
        
        # Exact subgradient
        if time.monotonic() < deadline - 3:
            a_f, sc_f = _exact_subgrad_polish(best_seq, deadline - 3, reserve=1.0,
                                               max_steps=20000)
            try_update(a_f, f"final_exact_{rnd}")
        
        # Micro L-BFGS
        if time.monotonic() < deadline - 3:
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-8
            max_iter = min(3000, max(100, int((deadline - 3 - time.monotonic()) / 0.001)))
            if max_iter > 50:
                a_f2 = _lbfgs_smooth(best_seq, beta, n_best, max_iter)
                try_update(a_f2, f"final_lbfgs_{rnd}")
        
        # Projected subgradient with very small active set
        if time.monotonic() < deadline - 3:
            a_f3, _ = _proj_subgrad(best_seq, deadline - 3, reserve=1.0,
                                     k_init=3, k_max=8, momentum=0.95, max_steps=20000)
            try_update(a_f3, f"final_proj_{rnd}")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
