        # and w solves (H+mu*I)w = 1
        # Then 1^T d = -1^T u + λ'*1^T w = 0
        # λ' = (1^T u) / (1^T w)
        
        def cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - hvp(x)  # = rhs since x=0
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r = r - alpha * Ap
                rs_new = np.dot(r, r)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r + (rs_new / rs) * p
                r = r_new  # BUG: should be r_new
                rs = rs_new
            return x
        
        # Fix the CG bug
        def cg_solve_fixed(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs.copy()  # hvp(0) = 0, so r = rhs
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x = x + alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x
        
        u = cg_solve_fixed(g_t)
        w = cg_solve_fixed(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            d = -u
            d = d - d.mean()
        else:
            lam = float(u.sum()) / sum_w
            d = -u + lam * w
            d = d - d.mean()  # Ensure sum(d) = 0 for numerical stability
        
        # Line search with non-negativity
        neg_mask = d < -1e-15
        if neg_mask.any():
            t_max = 0.99 * np.min(a[neg_mask] / (-d[neg_mask]))
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)
        
        f_old = _score(a)
        t = min(1.0, t_max)
        accepted = False
        for ls in range(50):
            a_try = a + t * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            f_try = _score(a_try)
            if f_try < f_old - 1e-12 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5
        
        if not accepted:
            break  # or try gradient step
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc
