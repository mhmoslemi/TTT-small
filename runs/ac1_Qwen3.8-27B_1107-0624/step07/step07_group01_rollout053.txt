"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating method (main innovation)
2. L-BFGS with log-sum-exp smoothing for polishing
3. Adaptive step-size control
4. Multi-length and multi-initialization exploration
5. Subgradient method for fine tuning
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lp_best_response(a):
    """
    Solve the LP: minimize t subject to (a*b)_k <= t for all k, b >= 0, sum(b) = 1.
    Returns the optimal b.
    """
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, t
    # Objective: minimize t → c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: (a*b)_k <= t for k = 0, ..., 2n-2
    # (a*b)_k = sum_i a_i * b_{k-i} (with b_j = 0 for j out of range)
    # So: sum_i a_i * b_{k-i} - t <= 0
    
    # Build the constraint matrix A (sparse would be better but let's start with dense for small n)
    # num_constraints = 2n - 1
    num_k = 2 * n - 1
    A = np.zeros((num_k, n + 1))
    b_vec = np.zeros(num_k)
    
    for k in range(num_k):
        # (a*b)_k = sum_{i} a_i * b_{k-i} where k-i must be in [0, n-1]
        # i.e., i must satisfy: 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i  # index into b
            A[k, j] = a[i]
        A[k, -1] = -1.0  # -t
    
    # Sum constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, t unbounded (or t >= 0)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A, b_ub=b_vec, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    return result.x[:n]

    current_a = a_start.copy()
    current_sc = _score(current_a)

                              2e-6, 1e-6, 5e-7])

    for i, beta in enumerate(betas):

Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating method (main innovation over parent)
2. L-BFGS with adaptive log-sum-exp smoothing for polishing
3. Dual-guided subgradient polishing
4. Multi-length exploration and interpolation
5. Basin hopping for diversification

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    Solve LP: minimize t s.t. (a*b)_k <= t for all k, b >= 0, sum(b) = 1.
    Returns optimal b, or None on failure.
    """
    n = len(a)
    num_k = 2 * n - 1

    if max_constraints is not None and max_constraints < num_k:
        # Use only top-k constraints (based on current a*a peaks as proxy)
        c_proxy = _full_convolve(a, a)
        k_indices = np.argsort(c_proxy)[::-1][:max_constraints]
    else:
        k_indices = np.arange(num_k)

    nrows = len(k_indices)

    rows = []
    cols = []
    vals = []

    for row, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(row)
            cols.append(j)
            vals.append(a[i])
        rows.append(row)
        cols.append(n)
        vals.append(-1.0)

    A = csr_matrix((vals, (rows, cols)), shape=(nrows, n + 1))
    b_vec = np.zeros(nrows)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None

    if not result.success:
        return None

    b = result.x[:n].copy()
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-15:
        return None
    b = b / s
    return b


def _lp_alternating(a_init, deadline, max_iters=100, alpha_init=0.15,
                    max_constraints=None, use_line_search=True):
    """
    LP-based alternating optimization.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    alpha = alpha_init
    t_lp = 0.0

    print(f"  LP alternating: n={n}, max_constraints={max_constraints}")

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        b = _lp_solve(a, max_constraints=max_constraints)
        t_lp = time.monotonic() - t0

        if b is None:
            print(f"    LP failed at iter {it}")
            break

        # Add tiny regularization to avoid extreme sparsity
        b = b + 1e-10
        b = b / b.sum()

        # Line search over step sizes
        if use_line_search:
            alphas = [alpha * 0.5, alpha, alpha * 1.5, alpha * 2, 0.3, 0.5, 1.0]
            best_local_a = a.copy()
            best_local_sc = _score(a)
            best_local_alpha = alpha

            for a_try_val in alphas:
                a_try = (1 - a_try_val) * a + a_try_val * b
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score(a_try)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local_a = a_try.copy()
                    best_local_alpha = a_try_val

            a = best_local_a
            alpha = best_local_alpha
        else:
            a_try = (1 - alpha) * a + alpha * b
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                break
            a_try = a_try / s
            a = a_try
            alpha = max(0.01, alpha * 0.95)

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if it % 10 == 0 or it < 3:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_sc:.10f}, "
                  f"lp_time={t_lp:.2f}s, alpha={alpha:.3f}")

        # Check convergence
        if np.allclose(a, best_a, atol=1e-12):
            print(f"    Converged at iter {it}")
            break

        # Gradually increase constraints if we started with fewer
        if max_constraints is not None and it % 20 == 19:
            max_constraints = min(2 * n - 1, max_constraints * 2)

    return best_a, best_sc


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(15000, max(200, int(remaining / (0.005 * scale * 25))))

        z0 = np.log(np.maximum(current_a, 1e-18))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 80,
                                       'maxls': 100, 'disp': False})
            a_out = _softmax(result.x)
            sc = _score(a_out)
            if sc < current_sc - 1e-15:
                current_a = a_out
                current_sc = sc
        except Exception:
            pass

        if current_sc < 1.5005:
            break

    return current_a


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=50000, lr_init=0.06):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    stall_threshold = max(10, k_active * 3)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = lr * a.sum()
        z = z - step * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.03, 0.2)
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= 0.6
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _direct_subgrad(a_init, deadline, k_active=3, max_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.3 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        a_try = a - step_size * g / np.sqrt(n)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        sc_cur = _score(a)
        if sc_try < sc_cur - 1e-16:
            a = a_try
            step_size = min(step_size * 1.02, 0.5)
        else:
            stall += 1
            step_size *= 0.93

        if stall > 400:
            break

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_optimized_taper(n, p1=0.7, p2=1.3):
    """A more sophisticated taper based on raised cosine with adjustable parameters."""
    t = np.linspace(0, 1, n)
    a = (1.0 - np.cos(np.pi * t)) / 2.0
    a = a ** p1
    # Add a second component
    b = (1.0 - np.cos(2.0 * np.pi * t)) / 4.0
    a = a + 0.1 * b ** p2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _kkt_refine(a, deadline, k_active=4, max_iter=100):
    """KKT-inspired iterative refinement."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        g = h - W
        g = g - g.mean()

        alpha = 1.0
        improved = False
        for _ in range(15):
            a_try = a - alpha * g / n
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:
            break

    return best_a, best_sc


def _basin_hop(a_current, deadline, rng, n_hops=4):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)

    for h in range(n_hops):
        if time.monotonic() > deadline:
            break

        pert_type = h % 3
        if pert_type == 0:
            sigma = 0.05 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2):
                freq = int(rng.integers(1, max(2, n // 16)))
                amp = 0.02 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 3 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
            z = z - z.mean()
            a_pert = _softmax(z)

        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        beta = mc * 0.01
        n_pert = len(a_pert)
        z0 = np.log(np.maximum(a_pert, 1e-18))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n_pert)

        def jac(z):
            return _smograd(z, beta, n_pert)

        try:
            max_iter = min(3000, max(200, int((deadline - time.monotonic()) / 30.0 / 0.005)))
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'display': False, 'disp': False})
            a_opt = _softmax(result.x)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining LP-based best-response dynamics with
    L-BFGS smoothing and subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    candidates = []

    if parent is not None:
        candidates.append(("parent", parent))
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Work at n=4096 for main optimization
    n_main = 4096

    if parent is not None and len(parent) != n_main:
        parent_interp = _interpolate_to_n(parent, n_main)
        candidates.append(("parent_interp", parent_interp))
        try_update(parent_interp, "parent_interp")
    elif parent is not None:
        pass  # already at n_main

    # Structured initializations
    structured_inits = [
        ("chirp_f1", _make_chirp(n_main, freq=1.0, eps=0.3)),
        ("chirp_f2", _make_chirp(n_main, freq=2.0, eps=0.25)),
        ("chirp_half", _make_chirp(n_main, freq=0.5, eps=0.35)),
        ("multi_13", _make_multiband(n_main, bands=[1, 3], eps=0.25)),
        ("multi_12", _make_multiband(n_main, bands=[1, 2], eps=0.3)),
        ("multi_25", _make_multiband(n_main, bands=[2, 5], eps=0.2)),
        ("tapered05", _make_tapered(n_main, power=0.5)),
        ("tapered07", _make_tapered(n_main, power=0.7)),
        ("tapered10", _make_tapered(n_main, power=1.0)),
        ("opt_taper", _make_optimized_taper(n_main)),
    ]

    for name, a_init in structured_inits:
        candidates.append((name, a_init))
        try_update(a_init, name)

    # Random initializations
    for r in range(4):
        a_r = _make_random_positive(n_main, rng)
        candidates.append((f"random_{r}", a_r))
        try_update(a_r, f"random_{r}")

    if best_seq is None:
        best_seq = structured_inits[0][1].copy()
        best_score_val = _score(best_seq)
        best_n = n_main

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: LP-based alternating optimization (MAIN PHASE) ===
    print(f"\n=== Phase 2: LP Alternating (t={time.monotonic() - t_start:.1f}s) ===")

    # Try LP at n=2048 first (faster), then n=4096
    n_lp_small = 2048
    a_lp_small = _interpolate_to_n(best_seq, n_lp_small)

    lp_deadline_small = min(t_start + 180, deadline - 600)
    if time.monotonic() < lp_deadline_small:
        a_lp1, sc_lp1 = _lp_alternating(
            a_lp_small, lp_deadline_small,
            max_iters=40, alpha_init=0.2,
            max_constraints=n_lp_small  # use n constraints initially
        )
        try_update(a_lp1, f"lp_n{n_lp_small}")

        # Interpolate back to n_main
        if len(a_lp1) != n_main:
            a_lp1_full = _interpolate_to_n(a_lp1, n_main)
            try_update(a_lp1_full, f"lp_n{n_lp_small}_interp")

    # LP at n=4096
    lp_deadline_main = min(t_start + 600, deadline - 300)
    if time.monotonic() < lp_deadline_main:
        a_lp2, sc_lp2 = _lp_alternating(
            best_seq, lp_deadline_main,
            max_iters=60, alpha_init=0.15,
            max_constraints=None  # use all constraints
        )
        try_update(a_lp2, "lp_n4096_full")

    # Another round of LP with different alpha
    if time.monotonic() < deadline - 300:
        a_lp3, sc_lp3 = _lp_alternating(
            best_seq, min(t_start + 900, deadline - 200),
            max_iters=30, alpha_init=0.08,
            max_constraints=None
        )
        try_update(a_lp3, "lp_n4096_alpha_small")

    # === Phase 3: L-BFGS polishing ===
    print(f"\n=== Phase 3: L-BFGS Polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_p3 = _adaptive_lbfgs(best_seq, deadline, reserve_time=200.0)
        try_update(a_p3, "lbfgs_after_lp")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 150,
                                      k_active=5, max_steps=30000, lr_init=0.05)
        try_update(a_sg, "sg_k5")

    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg2, sc_sg2 = _subgrad_polish(best_seq, deadline - 100,
                                        k_active=3, max_steps=20000, lr_init=0.04)
        try_update(a_sg2, "sg_k3")

    if best_seq is not None and time.monotonic() < deadline - 80:
        a_d, sc_d = _direct_subgrad(best_seq, deadline - 80,
                                    k_active=3, max_steps=15000)
        try_update(a_d, "sg_direct")

    # === Phase 5: KKT refinement ===
    print(f"\n=== Phase 5: KKT (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_kkt, sc_kkt = _kkt_refine(best_seq, deadline - 60, k_active=4)
        try_update(a_kkt, "kkt")

    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin Hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 60, rng, n_hops=4)
        try_update(a_hop, "basin_hop")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for target_n in [8192, 2048, 10240]:
            if time.monotonic() > deadline - 30:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            try_update(a_interp, f"interp_n{target_n}")

            if time.monotonic() < deadline - 30:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=25.0)
                try_update(a_opt, f"lbfgs_n{target_n}")

    # === Phase 8: Final aggressive polishing ===
    print(f"\n=== Phase 8: Final Polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not
