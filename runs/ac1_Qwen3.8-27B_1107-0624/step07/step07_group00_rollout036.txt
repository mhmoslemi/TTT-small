"""
Multi-strategy optimizer combining LP-based Frank-Wolfe updates,
Adam in log-space, coordinate descent, and multi-scale exploration.
Key innovation: LP-based direction finding for exact active-set handling.
"""
import numpy as np
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,m] = a[k-m] if valid."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            idx = k - np.arange(m_start, m_end + 1)
            A[k, m_start:m_end+1] = a[idx]
    return A

def _lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None

        M = float(c.max())

        g = _lp_direction(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-15:

        g = g / g.sum()  # normalize to sum=1

                a = a_cand

def _compute_smoothed_gradient(a, c, T):

        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        c = _fft_convolve(a, a)

            # Use active set gradient

                p[idx] = decay ** i
            p_sum = p.sum()
            if p_sum > 1e-300:
                p = p / p_sum
            else:
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0

            s = e.sum()

        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:

                patience = 0

                m_buf = np.zeros(n)

"""
Multi-strategy optimizer combining LP-based Frank-Wolfe updates (key innovation),
Adam in log-space with adaptive temperature, coordinate descent polishing,
and multi-scale exploration (n=2048/4096/8192). The LP direction-finding
provides exact active-set handling for the autocorrelation peak minimization.
"""
import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            idx = k - np.arange(m_start, m_end + 1)
            A[k, m_start:m_end + 1] = a[idx]
    return A


def _lp_direction(a, M, time_limit=20):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x
    return None


def _lp_optimize(a, deadline, max_iters=50, verbose=True, verbose_interval=3):
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())

        t0 = time.monotonic()
        g = _lp_direction(a, M, time_limit=15)
        lp_time = time.monotonic() - t0

        if g is None or g.sum() < 1e-15:
            if verbose:
                print(f"  LP iter {it}: LP failed, stopping")
            break

        g = g / g.sum()

        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.03]:
            a_cand = (1 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0)
            s_cand = a_cand.sum()
            if s_cand < 1e-15:
                continue
            a_cand = a_cand / s_cand
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-12):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break

        if verbose and it % verbose_interval == 0:
            print(f"  LP iter {it}: score={best_score:.10f}, M={M:.6f}, "
                  f"g_sum_ratio={g.sum()/a.sum():.4f}, lp_time={lp_time:.2f}s, imp={improved}")

        if not improved:
            sc_g = _score_norm(g)
            if sc_g < best_score * (1 - 1e-12):
                best_score = sc_g
                best_a = g.copy()
                a = g
            else:
                if verbose:
                    print(f"  LP iter {it}: no improvement, stopping")
                break

    return best_a, best_score


def _compute_gradient(a, c, T, active_k=5, decay=0.5):
    n = len(a)
    if T > 0 and T < 1e-10:
        k_indices = np.argsort(c)[::-1][:active_k]
        p = np.zeros_like(c)
        for i, idx in enumerate(k_indices):
            p[idx] = decay ** i
        p_sum = p.sum()
        if p_sum > 1e-300:
            p = p / p_sum
        else:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
    else:
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=5, decay=0.5,
                  eval_interval=200, clip_norm=0.5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _compute_gradient(a, c, T, active_k, decay)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fast_cd_polish(a, deadline, max_rounds=200, tol_frac=1.0005):
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * tol_frac
    active = np.where(c > tol)[0]

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for k in active:
            if len(active) == 0:
                break
            if k < n:
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
            else:
                j_start = k - (n - 1)
                j_end = n - 1
            for j in range(j_start, j_end + 1):
                for scale in [1.15, 1.08, 0.92, 0.85, 0.75, 1.0]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-13):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * tol_frac
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _peak_equalize(a, deadline, max_iters=200):
    """Try to equalize the top peaks of the autocorrelation."""
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        # Find all positions within 0.5% of max
        threshold = c_max * 0.9995
        active_k = np.where(c >= threshold)[0]
        if len(active_k) <= 1:
            break
        # Try to reduce the single highest peak
        top_k = active_k[np.argmax(c[active_k])]
        if top_k < n:
            j_start = max(0, top_k - n + 1)
            j_end = min(top_k, n - 1)
        else:
            j_start = top_k - (n - 1)
            j_end = n - 1
        # Find the j that contributes most to this peak
        contributions = np.array([best_a[j] * best_a[top_k - j] for j in range(j_start, j_end + 1)])
        if contributions.sum() < 1e-30:
            break
        worst_j_offset = np.argmax(contributions)
        worst_j = j_start + worst_j_offset
        # Try reducing that element
        for scale in [0.95, 0.9, 0.85, 0.8, 0.98]:
            a_cand = best_a.copy()
            a_cand[worst_j] *= scale
            s = a_cand.sum()
            a_cand /= s
            c_cand = _fft_convolve(a_cand, a_cand)
            sc = 2.0 * n * float(c_cand.max())
            if sc < best_score * (1.0 - 1e-13):
                best_score = sc
                best_a = a_cand.copy()
                break

    return best_a, best_score
