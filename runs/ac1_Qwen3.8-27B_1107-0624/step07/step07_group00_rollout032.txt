"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines Nesterov-projected gradient on simplex, exact coordinate descent,
structured perturbation restarts, and multi-scale exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _nesterov_pg(a_init, deadline, max_iter=50000, top_k=5, 
                  lr_scale=1.0, use_nesterov=True):
    """Nesterov-accelerated projected gradient on simplex."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    eval_interval = 100
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute smoothed gradient using top-k active lags
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        
        # Get top-k lags
        k_indices = np.argsort(c)[::-1][:top_k]
        # Weights: exponential decay
        weights = np.exp(-0.5 * np.arange(top_k))
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for w, k in zip(weights, k_indices):
            # grad[j] += w * 2 * a[k-j] if 0 <= k-j < n else 0
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                # grad[lo:hi+1] += w * 2 * a[k-hi : k-lo+1][::-1]
                partner = a[k - hi : k - lo + 1][::-1]
                grad[lo:hi+1] += w * 2.0 * partner
        
        # Step size
        step = lr_scale / (t + 1)
        # Or use a fixed small step
        # step = 0.01 / (1 + 0.001*t)
        
        if use_nesterov:
            a_new = y - step * grad
            a_new = _project_simplex(a_new)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
            y = _project_simplex(y)
            t_k = t_new
            a = a_new
        else:
            a = a - step * grad
            a = _project_simplex(a)
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Frank-Wolfe conditional gradient with Nesterov acceleration,
adaptive active-set smoothing, exact 1D coordinate refinement,
structured cosine perturbations for restarts, and multi-scale exploration.
Differs from previous approach by using linear (not log-space) optimization,
adaptive active set management, and structured rather than random restarts.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _compute_topk_gradient(a, c, top_k=8, decay_rate=0.4):
    """Compute weighted gradient using top-k active lags with exponential decay weights."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    weights = np.exp(-decay_rate * np.arange(top_k))
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights = weights / wsum

    grad = np.zeros(n)
    for w, k in zip(weights, k_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            partner = a[k - hi : k - lo + 1][::-1]
            grad[lo:hi + 1] += w * 2.0 * partner
    return grad


def _compute_smoothed_gradient(a, c, T):
    """Compute gradient of smooth max approximation."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s

    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            partner = a[k - hi : k - lo + 1][::-1]
            grad[lo:hi + 1] += p[k] * 2.0 * partner
    return grad


def _frank_wolfe_nesterov(a_init, deadline, max_iter=20000,
                           top_k=8, decay_rate=0.4,
                           lr_base=0.5, eval_interval=200,
                           use_adaptive_k=True):
    """Nesterov-accelerated Frank-Wolfe (conditional gradient) on the simplex."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    best_pat = best_score

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(y, y)
        c_max = float(c.max())

        if use_adaptive_k:
            effective_k = min(top_k, max(3, int(top_k * (1.0 - 0.5 * min(1.0, t / max_iter)))))
        else:
            effective_k = top_k

        grad = _compute_topk_gradient(y, c, effective_k, decay_rate)

        # Frank-Wolfe: find min gradient component
        g_min_idx = np.argmin(grad)
        # Step size for FW
        step = lr_base / (t + 1)
        # Or use the FW step: step = 2/(t+2)
        step_fw = 2.0 / (t + 2)
        step = min(step, step_fw)

        a_new = (1.0 - step) * a + step * np.zeros(n)
        a_new[g_min_idx] = step
        # Project to ensure non-negativity (FW update on simplex is already on simplex)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        if t_k < 1e10:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
            y = _project_simplex(y)
            t_k = t_new
        a = a_new

        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10:
                # Restart from best with Nesterov momentum reset
                a = best_a.copy()
                y = best_a.copy()
                t_k = 1.0
                patience = 0

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _mirror_descent_adaptive(a_init, deadline, max_iter=300000,
                              T_start_ratio=0.005, T_end_ratio=0.00005,
                              lr_start=0.03, lr_end=0.002,
                              top_k=5, decay_rate=0.5,
                              eval_interval=100, use_switch=True):
    """Mirror descent in log-space (equivalent to multiplicative weight updates)
    with adaptive switching between top-k and smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(best_a)
    patience = 0
    lr_mult = 1.0

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, (t + 1) / max_iter)
        # Temperature: exponential decay
        T = T_start_ratio * (T_end_ratio / T_start_ratio) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_from_z(z)
        c = _fft_convolve(a, a)

        # Switch between top-k (early) and smoothed (late)
        if use_switch and frac > 0.6:
            g = _compute_smoothed_gradient(a, c, T)
        else:
            g = _compute_topk_gradient(a, c, top_k, decay_rate)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** (t + 1))
        vh = v_buf / (1.0 - b2 ** (t + 1))
        update = lr * mh / (np.sqrt(vh) + eps)
        unorm = np.linalg.norm(update)
        if unorm > 1.0:
            update = update * (1.0 / unorm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_from_z(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.03:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_from_z(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _softmax_from_z(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _exact_cd_step(a, j, conv, tol=1e-10):
    """
    Exact 1D optimization for coordinate j using bisection on the subgradient.
    Returns new value of a[j] that minimizes max_k conv_new[k].
    """
    n = len(a)
    alpha = a[j]
    S = float(a.sum())

    # Compute slopes g_k for all k
    # g_k = 2*a[k-j] if valid, else 0 (for k != 2j)
    # For k = 2j: quadratic term v^2 - alpha^2

    # Determine valid k range for coordinate j
    # conv[k] involves a[j] when 0 <= k-j < n and max(0,k-n+1) <= j <= min(k,n-1)
    # Simplified: k-j in [0, n-1] means k in [j, j+n-1]
    # Also need max(0, k-n+1) <= j <= min(k, n-1)
    # For k in [j, j+n-1]: k-j in [0, n-1] (valid)
    #   and we need k-n+1 <= j, i.e., k <= j+n-1 (always true in our range)
    #   and j <= k (true since k >= j) and j <= n-1 (true)
    # So valid k range is [j, j+n-1] intersected with [0, 2n-2]

    k_lo = max(0, j)
    k_hi = min(2 * n - 2, j + n - 1)

    if k_lo > k_hi:
        return alpha

    # For each k in [k_lo, k_hi], the slope is 2*a[k-j]
    # (this is the linear part; the k=2j case has an additional quadratic)
    ks = np.arange(k_lo, k_hi + 1)
    partners = ks - j
    slopes = 2.0 * a[partners]

    # Precompute: for k != 2j, new_conv[k] = conv[k] + slopes[k-k_lo] * (v - alpha)
    # For k = 2j (if in range): new_conv[2j] = conv[2j] + v^2 - alpha^2

    k_2j = 2 * j
    has_quad = (k_lo <= k_2j <= k_hi)

    # Bisection to find optimal v in [0, alpha*3] (or wider if needed)
    v_lo = 0.0
    v_hi = alpha * 3.0
    if v_hi < 1e-10:
        v_hi = 0.01

    def obj_at(v):
        """Compute max conv_new at value v for coordinate j."""
        if has_quad:
            # Build the full conv_new
            idx_2j = k_2j - k_lo
            vals = conv[k_lo:k_hi + 1].astype(np.float64) + slopes * (v - alpha)
            vals[idx_2j] = conv[k_2j] + v * v - alpha * alpha
            return float(np.max(vals))
        else:
            vals = conv[k_lo:k_hi + 1].astype(np.float64) + slopes * (v - alpha)
            return float(np.max(vals))

    def subgrad_at(v):
        """Compute a subgradient of f at v."""
        if has_quad:
            idx_2j = k_2j - k_lo
            vals = conv[k_lo:k_hi + 1].astype(np.float64) + slopes * (v - alpha)
            vals[idx_2j] = conv[k_2j] + v * v - alpha * alpha
            max_val = np.max(vals)
            # Active set: all k where vals[k] == max_val
            active = np.where(vals >= max_val - 1e-12)[0]
            if idx_2j in active:
                # Subgradient includes 2v from quadratic
                sgs = []
                if len(active) == 1 and active[0] == idx_2j:
                    return 2.0 * v
                sgs.append(2.0 * v)
            for ai in active:
                if ai != idx_2j:
                    sgs.append(slopes[ai])
            # Return the subgradient that's closest to 0 (for bisection)
            sgs = np.array(sgs) if sgs else np.array([0.0])
            return float(np.clip(sgs, sgs.min(), sgs.max())[0])  # Use min element
        else:
            vals = conv[k_lo:k_hi + 1].astype(np.float64) + slopes * (v - alpha)
            max_val = np.max(vals)
            active = np.where(vals >= max_val - 1e-12)[0]
            sgs = slopes[active]
            if len(sgs) == 0:
                return 0.0
            # For minimization, if all active slopes > 0, best is to decrease v
            # If all < 0, best is to increase v
            # If mixed, optimal is where they cross
            return float(np.argmax(sgs)) if np.argmax(sgs) > 0 else float(np.argmin(sgs))

    # Simpler approach: golden section search
    # f(v) is convex (max of affine + convex quadratic), so golden section works
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = v_lo
    b_g = v_hi
    c_g = b_g - gr * (b_g - a_g)
    d_g = a_g + gr * (b_g - a_g)
    fc = obj_at(c_g)
    fd = obj_at(d_g)

    for _ in range(60):
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr * (b_g - a_g)
            fc = obj_at(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr * (b_g - a_g)
            fd = obj_at(d_g)

    v_opt = (a_g + b_g) / 2.0
    return max(0.0, v_opt)


def _coordinate_descent(a, deadline, max_rounds=50, n_trials_per_coord=1,
                        focus_on_active=True, active_tol=0.002):
    """Coordinate descent with exact 1D optimization on a subset of coordinates."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    conv = _fft_convolve(best_a, best_a)
    c_max = float(conv.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        if focus_on_active:
            threshold = c_max * (1.0 - active_tol)
            # Find coordinates that appear in high-conv lags
            active_lags = np.where(conv > threshold)[0]
            # For each active lag k, the contributing coordinates are j where 0<=k-j<n
            coord_set = set()
            for k in active_lags[:20]:  # Limit to top 20 lags
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1, max(1, (hi - lo) // 100)):
                    coord_set.add(j)
            coords = list(coord_set)
            np.random.shuffle(coords)
        else:
            coords = list(range(n))
            np.random.shuffle(coords)

        improved = False
        for j in coords:
            if time.monotonic() > deadline:
                break
            v_new = _exact_cd_step(best_a, j, conv)
            if abs(v_new - best_a[j]) > 1e-12:
                # Evaluate improvement
                old_val = best_a[j]
                best_a[j] = v_new
                S = best_a.sum()
                if S < 1e-15:
                    best_a[j] = old_val
                    continue
                best_a_norm = best_a / S
                sc = _score_norm(best_a_norm)
                if sc < best_score - 1e-12:
                    best_score = sc
                    conv = _fft_convolve(best_a_norm, best_a_norm)
                    c_max = float(conv.max())
                    improved = True
                    best_a = best_a_norm
                else:
                    best_a[j] = old_val
                    if best_a.sum() != 1.0:
                        best_a = best_a / best_a.sum()

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _cosine_perturb(a, freq, phase=0.0, amplitude=0.3):
    """Apply a cosine-based structured perturbation."""
    n = len(a)
    x = np.arange(n)
    mod = 1.0 + amplitude * np.cos(2.0 * np.pi * freq * x / n + phase)
    a_new = a * mod
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining Frank-Wolfe, mirror descent,
    coordinate descent, and structured restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a0 = parent / parent.sum()
        else:
            raise ValueError
    except Exception:
        n = 4096
        a0 = np.ones(n) / n
        print("Using uniform initialization (no parent)")

    n_parent = len(a0)
    best_a = a0.copy()
    best_score = _score(a0)
    print(f"Initial: n={n_parent}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # ===== Phase 1: Frank-Wolfe Nesterov (0-120s) =====
    phase1_end = min(deadline, t_start + 120.0)
    print(f"Phase 1: Frank-Wolfe (n={n_parent})")
    a_fw, sc_fw = _frank_wolfe_nesterov(a0, phase1_end, max_iter=50000,
                                         top_k=10, decay_rate=0.3,
                                         lr_base=1.0, eval_interval=500)
    full_sc = _score(a_fw)
    print(f"  FW result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_fw.copy()

    # ===== Phase 2: Mirror descent from FW result (120-400s) =====
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Mirror descent (n={len(best_a)})")
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_md, sc_md = _mirror_descent_adaptive(best_a, phase2_end, max_iter=500000,
                                                T_start_ratio=0.003 * max_c,
                                                T_end_ratio=0.00003 * max_c,
                                                lr_start=0.04, lr_end=0.002,
                                                top_k=6, decay_rate=0.5,
                                                eval_interval=200)
        full_sc = _score(a_md)
        print(f"  MD result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_md.copy()

    # ===== Phase 3: Coordinate descent polish (400-650s) =====
    phase3_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Coordinate descent (n={len(best_a)})")
        a_cd, sc_cd = _coordinate_descent(best_a, phase3_end, max_rounds=30,
                                           focus_on_active=True, active_tol=0.005)
        full_sc = _score(a_cd)
        print(f"  CD result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # ===== Phase 4: Structured restarts (650-1000s) =====
    phase4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Structured restarts")
        n_best = len(best_a)
        attempts = [
            (1, 0.0, 0.15), (2, 0.0, 0.12), (3, 0.5, 0.10),
            (5, 0.0, 0.08), (8, 0.3, 0.08), (13, 0.0, 0.06),
        ]
        for freq, phase, amp in attempts:
            if time.monotonic() > phase4_end - 30:
                break
            a_pert = _cosine_perturb(best_a, freq, phase, amp)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(phase4_end - 5, time.monotonic() + 40.0)
            a_opt, sc_opt = _mirror_descent_adaptive(a_pert, sub_dl, max_iter=50000,
                                                      T_start_ratio=0.002 * max_cp,
                                                      T_end_ratio=0.00002 * max_cp,
                                                      lr_start=0.02, lr_end=0.001,
                                                      top_k=5, decay_rate=0.5,
                                                      eval_interval=100)
            full_sc = _score(a_opt)
            print(f"  Restart (freq={freq}, phase={phase:.1f}): score={full_sc:.10f}, "
                  f"t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

    # ===== Phase 5: Try n=4096 if we're at 8192 (1000-1300s) =====
    phase5_end = min(deadline, t_start + 1300.0)
    n_current = len(best_a)
    if time.monotonic() < phase5_end and n_current != 4096:
        print(f"Phase 5: Try n=4096 (from n={n_current})")
        a_4k = _interp_to_length(best_a, 4096)
        c_4k = _fft_convolve(a_4k, a_4k)
        max_4k = float(c_4k.max())
        mid5 = min(phase5_end, time.monotonic() + 150.0)
        a_4k_opt, sc_4k = _mirror_descent_adaptive(a_4k, mid5, max_iter=200000,
                                                    T_start_ratio=0.003 * max_4k,
                                                    T_end_ratio=0.00003 * max_4k,
                                                    lr_start=0.03, lr_end=0.002,
                                                    top_k=6, decay_rate=0.5,
                                                    eval_interval=150)
        full_sc_4k = _score(a_4k_opt)
        print(f"  n=4096 MD: score={full_sc_4k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_4k) and full_sc_4k < best_score:
            best_score = full_sc_4k
            best_a = a_4k_opt.copy()

        if time.monotonic() < phase5_end - 20:
            a_4k_cd, _ = _coordinate_descent(best_a, phase5_end - 5, max_rounds=15,
                                              focus_on_active=True, active_tol=0.005)
            sc_4k_cd = _score(a_4k_cd)
            if sc_4k_cd < best_score:
                best_score = sc_4k_cd
                best_a = a_4k_cd.copy()

    # ===== Phase 6: Try n=8192 if we're at 4096 (1300-1500s) =====
    phase6_end = min(deadline, t_start + 1500.0)
    n_current = len(best_a)
    if time.monotonic() < phase6_end and n_current != 8192:
        print(f"Phase 6: Try n=8192 (from n={n_current})")
        a_8k = _interp_to_length(best_a, 8192)
        c_8k = _fft_convolve(a_8k, a_8k)
        max_8k = float(c_8k.max())
        a_8k_opt, sc_8k = _mirror_descent_adaptive(a_8k, phase6_end - 5, max_iter=150000,
                                                    T_start_ratio=0.002 * max_8k,
                                                    T_end_ratio=0.00002 * max_8k,
                                                    lr_start=0.025, lr_end=0.0015,
                                                    top_k=6, decay_rate=0.4,
                                                    eval_interval=150)
        full_sc_8k = _score(a_8k_opt)
        print(f"  n=8192 MD: score={full_sc_8k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_8k) and full_sc_8k < best_score:
            best_score = full_sc_8k
            best_a = a_8k_opt.copy()

    # ===== Phase 7: Long fine-tuning mirror descent (1500-1720s) =====
    phase7_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < phase7_end:
        print(f"Phase 7: Fine-tuning (n={len(best_a)})")
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_fin, sc_fin = _mirror_descent_adaptive(best_a, phase7_end, max_iter=400000,
                                                  T_start_ratio=0.0005 * max_cf,
                                                  T_end_ratio=0.000005 * max_cf,
                                                  lr_start=0.01, lr_end=0.0003,
                                                  top_k=8, decay_rate=0.3,
                                                  eval_interval=100,
                                                  use_switch=True)
        full_sc_fin = _score(a_fin)
        print(f"  Fine-tune: score={full_sc_fin:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_fin) and full_sc_fin < best_score:
            best_score = full_sc_fin
            best_a = a_fin.copy()

        # Quick CD at the end
        if time.monotonic() < phase7_end - 10:
            a_cd_fin, _ = _coordinate_descent(best_a, phase7_end - 3, max_rounds=10,
                                               focus_on_active=True, active_tol=0.003)
            sc_cd_fin = _score(a_cd_fin)
            if sc_cd_fin < best_score:
                best_score = sc_cd_fin
                best_a = a_cd_fin.copy()

    # ===== Phase 8: Final ultra-fine polish (1720s - deadline) =====
    if time.monotonic() < deadline - 5:
        print(f"Phase 8: Ultra-fine polish")
        c_ultra = _fft_convolve(best_a, best_a)
        max_cu = float(c_ultra.max())
        a_ultra, sc_ultra = _mirror_descent_adaptive(best_a, deadline - 4, max_iter=200000,
                                                      T_start_ratio=0.0001 * max_cu,
                                                      T_end_ratio=0.000001 * max_cu,
                                                      lr_start=0.003, lr_end=0.0001,
                                                      top_k=10, decay_rate=0.2,
                                                      eval_interval=50,
                                                      use_switch=False)
        full_sc_ultra = _score(a_ultra)
        print(f"  Ultra: score={full_sc_ultra:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_ultra) and full_sc_ultra < best_score:
            best_score = full_sc_ultra
            best_a = a_ultra.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"External evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
