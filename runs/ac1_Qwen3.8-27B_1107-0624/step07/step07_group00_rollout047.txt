"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal mixing directions,
combined with multi-scale interpolation and coordinate descent polishing.
Key innovation: LP-based direction finding that directly optimizes
the convolution constraint rather than using gradient approximations.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum = 1) or None if LP fails.
    """
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix A_ub (2n-1) x n
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n
    A_ub = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    A_ub = A_ub.tocsr()
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _lp_optimize(a_init, deadline, max_iters=200):

    LP-based iterative optimizer.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        
        g = _lp_direction(a, M)
        if g is None:
            print(f"  LP failed at iter {it}")
            break
        
        # Line search for best mixing parameter
        best_t = 0
        best_sc = _score_norm(a)
        best_a_cand = a
        
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]:
            a_cand = (1 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0)
            a_cand = _normalize(a_cand)
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
                best_a_cand = a_cand
        
        if best_t > 0:
            a = best_a_cand
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
                if it % 20 == 0:

                    found = True

            if found:

                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
            else:

def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=3, decay=0.6,
                  eval_interval=200, clip_norm=0.8, use_backtrack=True,
                  plateau_frac=0.3):
    """Adam optimizer in log-space (from parent, simplified)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / (1.0 - plateau_frac)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if T < T_start * 0.01:
            g = _compute_weighted_active_gradient(a, c, active_k, decay)
        else:
            g = _compute_smoothed_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15)) - np.log(np.maximum(best_a, 1e-15)).mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smoothed_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_weighted_active_gradient(a, c, top_k=5, decay=0.7):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _cd_polish(a, deadline, max_rounds=100):
    """Fast coordinate descent polish focusing on active constraints."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for k in active:
            if k < n:
                js = range(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = range(k - (n - 1), n)
            js = list(js)  # materialize
            if len(js) == 0:
                continue
            for j in js:
                for scale in [1.2, 1.5, 0.8, 0.5, 0.3]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-11):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_a, _score(best_a)

    magnitudes = np.maximum(magnitudes, 0.1)
    
    spectrum = np.zeros(N, dtype=complex)
    spectrum[:N // 2 + 1] = magnitudes * np.exp(1j * phases)
    # Conjugate symmetry
    spectrum[N // 2 + 1:] = np.conj(spectrum[N // 2:0:-1])
    
    sig = np.fft.ifft(spectrum).real[:n]
    sig = np.maximum(sig, 0)
    s = sig.sum()
    if s > 1e-10:
        sig /= s
    else:
        sig = np.ones(n) / n
    return sig
