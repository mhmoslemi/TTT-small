import numpy as np
import random

def evaluate_sequence(seq):
    """Evaluates a sequence of coefficients with normalized sum and clipped values."""
    # Handle empty sequence
    if not seq or len(seq) == 0:
        return np.inf
    
    # Convert to float and apply defensive clipping
    n = len(seq)
    sequence = [float(x) if isinstance(x, (int, float))
              elif isinstance(x, bool) or type(x) is bool: np.nan
              else: np.inf for x in seq]
    
    # Reject invalid elements
    if np.isnan(sequence).any() or np.isinf(sequence).any():
        return np.inf
        
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    
    # Reject sequences with sum too close to zero
    s = np.sum(sequence)
    if s < 0.01:
        return np.inf
        
    # Compute full convolution/cross-correlation
    # Note: Linear convolution for input sequence of length n returns a sequence of length 2*n - 1
    conv = np.convolve(sequence, sequence, mode='full')  # This is the linear convolution, returns 2*n-1 coefficients
    
    # Compute the score
    score = 2.0 * n * np.max(conv) / (s**2)
    return score

    def propose_candidate(self):
        """Generate and evaluate candidate sequences."""
        seed_time = time.time() - self.start_time
        max_time = 1800 if not hasattr(self, 'target_length') or self.target_length == 8192 else 120
        
        if seed_time > max_time:
            self.save_backup()
            return self.best_candidate
            
        if not hasattr(self, 'current_length'):
            self.current_length = 4096
            
        # Generate candidates using crossover and mutation
        # We'll focus on crossover between top performers and diversity operators

        if self.n_pop_candidate >= self.n_pop_candidate_max:
            # Periodically change search scales
            self.transition_length()

        # Generate child through crossover and mutation
        child = self.crossover_and_mutation()

        # Try to maintain diversity
        if random.random() < 0.1:
            self.try_diversity_mutation(child)

        # Evaluate and update
        score = evaluate_sequence(child)
        self.n_evals += 1

        if score < self.best_score:
            self.best_score = score
            self.best_candidate = child

        return child

    def crossover_and_mutation(self):
        """Perform crossover between two parent candidates with Gaussian mutation."""
        parent_a = np.array(self.population[np.random.randint(0, self.n_pop)])
        parent_b = np.array(self.population[np.random.randint(0, self.n_pop)])

        # Crossover at a random point
        crossover_point = np.random.randint(1, len(parent_a)-1)
        child = np.concatenate([parent_a[:crossover_point], parent_b[crossover_point:]])

        # Mutation probability
        may_mutate = np.random.rand() < self.mutation_rate

        if may_mutate:
            # Gaussian mutation on each element
            mutation_std = 0.0 # Adjusted so scores don't rise due to random mutation
            child = child + np.random.normal(0, mutation_std, len(child))

        return child.tolist()

    def transition_length(self):
        # Occasionally move to a larger scale
        if not hasattr(self, 'target_length') or time.time()-self.start_time > 1000:
            if self.validate_and_accept('exponential_length'):
                self.target_length *= 2
            elif self.validate_and_accept('linear_length'):

    def validate_and_accept(self, candidate_type):

        if self.best_score > self.score_threshold:

            # Check if improvement has been validated multiple times or against significant deviations
            if (score < self.best_score) or (score < self.best_score * 1.2 and random.random() < 0.5):
                self.accept_candidate(candidate, score)

    def save_backup(self):
        # Save a fallback uniform candidate
        self.backup_candidate = np.ones(len(self.best_candidate)) / float(len(self.best_candidate)).tolist()
        backup_score = evaluate_sequence(self.backup_candidate)
        if backup_score < self.best_score and backup_score > self.score_threshold:  # Hope it may be better
            self.best_candidate = self.backup_candidate
            self.best_score = backup_score
            
    def accept_candidate(self, candidate, score):

        if score < self.best_score * 0.95 and random.random() < 0.05:
