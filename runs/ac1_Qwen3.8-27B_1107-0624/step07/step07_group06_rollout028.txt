```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(a):
    """Build sparse convolution matrix where C @ b = convolve(a, b)"""
    n = len(a)
    # Column j has values a[0], a[1], ..., a[n-1] at rows j, j+1, ..., j+n-1
    rows = np.repeat(np.arange(n), n)
    rows = rows + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    vals = np.tile(a, n)
    shape = (2*n-1, n)
    return csr_matrix((vals, (rows, cols)), shape=shape)

def _lp_direction(a, M):
    """Solve LP: max 1^T b s.t. (a*b)[k] <= M, b >= 0"""
    n = len(a)
    C = _build_conv_matrix(a)
    c_obj = -np.ones(n)  # Minimize -sum(b)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return None

def _lp_iterate(a, n_lp_iters=50, deadline=None, time_per_lp=10.0):
    """Iterate the LP-based method."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(n_lp_iters):
        if deadline is not None and time.monotonic() > deadline - 5:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        b = _lp_direction(a, M)
        if b is None:
            break
        b = np.maximum(b, 0)
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b / b_sum
        
        # Mix
        t = 0.3  # Mixing parameter
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc
```
