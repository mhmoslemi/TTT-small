def hankel_matvec(w, a):
    """Compute H @ a where H[i,j] = w[i+j]."""
    n = len(a)
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w.astype(np.float64), N)
    A_flip = np.fft.rfft(a[::-1], N)
    result = np.fft.irfft(W * A_flip, N)
    return result[n-1:2*n-1]  # elements at indices n-1, n, ..., 2n-2
