        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_equalize(a, deadline, max_iter=5000, tol=1e-12):
    """
    Active-set multiplicative update to equalize active convolution values.
    Converges to KKT point where active conv values are all equal on support.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: widen as we converge
        conv_range = float(c.max() - c.min())
        if conv_range < 1e-15:
            break
        rel_tol = max(0.005, 0.02 * (1.0 - min(1.0, it / max(1, max_iter // 4))))
        abs_tol = rel_tol * c_max
        active = c >= (c_max - abs_tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            top_idx = np.argsort(c)[::-1][:2 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 2 * n

        # Gradient from active set: g[m] = sum_{k in S} a[k-m]
        a_rev = a[::-1]
        active_f = active.astype(np.float64)
        wc = _full_convolve(a_rev, active_f)
        g = wc[n - 1:2 * n - 1]

        support = a > 1e-12
        if support.sum() < 2:
            break
        g_support_mean = float(g[support].mean())
        if g_support_mean < 1e-15:
            break

        # Multiplicative update: push a[m] down where g[m] is high
        # a_new[m] = a[m] * (g_mean / g[m]) helps equalize
        ratio = g_support_mean / np.maximum(g, 1e-30)
        ratio = np.clip(ratio, 0.1, 10.0)
        a_new = a * ratio
        a_new[~support] = 0
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if abs(c_max_new - c_max) < tol:
            break
        if no_improve > 200:
            break

        a = a_new

    return best_a, best_sc


def _kkt_solve(a_init, active, deadline, max_cg=500, mu=1e-8):
    """
    Solve the KKT system A_T · a_T = λ·1, sum(a_T)=1 using CG.
    A_T[i,j] = 1 if t_i + t_j in active set S.
    Uses FFT-based matrix-vector product.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    support = a > 1e-10
    n_act = int(active.sum())
    if n_act < 2:
        return best_a, best_sc

    # Build the full-length indicator of active set
    q = active.astype(np.float64)  # length 2n-1

    def matvec(v):
        """Compute A·v where A[i,j] = 1 if i+j in S, for i,j in [0,n-1]."""
        # P[t_j] = v[j], P_rev[k] = P[2n-2-k]
        P = v.copy()
        P_rev = P[::-1]
        R = _full_convolve(P_rev, q)
        # (A·v)[i] = R[i + 2n - 2]
        result = R[n - 1:2 * n - 1]
        return result

    # Solve (A + mu*I) x = 1 using CG
    # Then a = x / sum(x) (approximately satisfies A·a = const)

    b = np.ones(n)

    def matvec_reg(v):
        return matvec(v) + mu * v

    x = np.zeros(n)
    r = b - matvec_reg(x)
    p = r.copy()
    rs = np.dot(r, r)

    for cg_it in range(max_cg):
        if time.monotonic() > deadline:
            break
        Ap = matvec_reg(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-10 * (1.0 + np.linalg.norm(b)):
            break
        if rs < 1e-30:
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new

    # Normalize x to get a
    x_pos = np.maximum(x, 0)
    s_x = x_pos.sum()
    if s_x < 1e-15:
        a_try = a.copy()
    else:
        a_try = x_pos / s_x

    sc_try = _score(a_try)
    if sc_try < best_sc:
        best_a = a_try.copy()
        best_sc = sc_try
        a = a_try

        # Iterate: update active set and re-solve
        for outer in range(20):
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            c_max = float(c.max())
            tol_act = max(0.005 * c_max, 1e-12)
            active_new = c >= (c_max - tol_act)
            if int(active_new.sum()) < 2:
                break
            q = active_new.astype(np.float64)

            x = np.zeros(n)
            r = b - (matvec_new := _full_convolve(a[::-1], q)[n-1:2*n-1] + mu * a)
            # Rebuild matvec with new q
            def matvec_new2(v):
                P_rev = v[::-1]
                R = _full_convolve(P_rev, q)
                return R[n - 1:2 * n - 1] + mu * v

            x = np.zeros(n)
            r = b - matvec_new2(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg // 2):
                if time.monotonic() > deadline:
                    break
                Ap = matvec_new2(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-10:
                    break
                if rs < 1e-30:
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new

            x_pos = np.maximum(x, 0)
            s_x = x_pos.sum()
            if s_x < 1e-15:
                break
            a_try = x_pos / s_x
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
            else:
                break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.002):
    """
    Damped Newton on active-set objective with line search.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient: g[i] = (2/m) * sum_j a[j] * A_vec[i+j]
        flip_a = a[::-1]
        out_len = n + len(A_vec) - 1
        N = _next_pow2(out_len)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)[:out_len]
        g = (2.0 / m) * result[n - 1:2 * n - 1]

        # Project to tangent space
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Hessian matvec: H*v = (2/m) * corr(v, A_vec)
        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            flip_v = v[::-1]
            fv = np.fft.rfft(flip_v, N)
            res = np.fft.irfft(fv * fA, N)[:out_len]
            return (2.0 / m) * res[n - 1:2 * n - 1] + mu * v

        # CG solve for Newton direction
        x = np.zeros(n)
        r = g_t - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(g_t)):
                break
            if rs < 1e-30:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        u = x  # Newton step direction (before null-space projection)

        # Project to null space of sum constraint
        w = np.zeros(n)
        r_w = np.ones(n) - hess_matvec(w)
        p_w = r_w.copy()
        rs_w = np.dot(r_w, r_w)
        for cg_it in range(100):
            Ap_w = hess_matvec(p_w)
            pAp_w = np.dot(p_w, Ap_w)
            if pAp_w < 1e-30:
                break
            alpha_w = rs_w / pAp_w
            w += alpha_w * p_w
            r_w_new = r_w - alpha_w * Ap_w
            rs_w_new = np.dot(r_w_new, r_w_new)
            if rs_w < 1e-30:
                break
            p_w = r_w_new + (rs_w_new / rs_w) * p_w
            r_w = r_w_new
            rs_w = rs_w_new
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-15:
            dx = -u - np.mean(u)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(30):
            a_try = a + t * dx
            if a_try.min() < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_check = _full_convolve(a, a)
        if abs(float(c_check.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=16):
    """L-BFGS with aggressive adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_iter if False else a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 50:
        gap = c_sorted[0] - c_sorted[49]
    else:
        gap = c_sorted[0] - (c_sorted[-1] if len(c_sorted) > 1 else 0)
    base_beta = max(gap * 0.5, max_c * 0.05)

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03,
        0.015, 0.0075, 0.004, 0.002, 0.001,
        0.0005, 0.00025, 0.0001, 0.00005, 0.00002
    ][:n_betas])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.003 * max(1, n / 1024.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "palindromic":
        half = n // 2
        t = np.linspace(0, 1, half)
        a_half = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t)
        a_half = np.maximum(a_half, 0.1)
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, a_half[-2::-1]])
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _legendre_init(p):
    """Quadratic residue (Legendre) sequence of prime length p, made non-negative."""
    a = np.zeros(p)
    for x in range(1, p):
        if pow(x, (p - 1) // 2, p) == 1:
            a[x] = 1.0
    a[0] = 1.0  # 0 is a quadratic residue
    s = a.sum()
    if s < 1e-10:
        return np.ones(p) / p
    return a / s


def _welch_init(n, rng):
    """Random phase sequence, squared magnitude as non-negative weights."""
    phases = rng.standard_normal(n)
    a = 1.0 + 0.5 * np.cos(phases * 2 * np.pi)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, active-set equalization,
    direct KKT solve, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=200.0, n_betas=16)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Active-set equalization
    print(f"\n[Phase 2] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        dl2 = min(deadline - 100, time.monotonic() + 60)
        a_eq, sc_eq = _active_equalize(best_seq, dl2, max_iter=3000, tol=1e-13)
        update_best(a_eq, sc_eq, "active_equalize")

    # Phase 3: Active-set Newton
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        dl3 = min(deadline - 80, time.monotonic() + 50)
        a_ns, sc_ns = _active_set_newton(best_seq, dl3, max_iter=300, tol_active=0.002)
        update_best(a_ns, sc_ns, "newton_1")

        if time.monotonic() < deadline - 60:
            dl3b = min(deadline - 40, time.monotonic() + 30)
            a_ns2, sc_ns2 = _active_set_newton(best_seq, dl3b, max_iter=200, tol_active=0.001)
            update_best(a_ns2, sc_ns2, "newton_tight")

    # Phase 4: KKT direct solve
    print(f"\n[Phase 4] KKT direct solve (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        c_best = _full_convolve(best_seq, best_seq)
        c_max = float(c_best.max())
        active = c_best >= (c_max - 0.01 * c_max)
        if active.sum() >= 2:
            dl4 = min(deadline - 60, time.monotonic() + 40)
            a_kkt, sc_kkt = _kkt_solve(best_seq, active, dl4, max_cg=300)
            update_best(a_kkt, sc_kkt, "kkt_direct")

    # Phase 5: Another L-BFGS round with tighter betas
    print(f"\n[Phase 5] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_r2 = _run_lbfgs(best_seq, deadline, min_time_remaining=40.0, n_betas=12)
        update_best(a_r2, _score(a_r2), "lbfgs_r2")

        # Active equalize after
        if time.monotonic() < deadline - 30:
            dl5 = min(deadline - 20, time.monotonic() + 30)
            a_eq2, sc_eq2 = _active_equalize(a_r2, dl5, max_iter=2000, tol=1e-13)
            update_best(a_eq2, sc_eq2, "equalize_r2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_list = [4096, 8192, 16384]
    if n_parent not in n_list:
        n_list.insert(0, n_parent)

    for n_try in n_list:
        if time.monotonic() > deadline - 30:
            break
        if n_try == n_parent and best_score < 1.51:
            continue

        if parent is not None:
            a_init_n = _interpolate_to_n(parent, n_try)
        else:
            a_init_n = _make_profile(n_try, "uniform")

        sc_init = _score(a_init_n)
        a_opt = _run_lbfgs(a_init_n, deadline, min_time_remaining=25.0, n_betas=10)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick active equalize
        if time.monotonic() < deadline - 15:
            dl6 = min(deadline - 10, time.monotonic() + 15)
            a_eqn, sc_eqn = _active_equalize(a_opt, dl6, max_iter=1000, tol=1e-12)
            update_best(a_eqn, sc_eqn, f"n_{n_try}_eq")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        restarts = [
            ("perturb_003", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_008", lambda: _perturb(best_seq, 0.06, rng)),
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=7)),
            ("perturb_015", lambda: _perturb(best_seq, 0.12, rng)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 10:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=8.0, n_betas=8)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # Phase 8: Structured initializations
    print(f"\n[Phase 8] Structured inits (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 15:
        # Legendre sequence (prime length)
        primes_near = [8191, 4093, 2053]
        for p in primes_near:
            if time.monotonic() > deadline - 10:
                break
            a_leg = _legendre_init(p)
            # Pad to power of 2 for efficiency
            n_pad = _next_pow2(p)
            a_padded = np.zeros(n_pad)
            a_padded[:p] = a_leg
            a_padded = a_padded / a_padded.sum()
            a_opt = _run_lbfgs(a_padded, deadline, min_time_remaining=5.0, n_betas=6)
            update_best(a_opt, _score(a_opt), f"legendre_p{p}")

    # Phase 9: Final precision polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Tightest active-set Newton
        dl9 = min(deadline - 2, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(best_seq, dl9, max_iter=300, tol_active=0.0005)
        update_best(a_fin, sc_fin, "final_newton")

        # Active equalize
        if time.monotonic() < deadline - 2:
            dl9b = min(deadline - 1, time.monotonic() + 20)
            a_fin2, sc_fin2 = _active_equalize(a_fin, dl9b, max_iter=2000, tol=1e-14)
            update_best(a_fin2, sc_fin2, "final_eq")

        # Tiny-beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_fin = _full_convolve(best_seq, best_seq)
            mb = float(c_fin.max())
            n_fin = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n_fin),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n_fin),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-15, 'maxcor': 60, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
