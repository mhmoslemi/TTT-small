"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: adaptive L-BFGS with log-sum-exp smoothing, FISTA acceleration,
peak-repulsion active-set refinement, structured (chirp/quadratic) initializations,
aggressive multi-n exploration, and zero-pattern local search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025, 0.000125])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 14))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _fista_smooth(a_init, deadline, min_time_remaining=30.0, n_iter_max=20000):
    """FISTA (Nesterov accelerated) on the smoothed objective in log-space."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap, max_c * 0.003)

    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Use moderate beta for FISTA
    beta = max_c * 0.005

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    x = z.copy()
    x_prev = z.copy()
    t = 1.0
    alpha = 1.0 / (4.0 * n * max_c / beta + 1e-10)
    alpha = min(alpha, 0.1)

    n_iter = 0
    no_improve = 0
    best_z = z.copy()
    best_sc = current_sc

    while n_iter < n_iter_max:
        if time.monotonic() + min_time_remaining > deadline:
            break

        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t_prev - 1.0) / t

        y = x + momentum * (x - x_prev)
        x_prev = x.copy()

        g = _smograd(y, beta, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step = alpha
        obj_y = _smoobj(y, beta, n)
        for _ in range(30):
            x_try = y - step * g
            obj_try = _smoobj(x_try, beta, n)
            if obj_try <= obj_y + step * np.dot(g, x_try - y) + 0.5 * step * step * gnorm * gnorm * 1.1:
                x = x_try
                break
            step *= 0.5
        else:
            x = y - step * g

        n_iter += 1
        if n_iter % 200 == 0:
            a_check = _softmax(x)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-14:
                best_sc = sc_check
                best_z = x.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 30:
                    break

    a_final = _softmax(best_z)
    sc_final = _score(a_final)
    if sc_final < current_sc:
        return a_final
    return current_a


def _peak_repulsion(a_init, deadline, n_repel=10, lr_init=0.08, min_time_remaining=10.0):
    """Directly reduce values that contribute to the top convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c.max())
    lr = lr_init
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        # Find top-n_repel peaks
        top_idx = np.argsort(c)[::-1][:n_repel]
        tol = c_max * 0.002
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[top_idx[0]] = True

        # Compute pressure on each a[i]
        pressure = np.zeros(n)
        a_clipped = np.clip(a, 0, None)
        for k in np.where(active)[0]:
            # c[k] = sum_i a[i] * a[k-i]
            # dc[k]/da[i] = 2*a[k-i] (roughly, for the symmetric part)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]
                    if i != j:
                        pass  # already counted via symmetry in the sum

        pressure = pressure / max(active.sum(), 1)
        pnorm = np.linalg.norm(pressure)
        if pnorm < 1e-15:
            break

        # Apply repulsion
        a_new = a - lr * pressure
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc - 1e-15:
            a = a_new
            lr = min(lr_init, lr * 1.03)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00003, lr_init=0.05, min_time_remaining=10.0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.65
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.02)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline, min_time_remaining=10.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.008, 0.012, 0.018, 0.025, 0.035, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        # U-shaped: high at edges, low in center
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        # Quadratic phase -> good autocorrelation properties
        t = np.arange(n)
        a = 1.0 + 0.3 * np.abs(np.sin(np.pi * t * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.2) ** 2) + np.exp(-8.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tripolar":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.15) ** 2) + np.exp(-12.0 * (t - 0.5) ** 2) + np.exp(-12.0 * (t - 0.85) ** 2) + 0.03
        return a / a.sum()
    elif kind == "edge_conc":
        # Concentrated at edges
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * t ** 2) + np.exp(-20.0 * (1 - t) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 6))
        amp = 0.04 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _zero_pattern_search(a, deadline, min_time_remaining=20.0):
    """Try random zero patterns to see if sparsity helps."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    n_trials = max(10, int((deadline - time.monotonic() - min_time_remaining) / 0.05))
    n_trials = min(n_trials, 500)

    for _ in range(n_trials):
        if time.monotonic() + min_time_remaining > deadline:
            break
        # Try different sparsity levels
        sparsity = rng.uniform(0.05, 0.3)
        n_zero = int(n * sparsity)
        a_try = a.copy()
        zeros = rng.choice(n, size=n_zero, replace=False)
        a_try[zeros] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc

    if time.monotonic() < deadline - 150:
        a_shr, _ = _shrinkage_optimize(best_seq, deadline, min_time_remaining=100.0)
        _update_best(a_shr, "shrinkage")
        if time.monotonic() < deadline - 100:
            a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=60.0)
            _update_best(a_shr2, "lbfgs_post_shr")
        if time.monotonic() < deadline - 50:
            a_shr3, _ = _active_set_polish(best_seq, deadline - 30,
                                           tol_frac_start=0.001, tol_frac_end=0.00002,
                                           lr_init=0.01, min_time_remaining=20.0)
            _update_best(a_shr3, "polish_post_shr")

    # === Phase 7: Multi-n exploration (aggressive) ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192, 10240, 12288, 16384]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        if n_try > 16384:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick L-BFGS
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0)
        _update_best(a_opt, f"multi_n_{n_try}")
        # Quick polish if close
        if time.monotonic() < deadline - 30:
            a_opt2, _ = _active_set_polish(a_opt, deadline - 15,
                                           tol_frac_start=0.003, tol_frac_end=0.0001,
                                           lr_init=0.012, min_time_remaining=10.0)
            _update_best(a_opt2, f"multi_n_{n_try}_pol")

    # === Phase 8: Structured restarts ===
    print(f"\n[P8] Structured restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        restarts = [
            ("pert3", _perturb(best_seq, 0.03, rng)),
            ("pert8", _perturb(best_seq, 0.08, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("quadratic", _make_profile(n_best, "quadratic")),
            ("chirp", _make_profile(n_best, "chirp")),
            ("edge_conc", _make_profile(n_best, "edge_conc")),
            ("tripolar", _make_profile(n_best, "tripolar")),
            ("peri11", _make_profile(n_best, "periodic")),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 25:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=15.0)
            _update_best(a_opt, f"restart_{label}")

    # === Phase 9: Zero pattern search ===
    print(f"\n[P9] Zero pattern search (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        a_zp, _ = _zero_pattern_search(best_seq, deadline, min_time_remaining=15.0)
        _update_best(a_zp, "zero_pattern")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, _ = _active_set_polish(best_seq, deadline - 3,
                                      tol_frac_start=0.0005, tol_frac_end=0.00001,
                                      lr_init=0.005, min_time_remaining=2.0)
        _update_best(a_fin, "final_polish")

        remaining = deadline - 3 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(30, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                _update_best(a_fin2, "final_lbfgs")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
