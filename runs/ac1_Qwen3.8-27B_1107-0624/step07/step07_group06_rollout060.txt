def _proj_simplex(v):
    """Project v onto the simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            topk_idx = np.argpartition(c, -kk)[-kk:]
            w = np.zeros(len(c))
            w[topk_idx] = 1.0 / kk
        grad_a = 2.0 * _hankel_matvec(a, w, n)
        lr = lr0 / (1.0 + 0.01 * i)
        z = z - lr * grad_a
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _newton_active_set(a_init, deadline, tol_frac=0.002, max_iter=200):
    """Newton's method on active-set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient of sum active C[k]: grad[m] = 2 * sum_j a[j] * active[j+m]
        grad = _hankel_matvec(a, active.astype(np.float64), n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        # Build Hessian matvec: (Hv)[m] = 2 * sum_j v[j] * active[mj] where mj = m+j
        N = _next_pow2(4 * n)
        fa = np.fft.rfft(active.astype(np.float64), N)

        def hess_vec(v):
            v_rev = v[::-1]
            fv = np.fft.rfft(v_rev, N)
            r = np.fft.irfft(fv * fa, N)
            return 2.0 * r[n - 1:2 * n - 1]

        # CG to solve (H + mu*I) dx = -grad_t
        mu = 1e-6
        x = np.zeros(n)
        r = -grad_t.copy()
        p = r.copy()
        rs = float(np.dot(r, r))
        b_norm = float(np.linalg.norm(r))
        if b_norm < 1e-30:
            break

        for cg_it in range(150):
            Hp = hess_vec(p) + mu * p
            pH = float(np.dot(p, Hp))
            if abs(pH) < 1e-30:
                break
            alpha = rs / pH
            if not np.isfinite(alpha) or alpha < 0:
                break
            x = x + alpha * p
            r = r - alpha * Hp
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-10 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p = r + beta_cg * p
            rs = r_new

        dx = x - np.mean(x)

        # Line search
        f_old = float(c[active].sum())
        alpha_ls = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha_ls * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha_ls *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-10 * alpha_ls * gnorm:
                a_new = a_try
                break
            alpha_ls *= 0.5
        else:
            if alpha_ls < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _frank_wolfe(a_init, deadline, n_steps=5000):
    """Frank-Wolfe (conditional gradient) on the max of C."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _hankel_matvec(a, w, n)

        # FW: minimum of grad_a over simplex is at the vertex with min grad_a
        m_star = int(np.argmin(grad_a))
        s_vertex = np.zeros(n)
        s_vertex[m_star] = 1.0

        # Step size
        gap = float(np.dot(grad_a, a - s_vertex))
        if gap < 1e-14:
            break
        t = 2.0 / (i + 2.0)
        a_new = (1.0 - t) * a + t * s_vertex
        a = a_new

        if i % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    a_final = a
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc
