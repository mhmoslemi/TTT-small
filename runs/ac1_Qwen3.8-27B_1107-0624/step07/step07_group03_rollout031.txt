"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Log-sum-exp smoothed L-BFGS with adaptive beta
- Active-set KKT solver using MINRES with FFT-based Hankel matvec
- Multi-n exploration (8192, 16384, 32768)
- Asymmetric profile initialization and perturbations
- Final coordinate-descent polish
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse.linalg import minres as _minres_sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hankel_matvec(w, v, n):
    """Compute H*v where H[i,j] = w[i+j] for i,j in [0,n-1].
    w has length 2*n-1 (indices 0..2n-2).
    Uses FFT-based cross-correlation.
    """
    N = _next_pow2(2 * (2 * n - 1))
    W = np.fft.rfft(w, N)
    V = np.fft.rfft(v, N)
    corr = np.fft.irfft(np.conj(W) * V, N)
    return corr[:n]


def _active_set_kkt_solve(a_init, active_set, n, max_iter=2000, tol=1e-10):
    """Solve the KKT system for the active set using MINRES.
    System: M*a = m*lambda*1, 1^T*a = 1
    where M[i,j] = 1 if i+j in active_set.
    Returns solution a or None if failed.
    """
    m = len(active_set)
    if m == 0:
        return None
    active_indicator = np.zeros(2 * n - 1)
    active_indicator[active_set] = 1.0

    def augmented_matvec(x_aug):
        x = x_aug[:n]
        s = x_aug[n]
        # First n components: M*x - m*s*1
        Mx = _hankel_matvec(active_indicator, x, n)
        out = np.empty(n + 1)
        out[:n] = Mx - m * s
        out[n] = x.sum()
        return out

    rhs = np.zeros(n + 1)
    rhs[n] = 1.0

    x0 = np.zeros(n + 1)
    x0[:n] = a_init
    x0[n] = float(a_init.sum()) / m

    try:
        sol, info = _minres_sp(
            augmented_matvec, rhs, x0=x0,
            tol=tol, maxiter=max_iter, restart=80
        )
    except Exception:
        return None

    if info != 0:
        # Try with more iterations or different tol
        try:
            sol, info = _minres_sp(
                augmented_matvec, rhs, x0=x0,
                tol=tol * 10, maxiter=max_iter * 2, restart=50
            )
        except Exception:
            return None

    a_sol = sol[:n]
    if np.any(a_sol < -1e-10):
        # Small negative components - clip and check
        if np.min(a_sol) > -1e-6:
            a_sol = np.maximum(a_sol, 0)
            s = a_sol.sum()
            if s > 1e-15:
                a_sol = a_sol / s
            return a_sol
        return None

    a_sol = np.maximum(a_sol, 0)
    s = a_sol.sum()
    if s < 1e-15:
        return None
    return a_sol / s


def _active_set_optimize(a_init, deadline, max_outer=50, tol_active_start=0.01, tol_active_end=0.001):
    """Iterative active-set optimization.
    Alternates between:
    1. Identifying active set from current convolution
    2. Solving KKT system on that active set
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Tolerance interpolates from start to end
        frac = outer / max(1, max_outer - 1)
        tol_act = tol_active_start + (tol_active_end - tol_active_start) * frac
        active_mask = c >= (c_max - tol_act * c_max)
        n_active = int(active_mask.sum())

        if n_active == 0:
            active_mask[int(np.argmax(c))] = True
            n_active = 1

        # Cap active set size to avoid ill-conditioning
        max_active = min(2 * n, n_active)
        if n_active > max_active:
            active_mask = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:max_active]
            active_mask[top_idx] = True
            n_active = max_active

        active_set = np.where(active_mask)[0]

        # Try KKT solve
        a_new = _active_set_kkt_solve(a, active_set, n, max_iter=1000, tol=1e-12)

        if a_new is not None:
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                a = a_new
                best_a = a_new.copy()
                best_sc = sc_new
            elif sc_new < best_sc + 1e-10:
                a = a_new
            # Check if active set is stable
            c_check = _full_convolve(a, a)
            c_max_check = float(c_check.max())
            active_check = c_check >= (c_max_check - 1e-8 * c_max_check)
            if np.array_equal(active_check, active_mask):
                # Converged
                break
        else:
            # KKT solve failed - fall back to gradient step
            w = active_mask.astype(np.float64) / n_active
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Projected gradient step
            t = 0.01
            a_try = a - t * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_a = a_try.copy()
                    best_sc = sc_try

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, n_sweeps=10):
    """Simple coordinate descent: adjust pairs of elements to reduce max convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        improved = False

        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() > deadline:
                break
            # Try small perturbation of a[i] and a[i+1]
            if i + 1 >= n:
                continue
            delta = 0.01 * a[i]
            for sign in [1, -1]:
                a_try = a.copy()
                a_try[i] += sign * delta
                a_try[i + 1] -= sign * delta
                if a_try[i] < 0 or a_try[i + 1] < 0:
                    continue
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-12:
                    a = a_try
                    best_a = a_try.copy()
                    best_sc = sc_try
                    improved = True
                    break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tilt":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "tilt_strong":
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * t
        return a / a.sum()
    elif kind == "tilt_weak":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * t
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t + 0.3 * t * t
        return a / a.sum()
    elif kind == "exp":
        t = np.linspace(0, 1, n)
        a = np.exp(1.0 * t)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t * (0.7 + 0.3 * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tilt")
        print(f"No parent, using tilted n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Improve at native n
    print(f"\n[Phase 1] Improve at n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = deadline - 600.0
    if time.monotonic() < phase1_deadline:
        a_p1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=400.0)
        update_best(a_p1, _score(a_p1), "lbfgs_native")

        if time.monotonic() < phase1_deadline - 30:
            a_p1b, sc_p1b = _active_set_optimize(a_p1, phase1_deadline - 30,
                                                  max_outer=40, tol_active_start=0.005, tol_active_end=0.0005)
            update_best(a_p1b, sc_p1b, "activeset_native")

    # Phase 2: Explore larger n
    print(f"\n[Phase 2] Explore larger n (t={time.monotonic() - t_start:.1f}s)")
    n_targets = [16384, 32768]
    if n_parent not in [8192] and n_parent > 4096:
        n_targets.insert(0, n_parent * 2)

    # De-duplicate and sort
    n_targets = sorted(set(n_targets))

    for n_try in n_targets:
        if time.monotonic() > deadline - 120:
            break

        print(f"  Trying n={n_try}...")
        a_interp = _interpolate_to_n(parent, n_try)
        sc_interp = _score(a_interp)
        print(f"    Interpolated score: {sc_interp:.10f}")
        update_best(a_interp, sc_interp, f"interp_{n_try}")

        # L-BFGS on interpolated
        phase2_dl = deadline - max(60, (len(n_targets) - n_targets.index(n_try)) * 120)
        if time.monotonic() < phase2_dl:
            a_opt = _run_lbfgs(a_interp, phase2_dl, min_time_remaining=60.0)
            update_best(a_opt, _score(a_opt), f"lbfgs_{n_try}")

            # Active-set refinement
            if time.monotonic() < deadline - 80:
                a_as, sc_as = _active_set_optimize(a_opt, deadline - 80,
                                                   max_outer=30, tol_active_start=0.005,
                                                   tol_active_end=0.0005)
                update_best(a_as, sc_as, f"as_{n_try}")

    # Phase 3: Diverse restarts
    print(f"\n[Phase 3] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        restarts = [
            ("tilt_perturb", lambda: _perturb_asymmetric(best_seq, rng, 0.08)),
            ("tilt_perturb2", lambda: _perturb_asymmetric(best_seq, rng, 0.15)),
            ("linear_bias", lambda: _make_profile(n_best, "tilt") * 0.3 + best_seq * 0.7),
            ("exp_mix", lambda: 0.4 * _make_profile(n_best, "exp") + 0.6 * best_seq),
        ]

        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            p_start = np.maximum(p_start, 0)
            s = p_start.sum()
            if s < 1e-15:
                continue
            p_start = p_start / s
            sc_start = _score(p_start)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # Phase 4: Try different n with asymmetric profiles
    print(f"\n[Phase 4] Asymmetric profiles at various n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [4096, 8192]:
        if time.monotonic() > deadline - 40:
            break
        for kind in ["tilt", "tilt_strong", "exp"]:
            if time.monotonic() > deadline - 20:
                break
            a_prof = _make_profile(n_try, kind)
            sc_prof = _score(a_prof)
            if sc_prof < best_score:
                update_best(a_prof, sc_prof, f"profile_{kind}_{n_try}")
                # Optimize from this profile
                a_opt = _run_lbfgs(a_prof, deadline, min_time_remaining=15.0)
                update_best(a_opt, _score(a_opt), f"opt_{kind}_{n_try}")

    # Phase 5: Final polish
    print(f"\n[Phase 5] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_final = len(best_seq)

        # Active-set with very tight tolerance
        if time.monotonic() < deadline - 10:
            a_fin, sc_fin = _active_set_optimize(best_seq, deadline - 10,
                                                 max_outer=30, tol_active_start=0.002,
                                                 tol_active_end=0.0002)
            update_best(a_fin, sc_fin, "final_as_tight")

        # Coordinate descent polish
        if time.monotonic() < deadline - 5:
            a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 5, n_sweeps=5)
            update_best(a_cd, sc_cd, "coord_descent")

        # One more tiny-beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 4 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n_final),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n_final),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 100, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tilt")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
