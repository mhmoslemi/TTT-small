def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_max(c, beta):
    c_max = c.max()
    if beta < 1e-15:
        return c_max
    shifted = (c - c_max) / beta
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    s = e.sum()
    return c_max + beta * np.log(s)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    val = _logsumexp_max(c, beta)
    return 2.0 * n * val


def _smograd(z, beta, n):

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-15)
    return a_interp / a_interp.sum()


def _perturb(a, rng, scale=0.01):
