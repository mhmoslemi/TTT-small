        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration
        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 10))))

        def obj(z):
            return _smoothed_obj(z, beta, n)

        def grad(z):
            return _smoothed_grad(z, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                    'eps': 1e-15,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception as e:
            print(f"  L-BFGS error at beta={beta:.6f}: {e}")

    return current_a


def _nesterov_gradient_descent(a_init, deadline, 
                                tol_start=0.01, tol_end=0.00005,
                                lr_init=0.02, momentum_init=0.9):
    """
    Nesterov-accelerated projected gradient descent with adaptive step size.
    Operates in the simplex (a >= 0, sum(a) = 1).
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    # Nesterov momentum state
    y = a.copy()  # lookahead point
    v = np.zeros(n)  # velocity
    momentum = momentum_init
    
    lr = lr_init
    tol_frac = tol_start
    no_improve = 0
    max_stall = 200
    
    # Adaptive step size (Barzilai-Borwein-like)
    prev_a = a.copy()
    prev_grad = None
    
    iter_count = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        # Compute gradient at y (Nesterov lookahead)
        c = _full_convolve(y, y)
        c_max = float(c.max())
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        
        # Gradient of max(c) w.r.t. a (on the simplex)
        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        # grad[i] = 2*h[i] - 2*W where W = sum(w*c)
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)
        # Center the gradient (project onto sum=0 subspace)
        grad = grad - np.mean(grad)
        
        # Nesterov step
        v = momentum * v + grad
        y_new = y - lr * v
        
        # Project to simplex (a >= 0, sum = 1)
        a_new = np.maximum(y_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())
        
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Adaptive learning rate
            if no_improve % 20 == 0:
                lr *= 0.8
                momentum = max(0.5, momentum * 0.95)
            
            if no_improve >= max_stall:
                # Restart from best
                a = best_a.copy()
                y = a.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5
                momentum = momentum_init
                tol_frac = max(tol_end, tol_frac * 0.5)
                no_improve = 0
                if tol_frac < tol_end * 1.1:
                    break
        
        a = a_new  # hmm, this should only be set if we accept
        # Actually, for Nesterov, we should always move y forward
        # Let me restructure
        
        # Update active set tolerance
        if no_improve == 0 and tol_frac > tol_end:
            tol_frac = max(tol_end, tol_frac * 0.98)
        
        iter_count += 1
        if iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NGD iter {iter_count}: score={best_sc:.10f}, "
                  f"lr={lr:.6f}, tol={tol_frac:.8f}, t={elapsed:.1f}s")
    
    return best_a, _score(best_a)


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    """Original active-set polish (kept for comparison)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 400
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc_cur = 2.0 * n * c_max

        if sc_cur < best_sc - 1e-14:
            best_sc = sc_cur
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False

            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)

            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < 2.0 * n * c_max - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.75)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr
