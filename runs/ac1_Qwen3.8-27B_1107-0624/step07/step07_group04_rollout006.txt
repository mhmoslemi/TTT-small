"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14, reg=0.0):
    """Solve (H + reg*I)*x = rhs using CG, H is Hankel with entries 2*W[j+m]."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp < 0:
            # Non-PSD; use gradient step instead
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs:
            # Restart
            p = r_new.copy()
            rs = float(p @ p)
            if rs < 1e-30:
                break
            x = x_new
            r = r_new
            continue
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_polish(a_init, deadline, max_rounds=200, tight=False):

    n = len(a_init)

            break

        c_max = float(c.max())

        active = c >= (c_max - tol)

        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for computational reasons
        if n_active > 2 * n:
            # This shouldn't happen since c has length 2n-1
            pass

        # Build W vector (uniform weights on active set)
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        # Try Newton step via CG with increasing regularization
        step_done = False
        for reg_factor in [0, 1e-8, 1e-6, 1e-4, 1e-2]:
            reg = reg_factor * np.max(g)
            Hg = _cg_solve(W, n, g_t, max_iter=300, reg=reg)
            H1 = _cg_solve(W, n, np.ones(n) / n, max_iter=300, reg=reg)
            H1_norm = float(np.linalg.norm(H1))
            if H1_norm < 1e-20 or not np.isfinite(H1_norm):
                continue
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1

                a_try = a + alpha * dx
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                c_max_try = float(c_try.max())
                if c_max_try < c_max - 1e-4 * alpha * gn:
                    a = a_try
                    step_done = True
                    break
                alpha *= 0.5
                if alpha < 1e-14:
                    break
            if step_done:
                break

        if not step_done:
            # Gradient step fallback
            alpha = 1.0 / (gn + 1e-30)
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)

                    a = a_try
                    step_done = True
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Early stopping if progress is negligible
        if best_sc - sc < 1e-16 * best_sc and rnd > 10:
            break

    return best_a, best_sc

def _smoobj(z, beta, n):

    c = _full_convolve(a, a)

def _smograd(z, beta, n):

    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        # Scale iterations based on remaining time and problem size
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        if aggressive:
            max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))
        else:
            max_iter = min(3000, max(30, int(remaining / (est_iter_time * 12))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            # If no improvement, still update z to the L-BFGS solution
            # (it might help for the next beta)
            elif i < len(betas) - 5:
                z = np.log(np.maximum(_softmax(result.x), 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _kkt_check(a, tol=0.001):
    """Check KKT conditions. Return the active set and weights if KKT is approximately satisfied."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Active set
    active = c >= (c_max - tol * c_max)
    if not active.any():
        active[int(np.argmax(c))] = True
    
    # Compute gradient on active set
    W = np.zeros(2 * n - 1)
    W[active] = 1.0 / active.sum()
    g = _hankel_matvec(a, W, n)
    
    # KKT: g should be constant where a > 0, and g >= const where a = 0
    support = a > 1e-10
    if support.sum() < 2:
        return c_max, active
    
    g_support = g[support]
    g_mean = np.mean(g_support)
    g_std = np.std(g_support)
    
    # KKT residual
    kkt_res = g_std / (np.abs(g_mean) + 1e-30)
    
    return kkt_res, active


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "flat_centered":
        a = np.ones(n)
        center = n // 2
        width = n // 4
        a[center - width: center + width] = 0.5
        return a / a.sum()
    elif kind == "u_shaped":
        t = np.linspace(0, 1, n)
        a = (t - 0.5)**2 + 0.1
        return a / a.sum()
    elif kind == "power":
        t = np.linspace(0, 1, n)
        a = t**0.3 + (1-t)**0.3 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.10, 0.20]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc
