"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Nesterov-accelerated gradient descent on log-sum-exp smoothed objective
- Adaptive temperature with backtracking line search
- Multi-scale length exploration (4096, 8192, 16384)
- Coordinate descent polishing
- Multi-restart with diverse perturbations
"""
import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def _fft_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    out_n = 2 * n - 1
    N = next_fast_len(out_n)
    fa = fft(a, N)
    return np.real(ifft(fa * fa, N)[:out_n])

def _score(a):
    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)

def _score_unit(a):
    """Score for unit-sum vector: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_conv(a)
    return 2.0 * n * float(c.max())

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = css[rho - 1] / rho
    return np.maximum(v - theta, 0.0)

def _smoothed_grad(a, c, T):
    """
    Compute smoothed gradient of log-sum-exp of convolution.
    Returns gradient of (1/T)*log(sum_k exp(c_k/T)) with respect to a.
    """
    n = len(a)
    max_c = float(c.max())
    shifted = (c - max_c) / T
    shifted = np.clip(shifted, -100, 0)
    p = np.exp(shifted)
    p_sum = float(p.sum())
    if p_sum < 1e-300:
        # Degenerate: just use argmax
        k = int(np.argmax(c))
        g = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        g[lo:hi] = 2.0 * a[k-lo:k-lo+(hi-lo)][::-1]
        return g / T
    p /= p_sum
    g = np.zeros(n)
    # g[m] = (2/T) * sum_k p[k] * a[k-m] for valid indices
    # Only iterate over significant p
    thresh = 0.5 / len(c)  # at least this many expected
    active_k = np.where(p > thresh)[0]
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        seg = a[k-lo:k-lo+(hi-lo)]
        g[lo:hi] += 2.0 * p[k] * seg[::-1]
    return g / T

def _active_set_grad(a, c, top_k=3):
    """Compute gradient using only top_k peaks (equal weight)."""
    n = len(a)
    idx = np.argsort(c)[::-1][:top_k]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        seg = a[k-lo:k-lo+(hi-lo)]
        g[lo:hi] += (2.0 / top_k) * seg[::-1]
    return g

def _nesterov_optimize(a_init, deadline, T_init=1e-3, T_min=1e-5, 
                        lr_init=0.01, max_iter=200000):
    """
    Nesterov-accelerated gradient descent on the smoothed objective.
    Works in the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = _score_unit(a)
    t_k = 1.0
    T = T_init
    lr = lr_init
    patience = 0
    best_patience = best_score
    
    c = _fft_conv(y)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Decay temperature
        T = max(T_min, T_init * (0.999 ** it))
        # Decay learning rate
        lr = lr_init / (1.0 + 0.001 * it)
        
        # Compute gradient at y
        if T > T_min * 10:
            g = _smoothed_grad(y, c, T)
        else:
            g = _active_set_grad(y, c, top_k=3)
        
        # Nesterov step
        step = lr / (np.linalg.norm(g) + 1e-10)
        y_new = y - step * g
        a_new = _project_simplex(y_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - y)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        
        # Update convolution
        c = _fft_conv(y)
        
        # Track best
        if it % 100 == 0:
            sc = _score_unit(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            
            # Learning rate adaptation
            if patience > 10:
                lr *= 0.7
                T = max(T_min, T * 0.7)
                patience = 0
                # Reset Nesterov
                t_k = 1.0
                y = best_a.copy()
    
    sc = _score_unit(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def _cd_polish(a, deadline, max_rounds=100):
    """
    Coordinate descent polishing: try to reduce the max convolution peak
    by adjusting individual coordinates.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_unit(best_a)
    c = _fft_conv(best_a)
    c_max = float(c.max())
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        
        # Find the peak location(s)
        peak_k = int(np.argmax(c))
        
        # For the peak at position k, the gradient wrt a[j] is 2*a[k-j]
        # We want to decrease the peak, so we should increase a[j] where 
        # a[k-j] is small, or decrease a[j] where a[k-j] is large
        # But we need to maintain sum(a) = 1
        
        # Strategy: find j that most reduces the peak when moved
        # Damping the peak: reduce a[j] for j in the support of the peak
        lo = max(0, peak_k - n + 1)
        hi = min(peak_k + 1, n)
        
        # Try moving mass from high-contributing positions to low ones
        contrib = c[peak_k]  # current peak value
        # Contribution of a[j] to peak: 2*a[j]*a[peak_k-j] (approximately, for small changes)
        
        # Try: reduce a[j] for j in [lo, hi) where a[peak_k-j] is large
        # and increase a[j'] for j' outside [lo, hi) where a[peak_k-j'] would be small
        
        # Simple heuristic: try reducing top contributors
        if hi > lo:
            seg = a[lo:hi].copy()
            # Weight by a[peak_k - j] (the conjugate)
            conjugate = np.array([a[peak_k - j] for j in range(lo, hi) if 0 <= peak_k - j < n])
            if len(conjugate) == (hi - lo):
                weights = seg * conjugate
                # Find the largest contributor
                j_idx = lo + int(np.argmax(weights))
                if best_a[j_idx] > 1e-6:
                    # Try reducing it
                    for frac in [0.7, 0.5, 0.3, 0.1]:
                        a_cand = best_a.copy()
                        old_val = a_cand[j_idx]
                        a_cand[j_idx] *= frac
                        # Redistribute to a random position with low contribution
                        # Find position with minimum a[peak_k - j]
                        min_pos = lo + int(np.argmin(conjugate))
                        a_cand[min_pos] += old_val * (1 - frac)
                        # Re-normalize
                        s = a_cand.sum()
                        if s > 1e-15:
                            a_cand /= s
                            c_cand = _fft_conv(a_cand)
                            sc = 2.0 * n * float(c_cand.max())
                            if sc < best_score * (1 - 1e-10):
                                best_score = sc
                                best_a = a_cand.copy()
                                c = c_cand
                                improved = True
                                break
        
        if not improved:
            # Try random coordinate changes
            j = np.random.randint(0, n)
            for scale in [0.5, 0.7, 1.3, 2.0]:
                a_cand = best_a.copy()
                a_cand[j] *= scale
                s = a_cand.sum()
                if s > 1e-15:
                    a_cand /= s
                    c_cand = _fft_conv(a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        improved = True
                        break
        
        if not improved and round_num > 20:
            break
    
    return best_a, best_score

def _interp_to_length(a, n_new):
    """Interpolate a from its current length to n_new."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()
