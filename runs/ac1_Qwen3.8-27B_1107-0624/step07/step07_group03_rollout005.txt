"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.
Focuses on aggressive active-set Newton with multi-scale tolerance refinement,
diverse parametric initializations, and coordinate descent polishing.
"""
import numpy as np
import time
import math
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b if n == m else a, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autocorr(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _autocorr(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _corr_vec(a, A_vec, n):
    """Compute g[i] = sum_j a[j] * A_vec[i+j] for i=0..n-1."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_matvec(v, A_vec, n, mu):
    """Hessian-vector product: (2/m)*corr(v, A) + mu*v"""
    return (2.0 / len(A_vec[A_vec > 0])) * _corr_vec(v, A_vec, n) + mu * v


def _cg_solve(rhs, matvec_fn, max_cg=500, tol=1e-10):
    """Conjugate Gradient solver."""
    n = len(rhs)
    x = np.zeros(n)
    r = rhs - matvec_fn(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_cg):
        Ap = matvec_fn(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < tol * (1.0 + rhs_norm):
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.001, verbose=False):
    """
    Damped Newton method on the active set.
    Minimizes mean of active autocorrelation values subject to sum=1, a>=0.
    """
    n = len(a_init)
    a = _normalize(np.maximum(a_init, 0.0))
    best_a = a.copy()
    best_sc = _score(a)
    c_best = float(_autocorr(best_a).max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _autocorr(a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        if n_active > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 4 * n

        A_vec = active.astype(np.float64)
        m = float(n_active)

        # Gradient: g[i] = (2/m) * sum_j a[j] * A_vec[i+j]
        g = (2.0 / m) * _corr_vec(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Hessian with regularization
        mu = 1e-8 * c_max + 1e-10

        def matvec(v):
            return (2.0 / m) * _corr_vec(v, A_vec, n) + mu * v

        # CG solve for Newton direction
        u = _cg_solve(g_t, matvec, max_cg=300, tol=1e-10)

        # Handle the equality constraint sum(a)=1
        ones = np.ones(n)
        w = _cg_solve(ones, matvec, max_cg=200, tol=1e-10)
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-14:
            dx = -u
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w

        dx = dx - np.mean(dx)

        # Line search with non-negativity
        f_old = float(c.max())
        neg_mask = dx < -1e-14
        if neg_mask.any():
            t_max = 0.95 * np.min(a[neg_mask] / (-dx[neg_mask]))
        else:
            t_max = 2.0
        t_max = max(t_max, 1e-12)

        accepted = False
        t = min(1.0, t_max)
        for ls in range(60):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-14:
                t *= 0.5
                if t < 1e-15:
                    break
                continue
            a_try = a_try / s_try
            c_try = _autocorr(a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5
            if t < 1e-15:
                break

        if not accepted:
            # Gradient fallback
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-14:
                    t *= 0.5
                    if t < 1e-15:
                        break
                    continue
                a_try = a_try / s_try
                c_try = _autocorr(a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            c_best = float(_autocorr(best_a).max())

        if verbose and it % 50 == 0:
            print(f"    ASN iter {it}: score={sc:.10f}, c_max={float(c.max()):.2e}, "
                  f"n_active={n_active}, gnorm={gnorm:.2e}")

        if abs(float(_autocorr(a).max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing and beta continuation."""
    n = len(a_init)
    c_cur = _autocorr(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                  0.01, 0.005, 0.002, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsobj_grad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lsobj_grad(z, beta, n):
    a = _softmax(z)
    c = _autocorr(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _coord_descent(a_init, deadline, max_rounds=200):
    """
    Coordinate descent: for each coordinate i, find the best value
    that minimizes the max autocorrelation (keeping others fixed).
    """
    n = len(a_init)
    a = _normalize(np.maximum(a_init, 0.0))
    best_a = a.copy()
    best_sc = _score(a)

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _autocorr(a)
        c_max = float(c.max())
        improved = False

        # Process coordinates in order of their current value (smallest first)
        order = np.argsort(a)

        for i in order:
            if time.monotonic() > deadline:
                break

            a_old = a[i]
            # The contribution of a[i] to autocorrelation at lag k:
            # delta_conv[k] = a[i] * (a[k-i] + a[i-k]) - a[i]^2 * delta(k, 2i)
            # We want to find a[i] that minimizes the new max autocorrelation.

            # Gather the "other" contributions to each lag from a[i]
            # conv_new[k] = conv_old[k] - a_old*(a[k-i]+a[i-k]) + a_new*(a[k-i]+a[i-k])
            #              = conv_old[k] + (a_new - a_old)*(a[k-i] + a[i-k])

            # For lag k, the coefficient of a_new is: a[k-i] + a[i-k] (with bounds)
            # Let coef[k] = a[k-i] + a[i-k] (treating out-of-bounds as 0)
            # Then conv_new[k] = c[k] + (a_new - a_old) * coef[k]

            # We want to minimize max_k conv_new[k] = max_k (c[k] + (a_new - a_old)*coef[k])
            # This is a 1D problem in a_new >= 0.

            # Build coef vector
            coef = np.zeros(2 * n - 1)
            # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
            lo1 = max(0, i)
            hi1 = min(2 * n - 2, i + n - 1)
            if lo1 <= hi1:
                idx1 = np.arange(lo1, hi1 + 1)
                coef[idx1] += a[np.clip(idx1 - i, 0, n - 1)]
            # a[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            lo2 = max(0, i - n + 1)
            hi2 = min(2 * n - 2, i)
            if lo2 <= hi2:
                idx2 = np.arange(lo2, hi2 + 1)
                coef[idx2] += a[np.clip(i - idx2, 0, n - 1)]

            # conv_new[k] = c[k] + (a_new - a_old) * coef[k]
            # = (c[k] - a_old * coef[k]) + a_new * coef[k]
            base = c - a_old * coef
            # We want min_{a_new >= 0} max_k (base[k] + a_new * coef[k])

            # The optimal a_new is where the max is minimized.
            # For a given a_new, the value is max_k (base[k] + a_new * coef[k]).
            # This is a piecewise linear function of a_new (convex, as max of linear functions).
            # We can find the minimum by ternary search or by finding the intersection.

            # Simple approach: ternary search on [0, a_old * 2]
            lo = 0.0
            hi = max(a_old * 2, c_max / (max(np.max(coef[coef > 0]), 1e-30)))
            hi = min(hi, 10.0)  # Cap the search range

            best_val = np.inf
            best_anew = a_old

            # Ternary search
            for _ in range(60):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = float(np.max(base + m1 * coef))
                v2 = float(np.max(base + m2 * coef))
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                if v1 < best_val:
                    best_val = v1
                    best_anew = m1
                if v2 < best_val:
                    best_val = v2
                    best_anew = m2

            if best_anew != a_old and best_anew >= 0:
                a[i] = best_anew
                a = _normalize(a)
                sc = _score(a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True

        if not improved and rd > 5:
            break

    return best_a, best_sc


def _parametric_init(n, family, param=1.0):
    """Generate parametric initial sequences."""
    t = np.arange(n) / n
    if family == "expo":
        a = np.exp(-param * t)
    elif family == "ramp":
        a = np.maximum(1.0 - param * t, 0.01)
    elif family == "cosine":
        a = np.cos(np.pi * t / 2.0) ** param
    elif family == "half_cos":
        a = np.where(t < 0.5, np.cos(np.pi * t), 0.0) + 0.01
    elif family == "gaussian":
        center = 0.7
        a = np.exp(-param * (t - center) ** 2)
    elif family == "bump_right":
        a = np.exp(-param * (1.0 - t) ** 3)
    elif family == "bump_left":
        a = np.exp(-param * t ** 3)
    elif family == "sinusoid":
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * param * t)
        a = np.maximum(a, 0.01)
    elif family == "asym_flat":
        a = np.where(t < 0.6, 1.0, np.maximum(1.0 - 3.0 * (t - 0.6), 0.01))
    else:
        a = np.ones(n) * 0.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _shrunk_and_reoptimize(a_init, deadline, min_time_remaining=50):
    """Try zeroing small elements then re-optimizing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    sorted_idx = np.argsort(a_init)

    for frac in [0.05, 0.10, 0.20, 0.30, 0.40]:
        if time.monotonic() > deadline - min_time_remaining:
            break
        n_zero = max(1, int(n * frac))
        a_try = a_init.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()

    return best_a, best_sc

    if best_seq is not None:

        tol_scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        a_current = best_seq.copy()

        for scale_i, tol_scale in enumerate(tol_scales):
            if time.monotonic() > phase2_deadline - 10:
                break
            time_per_scale = (phase2_deadline - 10 - time.monotonic()) / (len(tol_scales) - scale_i)
            scale_deadline = time.monotonic() + min(time_per_scale, 120.0)

            a_new, sc_new = _active_set_newton(a_current, scale_deadline,
                                                max_iter=200, tol_active=tol_scale)
            if sc_new < best_score - 1e-12:
                a_current = a_new
                update_best(a_new, sc_new, f"asn_tol_{tol_scale}")
            elif sc_new < _score(a_current) + 1e-10:
                # Slight improvement, keep it
                a_current = a_new

        # Additional Newton passes with tight tolerance
        if time.monotonic() < phase2_deadline - 30:
            for extra in range(3):
                if time.monotonic() > phase2_deadline - 20:
                    break
                a_new, sc_new = _active_set_newton(a_current,
                                                    min(phase2_deadline - 15, time.monotonic() + 40),
                                                    max_iter=100, tol_active=0.0001)
                if sc_new < best_score - 1e-14:
                    a_current = a_new
                    update_best(a_new, sc_new, f"asn_extra_{extra}")

    # === Phase 3: Coordinate descent + shrinkage (20% of budget) ===
    phase3_deadline = t_start + 0.90 * (deadline - t_start)
    print(f"\n[Phase 3] Coord descent + shrinkage (until t={phase3_deadline - t_start + t_start:.0f}s)")

    if best_seq is not None:
        # Coord descent
        if time.monotonic() < phase3_deadline - 30:
            a_cd, sc_cd = _coord_descent(best_seq, min(phase3_deadline - 20, time.monotonic() + 120),
                                         max_rounds=100)
            update_best(a_cd, sc_cd, "coord_descent")
            if sc_cd < best_score:
                # Follow up with Newton
                a_cd2, sc_cd2 = _active_set_newton(a_cd, min(phase3_deadline - 10, time.monotonic() + 60),
                                                    max_iter=100, tol_active=0.0002)
                update_best(a_cd2, sc_cd2, "cd+asn")

        # Shrinkage
        if time.monotonic() < phase3_deadline - 10:
            a_shr, sc_shr = _shrunk_and_reoptimize(best_seq, phase3_deadline - 5, min_time_remaining=5)
            update_best(a_shr, sc_shr, "shrunk")

    # === Phase 4: Final refinement (10% of budget) ===
    print(f"\n[Phase 4] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight Newton
        a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 3, time.monotonic() + 60),
                                           max_iter=300, tol_active=0.00005)
        update_best(a_fin, sc_fin, "final_asn_tight")

        # One more coord descent
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _coord_descent(best_seq, min(deadline - 3, time.monotonic() + 40),
                                             max_rounds=50)
            update_best(a_fin2, sc_fin2, "final_cd")

        # Tiny L-BFGS with very small beta
        if time.monotonic() < deadline - 5:
            c_best = _autocorr(best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 5:
                max_iter = min(2000, max(50, int(remaining / 0.02)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsobj_grad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs")
                except Exception:
                    pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
