"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _spectral_perturb(a, rng, n_pert=3):
    """Perturb in frequency domain: randomize phase of some FFT bins."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    
    # Random phase perturbation on a subset of frequency bins
    n_bins = len(A)
    n_perturb = min(n_pert, n_bins - 2)
    idx = rng.integers(1, n_bins - 1, size=n_perturb)
    phases = rng.uniform(-np.pi, np.pi, size=n_perturb)
    A[idx] *= np.exp(1j * phases)
    
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _kkt_equalize(a, deadline, k_active=5, max_iter=100):
    """Try to equalize the top k_active autocorrelation values."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        target = active_vals.mean()
        
        # Compute gradient to push active values toward target
        w = np.zeros_like(c)
        # Weight by how far above target each active value is
        for j, i in enumerate(idx):
            if c[i] > target:
                w[i] = (c[i] - target)
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()

        # Line search
        alpha = 1.0 / n
        improved = False
        for _ in range(30):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:
            break

    return best_a, best_sc


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Finer beta schedule

                              0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
                              1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        # More iterations for smaller betas
        scale = max(1.0, (n / 4096.0) ** 1.0)
        max_iter = min(8000, max(50, int(remaining / (0.003 * scale * 15))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            if i % 5 == 0:
                print(f"    beta={beta:.2e}: score={sc:.10f}")

        if current_sc < 1.5005:
            break

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive L-BFGS with spread-based beta scheduling and restarts
2. Nesterov momentum subgradient in log-space
3. Spectral (FFT phase) basin hopping for diversification
4. KKT active-set equalization
5. Direct subgradient with momentum in positive orthant
6. Multi-length exploration with interpolation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs_v2(a_start, deadline, reserve_time=60.0):
    """Adaptive L-BFGS with spread-based beta and restart logic."""
    n = len(a_start)
    current_a = a_start.copy()
    current_sc = _score(current_a)
    no_improve_count = 0
    beta_factor = 0.15

    for rnd in range(100):
        if time.monotonic() + reserve_time > deadline:
            break

        c = _full_convolve(current_a, current_a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        sorted_c = np.sort(c)[::-1]
        k_spread = min(20, len(sorted_c))
        spread = c_max - sorted_c[k_spread - 1]

        if spread < 1e-18:
            beta = c_max * 1e-8
        else:
            beta = max(c_max * 1e-10, spread * beta_factor)

        beta = min(beta, c_max * 0.2)

        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break
        scale = max(1.0, (n / 4096.0))
        max_iter = min(10000, max(50, int(remaining / (0.004 * scale * 12))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 3:
                # Restart with smaller beta factor
                beta_factor *= 0.4
                no_improve_count = 0
            elif no_improve_count > 1:
                beta_factor *= 0.7

        if rnd % 10 == 0:
            print(f"    L-BFGS round {rnd}: beta={beta:.2e}, score={current_sc:.10f}")

        if current_sc < 1.5005:
            break

    return current_a


def _momentum_subgrad(a_init, deadline, k_active=5, lr_init=0.04, momentum=0.75):
    """Subgradient with Nesterov momentum in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    vz = np.zeros_like(z)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    max_steps = 300000

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = lr / n
        vz = momentum * vz - step * grad / (a * n + 1e-30)
        z_new = z + vz
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -80.0, 80.0)

        if i % 200 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                lr = min(lr * 1.005, 0.3)
            else:
                stall += 1
                if stall % 5 == 0:
                    lr *= 0.9
                    momentum = min(momentum + 0.002, 0.95)
                if lr < 1e-12:
                    break
        z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _direct_subgrad_mom(a_init, deadline, k_active=4, max_steps=100000):
    """Direct subgradient with momentum in positive orthant."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros_like(a)
    step_size = 0.8 / np.sqrt(n)
    momentum = 0.4
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        v = momentum * v - step_size * g / np.sqrt(n)
        a_try = a + v
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            v = np.zeros_like(a)
            step_size *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        a_try = a_try / s

        if i % 100 == 0:
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                stall = 0
                step_size = min(step_size * 1.01, 3.0 / np.sqrt(n))
            else:
                stall += 1
                if stall % 5 == 0:
                    step_size *= 0.92
                    v *= 0.7
                if stall > 200:
                    break

        a = a_try

    return best_a, best_sc


def _spectral_perturb(a, rng, n_phase=3, n_amp=2):
    """Perturb in frequency domain: randomize some phase and amplitude."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    n_bins = len(A)

    # Phase perturbation
    if n_phase > 0:
        idx_p = rng.integers(1, max(2, n_bins - 1), size=min(n_phase, n_bins - 2))
        phases = rng.uniform(-np.pi / 3, np.pi / 3, size=len(idx_p))
        A[idx_p] *= np.exp(1j * phases)

    # Amplitude perturbation
    if n_amp > 0:
        idx_a = rng.integers(1, max(2, n_bins - 1), size=min(n_amp, n_bins - 2))
        factors = rng.uniform(0.95, 1.05, size=len(idx_a))
        A[idx_a] *= factors

    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _spectral_basin_hop(a_current, deadline, rng, n_hops=12):
    """Basin hopping using spectral phase/amplitude randomization."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)

    for h in range(n_hops):
        if time.monotonic() > deadline:
            break

        # Escalating perturbation strength
        n_ph = 2 + (h % 5)
        n_am = 1 + (h % 3)
        a_pert = _spectral_perturb(best_a, rng, n_ph, n_am)

        # Mix with small time-domain noise
        if h % 3 == 0:
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + 0.02 * rng.standard_normal(n)
            z = z - z.mean()
            a_mix = _softmax(z)
            a_pert = 0.5 * a_pert + 0.5 * a_mix
            a_pert = a_pert / a_pert.sum()

        # Quick optimization from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        beta = mc * 0.003
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(3000, max(100, int(remaining / 40 / 0.004)))
        a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc


def _kkt_equalize(a, deadline, k_active=5, max_iter=200):
    """Push active autocorrelation values toward their mean."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        target = active_vals.mean()

        # Weight active set by excess above target
        w = np.zeros_like(c)
        excess = np.maximum(active_vals - target, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        for j in range(len(idx)):
            w[idx[j]] = excess[j] / total_excess

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()

        # Line search
        alpha = 1.0 / n
        improved = False
        for _ in range(40):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:
            break

    return best_a, best_sc


def _mix_and_optimize(a1, a2, deadline, n=4096):
    """Find best convex combination of a1 and a2, then optimize."""
    best_a = None
    best_sc = np.inf
    for t in np.linspace(0, 1, 21):
        a_try = (1 - t) * a1 + t * a2
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()

    if best_a is not None:
        c = _full_convolve(best_a, best_a)
        mc = float(c.max())
        beta = mc * 0.005
        a_opt = _lbfgs_smooth(best_a, beta, len(best_a), 2000)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp
