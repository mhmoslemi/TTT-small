"""

FFT-accelerated Hankel matvec, CG-based QP solves, and multi-n exploration.
Improves on prior approach by: (1) progressive active-set tightening,
(2) direct KKT-system QP solves, (3) tighter CG convergence,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=800, tol=1e-14):
    """Solve H*x = rhs using CG with matrix-free Hankel Hessian."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.2:
            break
        x = x_new
        r = r_new
        p = r + (rs_new / rs) * p
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.002)

    if aggressive:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.00375, 0.0018, 0.0009, 0.00045, 0.0002, 0.0001,
                           0.00005, 0.00002, 0.00001, 0.000005, 0.000002,
                           0.000001, 0.0000005, 0.0000002, 0.0000001])
    else:
        ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                           0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                           0.00002, 0.00001, 0.000005, 0.000002, 0.000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _qp_active_solve(W, n, a_init, max_outer=100, cg_iter=600):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 using active-set on support."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = float(a @ _hankel_matvec(a, W, n) / 2.0)

    for outer in range(max_outer):
        support = a > 1e-10
        n_sup = int(support.sum())
        if n_sup == 0:
            support[:] = True
            n_sup = n
        if n_sup == n:
            # Equality constrained only: H*a = (lambda/2)*1, 1^T a = 1
            # Solve H*y = 1, then a = y/(1^T y)
            y = _cg_solve(W, n, np.ones(n), max_iter=cg_iter)
            oney = float(np.sum(y))
            if abs(oney) < 1e-20 or not np.isfinite(oney):
                break
            a_new = y / oney
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        else:
            # Inequality active: solve on support
            # Use projected gradient as fallback for partial support
            grad = _hankel_matvec(a, W, n)
            lam = grad[support].mean()
            step = 1.0 / (np.max(np.abs(grad)) + 1e-15)
            step = min(step, 0.1)
            a_new = a - step * (grad - lam)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new

        val_new = float(a_new @ _hankel_matvec(a_new, W, n) / 2.0)
        if not np.isfinite(val_new):
            break
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
        a = a_new

        # Check convergence
        grad = _hankel_matvec(a, W, n)
        support = a > 1e-10
        if support.any():
            lam = grad[support].mean()
            resid = np.max(np.abs(grad[support] - lam))
            off_s = ~support
            if off_s.any():
                resid = max(resid, np.max(lam - grad[off_s]))
            else:
                pass
        else:
            resid = np.inf
        if resid < 1e-10 * (np.max(np.abs(grad)) + 1e-15):
            break

    return best_a


def _newton_active_polish(a_init, deadline, max_rounds=100, tol_schedule=None):
    """CG-Newton polish with progressive active set tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    if tol_schedule is None:
        tol_schedule = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                        0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                        0.0000005, 0.0000002, 0.0000001]

    tol_idx = 0
    rounds_per_tol = 5

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Solve QP: min a^T H_W a s.t. sum(a)=1, a>=0
        a_qp = _qp_active_solve(W, n, a, max_outer=50, cg_iter=400)
        sc_qp = _score(a_qp)

        if sc_qp < best_sc - 1e-14:
            best_sc = sc_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Try damped Newton step instead
            g = _hankel_matvec(a, W, n)
            g_t = g - np.mean(g)
            gn = float(np.linalg.norm(g_t))
            if gn < 1e-15:
                pass
            else:
                Hg = _cg_solve(W, n, g, max_iter=300)
                H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
                H1_norm = float(np.linalg.norm(H1))
                if H1_norm > 1e-20 and np.isfinite(H1_norm):
                    lam = float(np.sum(Hg)) / float(np.sum(H1))
                    dx = -Hg + lam * H1
                    dx = dx - np.mean(dx)
                else:
                    dx = -g_t / (gn + 1e-30)

                alpha = 1.0
                for ls in range(40):
                    a_try = a + alpha * dx
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        alpha *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    c_max_try = float(c_try.max())
                    if c_max_try < c_max:
                        a = a_try
                        sc_try = _score(a)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a.copy()
                        break
                    alpha *= 0.5
                    if alpha < 1e-14:
                        break

        # Advance tolerance schedule
        if (rnd + 1) % rounds_per_tol == 0:
            tol_idx += 1

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, max_steps=3000):
    """Adaptive subgradient with Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.03
    momentum = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.0005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0005 * i)
        z_new = z - step * grad
        # Nesterov momentum
        z_new = z_prev + momentum * (z_new - z) + (1 - momentum) * (-step * grad)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z_prev = z.copy()
        z = z_new

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z_prev = z.copy()
                    patience = 0
                    lr *= 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _flatten_polish(a_init, deadline, max_iters=200):
    """Direct convolution flattening: push down high-conv entries."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_target = np.sum(c) / len(c)  # mean of autocorrelation
        c_max = float(c.max())

        # Compute "excess" autocorrelation
        excess = np.maximum(c - c_target, 0)
        excess_max = excess.max()
        if excess_max < 1e-15:
            break

        # Gradient of sum of squares of excess
        # d/d a_i [sum_k excess_k^2] = 2 * sum_k excess_k * d(c_k)/da_i
        # d(c_k)/da_i = a_{i+k} + a_{i-k} (with boundary handling)
        # This is just the "deconvolution" of excess with a
        a_rev = a[::-1]
        excess_conv = _full_convolve(excess, a_rev)
        grad_part = excess_conv[n - 1:2 * n - 1]

        step = 0.1 / (1.0 + 0.01 * it)
        a_new = a - step * grad_part
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a + 0.5 * (a_new - a)  # partial step
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, aggressive=True)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Active-set QP with progressive tightening
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _newton_active_polish(best_seq,
                                         min(deadline - 400, time.monotonic() + 300),
                                         max_rounds=80)
        update_best(a2, sc2, "active_qp_1")

    # Phase 3: Flatten polish
    print(f"\n[Phase 3] Flatten polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a3, sc3 = _flatten_polish(best_seq, min(deadline - 300, time.monotonic() + 100))
        update_best(a3, sc3, "flatten_1")

    # Phase 4: Subgradient with momentum
    print(f"\n[Phase 4] Subgradient+Nesterov (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _subgradient_polish(best_seq, min(deadline - 250, time.monotonic() + 120))
        update_best(a4, sc4, "subgrad_1")

    # Phase 5: Another round of active-set QP (tighter)
    print(f"\n[Phase 5] Active-set QP round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _newton_active_polish(best_seq,
                                         min(deadline - 200, time.monotonic() + 150),
                                         max_rounds=100)
        update_best(a5, sc5, "active_qp_2")

    # Phase 6: Shrinkage + re-optimize
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a6, sc6 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=100.0)
        update_best(a6, sc6, "shrinkage")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0, aggressive=True)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 60:
            a_np, sc_np = _newton_active_polish(a_opt,
                                                 min(deadline - 50, time.monotonic() + 80),
                                                 max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_qp")

    # Phase 8: Diverse restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_profiles = ["tapered", "cos", "centered", "bimodal"]
        for i, prof in enumerate(restart_profiles[:3]):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 1:
                n_r = len(best_seq)
                p_start = _make_profile(n_r, prof)
            else:
                # Shrink + perturb
                p_shrunk = best_seq.copy()
                idx = np.argsort(p_shrunk)
                p_shrunk[idx[:len(idx) // 5]] = 0
                p_shrunk = p_shrunk / p_shrunk.sum()
                p_start = _perturb(p_shrunk, 0.1, rng)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_active_polish(a_r,
                                                     min(deadline - 10, time.monotonic() + 20),
                                                     max_rounds=30)
                update_best(a_rp, sc_rp, f"restart_{i}_qp")

    # Phase 9: Final ultra-tight polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2,
                                              max_rounds=100,
                                              tol_schedule=[0.00001, 0.000005,
                                                            0.000002, 0.000001,
                                                            0.0000005, 0.0000002, 0.0000001])
        update_best(a_fin, sc_fin, "final_qp")

        # Final flatten
        if time.monotonic() < deadline - 3:
            a_fl, sc_fl = _flatten_polish(best_seq, deadline - 2, max_iters=100)
            update_best(a_fl, sc_fl, "final_flatten")

        # Ultra-fine L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-8
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 60}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
