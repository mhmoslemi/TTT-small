    # = sum_j a[j] * w[i+j]
    # Let's define w' = w (length 2n-1). Then sum_j a[j]*w[i+j] for j=0..n-1
    # This is the (i)-th element of the "valid" cross-correlation.
    
    # Alternative: pad a to length 2n-1, compute (a_pad * w_rev) and extract.
    # a_pad = [a, 0, 0, ..., 0] (length 2n-1)
    # w_rev = w[::-1] (length 2n-1)
    # (a_pad * w_rev)[k] = sum_j a_pad[j] * w_rev[k-j] = sum_{j=0}^{n-1} a[j] * w[2n-2-(k-j)]
    # We want sum_j a[j] * w[i+j]. Let 2n-2-(k-j) = i+j => k = 2n-2-i.
    # So (a_pad * w_rev)[2n-2-i] = sum_j a[j] * w[i+j]. 
    
    # Therefore: (W@a)_i = 2 * (a_pad * w_rev)[2n-2-i]
    # And we need i = 0, ..., n-1, so we extract indices 2n-2, 2n-3, ..., n-1 from the convolution.
    
    conv = _full_convolve(a, w[::-1])
    # conv has length (n + (2n-1) - 1) = 3n-2
    # We want conv[2n-2-i] for i = 0, ..., n-1
    # That's conv[n-1 : 2n-1] in reverse
    result = 2.0 * conv[n-1:2*n-1][::-1]
    return result


def _reweighting_optimize(a_init, deadline, max_iter=5000):
    """Iterative reweighting: alternate between computing peak weights
    and doing gradient steps on the weighted quadratic."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 1.0  # Will be adapted
    tau = 0.01  # Temperature for weight computation
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        s = a.sum()
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Compute weights: emphasize peaks close to the max
        shifted = (c - c_max) / (tau * c_max)
        w = np.exp(np.clip(shifted, -50, 0))
        w = w / w.sum()
        
        # Compute gradient of weighted quadratic
        grad = _weighted_grad(a, w)
        # Center the gradient (to maintain sum(a) = 1 constraint)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        
        if gnorm < 1e-15:
            break
        
        # Line search
        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.8
            tau = max(0.001, tau * 0.9)
            if lr < 1e-6:
                break
        else:
            lr = min(2.0, lr * 1.05)
    
    return best_a, _score(best_a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(20, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 400
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        grad = _weighted_grad(a, w)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.75)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 8:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.02)

    return best_a, _score(best_a)
