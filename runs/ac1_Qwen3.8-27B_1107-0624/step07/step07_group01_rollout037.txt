    ```python
        """
        Solve the KKT linear system for given active set and weights.
        Returns a, lambda, or None if the system is singular.
        """
        m = len(active_set)
        
        # Build the (n+1) x (n+1) system
        # Row i (0 to n-1): sum_k weights[k] * A[k-i] (if valid) + lambda = 0
        # Row n: sum a_i = 1
        
        # For efficiency, build the system as a sparse matrix
        # Each row i has at most m non-zero entries in the a-part
        # Plus one entry for lambda
        
        from scipy.sparse import csr_matrix, diags
        from scipy.sparse.linalg import spsolve
        
        # Build the matrix
        rows = []
        cols = []
        vals = []
        
        for i in range(n):
            for l, k in enumerate(active_set):
                j = k - i  # a[k-i]
                if 0 <= j <= n-1:
                    rows.append(i)
                    cols.append(j)
                    vals.append(2.0 * weights[l])
            # lambda term
            rows.append(i)
            cols.append(n)  # lambda is the (n+1)-th variable (0-indexed: n)
            vals.append(1.0)
        
        # Normalization row
        rows.append(n)
        for i in range(n):
            cols.append(i)
            vals.append(1.0)
        
        # Build sparse matrix
        A_mat = csr_matrix((vals, (rows, cols)), shape=(n+1, n+1))
        b_vec = np.zeros(n+1)
        b_vec[n] = 1.0
        
        # Solve
        sol = spsolve(A_mat, b_vec)
        if sol is None:
            return None, None
        
        a = sol[:n]
        lam = sol[n]
        
        return a, lam
    ```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _kkt_solve_linear_system(n, active_set, weights):
    """
    Solve the KKT linear system:
      sum_l w_l * a[active_l - i] + lambda = 0  for i = 0..n-1
      sum_i a_i = 1
    Returns (a, lambda) or (None, None) if singular.
    """
    m = len(active_set)
    # Build sparse system of size (n+1) x (n+1)
    # Variables: a[0], ..., a[n-1], lambda (index n)
    row_idx = []
    col_idx = []
    vals = []

    for i in range(n):
        for l in range(m):
            k = active_set[l]
            j = k - i
            if 0 <= j < n:
                row_idx.append(i)
                col_idx.append(j)
                vals.append(2.0 * weights[l])
        # lambda term (coefficient +1)
        row_idx.append(i)
        col_idx.append(n)
        vals.append(1.0)

    # Normalization row
    row_idx.append(n)
    for i in range(n):
        col_idx.append(i)
        vals.append(1.0)

    A_mat = csr_matrix((vals, (row_idx, col_idx)), shape=(n + 1, n + 1))
    b_vec = np.zeros(n + 1)
    b_vec[n] = 1.0

    try:
        sol = spsolve(A_mat, b_vec)
    except Exception:
        return None, None

    if sol is None or not np.all(np.isfinite(sol)):
        return None, None

    a = sol[:n]
    lam = sol[n]
    return a, lam


def _kkt_active_set_method(a_init, deadline, max_iter=200, tol=1e-10):
    """
    KKT-based active set optimization.
    Iteratively: identify active peaks, solve linear system, update active set.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify active set: peaks within 5% of max
        threshold = c_max * 0.95
        active_indices = np.where(c >= threshold)[0]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])
        # Limit to top 8
        if len(active_indices) > 8:
            vals = c[active_indices]
            top_k = np.argsort(vals)[::-1][:8]
            active_indices = active_indices[top_k]

        # Use soft weights proportional to peak values
        peak_vals = c[active_indices]
        weights = peak_vals / peak_vals.sum()

        # Solve the KKT linear system
        a_kkt, lam = _kkt_solve_linear_system(n, active_indices, weights)

        if a_kkt is None:
            # System singular, try with just the top 3 peaks
            top3 = np.argsort(c)[::-1][:3]
            pv = c[top3]
            w3 = pv / pv.sum()
            a_kkt, lam = _kkt_solve_linear_system(n, top3, w3)
            if a_kkt is None:
                break

        # Check if solution is non-negative
        if np.any(a_kkt < -1e-10):
            # Clamp and try again with fewer active peaks
            a_kkt = np.maximum(a_kkt, 0.0)
            s_try = a_kkt.sum()
            if s_try < 1e-15:
                break
            a_kkt = a_kkt / s_try
            sc_try = _score(a_kkt)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_kkt.copy()
            # Use as starting point for next iteration
            a = a_kkt.copy()
            continue

        # Normalize
        s_kkt = a_kkt.sum()
        if s_kkt < 1e-15:
            break
        a_kkt = a_kkt / s_kkt

        sc_kkt = _score(a_kkt)
        if sc_kkt < best_sc:
            best_sc = sc_kkt
            best_a = a_kkt.copy()

        # Check KKT feasibility: all peaks should be <= c_max (of the KKT solution)
        c_kkt = _full_convolve(a_kkt, a_kkt)
        c_kkt_max = float(c_kkt.max())

        # Update: if the KKT solution has a different peak structure, update a
        a = a_kkt.copy()

        # Convergence check
        if abs(sc_kkt - best_sc) < tol:
            # Check if we've converged
            if c_kkt_max < c_max * (1 + 1e-8):
                pass  # stable
        # If the KKT solution is worse, blend
        if sc_kkt > best_sc:
            a = 0.5 * best_a + 0.5 * a_kkt
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def _kkt_weight_update(a_init, deadline, max_iter=500):
    """
    KKT method with adaptive weight updates.
    Start with current active peaks, solve, update weights based on violations.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Initial active set: top 5 peaks
    active = list(np.argsort(c)[::-1][:5])
    weights = np.ones(len(active)) / len(active)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Solve KKT system
        a_kkt, lam = _kkt_solve_linear_system(n, active, weights)

        if a_kkt is None:
            # Try with top 3
            active = list(np.argsort(c)[::-1][:3])
            weights = np.ones(3) / 3
            a_kkt, lam = _kkt_solve_linear_system(n, active, weights)
            if a_kkt is None:
                break

        a_kkt = np.maximum(a_kkt, 0.0)
        s_kkt = a_kkt.sum()
        if s_kkt < 1e-15:
            break
        a_kkt = a_kkt / s_kkt

        sc_kkt = _score(a_kkt)
        if sc_kkt < best_sc:
            best_sc = sc_kkt
            best_a = a_kkt.copy()

        # Compute new autocorrelation
        c_kkt = _full_convolve(a_kkt, a_kkt)
        c_kkt_max = float(c_kkt.max())

        # Update active set and weights
        # Peaks that are at or near the max should have higher weight
        new_active = list(np.argsort(c_kkt)[::-1][:min(8, len(active) + 2)])
        new_peak_vals = c_kkt[new_active]
        # Soft weighting: emphasize peaks closer to max
        temp = 0.01 * c_kkt_max + 1e-10
        new_weights = np.exp(new_peak_vals / temp)
        new_weights = new_weights / new_weights.sum()

        # Check if we've made progress
        if c_kkt_max < c_max * 0.999:
            active = new_active
            weights = new_weights
            c = c_kkt
            c_max = c_kkt_max
            a = a_kkt.copy()
        else:
            # No progress, try different active set
            if it % 10 == 0:
                active = new_active[:max(2, len(active) - 1)] if len(active) > 3 else new_active[:3]
                weights = np.ones(len(active)) / len(active)

    return best_a


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # More aggressive beta schedule
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
                              1e-6, 5e-7, 2e-7, 1e-7, 5e-8])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, (n / 4096.0) ** 1.0)
        max_iter = min(20000, max(200, int(remaining / (0.003 * scale * 15))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5001:
            break

    return current_a


def _nesterov_subgrad(a_init, deadline, k_active=5, max_steps=20000, lr_init=0.03):
    """
    Nesterov-accelerated subgradient method with iterate averaging.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    theta = 2.0 / (2.0 + 1.0)  # momentum

    # Running average for output
    avg_z = z.copy()
    avg_count = 0

    stall = 0
    stall_threshold = 50

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        z_new = z + theta * (z - z_prev)
        z_prev = z.copy()
        z = z_new

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Subgradient: use top-k peaks
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Step size (Polyak-like)
        step = lr / (i + 1) ** 0.5
        z = z - step * grad / (n * a.sum() + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -200.0, 200.0)

        # Update average
        avg_count += 1
        avg_z = avg_z + (z - avg_z) / avg_count

        # Periodic check
        if i % 200 == 0:
            a_check = _softmax(avg_z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= 0.7
                    theta = 2.0 / (2.0 + i * 0.01)
                    if lr < 1e-10:
                        break

    a_final = _softmax(avg_z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a


def _direct_subgrad(a_init, deadline, k_active=3, max_steps=30000, lr_init=0.3):
    """Direct subgradient in a-space with line search."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    step = lr_init / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        a_try = a - step * g / np.sqrt(n)
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step *= 0.5
            stall += 1
            if stall > 200:
                break
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        sc_cur = _score(a)
        if sc_try < sc_cur - 1e-16:
            a = a_try
            step = min(step * 1.005, 0.5)
        else:
            stall += 1
            step *= 0.95

        if stall > 500:
            break

    return best_a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_recurrence(n, r=0.95, offset=0.5):
    """Create a sequence based on the KKT recurrence solution."""
    a = np.zeros(n)
    # a_j = A * r^j + offset (simple 2-peak KKT solution)
    t = np.arange(n)
    a = offset + (1.0 - offset) * r ** t
    # Also try symmetric version
    a2 = offset + (1.0 - offset) * r ** (n - 1 - t)
    a_avg = 0.5 * (a + a2)
    a_avg = np.maximum(a_avg, 1e-10)
    return a_avg / a_avg.sum()


def _make_damped_osc(n, freq=3.0, decay=0.8):
    """Damped oscillation (from 3-peak KKT solution)."""
    t = np.arange(n)
    a = 0.5 + 0.3 * np.exp(-decay * t / n) * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.25):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=8, reserve=30.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6

        pert_type = h % 4
        z = np.log(np.maximum(best_a, 1e-18))
        z = z - z.mean()

        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = z + sigma * rng.standard_normal(n)
        elif pert_type == 1:
            # Low-frequency perturbation
            for _ in range(3):
                freq = int(rng.integers(1, max(2, n // 64)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        elif pert_type == 2:
            # Block perturbation
            n_blocks = 4 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
        else:
            # High-frequency perturbation
            for _ in range(2):
                freq = int(rng.integers(n // 8, n // 2))
                amp = 0.01 * rng.uniform(0.5, 1.5)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)

        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick optimization from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.02
        max_it = min(2000, max(100, int((hop_deadline - time.monotonic()) / 0.01)))
        if max_it > 50:
            a_opt = _lbfgs_smooth(a_pert, beta, n, max_it)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a


def _peak_shaving(a_init, deadline, max_iter=5000):
    """
    Targeted peak shaving: reduce the highest peak by adjusting
    the elements that contribute most to it.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.1

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Compute contribution of each a_j to the peak
        # (a*a)[peak_idx] = sum_j a_j * a[peak_idx - j]
        contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] = a[k]  # d(a*a)[peak]/d(a_j) ~ a[k] (times 2 for symmetry)

        # Also account for the symmetric contribution
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] += a[j] * 0  # already counted above

        # Normalize contribution
        contrib = contrib / (contrib.sum() + 1e-30)

        # Gradient: we want to reduce elements that contribute to the peak
        # but also not make other peaks worse
        # Simple approach: reduce high-contributing elements, increase low ones
        g = contrib - 1.0 / n
        a_try = a - lr * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            lr *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.9

    return best_a


def _multi_kkt(a_init, deadline):
    """
    Try multiple different active sets and pick the best KKT solution.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Try various active sets
    top_peaks = np.argsort(c)[::-1]

    active_sets = [
        # Top 3
        list(top_peaks[:3]),
        # Top 5
        list(top_peaks[:5]),
        # Top 2
        list(top_peaks[:2]),
        # Center peak only
        [n - 1],
        # Center + adjacent
        [n - 1, n - 2, n],
        # Spread out
        list(top_peaks[[0, 2, 4]]),
    ]

    for active in active_sets:
        if time.monotonic() > deadline:
            break
        active = [k for k in active if 0 <= k <= 2 * n - 2]
        if len(active) < 1:
            continue
        m = len(active)
        for w_mode in ['uniform', 'proportional']:
            if w_mode == 'uniform':
                weights = np.ones(m) / m
            else:
                pv = c[active]
                weights = pv / pv.sum()

            a_kkt, lam = _kkt_solve_linear_system(n, active, weights)
            if a_kkt is None:
                continue

            a_kkt = np.maximum(a_kkt, 0.0)
            s_k = a_kkt.sum()
            if s_k < 1e-15:
                continue
            a_kkt = a_kkt / s_k
            sc_k = _score(a_kkt)
            if sc_k < best_sc:
                best_sc = sc_k
                best_a = a_kkt.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining KKT active-set, L-BFGS,
    Nesterov subgradient, peak shaving, and basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Try structured inits at n=4096 (quick)
    n_init = 4096
    quick_inits = [
        ("chirp1", _make_chirp(n_init, 1.0, 0.3)),
        ("rec095", _make_recurrence(n_init, 0.95, 0.4)),
        ("damp3", _make_damped_osc(n_init, 3.0, 0.8)),
        ("multi13", _make_multiband(n_init, [1, 3], 0.25)),
        ("tapered", _make_tapered(n_init, 0.7)),
    ]

    for name, a_init in quick_inits:
        sc = _score(a_init)
        try_update(a_init, name)

    if best_seq is None:
        best_seq = _make_chirp(n_init, 1.0, 0.3)
        best_score_val = _score(best_seq)
        best_n = n_init

    print(f"  Phase 1 done. Best: {best_score_val:.10f} (n={best_n})")

    # === Phase 2: KKT Active-Set Optimization ===
    print(f"\n=== Phase 2: KKT Active-Set (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 600:
        kkt_deadline = deadline - 600

        # First: multi-KKT (try different active sets)
        a_mk = _multi_kkt(best_seq, min(kkt_deadline, time.monotonic() + 30))
        try_update(a_mk, "multi_kkt")

        # Second: iterative KKT with weight updates
        if time.monotonic() < kkt_deadline - 60:
            a_kkt = _kkt_active_set_method(best_seq, min(kkt_deadline - 60, time.monotonic() + 120))
            try_update(a_kkt, "kkt_iter")

        # Third: KKT with adaptive weights
        if time.monotonic() < kkt_deadline - 30:
            a_kkt2 = _kkt_weight_update(best_seq, kkt_deadline - 30, max_iter=200)
            try_update(a_kkt2, "kkt_wt")

    # === Phase 3: L-BFGS Optimization ===
    print(f"\n=== Phase 3: L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 400:
        a_lb = _adaptive_lbfgs(best_seq, deadline, reserve_time=400)
        try_update(a_lb, "lbfgs_main")

    # === Phase 4: Nesterov Subgradient ===
    print(f"\n=== Phase 4: Nesterov Subgradient (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 300:
        a_ns = _nesterov_subgrad(best_seq, min(deadline - 300, time.monotonic() + 120),
                                  k_active=5, max_steps=15000, lr_init=0.04)
        try_update(a_ns, "nesterov_k5")

    if best_seq is not None and time.monotonic() < deadline - 250:
        a_ns2 = _nesterov_subgrad(best_seq, min(deadline - 250, time.monotonic() + 80),
                                   k_active=3, max_steps=10000, lr_init=0.03)
        try_update(a_ns2, "nesterov_k3")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 250:
        a_ps = _peak_shaving(best_seq, min(deadline - 250, time.monotonic() + 60))
        try_update(a_ps, "peak_shave")

    # === Phase 6: Direct Subgradient ===
    print(f"\n=== Phase 6: Direct Subgrad (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_dg = _direct_subgrad(best_seq, min(deadline - 200, time.monotonic() + 80),
                                k_active=3, max_steps=20000, lr_init=0.3)
        try_update(a_dg, "direct_sg")

    # === Phase 7: Second KKT round (from improved point) ===
    print(f"\n=== Phase 7: KKT Round 2 (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_kkt3 = _kkt_weight_update(best_seq, min(deadline - 200, time.monotonic() + 60), max_iter=100)
        try_update(a_kkt3, "kkt_r2")

        a_mk2 = _multi_kkt(best_seq, min(deadline - 180, time.monotonic() + 20))
        try_update(a_mk2, "multi_kkt_r2")

    # === Phase 8: Final L-BFGS ===
    print(f"\n=== Phase 8: Final L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 150:
        a_lb2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=100)
        try_update(a_lb2, "lbfgs_final")

    # === Phase 9: Basin Hopping ===
    print(f"\n=== Phase 9: Basin Hopping (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 80:
        a_hop = _basin_hop(best_seq, deadline - 50, rng, n_hops=6, reserve=40.0)
        try_update(a_hop, "basin_hop")

    # === Phase 10: Multi-length exploration ===
    print(f"\n=== Phase 10: Multi-length (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 50:
        current_n = len(best_seq)
        lengths_to_try = []
        for ln in [2048, 4096, 8192, 12288, 16384, 32768]:
            if ln != current_n:
                lengths_to_try.append(ln)

        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 30:
                break
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            # Quick L-BFGS
            if time.monotonic() < deadline - 20:
                c_i = _full_convolve(a_interp, a_interp)
                beta = c_i.max() * 0.005
                max_it = min(3000, max(100, int((deadline - 20 - time.monotonic()) / 0.01)))
                a_opt = _lbfgs_smooth(a_interp, beta, target_n, max_it)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 11: Final tight polishing ===
    print(f"\n=== Phase 11: Final Polish (t={time.monotonic()-t_start:.1f}s) ===")

    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 3:
                break
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())

            # Very tight L-BFGS
            beta = mb * 1e-8
            max_it = min(3000, max(100, int((deadline - 3 - time.monotonic()) / 0.01)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_it)
            try_update(a_f, f"polish_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 3:
                a_s = _direct_subgrad(best_seq, deadline - 3,
                                       k_active=3, max_steps=5000, lr_init=0.1)
                try_update(a_s, f"polish_sg_{rnd}")

            # KKT
            if time.monotonic() < deadline - 3:
                a_k = _kkt_weight_update(best_seq, deadline - 3, max_iter=50)
                try_update(a_k, f"polish_kkt_{rnd}")

    # === Final ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
