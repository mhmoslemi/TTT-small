import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_step(a, c, n, step_size):
    """One subgradient step on the true max, with simplex projection."""
    c_max = float(c.max())
    idx = np.argmax(c)
    a_rev = a[::-1]
    e = np.zeros(2 * n - 1)
    e[idx] = 1.0
    wc = _full_convolve(e, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = c_max
    grad = 4.0 * n * a * (h - W)
    a_new = a - step_size * grad
    # Project onto simplex: a >= 0, sum a = 1
    a_new = _project_simplex(a_new)
    return a_new


def _project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u * np.arange(1, n + 1) > cssv)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _fista_polish(a_init, deadline, n_iters=5000, beta_start=0.001):
    """FISTA (accelerated proximal gradient) with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Estimate Lipschitz constant
    a0 = _softmax(z)
    g0 = _smograd(z, beta_start, n)
    L = max(np.linalg.norm(g0) / max(np.linalg.norm(a0), 1e-10), 1.0)
    L = min(L * 2, 100.0)

    t = 1.0
    z_prev = z.copy()
    a_prev = a0.copy()

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break

        g = _smograd(z, beta_start, n)
        # Backtracking line search
        step = 1.0 / L
        z_try = z - step * g
        a_try = _softmax(z_try)
        f_new = _smoobj(z_try, beta_start, n)
        f_old = _smoobj(z, beta_start, n)
        backtracks = 0
        while f_new > f_old + 0.5 * step * np.dot(g, z_try - z) and backtracks < 20:
            step *= 0.5
            z_try = z - step * g
            a_try = _softmax(z_try)
            f_new = _smoobj(z_try, beta_start, n)
            backtracks += 1
            if step < 1e-12:
                break

        z_new = z_try
        a_new = _softmax(z_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z_y = z_new + ((t - 1.0) / t_new) * (z_new - z_prev)
        z_y = np.clip(z_y, -60.0, 60.0)

        z_prev = z
        z = z_y
        z = z - z.mean() if np.abs(z).max() > 100 else z
        a_prev = a_new
        t = t_new
        L = max(L * 0.99, np.linalg.norm(_smograd(z, beta_start, n)) / max(np.linalg.norm(a_new), 1e-10) * 1.1)
        L = min(L, 200.0)

        if i % 50 == 0:
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3):

        c_max = float(c.max())

        # Find top contributing coordinates to the max
        peak_idx = np.argmax(c)
        # Contribution of a_i to c[peak_idx]: a_i * a[peak_idx - i] (and symmetric)
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = 2.0 * a[i] * a[j]
            else:
                contrib[i] = a[i] * a[i] if 2 * i == peak_idx else 0.0

        # Try modifying top contributors
        n_try = min(n, 20)
        top_idx = np.argsort(contrib)[-n_try:]
        np.random.shuffle(top_idx)

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Try reducing (primary direction)
            for factor in [0.7, 0.8, 0.9]:
                a[i] = old_val * factor
                c_try = _full_convolve(a, a).max()
                if c_try < c_max - 1e-12:
                    improved = True
                    break
            else:
                # Try small increase
                for factor in [1.2, 1.5]:
                    a[i] = old_val * factor
                    c_try = _full_convolve(a, a).max()
                    if c_try < c_max - 1e-12:
                        improved = True
                        break
                else:
                    a[i] = old_val

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_theoretical_init(n):
    """
    Initialize based on theoretical analysis.
    The ideal a(x)^2 should be flat, meaning a(x) ~ sqrt(1+x+...+x^{2n-2}).
    The coefficients of sqrt(1+x+...+x^{2n-2}) decay slowly.
    Use a_i ~ 1/sqrt(i+1) as approximation, then taper edges.
    """
    i = np.arange(n, dtype=np.float64)
    # Base: inverse square root
    a = 1.0 / np.sqrt(i + 1.0)
    # Taper the last few elements to reduce edge effects
    taper = n // 10
    if taper > 0:
        for j in range(taper):
            a[n - 1 - j] *= (j + 1) / taper
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _make_tapered_init(n, power=0.5):
    """Parabolic/tapered initialization."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _fourier_perturb(a, rng, n_modes=3, amplitude=0.01):
    """Small Fourier perturbation for diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _multi_optimize(a_init, deadline, min_remaining=30.0):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: L-BFGS with aggressive beta
    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _lbfgs_optimize(current_a, deadline, min_time_remaining=min_remaining + 60)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        pe_deadline = min(deadline - min_remaining - 5, time.monotonic() + 90)
        a2, sc2_norm = _pressure_equalize_v2(current_a, pe_deadline, n_iters=200, gamma=0.35)
        sc2 = _score(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: L-BFGS again (finer)
    if time.monotonic() < deadline - min_remaining - 20:
        a3 = _lbfgs_optimize(current_a, deadline, min_time_remaining=min_remaining + 20)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: FISTA polish
    if time.monotonic() < deadline - min_remaining - 5:
        fista_deadline = min(deadline - min_remaining - 2, time.monotonic() + 60)
        a4, sc4 = _fista_polish(current_a, fista_deadline, n_iters=3000, beta_start=0.0005)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Coordinate polish
    if time.monotonic() < deadline - min_remaining:
        cd_deadline = min(deadline - min_remaining + 5, time.monotonic() + 60)
        a5, sc5 = _coordinate_polish_v2(current_a, cd_deadline, n_rounds=100)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    # Stage 6: Final L-BFGS
    if time.monotonic() < deadline - 5:
        a6 = _lbfgs_optimize(current_a, deadline, min_time_remaining=3)
        sc6 = _score(a6)
        if sc6 < current_sc:
            current_a = a6
            current_sc = sc6

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining L-BFGS, pressure equalization, FISTA,
    and coordinate descent. Exploits the structure of the optimal solution
    (slowly decaying coefficients) for better initialization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_theoretical_init(n_parent)
        print(f"No parent, using theoretical init n={n_parent}, score={_score(parent):.10f}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_optimize(parent, deadline, min_remaining=600)
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure equalization + re-optimize ---
    print(f"\n[Phase 2] Extended PE + re-opt (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        pe_deadline = min(deadline - 300, time.monotonic() + 120)
        a_pe, _ = _pressure_equalize_v2(best_seq, pe_deadline, n_iters=400, gamma=0.25)
        sc_pe = _score(a_pe)
        _update(a_pe, sc_pe)
        if sc_pe < best_score + 0.001:
            a_follow = _lbfgs_optimize(a_pe, deadline, min_time_remaining=250)
            _update(a_follow, _score(a_follow))

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 10240, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - 150 - time.monotonic()
        if remaining < 60:
            break

        # Interpolate parent to new length
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, best_seq if best_seq is not None else parent)
        a_init = np.maximum(a_init, 0)
        s_init = a_init.sum()
        if s_init > 1e-10:
            a_init = a_init / s_init
        else:
            a_init = _make_theoretical_init(n_try)

        min_rem = max(30, remaining * 0.4)
        a_opt, sc_opt = _multi_optimize(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        inits = [
            _make_theoretical_init(n_best),
            _make_tapered_init(n_best, power=0.3),
            _make_tapered_init(n_best, power=1.0),
        ]
        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            # Blend with best
            a_blend = 0.5 * best_seq + 0.5 * a_init
            a_blend = np.maximum(a_blend, 0) / max(a_blend.sum(), 1e-15)
            a_pert = _fourier_perturb(a_blend, rng, n_modes=3, amplitude=0.015)

            min_rem = max(20, (deadline - 30 - time.monotonic()) * 0.5)
            a_r, sc_r = _multi_optimize(a_pert, deadline, min_remaining=min_rem)
            _update(a_r, sc_r)

    # --- Phase 5: Intensive FISTA on best ---
    print(f"\n[Phase 5] Intensive FISTA (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        fista_deadline = min(deadline - 10, time.monotonic() + 300)
        # Try multiple beta values for FISTA
        for beta in [0.001, 0.0003, 0.0001]:
            if time.monotonic() > fista_deadline:
                break
            sub_deadline = min(fista_deadline, time.monotonic() + 100)
            a_f, sc_f = _fista_polish(best_seq, sub_deadline, n_iters=5000, beta_start=beta)
            if sc_f < best_score:
                _update(a_f, sc_f)

    # --- Phase 6: Final coordinate + L-BFGS polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_cd, sc_cd = _coordinate_polish_v2(best_seq, min(deadline - 5, time.monotonic() + 120), n_rounds=150)
        _update(a_cd, sc_cd)

        if time.monotonic() < deadline - 3:
            a_fin = _lbfgs_optimize(best_seq, deadline, min_time_remaining=2)
            _update(a_fin, _score(a_fin))

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_theoretical_init(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
