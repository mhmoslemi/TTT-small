"""
Optimizer combining LP-based power iteration (greedy LP), smoothed L-BFGS with
adaptive beta continuation, CG-Newton active set polish, and multi-n exploration
to minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key new idea: greedy LP approximates the "spreading" direction from the
state-of-the-art LP iteration, providing a different global search mechanism
complementary to gradient-based methods.

import numpy as np
import time
from scipy.optimize import minimize
from scipy import sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _greedy_lp_direction(f, M):
    """Greedy LP approximation: find b >= 0 with (f*b)_m <= M for all m,
    maximizing sum(b) greedily. Returns b (not normalized)."""
    n = len(f)
    c = _full_convolve(f, f)
    s = M - c  # slack, should be >= 0 (clip for numerical safety)
    s = np.maximum(s, 0.0)
    b = np.zeros(n)
    # Process in order; multiple passes for better solution
    for _pass in range(3):
        for j in range(n):
            # b_j limited by: s[m] / f[m-j] for m = j..j+n-1 where f[m-j] > 0
            f_window = f  # f[m-j] for m=j..j+n-1 is f[0..n-1]
            s_window = s[j:j + n]
            valid = f_window > 1e-30
            if not np.any(valid):
                continue
            deltas = s_window[valid] / f_window[valid]
            delta = float(np.min(deltas))
            if delta > 1e-15:
                b[j] += delta
                s[j:j + n] -= delta * f_window
                s[j:j + n] = np.maximum(s[j:j + n], 0.0)
    return b


def _lp_power_iteration(f, deadline, max_iters=30, min_time_remaining=300):
    """LP-based power iteration using greedy LP for direction finding."""
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 0)
    s_init = f.sum()
    if s_init < 1e-15:
        return f
    f = f / s_init
    best_f = f.copy()
    best_sc = _score(f)
    print(f"  LP-power: start score={best_sc:.10f}")

    for it in range(max_iters):
        if time.monotonic() > deadline - min_time_remaining:
            break
        t0 = time.monotonic()
        c = _full_convolve(f, f)
        M = float(c.max())

        # Greedy LP to find direction b
        b = _greedy_lp_direction(f, M)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum  # normalize to sum 1

        # Backtracking line search
        t_val = 1.0
        accepted = False
        for _ in range(15):
            f_try = (1 - t_val) * f + t_val * b
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                t_val *= 0.5
                continue
            f_try = f_try / s_try
            c_try = _full_convolve(f_try, f_try)
            c_max_try = float(c_try.max())
            if c_max_try <= M * (1 + t_val * 0.001):  # allow tiny increase

                accepted = True
                break
            t_val *= 0.5

        sc = _score(f)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()

        else:

            if it > 0 and not accepted:

        elapsed = time.monotonic() - t0
        if elapsed > 5.0 and it > 5:
            break  # LP getting too slow

def _newton_active_polish(a_init, deadline, max_rounds=80):
    """CG-Newton polish on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(a)

        if time.monotonic() > deadline:
            break

        c_max = float(c.max())

        tol = 0.0003 * c_max

        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        H1 = _cg_solve(W, n, np.ones(n), max_iter=250)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try <= c_max * (1 - 0.01 * alpha):
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.4, max_c * 0.003)

    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                       0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                       0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                       0.0000005, 0.0000002, 0.0000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 10))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, max_steps=8000):
    """Adaptive subgradient with dynamic active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.0008 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 25 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.04, 0.06)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.012
            z = z_new
        else:
            z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "trimodal":
        t = np.linspace(0, 1, n)
        a = (np.exp(-12.0 * (t - 0.2) ** 2) + np.exp(-12.0 * (t - 0.5) ** 2) +
             np.exp(-12.0 * (t - 0.8) ** 2) + 0.02)
        return a / a.sum()
    elif kind == "flat_centered":
        a = np.ones(n)
        center = n // 2
        width = n // 5
        a[center - width: center + width] = 0.7
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _nested_optimize(a_init, deadline, min_remaining, label=""):
    """Run L-BFGS + Newton polish pipeline on a starting point."""
    a1 = _run_lbfgs(a_init, deadline, min_time_remaining=max(min_remaining * 0.6, 10))
    sc1 = _score(a1)
    best_a = a1.copy()
    best_sc = sc1

    if time.monotonic() < deadline - max(min_remaining * 0.3, 5):
        a2, sc2 = _newton_active_polish(a1, deadline - max(min_remaining * 0.3, 5), max_rounds=60)
        if sc2 < best_sc:
            best_a = a2
            best_sc = sc2

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. LP-based power iteration (global search via greedy LP)
    2. Smoothed L-BFGS with aggressive beta continuation
    3. CG-Newton active set polish
    4. Subgradient polish
    5. Shrinkage + re-optimize
    6. Multi-n exploration
    7. Diverse restarts
    8. Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_work = n_parent
    a_work = parent.copy()

    # Phase 1: LP-based power iteration (NEW - global search)
    print(f"\n[Phase 1] LP-based power iteration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        a_lp = _lp_power_iteration(a_work, deadline, max_iters=20,
                                   min_time_remaining=700)
        sc_lp = _score(a_lp)
        update_best(a_lp, sc_lp, "lp_power")
        if sc_lp < _score(a_work) + 0.001:
            a_work = a_lp

    # Phase 2: Main L-BFGS
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a2 = _run_lbfgs(a_work, deadline, min_time_remaining=500)
    update_best(a2, _score(a2), "lbfgs_main")

    # Phase 3: CG-Newton active set polish
    print(f"\n[Phase 3] CG-Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 200),
                                         max_rounds=100)
        update_best(a3, sc3, "newton_1")

    # Phase 4: Subgradient polish
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _subgrad_polish(best_seq, min(deadline - 250, time.monotonic() + 150))
        update_best(a4, sc4, "sg_1")

    # Phase 5: Newton polish after subgradient
    print(f"\n[Phase 5] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a5, sc5 = _newton_active_polish(best_seq, min(deadline - 180, time.monotonic() + 100),
                                         max_rounds=80)
        update_best(a5, sc5, "newton_2")

    # Phase 6: Shrinkage + re-optimize
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a6, sc6 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=80)
        update_best(a6, sc6, "shrinkage")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = []
    if n_parent <= 4096:
        n_candidates = [8192, 16384]
    elif n_parent <= 8192:
        n_candidates = [16384, 4096]
    else:
        n_candidates = [n_parent // 2, n_parent * 2]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        if n_try < 1024 or n_try > 32768:
            continue
        print(f"  Trying n={n_try}")
        a_init = _resample(parent, n_try)
        a_opt, sc_opt = _nested_optimize(a_init, deadline, 60, f"n_{n_try}")
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 8: Diverse restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            (0.05, "perturb_small"),
            (0.15, "perturb_med"),
            (0.3, "perturb_large"),
        ]
        for sigma, label in restart_configs:
            if time.monotonic() > deadline - 25:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r, sc_r = _nested_optimize(p_start, deadline, 15, label)
            update_best(a_r, sc_r, f"restart_{label}")

        # Different profile
        if time.monotonic() < deadline - 20:
            p_start = _make_profile(len(best_seq), "bimodal")
            a_r, sc_r = _nested_optimize(p_start, deadline, 10, "profile_bimodal")
            update_best(a_r, sc_r, "profile_bimodal")

    # Phase 9: Final ultra-fine polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS with tiny beta
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-8
            remaining = deadline - 2 - time.monotonic()
            if remaining > 1:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
