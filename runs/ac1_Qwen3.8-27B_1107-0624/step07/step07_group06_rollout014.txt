"""
Improved optimizer: aggressive L-BFGS with log-sum-exp smoothing, Newton active-set
refinement, shrinkage with re-optimization, multi-n exploration, and structured
perturbation restarts.Targets beating 1.5057.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_matvec(v, active, n):
    """Compute H*v where H corresponds to the active set quadratic form."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=300, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0:
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        r_norm_sq = float(np.dot(r_new, r_new))
        if np.sqrt(r_norm_sq) < tol * b_norm:
            x = x_new
            break
        beta_cg = r_norm_sq / rs if rs > 1e-30 else 0.0
        p_new = r_new + beta_cg * p
        x = x_new
        r = r_new
        p = p_new
        rs = r_norm_sq
    return x

            return _hess_matvec(v, active, n) + mu * v

        dx = dx - np.mean(dx)

        if dx_norm < 1e-15:

        f_old = float(np.dot(optimize_dummy_weight(c, active), np.ones(1)))

            c_try = _full_convolve(a_try, a_try)

def optimize_dummy_weight(c, active):
    return c[active].sum()


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    # Very aggressive continuation schedule
    ratios = [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
              0.0025, 0.001, 0.0005, 0.00025, 0.0001, 0.00005, 0.00002, 0.00001]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration and set max_iter
        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time_per_iter * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, float(beta), n)

        def jac_local(z_local):
            return _smograd(z_local, float(beta), n)

        try:
            result = minimize(
                fun_local, z.copy(), jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_momentum(a_init, deadline, n_max_steps=5000, lr_init=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    momentum = np.zeros(n)
    beta_m = 0.9
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_grad(a, c, n, tol_frac=0.001)
        step = lr / (1.0 + 0.005 * i)
        momentum = beta_m * momentum - step * grad
        z = z + momentum
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_and_reopt(a, deadline, min_time_remaining=50.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.10, 0.15, 0.20, 0.30]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    # Re-optimize the shrunk version
    if best_sc < _score(a) - 1e-12 and time.monotonic() + min_time_remaining < deadline:
        a_opt = _run_lbfgs_aggressive(best_a, deadline, min_time_remaining=min_time_remaining)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "half_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_structured(a, rng, mode=0):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    if mode == 0:
        # Gaussian noise in log space
        z = z + 0.02 * rng.standard_normal(n)
    elif mode == 1:
        # Fourier perturbation
        t = np.arange(n)
        for _ in range(4):
            freq = rng.integers(1, max(2, n // 4))
            amp = 0.02 * rng.uniform(1.0, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == 2:
        # Block swap
        n_blocks = 3
        block_size = max(1, n // (2 * n_blocks))
        for _ in range(n_blocks):
            i = rng.integers(0, max(1, n - 2 * block_size))
            j = rng.integers(0, max(1, n - 2 * block_size))
            if abs(i - j) < block_size:
                continue
            tmp = z[i:i + block_size].copy()
            z[i:i + block_size] = z[j:j + block_size]
            z[j:j + block_size] = tmp
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: aggressive L-BFGS with log-sum-exp smoothing,
    Newton active-set, subgradient with momentum, shrinkage+re-optimize,
    multi-n exploration, and structured restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Main L-BFGS with aggressive continuation ===
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 700:
        a_main = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=700)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 2: Newton active-set ===
    print(f"\n[Phase 2] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        n_deadline = min(deadline - 500, time.monotonic() + 300)
        a_newton, sc_newton = _newton_active(best_seq, n_deadline, max_iter=300)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 3: Subgradient with momentum ===
    print(f"\n[Phase 3] Subgrad+momentum (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, time.monotonic() + 200)
        a_sg, sc_sg = _subgrad_momentum(best_seq, sg_deadline, n_max_steps=3000, lr_init=0.008)
        update_best(a_sg, sc_sg, "sg_momentum")

    # === Phase 4: Shrinkage + re-optimize ===
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_shr, sc_shr = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=300)
        update_best(a_shr, sc_shr, "shrinkage")

    # === Phase 5: Newton on best so far ===
    print(f"\n[Phase 5] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_deadline = min(deadline - 300, time.monotonic() + 200)
        a_n2, sc_n2 = _newton_active(best_seq, n_deadline, max_iter=200)
        update_best(a_n2, sc_n2, "newton_polish")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s < 1e-10:
                continue
            a_init = a_init / s
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        a_opt = _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=100)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 80:
            a_opt2, sc_opt2 = _newton_active(a_opt, min(deadline - 80, time.monotonic() + 80), max_iter=100)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 7: Structured restarts ===
    print(f"\n[Phase 7] Structured restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_restarts = 3
        for i in range(n_restarts):
            if time.monotonic() > deadline - 35:
                break
            p_start = _perturb_structured(best_seq, rng, mode=i)
            a_r = _run_lbfgs_aggressive(p_start, deadline, min_time_remaining=25)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _newton_active(a_r, min(deadline - 15, time.monotonic() + 30), max_iter=50)
                update_best(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 8: Final very-low-beta L-BFGS ===
    print(f"\n[Phase 8] Final ultra-low-beta L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.000005
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(5000, max(200, int(remaining / 0.005)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 100}
                )
                a_fin3 = _softmax(result.x)
                update_best(a_fin3, _score(a_fin3), "final_ultra")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _newton_active(best_seq, max(deadline - 2, time.monotonic()), max_iter=100)
            update_best(a_fin4, sc_fin4, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
