"""
Step sequence optimizer using adaptive active-set L-BFGS, spectral shaping,
mass redistribution, and diverse multi-restart to minimize
2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def identify_active_set(c, top_k_frac=0.02, tol_frac=5e-4):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    n_active = max(3, int(top_k_frac * len(c)))
    sorted_c = np.sort(c)[::-1]
    threshold_rank = sorted_c[min(n_active - 1, len(c) - 1)]
    tol_rank = c_max - threshold_rank
    tol_fixed = tol_frac * c_max
    tol = max(tol_fixed, min(tol_rank, 0.4 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def gradient_active_set(a, w, n):
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def objective_active_set(z, w, n, beta=0.0):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    if beta > 0:
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return float(np.dot(w, c)) + 2.0 * n * c_max
        return float(2.0 * n * (c_max + beta * np.log(s)))
    return float(2.0 * n * np.dot(w, c))


def gradient_obj_active(z, w, n, beta=0.0):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    if beta > 0:
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            ww = np.zeros_like(c)
            ww[int(np.argmax(c))] = 1.0
        else:
            ww = e / s
    else:
        ww = w
    g_direct = gradient_active_set(a, ww, n)
    W = float(np.dot(ww, c))
    return 4.0 * n * a * (g_direct / 2.0 - W)


def lbfgs_active_set(a_init, w, n, max_iter, ftol=1e-20, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return objective_active_set(z, w, n, beta=0.0)

    def jac(z):
        return gradient_obj_active(z, w, n, beta=0.0)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 100, 'disp': False})
        return softmax_fn(result.x), result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def lbfgs_smooth(z_init, beta, n, max_iter, ftol=1e-20, gtol=1e-14):
    a_init = softmax_fn(z_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        a = softmax_fn(z)
        c = full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def jac(z):
        a = softmax_fn(z)
        c = full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            ww = np.zeros_like(c)
            ww[int(np.argmax(c))] = 1.0
        else:
            ww = e / s
        a_rev = a[::-1]
        wc = full_convolve(ww, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(ww, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 100, 'disp': False})
        return softmax_fn(result.x), result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

    a = a / s

def spectral_shaping(a_init, deadline, max_iters=500):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score_fn(a)
    N = next_pow2(2 * n - 1)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A) ** 2
        power_mean = power.mean()
        power_std = power.std()
        if power_std < 1e-15:
            break

        residual = power - power_mean
        grad_freq = np.zeros_like(A)
        grad_freq[:] = 2.0 * residual * A
        grad_time = np.fft.irfft(grad_freq, N)[:n]
        grad_time = grad_time - grad_time.mean()

        g_norm = np.linalg.norm(grad_time)
        if g_norm < 1e-15:
            break
        grad_time = grad_time / g_norm

        best_lr = 0.0
        best_a_try = a.copy()
        best_sc_try = best_sc

        for frac in [1.0, 0.5, 0.25, 0.1]:
            step = lr * frac
            a_try = a - step * grad_time
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = score_fn(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_lr = step

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            lr = min(lr * 1.03, 0.02)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break

    return best_a, best_sc


def mass_redistribution(a_init, deadline, max_iters=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score_fn(a)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = full_convolve(a, a)
        c_max = c.max()
        peak_idx = int(np.argmax(c))

        contributors = []
        for j in range(n):
            k_j = peak_idx - j
            if 0 <= k_j < n:
                contributors.append((j, k_j, a[j] * a[k_j]))
        contributors.sort(key=lambda x: -x[2])

        if not contributors:
            break

        top_contribs = contributors[:min(8, len(contributors))]
        reduce_amount = 0.001 * c_max / max(1, (sum(a[j] for j, _, _ in top_contribs) + 1e-15))
        reduce_amount = min(reduce_amount, 0.01 * a[contributors[0][0]])

        new_a = a.copy()
        total_removed = 0.0
        for j, kj, val in top_contribs:
            removal = min(reduce_amount, new_a[j] * 0.5)
            if removal > 1e-15:
                new_a[j] -= removal
                total_removed += removal

        if total_removed < 1e-15:
            break

        non_peak_positions = []
        peak_positions = set()
        for k_peak in np.argsort(c)[::-1][:20]:
            for j in range(n):
                if 0 <= k_peak - j < n:
                    peak_positions.add(j)

        for j in range(n):
            if j not in peak_positions and new_a[j] < 0.01:
                non_peak_positions.append(j)

        if non_peak_positions:
            add_each = total_removed / len(non_peak_positions)
            for j in non_peak_positions:
                new_a[j] += add_each

        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-10:
            break
        new_a = new_a / s_new
        sc_new = score_fn(new_a)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = new_a.copy()
            a = new_a
        else:
            reduce_amount *= 0.5
            if reduce_amount < 1e-12:
                break

    return best_a, best_sc


def coordinate_descent_pairs(a_init, deadline, max_rounds=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = score_fn(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        fracs = np.linspace(0.1, 0.9, 9)
        for frac in fracs:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            if new_i < 0 or new_j < 0:
                continue
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = score_fn(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and rnd % 100 == 0:
            idx2 = int(rng.integers(0, n))
            jdx2 = int(rng.integers(0, n))
            kdx2 = int(rng.integers(0, n))
            if idx2 == jdx2 or idx2 == kdx2 or jdx2 == kdx2:
                continue
            s_trip = a[idx2] + a[jdx2] + a[kdx2]
            if s_trip < 1e-15:
                continue
            for f1 in [0.2, 0.4, 0.6]:
                for f2 in [0.1, 0.3, 0.5]:
                    new1 = s_trip * f1
                    new2 = s_trip * f2 * (1 - f1)
                    new3 = s_trip * (1 - f1) * (1 - f2)
                    if new1 < 0 or new2 < 0 or new3 < 0:
                        continue
                    a_try = a.copy()
                    a_try[idx2] = new1
                    a_try[jdx2] = new2
                    a_try[kdx2] = new3
                    sc_try = score_fn(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break

    return best_a, best_sc

def perturb_fourier(a, rng, n_modes=6, amp_scale=0.005):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = amp_scale * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return softmax_fn(z)


def perturb_gauss(a, rng, scale=0.03):

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return softmax_fn(z)


def perturb_block(a, rng, n_blocks=10):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    block_size = max(4, n // 32)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)
        z[start:end] += rng.uniform(-0.005, 0.005)
    z = z - z.mean()
    return softmax_fn(z)


def make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.5, 0.7)
        return a / a.sum()
    else:
        return np.ones(n) / n

    print(f"\n=== Phase 2: Smooth continuation ===")
    t_phase = time.monotonic()
    remaining_for_p2 = min(300.0, deadline - t_phase - 400.0)
    a_p2 = smooth_continuation(best_a, t_phase + max(30, remaining_for_p2), min_time_remaining=120.0)
    try_update(a_p2, "smooth_cont")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, score={score_fn(a_p2):.10f}")

    # Phase 3: Peak shaving
    print(f"\n=== Phase 3: Peak shaving ===")
    t_phase = time.monotonic()
    remaining_for_p3 = min(200.0, deadline - t_phase - 300.0)
    if remaining_for_p3 > 30:
        a_p3, sc_p3 = peak_shaving_adaptive(best_a, t_phase + remaining_for_p3, max_iters=40000)
        try_update(a_p3, "peak_shave")
        print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s, score={sc_p3:.10f}")
    else:
        print(f"  Phase 3 skipped (insufficient time)")

    # Phase 4: Spectral shaping
    print(f"\n=== Phase 4: Spectral shaping ===")
    t_phase = time.monotonic()
    remaining_for_p4 = min(150.0, deadline - t_phase - 200.0)
    if remaining_for_p4 > 20:
        a_p4, sc_p4 = spectral_shaping(best_a, t_phase + remaining_for_p4, max_iters=300)
        try_update(a_p4, "spectral")
        print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s, score={sc_p4:.10f}")
    else:
        print(f"  Phase 4 skipped (insufficient time)")

    # Phase 5: Mass redistribution
    print(f"\n=== Phase 5: Mass redistribution ===")
    t_phase = time.monotonic()
    remaining_for_p5 = min(100.0, deadline - t_phase - 150.0)
    if remaining_for_p5 > 20:
        a_p5, sc_p5 = mass_redistribution(best_a, t_phase + remaining_for_p5, max_iters=15000)
        try_update(a_p5, "mass_redist")
        print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s, score={sc_p5:.10f}")
    else:
        print(f"  Phase 5 skipped (insufficient time)")

    # Phase 6: Multi-restart with diverse perturbations
    print(f"\n=== Phase 6: Multi-restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        ("fourier1", "fourier", 6, 0.004),
        ("fourier2", "fourier", 8, 0.008),
        ("gauss1", "gauss", 0, 0.02),
        ("gauss2", "gauss", 0, 0.05),
        ("block1", "block", 8, 0.0),
        ("block2", "block", 16, 0.0),
        ("fourier3", "fourier", 4, 0.012),
    ]

    for name, mode, nat, scale in restart_configs:
        if time.monotonic() > deadline - 100:
            break
        t_restart = time.monotonic()
        remaining_r = min(120.0, deadline - t_restart - 80.0)
        if remaining_r < 20:
            break

        if mode == "fourier":
            p_start = perturb_fourier(best_a, rng, n_modes=nat, amp_scale=scale)
        elif mode == "gauss":
            p_start = perturb_gauss(best_a, rng, scale=scale)
        else:
            p_start = perturb_block(best_a, rng, n_blocks=nat)

        sc_start = score_fn(p_start)
        print(f"  Restart [{name}]: start={sc_start:.10f}")

        a_r = iterative_active_set_refine(p_start, t_restart + remaining_r * 0.6, max_rounds=20)
        try_update(a_r, f"restart_{name}")

        if time.monotonic() < deadline - 80:
            a_r2, _ = peak_shaving_adaptive(a_r, deadline - 80, max_iters=8000)
            try_update(a_r2, f"restart_{name}_ps")

        if time.monotonic() < deadline - 70:
            a_r3 = smooth_continuation(a_r, deadline - 70, min_time_remaining=20.0)
            try_update(a_r3, f"restart_{name}_smooth")

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # Phase 7: Final intensive polish cycle
    print(f"\n=== Phase 7: Final polish ===")
    t_phase = time.monotonic()
    for cycle in range(4):
        if time.monotonic() > deadline - 5:
            break

        if time.monotonic() < deadline - 10:
            a_f1 = iterative_active_set_refine(best_a, deadline - 5, max_rounds=15)
            try_update(a_f1, f"polish_asm_{cycle}")

        if time.monotonic() < deadline - 8:
            a_f2, _ = peak_shaving_adaptive(best_a, deadline - 5, max_iters=10000)
            try_update(a_f2, f"polish_ps_{cycle}")

        if time.monotonic() < deadline - 6:
            a_f3, _ = coordinate_descent_pairs(best_a, deadline - 4, max_rounds=5000)
            try_update(a_f3, f"polish_cd_{cycle}")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # Final coordinate descent
    print(f"\n=== Final coord descent ===")
    if time.monotonic() < deadline - 2:
        a_fd, _ = coordinate_descent_pairs(best_a, deadline - 1, max_rounds=8000)
        try_update(a_fd, "final_cd")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = make_profile(256, "uniform")
        sc_final = score_fn(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = make_profile(8192, "uniform")
    print(f"FALLBACK: score={score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
