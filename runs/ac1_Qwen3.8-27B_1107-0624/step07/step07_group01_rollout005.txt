import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_objective(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lse_gradient(z, beta, n):
    """Gradient of the log-sum-exp smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        except Exception:
            continue

    return current_a

def _remez_newton(a_start, deadline, k_active=8, max_iter=200, damp=1.0):
    """
    Remez-Newton method: iteratively equalize the active lags.
    Variables are in log-space (z), with a = softmax(z).
    """
    n = len(a_start)
    z = np.log(np.maximum(a_start, 1e-18))
    z = z - z.mean()
    best_a = a_start.copy()
    best_sc = _score(a_start)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Select active set: top-k lags
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        
        # Check convergence
        spread = active_vals.max() - active_vals.min()
        if spread < 1e-12 * c_max:
            break

        # Build Jacobian J of shape (k_active-1) x n
        # J[i, j] = dc[k_{i+1}]/dz[j] - dc[k_last]/dz[j]
        # = 2*a[j]*(a[k_{i+1}-j] - a[k_last-j]) (with bounds)
        m = k_active - 1
        J = np.zeros((m, n))
        
        for i in range(m):
            k1 = idx[i]
            k2 = idx[-1]
            # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n else 0
            # dc[k]/dz[j] = 2*a[j]*a[k-j] if 0 <= k-j < n else 0
            # So J[i,j] = 2*a[j]*(a[k1-j] - a[k2-j])
            
            # Vectorized: for each j, compute a[k1-j] and a[k2-j]
            # a[k1-j] is valid when 0 <= k1-j < n, i.e., k1-n+1 <= j <= k1
            # a[k2-j] is valid when 0 <= k2-j < n, i.e., k2-n+1 <= j <= k2
            
            v1 = np.zeros(n)
            v2 = np.zeros(n)
            
            j_start1 = max(0, k1 - n + 1)
            j_end1 = min(n - 1, k1)
            if j_start1 <= j_end1:
                v1[j_start1:j_end1+1] = a[k1 - np.arange(j_start1, j_end1+1)]
            
            j_start2 = max(0, k2 - n + 1)
            j_end2 = min(n - 1, k2)
            if j_start2 <= j_end2:
                v2[j_start2:j_end2+1] = a[k2 - np.arange(j_start2, j_end2+1)]
            
            J[i, :] = 2.0 * a * (v1 - v2)
        
        # Residual: F[i] = c[idx[i]] - c[idx[-1]]
        F = c[idx[:m]] - c[idx[-1]]
        
        # Solve J J^T x = F for x, then step = J^T x
        JJT = J @ J.T
        # Add small regularization for numerical stability
        JJT += 1e-10 * np.eye(m)
        
        try:
            x = np.linalg.solve(JJT, F)
            step = J.T @ x
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(JJT, F, rcond=None)[0]
            step = J.T @ x

        # Normalize step (it should preserve the softmax normalization automatically)
        # Apply damped step
        alpha = damp
        improved = False
        
        for _ in range(10):
            z_try = z - alpha * step
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(2.0 * n * c_try.max())  # sum(a)=1 so score = 2n*max(c)
            
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
            
            if sc_try <= _score(a) + 1e-10:
                z = z_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try without the Newton step - just do a subgradient step
            break

    return best_a, best_sc

def _mirror_descent(a_start, deadline, k_active=5, max_steps=50000):
    """Mirror descent (exponential weights) for the subgradient of max(conv)."""
    n = len(a_start)
    z = np.log(np.maximum(a_start, 1e-18))
    z = z - z.mean()
    best_a = a_start.copy()
    best_sc = _score(a_start)
    
    # Step size schedule
    lr = 0.02
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set with soft weights
        idx = np.argsort(c)[::-1][:k_active]
        # Soft weights: proportional to how close to max
        active_vals = c[idx]
        weights = active_vals / active_vals.sum()
        
        w = np.zeros_like(c)
        w[idx] = weights

        # Subgradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Subtract mean to stay on the simplex in log-space
        grad = grad - np.dot(grad, a)  # project onto tangent space

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Update
        z = z - lr * grad / max(gnorm, 1e-10)
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        # Periodic check
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.01
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc

def _make_raised_cosine(n, width=0.5):
    """Raised cosine window."""
    t = np.linspace(0, 1, n)
    a = np.maximum(0.0, np.cos(np.pi * (t - 0.5) / width))
    a = a ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_gaussian_window(n, sigma=0.3):
    """Gaussian window."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = a / a.sum()
    return a

def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

def _symmetrize(a):
    """Average a with its reverse (exploits symmetry of the problem)."""
    return (a + a[::-1]) / 2.0

def _polish_round(a_start, deadline, methods=None):
    """Apply a sequence of polishing methods."""
    if methods is None:
        methods = ['lbfgs', 'remez', 'mirror']
    
    a = a_start.copy()
    sc = _score(a)
    
    for method in methods:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        if method == 'lbfgs':
            a_new = _adaptive_lbfgs(a, deadline, reserve_time=max(30, remaining * 0.1))
            sc_new = _score(a_new)
            if sc_new < sc:
                a = a_new
                sc = sc_new
        elif method == 'remez':
            a_new, sc_new = _remez_newton(a, deadline, k_active=10, max_iter=100)
            if sc_new < sc:
                a = a_new
                sc = sc_new
        elif method == 'mirror':
            a_new, sc_new = _mirror_descent(a, deadline, k_active=5, max_steps=20000)
            if sc_new < sc:
                a = a_new
                sc = sc_new
    
    return a, sc
