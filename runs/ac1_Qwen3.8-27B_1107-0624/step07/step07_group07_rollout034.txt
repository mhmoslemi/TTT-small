"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2.
Uses edge-concentrated initialization, top-K active set gradient descent,
L-BFGS with adaptive beta continuation, direct-space peak shaving,
and coordinate descent polishing. Key innovation: structured initializations
based on theoretical analysis of flat-autocorrelation limits, combined with
a top-K peak focusing strategy for the active set.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    b = wc[n - 1:2 * n - 1]
    W = float(np.dot(b, a))
    return 4.0 * n * a * (b - W)


def _topk_gradient(a, c, n, K):
    """Compute gradient using top-K active set."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    # Get indices of top K peaks
    if K >= len(c):
        k_idx = np.arange(len(c))
        w = np.ones(len(c)) / len(c)
    else:
        k_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        w[k_idx] = 1.0 / K
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    b = wc[n - 1:2 * n - 1]
    W = float(np.dot(b, a))
    return 2.0 * n * (b - W)


def _make_edge_concentrated(n):
    """Create an edge-concentrated profile based on theoretical analysis.
    The ideal flat-autocorrelation sequence has values ~ 1/sqrt(2n) at edges."""
    t = np.linspace(0, 1, n)
    # V-shape: high at edges, low in middle
    a = np.abs(2 * t - 1) ** 0.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tent(n):
    """Tent-shaped profile: linear increase then decrease."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian_edges(n):
    """Gaussian concentrated at edges."""
    t = np.linspace(0, 1, n)
    sigma = 0.1
    a = np.exp(-((t - 0) / sigma) ** 2) + np.exp(-((t - 1) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cosine_taper(n, order=2):
    """Cosine-tapered profile."""
    t = np.linspace(0, 1, n)
    a = (1.0 + np.cos(np.pi * t)) / 2.0
    a = a ** order
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _perturb(a, rng, mode="gauss", scale=0.03):

def _lbfgs_smoothed(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-16):

        return _smoothed_grad(z, beta, n)

                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _topk_lbfgs(a_init, K, deadline, max_iter=5000):
    """L-BFGS using top-K active set gradient (not smoothed)."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_sorted = np.sort(c)[::-1]
        return 2.0 * n * c_sorted[:K].mean()

    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        g_direct = _topk_gradient(a, c, n, K)
        # Convert to z-gradient: grad_z = a * (grad_a - dot(grad_a, a))
        W = float(np.dot(g_direct, a))
        return a * (g_direct - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _coordinate_descent(a_init, deadline, max_rounds=20000):

                    if sc_try < best_sc - 1e-14:

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

                try_update(a_cd, f"final_cd_{cycle}")
    print(f"  Phase 10 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(n_default)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_uniform(n_default)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
