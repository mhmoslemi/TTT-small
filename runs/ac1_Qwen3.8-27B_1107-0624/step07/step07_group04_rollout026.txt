                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback
            alpha = 0.01
            for ls in range(30):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Adaptive subgradient method with dynamic active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _support_reduction(a, deadline, min_time_remaining=30.0):
    """Iteratively remove smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    print(f"  Support reduction start: n={n}, score={best_sc:.10f}")

    for frac in [0.02, 0.04, 0.06, 0.08, 0.10, 0.12, 0.15, 0.20]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(len(best_a) * frac))
        a_try = best_a.copy()
        a_try[np.argsort(a_try)[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc * 1.002:  # Accept if not much worse
            best_sc = min(best_sc, sc_try)
            best_a = a_try.copy()
            # Re-optimize from this point
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_opt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt
            # Quick Newton polish
            if time.monotonic() + min_time_remaining > deadline:
                a_np, sc_np = _newton_active_polish(best_a, deadline - min_time_remaining, max_rounds=30)
                if sc_np < best_sc:
                    best_sc = sc_np
                    best_a = a_np
            print(f"    frac={frac:.2f}: n_eff={int((best_a > 0).sum())}, score={best_sc:.10f}")

    return best_a, best_sc


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale search: L-BFGS with beta continuation + Newton active-set
    + support reduction + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # -------------------------------------------------------------------------
    # Phase 0: Load parent or create initial solution
    # -------------------------------------------------------------------------
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # -------------------------------------------------------------------------
    # Phase 1: Multi-scale refinement
    # -------------------------------------------------------------------------
    target_n = max(n_parent, 8192)
    # Build up from small n to target_n
    scale_ns = []
    n_cur = 512
    while n_cur < target_n:
        scale_ns.append(n_cur)
        n_cur *= 2
    if n_cur != target_n:
        scale_ns.append(n_cur)
    if target_n not in scale_ns:
        scale_ns.append(target_n)

    print(f"\n[Phase 1] Multi-scale: scales={scale_ns}")
    a_current = _resample(parent, scale_ns[0]) if scale_ns[0] != n_parent else parent.copy()

    for idx, n_scale in enumerate(scale_ns):
        if time.monotonic() > deadline - 300:
            break
        remaining_budget = deadline - 300 - time.monotonic()
        alloc = min(remaining_budget, max(30, 120 * (n_scale / 8192.0)))
        print(f"  Scale n={n_scale}, budget={alloc:.0f}s")

        # L-BFGS
        a_scale = _run_lbfgs(a_current, t_start + 1800 - 300 - (len(scale_ns) - 1 - idx) * 60,
                             min_time_remaining=min(30, alloc * 0.3))
        update_best(a_scale, _score(a_scale), f"scale_{n_scale}")

        # Quick Newton
        if time.monotonic() < deadline - 300:
            a_np, sc_np = _newton_active_polish(a_scale, deadline - 300 - (len(scale_ns) - 1 - idx) * 30,
                                                max_rounds=50)
            if sc_np < _score(a_scale):
                a_scale = a_np
                update_best(a_scale, sc_np, f"scale_{n_scale}_np")

        a_current = a_scale

    # Now a_current is at target_n
    if len(a_current) != target_n:
        a_current = _resample(a_current, target_n)

    # -------------------------------------------------------------------------
    # Phase 2: Full L-BFGS at target scale
    # -------------------------------------------------------------------------
    print(f"\n[Phase 2] Full L-BFGS n={target_n} (t={time.monotonic()-t_start:.1f}s)")
    a2 = _run_lbfgs(a_current, deadline, min_time_remaining=400.0)
    update_best(a2, _score(a2), "lbfgs_full")

    # -------------------------------------------------------------------------
    # Phase 3: Newton active-set polish
    # -------------------------------------------------------------------------
    print(f"\n[Phase 3] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a3, sc3 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 200),
                                        max_rounds=150, tight_tol=1e-7)
        update_best(a3, sc3, "newton_1")

    # -------------------------------------------------------------------------
    # Phase 4: Subgradient polish
    # -------------------------------------------------------------------------
    print(f"\n[Phase 4] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a4, sc4 = _subgrad_polish(best_seq, min(deadline - 200, time.monotonic() + 100),
                                  max_steps=2000)
        update_best(a4, sc4, "subgrad")

    # -------------------------------------------------------------------------
    # Phase 5: Second Newton pass
    # -------------------------------------------------------------------------
    print(f"\n[Phase 5] Newton 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a5, sc5 = _newton_active_polish(best_seq, min(deadline - 150, time.monotonic() + 120),
                                        max_rounds=100, tight_tol=5e-8)
        update_best(a5, sc5, "newton_2")

    # -------------------------------------------------------------------------
    # Phase 6: Support reduction
    # -------------------------------------------------------------------------
    print(f"\n[Phase 6] Support reduction (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a6, sc6 = _support_reduction(best_seq, deadline, min_time_remaining=80.0)
        update_best(a6, sc6, "support_red")

    # -------------------------------------------------------------------------
    # Phase 7: Diverse restarts
    # -------------------------------------------------------------------------
    print(f"\n[Phase 7] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restarts = [
            ("perturb_0.02", _perturb(best_seq, 0.02, rng)),
            ("perturb_0.05", _perturb(best_seq, 0.05, rng)),
            ("gauss_reopt", _make_profile(len(best_seq), "gauss")),
            ("cos_reopt", _make_profile(len(best_seq), "cos")),
        ]
        for label, p_start in restarts:
            if time.monotonic() > deadline - 30:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, label)
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 10, time.monotonic() + 40),
                                                    max_rounds=50)
                update_best(a_rp, sc_rp, label + "_np")

    # -------------------------------------------------------------------------
    # Phase 8: Final ultra-fine polish
    # -------------------------------------------------------------------------
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=200,
                                              tight_tol=1e-8)
        update_best(a_fin, sc_fin, "final_newton")

        if time.monotonic() < deadline - 1:
            # Ultra-tight L-BFGS
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            remaining = deadline - 1 - time.monotonic()
            if remaining > 1:
                _G_BETA = mb * 5e-8
                _G_N = n
                _G_SUPPORT = None
                max_iter = min(2000, max(20, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _smoobj, z, jac=_smograd, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 60, 'maxls': 60}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # -------------------------------------------------------------------------
    # Final validation
    # -------------------------------------------------------------------------
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
