"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Two-ramp (U-shape) initialization from binomial coefficients
2. Aggressive multi-scale L-BFGS with log-sum-exp smoothing
3. Projected subgradient with Nesterov momentum on simplex
4. Peak-shaving heuristic for direct peak reduction
5. Multi-length exploration and final squeeze phase
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import gammaln


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_two_ramp(n):
    k = np.arange(n, dtype=np.float64)
    log_f = gammaln(2 * k + 1) - 2 * gammaln(k + 1) - k * np.log(4.0)
    f = np.exp(log_f)
    a = f + f[::-1]
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_one_ramp(n):
    k = np.arange(n, dtype=np.float64)
    log_f = gammaln(2 * k + 1) - 2 * gammaln(k + 1) - k * np.log(4.0)
    f = np.exp(log_f)
    a = np.maximum(f, 1e-15)
    return a / a.sum()


def _make_power_u(n, p=0.3):
    k = np.arange(n, dtype=np.float64)
    a = (k + 1.0) ** (-p) + (n - k) ** (-p) + 0.01
    return a / a.sum()


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_mixed_ramp(n, blend=0.5):
    ramp1 = _make_one_ramp(n)
    ramp2 = _make_two_ramp(n)
    a = blend * ramp1 + (1.0 - blend) * ramp2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.0125, 0.00625,
                              0.003125, 0.0015625, 0.00078125,
                              0.000390625, 0.0001953125,
                              0.00009765625, 0.000048828125,
                              0.0000244140625, 0.00001220703125,
                              0.0000061035156, 0.0000030517578,
                              0.0000015258789, 0.0000007629395,
                              0.0000003814697, 0.0000001907349])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(30000, max(200, int(remaining / 0.002)))
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
        if current_sc < 1.5020:
            break
    return current_a


def _projected_subgrad(a_init, deadline, k_start=1, k_end=15, max_steps=50000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)
    lb = 1.0
    t = 1.0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        k = min(k_end, k_start + (i * (k_end - k_start + 1)) // max_steps)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[idx] = 1.0 / k

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        f_cur = c_max
        gnorm_sq = float(np.dot(h, h))
        if gnorm_sq < 1e-30:
            break

        alpha = (f_cur - lb / (2.0 * n)) / (gnorm_sq + 1e-15)
        alpha = min(alpha, 0.5)

        t_new = np.sqrt(t * t + 1.0)
        theta = (t - 1.0) / t_new
        y = a + theta * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a

        a_new = y - alpha * h
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            alpha *= 0.5
            a_new = y - alpha * h
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                t = t_new
                v = a - y
                continue
        a_new = a_new / sn
        v = a_new - y
        t = t_new

        sc_try = _score(a_new)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def _peak_shave(a_init, deadline, max_iter=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    alpha = 0.005
    stall = 0

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        j_star = int(np.argmax(c))

        g = np.zeros(n)
        lo = max(0, j_star - n + 1)
        hi = min(n, j_star + 1)
        valid_k = np.arange(lo, hi)
        g[valid_k] = a[j_star - valid_k]

        a_try = a - alpha * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            alpha *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        cur_sc = _score(a)
        if sc_try < cur_sc - 1e-16:
            a = a_try
            alpha = min(alpha * 1.02, 0.05)
        else:
            stall += 1
            alpha *= 0.8
            if stall > 200 or alpha < 1e-12:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=8, reserve=10.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    remaining = deadline - reserve - time.monotonic()
    if remaining < 30:
        return best_a, best_sc
    time_per_hop = max(20.0, remaining / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.7

        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(3):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.03 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 4 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            a_pert = best_a + 0.02 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
            else:
                a_pert = best_a.copy()

        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        a_opt = _lbfgs_smooth(a_pert, beta, n, 5000)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Phase 1: Initialization
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    n_init = len(parent) if parent is not None else 4096

    inits = [
        ("two_ramp", _make_two_ramp(n_init)),
        ("one_ramp", _make_one_ramp(n_init)),
        ("mixed_0.3", _make_mixed_ramp(n_init, 0.3)),
        ("mixed_0.7", _make_mixed_ramp(n_init, 0.7)),
        ("power_u_02", _make_power_u(n_init, 0.2)),
        ("power_u_04", _make_power_u(n_init, 0.4)),
        ("chirp_1", _make_chirp(n_init, 1.0, 0.3)),
        ("chirp_3", _make_chirp(n_init, 3.0, 0.2)),
    ]

    for name, a_init in inits:
        sc = _score(a_init)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_init, name)

    if best_seq is None:
        best_seq = _make_two_ramp(4096)
        best_score_val = _score(best_seq)

    print(f"  Phase 1 done. Best: {best_score_val:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # Phase 2: Main L-BFGS
    print(f"\n=== Phase 2: Main L-BFGS ===")
    a_p2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=500.0)
    try_update(a_p2, "lbfgs_main")

    # Phase 3: Projected subgradient
    print(f"\n=== Phase 3: Projected subgradient ===")
    if time.monotonic() < deadline - 350:
        a_sg, sc_sg = _projected_subgrad(best_seq, deadline - 350, k_start=1, k_end=12)
        try_update(a_sg, "proj_sg_1_12")

    if time.monotonic() < deadline - 250:
        a_sg2, sc_sg2 = _projected_subgrad(best_seq, deadline - 250, k_start=3, k_end=20)
        try_update(a_sg2, "proj_sg_3_20")

    # Phase 4: Peak shaving
    print(f"\n=== Phase 4: Peak shaving ===")
    if time.monotonic() < deadline - 200:
        a_ps, sc_ps = _peak_shave(best_seq, deadline - 200)
        try_update(a_ps, "peak_shave")

    # Phase 5: Second L-BFGS round (tighter)
    print(f"\n=== Phase 5: Tight L-BFGS ===")
    if time.monotonic() < deadline - 200:
        a_p5 = _adaptive_lbfgs(best_seq, deadline, reserve_time=150.0)
        try_update(a_p5, "lbfgs_tight")

    # Phase 6: Basin hopping
    print(f"\n=== Phase 6: Basin hopping ===")
    if time.monotonic() < deadline - 120:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 120, rng, n_hops=6, reserve=60.0)
        try_update(a_hop, "basin_hop")

    # Phase 7: Multi-length exploration
    print(f"\n=== Phase 7: Multi-length ===")
    if time.monotonic() < deadline - 100:
        current_n = len(best_seq)
        lengths_to_try = []
        for ln in [2048, 4096, 8192, 16384, 32768]:
            if ln != current_n:
                lengths_to_try.append(ln)
        # Sort by proximity to current_n to try similar sizes first
        lengths_to_try.sort(key=lambda x: abs(x - current_n))
        for target_n in lengths_to_try[:4]:
            if time.monotonic() > deadline - 60:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_int = _score(a_interp)
            print(f"    Interpolated: {sc_int:.10f}")
            try_update(a_interp, f"interp_{target_n}")
            if time.monotonic() < deadline - 50:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"lbfgs_{target_n}")

    # Phase 8: Final squeeze - alternate methods
    print(f"\n=== Phase 8: Final squeeze ===")
    for rnd in range(6):
        if time.monotonic() > deadline - 5:
            break

        # L-BFGS with very tight beta
        n_cur = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta = mb * (3e-7 * (0.5 ** rnd))
        remaining_iter = deadline - 5 - time.monotonic()
        max_iter = min(20000, max(500, int(remaining_iter / 0.002)))
        a_f = _lbfgs_smooth(best_seq, beta, n_cur, max_iter)
        try_update(a_f, f"squeeze_lbfgs_{rnd}")

        # Peak shave
        if time.monotonic() < deadline - 5:
            a_s, _ = _peak_shave(best_seq, deadline - 5, max_iter=5000)
            try_update(a_s, f"squeeze_ps_{rnd}")

        # Quick subgradient
        if time.monotonic() < deadline - 5:
            a_g, _ = _projected_subgrad(best_seq, deadline - 5, k_start=1, k_end=5, max_steps=5000)
            try_update(a_g, f"squeeze_sg_{rnd}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
