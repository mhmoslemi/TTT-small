Active set QP + log-sum-exp polish optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses matrix-free CG with FFT-based Hankel matvec for the inner QP,
combined with smooth L-BFGS polishing.

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """Compute M*v where M_{j,l} = s_vec[j+l]. v has length n, s_vec has length 2n-1."""
    v_flip = v[::-1]
    N = _next_pow2(2 * n)
    F_v = np.fft.rfft(v_flip, N)
    F_s = np.fft.rfft(s_vec, N)
    conv = np.fft.irfft(F_v * F_s, N)
    return conv[n-1:2*n-1]

    rhs has length len(F_free). x0 is optional initial guess (also on free set).

    if nf == 1:

            return np.array([0.0])
        return np.array([rhs[0] / m_val])

        v_full[free_idx] = v_free
        result_full = _hankel_matvec(v_full, s_vec, n)
        return result_full[free_idx]

    rs = float(r @ r)

        r = r_new
        rs = rs_new
        if rs < tol * tol * max(rs0, 1e-30):
