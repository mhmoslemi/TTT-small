"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set weighted gradient descent with FFT convolution,
exact simplex projection, coordinate descent polishing,
multi-n exploration, and structured restarts.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smoothed_gradient(a, c, T, top_k=10):
    n = len(a)
    c_centered = c - c.max()
    e = np.exp(c_centered / T)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    # Focus on top-k lags
    if top_k < len(c):
        top_idx = np.argsort(c)[::-1][:top_k]
        mask = np.zeros_like(p)
        mask[top_idx] = p[top_idx]
        mask_sum = mask.sum()
        if mask_sum > 1e-300:
            p = mask / mask_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=10, decay=0.8):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, active_k=10, decay=0.75,
                   eval_interval=100, clip_norm=0.5, use_backtrack=True,
                   plateau_frac=0.2, sa_prob=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    rng = np.random.default_rng(12345)
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / max(1e-10, (1.0 - plateau_frac))
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if T < T_start * 0.01:
            g = _compute_active_gradient(a, c, active_k, decay)
        else:
            g = _compute_smoothed_gradient(a, c, T, top_k=active_k)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        # Simulated annealing jump
        if sa_prob > 0 and rng.random() < sa_prob:
            jump_size = 0.05 * (1.0 - frac)
            z_new = z_new + jump_size * rng.standard_normal(n)
            z_new = z_new - z_new.mean()

        if use_backtrack and t % 30 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.02:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.03:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fast_cd_polish(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for k in active:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue
            # Try modifying each j in the active set
            for j in js[:50]:  # Limit to 50 per round for speed
                for scale in [1.2, 1.1, 0.9, 0.8, 0.7, 0.5, 1.5, 2.0, 0.0]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_a, best_score

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fista_simplex(a_init, deadline, max_iter=50000):
    """FISTA on the simplex for the smoothed objective."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    y = a.copy()
    t_k = 1.0
    step = 0.01
    eval_interval = 200

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        T = c_max * 0.003
        g = _compute_smoothed_gradient(y, c, T, top_k=10)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            step = min(step * 1.3, 2.0 / (g_norm + 1e-10))
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining Adam log-space, FISTA, active-set CD,
    multi-n exploration, and structured restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Multi-n Adam (0-500s) ===
    n_values = [4096, 8192, 16384]
    n_values = [n for n in n_values if n != len(best_a) or True]

    phase1_end = min(deadline, t_start + 500.0)
    time_per_n = (phase1_end - t_start) / len(n_values)

    for n_try in n_values:
        if time.monotonic() > phase1_end - 60:
            break
        if n_try == len(best_a):
            a_cur = best_a.copy()
        else:
            a_cur = _interp_to_length(best_a, n_try)

        c_cur = _fft_convolve(a_cur, a_cur)
        max_c = float(c_cur.max())

        sub_deadline = min(phase1_end, time.monotonic() + time_per_n - 5)
        a_opt, sc = _adam_optimize(a_cur, sub_deadline,
                                    T_start=0.004 * max_c, T_end=0.00002 * max_c,
                                    lr_start=0.04, lr_end=0.002,
                                    max_iter=300000, active_k=10,
                                    decay=0.75, eval_interval=100,
                                    use_backtrack=True, plateau_frac=0.2,
                                    sa_prob=0.005)
        full_sc = _score(a_opt)
        print(f"Phase1 n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # === Phase 2: FISTA on best (500s-700s) ===
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end - 10:
        a_fista, sc_fista = _fista_simplex(best_a, phase2_end - 5, max_iter=30000)
        full_sc = _score(a_fista)
        print(f"Phase2 FISTA: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fista.copy()

    # === Phase 3: Hot coordinate descent (700s-900s) ===
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end - 10:
        a_hot, sc_hot = _coord_descent_hot(best_a, phase3_end - 5, max_rounds=500)
        full_sc = _score(a_hot)
        print(f"Phase3 HotCD: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_hot.copy()

    # === Phase 4: Fast CD polish (900s-1050s) ===
    phase4_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase4_end - 10:
        a_cd, sc_cd = _fast_cd_polish(best_a, phase4_end - 5, max_rounds=300)
        full_sc = _score(a_cd)
        print(f"Phase4 CD: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # === Phase 5: Second Adam pass on best (1050s-1300s) ===
    phase5_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase5_end - 10:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_opt2, sc2 = _adam_optimize(best_a, phase5_end - 5,
                                      T_start=0.001 * max_c, T_end=0.00001 * max_c,
                                      lr_start=0.015, lr_end=0.0005,
                                      max_iter=300000, active_k=15,
                                      decay=0.8, eval_interval=80,
                                      use_backtrack=True, plateau_frac=0.0,
                                      sa_prob=0.0)
        full_sc = _score(a_opt2)
        print(f"Phase5 Adam2: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt2.copy()

    # === Phase 6: Structured restarts (1300s-1600s) ===
    phase6_end = min(deadline, t_start + 1600.0)
    n_fin = len(best_a)
    if time.monotonic() < phase6_end - 30:
        # Structured perturbations: target the peak region
        c_best = _fft_convolve(best_a, best_a)
        peak_idx = np.argmax(c_best)
        c_best_max = float(c_best.max())

        # Determine peak region width
        peak_width = max(10, n_fin // 100)

        n_restarts = 6
        time_per_restart = (phase6_end - time.monotonic()) / n_restarts

        for attempt in range(n_restarts):
            if time.monotonic() > phase6_end - 30:
                break

            if attempt == 0:
                # Small random perturbation
                sigma = 0.01
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + sigma * rng.standard_normal(n_fin)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            elif attempt == 1:
                # Perturb peak region
                a_pert = best_a.copy()
                if peak_idx < n_fin:
                    lo = max(0, peak_idx - n_fin + 1 - peak_width // 2)
                    hi = min(n_fin - 1, peak_idx + peak_width // 2)
                else:
                    lo = max(0, peak_idx - (n_fin - 1) - peak_width // 2)
                    hi = n_fin - 1
                a_pert[lo:hi+1] *= (1.0 + 0.1 * rng.standard_normal(hi - lo + 1))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
            elif attempt == 2:
                # Perturb non-peak region
                a_pert = best_a.copy()
                if peak_idx < n_fin:
                    lo = max(0, peak_idx - n_fin + 1)
                    hi = min(n_fin - 1, peak_idx)
                else:
                    lo = max(0, peak_idx - (n_fin - 1))
                    hi = n_fin - 1
                # Perturb everything EXCEPT the peak region
                mask = np.ones(n_fin, dtype=bool)
                if lo <= hi:
                    mask[lo:hi+1] = False
                a_pert[mask] *= (1.0 + 0.15 * rng.standard_normal(mask.sum()))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
            elif attempt == 3:
                # Larger random
                sigma = 0.03
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + sigma * rng.standard_normal(n_fin)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            elif attempt == 4:
                # Interpolate with uniform
                a_pert = 0.9 * best_a + 0.1 * np.ones(n_fin) / n_fin
                a_pert /= a_pert.sum()
            else:
                # Interpolate with ramp
                ramp = np.linspace(1, 2, n_fin)
                ramp /= ramp.sum()
                a_pert = 0.85 * best_a + 0.15 * ramp
                a_pert /= a_pert.sum()

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(phase6_end - 10, time.monotonic() + max(15, time_per_restart - 5))
            a_pert_opt, sc_pert = _adam_optimize(a_pert, sub_dl,
                                                   T_start=0.002 * max_cp,
                                                   T_end=0.00001 * max_cp,
                                                   lr_start=0.02, lr_end=0.001,
                                                   max_iter=100000, active_k=10,
                                                   decay=0.75, eval_interval=80,
                                                   use_backtrack=True, plateau_frac=0.0,
                                                   sa_prob=0.0)
            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{attempt}]: score={full_sc_pert:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_pert) and full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

    # === Phase 7: Final hot CD + ultra-fine Adam (1600s-deadline) ===
    phase7_end = deadline - 5
    if time.monotonic() < phase7_end - 30:
        a_hot2, _ = _coord_descent_hot(best_a, min(phase7_end, time.monotonic() + 120), max_rounds=300)
        full_sc = _score(a_hot2)
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_hot2.copy()

    if time.monotonic() < phase7_end - 10:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _adam_optimize(best_a, phase7_end - 3,
                                     T_start=0.0002 * max_cf, T_end=0.000005 * max_cf,
                                     lr_start=0.005, lr_end=0.0001,
                                     max_iter=200000, active_k=15,
                                     decay=0.85, eval_interval=50,
                                     use_backtrack=True, plateau_frac=0.0,
                                     sa_prob=0.0)
        full_sc = _score(a_final)
        print(f"Phase7 final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

    # Final quick CD
    if time.monotonic() < phase7_end - 2:
        a_cd_fin, _ = _fast_cd_polish(best_a, phase7_end - 1, max_rounds=30)
        sc_cd_fin = _score(a_cd_fin)
        if sc_cd_fin < best_score:
            best_score = sc_cd_fin
            best_a = a_cd_fin.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
