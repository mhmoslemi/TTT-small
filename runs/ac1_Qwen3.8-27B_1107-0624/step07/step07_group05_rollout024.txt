Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key new ideas:
1. Gerchberg-Saxton style spectral iteration (alternating time/freq domain)
2. Refined pressure equalization with adaptive active set
3. Continuation L-BFGS on log-sum-exp smoothed objective
4. Multi-resolution warm start strategy
5. Better diverse restarts using structured perturbations
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score for sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad of 2n*(cmax + beta*log(sum exp((c-cmax)/beta))) wrt z
    # = 2n * sum_k w_k * dc_k/dz
    # dc_k/dz_m = 2*a_m*(a_{k-m} - c_k)  [derived earlier]
    # So grad_m = 2n * sum_k w_k * 2*a_m*(a_{k-m} - c_k)
    # = 4n * a_m * (sum_k w_k * a_{k-m} - sum_k w_k * c_k)
    # = 4n * a_m * (h_m - W)
    # where h_m = sum_k w_k * a_{k-m}, W = sum_k w_k * c_k
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _spectral_iterate(a, deadline, n_iters=500, alpha_start=0.05, alpha_end=1.0):
    """
    Gerchberg-Saxton style iteration to flatten the power spectrum.
    Alternates between:
    - Time domain: enforce non-negativity and normalization
    - Frequency domain: push |A| towards flat
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        # Annealing: alpha goes from 0 (no freq modification) to 1 (full flat spectrum)
        t = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t * t  # quadratic ramp
        
        # Compute spectrum
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mag_sq = mag ** 2
        avg_mag_sq = float(np.sum(mag_sq) / N)  # target average
        
        # Update magnitude: |A_new| = |A|^(1-alpha) * avg_mag^(alpha)
        # This pushes towards flat spectrum
        with np.errstate(divide='ignore', invalid='ignore'):
            target_mag = np.where(mag > 1e-30, np.sqrt(avg_mag_sq), np.sqrt(avg_mag_sq))
            new_mag = mag ** (1 - alpha) * target_mag ** alpha
        
        # Preserve phase
        phase = np.where(mag > 1e-30, A / mag, 1.0 + 0j)
        A_new = new_mag * phase
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        if it % 10 == 0:
            sc = _score_norm(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        a = a_new
    
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3):
    """
    Improved pressure equalization.
    At KKT optimum, p_j = sum_{k in S} a_{k-j} should be constant on support.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_conv(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        # Adaptive active set: use top fraction
        # Start with tight tolerance, expand if needed
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        
        # If too few active points, expand
        if n_active < max(3, n // 100):
            k = max(3, n // 100)
            top_k = np.argsort(c)[-k:]
            active = np.zeros_like(c, dtype=bool)
            active[top_k] = True
            n_active = k
        
        # Compute pressure: p_j = sum_{k in active} a_{k-j}
        # This is cross-correlation of 'active' with 'a'
        active_f = active.astype(float)
        a_rev = a[::-1]
        p = _fft_conv(active_f, a_rev)[n-1:2*n-1]
        p = np.maximum(p, 1e-30)
        
        # Update: push towards uniform pressure
        p_target = np.mean(p)
        ratio = p_target / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _lbfgs_continue(a, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    
    # Estimate good starting beta
    c_sorted = np.sort(c)[::-1]
    if len(c_sorted) > 5:
        gap = c_sorted[0] - c_sorted[min(5, len(c_sorted)-1)]
    else:
        gap = c_max * 0.1
    base_beta = max(gap, c_max * 0.05)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                   0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1.0, n / 1024.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 10))))
        
        try:
            result = minimize(
                _smoobj, z, args=(beta,),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _log_subgrad(a_init, deadline, n_steps=3000, lr=0.01):
    """Subgradient method in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _fft_conv(a, a)
        c_max = float(c.max())
        
        # Use top-k active set for gradient
        k = max(5, n // 20)
        top_idx = np.argsort(c)[-k:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / k
        
        a_rev = a[::-1]
        wc = _fft_conv(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_init(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2 + 0.01
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t**2 * 2) + 0.05
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * t) + 0.15 * np.cos(2 * np.pi * 19 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(n)
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _full_pipeline(a_init, deadline, min_remaining=60.0):
    """Run all optimization stages on a sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    # Stage 1: Spectral iteration (new!)
    if time.monotonic() < deadline - min_remaining - 60:
        t_sub = min(deadline - min_remaining - 30, time.monotonic() + 90)
        a_sp, sc_sp = _spectral_iterate(current_a, t_sub, n_iters=300)
        sc_sp_full = _score(a_sp)
        if sc_sp_full < current_sc:
            current_a = a_sp
            current_sc = sc_sp_full
    
    # Stage 2: L-BFGS
    if time.monotonic() < deadline - min_remaining - 30:
        a_lb = _lbfgs_continue(current_a, deadline, min_time_remaining=min_remaining + 30)
        sc_lb = _score(a_lb)
        if sc_lb < current_sc:
            current_a = a_lb
            current_sc = sc_lb
    
    # Stage 3: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        t_sub = min(deadline - min_remaining - 10, time.monotonic() + 60)
        a_pe, sc_pe = _pressure_equalize_v2(current_a, t_sub, n_iters=200)
        sc_pe_full = _score(a_pe)
        if sc_pe_full < current_sc:
            current_a = a_pe
            current_sc = sc_pe_full
    
    # Stage 4: L-BFGS again
    if time.monotonic() < deadline - min_remaining - 20:
        a_lb2 = _lbfgs_continue(current_a, deadline, min_time_remaining=min_remaining + 20)
        sc_lb2 = _score(a_lb2)
        if sc_lb2 < current_sc:
            current_a = a_lb2
            current_sc = sc_lb2
    
    # Stage 5: Spectral iteration again
    if time.monotonic() < deadline - min_remaining - 10:
        t_sub = min(deadline - min_remaining - 5, time.monotonic() + 60)
        a_sp2, sc_sp2 = _spectral_iterate(current_a, t_sub, n_iters=200)
        sc_sp2_full = _score(a_sp2)
        if sc_sp2_full < current_sc:
            current_a = a_sp2
            current_sc = sc_sp2_full
    
    # Stage 6: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        a_sg, sc_sg = _log_subgrad(current_a, min(deadline - min_remaining, time.monotonic() + 30))
        if sc_sg < current_sc:
            current_a = a_sg
            current_sc = sc_sg
    
    # Stage 7: Final L-BFGS
    if time.monotonic() < deadline - min_remaining:
        a_fin = _lbfgs_continue(current_a, deadline, min_time_remaining=min_remaining)
        sc_fin = _score(a_fin)
        if sc_fin < current_sc:
            current_a = a_fin
            current_sc = sc_fin
    
    return current_a, current_sc

"""

New approaches:
1. Gerchberg-Saxton style spectral iteration for flat auto-correlation
2. Refined adaptive pressure equalization with cycling active sets

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _spectral_iterate(a, deadline, n_iters=400, alpha_max=1.0):
    """
    GS-style iteration: alternate between flattening auto-correlation in
    time domain and enforcing non-negativity.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        t = it / max(1, n_iters - 1)
        alpha = 0.01 + (alpha_max - 0.01) * t ** 1.5

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        c_full = np.fft.irfft(A * A, N)[:2 * n - 1]

        c_flat = float(np.mean(c_full))
        if c_flat < 1e-30:
            break

        with np.errstate(divide='ignore', invalid='ignore'):
            c_modified = c_full * np.power(np.maximum(c_flat / c_full, 1e-20), alpha)

        target_spec = np.fft.rfft(np.concatenate([c_modified,
                                                   np.zeros(N - len(c_modified))]))
        target_spec = np.maximum(np.real(target_spec), 0)
        target_mag = np.sqrt(target_spec)

        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = np.where(mag > 1e-30, target_mag / mag, 1.0)
            ratio = np.clip(ratio, 0.01, 100.0)

        A_new = A * ratio
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        if it % 15 == 0:
            sc = _score_norm(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

        a = a_new

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _pressure_equalize(a, deadline, n_iters=300, gamma=0.3):
    """
    KKT-motivated multiplicative update to equalize pressure.
    Adaptive active set with cycling tolerance.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    tol_schedule = [0.005, 0.01, 0.02, 0.01, 0.005, 0.002, 0.001]

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = tol_schedule[it % len(tol_schedule)]
        active = c >= (c_max - tol * c_max)
        n_active = active.sum()

        if n_active < 3:
            k = 5
            top_k = np.argsort(c)[-k:]
            active = np.zeros_like(c, dtype=bool)
            active[top_k] = True
            n_active = k
        elif n_active > n // 2:
            k = max(10, n // 20)
            top_k = np.argsort(c)[-k:]
            active = np.zeros_like(c, dtype=bool)
            active[top_k] = True
            n_active = k

        active_f = active.astype(float)
        a_rev = a[::-1]
        p = _fft_conv(active_f, a_rev)[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_target = np.mean(p[p < np.percentile(p, 95)])
        ratio = p_target / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 25 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _lbfgs_run(a, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta continuation."""
    n = len(a)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a

    c_sorted = np.sort(c)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    else:
        gap = c_max * 0.05
    base_beta = max(gap, c_max * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                   0.015, 0.007, 0.003, 0.001, 0.0004, 0.0001])

    current_a = a.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _smoobj, z, args=(beta,),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.015):
    """Subgradient descent in log-space with adaptive step."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _fft_conv(a, a)
        c_max = float(c.max())

        k = max(5, n // 30)
        top_idx = np.argsort(c)[-k:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / k

        a_rev = a[::-1]
        wc = _fft_conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_init(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2 + 0.01
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t ** 2 * 3) + 0.05
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 11 * t) + 0.1 * np.cos(2 * np.pi * 23 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + (t * 3) ** 4)
        return a / a.sum()
    else:
        return np.ones(n) / n

        n_list.append(10240)
    n_list = [x for x in n_list if x != n_parent]
    n_list.sort(reverse=True)

    for n_try in n_list:
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - 400 - time.monotonic()
        if remaining < 60:
            break

        if parent is not None and n_parent > 0:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_init(n_try, "sin_taper")
        else:
            a_init = _make_init(n_try, "sin_taper")

        min_rem = max(50, remaining * 0.5)
        a_opt, sc_opt = _pipeline(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 3: Focused refinement on best ---
    print(f"\n[Phase 3] Focused refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        # Extended spectral
        t_sub = min(deadline - 80, time.monotonic() + 100)
        a_sp, _ = _spectral_iterate(best_seq, t_sub, n_iters=400, alpha_max=1.0)
        sc_sp = _score(a_sp)
        _update(a_sp, sc_sp)

        if sc_sp < best_score:
            a_lb = _lbfgs_run(a_sp, deadline, min_time_remaining=80)
            _update(a_lb, _score(a_lb))

        # Extended pressure
        if time.monotonic() < deadline - 60:
            t_sub2 = min(deadline - 40, time.monotonic() + 80)
            a_pe, _ = _pressure_equalize(best_seq, t_sub2, n_iters=400, gamma=0.25)
            sc_pe = _score(a_pe)
            _update(a_pe, sc_pe)
            if sc_pe < best_score:
                a_lb2 = _lbfgs_run(a_pe, deadline, min_time_remaining=40)
                _update(a_lb2, _score(a_lb2))

    # --- Phase 4: Diverse restarts ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        kinds = ["gauss", "cos_mod", "plateau", "sin_taper"]

        for kind in kinds:
            if time.monotonic() > deadline - 30:
                break
            a_init = 0.6 * best_seq + 0.4 * _make_init(n_best, kind)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

            # Add small random perturbation
            a_init = a_init + 0.01 * rng.standard_normal(n_best)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

            min_rem = max(20, (deadline - time.monotonic()) * 0.05)
            a_r, sc_r = _pipeline(a_init, deadline, min_remaining=min_rem)
            _update(a_r, sc_r)

    # --- Phase 5: Final cycling ---
    print(f"\n[Phase 5] Final cycling (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        current = best_seq.copy()
        current_sc = best_score

        for cycle in range(6):
            if time.monotonic() > deadline - 10:
                break

            # L-BFGS
            if time.monotonic() < deadline - 8:
                a_new = _lbfgs_run(current, deadline, min_time_remaining=8)
                sc_new = _score(a_new)
                if sc_new < current_sc:
                    current = a_new
                    current_sc = sc_new

            # Pressure
            if time.monotonic() < deadline - 8:
                t_sub = min(deadline - 5, time.monotonic() + 20)
                a_new, _ = _pressure_equalize(current, t_sub, n_iters=100, gamma=0.2)
                sc_new = _score(a_new)
                if sc_new < current_sc:
                    current = a_new
                    current_sc = sc_new

            # Spectral
            if time.monotonic() < deadline - 5:
                t_sub = min(deadline - 3, time.monotonic() + 15)
                a_new, _ = _spectral_iterate(current, t_sub, n_iters=100, alpha_max=0.8)
                sc_new = _score(a_new)
                if sc_new < current_sc:
                    current = a_new
                    current_sc = sc_new

        _update(current, current_sc)

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_init(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_init(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
