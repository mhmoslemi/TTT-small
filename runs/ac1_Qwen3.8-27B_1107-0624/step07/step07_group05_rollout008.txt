Optimized step-function coefficient search using adaptive log-sum-exp L-BFGS,

exploration. Target: minimize 2*n*max(conv(a,a))/sum(a)^2.

import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=0.005):

    if c_max < 1e-30:

    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adaptive_lbfgs(a_init, deadline, min_remaining=30.0, n_betas=None):
    """Adaptive L-BFGS with beta continuation and annealing reheats."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    if n_betas is None:
        n_betas = 15

    # Create beta schedule: geometric decay from base_beta to base_beta * 1e-6
    betas = base_beta * np.power(1e-6, np.linspace(0, 1, n_betas))
    # Ensure last few betas are very small
    betas[-3:] = base_beta * np.array([1e-4, 1e-5, 1e-6])
    betas = np.maximum(betas, max_c * 1e-7)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    rng = np.random.default_rng(777)
    temperature = max_c * 0.5
    best_z = z.copy()
    best_sc_z = current_sc

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.008 * max(1.0, n / 4096.0)
        max_iter = min(12000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                best_z = z.copy()
                best_sc_z = sc_new
            else:
                # Simulated annealing reheat: perturb and try again
                if i < len(betas) - 3 and temperature > max_c * 1e-4:
                    z_pert = z + temperature * rng.standard_normal(n)
                    z_pert = z_pert - z_pert.mean()
                    z_pert = np.clip(z_pert, -60, 60)
                    a_pert = _softmax(z_pert)
                    sc_pert = _score(a_pert)

                    # Accept if better or with annealing probability
                    delta = sc_pert - current_sc
                    if delta < 0 or (np.random.random() < np.exp(-delta / (temperature + 1e-30)) and np.isfinite(delta)):
                        z = z_pert
                        current_a = a_pert
                        current_sc = sc_pert

                    temperature *= 0.7
        except Exception:
            pass

    # Final pass with very small beta
    if time.monotonic() + min_remaining < deadline:
        beta_final = max_c * 1e-7
        try:
            result = minimize(
                _smoobj, best_z, args=(beta_final, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200}
            )
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < current_sc:
                current_a = a_fin
                current_sc = sc_fin
        except Exception:
            pass

    return current_a


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma_init=0.4, gamma_end=0.1):
    """Improved pressure equalization with adaptive gamma and tolerance."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive tolerance: tighten as we iterate
        frac = it / max(1, n_iters)
        tol = (0.02 - 0.015 * frac) * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 2:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 5

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Adaptive gamma: start aggressive, end gentle
        gamma = gamma_init + (gamma_end - gamma_init) * frac

        # Multiplicative update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 10 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.025):
    """Subgradient descent in log-space with adaptive step size and restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc
    rng = np.random.default_rng(999)
    temperature = 0.1

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_grad(a, c, n, tol_frac=0.003)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + 0.0002 * i)

        # Add small noise for exploration (annealing)
        noise = temperature * rng.standard_normal(n) * (1.0 / (1.0 + 0.01 * i))
        z = z - step * grad + noise
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
                best_patience_sc = best_sc
            else:
                patience += 1
                lr *= 0.92
                temperature *= 0.95
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
                    temperature = 0.05

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_polish(a, deadline, n_rounds=80):
    """Coordinate descent focusing on peak-contributing coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(54321)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(20, n // 30))
        top_idx = np.argsort(contrib)[-n_try:]
        perm = rng.permutation(len(top_idx))
        top_idx = top_idx[perm[:min(len(top_idx), max(5, n_try // 4))]]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            c_old = _full_convolve(a, a).max()

            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    break
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-15:
                    improved = True
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 2.0 * t * t) + 1.8
        return a / a.sum()
    elif kind == "multi_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 4 * t) + 0.1 * np.cos(2 * np.pi * 11 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.15
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.04):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.5) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _upsample(a, new_n):
    """Upsample a sequence to new length via interpolation."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def _downsample(a, new_n):
    """Downsample a sequence to new length."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def _multi_stage(a_init, deadline, min_remaining=30.0):
    """Run the full multi-stage optimization pipeline."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: Adaptive L-BFGS
    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _adaptive_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 60)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 20:
        pe_deadline = min(deadline - min_remaining - 10, time.monotonic() + 90)
        a2, sc2_norm = _pressure_equalize_v2(current_a, pe_deadline, n_iters=200)
        sc2_full = _score(a2)
        if sc2_full < current_sc:
            current_a = a2
            current_sc = sc2_full

    # Stage 3: L-BFGS again (finer)
    if time.monotonic() < deadline - min_remaining - 10:
        a3 = _adaptive_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 10)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 50)
        a4, sc4 = _subgrad_polish(current_a, sg_deadline, n_max_steps=3000, lr_init=0.02)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: adaptive L-BFGS with annealing, pressure equalization,
    multi-n exploration, and coordinate descent polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main adaptive L-BFGS on parent ---
    print(f"\n[Phase 1] Main adaptive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        phase1_deadline = deadline - 500
        a1 = _adaptive_lbfgs(parent, phase1_deadline, min_remaining=30, n_betas=12)
        _update(a1, _score(a1))

    # --- Phase 2: Pressure equalization + L-BFGS ---
    print(f"\n[Phase 2] Pressure equalization + refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        pe_deadline = min(deadline - 300, time.monotonic() + 120)
        a_pe, _ = _pressure_equalize_v2(best_seq, pe_deadline, n_iters=250, gamma_init=0.35, gamma_end=0.08)
        _update(a_pe, _score(a_pe))

        if time.monotonic() < deadline - 200:
            a_ref = _adaptive_lbfgs(best_seq, min(deadline - 100, time.monotonic() + 120), min_remaining=10, n_betas=8)
            _update(a_ref, _score(a_ref))

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288, 16384]
    # Filter: skip if same as parent and already optimized
    n_candidates = [x for x in n_candidates if x != n_parent or len(set(n_candidates)) > 3]
    # Sort: try different sizes
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_parent))
    n_candidates = n_candidates[:4]  # Limit to 4 additional sizes

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 40:
            break

        # Initialize from parent via interpolation
        if parent is not None and n_parent != n_try:
            a_init = _upsample(parent, n_try) if n_try > n_parent else _downsample(parent, n_try)
        elif parent is not None:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(20, min(remaining_for_this * 0.4, 200))
        a_opt, sc_opt = _multi_stage(a_init, min(deadline - 100, time.monotonic() + remaining_for_this * 0.6),
                                      min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts from best ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        restart_kinds = ["chirp", "multi_cos", "gaussian_bump", "tapered"]

        for i, kind in enumerate(restart_kinds):
            if time.monotonic() > deadline - 30:
                break
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, kind)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=0.03)

            min_rem = max(15, deadline - 15 - time.monotonic())
            if min_rem < 20:
                break
            a_r, sc_r = _multi_stage(a_pert, min(deadline - 10, time.monotonic() + min(remaining_for_this, min_rem * 0.8)),
                                      min_remaining=min(remaining_for_this, 15))
            _update(a_r, sc_r)

    # --- Phase 5: Extended subgradient on best ---
    print(f"\n[Phase 5] Extended subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        sg_deadline = min(deadline - 5, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=5000, lr_init=0.015)
        _update(a_sg, sc_sg)

    # --- Phase 6: Coordinate descent ---
    print(f"\n[Phase 6] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        cd_deadline = min(deadline - 3, time.monotonic() + 90)
        a_cd, _ = _coordinate_polish(best_seq, cd_deadline, n_rounds=60)
        _update(a_cd, _score(a_cd))

    # --- Phase 7: Final L-BFGS polish ---
    print(f"\n[Phase 7] Final L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin = _adaptive_lbfgs(best_seq, deadline - 1, min_time_remaining=1, n_betas=6)
        _update(a_fin, _score(a_fin))

        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 1, n_max_steps=500, lr_init=0.008)
            _update(a_fin2, sc_fin2)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
