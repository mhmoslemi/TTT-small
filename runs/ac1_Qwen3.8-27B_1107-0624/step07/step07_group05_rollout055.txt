import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

"""
Multi-phase optimizer combining aggressive L-BFGS with log-sum-exp smoothing,
smooth pressure equalization, multi-n exploration, and coordinate polishing.
Focuses on tight convergence to squeeze out the last 0.2% improvement.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

            if it % 30 == 0:
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent in log-space with adaptive step size and restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_smooth(a, c, n, temp=0.01)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.01, lr_init)
                best_patience_sc = best_sc
            else:
                patience += 1
                lr *= 0.88
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_polish(a, deadline, n_rounds=60):
    """
    Coordinate descent: adjust individual coordinates to reduce max convolution.
    Focuses on coordinates that contribute most to the peak.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(98765)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find coordinates contributing most to the active peaks
        temp = 0.02 * c_max
        active = np.exp(np.clip((c - c_max) / temp, -30, 0))
        s_act = active.sum()
        if s_act > 1e-10:
            active = active / s_act
        else:
            active = np.zeros_like(c)
            active[int(np.argmax(c))] = 1.0

        a_rev = a[::-1]
        contrib = _full_convolve(active, a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        # Pick top coordinates
        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = rng.permutation(top_idx[:min(n_try, len(top_idx))])

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Try multiplicative adjustments
            best_factor = 1.0
            best_c = _full_convolve(a, a).max()

            for factor in [0.85, 0.9, 0.95, 1.02, 1.05, 1.1]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < best_c - 1e-16:
                    best_c = c_try
                    best_factor = factor
                a[i] = old_val

            if best_factor != 1.0:
                a[i] = old_val * best_factor
                improved = True

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.03):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1 and freq > 0:
                z_hat[len(z_hat) - 1 - freq + 1] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tilt":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * (t - 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tilt2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * (t - 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "wave":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "double_wave":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t + 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Load parent, optimize aggressively
    2. Smooth pressure equalization
    3. L-BFGS with tighter convergence
    4. Multi-n exploration (larger n)
    5. Diverse restarts
    6. Coordinate polish
    7. Final subgradient + L-BFGS
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p):
        nonlocal best_seq, best_score
        if p is None or len(p) == 0:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s", flush=True)

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent: n={n_parent}, score={_score(parent):.10f}", flush=True)
                _update(parent)
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tilt")
        print(f"No parent, using tilt n={n_parent}", flush=True)
        _update(parent)

    # --- Phase 1: Aggressive L-BFGS on parent ---
    print(f"\n[Phase 1] Aggressive L-BFGS on parent (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 800:
        a1 = _run_lbfgs_aggressive(parent, deadline, min_time_remaining=800)
        _update(a1)

    # --- Phase 2: Smooth pressure equalization ---
    print(f"\n[Phase 2] Pressure equalization (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 600:
        pe_deadline = min(deadline - 500, time.monotonic() + 120)
        a2, _ = _smooth_pressure_equalize(best_seq, pe_deadline, n_iters=300)
        _update(a2)

    # --- Phase 3: L-BFGS after pressure ---
    print(f"\n[Phase 3] L-BFGS after pressure (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3 = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=400)
        _update(a3)

    # --- Phase 4: Subgradient polish ---
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 200:
        sg_deadline = min(deadline - 150, time.monotonic() + 60)
        a4, _ = _subgrad_polish(best_seq, sg_deadline, n_max_steps=2000, lr_init=0.015)
        _update(a4)

    # --- Phase 5: L-BFGS again (finer) ---
    print(f"\n[Phase 5] Final L-BFGS round (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 150:
        a5 = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=100)
        _update(a5)

    # --- Phase 6: Multi-n exploration (larger n) ---
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_candidates = [12288, 16384]
    # Also try slightly different sizes
    if n_parent not in [12288, 16384]:
        n_candidates.insert(0, n_parent + n_parent // 4)

    for n_try in sorted(set(n_candidates), reverse=True):
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 20:
            break

        # Interpolate best to new size
        x_old = np.linspace(0, 1, len(best_seq) if best_seq is not None else n_parent)
        x_new = np.linspace(0, 1, n_try)
        src = best_seq if best_seq is not None else parent
        a_init = np.interp(x_new, x_old, src)
        a_init = np.maximum(a_init, 0)
        s_init = a_init.sum()
        if s_init > 1e-10:
            a_init = a_init / s_init
        else:
            a_init = _make_profile(n_try, "tilt")

        print(f"  Trying n={n_try}", flush=True)
        a_opt = _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=60)
        _update(a_opt)

    # --- Phase 7: Diverse restarts ---
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        restart_profiles = ["tilt2", "double_wave", "wave"]
        for prof in restart_profiles:
            if time.monotonic() > deadline - 15:
                break
            a_prof = _make_profile(n, prof)
            # Blend 60% best, 40% profile
            a_blend = 0.6 * best_seq + 0.4 * a_prof
            a_blend = np.maximum(a_blend, 0)
            s_b = a_blend.sum()
            if s_b < 1e-10:
                continue
            a_blend = a_blend / s_b
            # Small Fourier perturbation
            a_pert = _fourier_perturb(a_blend, rng, n_modes=4, amplitude=0.015)
            a_opt = _run_lbfgs_aggressive(a_pert, deadline, min_time_remaining=15)
            _update(a_opt)

    # --- Phase 8: Coordinate descent polish ---
    print(f"\n[Phase 8] Coordinate polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 10:
        cd_deadline = min(deadline - 5, time.monotonic() + 90)
        a_cd, _ = _coordinate_polish(best_seq, cd_deadline, n_rounds=40)
        _update(a_cd)

    # --- Phase 9: Final L-BFGS + subgradient ---
    print(f"\n[Phase 9] Final polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=2)
        _update(a_fin)
        if time.monotonic() < deadline - 1:
            a_fin2, _ = _subgrad_polish(a_fin, deadline - 1, n_max_steps=500, lr_init=0.005)
            _update(a_fin2)

    # --- Final validation ---
    print(f"\n{'='*60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        # Also validate with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal={sc_final:.10f}, evaluate={eval_score:.10f}, n={len(final)}, "
                  f"time={time.monotonic()-t_start:.1f}s", flush=True)
        except Exception as e:
            print(f"FINAL: internal={sc_final:.10f}, n={len(final)}, "
                  f"time={time.monotonic()-t_start:.1f}s, eval error: {e}", flush=True)
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
