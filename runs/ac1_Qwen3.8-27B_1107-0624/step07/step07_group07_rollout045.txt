import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        est_per_iter = 0.003 * (n / 4096.0) ** 1.5
        max_iter = min(15000, max(300, int(remaining / (est_per_iter * 25))))

        a_new = _lbfgs_fine(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            if i % 5 == 0:
                print(f"    beta={beta:.2e}: score={sc:.10f}")

    return current_a


def _exact_subgrad_peak_shave(a_init, deadline, max_iters=80000):
    """Peak shaving using exact subgradient of max(conv).
    At each step, compute the subgradient at the active set (peaks near max),
    then take a projected gradient step in the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.05
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set: all positions within 1% of max
        tol = 0.005 * c_max + 1e-12
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Subgradient of max(c) w.r.t. a:
        # d(c_k)/d(a_j) = a_{k-j} + a_{k-j} if 2j != k, else a_j
        # = 2*a_{k-j} (for most cases, accounting for symmetry)
        # For the full convolution c = a*a:
        #   c_k = sum_i a_i * a_{k-i}
        #   d(c_k)/d(a_j) = a_{k-j} (from i=j term) + a_{k-j} (from k-i=j term)
        #                    if both valid and j != k-j
        #                 = a_{k-j} (from i=j = k-j term, i.e. a_j^2)
        #                     if j = k-j

        # More carefully: for numpy convolve(a,a)[k] = sum_i a[i]*a[k-i]
        # d/d(a[j]) = a[k-j] if 0<=k-j<n (from i=j) + a[k-j] if 0<=k-j<n and k-j!=j (from i=k-j)
        # = 2*a[k-j] if 0<=k-j<n and 2*j!=k
        # = a[k-j] if 0<=k-j<n and 2*j==k (i.e., a[j])

        # For the subgradient of max: grad_j = (1/n_active) * sum_{k in active} d(c_k)/d(a_j)
        # This can be computed as: grad = (1/n_active) * convolve(w_active, a_rev)[n-1:2n-1]
        # where w_active is the indicator of the active set (with appropriate normalization)

        w = np.zeros(len(c))
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)

        # Remove mean component (for invariance to scaling)
        grad = grad - grad.mean()
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        grad_normed = grad / g_norm

        # Line search
        best_sc_try = best_sc
        best_a_try = best_a
        best_eta = eta

        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * frac
            a_try = a - step * grad_normed
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_eta = step

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.15)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-9:
                break
            if stall > 150:
                break

    return best_a, best_sc


def _multi_peak_shave(a_init, deadline, max_iters=50000):
    """Peak shaving that targets the top-K peaks simultaneously."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.03
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Use top 5% of peaks (at least 5)
        sorted_c = np.sort(c)[::-1]
        n_top = max(5, int(0.03 * len(c)))
        threshold = sorted_c[min(n_top - 1, len(c) - 1)]
        active = c >= threshold
        n_active = active.sum()

        w = np.zeros(len(c))
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)
        grad = grad - grad.mean()
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        grad_normed = grad / g_norm

        best_sc_try = best_sc
        best_a_try = best_a

        for frac in [1.0, 0.5, 0.25, 0.1]:
            step = eta * frac
            a_try = a - step * grad_normed
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.01, 0.12)
            stall = 0
        else:
            eta *= 0.93
            stall += 1
            if eta < 1e-9 or stall > 200:
                break

    return best_a, best_sc


def _coordinate_descent_aggressive(a_init, deadline, max_rounds=200000):
    """Coordinate descent with 2- and 3-index updates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    improved_any = 0
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick 2-3 random indices
        k = 2 if rnd % 3 == 0 else 3
        idxs = rng.integers(0, n, size=k)
        s_k = a[idxs].sum()

        # Try a few random redistributions
        improved = False
        for trial in range(5):
            if trial == 0:
                fracs = np.ones(k) / k
            else:
                raw = rng.dirichlet(np.ones(k) * 0.5, size=1).ravel()
                fracs = raw
            new_vals = s_k * fracs
            a_try = a.copy()
            for ii, val in zip(idxs, new_vals):
                a_try[ii] = val
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            improved_any += 1
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000 and rnd > 1000:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _spectral_restart(a, rng, n):
    """Create a perturbed version using spectral domain."""
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    # Add random phase perturbation (preserving |A|)
    amp = np.abs(A)
    phase = np.angle(A)
    n_modes = max(5, n // 200)
    for _ in range(n_modes):
        freq = rng.integers(1, len(amp))
        delta = rng.uniform(-0.02, 0.02)
        phase[freq] += delta
        if freq < len(amp) - 1:
            phase[N - freq] = -phase[freq]  # maintain real constraint
    A_new = amp * np.exp(1j * phase)
    # Ensure real by symmetry
    A_new_full = np.fft.fft(A_new)
    A_new_full[N//2+1:] = np.conj(A_new_full[N//2-1:0:-1])
    a_new = np.fft.irfft(A_new_full * 2 + A_new_full[0], N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _gaussian_perturb(a, rng, scale=0.03):
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean() + scale * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _block_perturb(a, rng, n_blocks=10):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(4, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)
        z[start:end] += rng.uniform(-0.01, 0.01)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with aggressive L-BFGS, exact subgradient peak shaving,
    spectral restarts, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            if prev - sc > 1e-6:
                print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s (delta={prev-sc:.2e})")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        parent = np.ones(8192) / 8192
        print(f"No parent. Using uniform n=8192, score={_score(parent):.10f}")
        try_update(parent, "init")

    # === Phase 1: Aggressive L-BFGS on parent length ===
    print(f"\n=== Phase 1: Aggressive L-BFGS (n={len(best_seq)}) ===")
    t_ph = time.monotonic()
    a_p1 = _lbfgs_aggressive_schedule(best_seq, deadline - 400, min_remaining=300.0)
    try_update(a_p1, "lbfgs_p1")
    print(f"  Phase 1 done: {time.monotonic()-t_ph:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: Exact subgradient peak shaving ===
    print(f"\n=== Phase 2: Exact subgradient peak shaving ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_ps, _ = _exact_subgrad_peak_shave(best_seq, deadline - 300, max_iters=100000)
        try_update(a_ps, "peak_shave_exact")
    print(f"  Phase 2 done: {time.monotonic()-t_ph:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 3: Multi-peak shaving ===
    print(f"\n=== Phase 3: Multi-peak shaving ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 250:
        a_mp, _ = _multi_peak_shave(best_seq, deadline - 250, max_iters=50000)
        try_update(a_mp, "multi_peak")
    print(f"  Phase 3 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 4: L-BFGS again on improved sequence ===
    print(f"\n=== Phase 4: Repeat L-BFGS on improved ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 250:
        a_p4 = _lbfgs_aggressive_schedule(best_seq, deadline - 250, min_remaining=200.0)
        try_update(a_p4, "lbfgs_p4")
    print(f"  Phase 4 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 5: Try larger n (16384) ===
    print(f"\n=== Phase 5: Larger n exploration ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 150:
        target_n = 16384
        if len(best_seq) < target_n:
            print(f"  Interpolating to n={target_n}...")
            a_large = _interpolate_to_n(best_seq, target_n)
            sc_large = _score(a_large)
            print(f"    Interpolated score: {sc_large:.10f}")
            try_update(a_large, f"interp_{target_n}")

            # Optimize at larger n
            if time.monotonic() < deadline - 120:
                a_large_opt = _lbfgs_aggressive_schedule(a_large, deadline - 120, min_remaining=80.0)
                try_update(a_large_opt, f"lbfgs_{target_n}")

                if time.monotonic() < deadline - 60:
                    a_large_ps, _ = _exact_subgrad_peak_shave(a_large_opt, deadline - 60, max_iters=30000)
                    try_update(a_large_ps, f"ps_{target_n}")

            print(f"  Best at n={target_n}: {_score(best_seq):.10f}")
        else:
            # Try smaller n
            for smaller_n in [16384, 12288]:
                if len(best_seq) > smaller_n and time.monotonic() < deadline - 100:
                    a_small = _interpolate_to_n(best_seq, smaller_n)
                    try_update(a_small, f"interp_{smaller_n}")
                    if time.monotonic() < deadline - 60:
                        a_small_opt = _lbfgs_aggressive_schedule(a_small, deadline - 60, min_remaining=30.0)
                        try_update(a_small_opt, f"lbfgs_{smaller_n}")
    print(f"  Phase 5 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Aggressive coordinate descent ===")
    t_ph = time.monotonic()
    if time.monotonic() < deadline - 80:
        a_cd, _ = _coordinate_descent_aggressive(best_seq, deadline - 80, max_rounds=100000)
        try_update(a_cd, "coord_descent")
    print(f"  Phase 6 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart with polish ===")
    t_ph = time.monotonic()
    restart_configs = [
        ("gauss_01", "gauss", 0.01),
        ("gauss_02", "gauss", 0.02),
        ("gauss_05", "gauss", 0.05),
        ("gauss_10", "gauss", 0.10),
        ("spectral", "spectral", 0.0),
        ("block", "block", 0.0),
    ]
    for name, mode, scale in restart_configs:
        if time.monotonic() > deadline - 30:
            break
        if mode == "gauss":
            p_start = _gaussian_perturb(best_seq, rng, scale)
        elif mode == "spectral":
            p_start = _spectral_restart(best_seq, rng, len(best_seq))
        elif mode == "block":
            p_start = _block_perturb(best_seq, rng)
        else:
            p_start = best_seq.copy()

        sc_start = _score(p_start)
        if time.monotonic() < deadline - 25:
            a_r = _lbfgs_aggressive_schedule(p_start, deadline - 25, min_remaining=10.0)
            try_update(a_r, f"restart_{name}")

            if time.monotonic() < deadline - 15:
                a_r2, _ = _exact_subgrad_peak_shave(a_r, deadline - 15, max_iters=10000)
                try_update(a_r2, f"restart_{name}_ps")

    print(f"  Phase 7 done: {time.monotonic()-t_ph:.1f}s")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final polish ===")
    t_ph = time.monotonic()
    for cycle in range(4):
        if time.monotonic() > deadline - 3:
            break

        # Peak shave
        if time.monotonic() < deadline - 3:
            a_f1, _ = _exact_subgrad_peak_shave(best_seq, deadline - 3, max_iters=20000)
            try_update(a_f1, f"final_ps_{cycle}")

        # L-BFGS with tiny beta
        if time.monotonic() < deadline - 3 and best_seq is not None:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta_tiny = mb * 1e-9
            remaining = max(10, deadline - 3 - time.monotonic())
            max_iter = min(10000, max(500, int(remaining / 0.005)))
            a_f2 = _lbfgs_fine(best_seq, beta_tiny, n, max_iter)
            try_update(a_f2, f"final_lbfgs_{cycle}")

        # Coordinate descent
        if time.monotonic() < deadline - 3:
            a_f3, _ = _coordinate_descent_aggressive(best_seq, deadline - 3, max_rounds=30000)
            try_update(a_f3, f"final_cd_{cycle}")

    print(f"  Phase 8 done: {time.monotonic()-t_ph:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(8192) / 8192
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
