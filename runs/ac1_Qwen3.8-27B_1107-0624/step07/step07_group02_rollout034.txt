Improved optimizer using adaptive smoothing, active-set Newton, and multi-scale search.
Key differences from previous: (1) curriculum-based beta schedule with convergence monitoring,
(2) active-set with exact line search and adaptive step, (3) multi-scale coarse-to-fine,

import numpy as np
import time
from scipy.optimize import minimize, line_search

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: 2n * convolve(w, a_reversed)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta schedule and convergence monitoring."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Determine initial beta from the spread of top values
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adaptive beta schedule
    beta = base_beta
    min_beta = max_c * 0.00001
    beta_ratio = 0.5
    
    while beta > min_beta and time.monotonic() + min_time_remaining < deadline:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations needed
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 15))))
        
        # Define objective and gradient
        obj = lambda zz: _smoobj(zz, beta, n)
        grad = lambda zz: _smograd(zz, beta, n)
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    beta={beta:.2e}, score={current_sc:.10f}, iters={result.nit}")
            else:
                print(f"    beta={beta:.2e}, no improvement (sc_new={sc_new:.10f})")
        except Exception as e:
            print(f"    beta={beta:.2e}, error: {e}")
        
        # Adaptive beta reduction
        # If we used all iterations, reduce beta slowly; otherwise reduce faster
        if result.nit >= max_iter * 0.9:
            beta *= beta_ratio
        else:
            beta *= beta_ratio ** 2  # Converged, reduce faster
    
    return current_a

"""
Improved optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key improvements: (1) adaptive beta curriculum with convergence monitoring,
(2) Nesterov momentum active-set refinement, (3) warm restart perturbations,
(4) more aggressive multi-n exploration, (5) log-space L-BFGS with tighter tolerances.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _SmoProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1) > 0
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(cond))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    min_beta = max_c * 0.000005
    stall_count = 0
    max_stall = 3

    while beta > min_beta and time.monotonic() + min_time_remaining < deadline:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.0025 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 18))))

        prob = _SmoProb(beta, n)
        prev_sc = current_sc
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                stall_count = 0
            else:
                stall_count += 1
        except Exception:
            stall_count += 1

        # Adaptive beta reduction
        if stall_count >= max_stall:
            # Warm restart: perturb and try again at same beta
            z = z + 0.05 * np.random.standard_normal(n)
            z = z - z.mean()
            stall_count = 0
            beta *= 0.7
        else:
            # Check if converged (lower bound on beta)
            if beta < max_c * 0.0001:
                # Use smaller beta with more iterations
                beta *= 0.3
            else:
                beta *= 0.45

    return current_a


def _nesterov_active_set(a_init, deadline, tol_frac_start=0.005, tol_frac_end=0.00001,
                         lr_init=0.03, momentum_init=0.9):
    n = len(a_init)
    p = a_init.copy()
    s = p.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    p = p / s
    best_p = p.copy()
    best_sc = 2.0 * n * float(_full_convolve(p, p).max())

    v = np.zeros(n)  # momentum
    lr = lr_init
    mu = momentum_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300

    while time.monotonic() < deadline:
        c = _full_convolve(p, p)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_p = p.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = p[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * (h - W)

        # Center the gradient (for simplex constraint)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Nesterov momentum update
        v = momentum_init * v - lr * grad_t
        p_new = p + v
        p_new = _project_simplex(p_new)

        # Check if we improved; if not, backtrack
        c_new = _full_convolve(p_new, p_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new > current_sc + tol_frac * c_max * 0.1:
            # Rejected step: reduce momentum and lr
            v *= 0.5
            lr *= 0.7
            no_improve += 1
        else:
            p = p_new
            # Tune lr and momentum
            if sc_new < current_sc - 1e-12:
                lr = min(0.1, lr * 1.03)
                momentum_init = min(0.99, momentum_init + 0.001)
            else:
                lr *= 0.95
            if no_improve > 0:
                no_improve = max(0, no_improve - 1)

        if no_improve % 50 == 0 and no_improve > 0:
            tol_frac = max(tol_frac_end, tol_frac * 0.7)
            lr *= 0.6
        if no_improve >= max_stall:
            p = best_p.copy()
            v = np.zeros(n)
            lr = lr_init * 0.3
            tol_frac = max(tol_frac_end, tol_frac * 0.5)
            no_improve = 0

        if tol_frac <= tol_frac_end and gnorm < 1e-12 * c_max:
            break

    return best_p, _score(best_p)


def _subgradient_pass(a_init, deadline, n_steps=5000):
    """Simple subgradient method with diminishing steps and averaging."""
    n = len(a_init)
    p = a_init.copy()
    s = p.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    p = p / s
    best_p = p.copy()
    best_sc = 2.0 * n * float(_full_convolve(p, p).max())
    p_avg = p.astype(np.float64).copy()

    alpha = 1.0 / (2.0 * n * 0.01)

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(p, p)
        c_max = float(c.max())

        # Top-k active set
        k = min(20, len(c))
        top_k = np.argsort(c)[-k:]
        w = np.zeros_like(c)
        w[top_k] = 1.0 / k

        a_rev = p[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * (h - W)
        grad_t = grad - np.mean(grad)

        step = alpha / np.sqrt(t + 1)
        p = p - step * grad_t
        p = _project_simplex(p)

        # Running average
        p_avg += (p - p_avg) / (t + 1)

        if (t + 1) % 100 == 0:
            sc_avg = _score(p_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_p = p_avg.copy()

    sc_avg = _score(p_avg)
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_p = p_avg.copy()

    return best_p, best_sc


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (2 * t - 1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")
                if time.monotonic() < deadline - 60:
                    a_shr3, sc_shr3 = _nesterov_active_set(best_seq, min(deadline - 50, time.monotonic() + 80),
                                                           tol_frac_start=0.0005, tol_frac_end=0.00001,
                                                           lr_init=0.008, momentum_init=0.95)
                    if sc_shr3 < best_score:
                        best_seq = a_shr3.copy()
                        best_score = sc_shr3
                        print(f"  NEW BEST: {best_score:.10f} [nesterov_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288]
    if n_parent not in n_candidates:
        n_candidates.sort()
        insert_pos = 0
        for i, nc in enumerate(n_candidates):
            if nc > n_parent:
                insert_pos = i
                break
        else:
            insert_pos = len(n_candidates)
        n_candidates.insert(insert_pos, n_parent)

    for n_try in sorted(set(n_candidates)):
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 40:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _nesterov_active_set(a_opt, min(deadline - 25, time.monotonic() + 30),
                                                   tol_frac_start=0.002, tol_frac_end=0.00005,
                                                   lr_init=0.008)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("cos", _make_profile(n_best, "raised_cos")),
            ("peri", _make_profile(n_best, "periodic")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 20:
                break
            a_opt = _run_lbfgs_adaptive(a_r, deadline, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _nesterov_active_set(best_seq, deadline - 3,
                                             tol_frac_start=0.0005, tol_frac_end=0.000005,
                                             lr_init=0.004, momentum_init=0.97)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_nesterov] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _SmoProb(beta_fin, n)
            max_iter = min(3000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One more ultra-tight Nesterov pass
        remaining2 = deadline - 2 - time.monotonic()
        if remaining2 > 5:
            a_fin3, sc_fin3 = _nesterov_active_set(best_seq, deadline - 1,
                                                   tol_frac_start=0.0002, tol_frac_end=0.000001,
                                                   lr_init=0.002, momentum_init=0.98)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [ultra_final] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        if sc_final > best_score + 1e-6:
            # Re-evaluate to make sure
            sc_final = best_score
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
