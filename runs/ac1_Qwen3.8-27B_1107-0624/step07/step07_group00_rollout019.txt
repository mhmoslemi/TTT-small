"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient with FFT, FISTA with backtracking,
ultra-fast active-set coordinate descent, symmetry exploitation,
multi-start with diverse initializations, and multi-n exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _auto_convolve(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _auto_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _auto_convolve(a)
    return float(2.0 * n * c.max())


def _subgradient_fft(a, active_mask):
    """Compute subgradient of max_k conv_k(a) using active set.
    Returns gradient vector of length n.
    """
    n = len(a)
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(active_mask, N)
    R = np.fft.irfft(np.conj(A) * P, N)[:n]
    return R


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(np.minimum(z, 50))
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _symmetrize(a):
    return 0.5 * (a + a[::-1])


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _fista_backtrack(a_init, deadline, max_iter=200000, use_symmetry=True):
    """FISTA with backtracking line search on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    if use_symmetry:
        a = _symmetrize(a)
        a = a / a.sum()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    L = 1.0
    alpha = 0.5
    eval_interval = 200

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_y = _auto_convolve(y)
        c_max = float(c_y.max())

        # Active set: positions within 1% of max
        threshold = c_max * 0.99
        active_mask = (c_y >= threshold).astype(np.float64)
        if active_mask.sum() == 0:
            active_mask[np.argmax(c_y)] = 1.0

        # Subgradient via FFT
        grad = _subgradient_fft(y, active_mask)
        if use_symmetry:
            grad = _symmetrize(grad)

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Backtracking
        step = 1.0 / max(L, 1e-10)
        a_trial = y - step * grad
        if use_symmetry:
            a_trial = _symmetrize(a_trial)
        a_trial = _project_simplex(a_trial)
        c_trial = _auto_convolve(a_trial)
        f_trial = float(c_trial.max())
        f_y = c_max

        # Quadratic model: f(a_trial) <= f(y) + grad*(a_trial-y) + L/2 * ||a_trial-y||^2
        rhs = f_y + float(np.dot(grad, a_trial - y)) + L / 2 * float(np.dot(a_trial - y, a_trial - y))
        iter_count = 0
        while f_trial > rhs + 1e-10 and iter_count < 20:
            L *= 2.0
            step = 1.0 / L
            a_trial = y - step * grad
            if use_symmetry:
                a_trial = _symmetrize(a_trial)
            a_trial = _project_simplex(a_trial)
            c_trial = _auto_convolve(a_trial)
            f_trial = float(c_trial.max())
            rhs = f_y + float(np.dot(grad, a_trial - y)) + L / 2 * float(np.dot(a_trial - y, a_trial - y))
            iter_count += 1

        # Don't increase L too much
        L = min(L * 1.5, L * 3.0)
        if iter_count > 0:
            L *= 1.1

        a_new = a_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        if use_symmetry:
            y = _symmetrize(y)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new

        if t % eval_interval == 0:

            if sc < best_score:

                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

            a_cur = _softmax_stable(z)
            if use_symmetry:
                a_cur = _symmetrize(a_cur)
                a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    if use_symmetry:
        a_final = _symmetrize(a_final)
        a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fast_cd_active(a, deadline, active_size=20, use_symmetry=True, max_rounds=5000):
    """Ultra-fast coordinate descent using active set without full FFT per trial."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _auto_convolve(best_a)
    c_max = float(c.max())
    total_sum = float(best_a.sum())

    # Find active set
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:active_size].copy()

    # Half range for symmetry
    if use_symmetry:
        half = (n + 1) // 2
    else:
        half = n

    round_num = 0
    no_improve_rounds = 0
    scales = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]

    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        for j in range(half):
            a_j = best_a[j]
            if a_j < 1e-15:
                # Try turning on
                best_trial_score = best_score
                best_trial_val = 0.0
                for scale in [0.01, 0.05, 0.1, 0.5]:
                    trial_val = scale * (total_sum / n)
                    delta = trial_val - 0.0
                    new_sum = total_sum + delta
                    if new_sum < 1e-10:
                        continue
                    new_max = c_max
                    for k in active:
                        # Change from a[j] = 0 to a[j] = trial_val
                        ch = 0.0
                        idx1 = k - j
                        if 0 <= idx1 < n:
                            ch += trial_val * best_a[idx1]
                        idx2 = k - j  # same due to symmetry of conv
                        # Actually conv[k] = sum_i a[i]*a[k-i]
                        # Adding a[j]*a[k-j] and a[k-j]*a[j] (same if j=k-j)
                        # Since a[j] was 0, new terms are: a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] if j != k-j
                        # Or a[j]^2 if j == k-j
                        if j == k - j and 0 <= j < n:
                            ch += trial_val * trial_val
                        # For symmetry: also a[n-1-j] changes
                        if use_symmetry and n - 1 - j != j:
                            idx3 = k - (n - 1 - j)
                            if 0 <= idx3 < n:
                                ch += trial_val * best_a[idx3]
                            if n - 1 - j == k - (n - 1 - j):
                                ch += trial_val * trial_val
                        new_val_k = c[k] + ch
                        if new_val_k > new_max:
                            new_max = new_val_k
                    trial_score = 2.0 * n * new_max / (new_sum * new_sum)
                    if trial_score < best_trial_score:
                        best_trial_score = trial_score
                        best_trial_val = trial_val
                if best_trial_val > 0 and best_trial_score < best_score * (1 - 1e-10):
                    best_a[j] = best_trial_val
                    if use_symmetry:
                        best_a[n - 1 - j] = best_trial_val
                    total_sum = float(best_a.sum())
                    best_score = best_trial_score
                    improved = True
                continue

            best_trial_score = best_score
            best_trial_val = a_j
            for scale in scales:
                trial_val = a_j * scale
                if trial_val < 0:
                    continue
                delta = trial_val - a_j
                new_sum = total_sum + delta
                if use_symmetry and n - 1 - j != j:
                    new_sum += delta  # symmetry partner also changes

                if new_sum < 1e-10:
                    continue

                new_max = c_max
                for k in active:
                    ch = 0.0
                    # Change in conv[k] from changing a[j] by delta
                    idx1 = k - j
                    if 0 <= idx1 < n:
                        # a[j] changes: contributes delta * a[k-j] (from a[k-j]*a[j] term)
                        # and a[j] changes: contributes a[k-j] * delta (from a[j]*a[k-j] term)
                        # But if j == k-j, it's a[j]^2 changing to (a[j]+delta)^2
                        if j == idx1:
                            ch += 2 * delta * a_j + delta * delta
                        else:
                            ch += 2 * delta * best_a[idx1]
                    # Symmetry partner
                    if use_symmetry and n - 1 - j != j:
                        jp = n - 1 - j
                        idx2 = k - jp
                        if 0 <= idx2 < n:
                            if jp == idx2:
                                ch += 2 * delta * a_j + delta * delta
                            else:
                                ch += 2 * delta * best_a[idx2]

                    new_val_k = c[k] + ch
                    if new_val_k > new_max:
                        new_max = new_val_k

                trial_score = 2.0 * n * new_max / (new_sum * new_sum)
                if trial_score < best_trial_score:
                    best_trial_score = trial_score
                    best_trial_val = trial_val

            if best_trial_val != a_j and best_trial_score < best_score * (1 - 1e-10):
                best_a[j] = best_trial_val
                if use_symmetry:
                    best_a[n - 1 - j] = best_trial_val
                total_sum = float(best_a.sum())
                best_score = best_trial_score
                improved = True
                # Update c for affected positions (or just mark for recomputation)
                # For simplicity, recompute c periodically

        # Periodic recomputation of c and active set
        if round_num % 5 == 0 or improved:
            c = _auto_convolve(best_a)
            c_max = float(c.max())
            total_sum = float(best_a.sum())
            # Verify score
            actual_score = 2.0 * n * c_max / (total_sum * total_sum)
            if actual_score < best_score:
                best_score = actual_score
            sorted_idx = np.argsort(c)[::-1]
            active = sorted_idx[:active_size].copy()
            if not improved:
                no_improve_rounds += 1
            else:
                no_improve_rounds = 0

        if no_improve_rounds > 3:
            break

    # Final exact score
    final_score = _score_full(best_a)
    return best_a, final_score


def _exp_gradient(a_init, deadline, use_symmetry=True, max_iter=50000, eta_init=0.5):
    """Exponentiated gradient (multiplicative weights) with exact subgradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    if use_symmetry:
        a = _symmetrize(a)
        a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta_init
    avg_a = a.copy()
    count = 0

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        count += 1
        # Decay eta
        eta = eta_init / (1.0 + 0.01 * t)

        c = _auto_convolve(a)
        c_max = float(c.max())

        # Active set with tolerance
        tol = 0.005
        threshold = c_max * (1.0 - tol)
        active_mask = (c >= threshold).astype(np.float64)
        if active_mask.sum() == 0:
            active_mask[np.argmax(c)] = 1.0
        active_count = float(active_mask.sum())

        # Subgradient
        grad = _subgradient_fft(a, active_mask) / active_count
        if use_symmetry:
            grad = _symmetrize(grad)

        # Exponentiated gradient update
        a_new = a * np.exp(-eta * grad)
        if use_symmetry:
            a_new = _symmetrize(a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        avg_a = ((count - 1) / count) * avg_a + (1.0 / count) * a

        if t % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            sc_avg = _score_norm(avg_a)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()

    # Check both a and avg_a
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    sc_avg = _score_norm(avg_a)
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = avg_a.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian centered
    x = np.arange(n)
    mu = (n - 1) / 2.0
    sigma = n / 4.0
    g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    t = np.linspace(0, np.pi, n)
    rc = 0.5 * (1 + np.cos(t))
    inits.append(rc / rc.sum())

    # Bimodal (two peaks)
    mu1, mu2 = n * 0.25, n * 0.75
    sig = n * 0.1
    b = np.exp(-0.5 * ((x - mu1) / sig) ** 2) + np.exp(-0.5 * ((x - mu2) / sig) ** 2)
    inits.append(b / b.sum())

    # Power law (centered)
    d = np.abs(x - mu) + 1
    p = d ** (-1.5)
    inits.append(p / p.sum())

    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Log-uniform random
    lr = 10 ** rng.uniform(-2, 0, n)
    inits.append(lr / lr.sum())

    # Sine wave
    s = np.abs(np.sin(np.pi * x / n)) ** 2
    inits.append(s / s.sum())

    # Triangle
    tri = 1.0 - np.abs(x - mu) / (n / 2.0)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: FISTA from parent (0-80s)
    Phase 2: Exp-grad from parent (80-150s)
    Phase 3: Fast CD polish (150-250s)
    Phase 4: Multi-start at n=2048 (250-500s)
    Phase 5: Interpolate to n=4096, optimize (500-800s)
    Phase 6: Interpolate to n=8192, optimize (800-1100s)
    Phase 7: Multi-start at best n (1100-1500s)
    Phase 8: Final intensive CD (1500-1780s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_parent = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 2048
        a_parent = np.ones(n) / n
        print("Using default uniform init")

    n_parent = len(a_parent)
    best_a = a_parent.copy()
    best_score = _score_full(a_parent)
    print(f"Initial: n={n_parent}, score={best_score:.10f}")

    # Phase 1: FISTA from parent
    p1_end = min(deadline, t_start + 80.0)
    print(f"Phase 1: FISTA n={n_parent}")
    a_fista, sc = _fista_backtrack(a_parent, p1_end, max_iter=100000)
    full_sc = _score_full(a_fista)
    print(f"  FISTA: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_fista.copy()

    # Phase 2: Exp gradient
    p2_end = min(deadline, t_start + 150.0)
    if time.monotonic() < p2_end:
        print(f"Phase 2: Exp-Grad n={n_parent}")
        a_eg, sc_eg = _exp_gradient(a_parent, p2_end, max_iter=50000)
        full_sc_eg = _score_full(a_eg)
        print(f"  ExpGrad: score={full_sc_eg:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_eg) and full_sc_eg < best_score:
            best_score = full_sc_eg
            best_a = a_eg.copy()

    # Phase 3: Fast CD polish
    p3_end = min(deadline, t_start + 250.0)
    if time.monotonic() < p3_end:
        print(f"Phase 3: Fast CD n={n_parent}")
        a_cd, sc_cd = _fast_cd_active(best_a, p3_end, active_size=30, max_rounds=3000)
        full_sc_cd = _score_full(a_cd)
        print(f"  CD: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 4: Multi-start at n=2048
    p4_end = min(deadline, t_start + 500.0)
    if time.monotonic() < p4_end:
        print(f"Phase 4: Multi-start n=2048")
        n2k = 2048
        inits = _generate_diverse_inits(n2k, rng)
        # Include parent interpolated to 2048
        inits.append(_interp_to_length(best_a, n2k))

        time_per_init = max(15.0, (p4_end - time.monotonic()) / len(inits))
        for i, init in enumerate(inits):
            if time.monotonic() > p4_end - 10:
                break
            sub_dl = min(p4_end - 5, time.monotonic() + time_per_init)
            # Quick FISTA
            a_opt, sc_opt = _fista_backtrack(init, sub_dl, max_iter=30000)
            full_sc_opt = _score_full(a_opt)
            print(f"  Init[{i}]: score={full_sc_opt:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_opt) and full_sc_opt < best_score:
                best_score = full_sc_opt
                best_a = a_opt.copy()

    # Phase 5: Interpolate to n=4096
    p5_end = min(deadline, t_start + 800.0)
    if time.monotonic() < p5_end:
        n4k = 4096
        a_4k = _interp_to_length(best_a, n4k)
        print(f"Phase 5: Optimize n=4096, initial score={_score_full(a_4k):.10f}")

        # Adam log-space
        mid5 = min(p5_end, time.monotonic() + 150.0)
        a_4k_opt, sc_4k = _adam_logspace(a_4k, mid5, max_iter=100000, lr_start=0.025, lr_end=0.002)
        full_sc_4k = _score_full(a_4k_opt)
        print(f"  Adam n=4096: score={full_sc_4k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_4k) and full_sc_4k < best_score:
            best_score = full_sc_4k
            best_a = a_4k_opt.copy()

        # FISTA
        if time.monotonic() < p5_end - 30:
            mid5b = min(p5_end - 10, time.monotonic() + 80.0)
            a_4k_f, sc_4kf = _fista_backtrack(best_a, mid5b, max_iter=50000)
            full_sc_4kf = _score_full(a_4k_f)
            print(f"  FISTA n=4096: score={full_sc_4kf:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_4kf) and full_sc_4kf < best_score:
                best_score = full_sc_4kf
                best_a = a_4k_f.copy()

        # CD polish
        if time.monotonic() < p5_end - 5:
            a_4k_cd, sc_4kcd = _fast_cd_active(best_a, p5_end - 2, active_size=30, max_rounds=2000)
            full_sc_4kcd = _score_full(a_4k_cd)
            print(f"  CD n=4096: score={full_sc_4kcd:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_4kcd) and full_sc_4kcd < best_score:
                best_score = full_sc_4kcd
                best_a = a_4k_cd.copy()

    # Phase 6: Try n=8192
    p6_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < p6_end:
        n8k = 8192
        a_8k = _interp_to_length(best_a, n8k)
        init_sc_8k = _score_full(a_8k)
        print(f"Phase 6: Optimize n=8192, initial score={init_sc_8k:.10f}")

        # Adam log-space (less time since n is larger)
        mid6 = min(p6_end, time.monotonic() + 120.0)
        a_8k_opt, sc_8k = _adam_logspace(a_8k, mid6, max_iter=50000, lr_start=0.02, lr_end=0.001)
        full_sc_8k = _score_full(a_8k_opt)
        print(f"  Adam n=8192: score={full_sc_8k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_8k) and full_sc_8k < best_score:
            best_score = full_sc_8k
            best_a = a_8k_opt.copy()

        # FISTA
        if time.monotonic() < p6_end - 20:
            mid6b = min(p6_end - 5, time.monotonic() + 60.0)
            a_8k_f, sc_8kf = _fista_backtrack(best_a, mid6b, max_iter=30000)
            full_sc_8kf = _score_full(a_8k_f)
            print(f"  FISTA n=8192: score={full_sc_8kf:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_8kf) and full_sc_8kf < best_score:
                best_score = full_sc_8kf
                best_a = a_8k_f.copy()

        # CD polish
        if time.monotonic() < p6_end - 5:
            a_8k_cd, sc_8kcd = _fast_cd_active(best_a, p6_end - 2, active_size=30, max_rounds=1000)
            full_sc_8kcd = _score_full(a_8k_cd)
            print(f"  CD n=8192: score={full_sc_8kcd:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_8kcd) and full_sc_8kcd < best_score:
                best_score = full_sc_8kcd
                best_a = a_8k_cd.copy()

    # Phase 7: Multi-start at best n
    p7_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < p7_end:
        n_best = len(best_a)
        print(f"Phase 7: Multi-start n={n_best}")
        inits = _generate_diverse_inits(n_best, rng)
        inits.append(best_a.copy())

        time_per = max(20.0, (p7_end - time.monotonic()) / max(1, len(inits)))
        for i, init in enumerate(inits):
            if time.monotonic() > p7_end - 15:
                break
            sub_dl = min(p7_end - 5, time.monotonic() + time_per)
            a_opt, sc_opt = _fista_backtrack(init, sub_dl, max_iter=30000)
            # Quick CD
            if time.monotonic() < sub_dl - 5:
                a_opt, sc_opt2 = _fast_cd_active(a_opt, sub_dl, active_size=20, max_rounds=500)
            full_sc_opt = _score_full(a_opt)
            print(f"  MS[{i}] n={n_best}: score={full_sc_opt:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_opt) and full_sc_opt < best_score:
                best_score = full_sc_opt
                best_a = a_opt.copy()

    # Phase 8: Final intensive CD
    if time.monotonic() < deadline - 10:
        print(f"Phase 8: Final CD n={len(best_a)}")
        a_final_cd, sc_final = _fast_cd_active(best_a, deadline - 5, active_size=50, max_rounds=5000)
        full_sc_final = _score_full(a_final_cd)
        print(f"  Final CD: score={full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_final) and full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final_cd.copy()

        # One more Adam pass on the best
        if time.monotonic() < deadline - 15:
            a_adam_final, _ = _adam_logspace(best_a, deadline - 5,
                                             lr_start=0.005, lr_end=0.0005,
                                             max_iter=50000)
            full_sc_af = _score_full(a_adam_final)
            print(f"  Final Adam: score={full_sc_af:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_af) and full_sc_af < best_score:
                best_score = full_sc_af
                best_a = a_adam_final.copy()
            # Quick CD after
            if time.monotonic() < deadline - 3:
                a_cd_fin, _ = _fast_cd_active(best_a, deadline - 2, active_size=30, max_rounds=500)
                sc_cd_fin = _score_full(a_cd_fin)
                if sc_cd_fin < best_score:
                    best_score = sc_cd_fin
                    best_a = a_cd_fin.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score_full(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
