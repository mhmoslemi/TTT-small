"""
Dual-game optimization with multiplicative weights, FISTA inner solver,
L-BFGS polish, Newton active-set refinement, and multi-n exploration.
Key innovation: formulates the problem as a zero-sum game between lag
distributions (lambda) and sequences (a), solved via multiplicative
weights with efficient FFT-based Hankel matrix-vector products.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_matvec(a, lam, N_mat):
    """Compute M @ a where M[i,j] = lam[i+j]."""
    n = len(a)
    fa = np.fft.rfft(a, N_mat)
    fl = np.fft.rfft(lam, N_mat)
    corr = np.fft.irfft(np.conj(fa) * fl, N_mat)
    return corr[:n]


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_inner(a_init, lam, N_mat, max_iter=30, step=0.35):
    """FISTA to solve min a^T M a s.t. sum a = 1, a >= 0."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    t = 1.0
    a_prev = a.copy()

    for _ in range(max_iter):
        Ma = _hankel_matvec(y, lam, N_mat)
        grad = 2.0 * Ma
        a_new = _proj_simplex(y - step * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = a_new + beta * (a_new - a_prev)
        a_prev = a_new
        a = a_new
        t = t_new
    return a


def _dual_game(a_init, deadline, max_inner_iters=25, verbose=True):
    """
    Dual game: multiplicative weights on lambda (lag distribution)
    with FISTA inner solve for best response a.
    """
    n = len(a_init)
    N_mat = _next_pow2(3 * n + 2)

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize lambda from initial autocorrelation
    c_init = _fft_convolve(a, a)
    lam = np.maximum(c_init, 1e-15)
    lam = lam / lam.sum()

    best_a = a.copy()
    best_c_max = float(c_init.max())
    best_score = 2.0 * n * best_c_max

    t_start = time.monotonic()
    outer_iter = 0
    report_interval = max(1, int(outer_iter // 10) + 100)

    while time.monotonic() < deadline:
        # Inner solve: FISTA
        a = _fista_inner(a, lam, N_mat, max_iter=max_inner_iters, step=0.35)

        # Evaluate
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if c_max < best_c_max:
            best_c_max = c_max
            best_a = a.copy()
            best_score = sc
            if verbose and outer_iter % 200 == 0:
                print(f"    Dual iter {outer_iter}: NEW BEST score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

        # Update lambda: concentrate on high c_k lags
        # Use adaptive eta
        c_sorted = np.sort(c)
        c_95 = c_sorted[int(0.95 * len(c_sorted))]
        c_50 = c_sorted[int(0.50 * len(c_sorted))]
        gap = c_95 - c_50
        if gap < 1e-15:
            eta_out = 1.0
        else:
            eta_out = 1.0 / gap
            eta_out = min(eta_out, 200.0)

        # Multiplicative weights update
        log_lam = np.log(np.maximum(lam, 1e-300)) + eta_out * c
        log_lam -= log_lam.max()
        lam_new = np.exp(log_lam)
        lam_new = np.maximum(lam_new, 1e-300)
        lam_new = lam_new / lam_new.sum()

        # Smooth mixing to avoid instability
        alpha_mix = min(1.0, 0.3 + 0.7 * (outer_iter / 500.0))
        lam = alpha_mix * lam_new + (1.0 - alpha_mix) * lam
        lam = lam / lam.sum()

        outer_iter += 1

        if verbose and outer_iter % 500 == 0:
            print(f"    Dual iter {outer_iter}: c_max={c_max:.8e}, score={sc:.10f}, "
                  f"best={best_score:.10f}, eta_out={eta_out:.2f}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_c_max, lam


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta continuation on the simplex (log-parametrized)."""
    n = len(a_init)
    c_cur = _fft_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001,
                                  0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time = 0.004 * max(1, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 10))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxcor': 60, 'maxls': 100, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _hessian_matvec(v, active, n):
    """Compute H*v where (Hv)_i = 2 * sum_j v[j] * active[i+j]."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _newton_active(a_init, deadline, max_iter=100):
    """Newton's method on the active set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.0005 * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-8
        # CG to solve (H + mu*I) dx = -grad_t
        x = np.zeros(n)
        r = -grad_t.copy()
        p = r.copy()
        rs = float(np.dot(r, r))
        b_norm = float(np.linalg.norm(-grad_t))
        if b_norm < 1e-30:
            break
        for _ in range(150):
            Hp = _hessian_matvec(p, active, n) + mu * p
            pH = float(np.dot(p, Hp))
            if abs(pH) < 1e-30:
                break
            alpha_cg = rs / pH
            x = x + alpha_cg * p
            r = r - alpha_cg * Hp
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-10 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p = r + beta_cg * p
            rs = r_new

        dx = x - np.mean(x)

        # Line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _perturb_fourier(a, rng, n_modes=5, amp_scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = amp_scale * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _pherbo(a, rng, sigma=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Dual-game optimization with FISTA inner solver, L-BFGS polish,
    Newton active-set, multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase A: Dual game (main) ===
    print(f"\n[Phase A] Dual game n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    dual_deadline_A = min(deadline - 600, t_start + 500)
    if time.monotonic() < dual_deadline_A:
        a_dual, c_max_dual, lam_dual = _dual_game(
            parent, dual_deadline_A, max_inner_iters=25, verbose=True)
        sc_dual = 2.0 * n_parent * c_max_dual / (a_dual.sum() ** 2)
        update_best(a_dual, sc_dual, "dual_game")

        # L-BFGS polish on dual result
        if time.monotonic() < deadline - 400:
            a_pol = _lbfgs_smooth(a_dual, deadline, min_time_remaining=400)
            update_best(a_pol, _score(a_pol), "dual+lbfgs")

    # === Phase B: L-BFGS from parent ===
    print(f"\n[Phase B] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a_main = _lbfgs_smooth(parent, deadline, min_time_remaining=350)
        update_best(a_main, _score(a_main), "lbfgs_parent")

    # === Phase C: Newton active set ===
    print(f"\n[Phase C] Newton active set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_newton, sc_newton = _newton_active(best_seq, min(deadline - 250, time.monotonic() + 120), max_iter=80)
        update_best(a_newton, sc_newton, "newton")

    # === Phase D: Multi-n exploration ===
    print(f"\n[Phase D] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break

        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = np.ones(n_try) / n_try

        # Try dual game for this n
        if time.monotonic() < deadline - 80:
            dd = min(deadline - 80, time.monotonic() + 120)
            a_d, cm_d, _ = _dual_game(a_init, dd, max_inner_iters=20, verbose=False)
            sc_d = 2.0 * n_try * cm_d / (a_d.sum() ** 2)
            update_best(a_d, sc_d, f"dual_n{n_try}")

        # L-BFGS
        if time.monotonic() < deadline - 60:
            a_opt = _lbfgs_smooth(a_init, deadline, min_time_remaining=50)
            update_best(a_opt, _score(a_opt), f"lbfgs_n{n_try}")

    # === Phase E: Diverse restarts ===
    print(f"\n[Phase E] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _pherbo(best_seq, rng, sigma=0.02)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3, amp_scale=0.02)
            elif i == 2:
                p_start = _perturb_fourier(best_seq, rng, n_modes=8, amp_scale=0.01)
            elif i == 3:
                p_start = _pherbo(best_seq, rng, sigma=0.01)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=2, amp_scale=0.04)

            a_r = _lbfgs_smooth(p_start, deadline, min_time_remaining=10)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 8:
                a_r2, sc_r2 = _newton_active(a_r, min(deadline - 5, time.monotonic() + 20), max_iter=30)
                update_best(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase F: Final ultra-low-beta L-BFGS ===
    print(f"\n[Phase F] Final ultra polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _fft_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00001
        remaining = deadline - 3 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(100, int(remaining / 0.02)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                             'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _newton_active(best_seq, max(deadline - 2, time.monotonic()), max_iter=30)
            update_best(a_fin2, sc_fin2, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
                print(f"  Evaluator gave: {sc_eval:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
