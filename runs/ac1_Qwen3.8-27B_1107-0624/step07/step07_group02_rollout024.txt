def _score_and_topk(a, k):
    """Compute score and top-k gradient."""
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    max_c = float(c.max())
    score = 2.0 * n * max_c / (s * s)
    
    # Top-k weights
    if k >= len(c):
        k = len(c)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[top_idx] = 1.0 / k
    
    # Gradient
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    
    # Full gradient with scaling
    # d(score)/da = d(2n*max_c/s^2)/da = (2n/s^2)*dc_max/da - (4n*max_c/s^3)*sum(da)
    # For the smoothed version with top-k:
    # d(score)/da[i] = (2n/s^2)*h[i] - (4n*W/s^3)*s where W = sum(w*c) = mean of top-k c values
    # Actually: score = 2n * W / s^2 where W = sum(w*c)
    # d(score)/da[i] = 2n * (dW/da[i] * s^2 - W * 2s * da_s/da[i]) / s^4
    # = 2n * (h[i]*s - 2*W) / s^3
    # where da_s/da[i] = 1 (since s = sum(a))
    
    W = float(np.dot(w, c))
    grad = (2.0 * n * (h * s - 2.0 * W)) / (s ** 3)
    
    return score, grad, c


def _OptimizeEventTopK(a_init, deadline, k=100, lr_init=0.02, momentum=0.9):
    """Optimize using top-k gradient with momentum and backtracking."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    vel = np.zeros(n)
    lr = lr_init
    no_improve = 0
    
    t_start = time.monotonic()
    iter_count = 0
    
    while time.monotonic() < deadline:
        sc, grad, c = _score_and_topk(a, k)
        iter_count += 1
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Subtract mean to stay on the "sum = 1" manifold (approximately)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        
        if gnorm < 1e-15:
            break
        
        # Nesterov momentum
        vel = momentum * vel - lr * grad_centered
        step = vel.copy()
        
        # Backtracking line search
        improved = False
        alpha = 1.0
        for _ in range(30):
            a_try = a + alpha * step
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            vel *= 0.5
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.7
                if no_improve > 200:
                    # Restart with fresh momentum
                    vel = np.zeros(n)
                    lr = lr_init * 0.5
                    no_improve = 0
                    a = best_a.copy()
            if no_improve > 500:
                break
        else:
            lr = min(lr_init, lr * 1.01)
        
        # Periodic k adjustment
        if iter_count % 100 == 0:
            # Reduce k over time for sharper gradient
            k = max(10, k * 0.95)
    
    return best_a, best_sc

def _resample(a, n):
    old_n = len(a)
    if old_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_ramp(n, alpha=1.0):
    """Create a ramp-like profile."""
    t = np.linspace(0, 1, n)
    a = t ** alpha
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_skew_cos(n, freq=3.0, amp=0.3):
    """Skewed cosine profile."""
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t) + 0.2 * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _perturb(a, sigma, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()

"""
Multi-scale optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-k gradient with Nesterov momentum, backtracking line search,
log-sum-exp smoothing, multi-n exploration (up to 32768), and diverse
initializations. Key innovation: adaptive top-k schedule with momentum
for sharper peak balancing, plus aggressive exploration of larger n
where the theoretical lower bound 2n/(2n-1) is closer to 1.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_grad_topk(a, k):
    """Compute score and gradient using top-k averaging."""
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    max_c = float(c.max())
    score = 2.0 * n * max_c / (s * s)

    if k >= len(c):
        k = len(c)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k

    W = float(np.dot(w, c))
    a_flip = a[::-1]
    wc = _full_convolve(w, a_flip)
    h = wc[n - 1:2 * n - 1].copy()

    # Gradient: 4n * (h * s - W) / s^3
    grad = 4.0 * n * (h * s - W) / (s ** 3)
    return score, grad, c


def _score_grad_lse(a, beta):
    """Compute score and gradient using log-sum-exp smoothing."""
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    max_c = float(c.max())
    score = 2.0 * n * max_c / (s * s)

    shifted = (c - max_c) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    S = e.sum()
    if S < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        W = max_c
    else:
        w = e / S
        W = max_c + beta * np.log(S)

    a_flip = a[::-1]
    wc = _full_convolve(w, a_flip)
    h = wc[n - 1:2 * n - 1].copy()

    grad = 4.0 * n * (h * s - W) / (s ** 3)
    return score, grad, c


def _optimize_momentum(a_init, deadline, mode="topk", k_sched=None,
                       beta_sched=None, lr_init=0.02, momentum=0.9,
                       max_stall=300):
    """Optimize with momentum-based gradient descent and backtracking."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = float(a.sum())
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    vel = np.zeros(n)
    lr = lr_init
    no_improve = 0
    stall_count = 0

    if mode == "topk" and k_sched is None:
        k_sched = [300, 150, 80, 40, 20, 10, 5, 3, 2]
    if mode == "lse" and beta_sched is None:
        beta_sched = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    if mode == "topk":
        sched = k_sched
    else:
        sched = beta_sched

    for phase in range(len(sched)):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        param = sched[phase]

        while time.monotonic() < deadline and (time.monotonic() + 10 > deadline - remaining * 0.05):
            if mode == "topk":
                sc, grad, c = _score_grad_topk(a, k=param)
            else:
                sc, grad, c = _score_grad_lse(a, beta=param)

            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            grad_t = grad - grad.mean()
            gnorm = np.linalg.norm(grad_t)
            if gnorm < 1e-16:
                break

            vel = momentum * vel - lr * grad_t
            step = vel.copy()
            step_norm = np.linalg.norm(step)
            if step_norm < 1e-18:
                break

            improved = False
            alpha = min(1.0, 0.5 / (step_norm + 1e-15))
            for _ in range(40):
                a_try = a + alpha * step
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-10:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < sc - 1e-15:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                vel *= 0.5
                no_improve += 1
                if no_improve % 15 == 0:
                    lr *= 0.75
                if no_improve > max_stall:
                    a = best_a.copy()
                    vel = np.zeros(n)
                    lr = lr_init * 0.5
                    no_improve = 0
                    stall_count += 1
                    if stall_count > 6:
                        break
                    max_stall = int(max_stall * 0.7)
            else:
                lr = min(lr_init, lr * 1.005)

    return best_a, best_sc


def _optimize_lbfgs_lse(a_init, deadline, min_time=10.0):
    """L-BFGS with log-sum-exp smoothing, decreasing beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 10))))

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            c_max = float(c_.max())
            shifted = (c_ - c_max) / beta
            e_ = np.exp(np.clip(shifted, -700, 0))
            S_ = e_.sum()
            if S_ < 1e-300 or not np.isfinite(S_):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(S_))

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            c_max = float(c_.max())
            shifted = (c_ - c_max) / beta
            e_ = np.exp(np.clip(shifted, -700, 0))
            S_ = e_.sum()
            if S_ < 1e-300 or not np.isfinite(S_):
                w_ = np.zeros_like(c_)
                w_[int(np.argmax(c_))] = 1.0
            else:
                w_ = e_ / S_
            a_rev = a_[::-1]
            wc = _full_convolve(w_, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w_, c_))
            return 4.0 * n * a_ * (h - W)

        try:
            result = minimize(obj, z, jac=grad_fn, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-13, 'maxcor': 50, 'maxls': 30})
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind, *args):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        alpha = args[0] if args else 1.5
        t = np.linspace(0, 1, n)
        a = t ** alpha
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "s_curve":
        steep = args[0] if args else 5.0
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-steep * (t - 0.5)))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "skew_cos":
        freq = args[0] if args else 3.0
        amp = args[1] if len(args) > 1 else 0.3
        t = np.linspace(0, 1, n)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t) + 0.3 * t
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "power_ramp":
        alpha = args[0] if args else 2.0
        t = np.linspace(0, 1, n)
        a = (1 - t) ** alpha + 0.1
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2) + 0.05 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax_fn(z)


def _active_set_polish(a_init, deadline, tol_start=0.01, tol_end=0.0001, lr_init=0.03):
    """Simple active-set gradient descent for final polish."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    tol_frac = tol_start
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))
        a_flip = a[::-1]
        wc = _full_convolve(w, a_flip)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * (h * s - W) / (s ** 3)
        grad_t = grad - grad.mean()
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a + alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve > 300:
                a = best_a.copy()
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                if no_improve > 1000:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.01)

    return best_a, best_sc
