"""
LP-based active-set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses CVXPY LP solving for exact active-set directions, combined with
L-BFGS log-sum-exp smoothing, active-set gradient descent, and
multi-n exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = a.sum()
    if s > 1e-15:
        return a / s
    return a


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(20, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lp_direction(a, M):
    """Solve the LP: max sum(b) s.t. b >= 0, conv(a,b) <= M.
    Returns the optimal b (normalized to sum 1) or None on failure."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Compute conv(a, b) using the constraint structure
    # conv(a,b)[m] = sum_j a[j] * b[m-j] for valid j
    # We need: sum_j a[m-j] * b[j] <= M for all m (with a indexed appropriately)
    # Actually: conv(a,b)[m] = sum_j b[j] * a[m-j]
    # So the constraint is: for each m, sum_j b[j] * a[m-j] <= M
    
    # Build the constraint matrix implicitly
    # A[m, j] = a[m-j] if 0 <= m-j < n, else 0
    # conv = A @ b <= M
    
    # Instead of building the full matrix, use the fact that
    # conv(a,b) = np.convolve(a, b) and express constraints
    # In CVXPY, we can use:
    # For each m in range(2*n-1): sum over valid j of a[m-j]*b[j] <= M
    
    # This is equivalent to: the convolution of a and b is element-wise <= M
    # We can express this using a block-diagonal / Toeplitz structure
    
    # CVXPY approach: use exp обслуживание
    # Actually, the simplest way in CVXPY is to build the constraint matrix
    
    # For n up to ~10000, building the matrix might be memory-intensive
    # Let's use a different approach: express conv(a,b) <= M as a set of linear constraints
    
    # conv(a,b) = ifft(fft(a) * fft(b))
    # But that's not linear in the CVXPY sense (it's the product of twoCVXPY expressions)
    
    # The correct approach: conv(a,b)[m] = sum_j a[m-j] * b[j]
    # This IS linear in b (for fixed a). So we need:
    # For each m: sum_j a[m-j] * b[j] <= M
    
    # In CVXPY, we can write this as:
    # Let A_mat be the matrix where A_mat[m, j] = a[m-j] if valid
    # Then A_mat @ b <= M * ones(2*n-1)
    
    # Building A_mat explicitly for n=8192: 16383 x 8192 matrix with ~67M non-zeros

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())

    b_val = None
    for round_num in range(15):
        if deadline is not None and time.monotonic() > deadline:
            break

        active_idx = np.unique(active_idx)
        n_c = len(active_idx)
        if n_c > max_constraints:
            # Keep the worst constraint violations
            active_idx = active_idx[np.argsort(c[active_idx])[::-1][:max_constraints]]
            n_c = max_constraints

        # Build constraint matrix
        t0 = time.monotonic()
        A = _build_constraint_matrix(a, active_idx)
        build_time = time.monotonic() - t0

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M * np.ones(n_c)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if b_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        b_raw = np.maximum(0, np.asarray(b_var.value, dtype=np.float64))
        s = b_raw.sum()
        if s < 1e-15:
            break
        b_val = b_raw / s

        # Check if all constraints are satisfied
        conv_check = _full_convolve(a, b_val * s)
        violations = conv_check > M * (1 + 1e-6)
        if not violations.any():
            return b_val, s

        # Add violated constraints (worst ones)
        viol_idx = np.where(violations)[0]
        # Keep at most max_constraints total
        all_candidates = np.concatenate([active_idx, viol_idx])
        all_candidates = np.unique(all_candidates)
        if len(all_candidates) > max_constraints:
            scores = np.zeros(n_total)
            scores[all_candidates] = c[all_candidates]
            active_idx = np.argsort(scores)[::-1][:max_constraints]
        else:
            active_idx = all_candidates

    if b_val is not None:
        s = 1.0  # already normalized
        return b_val, s
    return None, 0


def _lp_iterate(a_init, deadline, min_time_remaining=60.0):
    """LP-based active-set iteration with line search."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    stagnation = 0
    it = 0
    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        M = float(c.max())
        current_sc = 2.0 * n * M

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            stagnation = 0

        # LP with constrained deadline
        lp_deadline = min(deadline - min_time_remaining, time.monotonic() + 120.0)
        b_dir, opt_val = _lp_direction(a, M, max_constraints=3000, deadline=lp_deadline)

        if b_dir is None:
            break

        diff = np.linalg.norm(b_dir - a)
        if diff < 1e-7:
            stagnation += 1
            if stagnation > 15:
                break
            # Perturb to escape
            a_pert = a + 0.005 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            a = _normalize(a_pert)
            it += 1
            continue

        # Line search
        best_t = 0.0
        best_t_sc = current_sc
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * b_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b_dir
            a = np.maximum(a, 0.0)
            a = _normalize(a)
        else:
            stagnation += 1
            if stagnation > 15:
                break

        it += 1
        if it % 5 == 0:
            print(f"    LP it={it}: cur={current_sc:.10f} best={best_sc:.10f} "
                  f"t={best_t:.3f} diff={diff:.6f}")

    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.6
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS + LP active-set + constraint generation + polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS initial polish ===
    print(f"\n[P1] L-BFGS init (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1000.0, t_start + 250.0)
    a_main = _run_lbfgs(parent, p1_deadline, min_time_remaining=50.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_init]")

    # === Phase 2: LP-based active-set iteration (main workhorse) ===
    print(f"\n[P2] LP active-set (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 500.0, time.monotonic() + 900.0)
    if time.monotonic() < p2_deadline - 30:
        a_lp, sc_lp = _lp_iterate(best_seq, p2_deadline, min_time_remaining=30.0)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lp_main]")

    # === Phase 3: L-BFGS after LP (to smooth out LP artifacts) ===
    print(f"\n[P3] L-BFGS post-LP (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350.0:
        p3_deadline = min(deadline - 350.0, time.monotonic() + 200.0)
        a_post = _run_lbfgs(best_seq, p3_deadline, min_time_remaining=30.0, aggressive=True)
        sc_post = _score(a_post)
        if sc_post < best_score:
            best_seq = a_post.copy()
            best_score = sc_post
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_lp]")

    # === Phase 4: Second LP round ===
    print(f"\n[P4] LP round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250.0:
        p4_deadline = min(deadline - 250.0, time.monotonic() + 350.0)
        a_lp2, sc_lp2 = _lp_iterate(best_seq, p4_deadline, min_time_remaining=30.0)
        if sc_lp2 < best_score:
            best_seq = a_lp2.copy()
            best_score = sc_lp2
            print(f"  NEW BEST: {best_score:.10f} [lp_round2]")

    # === Phase 5: Active-set polish (fine) ===
    print(f"\n[P5] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150.0:
        p5_deadline = min(deadline - 150.0, time.monotonic() + 150.0)
        a_pol, sc_pol = _active_set_polish(best_seq, p5_deadline,
                                           tol_frac_start=0.004, tol_frac_end=0.00005,
                                           lr_init=0.02)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish_fine]")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120.0:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 120.0)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage]")
            # Quick re-optimize after shrinkage
            if time.monotonic() < deadline - 100.0:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=60.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink]")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    n_candidates = []
    for n_try in [4096, 6144, 8192, 10240, 12288]:
        if n_try != n_best:
            n_candidates.append(n_try)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60.0:
            break
        remaining = deadline - 60.0 - time.monotonic()
        if remaining < 40.0:
            break
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        if sc_init < best_score - 0.01:
            # Worth optimizing
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
            elif time.monotonic() < deadline - 35.0:
                a_pol2, sc_pol2 = _active_set_polish(a_opt, min(deadline - 30.0, time.monotonic() + 25.0),
                                                     tol_frac_start=0.005, tol_frac_end=0.0001,
                                                     lr_init=0.015)
                if sc_pol2 < best_score:
                    best_seq = a_pol2.copy()
                    best_score = sc_pol2
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish]")

    # === Phase 8: Diverse restarts ===
    print(f"\n[P8] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30.0:
        n_cur = len(best_seq)
        restart_starts = [
            _perturb(best_seq, 0.02, rng),
            _perturb(best_seq, 0.05, rng),
            _fourier_perturb(best_seq, rng, n_modes=3),
        ]
        for i, a_r in enumerate(restart_starts):
            if time.monotonic() > deadline - 15.0:
                break
            remaining = deadline - 15.0 - time.monotonic()
            if remaining < 15.0:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}]")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3.0:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 2.0,
                                           tol_frac_start=0.001, tol_frac_end=0.00001,
                                           lr_init=0.005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
