"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements:
1. Adaptive beta continuation (decrease beta only upon convergence)
2. Sparsification: zero out small entries to exploit boundary KKT conditions

5. Active-set-aware coordinate polishing
6. Dual averaging for better iterates from subgradient phases

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk(a, c, n, k=20):
    """Gradient using top-k active set."""
    n_c = len(c)
    k_use = min(k, n_c)
    idx = np.argpartition(c, -k_use)[-k_use:]
    w = np.zeros(n_c)
    w[idx] = 1.0 / k_use
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_adaptive(a, c, n, rel_tol=0.005):
    """Gradient using adaptive active set (within rel_tol of max)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = rel_tol * c_max
    active = c >= (c_max - tol)
    n_active = active.sum()
    if n_active < 3:
        idx = np.argsort(c)[-5:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
        n_active = 5
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_adaptive(a_init, deadline, min_remaining=30.0, max_beta_factor=1.0, min_beta_factor=0.0001):
    """
    L-BFGS with adaptive beta continuation.
    Decreases beta only when the current level is sufficiently converged.
    """
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Initial beta from gap
    c_cur = _full_convolve(current_a, current_a)
    sorted_c = np.sort(c_cur)[::-1]
    if len(sorted_c) > 50:
        gap = sorted_c[0] - sorted_c[50]
    else:
        gap = sorted_c[0] - sorted_c[-1]
    base_beta = max(gap, sorted_c[0] * 0.01)

    # Beta schedule: geometric, but we only move to next when converged
    n_levels = 18
    beta_ratios = np.geomspace(1.0, 0.0001, n_levels)
    max_beta = base_beta * max_beta_factor
    min_beta = base_beta * min_beta_factor

    level = 0
    best_z = z.copy()
    best_sc = current_sc

    while level < n_levels and time.monotonic() < deadline - min_remaining:
        beta = max(min_beta, max_beta * beta_ratios[level])

        # For this beta level, run L-BFGS until convergence or time limit
        time_level_start = time.monotonic()
        time_budget = min(120.0, (deadline - min_remaining - time.monotonic()) * 0.7)
        est_iter_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(20000, max(100, int(time_budget / est_iter_time)))

        converged = False
        try:
            prev_obj = _smoobj(z, beta, n)
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50}
            )
            z = result.x
            final_obj = _smoobj(z, beta, n)
            if abs(prev_obj - final_obj) < 1e-12 * max(1.0, abs(final_obj)):
                converged = True
        except Exception:
            pass

        a_new = _softmax(z)
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_z = z.copy()
            current_a = a_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()

        # Check if we should continue to next beta level
        if not converged:
            # Check time
            if time.monotonic() > deadline - min_remaining:
                break
            # If we used most of our time budget without converging, still move on
            if level < n_levels - 1:
                level += 1
        else:
            if level < n_levels - 1:
                level += 1
            else:
                break

        # Reset z from best for next level (warm start)
        if level < n_levels:
            z = np.log(np.maximum(best_z + 1e-30, 1e-30))
            z = np.log(np.maximum(_softmax(best_z), 1e-30))
            z = z - z.mean()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        current_a = a_final
    return current_a, best_sc


def _fista_smooth(a_init, deadline, beta=None, n_iter=5000):
    """FISTA for the smoothed problem in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()
    t = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)

    if beta is None:
        c = _full_convolve(a_init, a_init)
        c_max = float(c.max())
        beta = c_max * 0.05

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        grad = _smograd(y, beta, n)
        # Step size: for the smoothed problem, L ~ 4n * max|h| / beta (rough)
        # Use a conservative step size
        step = (beta / (4.0 * n * max(1.0, np.max(np.abs(grad)) / max(n, 1))))
        step = min(step, 1.0)
        z_new = y - step * grad
        z_new = np.clip(z_new, -100, 100)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = z_new + ((t - 1.0) / t_new) * (z_new - z)
        z = z_new
        t = t_new

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

        a_try = a_try / a_try.sum() * n  # normalize to sum=n for fair comparison
        # Actually, score is scale-invariant, so just check
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3):
    """
    Improved pressure equalization:
    - Adaptive active set based on top-K AND relative tolerance
    - Smoother updates with momentum
    - Periodic scoring
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    momentum = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive active set: use top-K and relative tolerance
        K = max(10, min(200, len(c) // 100))
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() < K:
            idx = np.argsort(c)[-K:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Momentum-based update
        p_mean = np.mean(p)
        ratio = p_mean / p
        target = np.power(ratio, gamma)
        momentum = 0.7 * momentum + 0.3 * (target - 1.0)
        a = a * np.exp(momentum)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 30 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent with dual averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    # Dual averaging
    z_avg = np.zeros(n)
    t = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_adaptive(a, c, n, rel_tol=0.01)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Dual averaging update
        t += 1
        z_avg += z

        if i % 50 == 0:
            # Check both current and averaged
            a_check = _softmax(z)
            sc = _score(a_check)
            a_avg = _softmax(z_avg / t)
            sc_avg = _score(a_avg)

            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.01, lr_init)
            elif sc_avg < best_sc - 1e-13:
                best_sc = sc_avg
                best_a = a_avg.copy()
                lr = min(lr * 1.01, lr_init)
            else:
                lr *= 0.95
                if lr < lr_init * 0.01:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    # Also check averaged
    if t > 0:
        a_avg = _softmax(z_avg / t)
        sc_avg = _score(a_avg)
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = a_avg.copy()

    return best_a, best_sc


def _coordinate_polish(a, deadline, n_rounds=80):
    """Coordinate descent focusing on high-impact coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(999)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find high-impact coordinates via pressure
        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1]

        # Try modifying top-contributing coordinates
        n_try = min(n, max(20, n // 20))
        top_idx = np.argsort(contrib)[-n_try:]
        rng.shuffle(top_idx)
        top_idx = top_idx[:min(len(top_idx), n_try // 3)]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue

            c_old = c_max  # reuse
            best_factor = 1.0
            best_c = c_old

            # Try reduction factors (most useful since we want to reduce peaks)
            for factor in [0.3, 0.5, 0.7, 0.9]:
                a[i] = old_val * factor
                # Quick check: only recompute affected c values
                # For efficiency, just check a few key positions
                c_partial = _full_convolve(a, a)
                c_try = c_partial.max()
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    best_factor = factor
                else:
                    a[i] = old_val
                    break
            else:
                a[i] = old_val
                # Try the best factor found
                if best_factor != 1.0:
                    a[i] = old_val * best_factor
                    improved = True
                continue

            if best_factor != 1.0:
                a[i] = old_val * best_factor
                improved = True
            else:
                a[i] = old_val

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 2 * t * t) + 2.0
        return a / a.sum()
    elif kind == "multi_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 7 * t) + 0.2 * np.cos(2 * np.pi * 19 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "asymmetric_bell":
        t = np.linspace(0, 1, n)
        # Asymmetric: peak at 0.4
        a = np.exp(-12.0 * (t - 0.4) ** 2) + 0.03
        return a / a.sum()
    elif kind == "two_lobe":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.03):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _interpolate_to_n(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search:
    Phase 1: Quick optimization of parent (GAP in offspring.

    Phase 2: Intensive L-BFGS with adaptive beta + sparsification.
    Phase 3: Multi-n exploration.
    Phase 4: Diverse restarts.
    Phase 5: Final ultra-fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={elapsed:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        _update(parent, parent_sc)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    n = len(parent)

    # === Phase 1: Quick L-BFGS on parent (budget: ~150s) ===
    print(f"\n[Phase 1] Quick L-BFGS on parent n={n} (t={time.monotonic()-t_start:.1f}s)")
    t_phase1_end = t_start + 150.0
    a_p1, sc_p1 = _lbfgs_adaptive(parent, min(deadline, t_phase1_end), min_remaining=600)
    _update(a_p1, sc_p1)

    # Quick pressure equalization
    if time.monotonic() < deadline - 500:
        t_pe = min(deadline - 500, time.monotonic() + 40)
        a_pe, sc_pe_norm = _pressure_equalize_v2(a_p1, t_pe, n_iters=100, gamma=0.25)
        sc_pe = _score(a_pe)
        _update(a_pe, sc_pe)
        if sc_pe < best_score:
            a_p1 = a_pe

    # === Phase 2: Intensive optimization (budget: ~600s) ===
    print(f"\n[Phase 2] Intensive optimization (t={time.monotonic()-t_start:.1f}s)")
    t_phase2_end = t_start + 700.0

    # Multiple L-BFGS passes with sparsification between
    current_best = a_p1.copy()
    for pass_idx in range(4):
        if time.monotonic() > min(deadline - 300, t_phase2_end):
            break

        # L-BFGS pass
        a_opt, sc_opt = _lbfgs_adaptive(current_best, min(deadline - 300, t_phase2_end),
                                         min_remaining=200, max_beta_factor=0.3)
        _update(a_opt, sc_opt)
        if sc_opt < _score(current_best):
            current_best = a_opt

        # Sparsification
        if time.monotonic() < min(deadline - 300, t_phase2_end) - 30:
            a_sp, sc_sp = _sparsify(current_best, min(deadline - 300, t_phase2_end) - 30)
            _update(a_sp, sc_sp)
            if sc_sp < _score(current_best):
                current_best = a_sp
                # Re-optimize after sparsification
                if time.monotonic() < min(deadline - 300, t_phase2_end) - 60:
                    a_reopt, sc_reopt = _lbfgs_adaptive(current_best,
                                                         min(deadline - 300, t_phase2_end) - 60,
                                                         min_remaining=150, max_beta_factor=0.1)
                    _update(a_reopt, sc_reopt)
                    if sc_reopt < _score(current_best):
                        current_best = a_reopt

    # Subgradient polish
    if time.monotonic() < deadline - 250:
        t_sg = min(deadline - 250, time.monotonic() + 60)
        a_sg, sc_sg = _subgrad_polish(current_best, t_sg, n_max_steps=2000, lr_init=0.015)
        _update(a_sg, sc_sg)
        if sc_sg < _score(current_best):
            current_best = a_sg

    # === Phase 3: Multi-n exploration (budget: ~400s) ===
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    t_phase3_end = t_start + 1150.0

    n_candidates = [6144, 7168, 7680, 8704, 9216, 10240, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != len(current_best)]
    n_candidates.sort(key=lambda x: abs(x - len(current_best)))

    for n_try in n_candidates:
        if time.monotonic() > t_phase3_end:
            break
        remaining = t_phase3_end - time.monotonic()
        if remaining < 40:
            break

        # Interpolate parent to new n
        a_init = _interpolate_to_n(current_best, n_try)

        # Quick optimization
        min_rem = max(20, (deadline - time.monotonic()) * 0.25)
        a_opt, sc_opt = _lbfgs_adaptive(a_init, min(deadline, t_phase3_end),
                                         min_remaining=min_rem, max_beta_factor=0.5)
        _update(a_opt, sc_opt)

        # Try sparsification
        if sc_opt < best_score + 0.01 and time.monotonic() < t_phase3_end - 20:
            a_sp, sc_sp = _sparsify(a_opt, t_phase3_end - 20, fracs=[0.02, 0.05, 0.10])
            _update(a_sp, sc_sp)

    # === Phase 4: Diverse restarts (budget: ~200s) ===
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    t_phase4_end = min(deadline - 50, t_start + 1400.0)

    if best_seq is not None:
        n_best = len(best_seq)
        profiles = ["asymmetric_bell", "two_lobe", "multi_cos", "gaussian", "chirp"]

        for i, prof in enumerate(profiles):
            if time.monotonic() > t_phase4_end:
                break

            # Blend best with profile
            blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()

            # Fourier perturbation
            a_pert = _fourier_perturb(blend, rng, n_modes=5, amplitude=0.025)

            # Quick optimization
            min_rem = max(15, (deadline - time.monotonic()) * 0.2)
            a_r, sc_r = _lbfgs_adaptive(a_pert, min(deadline, t_phase4_end),
                                         min_remaining=min_rem, max_beta_factor=0.3)
            _update(a_r, sc_r)

    # === Phase 5: Final ultra-fine polish (remaining time) ===
    print(f"\n[Phase 5] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        remaining = deadline - time.monotonic()

        # Ultra-fine L-BFGS
        if remaining > 30:
            t_ultra = min(deadline - 5, t_start + budget_s - 15)
            a_ultra, sc_ultra = _lbfgs_adaptive(best_seq, t_ultra,
                                                 min_remaining=3, max_beta_factor=0.05)
            _update(a_ultra, sc_ultra)

        # Pressure equalization (final)
        if time.monotonic() < deadline - 15:
            t_pe_final = min(deadline - 15, time.monotonic() + 30)
            a_pe_f, _ = _pressure_equalize_v2(best_seq, t_pe_final, n_iters=200, gamma=0.2)
            sc_pe_f = _score(a_pe_f)
            _update(a_pe_f, sc_pe_f)

        # Coordinate polish
        if time.monotonic() < deadline - 10:
            t_cd = min(deadline - 10, time.monotonic() + 30)
            a_cd, _ = _coordinate_polish(best_seq, t_cd, n_rounds=50)
            sc_cd = _score(a_cd)
            _update(a_cd, sc_cd)

        # Final sparsification
        if time.monotonic() < deadline - 5:
            a_sp_f, sc_sp_f = _sparsify(best_seq, deadline - 5, fracs=[0.01, 0.03, 0.05, 0.10])
            _update(a_sp_f, sc_sp_f)

        # Very last L-BFGS
        if time.monotonic() < deadline - 3:
            a_last, sc_last = _lbfgs_adaptive(best_seq, deadline - 3,
                                               min_remaining=1, max_beta_factor=0.01)
            _update(a_last, sc_last)

    # === Final validation and return ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal_score={sc_final:.10f}, eval_score={eval_sc:.10f}, "
                  f"n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        except Exception:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(1024, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
