"""
Optimizer using exact subgradient FISTA with adaptive steps, symmetric
enforcement, multi-resolution (coarse-to-fine), simulated annealing restarts,
and block coordinate descent polishing. Exploits the non-convex structure of
minimizing max autocorrelation peak over the probability simplex.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _exact_subgradient(a, c, top_k=10, thresh_frac=0.999):
    """Compute exact subgradient using active set (positions where c is close to max)."""
    n = len(a)
    c_max = c.max()
    # Active set: positions where c_k >= thresh_frac * c_max
    active_thresh = thresh_frac * c_max
    active = np.where(c >= active_thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Limit to top_k
    if len(active) > top_k:
        active = np.argsort(c[active])[::-1][:top_k] + active[np.argsort(c[active])[::-1][:top_k]]
        # Safer: just take top_k by value
        top_indices = np.argsort(c)[::-1][:top_k]
        active = top_indices[top_indices < len(c)]
    
    # Gradient of c_k w.r.t. a: (2/|S|) * sum_{k in S} [a_{k-j} for valid j]
    # For each active k, the gradient contribution at position j is 2*a_{k-j} if 0 <= k-j < n
    g = np.zeros(n)
    for k in active:
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            # a_{k-j} for j in [j_min, j_max]
            # k-j ranges from k-j_max to k-j_min
            vals = a[k - j_max : k - j_min + 1][::-1]
            g[j_min:j_max+1] += 2.0 * vals
    g /= len(active)
    return g, active


def _project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _fista_optimize(a_init, deadline, max_iter=200000, use_symmetry=False,
                    lr_init=0.1, active_top=10, thresh_frac=0.999,
                    restart_every=5000, patience_limit=3000):
    """FISTA with exact subgradient and adaptive step size."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    if use_symmetry:
        # Work with half the variables
        half = (n + 1) // 2
        def unpack(x):
            out = np.zeros(n)
            out[:half] = x
            out[half:] = x[:n - half][::-1] if n % 2 == 0 else x[:n - half][::-1]
            # Fix: for even n, out[half:] = x[:half][::-1]; for odd, out[half+1:] = x[1:half][::-1]
            if n % 2 == 0:
                out[half:] = x[:half][::-1]
            else:
                out[half:] = x[1:half][::-1]
                out[half] = x[half]
            return out
        def pack(a_full):
            return a_full[:half].copy()
        x = pack(a).copy()
        x = x / x.sum()  # This isn't quite right for symmetric case, let me handle differently
        # Actually for symmetric a with sum=1: 2*sum(x[:half-1]) + x[half-1] = 1 (if n odd)
        # or 2*sum(x[:half]) = 1 (if n even)
        # Let's just work in full space but enforce symmetry after each step
        pass
    
    t = 0
    y = a.copy()
    t_k = 1.0
    step = lr_init
    prev_grad = None
    prev_y = None
    patience = 0
    best_patience_score = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        c = _fft_convolve(y, y)
        g, active_set = _exact_subgradient(y, c, top_k=active_top, thresh_frac=thresh_frac)
        
        # Barzilai-Borwein step size
        if prev_grad is not None and prev_y is not None:
            sy = y - prev_y
            yg = g - prev_grad
            sy_norm = np.dot(sy, sy)
            yg_norm = np.dot(yg, yg)
            if sy_norm > 1e-30 and yg_norm > 1e-30:
                syyg = np.dot(sy, yg)
                yygg = np.dot(yg, yg)
                if syyg > 1e-15 and yygg > 1e-15:
                    step_bb = syyg / yygg
                    step = 0.5 * (step + np.clip(step_bb, 0.01 * step, 10.0 * step))
        
        # FISTA update
        a_new = y - step * g
        if use_symmetry:
            # Enforce symmetry
            a_new = 0.5 * (a_new + a_new[::-1])
        a_new = _project_simplex(a_new)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - y)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a_new.copy()
        else:
            y /= y_sum
        if use_symmetry:
            y = 0.5 * (y + y[::-1])
            y /= y.sum()
        t_k = t_new
        prev_y = y.copy()
        prev_grad = g.copy()
        
        # Evaluation
        if t % 100 == 0:
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
        
        # Adaptive restart
        if patience >= patience_limit:
            y = best_a.copy()
            t_k = 1.0
            step = max(lr_init * 0.5, step * 0.3)
            patience = 0
            prev_grad = None
            prev_y = None
        
        # Periodic restart from best
        if t % restart_every == 0:
            if patience > patience_limit // 2:
                y = best_a.copy()
                t_k = 1.0
                step = lr_init
                patience = 0
                prev_grad = None
                prev_y = None
    
    # Final evaluation
    sc = _score_norm(a_new) if t > 0 else best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_new.copy()
    
    return best_a, best_score


def _block_cd_polish(a, deadline, block_size=128, max_rounds=50):
    """Block coordinate descent: optimize blocks of variables at a time."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            if end - start < 2:
                continue
            block = best_a[start:end].copy()
            # Try to improve the block using a few local gradient steps
            for local_iter in range(20):
                c_cur = _fft_convolve(best_a, best_a)
                g_full, _ = _exact_subgradient(best_a, c_cur, top_k=5, thresh_frac=0.9995)
                g_block = g_full[start:end]
                # Multiplicative update within block
                lr = 0.1
                new_block = block * np.exp(-lr * g_block)
                new_block = np.maximum(new_block, 0.0)
                if new_block.sum() < 1e-15:
                    new_block = block.copy()
                # Update full array and renormalize
                cand = best_a.copy()
                cand[start:end] = new_block
                cand /= cand.sum()
                sc = _score_norm(cand)
                if sc < best_score - 1e-12:
                    best_a = cand
                    best_score = sc
                    block = new_block
                    improved = True
                else:
                    lr *= 0.5
                    if lr < 1e-6:
                        break
        if not improved:
            break
    
    return best_a, best_score


def _annealing_restart(a, deadline, rng, sigma=0.05, max_restarts=5):
    """Simulated annealing style restarts."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for i in range(max_restarts):
        if time.monotonic() > deadline:
            break
        # Perturb in log space
        z = np.log(np.maximum(a, 1e-15))
        z += sigma * rng.standard_normal(n)
        z -= z.mean()
        a_pert = np.exp(z)
        a_pert /= a_pert.sum()
        
        # Quick local optimization
        sub_deadline = min(deadline, time.monotonic() + max(10.0, (deadline - time.monotonic()) / (max_restarts - i)))
        a_opt, sc = _fista_optimize(a_pert, sub_deadline, max_iter=50000,
                                     lr_init=0.05, active_top=5, patience_limit=1000)
        full_sc = _score_norm(a_opt)
        print(f"  Anneal[{i}] sigma={sigma:.3f}: score={full_sc:.10f}")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()
        sigma *= 0.7
    
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    p_new /= p_new.sum()
    return p_new


def _coordinate_descent_single(a, deadline, max_rounds=100, top_positions=50):
    """Coordinate descent on individual elements, focusing on positions that affect active set."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        c_max = c.max()
        # Find positions that most affect the max
        active_k = np.where(c >= 0.999 * c_max)[0]
        # Find a-indices that appear in these convolution positions
        affected = set()
        for k in active_k:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1, max(1, (j_max - j_min) // top_positions)):
                affected.add(j)
        affected = list(affected)[:top_positions * 2]
        
        improved = False
        for j in affected:
            old_val = best_a[j]
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.0]:
                new_val = old_val * frac if frac > 0 else 0.0
                if new_val < 0:
                    continue
                cand = best_a.copy()
                cand[j] = new_val
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand /= s
                sc = _score_norm(cand)
                if sc < best_score - 1e-14:
                    best_a = cand
                    best_score = sc
                    improved = True
                    break
        if not improved:
            break
    
    return best_a, best_score
