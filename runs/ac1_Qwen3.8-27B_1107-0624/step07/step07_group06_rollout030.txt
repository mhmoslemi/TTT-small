Dual Minimax Optimizer with FFT-based QP solving and multi-resolution
embedded constraint projection for minimizing the peak autocorrelation
of non-negative step functions.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _hankel_matvec(v, p):
    """Compute H_p @ v where (H_p)_{ij} = p_{i+j}.
    Uses FFT: result[i] = sum_j p[i+j] * v[j] = (v_rev * p)[n-1+i]
    where v_rev[j] = v[n-1-j].
    """
    n = len(v)
    v_rev = v[::-1]
    N = _next_pow2(n + len(p) - 1)
    fv = np.fft.rfft(v_rev, N)
    fp = np.fft.rfft(p, N)
    r = np.fft.irfft(fv * fp, N)
    return r[n-1:2*n-1]  # Wait, this gives length n, indices n-1 to 2n-2


def _proj_simplex(z):
    """Project z onto the simplex {a >= 0, sum a = 1}."""
    n = len(z)
    # Sort in descending order
    u = np.sort(z)[::-1]
    # Find the largest m such that u[m-1] > (sum(u[0:m]) - 1) / m
    cssv = np.cumsum(u)
    cond = u > (cssv - 1.0) / np.arange(1, n + 1)
    rho = int(np.sum(cond))
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(z - theta, 0.0)


def _fista_qp(p, a_init, n_iters=500, L=None):
    """Solve min a^T H_p a subject to a >= 0, sum a = 1 using FISTA.
    H_p is the Hankel matrix with (H_p)_{ij} = p_{i+j}.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    a_prev = a.copy()
    
    if L is None:
        # Estimate L = 2 * lambda_max(H_p)
        # Upper bound: 2 * max_j sum_i p_{i+j} = 2 * max row sum
        # Row i: sum_j p_{i+j} for j=0..n-1
        # This is at most sum_j p_j = 1
        L = 2.0  # Conservative upper bound
    
    t = 1.0
    for it in range(n_iters):
        # Momentum
        y = a + ((t - 1) / (t + 2)) * (a - a_prev)
        # Gradient: 2 * H_p @ y
        Hy = _hankel_matvec(y, p)
        grad = 2.0 * Hy
        # Gradient step
        a_new = y - (1.0 / L) * grad
        # Project onto simplex
        a_new = _proj_simplex(a_new)
        # Update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_prev = a
        a = a_new
        t = t_new
    
    return a


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    Primal: min_a max_k c_k(a) s.t. sum a = 1, a >= 0
    Dual:   max_p min_a sum_k p_k c_k(a) s.t. p >= 0, sum p = 1

    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score(a)
    
    # Initialize p from the autocorrelation of a
    c = _full_convolve(a, a)
    beta = 1.0
    p = _softmax(beta * c)
    
    if beta_schedule is None:
        # Gradually increase beta
        n_outer = min(max_outer, 100)
        beta_schedule = np.logspace(0, 4, n_outer)  # from 1 to 10000
        # Actually, let's use a more moderate schedule

            np.linspace(1, 10, 20),
            np.linspace(10, 100, 30),
            np.linspace(100, 1000, 50),
        ])
    
    for outer_it, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline - 10:
            break
        
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        # How many FISTA iterations can we afford?
        est_time_per_iter = 0.002 * (n / 4096.0)  # Rough estimate
        max_fista = min(500, max(20, int(remaining * 0.8 / (len(beta_schedule) - outer_it) / est_time_per_iter)))
        max_fista = min(max_fista, fista_iters_per_outer)
        
        # Solve inner QP

3. Optimize the fine sequence subject to this constraint
4. Project the fine solution back to the coarse level
5. Update the coarse sequence
6. Repeat

- Lagrange multipliers (add a penalty for violating the constraint)
- Projection (project onto the affine subspace defined by the block sum constraints)
- Alternating projections (alternate between projecting onto the simplex and projecting onto the affine constraint)

- B is an m×n_f matrix where B_{r,j} = 1/n_c if j is in block r, else 0

    return r[n - 1:2 * n - 1]


def _proj_simplex(z):
    """Project z onto the probability simplex {a >= 0, sum a = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - 1.0) / idx
    rho = int(np.sum(cond))
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(z - theta, 0.0)


def _fista_qp(p, a_init, n_iters=300, L=None, deadline=None):
    """Solve min a^T H_p a s.t. a >= 0, sum a = 1 using FISTA.
    H_p is the n x n Hankel matrix with (H_p)_{ij} = p_{i+j}.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    a_prev = a.copy()

    if L is None:
        L = 2.0

    t = 1.0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        y = a + ((t - 1.0) / (t + 2.0)) * (a - a_prev)
        Hy = _hankel_matvec(y, p)
        grad = 2.0 * Hy
        a_new = y - (1.0 / L) * grad
        a_new = _proj_simplex(a_new)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_prev = a
        a = a_new
        t = t_new
    return a


def _dual_minimax(a_init, deadline, max_outer=80, fista_iters=300):
    """Dual ascent on the minimax problem.
    Primal: min_a max_k c_k(a) s.t. sum a = 1, a >= 0
    Dual:   max_p min_a sum_k p_k c_k(a) s.t. p >= 0, sum p = 1
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    betas = np.concatenate([
        np.linspace(0.5, 5.0, 15),
        np.linspace(5.0, 50.0, 25),
        np.linspace(50.0, 500.0, 40),
    ])

    for outer_it, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        n_outer_remaining = len(betas) - outer_it
        per_iter_time = remaining / max(1, n_outer_remaining)
        fista_n = min(fista_iters, max(30, int(per_iter_time / 0.005)))

        a = _fista_qp(p, a, n_iters=fista_n, deadline=deadline)
        c = _full_convolve(a, a)
        p = _softmax_beta(c, beta)

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if outer_it % 20 == 0:
            print(f"    Dual outer={outer_it}, beta={beta:.1f}, "
                  f"score={sc:.10f}, max_c={c.max():.4e}")

    return best_a, best_sc


def _softmax_beta(z, beta):
    z_shifted = beta * z
    z_shifted = z_shifted - z_shifted.max()
    e = np.exp(np.clip(z_shifted, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _coarse_constraint_project(a_fine, a_coarse, n_blocks):
    """Project a_fine onto the affine subspace where block means match a_coarse.
    a_fine has length n_blocks * len(a_coarse).
    """
    n_c = len(a_coarse)
    n_f = len(a_fine)
    block_size = n_f // n_c
    a_proj = a_fine.copy()
    for r in range(n_c):
        start = r * block_size
        end = start + block_size
        block_mean = a_proj[start:end].mean()
        a_proj[start:end] += (a_coarse[r] - block_mean)
    return a_proj


def _fista_constrained(p, a_init, a_coarse, n_iters=200, L=None, deadline=None):
    """FISTA with alternating projection onto simplex and block-sum constraint."""
    n = len(a_init)
    n_c = len(a_coarse)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    a_prev = a.copy()

    if L is None:
        L = 2.0

    t = 1.0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        y = a + ((t - 1.0) / (t + 2.0)) * (a - a_prev)
        Hy = _hankel_matvec(y, p)
        grad = 2.0 * Hy
        a_new = y - (1.0 / L) * grad
        a_new = _proj_simplex(a_new)
        a_new = _coarse_constraint_project(a_new, a_coarse, n_c)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = _proj_simplex(a_new)
        else:
            a_new = a_new / s
            a_new = _coarse_constraint_project(a_new, a_coarse, n_c)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_prev = a
        a = a_new
        t = t_new
    return a


def _multi_res_optimize(a_coarse, n_fine, deadline, min_remaining=60):
    """Fine optimization with coarse constraint, then release constraint."""
    n_c = len(a_coarse)
    m = n_fine // n_c
    if m < 2:
        return a_coarse, _score(a_coarse)

    # Upsample coarse to fine
    x_coarse = np.linspace(0, 1, n_c)
    x_fine = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_fine, x_coarse, a_coarse)
    a_fine = np.maximum(a_fine, 0.0)
    a_fine = a_fine / a_fine.sum()

    # Phase A: Optimize with constraint
    best_a = a_fine.copy()
    best_sc = _score(a_fine)

    c = _full_convolve(a_fine, a_fine)
    p = np.ones(2 * n_fine - 1) / (2 * n_fine - 1)

    betas = np.concatenate([
        np.linspace(1.0, 10.0, 10),
        np.linspace(10.0, 100.0, 20),
        np.linspace(100.0, 500.0, 30),
    ])

    for outer_it, beta in enumerate(betas):
        if time.monotonic() > deadline - min_remaining:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        n_rem = len(betas) - outer_it
        per_iter = remaining / max(1, n_rem)
        fista_n = min(200, max(20, int(per_iter / 0.01)))

        a_fine = _fista_constrained(p, a_fine, a_coarse, n_iters=fista_n,
                                     deadline=deadline)
        c = _full_convolve(a_fine, a_fine)
        p = _softmax_beta(c, beta)

        sc = _score(a_fine)
        if sc < best_sc:
            best_sc = sc
            best_a = a_fine.copy()

        if outer_it % 15 == 0:
            print(f"      Multi-res constrained outer={outer_it}, "
                  f"beta={beta:.1f}, score={sc:.10f}")

    # Phase B: Release constraint, continue dual minimax
    if time.monotonic() < deadline - min_remaining * 0.5:
        a_fine = best_a.copy()
        c = _full_convolve(a_fine, a_fine)
        p = _softmax_beta(c, 100.0)

        betas2 = np.linspace(100.0, 1000.0, 30)
        for outer_it, beta in enumerate(betas2):
            if time.monotonic() > deadline - min_remaining * 0.3:
                break
            remaining = deadline - min_remaining * 0.3 - time.monotonic()
            if remaining < 2:
                break
            fista_n = min(200, max(20, int(remaining / 0.01)))
            a_fine = _fista_qp(p, a_fine, n_iters=fista_n, deadline=deadline)
            c = _full_convolve(a_fine, a_fine)
            p = _softmax_beta(c, beta)
            sc = _score(a_fine)
            if sc < best_sc:
                best_sc = sc
                best_a = a_fine.copy()

    return best_a, best_sc


def _lbfgs_polish(a_init, deadline, min_remaining=30):
    """L-BFGS polish with log-sum-exp smooth approximation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 2:
            break
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 10))))

        a_new = _lbfgs_single(z, beta, n, max_iter, deadline, min_remaining)
        if a_new is not None:
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()

    return current_a


def _lbfgs_single(z, beta, n, max_iter, deadline, min_remaining):
    """Single L-BFGS run at a given beta."""
    a = _softmax_logit(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return None
    obj_start = 2.0 * n * (c_max + beta * np.log(s))

    def fun_local(z_local):
        a_loc = _softmax_logit(z_local)
        c_loc = _full_convolve(a_loc, a_loc)
        cm = float(c_loc.max())
        sh = (c_loc - cm) / beta
        el = np.exp(np.clip(sh, -700, 0))
        sl = el.sum()
        if sl < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(sl))

    def jac_local(z_local):
        a_loc = _softmax_logit(z_local)
        c_loc = _full_convolve(a_loc, a_loc)
        cm = float(c_loc.max())
        sh = (c_loc - cm) / beta
        el = np.exp(np.clip(sh, -700, 0))
        sl = el.sum()
        if sl < 1e-300:
            w = np.zeros_like(c_loc)
            w[int(np.argmax(c_loc))] = 1.0
        else:
            w = el / sl
        a_rev = a_loc[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c_loc))
        return 4.0 * n * a_loc * (h - W)

    try:
        result = minimize(
            fun_local, z, jac=jac_local, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxcor': 50, 'maxls': 80}
        )
        a_new = _softmax_logit(result.x)
        return a_new
    except Exception:
        return None


def _softmax_logit(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.015):
    """Subgradient descent on the active-set gradient of max autocorrelation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax_logit(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax_logit(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax_logit(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _spectral_reshape(a, deadline):
    """Try to flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    power = np.abs(A) ** 2

    # Try to equalize the power spectrum (reduce dynamic range)
    for gamma in [0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() > deadline:
            break
        p_min = power.min()
        p_max = power.max()
        if p_max - p_min < 1e-10:
            continue
        # Compress dynamic range
        power_new = p_min + (p_max - p_min) * ((power - p_min) / (p_max - p_min)) ** gamma
        # Preserve DC
        power_new[0] = power[0]
        A_new = np.sqrt(power_new)
        A_new[0] = A[0] if np.isreal(A[0]) else A[0]
        # Keep phase
        phase = np.angle(A)
        A_new = A_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    Spectral reshape gamma={gamma}: score={sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-resolution dual minimax optimizer with embedded constraint projection.
    Combines FFT-accelerated QP solving, coarse-to-fine constraint projection,
    spectral reshaping, and L-BFGS/subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Coarse optimization ===
    print(f"\n[Phase 1] Coarse optimization (t={time.monotonic() - t_start:.1f}s)")
    n_coarse = 1024
    if time.monotonic() < deadline - 1200:
        if best_seq is not None and len(best_seq) > n_coarse:
            x_f = np.linspace(0, 1, len(best_seq))
            x_c = np.linspace(0, 1, n_coarse)
            a_c = np.interp(x_c, x_f, best_seq)
            a_c = np.maximum(a_c, 0.0)
            a_c = a_c / a_c.sum()
        else:
            a_c = np.ones(n_coarse) / n_coarse

        coarse_deadline = min(deadline - 1200, time.monotonic() + 180)
        a_c_opt, sc_c = _dual_minimax(a_c, coarse_deadline, max_outer=60, fista_iters=200)
        update_best(a_c_opt, sc_c, f"coarse_{n_coarse}")

        # Also try n=2048
        if time.monotonic() < deadline - 1100:
            n_c2 = 2048
            if best_seq is not None and len(best_seq) > n_c2:
                x_f = np.linspace(0, 1, len(best_seq))
                x_c = np.linspace(0, 1, n_c2)
                a_c2 = np.interp(x_c, x_f, best_seq)
                a_c2 = np.maximum(a_c2, 0.0)
                a_c2 = a_c2 / a_c2.sum()
            else:
                a_c2 = np.ones(n_c2) / n_c2
            coarse_deadline2 = min(deadline - 1100, time.monotonic() + 120)
            a_c2_opt, sc_c2 = _dual_minimax(a_c2, coarse_deadline2, max_outer=50, fista_iters=150)
            update_best(a_c2_opt, sc_c2, f"coarse_{n_c2}")

    # === Phase 2: Multi-resolution with constraint projection ===
    print(f"\n[Phase 2] Multi-res constraint projection (t={time.monotonic() - t_start:.1f}s)")
    n_fine = 8192
    if best_seq is not None and time.monotonic() < deadline - 600:
        # Use the best coarse solution as constraint
        # Find best coarse (from phase 1 or downsample of best_seq)
        if len(best_seq) > n_coarse:
            x_f = np.linspace(0, 1, len(best_seq))
            x_c = np.linspace(0, 1, n_coarse)
            a_coarse_for_constraint = np.interp(x_c, x_f, best_seq)
            a_coarse_for_constraint = np.maximum(a_coarse_for_constraint, 0.0)
            a_coarse_for_constraint = a_coarse_for_constraint / a_coarse_for_constraint.sum()
        else:
            a_coarse_for_constraint = best_seq.copy()

        mr_deadline = min(deadline - 400, time.monotonic() + 500)
        a_mr, sc_mr = _multi_res_optimize(a_coarse_for_constraint, n_fine, mr_deadline,
                                          min_remaining=200)
        update_best(a_mr, sc_mr, "multi_res")

    # === Phase 3: Direct dual minimax at full resolution ===
    print(f"\n[Phase 3] Full-res dual minimax (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        dm_deadline = min(deadline - 300, time.monotonic() + 400)
        a_dm, sc_dm = _dual_minimax(best_seq, dm_deadline, max_outer=80, fista_iters=250)
        update_best(a_dm, sc_dm, "dual_full")

    # === Phase 4: Spectral reshape ===
    print(f"\n[Phase 4] Spectral reshape (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        sr_deadline = min(deadline - 200, time.monotonic() + 100)
        a_sr, sc_sr = _spectral_reshape(best_seq, sr_deadline)
        update_best(a_sr, sc_sr, "spectral")
        if sc_sr < best_score:
            # Polish the spectral result
            if time.monotonic() < deadline - 150:
                a_sr2 = _lbfgs_polish(a_sr, deadline, min_remaining=100)
                update_best(a_sr2, _score(a_sr2), "spectral_lbfgs")

    # === Phase 5: L-BFGS polish ===
    print(f"\n[Phase 5] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        lf_deadline = min(deadline - 80, time.monotonic() + 200)
        a_lf = _lbfgs_polish(best_seq, lf_deadline, min_remaining=60)
        update_best(a_lf, _score(a_lf), "lbfgs")

    # === Phase 6: Subgradient polish ===
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sg_deadline = min(deadline - 30, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_steps=5000, lr_init=0.01)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_restarts = 3
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            n = len(best_seq)
            if i == 0:
                # Small random perturbation
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z + 0.02 * rng.standard_normal(n)
                z = z - z.mean()
                a_r = _softmax_logit(z)
            elif i == 1:
                # Fourier perturbation
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z - z.mean()
                t = np.arange(n)
                for _ in range(3):
                    freq = rng.integers(1, max(2, n // 5))
                    amp = 0.02 * rng.uniform(1, 3)
                    phase = rng.uniform(0, 2 * np.pi)
                    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
                z = z - z.mean()
                a_r = _softmax_logit(z)
            else:
                # Block swap
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z - z.mean()
                block = n // 8
                i1 = rng.integers(0, n - 2 * block)
                i2 = rng.integers(0, n - 2 * block)
                if abs(i1 - i2) > block:
                    tmp = z[i1:i1 + block].copy()
                    z[i1:i1 + block] = z[i2:i2 + block]
                    z[i2:i2 + block] = tmp
                a_r = _softmax_logit(z)

            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / a_r.sum()
            # Quick dual minimax from perturbed start
            r_deadline = min(deadline - 15, time.monotonic() + 60)
            a_r_opt, sc_r = _dual_minimax(a_r, r_deadline, max_outer=30, fista_iters=100)
            update_best(a_r_opt, sc_r, f"restart_{i}")

    # === Phase 8: Final ultra-low beta L-BFGS ===
    print(f"\n[Phase 8] Final ultra polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0005
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            a_fin = _lbfgs_single(z, beta_final, n, max_iter, deadline, 3)
            if a_fin is not None:
                update_best(a_fin, _score(a_fin), "final_ultra")

        # One more subgradient pass
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 3, n_steps=2000, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
