def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _self_convolve(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_matvec(v, A_idx, n, m):
    """Compute g[i] = sum_{j in A} v[i+j] + v[|i-j|] style operation via FFT."""
    # For active set A (indices in [0, 2n-2]), compute:
    # (H v)[i] = sum_{j in A} (v[j-i] + v[i-j]) with v[m]=0 for m outside [0,n-1]
    # This equals: sum_k v[k] * A_count[i+k] + sum_k v[k] * A_count[|i-k|]
    # where A_count is the indicator of the active set (as a convolution kernel)
    # Actually: (H v)[i] = sum_{j in A} v[j-i] (if 0<=j-i<n) + sum_{j in A} v[i-j] (if 0<=i-j<n)
    # First sum: let k = j-i, so j = k+i, range: k from 0 to min(n-1, 2n-2-i)
    #   = sum_k v[k] * I[k+i in A] = sum_k v[k] * A_ext[k+i]
    # Second sum: let k = i-j, so j = i-k, range: k from 0 to min(i, n-1)
    #   = sum_k v[k] * I[i-k in A] = sum_k v[k] * A_ext[i-k]
    # Combined: (Hv)[i] = sum_k v[k] * (A_ext[k+i] + A_ext[i-k])
    # This is: (Hv)[i] = (v * A_ext)[i] + (v * A_ext_flip)[i] ... 
    # Actually it's a correlation: (Hv)[i] = sum_k v[k] * B[i+k] + sum_k v[k] * B[i-k]
    # where B = A_ext (the active indicator on [0, 2n-2])
    # First term: sum_k v[k]*B[i+k] -> this is cross-correlation of v and B, evaluated at lag i
   correo = (second term B[i-k])
    # Let me just use the same trick as the gradient computation:
    # Create B_ext = A_ext (length 2n-1, padded)
    # (Hv)[i] = sum_j B[j] * (v[j-i] + v[i-j])
    # = sum_j B[j]*v[j-i] + sum_j B[j]*v[i-j]
    # First: let the kernel be B, signal be v. sum_j B[j]*v[j-i] = (flip(B) * v)[n-1-... 
    # I'll just use the same FFT trick:
    v_rev = v[::-1]
    # sum_j B[j]*v[j-i]: let m = j-i, j = m+i: sum_m B[m+i]*v[m]
    # This is the "sliding dot product" of B (shifted) with v.
    # = (B * flip(v))[2n-2-... hmm.
    # Let me use: conv(B, v_rev) gives C[m] = sum_j B[j]*v_rev[m-j] = sum_j B[j]*v[n-1-m+j]
    # C[n-1+i] = sum_j B[j]*v[i+j]  -- this is "forward" correlation
    # C[n-1-i] = sum_j B[j]*v[-i+j] = sum_j B[j]*v[j-i] -- this is "backward" correlation
    # So (Hv)[i] = C[n-1+i] + C[n-1-i] (with bounds)
    
    out_len = (2 * n - 1) + n - 1  # B has length 2n-1, v_rev has length n
    N = _next_pow2(out_len)
    fB = np.fft.rfft(A_idx, N)
    fv = np.fft.rfft(v_rev, N)
    C = np.fft.irfft(fB * fv, N)[:out_len]
    # C[m] = sum_j A_idx[j] * v_rev[m-j] = sum_j A_idx[j] * v[n-1-m+j]
    # C[n-1+i] = sum_j A_idx[j] * v[i+j]  (for i+j in [0, 2n-2] and n-1-m+j in [0,n-1])
    # C[n-1-i] = sum_j A_idx[j] * v[j-i]
    # So (Hv)[i] = C[n-1+i] + C[n-1-i]
    hv = C[n - 1:n - 1 + n] + C[n - 1 - (n - 1):n - 1 - (n - 1) + n][::-1]
    # Wait, C[n-1-i] for i=0..n-1 is C[n-1], C[n-2], ..., C[0], which is C[:n][::-1]
    hv = C[n - 1:2 * n - 1] + C[n - 1::-1][:n]
    return hv

def _coord_descent(a_init, deadline, max_rounds=20):
    """Coordinate descent: optimize pairs (i, ref) to reduce max convolution."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _self_convolve(a)
        c_max = float(c.max())
        # Try optimizing coordinates that are at or near the "peak"
        peak_idx = np.argsort(c)[::-1][:min(100, 2 * n)]
        # For each peak position j, the contributing pair is (i, j-i) 
        # Try adjusting a[i] for i that contribute to the peak
        candidates = set()
        for j in peak_idx[:50]:
            j = int(j)
            for i in range(n):
                if 0 <= j - i < n:
                    candidates.add(i)
                    break
        # Also add random coordinates
        for _ in range(min(50, n // 10)):
            candidates.add(int(rng.integers(n)))
        
        improved = False
        for i in list(candidates)[:200]:
            if time.monotonic() > deadline:
                break
            # Try different values of a[i], compensating with a uniform adjustment
            # of all other elements
            base_sum = a.sum() - a[i]
            if base_sum < 1e-15:
                continue
            # a[i] new value: try several
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_ai = a[i] * scale
                if new_ai < 0:
                    continue
                a_try = a.copy()
                a_try[i] = new_ai
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-12:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

        if not improved and rnd > 3:
            break

    return best_a, best_sc


def _shrinkage_polish(a, deadline):
    """Try zeroing out smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.10, 0.15]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp_left":
        t = np.linspace(0, 1, n)
        a = (1.0 - t) ** 2 + 0.01
        return a / a.sum()
    elif kind == "ramp_right":
        t = np.linspace(0, 1, n)
        a = t ** 2 + 0.01
        return a / a.sum()
    elif kind == "gaussian_offcenter":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + 1e-6
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.5) ** 2) + 1e-6
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = (1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t)) ** 2 + 1e-6
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _downsample(a, n_new):
    """Downsample by averaging blocks."""
    n_old = len(a)
    if n_new >= n_old:
        return _interpolate_to_n(a, n_new)
    block_size = n_old // n_new
    if block_size < 1:
        return _interpolate_to_n(a, n_new)
    a_blocks = a[:block_size * n_new].reshape(n_new, block_size)
    a_new = a_blocks.mean(axis=1)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_structured(a, rng, n_perturb=5):
    """Add structured sinusoidal perturbations."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_perturb):
        freq = rng.integers(1, max(2, n // 8))
        amp = rng.uniform(0.01, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent loaded: n={n_parent}")
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        update_best(parent, sc_parent, "parent")
        print(f"  Parent score: {sc_parent:.10f}")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # Phase 1: L-BFGS at parent's n
    print(f"\n[Phase 1] L-BFGS at n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a1 = _run_lbfgs(parent, deadline, min_time_remaining=250.0, use_active=True)
        update_best(a1, _score(a1), f"lbfgs_n{n_parent}")

    # Phase 2: Active-set Newton at parent's n
    print(f"\n[Phase 2] Active-set Newton at n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as_dl = min(deadline - 180, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=300, tol_active=0.001)
        update_best(a_as, sc_as, f"newton_n{n_parent}")
        # Tighter
        if time.monotonic() < deadline - 100:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 80, time.monotonic() + 80),
                                                max_iter=200, tol_active=0.0005)
            update_best(a_as2, sc_as2, f"newton_tight_n{n_parent}")

    # Phase 3: Try n=16384 (interpolate from best at n_parent)
    print(f"\n[Phase 3] L-BFGS at n=16384 (t={time.monotonic() - t_start:.1f}s)")
    n_large = 16384
    if time.monotonic() < deadline - 200:
        if best_seq is not None:
            a_large_init = _interpolate_to_n(best_seq, n_large)
        else:
            a_large_init = _make_profile(n_large, "uniform")
        
        a_large = _run_lbfgs(a_large_init, deadline, min_time_remaining=150.0, use_active=True)
        update_best(a_large, _score(a_large), f"lbfgs_n{n_large}")

        # Newton on large
        if time.monotonic() < deadline - 100:
            a_large_as, sc_large_as = _active_set_newton(a_large, min(deadline - 80, time.monotonic() + 100),
                                                         max_iter=300, tol_active=0.001)
            update_best(a_large_as, sc_large_as, f"newton_n{n_large}")

    # Phase 4: Shrinkage
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_shr, sc_shr = _shrinkage_polish(best_seq, deadline - 60)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score and time.monotonic() < deadline - 40:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=30.0, use_active=True)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # Phase 5: Coordinate descent
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cd, sc_cd = _coord_descent(best_seq, deadline - 20, max_rounds=30)
        update_best(a_cd, sc_cd, "coord_descent")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 25:
        restarts = [
            ("struct_perturb_1", lambda: _perturb_structured(best_seq, rng, n_perturb=3)),
            ("struct_perturb_2", lambda: _perturb_structured(best_seq, rng, n_perturb=6)),
            ("gaussian_offcenter", lambda: _make_profile(len(best_seq), "gaussian_offcenter")),
            ("ramp_left", lambda: _make_profile(len(best_seq), "ramp_left")),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 15:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0, use_active=True)
            update_best(a_r, _score(a_r), f"restart_{label}")
            if _score(a_r) < best_score + 0.01 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 5, time.monotonic() + 30),
                                                  max_iter=100, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_newton")

    # Phase 7: Final intensive Newton
    print(f"\n[Phase 7] Final Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 2, max_iter=500, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_newton")
        
        # One more L-BFGS with tiny beta
        if time.monotonic() < deadline - 2:
            c_b = _self_convolve(best_seq)
            mb = float(c_b.max())
            n_best = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda zl: _ls_objective_full(zl, beta_final, n_best),
                        z,
                        jac=lambda zl: _ls_gradient_full(zl, beta_final, n_best),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxcor': 100}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs")
                except Exception:
                    pass

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
