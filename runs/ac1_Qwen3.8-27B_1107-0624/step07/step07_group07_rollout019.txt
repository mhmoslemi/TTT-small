- For each peak at position k, find the elements a[i] that contribute most to it (i.e., where a[i]*a[k-i] is large)
- Reduce those elements slightly and redistribute the mass to elements that don't contribute to any peak

This is a greedy approach but might work well for the final push.

If a is the optimal sequence, then reverse(a) = [a[n-1], ..., a[0]] is also optimal (the problem is symmetric under reversal, since (reverse(a) * reverse(a))[k] = (a*a)[2n-2-k]).

What about the "folded" sequence: fold(a)[i] = (a[i] + a[n-1-i]) / 2?
The convolution of fold(a) with itself is NOT the same as the average of the convolutions. But fold(a) is symmetric, and for symmetric sequences, the max of the convolution is at the center (position n-1), which equals sum a[i]^2.

So folding would likely make things worse (since it makes the sequence more symmetric, increasing the central peak).

What about the "anti-folded" sequence: anti(a)[i] = (a[i] - a[n-1-i]) / 2?
This would be antisymmetric and have a[center] = 0. But it might have negative values, which isn't allowed.

So the optimal sequence is likely NOT symmetric, which is why we can get below 2.

The convolution of b with b will have the same "shape" as a*a but stretched. The max will be the same (approximately), and the sum will be the same, so the score will be 2*m*max_c/S^2 which is worse (larger m).

So upscaling doesn't help directly. But it could help as an intermediate step in optimization.

"""
Step sequence optimizer v2: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
adaptive L-BFGS with peak-redirecting local search, simulated annealing
restarts, and frequency-domain perturbation. Key improvements: direct
peak-redirecting steps that target specific convolution highs, adaptive
beta continuation with progress detection, and diverse restart strategies.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _convolve_a(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_a(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_direct(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _project_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones_like(v) / len(v)
    return v / s


def _peak_redirect(a, n, deadline, max_iters=20000, top_k=5):
    a = a.copy()
    a = _project_simplex(a)
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.05
    stall = 0
    stall_limit = 50

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _convolve_a(a)
        c_max = c.max()

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]
        thr = c[active[-1]] if len(active) > 1 else c_max * 0.95
        active_mask = c >= (c_max - 0.02 * c_max)
        if active_mask.sum() == 0:
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active_mask] = 1.0 / active_mask.sum()

        g = _grad_direct(a, c, n, w)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        best_eta = eta
        best_a_try = a
        best_sc_try = best_sc

        for frac in [1.0, 0.7, 0.5, 0.3, 0.15]:
            step = eta * frac
            a_try = a - step * g
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-15:
                best_sc_try = sc_try
                best_a_try = a_try
                best_eta = step

        if best_sc_try < best_sc - 1e-14:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.15)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-9 or stall > stall_limit:
                break

    return best_a, best_sc


def _targeted_peak_reduce(a, n, deadline, max_rounds=5000):
    a = a.copy()
    a = _project_simplex(a)
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _convolve_a(a)
        k_peak = int(np.argmax(c))
        c_max = c[k_peak]

        contrib = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = a[j]
            j2 = i - k_peak
            if 0 <= j2 < n:
                contrib[i] += a[j2]

        pos_mask = contrib > 1e-10
        if not pos_mask.any():
            break

        mass_to_remove = 0.01 * a.sum()
        remove_amt = mass_to_remove * contrib[pos_mask]
        remove_amt = remove_amt / remove_amt.sum() * mass_to_remove
        remove_vec = np.zeros(n)
        remove_vec[pos_mask] = remove_amt

        a_try = a - remove_vec
        a_try = _project_simplex(a_try)
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            a_half = a - 0.5 * remove_vec
            a_half = _project_simplex(a_half)
            sc_half = _score(a_half)
            if sc_half < best_sc - 1e-14:
                best_sc = sc_half
                best_a = a_half.copy()
                a = a_half
            else:
                return best_a, best_sc

    return best_a, best_sc


def _freq_perturb(a, rng, n_modes=8, scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _gauss_perturb(a, rng, scale=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _lbfgs_smooth(z0, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 120,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        a_out = _softmax(z0)
        return a_out, False, 0


def _lbfgs_continuation(a_start, deadline, reserve_s=120.0):
    n = len(a_start)
    c_cur = _convolve_a(a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.5, 0.25, 0.12, 0.06, 0.03,
                              0.015, 0.008, 0.004, 0.002, 0.001,
                              5e-4, 2.5e-4, 1.2e-4, 6e-5, 3e-5,
                              1.5e-5, 7e-6, 3.5e-6, 1.8e-6, 9e-7])

    current_a = a_start.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-15))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_s > deadline:
            break
        remaining = deadline - reserve_s - time.monotonic()
        if remaining < 20:
            break

        scale = n / 4096.0
        est_time_per_iter = 0.003 * scale ** 1.3
        max_iter = min(15000, max(300, int(remaining / (est_time_per_iter * 25))))

        a_new, success, niter = _lbfgs_smooth(z, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            z = np.log(np.maximum(current_a, 1e-15))
            z = z - z.mean()
            print(f"    beta={beta:.3e}: score={sc:.10f} (iter={niter}, ok={success})")
        else:
            z = np.log(np.maximum(current_a, 1e-15))
            z = z - z.mean()

    return current_a


def _pair_swap_search(a_init, deadline, max_rounds=30000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng_local = np.random.default_rng(12345)

    imp_count = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        i = int(rng_local.integers(0, n))
        j = int(rng_local.integers(0, n))
        if i == j:
            continue

        s_pair = a[i] + a[j]
        if s_pair < 1e-15:
            continue

        best_frac = 0.5
        best_sc_try = best_sc
        best_a_try = a

        for frac in [0.5, 0.3, 0.7, 0.2, 0.8, 0.1, 0.9, 0.4, 0.6]:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[i] = new_i
            a_try[j] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-14:
                best_sc_try = sc_try
                best_frac = frac
                best_a_try = a_try

        if best_sc_try < best_sc - 1e-14:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            imp_count += 1
        elif imp_count > 200 and rnd - imp_count > 5000:
            break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _block_perturb(a, rng, n_blocks=6, block_scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    block_size = max(4, n // (2 * n_blocks))
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)
        z[start:end] += block_scale * rng.uniform(-1, 1)
    z = z - z.mean()
    return _softmax(z)


def _anneal_restart(best_a, rng, deadline, reserve_s=60.0, n_restarts=5):
    n = len(best_a)
    best_sc = _score(best_a)
    best = best_a.copy()
    t_start = time.monotonic()

    for r in range(n_restarts):
        if time.monotonic() + reserve_s > deadline:
            break
        remaining = deadline - reserve_s - time.monotonic()
        if remaining < 30:
            break

        scale = 0.03 + 0.04 * (r / max(1, n_restarts - 1))
        mode = r % 3
        if mode == 0:
            perturbed = _gauss_perturb(best, rng, scale)
        elif mode == 1:
            perturbed = _freq_perturb(best, rng, n_modes=6, scale=scale * 0.8)
        else:
            perturbed = _block_perturb(best, rng, n_blocks=4, block_scale=scale)

        sc_start = _score(perturbed)

        a_opt = _lbfgs_continuation(perturbed, deadline - reserve_s + 30,
                                     reserve_s=reserve_s)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc - 1e-14:
            best_sc = sc_opt
            best = a_opt.copy()
            print(f"    Anneal restart {r}: improved to {best_sc:.10f}")

        a_ps, sc_ps = _peak_redirect(a_opt, n, deadline - reserve_s + 15,
                                     max_iters=5000, top_k=4)
        if sc_ps < best_sc - 1e-14:
            best_sc = sc_ps
            best = a_ps.copy()
            print(f"    Anneal restart {r} + PS: improved to {best_sc:.10f}")

    return best, best_sc

        if time.monotonic() < deadline - 5:
            a_f3, _ = _pair_swap_search(best_seq, deadline - 5, max_rounds=10000)
            try_update(a_f3, f"final_swap_{cycle}")

    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
