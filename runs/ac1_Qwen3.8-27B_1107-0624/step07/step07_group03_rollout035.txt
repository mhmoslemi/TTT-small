import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)

    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = base_beta
    patience = 0
    max_patience = 5

    while time.monotonic() + min_time_remaining < deadline:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.004 * max(1, n / 4096.0)
        max_iter = min(6000, max(100, int(remaining / (est_time * 10))))

        z_old = z.copy()
        sc_old = current_sc

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < sc_old - 1e-14:
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                current_a = a_new
                current_sc = sc_new
                patience = 0
                beta *= 0.5
            else:
                patience += 1
                if patience >= max_patience:
                    z = z_old.copy()
                    z = z + 0.05 * np.random.default_rng(int(time.monotonic() * 1e6) % 100000).standard_normal(n)
                    z = z - z.mean()
                    patience = 0
                    beta *= 0.3
                else:
                    beta *= 0.6
        except Exception:
            beta *= 0.5

    return current_a

            for k in idx:
                if 0 <= k - j <= n - 1:
                    d[k] = 2.0 * a[k - j]
                if k == 2 * j and 0 <= j <= n - 1:
                    d[k] += a[j]
            # c[k] new = c[k] + a[j]*d[k] - a[j]^2 (if k==2j, the a[j]^2 was already in c)
            # Actually: c[k] = c_base[k] + 2*a[j]*a[k-j] (for k!=2j) or c_base_excl + a[j]^2 (for k=2j)
            # Let me just compute the max as a function of a[j] = t
            # c[k](t) = c[k] - d[k]*a[j] + d[k]*t for k != 2j (d[k] = 2*a[k-j])
            # c[2j](t) = c[2j] - a[j]^2 + t^2 (the a[j]^2 term)
            # Actually simpler: just try a few values
            old_val = a[j]
            # Try small perturbations
            candidates = []
            for frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5, 1.0, 2.0]:
                t = old_val * (1 + frac) if old_val > 0 else 0.01 / n
                t = max(0, t)
                candidates.append(t)
            # Also try the value that equalizes with c_max
            # c[k] + d[k]*(t - old_val) = c_max => t = old_val + (c_max - c[k])/d[k]
            for k in range(2 * n - 1):
                if d[k] != 0:
                    t = old_val + (c_max - c[k]) / d[k]
                    if t > 0:
                        candidates.append(min(t, 10.0 / n))

            best_t = old_val
            best_k_max = c_max
            for t in candidates:
                t = max(0.0, min(t, 5.0))
                a[j] = t
                s_try = a.sum()
                if s_try < 1e-15:
                    a[j] = old_val
                    continue
                a_norm = a / s_try
                c_try = _full_convolve(a_norm, a_norm)
                k_max_try = float(c_try.max())
                if k_max_try < best_k_max:
                    best_k_max = k_max_try
                    best_t = t
            a[j] = best_t
            if abs(best_t - old_val) > 1e-10:
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        if not improved:
            break

    return best_a, best_sc


def _freq_init(n, rng):
    """Generate a non-negative sequence with approximately flat spectrum."""
    N = n
    # Random phases with conjugate symmetry
    phases = rng.standard_normal(N // 2 + 1)
    A = np.zeros(N // 2 + 1, dtype=np.complex128)
    A[0] = np.sqrt(n)  # DC component large enough for non-negativity
    A[1:] = np.exp(1j * phases[1:]) * (np.sqrt(n) * 0.5)
    # Handle Nyquist if N is even
    if N % 2 == 0:
        # A[N//2] is real for real signal
        pass

    # Build full spectrum
    A_full = np.zeros(N, dtype=np.complex128)
    A_full[:N // 2 + 1] = A
    for k in range(1, N // 2 + (1 if N % 2 == 0 else 0)):
        if k < N - k:
            A_full[N - k] = np.conj(A[k])
    if N % 2 == 0:
        A_full[N // 2] = np.real(A_full[N // 2])

    a = np.fft.ifft(A_full).real
    # Make non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                                  0.00002, 0.00001, 0.000005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
