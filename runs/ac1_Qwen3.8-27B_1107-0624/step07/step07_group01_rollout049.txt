Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive L-BFGS with aggressive log-sum-exp smoothing
2. Momentum-based subgradient polishing on the simplex
3. Diverse multi-start with structured and random initializations
4. Multi-length exploration (n=8192 to 32768)
5. Final micro-polishing with very tight beta

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_and_grad(z, beta, n):
    """Compute smoothed objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _lbfgs_step(a_init, beta, n, max_iter=5000):
    """Run L-BFGS on the smoothed problem."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        obj, _ = _smoobj_and_grad(z, beta, n)
        return obj

    def jac(z):
        _, g = _smoobj_and_grad(z, beta, n)
        return g

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

    n = len(a_start)

    current_a = a_start.copy()

def _momentum_subgrad(a_init, deadline, k_active=3, momentum=0.9, max_steps=50000):
    """Subgradient method with Nesterov momentum on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Get gradient from top-k peaks
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        
        # Project gradient to tangent space of simplex
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Normalize gradient
        grad = grad / gnorm

        # Nesterov momentum
        v = momentum * v + grad
        v = v - v.mean()  # stay in tangent space

        # Step
        a_try = a + step_size * v
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s

        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
            step_size = min(step_size * 1.005, 2.0 / np.sqrt(n))
        else:
            stall += 1
            if stall % 50 == 0:
                step_size *= 0.8
                momentum = max(0.5, momentum * 0.99)
            if step_size < 1e-10:
                break

    return best_a, best_sc


def _plain_subgrad(a_init, deadline, k_active=2, max_steps=30000):

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random(n, rng, dist='uniform'):
    if dist == 'uniform':
        a = rng.uniform(0.1, 1.0, n)
    elif dist == 'exponential':
        a = rng.exponential(1.0, n) + 0.1
    else:
        a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _peak_shave(a, deadline, k_active=3, max_iters=10000):
    """
    Peak shaving: iteratively reduce the top peaks by adjusting a
    in the direction opposite to the peak's gradient.
    Uses a line search for step size.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Try different step sizes
        improved = False
        for alpha_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            alpha = alpha_mult / (n * gnorm)
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-30:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            break

    return best_a, best_sc

The problem is: min_a f(a) = max_k c[k] subject to a ≥ 0, sum(a) = 1.

The Lagrangian is: L = max_k c[k] - λ(sum(a) - 1) - sum_j μ_j * a_j

- The subgradient of max_k c[k] must contain λ * 1 (the gradient of the constraint)
- That is, there exists w (a probability vector on the active set) such that:
  sum_k w[k] * grad_k c = λ * 1
  where grad_k c[j] = 2*a[k-j] (appropriate bounds)

=> sum_k w[k] * a[k-j] = λ for all j

This means that the weighted average of the "shifted" sequences a[k-j] (over the active k values, weighted by w) is constant (equal to λ) for all j.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive L-BFGS with aggressive log-sum-exp smoothing (very tight betas)
2. Symmetrization heuristic (exploiting KKT structure)
3. Momentum-based subgradient polishing with adaptive step size
4. Peak-shaving line search
5. Multi-length exploration (n=8192 to 32768)
6. Final micro-polish with extremely tight beta
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_solve(a_init, beta, n, max_iter=5000):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    max_c = _conv_max(a_start)
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Very aggressive beta schedule going to extremely small values
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7, 2e-7,
                              1e-7, 5e-8, 1e-8, 5e-9])

    for i, beta in enumerate(betas):

        remaining = deadline - reserve_time - time.monotonic()

def _momentum_subgrad(a_init, deadline, k_active=3, momentum=0.9, max_steps=60000):
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        grad = grad / gnorm
        v = momentum * v + grad
        v = v - v.mean()

        a_try = a + step_size * v
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s

        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
            step_size = min(step_size * 1.005, 3.0 / np.sqrt(n))
        else:
            stall += 1
            if stall % 50 == 0:
                step_size *= 0.85
                momentum = max(0.5, momentum * 0.995)
            if step_size < 1e-10:
                break

    return best_a, best_sc


def _plain_subgrad(a_init, deadline, k_active=2, max_steps=40000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        else:
            stall += 1

def _peak_shave(a, deadline, k_active=3, max_iters=10000):

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W

def _symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-30:
        return a.copy()
    return a_sym / s


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp
