Let me also add a novel component: **iterative reweighted least squares** (IRLS) approach.

The idea: we want to minimize max_k c_k. This is equivalent to finding t and a such that c_k <= t for all k, minimizing t. This is a QCQP (quadratically constrained quadratic program).

We can solve it approximately using an IRLS approach:
1. Start with weights w_k (initially uniform)
2. Solve: minimize sum_k w_k * c_k subject to a >= 0, sum(a) = 1
   This is a quadratic program: minimize a^T (sum_k w_k M_k) a subject to constraints
   The matrix A = sum_k w_k M_k is a symmetric Toeplitz-like matrix
   We can solve this using projected gradient or conjugate gradient
3. Update weights: w_k = 1 / max(c_k, ε) (Em ridicule / inverse of the constraint violation)
4. Repeat

This is similar to the "capacity" or "water-filling" approach.

For step 2, the QP: minimize a^T A a subject to a >= 0, sum(a) = 1.
The KKT conditions give: 2Aa + λ*1 + μ = 0 where μ_i >= 0 and μ_i * a_i = 0.
This is a standard QP that can be solved with active set methods.

But implementing a full QP solver is complex. Instead, I'll use projected gradient:
- Gradient of a^T A a = 2Aa
- Update: a_new = a - lr * 2Aa
- Project onto {a >= 0, sum(a) = 1} (simplex projection)

The computation of Aa can be done efficiently if A has a known structure. Since A = sum_k w_k M_k and M_k is a "shift" matrix, A is a weighted sum of shifts, which means A is a polynomial in the shift operator. Computing Aa can be done using FFT if w has a simple structure.

Actually, Aa at index i = sum_k w_k * (2 * a_{k-i} + 2 * a_{i-k}) (from our gradient formula, which was 2 * conv(w, a_rev)[n-1:2n-1]).

So Aa = 2 * conv(w, a[::-1])[n-1:2n-1] (the same formula as before!).

This means the IRLS iteration is:
1. Compute c = conv(a, a)
2. Update w = 1 / (c + ε) (or some other weighting that emphasizes peaks)
3. Compute Aa = 2 * conv(w, a[::-1])[n-1:2n-1]
4. Update a = project_simplex(a - lr * Aa)
5. Normalize a
6. Repeat

This is elegant and should work well! The key insight is that by reweighting the lags inversely proportional to their current values, we're "pushing up" the valleys and "pushing down" the peaks of the autocorrelation.

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses analytically-derived FFT-based gradients, Adam in log-space,
IRLS peak-shaving, coordinate descent polish, and multi-scale exploration (n=8192, 16384).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_of_smoothed_max(a, c, T, top_k=None):
    """Compute gradient of softmax-weighted sum of convolution lags.
    
    Uses the derived formula: grad = 2 * conv(w, a[::-1])[n-1:2n-1]
    where w = softmax(c/T) or top-k weights.
    """
    n = len(a)
    if top_k is not None:
        idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(len(c))
        for i, j in enumerate(idx):
            w[j] = (0.6 ** i)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
    else:
        shifted = (c - c.max()) / T
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    conv_wa = _fft_convolve(w, a_rev)
    grad = 2.0 * conv_wa[n - 1:2 * n - 1]
    return grad


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, top_k_switch=0.4, active_k=3,
                   eval_interval=200, clip_norm=1.0):
    """Adam optimizer in log-space with adaptive temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        if frac > top_k_switch:
            g = _gradient_of_smoothed_max(a, c, top_k=active_k)
        else:
            T_eff = max(T, c_max * 0.0001)
            g = _gradient_of_smoothed_max(a, c, T_eff)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _irls_peak_shave(a_init, deadline, max_iters=20000, lr_start=0.01, lr_end=0.001,
                     regularize=1e-6):
    """IRLS: iteratively reweight lags inversely proportional to their values."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    t = 0
    while t < max_iters:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iters)
        lr = lr_start * (lr_end / lr_start) ** frac
        c = _fft_convolve(a, a)
        c_pos = np.maximum(c, 1e-12)
        w = 1.0 / (c_pos + regularize)
        w = w / w.sum()
        a_rev = a[::-1]
        Aa = 2.0 * _fft_convolve(w, a_rev)[n - 1:2 * n - 1]
        a_new = a - lr * Aa
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        if t % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _cd_polish(a_init, deadline, max_rounds=2000, top_k=3):
    """Coordinate descent: find active lags, adjust contributing coordinates."""
    n = len(a_init)
    best_a = a_init.copy()
    best_a = best_a / best_a.sum()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        k_indices = np.argsort(c)[::-1][:top_k]
        for k in k_indices:
            if k < 0 or k >= 2 * n - 1:
                continue
            js = range(max(0, k - n + 1), min(k + 1, n))
            for j in js:
                a_part = best_a[j + k - j] if 0 <= k - j < n else 0.0
                if a_part < 1e-12:
                    continue
                for scale in [0.85, 0.7, 0.5, 1.2]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_a, _score_norm(best_a)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _gaussian_init(n, width_frac=0.3):
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = n * width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _chirp_init(n, beta=0.5):
    x = np.arange(n)
    a_phasor = np.exp(1j * 2 * np.pi * beta * x * x / n)
    a = np.abs(a_phasor)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * x / n * 3)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _smooth_bump_init(n):
    x = np.arange(n) / (n - 1)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _spectral_equalize(a, deadline, max_iters=5000):
    """Try to flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a_cur = a.copy()
    a_cur = a_cur / a_cur.sum()
    best_a = a_cur.copy()
    best_score = _score_norm(a_cur)
    N = _next_pow2(2 * n)
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        X = np.fft.rfft(a_cur, N)
        P = np.abs(X) ** 2
        P_flat = P.mean()
        # Soft-clamp the power spectrum
        alpha = 0.3
        P_new = (1 - alpha) * P + alpha * P_flat / (P/P_flat + 0.1) * np.ones_like(P)
        # Target: P_target = P^0.5 * P_flat^0.5 (geometric mean with flat)
        P_target = np.sqrt(P * P_flat)
        # Interpolate
        gamma = 0.1
        P_interp = (1 - gamma) * P + gamma * P_target
        # Reconstruct with same phase
        X_new = X / np.sqrt(np.maximum(P, 1e-30)) * np.sqrt(P_interp)
        a_recon = np.fft.irfft(X_new, N)[:n]
        a_recon = np.maximum(a_recon, 0.0)
        s = a_recon.sum()
        if s < 1e-15:
            break
        a_recon = a_recon / s
        a_cur = a_recon
        if t % 200 == 0:
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
    sc = _score_norm(a_cur)
    if sc < best_score:
        best_score = sc
        best_a = a_cur.copy()
    return best_a, best_score

                                            active_k=3, eval_interval=200, clip_norm=0.7)
            full_sc = _score(a_try)
            print(f"  Init '{name}': score={full_sc:.10f} (init={sc_init:.4f}), t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_try.copy()

    # === Phase 7: Final polish (1500-deadline) ===
    if time.monotonic() < deadline - 20:
        n_fin = len(best_a)
        # Adam with very low temperature
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_pol, sc_pol = _adam_logspace(best_a, deadline - 30,
                                       T_start=0.0005 * max_cf, T_end=0.000005 * max_cf,
                                       lr_start=0.01, lr_end=0.0003,
                                       max_iter=200000, top_k_switch=0.2,
                                       active_k=3, eval_interval=100, clip_norm=0.5)
        full_sc_pol = _score(a_pol)
        print(f"Phase 7 polish: score={full_sc_pol:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_pol) and full_sc_pol < best_score:
            best_score = full_sc_pol
            best_a = a_pol.copy()

        # IRLS final
        if time.monotonic() < deadline - 15:
            a_irls_f, _ = _irls_peak_shave(best_a, deadline - 10,
                                            max_iters=10000, lr_start=0.002, lr_end=0.0002)
            full_sc_f = _score(a_irls_f)
            print(f"Phase 7 IRLS final: score={full_sc_f:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_f) and full_sc_f < best_score:
                best_score = full_sc_f
                best_a = a_irls_f.copy()

        # CD final
        if time.monotonic() < deadline - 5:
            a_cd_f, _ = _cd_polish(best_a, deadline - 3, max_rounds=500, top_k=5)
            full_sc_cd = _score(a_cd_f)
            print(f"Phase 7 CD final: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
                best_score = full_sc_cd
                best_a = a_cd_f.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
