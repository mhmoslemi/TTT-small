"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) L-BFGS with log-sum-exp smoothing, (2) Active-set Newton with iterative refinement,
(3) Frank-Wolfe transport method (new), (4) Multi-n exploration, (5) Final precision polish.
Key innovation: Frank-Wolfe/transport method that directly operates on the max-correlation
structure, combined with adaptive active-set Newton for final precision.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _corr_a_A(a, A_vec):
    """Compute g[i] = sum_j a[j] * A_vec[i+j] for i=0..n-1 via FFT."""
    n = len(a)
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_matvec(v, A_vec, m):
    """Compute H*v where H[i,j] = (1/m) if (i+j) in active set."""
    return _corr_a_A(v, A_vec) / m


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.001):
    """Active-set Newton with CG and adaptive active set."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Cap active set size for efficiency
        max_active = min(2 * n, 4096)
        if n_active > max_active:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:max_active]
            active[top_idx] = True
            n_active = max_active

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of f(a) = (1/m) * sum_{k in active} c[k]
        g = _hess_matvec(a, A_vec, m) * 2.0
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # CG to solve (2H + mu*I) u = g_t
        mu = 1e-8 * c_max / max(m, 1) + 1e-10

        def _hvp(w):
            return 2.0 * _hess_matvec(w, A_vec, m) + mu * w

        # Solve (2H + mu*I) u = g_t via CG
        x = np.zeros(n)
        r = g_t - _hvp(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = _hvp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        u = x

        # Solve (2H + mu*I) w = 1 for the constraint correction
        x2 = np.zeros(n)
        r2 = np.ones(n) - _hvp(x2)
        p2 = r2.copy()
        rs2 = np.dot(r2, r2)
        for cg_it in range(300):
            Ap2 = _hvp(p2)
            pAp2 = np.dot(p2, Ap2)
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            x2 += alpha2 * p2
            r2_new = r2 - alpha2 * Ap2
            rs2_new = np.dot(r2_new, r2_new)
            if np.sqrt(rs2_new) < 1e-12:
                break
            p2 = r2_new + (rs2_new / rs2) * p2
            r2 = r2_new
            rs2 = rs2_new
        w = x2

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-12 or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback gradient step
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _frank_wolfe_transport(a_init, deadline, max_iter=50000):
    """Frank-Wolfe / transport method for the max-correlation problem."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    iteration = 0

    while time.monotonic() < deadline and iteration < max_iter:
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use a small active set for the gradient
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        A_vec = active.astype(np.float64)
        m = int(active.sum())

        # Gradient: g[i] = (2/m) * sum_j a[j] * A_vec[i+j]
        g = 2.0 * _corr_a_A(a, A_vec) / m

        # Project to tangent space
        g_proj = g - np.mean(g)

        # Away-step Frank-Wolfe: transfer from argmax(g_proj) to argmin(g_proj)
        i_src = int(np.argmax(g_proj))
        i_dst = int(np.argmin(g_proj))

        if i_src == i_dst:
            break
        if a[i_src] < 1e-15:
            break

        # Step size: 2/(iteration+2) for standard FW, but we use a custom one
        # Limited by a[i_src] (can't go negative)
        t_fw = 2.0 / (iteration + 2.0)
        t_max = a[i_src] * 0.99  # Can't decrease a[i_src] below 0
        t_step = min(t_fw, t_max)

        # Also limit by the peak not increasing (linear approximation)
        delta_g = g[i_src] - g[i_dst]
        if delta_g > 1e-15:
            # The peak should decrease by approximately t_step * delta_g * c_max
            # Limit so we don't overshoot
            t_limit = 0.5 * c_max / max(delta_g, 1e-15)
            t_step = min(t_step, t_limit)

        a[i_src] -= t_step
        a[i_dst] += t_step
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        iteration += 1

        if iteration % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()

    # Final check
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def _frank_wolfe_batch(a_init, deadline, max_iter=2000, batch_size=10):
    """Batch Frank-Wolfe: transfer from top-k sources to top-k destinations."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for iteration in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        A_vec = active.astype(np.float64)
        m = int(active.sum())

        g = 2.0 * _corr_a_A(a, A_vec) / m
        g_proj = g - np.mean(g)

        # Find top-k sources and top-k destinations
        k = min(batch_size, n // 2)
        src_indices = np.argsort(g_proj)[::-1][:k]
        dst_indices = np.argsort(g_proj)[:k]

        # Only transfer from sources with sufficient mass
        valid_src = [i for i in src_indices if a[i] > 1e-10]
        if not valid_src:
            break

        # Equal transfer amount
        total_transfer = min(a[valid_src[0]] * 0.1, 0.05 / len(valid_src))
        per_src = total_transfer / len(valid_src)
        per_dst = total_transfer / k

        for i in valid_src:
            a[i] -= per_src
        for j in dst_indices:
            a[j] += per_dst

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if (iteration + 1) % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a.copy()

    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    # More gradual beta reduction
    betas = base_beta * np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.10,
                                  0.06, 0.04, 0.025, 0.015, 0.010,
                                  0.006, 0.004, 0.0025, 0.0015, 0.001,
                                  0.0006, 0.0004, 0.00025, 0.00015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time_per_iter * 6))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _power_iteration_active(a_init, deadline, max_iter=200):
    """Power iteration on the active-set Hessian to find the eigenvector solution."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        A_vec = active.astype(np.float64)
        m = int(active.sum())

        # Power iteration: v -> H*v, normalize
        v = _hess_matvec(a, A_vec, m)
        v = np.maximum(v, 0)
        sv = v.sum()
        if sv < 1e-15:
            break
        v = v / sv

        # Damped update
        a_new = 0.7 * a + 0.3 * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # Convergence
        if np.linalg.norm(a_new - a) < 1e-12:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient method with active-set tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, active-set Newton,
    Frank-Wolfe transport, power iteration, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_deadline = min(deadline - 400, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=400, tol_active=0.001)
        update_best(a_as, sc_as, "active_newton_1")

    if best_seq is not None and time.monotonic() < deadline - 300:
        as_deadline2 = min(deadline - 300, time.monotonic() + 90)
        a_as2, sc_as2 = _active_set_newton(best_seq, as_deadline2, max_iter=300, tol_active=0.0005)
        update_best(a_as2, sc_as2, "active_newton_2")

    # Phase 3: Frank-Wolfe transport (NEW)
    print(f"\n[Phase 3] Frank-Wolfe Transport (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        fw_deadline = min(deadline - 250, time.monotonic() + 120)
        a_fw, sc_fw = _frank_wolfe_transport(best_seq, fw_deadline, max_iter=30000)
        update_best(a_fw, sc_fw, "frank_wolfe")

        # Run active-set Newton after FW
        if sc_fw < best_score + 0.01 and time.monotonic() < deadline - 200:
            a_fw2, sc_fw2 = _active_set_newton(a_fw, min(deadline - 200, time.monotonic() + 60),
                                                max_iter=200, tol_active=0.001)
            update_best(a_fw2, sc_fw2, "fw_then_newton")

    # Phase 4: Batch Frank-Wolfe (NEW)
    print(f"\n[Phase 4] Batch FW (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        bw_deadline = min(deadline - 150, time.monotonic() + 60)
        a_bw, sc_bw = _frank_wolfe_batch(best_seq, bw_deadline, max_iter=5000, batch_size=20)
        update_best(a_bw, sc_bw, "batch_fw")

    # Phase 5: Power iteration on active set (NEW)
    print(f"\n[Phase 5] Power Iteration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        pi_deadline = min(deadline - 100, time.monotonic() + 60)
        a_pi, sc_pi = _power_iteration_active(best_seq, pi_deadline, max_iter=200)
        update_best(a_pi, sc_pi, "power_iter")

    # Phase 6: Multi-n search
    print(f"\n[Phase 6] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        if time.monotonic() < deadline - 40:
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")

            if sc_opt < best_score + 0.02 and time.monotonic() < deadline - 30:
                a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 40),
                                                      max_iter=150, tol_active=0.001)
                update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # Phase 7: Subgradient polish
    print(f"\n[Phase 7] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sg_deadline = min(deadline - 60, time.monotonic() + 60)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=5000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # Phase 8: Perturbation restarts
    print(f"\n[Phase 8] Perturbation restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for sigma, label in [(0.02, "small"), (0.05, "mid"), (0.1, "large")]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            if time.monotonic() < deadline - 10:
                a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{label}")

    # Phase 9: Final precision polish
    print(f"\n[Phase 9] Final precision (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active-set Newton
        as_final = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=300, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_newton_tight")

        # Another FW pass on the result
        if time.monotonic() < deadline - 3:
            fw_final = min(deadline - 2, time.monotonic() + 10)
            a_fin2, sc_fin2 = _frank_wolfe_transport(a_fin, fw_final, max_iter=10000)
            update_best(a_fin2, sc_fin2, "final_fw")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 4 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 80}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
