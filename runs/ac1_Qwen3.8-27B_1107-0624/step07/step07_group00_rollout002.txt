```python
"""
Multi-strategy optimizer with exact active-set gradients.
Uses top-k peak tracking for exact subgradients, Adam in log-space
with backtracking, aggressive coordinate descent, and multi-n search.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _fft_conv(a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Score for arbitrary a."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)

def _active_gradient(a, c, top_k=10, power=5):
    """
    Compute subgradient of max(c) using top-k peaks with power weighting.
    a is normalized (sum=1), c = convolve(a, a).
    Returns gradient wrt a.
    """
    n = len(a)
    # Find top-k indices by value
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    c_max = c[top_idx[0]]
    
    # Weighting: w[i] = (c[i]/c_max)^power
    w = np.zeros(len(c))
    for idx in top_idx:
        w[idx] = (c[idx] / c_max) ** power
    w /= w.sum()
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] (where 0 <= k-j < n)
    grad = np.zeros(n)
    for idx in top_idx:
        wi = w[idx]
        lo = max(0, idx - n + 1)
        hi = min(idx, n - 1)
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * wi * a[idx - js]
    
    return grad

def _adam_logspace(a_init, deadline, top_k=10, power=5,
                   lr=0.02, max_iter=100000, eval_interval=100,
                   clip_norm=0.5, backtrack_every=30, plateau_check=5):
    """
    Adam optimization in log-space with exact active-set gradient.
    a_init is normalized (sum=1).
    """
    n = len(a_init)
    log_a = np.log(np.maximum(a_init, 1e-20))
    log_a = log_a - log_a.mean()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Current a
        a = np.exp(log_a - np.max(log_a))
        a = a / a.sum()
        
        # Compute convolution
        c = _fft_conv(a)
        
        # Gradient
        g = _active_gradient(a, c, top_k, power)
        
        # Project gradient for log-space: d(log_a)/d(a) * g = g / a
        # But we use the softmax parameterization: z = log(a) (unnormalized)
        # a = softmax(z), so da/dz = a * (I - a*a^T)
        # dz (gradient in z-space) = g - a * (a^T * g) ... 
        # Actually, for softmax: d/ dz J(softmax(z)) = a * (g - sum(a*g))
        # Wait, that's for the case where we're optimizing J(softmax(z)) wrt z.
        # dJ/dz = a * (g - a^T * g / a^T * 1) ... 
        # Actually: if a = softmax(z), then da_i/dz_j = a_i * (delta_ij - a_j)
        # So dJ/dz_j = sum_i (dJ/da_i) * da_i/dz_j = sum_i g_i * a_i * (delta_ij - a_j)
        # = a_j * g_j - a_j * sum_i g_i * a_i = a_j * (g_j - sum_i a_i * g_i)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        effective_lr = lr * lr_mult
        update = effective_lr * mh / (np.sqrt(vh) + eps)
        
        # Clip
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        log_a_new = log_a - update
        log_a_new = np.clip(log_a_new, -50.0, 50.0)
        log_a_new = log_a_new - log_a_new.mean()
        
        # Backtrack check
        if backtrack_every > 0 and t % backtrack_every == 0:
            a_cand = np.exp(log_a_new - np.max(log_a_new))
            a_cand = a_cand / a_cand.sum()
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.02:
                log_a_new = log_a  # Reject
        log_a = log_a_new
        
        # Evaluate
        if t % eval_interval == 0:
            a_cur = np.exp(log_a - np.max(log_a))
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience >= plateau_check and lr_mult > 0.1:
                lr_mult *= 0.5
                patience = 0
                log_a = np.log(np.maximum(best_a, 1e-20))
                log_a = log_a - log_a.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                t = 0  # Reset Adam state
    
    # Final
    a_final = np.exp(log_a - np.max(log_a))
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _cd_polish(a, deadline, max_rounds=200, top_peaks=5):
    """
    Coordinate descent: for each of the top peaks, try adjusting the contributing coordinates.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_conv(best_a)
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        # Get top peaks
        top_indices = np.argsort(c)[::-1][:top_peaks]
        
        for k in top_indices:
            # Find contributing j values
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            if len(js) == 0:
                continue
            
            # For each j, try different scales
            for j in js:
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.5, 0.7, 0.85, 1.2, 1.5]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_conv(a_cand)
                    sc = float(2.0 * n * c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)
```
