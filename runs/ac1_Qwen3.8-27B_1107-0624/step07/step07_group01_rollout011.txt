import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_softmax_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max, None
    return 2.0 * n * (c_max + beta * np.log(s)), c


def _log_softmax_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W), w


def _active_grad(a, c, n, k_active):
    """Compute subgradient using top-k active lags with optimal weights."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    # Use soft weights: weight ∝ exp(alpha * (c[k] - c_max) / temperature)
    # This gives the "tightest" subgradient
    idx_all = np.arange(len(c))
    c_shifted = (c - c_max) / max(c_max * 1e-4, 1e-30)
    w_soft = np.exp(c_shifted * 50)  # sharp peak near max
    w_soft = w_soft / w_soft.sum()
    
    # Focus on top-k
    top_k = np.argsort(c)[::-1][:k_active]
    for i in range(len(c)):
        if i not in top_k:
            w_soft[i] *= 0.01
    
    w_soft = w_soft / w_soft.sum()
    
    a_rev = a[::-1]
    wc = _full_convolve(w_soft, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w_soft, c))
    return 4.0 * n * a * (h - W), w_soft


def _hard_active_grad(a, c, n, k_active):

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-16):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        val, _ = _log_softmax_obj(z, beta, n)
        return val

    def jac(z):
        g, _ = _log_softmax_grad(z, beta, n)
        return g

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _momentum_subgrad(a_init, deadline, k_active_init=5, k_active_max=50,
                      max_steps=100000, lr_init=0.03, momentum_init=0.9):
    """
    Heavy-ball momentum subgradient with adaptive active set.
    Uses dual-fitting for optimal gradient direction.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    lr = lr_init
    momentum = momentum_init
    stall = 0
    v = np.zeros(n)  # momentum buffer
    eval_interval = max(100, n // 50)
    stall_threshold = 20

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: increase k as we converge
        k_active = min(k_active_max, k_active_init + i // 2000)
        
        # Compute dual-fitted gradient
        g, w = _active_grad(a, c, n, k_active)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Heavy ball update
        v = momentum * v - lr * g / (a.sum() * n)
        z = z + v
        z = z - z.mean()
        z = np.clip(z, -200.0, 200.0)

        # Periodic evaluation
        if i % eval_interval == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-16:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                # Slightly increase momentum when making progress
                momentum = min(0.98, momentum + 0.005)
                lr = min(lr * 1.01, 0.1)
            else:
                stall += 1
                if stall >= stall_threshold:
                    lr *= 0.7
                    momentum = max(0.5, momentum - 0.05)
                    stall = 0
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95
                    momentum = max(0.5, momentum - 0.01)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _dual_fitted_polish(a_init, deadline, k_active=10, max_steps=50000):
    """
    Polishing using optimal dual weights (Feynman-Hellmann style).
    At the optimum, the gradient should be constant on the support.
    We exploit this: find weights w over active lags such that
    the induced gradient is as "flat" as possible on the support.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / n

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify active set with soft threshold
        threshold = c_max - 5 * c_max / n
        active_mask = c >= threshold
        if active_mask.sum() < 2:
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[np.argsort(c)[::-1][:k_active]] = True
        active_idx = np.where(active_mask)[0]
        k_act = len(active_idx)

        # Compute optimal weights: minimize variance of induced gradient on support
        # The gradient from lag k is: g_k[j] = a[k-j] + a[j-k] (centered)
        # We want sum_k w_k * g_k to be as constant as possible on {j: a[j] > 0}
        
        # Approximate: use weights proportional to (c[k] - threshold) for active k
        c_active = c[active_idx]
        w = np.maximum(c_active - threshold, 0)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_act)
        else:
            w = w / w_sum

        w_full = np.zeros(len(c))
        w_full[active_idx] = w

        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w_full, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w_full, c))
        g = 2.0 * n * (h - W)

        # Remove mean (works on the simplex)
        g = g - g.mean()

        # Normalized step
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Line search
        improved = False
        for alpha_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = step_size * alpha_factor
            a_try = a - alpha * g / gnorm
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            if sc_try < _score(a):
                a = a_try
                improved = True
                break

        if not improved:
            step_size *= 0.5
            if step_size < 1e-15 / n:
                break

    return best_a, best_sc

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):

    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=30.0):
    """Diversification via perturbation + local optimization."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    
    if deadline - time.monotonic() < reserve + 60:
        return best_a, best_sc
    
    time_per_hop = (deadline - reserve - time.monotonic()) / n_hops

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.75

        # Create perturbation in log-space
        pert_type = h % 3
        z = np.log(np.maximum(best_a, 1e-18))
        
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = z + sigma * rng.standard_normal(n)
        elif pert_type == 1:
            for _ in range(3):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        else:
            n_blocks = 4 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
        
        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        max_iter = min(3000, max(200, int((hop_deadline - time.monotonic()) / 0.005)))
        a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Basin hop {h}: improved to {sc_opt:.10f}")

    return best_a, best_sc


def _final_micro_polish(a_init, deadline, iterations=20000):
    """
    Ultra-fine polishing: very small steps with optimal gradient direction.
    Uses the fact that near optimum, the active set is stable.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    # Start with small step
    step = 0.3 / np.sqrt(n)
    momentum_vec = np.zeros(n)
    mu = 0.85  # momentum
    
    for i in range(iterations):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Use top-k with soft weighting
        k_act = min(20, 5 + i // 2000)
        c_shifted = (c - c_max) / (c_max * 1e-5 + 1e-30)
        w = np.exp(np.maximum(c_shifted * 20, -100))
        w = w / w.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Momentum update
        momentum_vec = mu * momentum_vec - step * g / gnorm
        a_try = a + momentum_vec
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-12:
            step *= 0.5
            momentum_vec *= 0.5
            continue
        a_try = a_try / s

        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            # Slightly increase step on success
            step = min(step * 1.005, 1.0 / np.sqrt(n))
        else:
            # Revert and decrease step
            a_try = a  # keep old
            step *= 0.7
            momentum_vec *= 0.5
            if step < 1e-12 / n:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimizer exploiting the convexity of the problem.
    Phases:
    1. Initialization (parent + structured)
    2. Coarse multi-scale L-BFGS
    3. Fine multi-scale L-BFGS  
    4. Momentum subgradient polishing
    5. Dual-fitted refinement
    6. Basin hopping diversification
    7. Multi-length exploration
    8. Final micro-polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 16384

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")
    
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={best_score_val:.10f}")
    else:
        # Generate initializations
        n_init = 16384
        inits = [
            _make_chirp(n_init, freq=1.0, eps=0.3),
            _make_chirp(n_init, freq=2.0, eps=0.25),
            _make_multiband(n_init, bands=[1, 3], eps=0.25),
            _make_tapered(n_init, power=0.5),
            _make_random_positive(n_init, rng),
        ]
        for i, a_inp in enumerate(inits):
            try_update(a_inp, f"init_{i}")

    if best_seq is None:
        best_seq = np.ones(16384) / 16384
        best_score_val = _score(best_seq)
        best_n = 16384

    print(f"  Phase 1 done. Best: {best_score_val:.10f} at n={best_n}")

    # === Phase 2: Main multi-scale L-BFGS ===
    print(f"\n=== Phase 2: Multi-scale L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    a_p2 = _multi_scale_lbfgs(best_seq, deadline, reserve_time=600.0)
    try_update(a_p2, "multi_scale_lbfgs")

    # === Phase 3: Momentum subgradient ===
    print(f"\n=== Phase 3: Momentum subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        a_sg, sc_sg = _momentum_subgrad(best_seq, deadline - 400,
                                         k_active_init=5, k_active_max=30,
                                         max_steps=80000, lr_init=0.025,
                                         momentum_init=0.9)
        try_update(a_sg, "momentum_sg")

    # === Phase 4: Dual-fitted polish ===
    print(f"\n=== Phase 4: Dual-fitted polish (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        a_df, sc_df = _dual_fitted_polish(best_seq, deadline - 300,
                                           k_active=10, max_steps=30000)
        try_update(a_df, "dual_fitted")

    # === Phase 5: More L-BFGS (finer) ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 250:
        a_p5 = _multi_scale_lbfgs(best_seq, deadline, reserve_time=200.0)
        try_update(a_p5, "fine_lbfgs")

    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 150, rng,
                                    n_hops=4, reserve=80.0)
        try_update(a_hop, "basin_hop")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Length exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        lengths_to_try = [8192, 32768, 2048]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 60:
                break
            if target_n == best_n:
                continue
            # Only try smaller lengths if we're running low on time
            if target_n > best_n and time.monotonic() > deadline - 80:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")
            if sc_interp < best_score_val:
                try_update(a_interp, f"interp_n{target_n}")
            # Quick optimization
            if time.monotonic() < deadline - 40:
                a_opt = _multi_scale_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"opt_n{target_n}")

    # === Phase 8: Final micro-polishing ===
    print(f"\n=== Phase 8: Final micro-polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 2:
        for rnd in range(8):
            if time.monotonic() > deadline - 2:
                break
            a_f, sc_f = _final_micro_polish(best_seq, deadline - 2,
                                             iterations=15000)
            try_update(a_f, f"micro_{rnd}")
            if best_score_val < 1.5028:
                break

        # One more L-BFGS with ultra-tight beta
        if time.monotonic() < deadline - 1:
            n = len(best_seq)
            _, mc = _conv_and_max(best_seq)
            beta = mc * 1e-12
            max_iter = min(3000, max(100, int((deadline - 1 - time.monotonic()) / 0.004)))
            if max_iter > 50:
                a_ultra = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_ultra, "ultra_lbfgs")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
