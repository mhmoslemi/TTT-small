"""
Optimizer combining dual mirror descent (game-theoretic), FISTA inner QP solver,
L-BFGS log-sum-exp smoothing, active-set refinement, and multi-length exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _simplex_proj(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _q_matvec(q, x):
    """Compute Q(q) @ x where Q_{i,j} = q_{i+j}. Uses FFT-based convolution."""
    n = len(x)
    b = q[::-1]
    c = _full_convolve(b, x)
    return c[n - 1:2 * n - 1][::-1]


def _inner_fista(q, a_init, n_inner=25, L=None):
    """Solve min a^T Q(q) a s.t. a in simplex using FISTA."""
    n = len(a_init)
    if L is None:
        L = 2.0 * float(q.sum())
    step = 1.0 / max(L, 1e-12)

    a = a_init.copy()
    y = a.copy()
    t = 1.0

    for _ in range(n_inner):
        grad_y = 2.0 * _q_matvec(q, y)
        a_new = _simplex_proj(y - step * grad_y)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (t - 1.0) / t_new * (a_new - a)
        a = a_new
        t = t_new

    return a

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-12):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

def _active_set_optimize(a_init, deadline, k_active=5, max_rounds=200):
    """
    Iteratively optimize the active set:
    1. Identify top-k lags at max
    2. Solve min sum_{j in active} lambda_j * c_{k_j}(a) via FISTA
    3. Update lambdas based on which constraints are binding
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]

        # Build q concentrated on active set
        q = np.zeros(2 * n - 1)
        # Give weights inversely proportional to how "below max" each is
        c_active = c[idx]
        c_spread = c_active.max() - c_active.min()
        if c_spread < 1e-15:
            w_active = np.ones(k_active) / k_active
        else:
            w_active = 1.0 / (c_active.max() - c_active + 1e-15)
            w_active = w_active / w_active.sum()
        q[idx] = w_active

        # Solve inner QP with FISTA
        L = 2.0 * float(q.sum())
        a = _inner_fista(q, a, n_inner=50, L=L)

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        c_new_active = c_new[idx]
        if c_new_active.max() - c_new_active.min() < 1e-12 * c_new_active.max():
            break

    return best_a, best_sc

    t = 1.0
    step_base = 2.0 / np.sqrt(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at current a
        c_a = _full_convolve(a, a)
        idx = np.argsort(c_a)[::-1][:k_active]
        w = np.zeros_like(c_a)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c_a))
        grad = 2.0 * (h - W)
        # Project gradient to be tangent to simplex
        grad = grad - np.dot(grad, a) * a  # make orthogonal to a (tangent space)

        # Step
        step = step_base / (1.0 + t)
        a_new = _simplex_proj(a - step * grad * n)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = a_new + (t - 1.0) / t_new * (a_new - a)

        a = _simplex_proj(y_new)
        t = t_new

        if i % 500 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _make_chirp(n, freq=3.0, eps=0.35):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
