        rs = rs_new

                stall = 0
            if stall > 60:
                print(f"    iter {it}: stalled, breaking")
                break

    # Final score check
    final_sc = _score(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax_z(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax_z(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_fine(a_init, deadline, min_remaining=60.0):
    """L-BFGS with log-sum-exp smoothing, fine beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
                              5e-7, 2e-7, 1e-7, 5e-8])
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        try:
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxcor': 100,
                                       'maxls': 50, 'disp': False})
            a_out = _softmax_z(result.x)
            sc = _score(a_out)
            if sc < current_sc - 1e-15:
                current_a = a_out
                current_sc = sc
                print(f"    L-BFGS beta={beta:.2e}: score={sc:.10f}")
        except Exception:
            pass

    return current_a


def _peak_shaving_fine(a_init, deadline):
    """Direct-space peak shaving with adaptive step size."""
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0

    for i in range(50000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Top-10 active set
        sorted_idx = np.argsort(c)[::-1]
        K = min(20, len(sorted_idx))
        active = sorted_idx[:K]

        # Weighted gradient
        c_act = c[active]
        w = np.exp(c_act / (0.01 * c_max) - c_act.max())
        w = w / w.sum()

        g = np.zeros(len(a))
        for m, wi in zip(active, w):
            g += wi * _grad_f_m_vec(a, int(m))
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        # Line search
        best_eta_local = eta
        best_sc_local = _score(a)
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_local:
                best_sc_local = sc_try
                best_eta_local = step

        if best_sc_local < best_sc - 1e-14:
            best_sc = best_sc_local
            a = a - best_eta_local * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_a = a.copy()
            eta = min(eta * 1.05, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.85
            if stall > 50 or eta < 1e-9:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: active-set Newton + L-BFGS polish + peak shaving.
    Exploits convexity of the problem with second-order methods.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s (improvement: {prev-sc:.6e})")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(8192, "uniform")
        print(f"No parent. Using uniform n=8192, score={_score(a0):.10f}")
        try_update(a0, "init")

    # === Phase 1: Active-set Newton (main workhorse) ===
    print(f"\n=== Phase 1: Active-set Newton ===")
    t_phase = time.monotonic()
    a_newton, sc_newton = _newton_phase(best_seq, deadline - 400,
                                         max_iters=150, K_start=20, K_max=150)
    try_update(a_newton, "newton")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={sc_newton:.10f}")

    # === Phase 2: L-BFGS fine polish ===
    print(f"\n=== Phase 2: L-BFGS fine polish ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 200:
        a_lbfgs = _lbfgs_fine(best_seq, deadline - 200, min_remaining=100.0)
        try_update(a_lbfgs, "lbfgs_fine")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 150:
        a_ps, sc_ps = _peak_shaving_fine(best_seq, deadline - 150)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Second Newton pass with refined active set ===
    print(f"\n=== Phase 4: Refined Newton ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 100:
        a_newton2, sc_n2 = _newton_phase(best_seq, deadline - 100,
                                          max_iters=100, K_start=50, K_max=200,
                                          tau_init=0.005, reg_init=1e-8)
        try_update(a_newton2, "newton2")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Ultra-fine L-BFGS ===
    print(f"\n=== Phase 5: Ultra-fine L-BFGS ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 50:
        a_ultra = _lbfgs_fine(best_seq, deadline - 50, min_remaining=30.0)
        try_update(a_ultra, "ultra_lbfgs")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Try slightly different starting points and re-optimize ===
    print(f"\n=== Phase 6: Perturbation + re-optimize ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 40:
        for attempt in range(3):
            if time.monotonic() > deadline - 35:
                break
            # Small perturbation in log space
            z = np.log(np.maximum(best_seq, 1e-15))
            scale = 0.01 * (attempt + 1)
            z_pert = z + scale * rng.standard_normal(len(z))
            z_pert = z_pert - z_pert.mean()
            a_pert = _softmax_z(z_pert)
            sc_pert = _score(a_pert)
            print(f"  Perturb {attempt+1}: score={sc_pert:.10f} (scale={scale})")

            if sc_pert < best_score_val + 0.01:  # Only if not too far
                # Quick Newton polish
                a_opt, _ = _newton_phase(a_pert, deadline - 35 - attempt * 10,
                                          max_iters=50, K_start=30, K_max=100)
                try_update(a_opt, f"perturb_{attempt+1}_newton")

                if time.monotonic() < deadline - 35:
                    a_opt2 = _lbfgs_fine(a_opt, deadline - 35, min_remaining=15.0)
                    try_update(a_opt2, f"perturb_{attempt+1}_lbfgs")

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Final intensive cycle ===
    print(f"\n=== Phase 7: Final polish cycle ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(3):
            if time.monotonic() > deadline - 5:
                break
            # Quick peak shave
            if time.monotonic() < deadline - 5:
                a_f1, _ = _peak_shaving_fine(best_seq, deadline - 5)
                try_update(a_f1, f"final_ps_{cycle}")

            # Quick Newton
            if time.monotonic() < deadline - 5:
                a_f2, _ = _newton_phase(best_seq, deadline - 5,
                                         max_iters=30, K_start=30, K_max=100,
                                         tau_init=0.003, reg_init=1e-10)
                try_update(a_f2, f"final_newton_{cycle}")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
