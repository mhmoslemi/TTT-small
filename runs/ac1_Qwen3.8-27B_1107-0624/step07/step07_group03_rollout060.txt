def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton_variable(a_init, deadline, max_iter=500, tol_active=0.002):
    """Variable-active-set Newton with dynamic set updates and non-negativity."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_c_max = float(_full_convolve(a, a).max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 4 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-5 * (c_max / max(m, 1)) + 1e-8

        def hv(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve for (H + mu I) u = g_t
        x = np.zeros(n)
        r = g_t - hv(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = hv(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13 * (1.0 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        u = x

        # Solve for sum constraint
        w_cg = np.zeros(n)
        r = np.ones(n) - hv(w_cg)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = hv(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            w_cg += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        w = w_cg
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search with non-negativity
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_now = _full_convolve(a, a)
        c_now_max = float(c_now.max())
        if abs(c_now_max - c_max) < 1e-13 * c_max:
            break

        # Tighten active set every 20 iterations
        if it % 20 == 19 and it > 0:
            tol_active = max(tol_active * 0.7, 0.0001)

    return best_a, best_sc


def _coordinate_polish(a_init, deadline, n_sweeps=5):

    a = np.maximum(a, 0)

                # If only one valid: one term

            C = np.zeros(2 * n - 1)

            B = np.zeros(2 * n - 1)

            # conv_new[k] = sum_j new_a[j] * new_a[k-j]
            # = conv_old[k] - a[i]*a[k-i]*(if valid) - a[k-i]*a[i]*(if valid, k-i != i)

            # conv_new[k] = conv_old[k] + delta * B_k + delta^2 * C_k
            # where B_k = a[k-i] if (0<=k-i<n and k-i != i) + a[k-i] if (0<=k-i<n and k-i != i)
            #            = 2*a[k-i] if (0<=k-i<n and k-i != i)
            #            = a[k-i] if (0<=k-i<n and k-i == i)  [in this case C_k=1]

            #   The term is a[i]*a[i] = a[i]^2. Change: (a[i]+delta)^2 - a[i]^2 = 2*a[i]*delta + delta^2

            #   The terms are a[i]*a[k-i] and a[k-i]*a[i]. Change: 2*a[k-i]*delta

            jj_arr = np.arange(2 * n - 1) - i
            valid = (jj_arr >= 0) & (jj_arr < n)
            is_self = (jj_arr == i) & valid
            B[is_self] = 2.0 * a[i]  # when k=2i, B = 2*a[i], C = 1
            C[is_self] = 1.0
            not_self_valid = valid & ~is_self
            B[not_self_valid] = 2.0 * a[jj_arr[not_self_valid]]

            # Now minimize max_k (c[k] + B[k]*delta + C[k]*delta^2) over delta >= -a[i]
            # The lower bound is delta >= -a[i] (to keep a[i] >= 0)
            # Find optimal delta by ternary search

            lo = -a[i]
            # Upper bound: the new conv should not be too large
            # A reasonable upper bound is the current max / a[i] (if a[i] > 0)
            hi = max(0.0, c.max() / max(a[i], 1e-10)) if a[i] > 0 else 0.0
            hi = min(hi, 10.0)  # Cap it
            if hi <= lo:
                hi = lo + 0.01

            # Evaluate the objective at a few points to bracket
            def obj_1d(delta):
                c_new = c + B * delta + C * delta * delta
                return float(c_new.max())

            # Ternary search
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = obj_1d(m1)
                f2 = obj_1d(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            optimal_delta = (lo + hi) / 2.0
            new_ai = a[i] + optimal_delta
            if new_ai < 0:
                new_ai = 0.0

            # Check if this improves
            a[i] = new_ai
            s_new = a.sum()
            if s_new < 1e-15:
                a[i] = a[i] - optimal_delta  # revert
                continue
            a_norm = a / s_new
            sc_new = _score(a_norm)
            if sc_new < best_sc - 1e-12:
                a = a_norm
                best_sc = sc_new
                best_a = a.copy()
                improved = True
            else:
                # Revert
                a[i] = a[i] - optimal_delta / s_new if s_new > 0 else 0
                # Actually we already modified a, need to revert properly
                a = a * (s_new / s_new)  # no-op, we need to undo
                pass

        if not improved and sweep > 0:
            break

    # Renormalize
    a = best_a.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    return a, _score(a)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Adaptive L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(12000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.4e}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient method with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.005, 0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-11:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_window":
        t = np.linspace(0, 1, n)
        # Flat in the middle, taper at edges
        edge = 0.1
        a = np.ones(n)
        n_edge = int(n * edge)
        if n_edge > 0:
            for i in range(n_edge):
                a[i] = i / n_edge
                a[n - 1 - i] = i / n_edge
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _symmetrize(a):
    """Make sequence palindromic by averaging with its reverse."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym / s

            best_seq = p.copy()
            best_score = float(sc)

                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    except Exception:
