def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _cg_solve(hess_vec, b, n, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _make_profile(n, kind="uniform"):

    elif kind == "wheatstone":

    elif kind == "u_shape":

def _spectral_perturb(a, rng, n_modes=8, amp_range=(0.01, 0.15)):
    n = len(a)
    A = np.fft.rfft(a)
    amp = np.abs(A)
    phase = np.angle(A)
    for _ in range(n_modes):
        k = rng.integers(1, len(A))
        da = rng.uniform(*amp_range) * np.sign(rng.uniform(-1, 1))
        amp[k] = np.maximum(amp[k] + da, amp[k] * 0.1)
        if rng.uniform() < 0.5:
            phase[k] += rng.uniform(-0.3, 0.3)
    A_new = amp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _block_swap(a, rng, n_swaps=3, min_block=8, max_block_frac=0.1):
    n = len(a)
    a_work = a.copy()
    for _ in range(n_swaps):
        block_size = rng.integers(min_block, max(10, int(n * max_block_frac)))
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        a_work[i:i + block_size], a_work[j:j + block_size] = a_work[j:j + block_size].copy(), a_work[i:i + block_size].copy()
    return a_work / a_work.sum()


def _mirror_antisym_perturb(a, rng, strength=0.1):
    n = len(a)
    a_work = a.copy()
    half = n // 2
    mirror = a_work[half:][::-1]
    diff = a_work[:half] - mirror[:len(a_work[:half])]
    correction = strength * diff * rng.uniform(0.5, 1.5)
    a_work[:half] += correction
    a_work[half:] -= correction
    a_work = np.maximum(a_work, 1e-15)
    return a_work / a_work.sum()

        a = _softmax(z)
        c = _full_convolve(a, a)
        tol_frac = max(0.0005, 0.002 * np.exp(-i / 2000.0))
        grad = _gradient_active(a, c, n, tol_frac=tol_frac)
        step = lr / (1.0 + 0.005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                no_improve_count = 0
                lr = min(lr * 1.02, lr_init)
            else:
                no_improve_count += 1
                if no_improve_count > 30:
                    lr *= 0.85
                    if no_improve_count > 80:
                        z = np.log(np.maximum(best_a, 1e-30))
                        z = z - z.mean()
                        no_improve_count = 0
                        lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_descent_polish(a_init, deadline, n_rounds=20):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, max(1, n // 2000)):
            if time.monotonic() > deadline:
                break
            idx = slice(i, i + max(1, n // 2000))
            old_val = a[idx].copy()
            for trial in range(3):
                perturbation = 0.01 * (trial - 1)
                a[idx] = old_val * (1.0 + perturbation)
                a[idx] = np.maximum(a[idx], 0)
                s = a.sum()
                if s < 1e-15:
                    a[idx] = old_val
                    break
                a_norm = a / s
                sc = _score(a_norm)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_norm.copy()
                    improved = True
                    a = a_norm
                    break
            a[idx] = old_val
        if not improved:
            break

    return best_a, best_sc


def _shrinkage_optim(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.20, 0.30]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Extended L-BFGS on parent ===
    print(f"\n[Phase 1] Extended L-BFGS on parent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_p1 = _run_lbfgs_extended(best_seq, deadline, min_time_remaining=550)
        update_best(a_p1, _score(a_p1), "lbfgs_p1")

    # === Phase 2: Newton active-set ===
    print(f"\n[Phase 2] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        n_deadline = min(deadline - 500, time.monotonic() + 250)
        a_newton, sc_newton = _newton_active_set(best_seq, n_deadline, max_iter=400, tol_schedule="adaptive")
        update_best(a_newton, sc_newton, "newton_p2")

    # === Phase 3: Subgradient polish ===
    print(f"\n[Phase 3] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=4000, lr_init=0.012)
        update_best(a_sg, sc_sg, "sg_p3")

    # === Phase 4: L-BFGS after Newton ===
    print(f"\n[Phase 4] L-BFGS post-Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_p4 = _run_lbfgs_extended(best_seq, deadline, min_time_remaining=300)
        update_best(a_p4, _score(a_p4), "lbfgs_p4")

    # === Phase 5: Shrinkage ===
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_shr, sc_shr = _shrinkage_optim(best_seq, deadline - 280)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score * 1.002:
            a_shr2 = _run_lbfgs_extended(a_shr, deadline, min_time_remaining=250)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # === Phase 6: Spectral restarts ===
    print(f"\n[Phase 6] Spectral restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        for i in range(5):
            if time.monotonic() > deadline - 180:
                break
            a_pert = _spectral_perturb(best_seq, rng, n_modes=rng.integers(3, 10),
                                       amp_range=(0.005, 0.08))
            a_opt = _run_lbfgs_extended(a_pert, deadline, min_time_remaining=150)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"spectral_{i}")
            if sc_opt < best_score * 1.003 and time.monotonic() < deadline - 150:
                a_opt2, sc_opt2 = _newton_active_set(a_opt, min(deadline - 150, time.monotonic() + 60), max_iter=80)
                update_best(a_opt2, sc_opt2, f"spectral_{i}_newton")

    # === Phase 7: Block swap restarts ===
    print(f"\n[Phase 7] Block swap restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        for i in range(4):
            if time.monotonic() > deadline - 130:
                break
            a_pert = _block_swap(best_seq, rng, n_swaps=rng.integers(2, 5))
            a_opt = _run_lbfgs_extended(a_pert, deadline, min_time_remaining=120)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"block_{i}")

    # === Phase 8: Mirror/antisymmetric perturbations ===
    print(f"\n[Phase 8] Mirror perturbations (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        for i in range(3):
            if time.monotonic() > deadline - 100:
                break
            a_pert = _mirror_antisym_perturb(best_seq, rng, strength=rng.uniform(0.05, 0.15))
            a_opt = _run_lbfgs_extended(a_pert, deadline, min_time_remaining=90)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"mirror_{i}")

    # === Phase 9: Multi-n exploration ===
    print(f"\n[Phase 9] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        a_opt = _run_lbfgs_extended(a_init, deadline, min_time_remaining=40)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _newton_active_set(a_opt, min(deadline - 30, time.monotonic() + 40), max_iter=60)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 10: Coordinate descent polish ===
    print(f"\n[Phase 10] Coordinate descent polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd, sc_cd = _coordinate_descent_polish(best_seq, deadline - 10, n_rounds=10)
        update_best(a_cd, sc_cd, "coord_desc")

    # === Phase 11: Final ultra-precision L-BFGS + Newton ===
    print(f"\n[Phase 11] Final precision phase (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        # Ultra-low beta L-BFGS
        beta_final = mb * 0.000005
        remaining = max(2.0, deadline - 4 - time.monotonic())
        if remaining > 5:
            max_iter = min(5000, max(200, int(remaining / 0.004)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                             'maxcor': 100, 'maxls': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra_lbfgs")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(best_seq, max(deadline - 2, time.monotonic()), max_iter=100)
            update_best(a_fin2, sc_fin2, "final_newton")

        # Final subgradient with very low learning rate
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 1, n_max_steps=2000, lr_init=0.003)
            update_best(a_fin3, sc_fin3, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
