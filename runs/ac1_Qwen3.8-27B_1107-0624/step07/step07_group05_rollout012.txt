def _next_pow2(n):

    while k < n:

    """Fast linear convolution using FFT."""

    rho = rho[-1]

    current_a = a0.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Compute initial gap for beta
    A_hat = np.fft.rfft(current_a, N1)
    c = np.fft.irfft(A_hat * A_hat, N1)[:2 * n - 1]
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, c.max() * 0.01)

    betas = [base_beta * f for f in [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001]]

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining * 0.8 / max(est_time_per_iter, 0.001))))

        try:
            result = minimize(
                _smoobj, z, args=(beta,),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 30, 'maxls': 50}
            )
            a_new = np.exp(result.x - np.max(result.x))
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_reinit(n, rng, strength=0.3):
    """
    Create a sequence whose power spectrum is designed to minimize peak autocorrelation.
    Uses a 'flat spectrum' target with small random perturbations.
    """
    N = _next_pow2(2 * n - 1)
    # Target: flat power spectrum (uniform |A|^2)
    # In time domain, this would be a delta, but we need a >= 0
    # Compromise: use a raised cosine spectrum
    freqs = np.arange(N // 2 + 1)
    # Power spectrum: raised cosine centered at DC with width ~n/2
    width = n * 0.6
    P = np.ones(N // 2 + 1)
    for f in range(len(P)):
        if f > 0:
            # Attenuate high frequencies to make time-domain signal smoother
            x = f / (N / 4)
            P[f] = np.exp(-x * x * 2.0)

    # Add random phase
    phases = rng.standard_normal(N // 2 + 1) * np.pi
    # DC component is real and positive
    phases[0] = 0
    phases[-1] = 0  # Nyquist

    A = P * np.exp(1j * phases)
    # Make sure it's valid for real signal
    a_complex = np.fft.irfft(A * np.sqrt(N), N)[:n]

    # Project to non-negative
    a = np.maximum(a_complex, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s

    # Blend with random for diversity
    a_rand = rng.exponential(1.0, n)
    a_rand = a_rand / a_rand.sum()
    a = (1 - strength) * a + strength * a_rand
    a = np.maximum(a, 0)
    a = a / a.sum()

    return a


def _fourier_perturb(a, rng, n_modes=8, amplitude=0.02):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            old = z_hat[freq].copy()
            z_hat[freq] += delta * np.exp(1j * phase)
            # Maintain Hermitian symmetry (rfft handles this)
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    a_new = np.exp(z_new - np.max(z_new))
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "double_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.3) ** 2) + np.exp(-40.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _upsample(a, new_n):
    """Upsample sequence from len(a) to new_n using spline-like interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _downsample(a, new_n):
    """Downsample sequence from len(a) to new_n."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _pressure_flatten(a, deadline, n_iters=500, gamma=0.3):
    """
    KKT-motivated multiplicative update to flatten pressure.
    At optimum, sum_{k in S} a_{k-i} should be constant for all i.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    N1 = _next_pow2(2 * n - 1)
    N2 = _next_pow2(3 * n - 2)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A_hat = np.fft.rfft(a, N1)
        c = np.fft.irfft(A_hat * A_hat, N1)[:2 * n - 1]
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Active set
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros(2 * n - 1, dtype=bool)
            active[idx] = True

        # Pressure: p_i = sum_{k in S} a_{k-i}
        s_padded = np.zeros(N2)
        s_padded[:2 * n - 1] = active.astype(np.float64)
        Y_hat = np.fft.rfft(a, N2)
        S_hat = np.fft.rfft(s_padded, N2)
        p = np.fft.irfft(Y_hat * np.conj(S_hat), N2)[:n]
        p = np.maximum(p, 1e-30)

        # Multiplicative update: a_i <- a_i * (mean(p)/p_i)^gamma
        p_mean = np.mean(p)
        ratio = (p_mean / p) ** gamma
        a = a * ratio
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def _coordinate_polish(a, deadline, n_rounds=50):
    """
    Local coordinate descent: adjust top-contributing coordinates.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    N1 = _next_pow2(2 * n - 1)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        A_hat = np.fft.rfft(a, N1)
        c = np.fft.irfft(A_hat * A_hat, N1)[:2 * n - 1]
        c_max = float(c.max())

        # Find which coordinates contribute most to the peak
        # Contribution of a_i to c_k is a_i * a_{k-i}
        # Total peak contribution: sum_{k near max} a_i * a_{k-i}
        tol = 0.1 * c_max
        active = c >= (c_max - tol)
        # Pressure tells us which coordinates to modify
        N2 = _next_pow2(3 * n - 2)
        s_padded = np.zeros(N2)
        s_padded[:2 * n - 1] = active.astype(np.float64)
        Y_hat = np.fft.rfft(a, N2)
        S_hat = np.fft.rfft(s_padded, N2)
        contrib = np.fft.irfft(Y_hat * np.conj(S_hat), N2)[:n] * a

        # Try reducing top contributors
        n_try = min(n, max(20, n // 32))
        top_idx = np.argsort(contrib)[-n_try:]
        np.random.shuffle(top_idx)

        improved = False
        for i in top_idx[:min(n_try, 30)]:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            # Try reducing
            for factor in [0.7, 0.85, 0.95]:
                a[i] = old_val * factor
                A_hat = np.fft.rfft(a, N1)
                c_try = np.fft.irfft(A_hat * A_hat, N1)[:2 * n - 1].max()
                if c_try < c_max - 1e-15:
                    improved = True
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining Nesterov subgradient, L-BFGS smoothing,
    pressure flattening, coordinate descent, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # === Phase 1: Nesterov subgradient on parent ===
    print(f"\n[Phase 1] Nesterov subgradient on n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    t_phase1_end = deadline - 600.0
    if time.monotonic() < t_phase1_end:
        a_nest = _nesterov_subgrad(parent, t_phase1_end, max_iters=200000, eta=0.8,
                                   tol_init=0.03, tol_min=0.001)
        _update(a_nest, _score(a_nest))
        print(f"  Nesterov done: score={_score(a_nest):.10f}")

    # === Phase 2: Pressure flattening ===
    print(f"\n[Phase 2] Pressure flattening (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_pf = _pressure_flatten(best_seq, min(deadline - 350, time.monotonic() + 90),
                                 n_iters=300, gamma=0.25)
        _update(a_pf, _score(a_pf))
        print(f"  Pressure flatten done: score={_score(a_pf):.10f}")

    # === Phase 3: L-BFGS smoothing ===
    print(f"\n[Phase 3] L-BFGS smoothed optimization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_lb = _smoothed_lbfgs(best_seq, deadline, min_remaining=150)
        _update(a_lb, _score(a_lb))
        print(f"  L-BFGS done: score={_score(a_lb):.10f}")

    # === Phase 4: Another Nesterov pass (tighter tol) ===
    print(f"\n[Phase 4] Fine Nesterov (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_nest2 = _nesterov_subgrad(best_seq, min(deadline - 250, time.monotonic() + 120),
                                    max_iters=100000, eta=0.1, tol_init=0.005, tol_min=0.0001)
        _update(a_nest2, _score(a_nest2))
        print(f"  Fine Nesterov done: score={_score(a_nest2):.10f}")

    # === Phase 5: Coordinate polish ===
    print(f"\n[Phase 5] Coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_cd = _coordinate_polish(best_seq, min(deadline - 80, time.monotonic() + 60), n_rounds=40)
        _update(a_cd, _score(a_cd))
        print(f"  Coord polish done: score={_score(a_cd):.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_to_try = [4096, 12288, 16384]
    # Filter out parent n
    n_to_try = [x for x in n_to_try if x != n_parent]
    n_to_try.sort(reverse=True)  # Largest first (more interesting)

    for n_try in n_to_try:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - 80 - time.monotonic()
        if remaining < 20:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.1f}s)")

        # Initialize from best by interpolation
        if best_seq is not None:
            a_init = _upsample(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "gaussian")

        # Quick Nesterov
        t_end = time.monotonic() + min(remaining * 0.5, 60.0)
        a_opt = _nesterov_subgrad(a_init, t_end, max_iters=50000, eta=0.5,
                                  tol_init=0.03, tol_min=0.002)
        sc_opt = _score(a_opt)
        _update(a_opt, sc_opt)
        print(f"    n={n_try}: score={sc_opt:.10f}")

        # If good, follow up with L-BFGS
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_lb2 = _smoothed_lbfgs(a_opt, deadline, min_remaining=25)
            _update(a_lb2, _score(a_lb2))

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restarts = [
            ("spectral", lambda: _spectral_reinit(n_best, rng, strength=0.4)),
            ("gaussian", lambda: _make_profile(n_best, "gaussian")),
            ("double_hump", lambda: _make_profile(n_best, "double_hump")),
            ("perturbed", lambda: _fourier_perturb(best_seq, rng, n_modes=12, amplitude=0.03)),
        ]
        for name, gen_fn in restarts:
            if time.monotonic() > deadline - 20:
                break
            try:
                a_r = gen_fn()
            except Exception:
                continue
            # Blend with current best
            a_blend = 0.5 * best_seq + 0.5 * a_r
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()

            t_end = min(deadline - 15, time.monotonic() + 30)
            a_opt = _nesterov_subgrad(a_blend, t_end, max_iters=30000, eta=0.3,
                                      tol_init=0.02, tol_min=0.002)
            _update(a_opt, _score(a_opt))

    # === Phase 8: Final polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very fine Nesterov
        a_fin = _nesterov_subgrad(best_seq, deadline - 1, max_iters=50000, eta=0.01,
                                  tol_init=0.001, tol_min=0.0001)
        _update(a_fin, _score(a_fin))

        # Pressure flatten
        if time.monotonic() < deadline - 1:
            a_fin2 = _pressure_flatten(best_seq, deadline - 0.5, n_iters=100, gamma=0.1)
            _update(a_fin2, _score(a_fin2))

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
