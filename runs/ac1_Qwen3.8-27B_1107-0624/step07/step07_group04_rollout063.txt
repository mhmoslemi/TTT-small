The active set Newton method is essentially this: at each step, identify the active constraints (those k where (a*a)[k] is close to max), and solve the QP:

(2/|S|) H a + lambda * 1 = 0, where H is the Hessian matrix

So H a = -(lambda|S|/2) * 1

We solve for a using the linear system H a = rhs, where rhs is proportional to 1.

"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: extended n exploration, preconditioned CG-Newton, adaptive
active set, more aggressive beta continuation, diverse initializations,
symmetry exploitation, and better time management.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_precond(W, n, rhs, max_iter=500, tol=1e-14):
    """Preconditioned CG for H*x = rhs with diagonal preconditioner from row sums."""
    n_active = int(W.sum() > 0)
    if n_active == 0:
        return np.zeros(n)
    # Preconditioner: approximate diagonal from row sums
    # row_sum[j] = 2 * sum_{m=0}^{n-1} W[j+m]
    # Compute efficiently via prefix sums
    W_prefix = np.concatenate([[0.0], np.cumsum(W)])
    row_sums = np.zeros(n)
    for j in range(n):
        lo = j
        hi = min(j + n, len(W))
        row_sums[j] = 2.0 * (W_prefix[hi] - W_prefix[lo])
    row_sums = np.maximum(row_sums, 1e-30)
    diag_precond = 1.0 / row_sums

    x = np.zeros(n)
    r = rhs.copy()
    z = diag_precond * r
    p = z.copy()
    rz = float(r @ z)
    rz0 = rz
    if abs(rz0) < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Fallback: unpreconditioned step
            alpha = rz / max(float(r @ r), 1e-30)
            x = x + alpha * r
            return x
        alpha = rz / pHp
        x = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rz0 * 0.5:
            break
        z_new = diag_precond * r_new
        rz_new = float(r_new @ z_new)
        if abs(rz_new) < tol * abs(rz0):
            break
        beta = rz_new / max(rz, 1e-30)
        p = z_new + beta * p
        r = r_new
        rz = rz_new
    return x


def _smoobj(z_val, beta_val, n_val):
    a = _softmax(z_val)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta_val
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n_val * c_max
    return 2.0 * n_val * (c_max + beta_val * np.log(s))


def _smograd(z_val, beta_val, n_val):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    wc = _full_convolve(w, a_rev)
    h = wc[n_val - 1:2 * n_val - 1]
    W = float(np.dot(w, c))

def _newton_polish(a_init, deadline, max_rounds=80):
    """Active set Newton with preconditioned CG and adaptive tolerance."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: tighten as we converge
        if rnd < 5:
            tol_frac = 0.002
        elif rnd < 20:
            tol_frac = 0.0005
        else:
            tol_frac = 0.0001
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        # Compute gradient and Newton direction
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * np.sqrt(n):
            break

        # Newton step via preconditioned CG
        Hg = _cg_solve_precond(W, n, g, max_iter=300)
        H1 = _cg_solve_precond(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.sum(Hg)) / float(np.sum(H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Damped step with backtracking
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.01 * alpha * gn * 0.001:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback
            alpha = 0.02
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    step_done = True
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient with adaptive step and periodic Newton refresh."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015

    for i in range(max_steps):

            break

        c = _full_convolve(a, a)

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        grad = 4.0 * n * a * (h - W)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 30 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.01
        z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    # More aggressive schedule with 30 steps
    ratios = [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.025, 0.015,
              0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0007, 0.0004,
              0.0002, 0.0001, 0.00007, 0.00004, 0.00002, 0.00001,
              0.000007, 0.000004, 0.000002, 0.000001, 0.0000007, 0.0000004]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.004 * max(1.0, n / 4096.0)
        # More iterations at small beta (final polish is more important)
        if i < 10:
            max_iter = min(2000, max(30, int(remaining / (est_iter_time * 20))))
        else:
            max_iter = min(3000, max(50, int(remaining / (est_iter_time * 15))))

        obj_func = _make_smoothed_obj(float(beta), n)
        grad_func = _make_smoothed_grad(float(beta), n)

        try:
            result = minimize(
                obj_func,
                z,
                jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13,

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.8
    elif kind == "tapered2":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "bimodal":
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.02
    elif kind == "hamming":
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "hanning":
        a = 0.5 - 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "flat_top":
        # Flat-top window approximation
        a = 0.2156 - 0.4160 * np.cos(2 * np.pi * t) + 0.2786 * np.cos(4 * np.pi * t) \
            - 0.0835 * np.cos(6 * np.pi * t) + 0.0069 * np.cos(8 * np.pi * t)
        a = np.abs(a) + 0.01
    else:
        a = np.sin(np.pi * t) ** 0.8
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _resample_symmetric(a, n_new):
    """Resample and enforce symmetry."""
    a_r = _resample(a, n_new)
    a_sym = (a_r + a_r[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_sym / s

            ("perturb_small", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_med", lambda: _perturb(best_seq, 0.08, rng)),
            ("shrink_perturb", None),  # handled specially
            ("hamming", lambda: _make_profile(n_best, "hamming")),
            ("flat_top", lambda: _make_profile(n_best, "flat_top")),
        ]
        for label, gen in restart_configs:
            if time.monotonic() > deadline - 25:
                break
            if label == "shrink_perturb":
                p_shrunk = best_seq.copy()
                n_shrink = len(p_shrunk) // 5
                p_shrunk[np.argsort(p_shrunk)[:n_shrink]] = 0
                p_shrunk = p_shrunk / p_shrunk.sum()
                p_start = _perturb(p_shrunk, 0.04, rng)
            else:
                try:
                    p_start = gen()
                except Exception:
                    continue

            t6_budget = min(40.0, deadline - 15.0)
            a_r = _run_lbfgs(p_start, t_start + t6_budget, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_polish(a_r, min(t_start + 25.0, deadline - 10.0), max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{label}_np")

    # Phase 7: Shrinkage
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a7, sc7 = _shrink_and_reopt(best_seq, deadline, min_time_remaining=20.0)
        update_best(a7, sc7, "shrinkage")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Ultra-tight Newton
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 3.0, max_rounds=120)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS with tiny beta
        if time.monotonic() < deadline - 4:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-8
            remaining = deadline - 4 - time.monotonic()
            if remaining > 3:
                max_iter = min(2000, max(50, int(remaining / 0.03)))
                obj_func = _make_smoothed_obj(beta_fin, n)
                grad_func = _make_smoothed_grad(beta_fin, n)
                try:
                    result = minimize(
                        obj_func,
                        z,
                        jac=grad_func,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 100, 'maxls': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

        # One more Newton pass
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _newton_polish(best_seq, deadline - 1.0, max_rounds=50)
            update_best(a_fin2, sc_fin2, "final_newton2")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        # Validate with evaluate_sequence if available
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
