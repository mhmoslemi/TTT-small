import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _max_conv(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_gradient(a, lam, n):
    """
    Compute the gradient of 2*n*lam*log(sum exp(C/lam)) w.r.t. z (where a=softmax(z)).
    Returns gradient vector of length n.
    """
    c = _full_convolve(a, a)
    # Soft weights
    c_shifted = (c - c.max()) / lam
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smooth_objective(z, lam, n):
    """Smooth objective in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / lam
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + lam * np.log(s))

def _lbfgs_smoothed(z0, lam, n, max_iter, deadline):
    """L-BFGS on the smoothed objective in z-space."""

    def fun(z):
        if time.monotonic() > deadline:
            return _smooth_objective(z, lam, n)
        return _smooth_objective(z, lam, n)

        return _soft_gradient(_softmax(z), lam, n)

                                   'maxls': 100, 'disp': False})

def _adaptive_lbfgs(a_start, deadline, reserve=60.0):

    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-18))

                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve > deadline:
            break
        remaining = deadline - reserve - time.monotonic()
        if remaining < 10:
            break

        # Estimate cost per iteration

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    # === Phase 3: Main optimization loop ===
    print(f"\n=== Phase 3: Main optimization (t={time.monotonic()-t_start:.1f}s) ===")

    n_work = len(a_work)
    z_work = np.log(np.maximum(a_work, 1e-18))
    z_work = z_work - z_work.mean()

    # Soft subgradient with momentum - main workhorse
    # Use multiple passes with decreasing lambda
    mc = _max_conv(a_work)
    lam_schedule = mc * np.array([0.05, 0.02, 0.01, 0.005, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5])

    phase3_deadline = deadline - 300.0
    per_pass_time = max(10.0, (phase3_deadline - time.monotonic()) / len(lam_schedule))

    for i, lam in enumerate(lam_schedule):
        if time.monotonic() > phase3_deadline:
            break
        pass_deadline = min(phase3_deadline, time.monotonic() + per_pass_time * 1.5)

        a_work, sc_work, z_work = _soft_subgrad_optimize(
            z_work, n_work, pass_deadline,
            lam_init=lam, lam_min=lam * 0.3,
            momentum=0.9, max_iter=100000, ls_init=0.5
        )
        try_update(a_work, f"soft_sg_lam{i}")
        print(f"  Soft subgrad pass {i}: lam={lam:.2e}, score={sc_work:.10f}")

    # === Phase 4: L-BFGS refinement ===
    print(f"\n=== Phase 4: L-BFGS refinement (t={time.monotonic()-t_start:.1f}s) ===")
    a_lbfgs = _adaptive_lbfgs(best_seq, deadline, reserve=200.0)
    try_update(a_lbfgs, "lbfgs_main")

    # === Phase 5: More soft subgradient rounds ===
    print(f"\n=== Phase 5: Additional soft subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        n_work = len(best_seq)
        z_work = np.log(np.maximum(best_seq, 1e-18))
        z_work = z_work - z_work.mean()
        mc = _max_conv(best_seq)
        a_s5, sc_s5, z_work = _soft_subgrad_optimize(
            z_work, n_work, deadline - 150,
            lam_init=mc * 0.001, lam_min=mc * 1e-5,
            momentum=0.92, max_iter=200000, ls_init=0.3
        )
        try_update(a_s5, "soft_sg_round2")

    # === Phase 6: Diversification ===
    print(f"\n=== Phase 6: Diversification (t={time.monotonic()-t_start:.1f}s) ===")
    n_divers = 8
    div_time = max(5.0, (deadline - time.monotonic() - 100.0) / n_divers)

    if time.monotonic() < deadline - 100:
        for d in range(n_divers):
            if time.monotonic() > deadline - 100:
                break
            a_opt, sc_opt, a_pert, sc_pert = _perturb_and_optimize(
                best_seq, len(best_seq), rng,
                deadline - 100, div_time * 0.8
            )
            try_update(a_opt, f"diverse_{d}")
            print(f"  Diverse {d}: perturbed={sc_pert:.10f}, optimized={sc_opt:.10f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length exploration (t={time.monotonic()-t_start:.1f}s) ===")
    lengths_to_try = [4096, 8192, 16384, 32768, 6144, 12288]
    tried_n = set([len(best_seq)])

    if time.monotonic() < deadline - 80:
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 80:
                break
            if target_n in tried_n:
                continue
            tried_n.add(target_n)

            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"  n={target_n}: interpolated score={sc_interp:.10f}")

            if time.monotonic() < deadline - 60:
                # Quick soft subgradient at new length
                n_t = target_n
                z_t = np.log(np.maximum(a_interp, 1e-18))
                z_t = z_t - z_t.mean()
                mc_t = _max_conv(a_interp)
                sub_dl = min(deadline - 60, time.monotonic() + 40.0)
                a_t, sc_t, _ = _soft_subgrad_optimize(
                    z_t, n_t, sub_dl,
                    lam_init=mc_t * 0.02, lam_min=mc_t * 0.001,
                    momentum=0.9, max_iter=30000, ls_init=0.4
                )
                try_update(a_t, f"n_{target_n}_sg")

                # L-BFGS at new length
                if time.monotonic() < deadline - 60:
                    a_t2 = _adaptive_lbfgs(a_t, deadline, reserve=40.0)
                    try_update(a_t2, f"n_{target_n}_lbfgs")

    # === Phase 8: Final intensive polishing ===
    print(f"\n=== Phase 8: Final polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 5:
        n_fin = len(best_seq)
        z_fin = np.log(np.maximum(best_seq, 1e-18))
        z_fin = z_fin - z_fin.mean()
        mc_fin = _max_conv(best_seq)

        # Very tight soft subgradient
        a_f1, sc_f1, z_fin = _soft_subgrad_optimize(
            z_fin, n_fin, deadline - 2,
            lam_init=mc_fin * 1e-4, lam_min=mc_fin * 1e-7,
            momentum=0.95, max_iter=100000, ls_init=0.2
        )
        try_update(a_f1, "final_sg_tight")

        # Final L-BFGS with very small beta
        if time.monotonic() < deadline - 2:
            a_f2 = _adaptive_lbfgs(best_seq, deadline, reserve=2.0)
            try_update(a_f2, "final_lbfgs")

    # === Final ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
