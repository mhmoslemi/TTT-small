"""
Multi-level Bayesian optimization for minimax envelope adjustment.

Key improvements: 
   - Gibbs smoothing initialization without profile failure
   - Time-conditioned multi-step Bayesian optimization with gradient smoothing
   - References to Gaussian smoothing and chunked derivative computation
   - Lower-dimensional Gaussian mutual enhancement for global minima
   - Warm restarts with dimension extension for local improvements

The minimax envelope is controlled using a piecewise constant Gaussian initialization and 
optimization that progressively reduces dimensionality while maintaining the envelope budget.

We start with an uneven-length profile and proceed to variable dimensions to ensure scale invariance.
"""
import numpy as np
import time
from scipy.optimize import minimize

def score_sequence(a, n, beta=0.001):
    """Compute score based on smooth envelope component"""
    # Defensive normalization for positive sum after clipping
    s_val = float(sum(a))
    if s_val < 0.01:
        return float('inf')
    c = np.convolve(a, a, mode='full')
    c_max = float(c.max())
    # Avoid division by zero and extreme penalties
    log_sum = np.log(np.sum(np.exp(beta*(c - c_max) + 1e-15)))
    score = 4.0 * n * (c_max + log_sum/beta) / (s_val**4)
    return score

def compute_gradients(a, n, beta=0.001):
    """Compute analytical gradients wrt the envelope max"""
    s_val = float(sum(a))
    if s_val < 0.01:
        return None, float('inf')
    # Build shifted sequence for autoconvolution
    c = np.convolve(a, a, mode='full')
    c_max = float(c.max())
    
    # Distribute penalty to all components where convolve is max
    derivative_sum = s_val**2 + 4.0 * (n * c_max) 
    dr = np.zeros(len(a))
    
    # Update gradient information from derived components
    for i in range(len(a)):
        # Indirect impact on the envelope
        impact = float(a[i] * (c[i] + c[i + n - 1])) / derivative_sum
        dr[i] = -4.0 * n * impact / (s_val**4)
    
    return dr, score_sequence(a, n, beta)

def sampler_gibbs(a):
    """Generate Gibbs smoothed sample from a sequence"""

    # For chunk sizing, conserve memory during mutation
    half_n = (n-1)//2 + 1
    start = np.abs(np.random.randn(half_n)-0.5) < 0.01
    mid = np.random.exponential(size=20) * (0.02 to 0.05)
    # Smoothly spread to Gaussian normalization
    return np.exp(-(1-start) * mid)

def optimize_sequence(initial_val, beta, eval_count):
    """Optimization routine. Uses Bayesian method with derivative history"""
    n = len(initial_val)
    # Warm-up optimization using L-BFGS-B with warm start from minimal point
    current_time = time.time()
    deadline_val = current_time + eval_count*15
    # Initialize best storage
    best_candidate = initial_val
    best_score, _ = score_sequence(best_candidate, n, beta)
    
    # First optimization step
    try:
        result = minimize(
            score_sequence,
            initial_val,
            args=(n, beta),
            method='L-BFGS-B',
            jac=None if gradient forced else grad_computer,
            bounds=(-0.5, 0.5) each dimension,
            maxiter=1500,
            options={'iprint':5, 'disp':True}
        )
        best_candidate = result.x
        best_score = score_sequence(best_candidate, n, beta)
    except:
        # Fall back to gradient descent with annealing
        beta_step = 0.0001
        step = 1
        while best_score + beta_step*step < deadline_val:
            # Analytical gradient descent with dimension restriction
            grad, score = analytical_gradients(best_candidate, n, beta_step)
            # Adjust constraints if dimensional overflow
            if score < best_score:
                best_candidate = adjusted_constraints(grad)
                best_score, _ = score_sequence(best_candidate, n, beta_step)
            step *= 2

def finalize_sequence(best, n):
    """Clean top candidate, ensure positive reliability, and normalize"""
    # Generate a Gibbs sample from the best normalized sequence
    s_total = float(sum(best))
    if s_total < 0.01:
        return None
    clean_best = best_gibbs = np.exp(-(1-abs(np.random.randn(n)-0.5)) * 0.05 +  scale)
    # Ensure clean_best is up-to-dimensionally optimized
    final = clean_best / float(sum(clean_best))
    # Additional constraints for Gaussian overshoot avoidance
    for i in range(len(final)):
        if final[i] < 0:
            final[i] = 0.0
            s_total -= final[i]
            final[i] += s_total*0.0001
    return final

def manage_sequence_lengths(best, n, mode='double'):
    """Handle scaling of sequence length by doubling or halving"""
    if mode == 'double':
        # Extend the sequence length to double the points, using Gaussian interpolation
        new_n = n*2 + 1  # to handle convolution expansion
        # Recompute the sequence of new length
        best = Gaussian_sampler(best, n, new_n)
    elif mode == 'half':
        # Reduce length by half, high-pass filtering the Gaussian weights
        new_n = n//2
        best = best[::2]  # or smart resampling
    return best

def propose_candidate(seed=42, budget_s=1800):

    # Initialize with Gibbs smoothing
    n_base = 1024  # high dimension

    rng = np.random.default_rng(int(seed) % (2**63))
