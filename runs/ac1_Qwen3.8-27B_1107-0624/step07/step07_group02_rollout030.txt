"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Frank-Wolfe, active-set gradient descent with Nesterov momentum,
log-sum-exp smoothed L-BFGS, structured initializations (window functions,

Key difference from prior: Frank-Wolfe as primary optimizer, Nesterov momentum,
and paired-coordinate polish.

import numpy as np
import time
import math
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) is already 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_half_sine(n):
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.sin(np.pi * (t + 0.5 / n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.25):
    t = np.linspace(0, 1, n, endpoint=False)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_chirp(n, chirp_rate=1.0):

    a = 1.0 + 0.4 * np.cos(2 * np.pi * chirp_rate * t * t / n)

def _make_quadratic_residue(n):
    """Use quadratic residues as a structured initialization."""
    # Find a prime near n
    def is_prime(x):
        if x < 2:
            return False
        if x % 2 == 0:
            return x == 2
        for i in range(3, int(math.isqrt(x)) + 1, 2):
            if x % i == 0:
                return False
        return True

    p = n
    while p > 2 and not is_prime(p):
        p -= 2

    if p < 17:
        return _make_half_sine(n)

    qr = set()
    for x in range(1, p):
        qr.add((x * x) % p)
    # Include 0 as a residue
    qr.add(0)

    a = np.zeros(n)
    for i in range(min(n, p)):
        if i in qr:
            a[i] = 1.0
        else:
            a[i] = 0.1  # small floor to keep it dense
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _lbfgs_smoothed(a_init, deadline, min_remaining=30.0):
    """L-BFGS on log-softmax parameterization with adaptive beta annealing."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.15,
                                  0.1, 0.07, 0.05, 0.03, 0.02, 0.01,
                                  0.006, 0.003, 0.0015, 0.0008, 0.0004,
                                  0.0002, 0.0001])

    current_a = a.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 2048.0)
        # Allocate proportional time to each beta
        frac_time = max(0.02, remaining / (len(betas) - i))
        max_iter = min(5000, max(30, int(frac_time / (est_time_per_iter * 10))))

        def obj(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            c_max = float(c_.max())
            shifted = (c_ - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad(z_):
            a_ = _softmax(z_)
            c_ = _full_convolve(a_, a_)
            c_max = float(c_.max())
            shifted = (c_ - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e / s
            a_rev = a_[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_))
            return 4.0 * n * a_ * (h - W)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _frank_wolfe(a_init, deadline, max_iter=50000):
    """Frank-Wolfe algorithm for the normalized problem."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    t_start = time.monotonic()
    step = 2.0 / (t_start + 1)  # standard FW step size (will adjust)
    momentum = 0.0
    prev_grad = None

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Adaptive active set
        tol = max(c_max * 0.01, 1e-10)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient (subgradient of max at active points)
        # grad[m] = (2*n/n_active) * sum_{k in active} 2*a[k-m]
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # a[k-m] for m in [lo, hi] is a[k-zi] for zi in [k-hi, k-lo] = [k-hi, lo]
            # Actually: grad[m] += 2*a[k-m] for m where 0<=k-m<n, i.e., k-n+1<=m<=k
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            # a[k-m] for m in [m_lo, m_hi]: index k-m ranges from k-m_hi to k-m_lo
            idx_lo = k - m_hi
            idx_hi = k - m_lo
            grad[m_lo:m_hi + 1] += 2.0 * a[idx_lo:idx_hi + 1][::-1]

        grad *= (2.0 * n / n_active)

        # Add negative gradient of the normalization (for the -2*max/s^2 term)
        # Since we're working with sum=1, this term is: -2*n*c_max*2*a / s^2 = -4*n*c_max*a (s=1)
        # But actually the score is 2n*max/s^2, d/d[a[m]] = 2n*(dmax/d[a[m]]/s^2 - 2*max/s^3 * 1)
        # With s=1: = 2n*(grad_raw[m] - 2*c_max)
        # Where grad_raw[m] = (1/n_active)*sum_{k active} 2*a[k-m]
        # So the full gradient is: 2n * ((1/n_active)*sum 2a[k-m] - 2*c_max)
        # Let me recompute properly

        # Actually for the score 2n*max_conv/s^2 with s=sum(a):
        # d(score)/d[a[m]] = 2n * [s^2 * dmax/d[a[m]] - max_conv * 2s] / s^4
        #                   = 2n * [dmax/d[a[m]]/s - 2*max_conv/s^2]
        # With s=1: = 2n * [dmax/d[a[m]] - 2*max_conv]
        # dmax/d[a[m]] = (1/n_active) * sum_{k in active} 2*a[k-m]
        # So grad[m] = 2n * [(1/n_active)*sum_{k} 2*a[k-m] - 2*c_max]
        #            = 4n * [(1/n_active)*sum_{k} a[k-m] - c_max]

        # Let me recompute
        raw_grad = np.zeros(n)
        for k in np.where(active)[0]:
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            idx_lo = k - m_hi
            idx_hi = k - m_lo
            raw_grad[m_lo:m_hi + 1] += a[idx_lo:idx_hi + 1][::-1]

        raw_grad /= n_active
        full_grad = 4.0 * n * (raw_grad - c_max)

        # Nesterov momentum
        if prev_grad is not None and momentum > 0:
            full_grad = full_grad + momentum * (full_grad - prev_grad)
        prev_grad = full_grad.copy()

        # Adaptive momentum (Nesterov)
        momentum = 0.9 * (it + 1) / (it + 1 + 25)  # slowly approaches 0.9

        # Frank-Wolfe: find vertex (min gradient coordinate)
        v = int(np.argmin(full_grad))

        # Standard FW step size: t_k = 2/(k+2)
        # But we can also use line search
        t_fw = 2.0 / (it + 2)

        # Try FW step
        a_try = (1 - t_fw) * a + t_fw * np.zeros(n)
        a_try[v] = 1.0  # vertex is e_v
        # Actually: a_try = (1-t)*a + t*e_v
        a_try = (1.0 - t_fw) * a
        a_try[v] += t_fw
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                continue

        # Gradient descent step (alternative)
        gnorm = np.linalg.norm(full_grad)
        if gnorm > 1e-15:
            # Backtracking line search
            alpha = 1.0 / (gnorm + 1e-15)
            improved = False
            for _ in range(30):
                a_try = a - alpha * full_grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                # Try a smaller step
                alpha = 1e-6 / (gnorm + 1e-15)
                a_try = a - alpha * full_grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < current_sc:
                        a = a_try

        # Decay momentum if stuck
        if it % 100 == 99 and not improved:
            momentum *= 0.8

    return best_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.005,
                       tol_frac_end=0.00001, lr_init=0.03, use_momentum=True):
    """Active-set gradient descent with Nesterov momentum and adaptive tolerance."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0
    v = np.zeros(n)  # momentum velocity
    momentum = 0.85

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient
        raw_grad = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            idx_lo = k - m_hi
            idx_hi = k - m_lo
            raw_grad[m_lo:m_hi + 1] += a[idx_lo:idx_hi + 1][::-1]

        raw_grad /= n_active
        # Full gradient of score (with sum=1 normalization)
        grad = 4.0 * n * (raw_grad - c_max)

        # Center (remove mean to maintain sum constraint approximately)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-16:
            break

        # Nesterov momentum
        if use_momentum:
            v = momentum * v + (1.0 - momentum) * grad
            step_dir = v
        else:
            step_dir = grad

        # Backtracking line search
        alpha = lr / (np.linalg.norm(step_dir) + 1e-15)
        alpha = min(alpha, 1.0)
        improved = False
        for _ in range(50):
            a_try = a - alpha * step_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
                if use_momentum:
                    v *= 0.5
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.01)

    return best_a, _score(_normalize(best_a))


def _paired_coordinate_polish(a_init, deadline):
    """Polish by moving mass between pairs of coordinates."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Find the active peak position
    peak_pos = int(np.argmax(c))

    # Strategy: for each position m, try moving a small amount to position m'
    # where the gradient effect is most favorable
    # grad[m] ~ a[peak_pos - m] (reflected)
    # We want to move mass from high-grad positions to low-grad positions

    raw_grad = np.zeros(n)
    m_lo = max(0, peak_pos - (n - 1))
    m_hi = min(n - 1, peak_pos)
    idx_lo = peak_pos - m_hi
    idx_hi = peak_pos - m_lo
    raw_grad[m_lo:m_hi + 1] = a[idx_lo:idx_hi + 1][::-1]

    # Sort by gradient: high grad = bad (contributes to peak), low grad = good
    order = np.argsort(raw_grad)
    n_top = max(1, n // 10)
    n_bot = max(1, n // 10)
    top_indices = order[-n_top:]  # high gradient (should reduce)
    bot_indices = order[:n_bot]   # low gradient (should increase)

    # Try moving mass from top to bottom
    epsilons = [0.001, 0.0005, 0.0002, 0.0001, 5e-5]

    for eps in epsilons:
        if time.monotonic() > deadline:
            break
        a_try = a.copy()
        # Move mass from top to bottom (distributed)
        total_move = 0.0
        for m in top_indices:
            move = min(eps * a[m], a[m] * 0.1)
            a_try[m] -= move
            total_move += move
        if total_move < 1e-15:
            continue
        # Distribute to bottom
        for m in bot_indices:
            a_try[m] += total_move / len(bot_indices)

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            c = _full_convolve(a, a)
            peak_pos = int(np.argmax(c))
            # Recompute gradient
            raw_grad = np.zeros(n)
            m_lo = max(0, peak_pos - (n - 1))
            m_hi = min(n - 1, peak_pos)
            idx_lo = peak_pos - m_hi
            idx_hi = peak_pos - m_lo
            raw_grad[m_lo:m_hi + 1] = a[idx_lo:idx_hi + 1][::-1]
            order = np.argsort(raw_grad)
            top_indices = order[-n_top:]
            bot_indices = order[:n_bot]

    return best_a, _score(_normalize(best_a))


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _local_search_kicks(a_init, deadline, n_kicks=5, kick_sigma=0.05):
    """Try random perturbations and optimize each briefly."""
    n = len(a_init)
    rng = np.random.default_rng(12345)
    best_a = _normalize(a_init.copy())
    best_sc = _score(best_a)

    for i in range(n_kicks):
        if time.monotonic() > deadline:
            break
        a_kick = _perturb(best_a, kick_sigma * (1 + i * 0.3), rng)
        # Quick optimization
        kick_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / (n_kicks - i))
        a_opt = _lbfgs_smoothed(a_kick, kick_deadline, min_remaining=5.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Kick {i+1} improved: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search:
    P1: Diverse initializations + quick L-BFGS
    P2: Deep active-set polish with momentum
    P3: Frank-Wolfe refinement
    P4: Paired coordinate polish
    P5: Multi-n exploration
    P6: Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    n_parent = 8192
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                parent_sc = _score(parent)
                best_seq = parent.copy()
                best_score = parent_sc
                print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_half_sine(n_parent)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, half-sine n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Diverse initializations ===
    print(f"\n[P1] Diverse initializations (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 200)

    init_methods = [
        ("half_sine", lambda n: _make_half_sine(n)),
        ("raised_cos", lambda n: _make_raised_cosine(n)),
        ("gaussian", lambda n: _make_gaussian(n, 0.25)),
        ("chirp", lambda n: _make_chirp(n, 1.5)),
        ("quadratic_res", lambda n: _make_quadratic_residue(n)),
        ("bimodal", lambda n: _make_bimodal_gaussian(n)),
        ("periodic", lambda n: _make_periodic_mix(n)),
        ("tapered_03", lambda n: _make_tapered_power(n, 0.3)),
        ("tapered_1_0", lambda n: _make_tapered_power(n, 1.0)),
    ]

    for name, func in init_methods:
        if time.monotonic() > p1_deadline:
            break
        a_init = func(n_parent)
        sc_init = _score(a_init)
        if sc_init < best_score:
            best_seq = a_init.copy()
            best_score = sc_init
            print(f"  {name}: init score={sc_init:.10f} NEW BEST")
        # Quick L-BFGS on this init
        quick_deadline = min(p1_deadline, time.monotonic() + 15)
        if time.monotonic() < p1_deadline - 10:
            a_opt = _lbfgs_smoothed(a_init, quick_deadline, min_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  {name}: opt score={sc_opt:.10f} NEW BEST")

    # Also optimize the parent
    print(f"\n[P1b] Optimizing parent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_parent_opt = _lbfgs_smoothed(parent, min(deadline - 500, time.monotonic() + 120),
                                       min_remaining=10.0)
        sc_parent = _score(a_parent_opt)
        if sc_parent < best_score:
            best_seq = a_parent_opt.copy()
            best_score = sc_parent
            print(f"  Parent optimized: {best_score:.10f} NEW BEST")

    # === Phase 2: Deep active-set polish ===
    print(f"\n[P2] Active-set polish with momentum (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, time.monotonic() + 250)
    a_pol, sc_pol = _active_set_polish(best_seq, p2_deadline,
                                       tol_frac_start=0.008, tol_frac_end=0.00003,
                                       lr_init=0.04, use_momentum=True)
    if sc_pol < best_score:
        best_seq = a_pol.copy()
        best_score = sc_pol
        print(f"  Active-set polished: {best_score:.10f} NEW BEST")

    # Second round of polish (finer)
    if time.monotonic() < deadline - 300:
        p2b_deadline = min(deadline - 300, time.monotonic() + 150)
        a_pol2, sc_pol2 = _active_set_polish(best_seq, p2b_deadline,
                                             tol_frac_start=0.003, tol_frac_end=0.00001,
                                             lr_init=0.015, use_momentum=True)
        if sc_pol2 < best_score:
            best_seq = a_pol2.copy()
            best_score = sc_pol2
            print(f"  Fine polish: {best_score:.10f} NEW BEST")

    # === Phase 3: Frank-Wolfe ===
    print(f"\n[P3] Frank-Wolfe (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        fw_deadline = min(deadline - 250, time.monotonic() + 100)
        a_fw = _frank_wolfe(best_seq, fw_deadline)
        sc_fw = _score(a_fw)
        if sc_fw < best_score:
            best_seq = a_fw.copy()
            best_score = sc_fw
            print(f"  Frank-Wolfe: {best_score:.10f} NEW BEST")

    # === Phase 4: Paired coordinate polish ===
    print(f"\n[P4] Paired coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pc_deadline = min(deadline - 200, time.monotonic() + 60)
        a_pc, sc_pc = _paired_coordinate_polish(best_seq, pc_deadline)
        if sc_pc < best_score:
            best_seq = a_pc.copy()
            best_score = sc_pc
            print(f"  Paired coord: {best_score:.10f} NEW BEST")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([4096, 5120, 6144, 8192, 10240, 12288, 16384]))
    if n_parent in n_candidates:
        n_candidates.remove(n_parent)
    n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: resampled score={sc_init:.10f}")

        # Quick L-BFGS
        n_deadline = min(deadline - 50, time.monotonic() + 40)
        a_opt = _lbfgs_smoothed(a_init, n_deadline, min_remaining=5.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  n={n_try}: optimized score={sc_opt:.10f} NEW BEST")

        # Active-set polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            as_deadline = min(deadline - 25, time.monotonic() + 25)
            a_ap, sc_ap = _active_set_polish(a_opt, as_deadline,
                                             tol_frac_start=0.003, tol_frac_end=0.0001,
                                             lr_init=0.012, use_momentum=True)
            if sc_ap < best_score:
                best_seq = a_ap.copy()
                best_score = sc_ap
                print(f"  n={n_try}: polished score={sc_ap:.10f} NEW BEST")

    # === Phase 6: Local search kicks ===
    print(f"\n[P6] Local search kicks (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        kick_deadline = min(deadline - 50, time.monotonic() + 120)
        a_kick, sc_kick = _local_search_kicks(best_seq, kick_deadline,
                                              n_kicks=4, kick_sigma=0.03)
        if sc_kick < best_score:
            best_seq = a_kick.copy()
            best_score = sc_kick
            print(f"  After kicks: {best_score:.10f} NEW BEST")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[P7] Final ultra-fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        fin_deadline = deadline - 3
        # Very tight active-set
        a_fin, sc_fin = _active_set_polish(best_seq, fin_deadline,
                                           tol_frac_start=0.001, tol_frac_end=0.000001,
                                           lr_init=0.005, use_momentum=True)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  Final active-set: {best_score:.10f} NEW BEST")

        # Very small beta L-BFGS
        remaining = max(0, deadline - 3 - time.monotonic())
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00005

            def obj_final(z_):
                a_ = _softmax(z_)
                c_ = _full_convolve(a_, a_)
                c_max = float(c_.max())
                shifted = (c_ - c_max) / beta_fin
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300:
                    return 2.0 * n * c_max
                return 2.0 * n * (c_max + beta_fin * np.log(s))

            def grad_final(z_):
                a_ = _softmax(z_)
                c_ = _full_convolve(a_, a_)
                c_max = float(c_.max())
                shifted = (c_ - c_max) / beta_fin
                e = np.exp(np.clip(shifted, -700, 0))
                s = e.sum()
                if s < 1e-300:
                    w = np.zeros_like(c_)
                    w[int(np.argmax(c_))] = 1.0
                else:
                    w = e / s
                a_rev = a_[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W = float(np.dot(w, c_))
                return 4.0 * n * a_ * (h - W)

            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(obj_final, z, jac=grad_final, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-17,
                                           'gtol': 1e-15, 'maxcor': 100})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  Final L-BFGS: {best_score:.10f} NEW BEST")
            except Exception:
                pass

    # Final validation and output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_half_sine(256)
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence score: {ev_score:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_half_sine(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
