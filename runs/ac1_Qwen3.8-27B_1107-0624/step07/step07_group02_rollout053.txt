    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(v)
    if n == 0:

    u = np.sort(v)[::-1]

    rho = np.where(u * np.arange(1, n+1) > (cssv - l))[0]
    if len(rho) == 0:
        return np.full(n, l / n)
    rho = rho[-1]
    theta = (cssv[rho] - l) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Enhanced optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with adaptive log-sum-exp, structured mirror descent,
active-set polishing with KKT-based refinement, and multi-scale exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _simproj(v, l=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cand = (cssv - l) / np.arange(1, n + 1)
    rho = np.where(u > cand)[0]
    if len(rho) == 0:
        return np.full(n, l / n)
    rho = rho[-1]
    theta = (cssv[rho] - l) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_structured(a, k_star):
    """Compute structured subgradient of max_k c_k at a, where k_star is the active lag."""
    n = len(a)
    # grad[j] = 2 * a[k_star - j] with zero padding
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g


def _subgrad_topk(a, top_k=5):
    """Compute weighted subgradient using top-k active constraints."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Find top-k positions
    idx = np.argsort(c)[::-1][:top_k]
    weights = c[idx] - c.max() + 1e-15
    weights = weights / weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, idx):
        for j in range(n):
            idx2 = int(k) - j
            if 0 <= idx2 < n:
                g[j] += w * 2.0 * a[idx2]
    return g


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                      0.03, 0.015, 0.0075, 0.004,
                                      0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                      0.03, 0.015, 0.0075, 0.004,
                                      0.002, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.002 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(20, int(remaining / (est_time_per_iter * 10))))

        def obj(z):
            return _smoobj(z, beta, n)

        def grad(z):
            return _smograd(z, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _mirror_descent(a_init, deadline, min_time_remaining=30.0):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    eta = 0.1
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                break

        # Compute subgradient using top-3 active constraints
        top_k = min(3, 2 * n - 1)
        idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(2 * n - 1)
        cw = c[idx]
        wv = cw / cw.sum()
        g = np.zeros(n)
        for wi, ki in zip(wv, idx):
            ki = int(ki)
            for j in range(n):
                idx2 = ki - j
                if 0 <= idx2 < n:
                    g[j] += wi * 2.0 * a[idx2]

        # Nesterov momentum
        v = momentum * v + (1 - momentum) * g
        gnorm = np.linalg.norm(v)
        if gnorm < 1e-15:
            break

        # Multiplicative update: a_new = a * exp(-eta * v) / Z
        # Work in log-space for numerical stability
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * v
        log_a -= log_a.max()
        exp_a = np.exp(log_a)
        s_exp = exp_a.sum()
        if s_exp < 1e-300:
            break
        a_new = exp_a / s_exp

        # Clip tiny values to zero for sparsity
        a_new[a_new < 1e-15] = 0.0
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        a = a_new

        # Adaptive eta
        sc_new = _score(a)
        if sc_new > current_sc:
            eta *= 0.8
        elif no_improve == 0:
            eta = min(eta * 1.05, 0.5)

    return best_a, best_sc


def _active_set_polish(a_init, deadline, min_time_remaining=10.0,
                       tol_init=0.005, tol_min=0.00002, lr_init=0.03):
    """Enhanced active-set polishing with KKT-based refinement."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_init
    tol_frac = tol_init
    no_improve = 0
    stall_escalations = 0

    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        # Cap active set size for computational efficiency
        if n_active > 50:
            top_idx = np.argsort(c)[::-1][:50]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 50

        # Weighted subgradient
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * a * (h - W)
        # Remove mean component (scale invariance)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        # Backtracking line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_min, tol_frac * 0.7)
                lr *= 0.8
            if no_improve >= 300:
                stall_escalations += 1
                if stall_escalations > 10:
                    break
                a = best_a.copy()
                tol_frac = max(tol_min, tol_frac * 0.5)
                lr = lr_init * 0.2
                no_improve = 0
        else:
            if tol_frac > tol_min:
                tol_frac = max(tol_min, tol_frac * 0.95)
            lr = min(lr_init, lr * 1.03)

    return best_a, best_sc


def _shrinkage_optimize(a, deadline, min_time_remaining=30.0):
    """Progressive shrinkage with re-optimization."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.04, 0.05, 0.07]
    for frac in fractions:
        if time.monotonic() > deadline - min_time_remaining:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _direct_gradient_descent(a_init, deadline, min_time_remaining=30.0):
    """Direct projected subgradient descent with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Nesterov momentum
    y = a.copy()
    beta_mom = 0.9
    eta = 0.05
    no_improve = 0

    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(y, y)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Subgradient at y for the active constraint
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = 2.0 * y[idx]

        # Polyak step size
        sc_y = 2.0 * n * c_max
        gap = sc_y - best_sc
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        if gap > 1e-10:
            eta_p = gap / (gnorm * gnorm)
            eta = min(max(eta_p, 0.001), 0.5)

        # Gradient step with projection
        a_new = _simproj(y - eta * g, 1.0)

        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
        y = a_new + (beta_mom / beta_new) * (a_new - a)
        a = a_new
        beta_mom = beta_new

        sc_new = _score(a)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 500:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search combining L-BFGS, mirror descent, subgradient,
    and active-set polishing with multi-scale exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main optimization ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0, aggressive=True)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 1 done: score={sc_main:.10f}, elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 2: Mirror descent ===
    print(f"\n[P2] Mirror descent (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400:
        md_deadline = min(deadline - 400, time.monotonic() + 250)
        a_md, sc_md = _mirror_descent(best_seq, md_deadline)
        if sc_md < best_score:
            best_seq = a_md.copy()
            best_score = sc_md
            print(f"  NEW BEST: {best_score:.10f} [mirror_descent] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 2 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Direct gradient descent ===
    print(f"\n[P3] Projected subgradient (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 350:
        gd_deadline = min(deadline - 350, time.monotonic() + 200)
        a_gd, sc_gd = _direct_gradient_descent(best_seq, gd_deadline)
        if sc_gd < best_score:
            best_seq = a_gd.copy()
            best_score = sc_gd
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 3 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Active-set polish (coarse) ===
    print(f"\n[P4] Active-set coarse (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 180)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_init=0.008, tol_min=0.0002, lr_init=0.04)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish_coarse] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 4 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Shrinkage ===
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 250)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 180:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=130.0, aggressive=True)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")
                if time.monotonic() < deadline - 100:
                    a_shr3, sc_shr3 = _active_set_polish(best_seq, min(deadline - 100, time.monotonic() + 80),
                                                         tol_init=0.002, tol_min=0.00005, lr_init=0.015)
                    if sc_shr3 < best_score:
                        best_seq = a_shr3.copy()
                        best_score = sc_shr3
                        print(f"  NEW BEST: {best_score:.10f} [polish_post_shrink] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 5 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    n_candidates = [4096, 5120, 6144, 8192, 10240, 12288]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 35, time.monotonic() + 35),
                                                 tol_init=0.003, tol_min=0.0001, lr_init=0.012)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 6 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("bimo", _make_profile(n_best, "bimodal")),
            ("peri", _make_profile(n_best, "periodic")),
            ("gaur", _make_profile(n_best, "gaussian_center")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 20:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=12.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")
    print(f"  Phase 7 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 3:
        # Tightest active-set polish
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 2,
                                           tol_init=0.0005, tol_min=0.000005, lr_init=0.004)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS with very small beta
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.00002, 1e-10)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()

            def obj_final(z):
                return _smoobj(z, beta_fin, n)

            def grad_final(z):
                return _smograd(z, beta_fin, n)

            max_iter = min(3000, max(20, int(remaining / 0.03)))
            try:
                result = minimize(obj_final, z, jac=grad_final, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One more micro active-set pass
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            a_fin3, sc_fin3 = _active_set_polish(best_seq, deadline - 1,
                                                 tol_init=0.0003, tol_min=0.000002, lr_init=0.002)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [micro_polish] t={time.monotonic()-t_start:.1f}s")

    print(f"  Phase 8 done: elapsed={time.monotonic()-t_phase:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
