def _solve_direction_lp(a, K, M):
    """Solve LP to find optimal direction.
    max sum(b) s.t. (a*b)[k] <= M for k in top_K, b >= 0, sum(b) = 1
    """

    c_conv = _full_convolve(a, a)
    idx = np.argsort(c_conv)[::-1][:K]

    # sum_j a[k-j] * b[j] <= M (with appropriate bounds on j)

            if 0 <= jj < n:
                A_ub[i, j] = a[jj]

                  A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with adaptive log-sum-exp smoothing for initial optimization
2. LP-based active set refinement (BPS-inspired) for fine-tuning - main innovation
3. Multi-resolution coarse-to-fine approach
4. Diverse structured initializations
5. Asymmetry-exploiting perturbations and diverse restarts

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.2)
        max_iter = min(12000, max(100, int(remaining / (0.004 * scale * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _solve_lp_direction(a, K, M):
    """Solve LP: max 1^T b s.t. (a*b)[k] <= M for top K lags, b >= 0.
    Returns the optimal b normalized to sum=1, or None if failed."""
    n = len(a)
    c_conv = _full_convolve(a, a)
    idx = np.argsort(c_conv)[::-1][:K]

    # Build constraint matrix: A[i,j] = a[k_i - j] if 0 <= k_i - j < n
    A_ub = np.zeros((K, n))
    for i, k in enumerate(idx):
        j_start = max(0, k - n + 1)
        j_end = min(k + 1, n)
        for j in range(j_start, j_end):
            A_ub[i, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.ones(K) * M

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})

    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None


def _lp_refine(a_init, deadline, K=20, max_iter=500, t_init=0.3):
    """LP-based active set refinement (BPS-inspired)."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    t = t_init
    stall = 0
    stall_limit = 50

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        b = _solve_lp_direction(a, K, M)
        if b is None:
            t *= 0.5
            if t < 1e-6:
                break
            continue

        # Try different step sizes
        improved = False
        for trial_t in [t, t * 0.5, t * 0.1, t * 0.01]:
            a_try = (1 - trial_t) * a + trial_t * b
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-14:
                best_a = a_try.copy()
                best_sc = sc_try
                a = a_try
                improved = True
                t = min(t * 1.05, 0.8)
                stall = 0
                break

        if not improved:
            stall += 1
            t *= 0.7
            if stall > stall_limit:
                break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=50000, lr_init=0.06):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    stall_threshold = max(10, k_active * 3)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = lr * a.sum()
        z = z - step * grad / (a * n)
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= 0.6
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_asymmetric_chirp(n, freq=1.0, eps=0.4):
    """Chirp with asymmetric amplitude modulation."""
    t = np.arange(n)
    chirp = np.cos(2.0 * np.pi * freq * t * t / n)
    # Asymmetric window: one side rises fast, other falls slow
    win = np.exp(-((t - 0.3 * n) ** 2) / (2 * (0.25 * n) ** 2))
    a = win * (1.0 + eps * chirp)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_linear_asym(n, slope=0.5):
    """Linearly ramped asymmetric sequence."""
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * (2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    # Add some chirp for interest
    a += 0.1 * np.cos(2.0 * np.pi * t * t * n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _kkt_refine(a, deadline, k_active=4, max_iter=200):

def _perturb_and_refine(a_current, rng, n_perturbations=5, deadline=None, reserve=30.0):

        pert_type = p % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * p
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            freq = int(rng.integers(1, max(2, n // 32)))

            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))

    if best_seq is not None and time.monotonic() < deadline - 150:
        a_lp2, _ = _lp_refine_multiK(best_seq, deadline - 150,
                                      K_list=[20, 30, 40, 50], max_iter_each=100)
        try_update(a_lp2, "lp_fine")

    # === Phase 6: Diversification ===
    print(f"\n=== Phase 6: Diversification (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_div, _ = _perturb_and_refine(best_seq, rng, n_perturbations=6,
                                        deadline=deadline, reserve=80.0)
        try_update(a_div, "diversify")

    # After diversification, if improved, do another LP round
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_lp3, _ = _lp_refine_multiK(best_seq, deadline - 80,
                                      K_list=[30, 40, 60], max_iter_each=80)
        try_update(a_lp3, "lp_post_div")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        lengths_to_try = [8192, 16384, 10240, 2048]
        # Sort by priority (prefer larger if current is small, smaller if current is large)
        if best_n < 8192:
            lengths_to_try.sort(reverse=True)
        else:
            lengths_to_try.sort()

        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 50:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            if time.monotonic() < deadline - 40:
                # Quick L-BFGS at new length
                c_i = _full_convolve(a_interp, a_interp)
                mc = c_i.max()
                beta = mc * 0.01
                max_it = max(500, int((deadline - 50 - time.monotonic()) / 0.01))
                a_opt = _lbfgs_smooth(a_interp, beta, target_n, min(max_it, 5000))
                try_update(a_opt, f"n_{target_n}_lbfgs")

            # LP at new length
            if best_seq is not None and time.monotonic() < deadline - 30:
                a_lpo, _ = _lp_refine(best_seq if len(best_seq) == target_n
                                       else _interpolate_to_n(best_seq, target_n),
                                       deadline - 30, K=20, max_iter=50)
                try_update(a_lpo, f"n_{target_n}_lp")

    # === Phase 8: Final ultra-tight polishing ===
    print(f"\n=== Phase 8: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Multiple rounds of very tight LP
        for rnd in range(8):
            if time.monotonic() > deadline - 5:
                break

            # Very tight LP
            K_final = min(50 + 10 * rnd, 100)
            a_f, _ = _lp_refine(best_seq, deadline - 5, K=K_final,
                                 max_iter=50, t_init=0.05)
            try_update(a_f, f"final_lp_{rnd}_K{K_final}")

            # Micro L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * (1e-7 * (0.5 ** rnd))
            max_iter = min(3000, max(200, int((deadline - 5 - time.monotonic()) / 0.005)))
            if max_iter > 100:
                a_f2 = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_f2, f"final_lbfgs_{rnd}")

            # KKT
            if time.monotonic() < deadline - 5:
                a_k, _ = _kkt_refine(best_seq, deadline - 5, k_active=3, max_iter=100)
                try_update(a_k, f"final_kkt_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
