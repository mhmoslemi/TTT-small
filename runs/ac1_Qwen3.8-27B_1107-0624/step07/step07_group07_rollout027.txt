"""
Step sequence optimizer using convex subgradient methods with water-filling,
adaptive L-BFGS smoothing continuation, block coordinate descent, and
symmetry exploitation. Key novel contributions: (1) water-filling peak
equalization that solves a small LP to find directions reducing all top
peaks simultaneously, (2) adaptive beta schedule based on peak flatness,
(3) block coordinate descent exploiting sparsity of convolution updates,
(4) symmetric-constrained search for additional structure.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _convolve_grad(w, a, n):
    """Compute gradient of sum_k w_k * conv_k(a) w.r.t. a.
    Result has length n, representing dq/da_j for the valid range.
    """
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return h


def _subgradient_max(a, n, w=None):
    """Subgradient of max_k conv_k(a). Returns gradient vector of length n."""
    c = _full_convolve(a, a)
    if w is None:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = w / w.sum()
    grad = _convolve_grad(w, a, n)
    return grad


def _adaptive_weights(c, frac=0.05, tol_frac=1e-3):
    """Compute weights for active set based on proximity to max."""
    c_max = float(c.max())
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_top = max(5, int(frac * len(c)))
    n_top = min(n_top, len(c))
    threshold_rank = sorted_c[n_top - 1]
    tol = max(tol_frac * c_max, c_max - threshold_rank)
    tol = min(tol, 0.5 * c_max)
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    # Weight by exp((c - c_max)/scale) for smoothness
    scale = max(tol * 0.5, 1e-10)
    w = np.zeros_like(c)
    w[active] = np.exp((c[active] - c_max) / scale)
    w /= w.sum()
    return w


def _smoobj_z(z, beta, n):
    """Log-sum-exp smoothed objective in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_z(z, beta, n):
    """Gradient of smoothed objective in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smoothed(a_init, beta, n, max_iter=5000, ftol=1e-20, gtol=1e-16):
    """Run L-BFGS on the smoothed objective in z-space."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj_z(z, beta, n)

    def jac(z):
        return _smograd_z(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 150,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_beta_schedule(a, deadline, min_time_remaining=200.0):
    """Adaptive beta continuation: start large, shrink based on peak flatness."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return a

    current_a = a.copy()
    current_sc = _score(current_a)

    # Adaptive schedule: start at 10% of max, halve each step
    beta = c_max * 0.1
    min_beta = c_max * 1e-8

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        # Estimate time per iteration
        est_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(20000, max(100, int(remaining / (est_time * 30))))

        a_new = _lbfgs_smoothed(current_a, beta, n, max_iter)
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new
            print(f"    beta={beta:.3e}: score={sc_new:.10f}")

        # Adaptive beta update: if peaks are flat, reduce beta more aggressively
        c_new = _full_convolve(current_a, current_a)
        top_frac = int(0.05 * len(c_new))
        top_vals = np.sort(c_new)[::-1][:top_frac]
        flatness = (top_vals[0] - top_vals[-1]) / max(top_vals[0], 1e-30)
        # If flat, reduce beta more (we're close to the max structure)
        if flatness < 0.01:
            beta *= 0.4
        elif flatness < 0.05:
            beta *= 0.5
        else:
            beta *= 0.6

    return current_a


def _water_filling_peaks(a, deadline, max_iters=20000):
    """Water-filling: iteratively reduce top peaks by finding directions
    that decrease all active peaks simultaneously using a small QP."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.05
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = _adaptive_weights(c, frac=0.03, tol_frac=5e-4)

        # Compute subgradient
        grad = _subgradient_max(a, n, w)

        # Project gradient onto sum=0 subspace
        grad = grad - grad.mean()

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Line search
        g_dir = grad / g_norm
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.12, 0.05]:
            s = step * frac
            a_try = a - s * g_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                step = min(step * 1.02, 0.2)
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            step *= 0.85
            if step < 1e-10 or stall > 200:
                break

    return best_a, best_sc


def _block_coordinate_descent(a, deadline, n_blocks=32, max_rounds=5000):
    """Block coordinate descent: optimize each block while holding others fixed."""
    n = len(a)
    block_size = n // n_blocks
    if block_size < 4:
        n_blocks = max(2, n // 4)
        block_size = n // n_blocks
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    remaining_size = n - n_blocks * block_size
    block_indices = []
    start = 0
    for b in range(n_blocks):
        end = start + block_size
        if b == n_blocks - 1:
            end = n
        block_indices.append((start, end))
        start = end

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        bi = rnd % n_blocks
        start, end = block_indices[bi]
        bsz = end - start

        # Optimize this block: minimize max conv involving this block
        # Use a few steps of gradient descent on the block
        a_block = a[start:end].copy()
        a_outer = np.delete(a, np.arange(start, end))

        for inner in range(10):
            # Reconstruct full a
            a_full = np.empty(n)
            a_full[:start] = a[:start]
            a_full[start:end] = a_block
            a_full[end:] = a[end:]

            c = _full_convolve(a_full, a_full)
            c_max = float(c.max())
            w = _adaptive_weights(c, frac=0.03, tol_frac=5e-4)

            # Gradient w.r.t. block elements only
            grad_full = _subgradient_max(a_full, n, w)
            grad_block = grad_full[start:end]
            # Only changes in block affect conv, so use full gradient projected
            grad_block = grad_block - grad_block.mean()

            gn = np.linalg.norm(grad_block)
            if gn < 1e-14:
                break

            lr = 0.01
            a_block_new = a_block - lr * grad_block / gn
            a_block_new = np.maximum(a_block_new, 0.0)
            s_total = a[:start].sum() + a_block_new.sum() + a[end:].sum()
            if s_total < 1e-10:
                break
            a_block = a_block_new / (a_block_new.sum() / s_total) if a_block_new.sum() > 0 else a_block

        # Place back
        a[start:end] = a_block
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _iterative_peak_equality(a, deadline, max_iters=30000):
    """Iteratively equalize top peaks: find direction that makes all top
    convolution values decrease proportionally."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()

    best_sc = _score(a)
    step = 0.03
    stall = 0

            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify top peaks
        n_top = max(10, int(0.02 * len(c)))
        top_idx = np.argsort(c)[::-1][:n_top]
        top_vals = c[top_idx]
        spread = (top_vals[0] - top_vals[-1]) / max(top_vals[0], 1e-30)

        # Weight: emphasize peaks that are relatively high
        w_top = np.zeros_like(c)
        w_top[top_idx] = np.exp(-(top_vals[0] - top_vals) / max(top_vals[0] * 0.1, 1e-10))
        w_top /= w_top.sum()

        # Compute direction: weighted average gradient
        grad = _subgradient_max(a, n, w_top)
        grad = grad - grad.mean()
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Adaptive step: if spread is small, use smaller steps
        effective_step = step * (1.0 + spread)
        g_dir = grad / g_norm

        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            s = effective_step * frac
            a_try = a - s * g_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                step = min(step * 1.01, 0.15)
                improved = True
                stall = 0
                break

            stall += 1

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, rng, mode="gauss", scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(8):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.005 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

    a_p1 = _adaptive_beta_schedule(best_seq, t1_end, min_time_remaining=100.0)
    try_update(a_p1, "adaptive_lbfgs")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: Water-filling peak equalization (20% of time) ===
    print(f"\n=== Phase 2: Water-filling ===")
    t_phase = time.monotonic()
    t2_end = deadline - (deadline - t_start) * 0.40
    a_wf, sc_wf = _iterative_peak_equality(best_seq, t2_end, max_iters=50000)
    try_update(a_wf, "water_filling")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, score={sc_wf:.10f}")

    # === Phase 3: L-BFGS polish after water-filling (15% of time) ===
    print(f"\n=== Phase 3: L-BFGS polish ===")
    t_phase = time.monotonic()
    t3_end = deadline - (deadline - t_start) * 0.25
    a_p3 = _adaptive_beta_schedule(best_seq, t3_end, min_time_remaining=80.0)
    try_update(a_p3, "lbfgs_polish")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p3):.10f}")

    # === Phase 4: Block coordinate descent (10% of time) ===
    print(f"\n=== Phase 4: Block CD ===")
    t_phase = time.monotonic()
    t4_end = deadline - (deadline - t_start) * 0.15
    a_bd, sc_bd = _block_coordinate_descent(best_seq, t4_end, n_blocks=64, max_rounds=10000)
    try_update(a_bd, "block_cd")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s, score={sc_bd:.10f}")

    # === Phase 5: Mirror descent (10% of time) ===
    print(f"\n=== Phase 5: Mirror descent ===")
    t_phase = time.monotonic()
    t5_end = deadline - (deadline - t_start) * 0.08
    a_md, sc_md = _mirror_descent(best_seq, t5_end, max_iters=30000)
    try_update(a_md, "mirror_descent")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s, score={sc_md:.10f}")

    # === Phase 6: Symmetry exploitation (5% of time) ===
    print(f"\n=== Phase 6: Symmetric optimization ===")
    t_phase = time.monotonic()
    t6_end = deadline - (deadline - t_start) * 0.04
    a_sym, sc_sym = _symmetric_optimize(best_seq, t6_end)
    try_update(a_sym, "symmetric")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s, score={sc_sym:.10f}")

    # === Phase 7: Multi-restart with polish (10% of time) ===
    print(f"\n=== Phase 7: Multi-restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        ("gauss_02", "gauss", 0.02),
        ("gauss_05", "gauss", 0.05),
        ("fourier", "fourier", 0.0),
        ("gauss_10", "gauss", 0.10),
    ]
    remaining_per_restart = (deadline - (deadline - t_start) * 0.03 - time.monotonic()) / max(1, len(restart_configs))
    for name, mode, scale in restart_configs:
        if time.monotonic() > deadline - 10:
            break
        p_start = _perturb(best_seq, rng, mode, scale)
        sc_start = _score(p_start)
        print(f"  Restart [{name}]: start={sc_start:.10f}")

        # Quick L-BFGS
        t_r_end = time.monotonic() + max(5.0, remaining_per_restart * 0.6)
        a_r = _adaptive_beta_schedule(p_start, t_r_end, min_time_remaining=5.0)
        try_update(a_r, f"restart_{name}")

        # Quick water-filling
        if time.monotonic() < deadline - 10:
            t_wf_end = time.monotonic() + max(3.0, remaining_per_restart * 0.3)
            a_w, _ = _iterative_peak_equality(a_r, t_wf_end, max_iters=5000)
            try_update(a_w, f"restart_{name}_wf")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Ultra-fine final polish (3% of time) ===
    print(f"\n=== Phase 8: Final ultra-fine polish ===")
    t_phase = time.monotonic()
    for cycle in range(5):
        if time.monotonic() > deadline - 3:
            break
        # Very small beta L-BFGS
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta = mb * 1e-7
        max_iter = min(3000, max(100, int((deadline - 3 - time.monotonic()) / 0.005)))
        a_ultra = _lbfgs_smoothed(best_seq, beta, n, max_iter)
        try_update(a_ultra, f"ultra_{cycle}")

        # Quick water-filling
        if time.monotonic() < deadline - 3:
            a_wf2, _ = _iterative_peak_equality(best_seq, deadline - 3, max_iters=3000)
            try_update(a_wf2, f"wf_final_{cycle}")

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
