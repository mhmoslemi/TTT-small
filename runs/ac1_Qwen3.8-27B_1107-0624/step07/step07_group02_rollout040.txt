"""
Optimized step function sequence search using:
1. LP-based conditional gradient (Frank-Wolfe) method
2. Log-sum-exp smoothed L-BFGS with adaptive temperature
3. Active-set subgradient descent with adaptive step
4. Multi-n exploration and diverse restarts
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_prob_obj(z, beta, n):
    return _smoobj(z, beta, n)


def _lbfgs_prob_grad(z, beta, n):
    return _smograd(z, beta, n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.002)
    
    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))
        
        try:
            result = minimize(
                lambda z: _lbfgs_prob_obj(z, beta, n),
                z,
                jac=lambda z: _lbfgs_prob_grad(z, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

We want to minimize `f(a) = 2n * max(conv(a,a))` subject to `a >= 0, sum(a) = 1`.

This is equivalent to: find the minimum `M` such that there exists `a >= 0, sum(a) = 1` with `conv(a,a)[k] <= M` for all k.

But the constraints are quadratic in a, so we can't solve this directly as an LP.

However, we can use the following iterative approach (similar to the "state of the art" mentioned):

1. Start with current a (normalized to sum=1).
2. Compute M = max(conv(a,a)).
3. Solve: find b >= 0 that maximizes sum(b) subject to conv(a, b)[k] <= M for all k.
   (Note: conv(a,b) is LINEAR in b, so this IS a linear program.)
4. Normalize b to sum=1.
5. Update: a_new = (1-t)*a + t*b.
6. Repeat.

The key insight: by optimizing b against the FIXED a, we find a direction that, when mixed with a, tends to reduce the overall max convolution.

Why does this work? Consider the expansion:
`conv((1-t)a + tb, (1-t)a + tb) = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,b) + t^2 conv(b,b)`

The cross term `conv(a,b)` is bounded by M (from the LP constraint). At t=0, the max is at most (1-t)^2 * M + 2t(1-t) * M + t^2 * max(conv(b,b)).

For small t, the dominant change is in the cross term. If max(conv(a,b)) < M (which it should be, since we're optimizing b to have low convolution with a), then the combined sequence has lower max convolution.

Actually, the LP constraint is conv(a,b) <= M, not < M. So at the constraint boundary, the cross term is at most M, same as the original. The improvement comes from the `t^2 * conv(b,b)` term being potentially smaller, and from the structure of b.

def _lp_direction(a, top_k=200, M_mult=1.0):
    """LP-based conditional gradient direction finding."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    M = c_max * M_mult

    # Select top_k lags to use as active constraints
    if top_k < 2 * n - 1:
        top_lags = np.partition(c, -(top_k))[-top_k:]
        threshold = top_lags[0]
        active_lags = np.where(c >= threshold * 0.999)[0]
        if len(active_lags) > top_k * 2:
            active_lags = np.argsort(c)[::-1][:top_k]
    else:
        active_lags = np.arange(2 * n - 1)

    K = len(active_lags)
    if K == 0:
        return None

    # Build constraint matrix A: A[i, j] = a[k_i - j] if 0 <= k_i - j < n
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1) + 1
        if j_end > j_start:
            idx = np.arange(j_start, j_end)
            A[i, idx] = a[k - idx]

    # Solve LP: maximize sum(b) s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = float(b.sum())
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None

        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= 300:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                if tol_frac <= tol_frac_end * 1.01 and lr < lr_init * 0.01:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t + 0.1 * np.sin(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.002, 0.005, 0.008, 0.012, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    return best_a, best_sc


def _refine_loop(a, dl):
    """Tight iterative refinement with aggressive tolerance reduction."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    tol = 0.001
    lr = 0.03
    stall = 0

    while time.monotonic() < dl:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        active = c >= c_max - tol * c_max
        na = int(active.sum())
        if na == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            na = 1
        if na > 3000:
            top_idx = np.argsort(c)[::-1][:3000]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            na = 3000

        w = np.zeros_like(c)
        w[active] = 1.0 / na
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gn = np.linalg.norm(grad_t)
        if gn < 1e-16:
            break

        alpha = min(lr, 1.0 / (gn + 1e-15))
        imp = False
        for _ in range(100):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < sc - 1e-14:
                a = a_try
                imp = True
                break
            alpha *= 0.5

        if imp:
            tol = max(5e-6, tol * 0.95)
            lr = min(0.03, lr * 1.02)
        else:
            tol = max(5e-6, tol * 0.6)
            lr *= 0.7
            if stall > 200:
                a = best_a.copy()
                stall = 0
                if tol <= 6e-6 and lr < 0.002:
                    break
    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: L-BFGS with adaptive log-sum-exp smoothing
    P2: LP-based conditional gradient (Frank-Wolfe)
    P3: Active-set coarse polish
    P4: Active-set fine polish
    P5: Shrinkage + re-optimize
    P6: Multi-n exploration
    P7: Diverse restarts
    P8: Final tight-loop refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    n_parent = 0
    a_parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                a_parent = parent_raw / s
                n_parent = len(a_parent)
    except Exception:
        pass

    if a_parent is not None:
        n_start = n_parent
        a_init = a_parent.copy()
        best_score = _score(a_init)
        best_seq = a_init.copy()
        print(f"Parent: n={n_start}, score={best_score:.10f}")
    else:
        n_start = 4096
        a_init = _make_profile(n_start, "uniform")
        best_score = _score(a_init)
        best_seq = a_init.copy()
        print(f"No parent. Uniform n={n_start}, score={best_score:.10f}")

    # P1: L-BFGS main (with more aggressive small-beta phase)
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    t1_end = min(deadline - 600, t_start + 500)
    a1 = _run_lbfgs(a_init, t1_end, min_time_remaining=100)
    sc1 = _score(a1)
    print(f"  P1 score={sc1:.10f} (t={time.monotonic()-t_start:.1f}s)")
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_main]")

    # P2: LP-based conditional gradient
    print(f"\n[P2] LP-based FW (t={time.monotonic()-t_start:.1f}s)")
    t2_end = min(deadline - 400, t_start + 700)
    a_fw = best_seq / best_seq.sum()
    sc_fw = _score(a_fw)
    fw_stall = 0
    while time.monotonic() < t2_end:
        a_try, sc_try = _fw_iteration(a_fw, top_k=200, M_mult=1.0)
        if sc_try < sc_fw - 1e-12:
            a_fw = a_try
            sc_fw = sc_try
            fw_stall = 0
            if sc_fw < best_score:
                best_seq = a_fw.copy()
                best_score = sc_fw
                print(f"  NEW BEST: {best_score:.10f} [fw] t={time.monotonic()-t_start:.1f}s")
        else:
            sc_try2 = _score(a_fw)
            a2, sc2 = _fw_iteration(a_fw, top_k=300, M_mult=0.95)
            if sc2 < sc_fw - 1e-12:
                a_fw = a2
                sc_fw = sc2
                fw_stall = 0
            else:
                fw_stall += 1
                if fw_stall > 10:
                    break
    print(f"  P2 score={_score(a_fw):.10f} (t={time.monotonic()-t_start:.1f}s)")

    # P3: Active-set coarse
    print(f"\n[P3] Active-set coarse (t={time.monotonic()-t_start:.1f}s)")
    t3_end = min(deadline - 250, t_start + 850)
    a3, sc3 = _active_set_poly(best_seq, t3_end, tol_frac_start=0.006, tol_frac_end=0.0002, lr_init=0.04)
    print(f"  P3 score={sc3:.10f} (t={time.monotonic()-t_start:.1f}s)")
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [polish_coarse]")

    # P4: Active-set fine
    print(f"\n[P4] Active-set fine (t={time.monotonic()-t_start:.1f}s)")
    t4_end = min(deadline - 180, t_start + 1000)
    a4, sc4 = _active_set_poly(best_seq, t4_end, tol_frac_start=0.0015, tol_frac_end=0.00004, lr_init=0.015)
    print(f"  P4 score={sc4:.10f} (t={time.monotonic()-t_start:.1f}s)")
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} [polish_fine]")

    # P5: Shrinkage + re-optimize
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a5, sc5 = _shrinkage_optimize(best_seq, deadline - 120)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [shrinkage]")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 80:
                a5b = _run_lbfgs(best_seq, deadline, min_time_remaining=50)
                sc5b = _score(a5b)
                if sc5b < best_score:
                    best_seq = a5b.copy()
                    best_score = sc5b
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink_lbfgs]")
                if time.monotonic() < deadline - 40:
                    a5c, sc5c = _active_set_poly(best_seq, deadline, tol_frac_start=0.001, tol_frac_end=0.00003, lr_init=0.01)
                    if sc5c < best_score:
                        best_seq = a5c.copy()
                        best_score = sc5c
                        print(f"  NEW BEST: {best_score:.10f} [post_shrink_polish]")

    # P6: Multi-n exploration
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_cands = sorted(set([len(best_seq)] + [2048, 3072, 4096, 5120, 6144, 8192]))
    for n_try in n_cands:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 30:
            continue
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() < deadline - 40:
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=25)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # P7: Diverse restarts
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restarts = [
            ("pert3", _perturb(best_seq, 0.03, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("peri", _make_profile(n_best, "periodic")),
            ("bimo", _make_profile(n_best, "bimodal")),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 20:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=15)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # P8: Final tight-loop refinement
    print(f"\n[P8] Final tight refinement (t={time.monotonic()-t_start:.1f}s)")
    t8_end = deadline - 5
    a8, sc8 = _refine_loop(best_seq, t8_end)
    print(f"  P8 score={sc8:.10f} (t={time.monotonic()-t_start:.1f}s)")
    if sc8 < best_score:
        best_seq = a8.copy()
        best_score = sc8
        print(f"  NEW BEST: {best_score:.10f} [final_refine]")

    # Final tiny L-BFGS
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        _g_beta = mb * 0.0001
        _g_n_val = n
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        max_iter = min(3000, max(50, int(remaining / 0.03)))
        try:
            result = minimize(_obj_wrapper, z, jac=_grad_wrapper, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxcor': 100})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
