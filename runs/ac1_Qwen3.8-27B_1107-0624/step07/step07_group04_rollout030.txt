"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Frank-Wolfe with away steps, Nesterov acceleration, log-sum-exp smoothing,
and active set refinement. Exploits the convexity of the problem.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgrad_conv(a, k_star):
    """Compute subgradient of max_k conv(a,a)[k] at the peak lag k_star.

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    mb = float(c.max())
    return float(2.0 * n * mb / (s * s))

"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses mild-greedy Frank-Wolfe with away steps, Nesterov-accelerated subgradient,
log-sum-exp smoothed L-BFGS, and active set Newton refinement.
Exploits the convexity of the problem (min max of convex quadratics over simplex).
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    mb = float(c.max())
    return float(2.0 * n * mb / (s * s))


def _max_conv(a):
    c = _full_convolve(a, a)
    return float(c.max()), int(np.argmax(c))


def _subgrad_at_k(a, k_star):
    n = len(a)
    ar = a[::-1]
    offset = n - 1 - k_star
    g = np.zeros(n)
    if offset >= 0:
        end = n - offset
        if end > 0:
            g[:end] = 2.0 * ar[offset:]
    else:
        start = -offset
        if start < n:
            g[start:] = 2.0 * ar[:n - start]
    return g


def _smooth_subgrad(a, w):
    n = len(a)
    ar = a[::-1]
    conv_w_ar = _full_convolve(w, ar)
    g = 2.0 * conv_w_ar[n - 1:2 * n - 1]
    return g


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_weights(c, beta):
    c_shifted = c - c.max()
    e = np.exp(np.clip(c_shifted / beta, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _fw_step(a, i_fwd, t):
    a_new = (1.0 - t) * a
    a_new[i_fwd] += t
    return a_new


def _away_step(a, i_away, t):
    denom = 1.0 - t * a[i_away]
    if denom < 1e-12:
        return None
    a_new = (a - t * a[i_away] * np.zeros(len(a))) / denom
    a_new[i_away] = 0.0
    return a_new


def _run_fw(a_init, deadline, max_iters=50000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_mc = _max_conv(a)[0]
    best_sc = 2.0 * n * best_mc

    beta = best_mc * 0.1
    use_smooth = True

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        mc = float(c.max())
        ks = int(np.argmax(c))
        sc = 2.0 * n * mc
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_mc = mc
            best_a = a.copy()

        if use_smooth:
            w = _softmax_weights(c, beta)
            g = _smooth_subgrad(a, w)
        else:
            g = _subgrad_at_k(a, ks)

        # Forward step direction
        i_fwd = int(np.argmin(g))

        # Away step direction
        support = np.where(a > 1e-8)[0]
        i_away = -1
        if len(support) > 1:
            g_sup = g[support]
            i_away = support[int(np.argmax(g_sup))]

        # Step size
        t = 2.0 / (it + 2.0)

        # Try forward
        a_fwd = _fw_step(a, i_fwd, t)
        mc_fwd = float(_full_convolve(a_fwd, a_fwd).max())

        # Try away
        a_away = None
        mc_away = np.inf
        if i_away >= 0 and a[i_away] > 1e-8:
            a_away = _away_step(a, i_away, t)
            if a_away is not None:
                mc_away = float(_full_convolve(a_away, a_away).max())

        # Pick best
        if mc_away < mc_fwd:
            a = a_away
        else:
            a = a_fwd

        # Adapt beta: decrease as we make progress
        if it % 500 == 0 and it > 0:
            beta *= 0.8
            if beta < best_mc * 1e-4:
                use_smooth = False

        # Periodic check
        if it % 2000 == 0:
            pass

    return best_a


def _run_nesterov(a_init, deadline, max_iters=50000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    a_prev = a.copy()
    beta_th = 1.0
    best_a = a.copy()
    best_mc = _max_conv(a)[0]
    best_sc = 2.0 * n * best_mc

    step_base = 1.0
    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        mc, ks = _max_conv(a)
        sc = 2.0 * n * mc
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_mc = mc
            best_a = a.copy()

        g = _subgrad_at_k(a, ks)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break

        # Nesterov momentum
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_th * beta_th)) / 2.0
        momentum = (beta_th / beta_new - 1.0)

        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)

        # Step size: adaptive
        step = step_base / np.sqrt(t)
        # Cap the step to avoid going too far
        step = min(step, 0.5 / max(g_norm, 1e-10))

        a_new = _proj_simplex(y - step * g)

        # Check for improvement, reduce step if not
        mc_new = float(_full_convolve(a_new, a_new).max())
        if mc_new > mc * 1.05:
            step *= 0.25
            a_new = _proj_simplex(y - step * g)

        a_prev = a
        a = a_new
        beta_th = beta_new

    # Final check
    mc_fin, _ = _max_conv(a)
    if 2.0 * n * mc_fin < best_sc:
        best_sc = 2.0 * n * mc_fin
        best_mc = mc_fin
        best_a = a.copy()

    return best_a
