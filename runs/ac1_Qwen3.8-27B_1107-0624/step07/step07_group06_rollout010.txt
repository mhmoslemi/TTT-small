"""
Multi-phase optimizer using L2-norm continuation, FISTA momentum,
smooth max (log-sum-exp) with adaptive beta, Newton active-set polish,
peak shaving, and multi-resolution constraint embedding.
Key innovations: L2-to-Linf continuation smoothing, proper LSE gradient,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _l2_sq_obj_val(a):
    c = _full_convolve(a, a)
    return float(np.sum(c * c))


def _l2_sq_grad_a(a):
    n = len(a)
    c = _full_convolve(a, a)
    a_flip = a[::-1]
    d = _full_convolve(c, a_flip)
    return 4.0 * d[n - 1:2 * n - 1]


def _l2_sq_grad_z(z):
    a = _softmax(z)
    grad_a = _l2_sq_grad_a(a)
    w = float(np.dot(a, grad_a))
    return a * (grad_a - w)


def _fista_l2(z_init, deadline, max_iter=5000):
    """FISTA (Nesterov accelerated gradient) on L2 norm of autocorrelation."""
    n = len(z_init)
    y = z_init.copy()
    x = z_init.copy()
    t = 1.0
    step = 0.01
    best_a = _softmax(x)
    best_val = _l2_sq_obj_val(best_a)

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad = _l2_sq_grad_z(y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Adaptive step size
        step = min(step * 1.001, 1.0 / max(gnorm * 0.1, 1e-10))
        step = min(step, 0.05)

        x_new = y - step * grad
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        x = x_new
        t = t_new

        if i % 100 == 0:
            a_check = _softmax(x)
            v = _l2_sq_obj_val(a_check)
            if v < best_val:
                best_val = v
                best_a = a_check.copy()

    a_final = _softmax(x)
    v_final = _l2_sq_obj_val(a_final)
    if v_final < best_val:
        best_a = a_final.copy()
    return best_a


def _lse_obj(z, beta, n):
    """Score = 2n * LSE(c/beta) where LSE = log-sum-exp."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    mc = float(c.max())
    if mc < 1e-30:
        return 0.0
    shifted = beta * (c - mc)
    e = np.exp(np.clip(shifted, -700.0, 0.0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * mc
    return 2.0 * n * (mc + (1.0 / beta) * np.log(s))


def _lse_grad(z, beta, n):
    """Gradient of 2n*LSE(c/beta) w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    mc = float(c.max())
    shifted = beta * (c - mc)
    e = np.exp(np.clip(shifted, -700.0, 0.0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lse_fista(z_init, beta, n, deadline, max_iter=3000):
    """FISTA on the LSE smooth max objective."""
    y = z_init.copy()
    x = z_init.copy()
    t = 1.0
    step = 0.01
    best_a = _softmax(x)
    best_sc = _score(best_a)

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad = _lse_grad(y, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step_eff = step / (1.0 + 0.0001 * i)
        x_new = y - step_eff * grad
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        x = x_new
        t = t_new

        if i % 200 == 0:
            a_check = _softmax(x)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                step = min(step * 1.02, 0.05)
            else:
                step *= 0.9

    a_final = _softmax(x)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
    return best_a


def _lbfgs_lse(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on LSE smooth max with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    mc = float(c_cur.max())
    if mc < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)] if len(c_sorted) > 5 else c_sorted[0]
    base_beta = max(gap / (2.0 * mc + 1e-30), 0.01)

    # Power schedule: beta_i = base_beta * (i+1)^(-1.5) for i=0,...
    n_stages = 12
    betas = []
    for i in range(n_stages):
        b = base_beta / ((i + 1) ** 1.3)
        betas.append(max(b, base_beta * 1e-6))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate iterations: each LSE iteration costs ~3 FFTs of size 2n
        est_time_per_iter = 0.002 * max(1, n / 4096.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 6))))

        def fun_lse(z_local):
            return _lse_obj(z_local, beta, n)

        def jac_lse(z_local):
            return _lse_grad(z_local, beta, n)

        try:
            result = minimize(
                fun_lse, z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 80, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _hankel_matvec(v, active, n):
    """Compute H*v where H[i,j] = active[i+j]. Uses FFT."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    return r[n - 1:2 * n - 1]


def _newton_active(a_init, deadline, max_iter=200, tol_frac_init=0.005, tol_frac_min=0.0001):
    """Adaptive active-set Newton method with shrinking tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = tol_frac_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Progressively tighten the active set
        tol_frac = max(tol_frac * 0.95, tol_frac_min)

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Gradient of sum_{k in active} c_k / n_active
        # = (1/n_active) * hankel_matvec(a, active, n) * 2 ... 
        # grad[i] = (2/n_active) * sum_j a[j] * active[i+j]
        grad = (2.0 / n_active) * _hankel_matvec(a, active.astype(np.float64), n)

        # Project to sum=0 (tangent space)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        # Newton step: (H/n_active + mu*I) dx = -grad_t
        mu = 1e-6
        active_f = active.astype(np.float64) / n_active

        def hess_vec(v):
            return _hankel_matvec(v, active_f, n) + mu * v

        # CG solve
        x_new = np.zeros(n)
        r = -grad_t.copy()
        p_vec = r.copy()
        rs = float(np.dot(r, r))
        b_norm = float(np.linalg.norm(-grad_t))
        max_cg = min(100, max(10, n // 100))
        for _ in range(max_cg):
            Hp = hess_vec(p_vec)
            pH = float(np.dot(p_vec, Hp))
            if abs(pH) < 1e-30:
                break
            alpha_cg = rs / pH
            if alpha_cg <= 0 or not np.isfinite(alpha_cg):
                break
            x_new = x_new + alpha_cg * p_vec
            r = r - alpha_cg * Hp
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-12 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p_vec = r + beta_cg * p_vec
            rs = r_new

        dx = x_new - np.mean(x_new)

        # Line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _peak_shave(a_init, deadline, n_iter=5000, lr_init=0.02):
    """Iterative peak shaving: reduce the maximum convolution coefficient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience = 0

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: top fraction of lags
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        n_act = max(1, active.sum())
        w = active.astype(np.float64) / n_act

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n * a * (h - W)
        # Chain rule to z
        w_z = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - w_z)

        step = lr / (1.0 + 0.002 * i)
        z = z - step * grad_z
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = _softmax(z).copy()
                patience = 0
                best_patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    best_patience += 1
                    lr = lr_init * 0.5
                    if best_patience > 5:
                        break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
    return best_a, best_sc


def _spectral_init(n, n_modes, rng):
    """Initialize with energy concentrated in a few Fourier modes."""
    k = rng.integers(1, n // 2, size=n_modes)
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    amps = rng.uniform(0.3, 1.0, size=n_modes)
    t = np.arange(n)
    a = np.ones(n) * 0.5
    for j in range(n_modes):
        a = a + amps[j] * np.cos(2.0 * np.pi * k[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _tapered_init(n, kind="sin"):
    t = np.linspace(0, 1, n)
    if kind == "sin":
        a = np.sin(np.pi * t) ** 0.7
    elif kind == "cos2":
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
    elif kind == "gauss":
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif kind == "asym":
        a = t ** 0.3 * (1 - t) ** 0.3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _binary_init(n, density, rng):
    """Random binary-like initialization."""
    a = (rng.random(n) < density).astype(np.float64)
    a = a + 0.05 * rng.random(n)
    return a / a.sum()


def _shrinkage(a, deadline):
    """Try zeroing out smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    for frac in [0.005, 0.01, 0.02, 0.05, 0.10, 0.15, 0.25, 0.40]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def _multi_scale_refine(a_best, deadline, min_remaining=60.0):
    """Downsample, optimize, upsample, re-optimize."""
    n = len(a_best)
    if n < 1024:
        return a_best

    best_a = a_best.copy()
    best_sc = _score(a_best)

    for factor in [4, 2]:
        if time.monotonic() > deadline - min_remaining:
            break
        n_coarse = n // factor
        x_fine = np.linspace(0, 1, n)
        x_coarse = np.linspace(0, 1, n_coarse)
        a_coarse = np.interp(x_coarse, x_fine, best_a)
        a_coarse = np.maximum(a_coarse, 0)
        a_coarse = a_coarse / a_coarse.sum()

        # Optimize coarse
        a_coarse_opt = _lbfgs_lse(a_coarse, deadline, min_time_remaining=min_remaining // 2)

        # Upsample
        a_up = np.interp(x_fine, x_coarse, a_coarse_opt)
        a_up = np.maximum(a_up, 0)
        a_up = a_up / a_up.sum()

        # Re-optimize fine
        a_fine_opt = _lbfgs_lse(a_up, deadline, min_time_remaining=min_remaining // 3)
        sc = _score(a_fine_opt)
        if sc < best_sc:
            best_a = a_fine_opt
            best_sc = sc

    return best_a


def _perturb(a, sigma, rng):
    """Log-space perturbation."""
    n = len(a)
    z = np.log(np.maximum(a,
