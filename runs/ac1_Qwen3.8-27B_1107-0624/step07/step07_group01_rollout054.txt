"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP active set method
2. Fourier-domain projection (Gerchberg-Saxton variant)
3. L-BFGS with adaptive log-sum-exp smoothing
4. Multi-resolution exploration
5. Diverse basin hopping
6. Nesterov-accelerated gradient descent
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP active set method (dual approach)
2. Fourier-domain peak-clipping projection
3. L-BFGS with adaptive log-sum-exp smoothing
4. Nesterov-accelerated gradient in log-space
5. Multi-resolution exploration with diverse basin hopping
6. Tight final polishing with very small beta
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max_only(a):
    c = _full_convolve(a, a)
    return float(c.max()), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    if shifted.max() > 500:
        return 2.0 * n * c_max
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    if shifted.max() > 500:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.008, 0.004, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.2)
        max_iter = min(20000, max(200, int(remaining / (0.006 * scale * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new.copy()
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _lp_active_set(a, deadline, max_rounds=50, k_max=200):
    """LP-based active set method with cutting planes."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    M = float(_full_convolve(a, a).max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        # Get active set (top k convolution values)
        k_active = min(k_max, 2 * n - 1)
        idx = np.argsort(c)[::-1][:k_active]

        # Build constraint matrix: for each active index k, row is a[k-j] for j=0..n-1
        # A_ub[k, j] = a[k-j] if 0 <= k-j <= n-1 else 0
        A_ub = np.zeros((len(idx), n))
        for i, k in enumerate(idx):
            # For row i (active index k): coefficient of g[j] is a[k-j] if valid
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            if j_max >= j_min:
                coeffs = a[k - j_max:k - j_min + 1][::-1]
                A_ub[i, j_min:j_max + 1] = coeffs

        b_ub = np.full(len(idx), M)
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
            if res.success:
                g = res.x
                g_sum = float(g.sum())
                if g_sum > 1e-10:
                    g = g / g_sum
                    # Line search: find best blend
                    best_t = 0.0
                    best_blend_sc = _score(a)
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                        a_blend = (1 - t) * a + t * g
                        a_blend = np.maximum(a_blend, 0)
                        bs = a_blend.sum()
                        if bs < 1e-15:
                            continue
                        a_blend = a_blend / bs
                        sc_blend = _score(a_blend)
                        if sc_blend < best_blend_sc:
                            best_blend_sc = sc_blend
                            best_t = t
                    if best_t > 0:
                        a_new = (1 - best_t) * a + best_t * g
                        a_new = np.maximum(a_new, 0)
                        a_new = a_new / a_new.sum()
                        if best_blend_sc < best_sc:
                            best_sc = best_blend_sc
                            best_a = a_new.copy()
                        a = a_new
                        M = float(_full_convolve(a, a).max())
        except Exception:
            break

    return best_a


def _fourier_peak_clip(a, deadline, n_iter=200, alpha_init=0.3):
    """Fourier-domain peak clipping projection."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    alpha = alpha_init

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = float(c.max())
        c_mean = float(c.mean())

        if c_max <= 0:
            break

        # Clip the peak
        threshold = c_mean + 0.1 * c_max
        c_mod = c.copy()
        excess = np.maximum(c_mod - threshold, 0)
        c_mod = c_mod - alpha * excess

        # Ensure non-negativity of modified correlation
        c_mod = np.maximum(c_mod, 0)

        # Compute new power spectrum
        P_new = np.fft.rfft(c_mod, N)
        # P_new should be real (it is, since c_mod is real), but might have negative values
        if P_new.min() < 0:
            # Reduce alpha and retry
            alpha *= 0.5
            if alpha < 1e-6:
                break
            continue

        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            alpha *= 0.5
            continue
        a_new = a_new / a_new_s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.05, 0.8)
        else:
            alpha *= 0.7

    return best_a


def _nesterov_grad(a_init, deadline, lr_init=0.3, momentum=0.9):
    """Nesterov-accelerated gradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    v = np.zeros_like(z)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for it in range(50000):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        z_look = z - momentum * v
        a = _softmax(z_look)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Subgradient of max (use top-3 for robustness)
        idx = np.argsort(c)[::-1][:3]
        w = np.zeros_like(c)
        w[idx] = 1.0 / 3.0

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        v = momentum * v + lr * grad / n
        z = z - v
        z = z - z.mean()

        if it % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.01, 1.0)
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.7
                    momentum = min(momentum + 0.01, 0.99)
                    if lr < 1e-8:
                        break

    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_u_shape(n, sharpness=2.0):
    """U-shaped initialization (concentrated at ends)."""
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + sharpness * np.sin(np.pi * t) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, width=0.15):
    """Bimodal: concentrated near positions 0 and n-1."""
    t = np.arange(n)
    center1 = n * 0.1
    center2 = n * 0.9
    a = np.exp(-((t - center1) / (n * width)) ** 2) + np.exp(-((t - center2) / (n * width)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop_diverse(a_current, deadline, rng, n_hops=8, reserve=30.0):
    """Diverse basin hopping with multiple perturbation strategies."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    remaining = deadline - reserve - time.monotonic()
    if remaining < 10:
        return best_a

    time_per_hop = max(10.0, remaining / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_time = min(time_per_hop, deadline - reserve - time.monotonic())
        if hop_time < 5:
            break
        hop_deadline = time.monotonic() + hop_time

        pert_type = h % 5
        if pert_type == 0:
            # Log-space Gaussian perturbation
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier amplitude perturbation
            N = _next_pow2(2 * n - 1)
            A = np.fft.rfft(best_a, N)
            P = np.abs(A) ** 2
            # Perturb the power spectrum
            noise = np.abs(rng.standard_normal(len(P))) * 0.1 * P.mean()
            P_pert = P + noise
            P_pert = np.maximum(P_pert, 0)
            phase = np.angle(A)
            A_pert = np.sqrt(P_pert) * np.exp(1j * phase)
            a_pert = np.fft.irfft(A_pert, N)[:n]
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() < 1e-15:
                a_pert = best_a.copy()
            else:
                a_pert = a_pert / a_pert.sum()
        elif pert_type == 2:
            # Block swap
            a_pert = best_a.copy()
            n_blocks = 4
            block_sz = n // n_blocks
            for _ in range(n_blocks // 2):
                i = int(rng.integers(0, n_blocks))
                j = int(rng.integers(0, n_blocks))
                while j == i:
                    j = int(rng.integers(0, n_blocks))
                si, sj = i * block_sz, j * block_sz
                ei, ej = si + block_sz, sj + block_sz
                a_pert[si:ei], a_pert[sj:ej] = a_pert[sj:ej].copy(), a_pert[si:ei].copy()
        elif pert_type == 3:
            # Frequency-domain phase randomization (partial)
            N = _next_pow2(2 * n - 1)
            A = np.fft.rfft(best_a, N)
            # Randomize phase of a random subset of frequencies
            n_freqs = len(A)
            n_rand = max(10, n_freqs // 10)
            rand_idx = rng.choice(n_freqs, size=n_rand, replace=False)
            for idx in rand_idx:
                A[idx] = A[idx] * np.exp(1j * rng.uniform(-0.3, 0.3))
            a_pert = np.fft.irfft(A, N)[:n]
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() < 1e-15:
                a_pert = best_a.copy()
            else:
                a_pert = a_pert / a_pert.sum()
        else:
            # Anti-correlation push: try to make a more "anti-symmetric"
            z = np.log(np.maximum(best_a, 1e-18))
            # Push mass towards the ends
            t = np.linspace(0, 1, n)
            push = 0.1 * (1 - np.sin(np.pi * t))  # high at ends, low in middle
            z = z + push
            z = z - z.mean()
            a_pert = _softmax(z)

        # Quick optimization from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        if mc > 1e-30:
            beta = mc * 0.005
            max_iter = min(3000, max(200, int(hop_time / 0.01)))
            a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a


def _tight_polish(a, deadline):
    """Very tight final polishing with extremely small beta."""
    n = len(a)
    c = _full_convolve(a, a)
    mc = float(c.max())
    if mc < 1e-30:
        return a

    best_a = a.copy()
    best_sc = _score(a)

    # Very tight betas
    tight_betas = mc * np.array([1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8])

    for beta in tight_betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(5000, max(100, int(remaining / 0.01)))
        a_new = _lbfgs_smooth(best_a, beta, n, max_iter)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

    return best_a


def _subgrad_direct(a_init, deadline, k_active=3, max_steps=20000):
    """Direct subgradient in a-space with line search."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        a_try = a - step_size * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            stall += 1
            if stall > 200:
                break
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        # Line search
        alpha = 1.0
        found = False
        sc_cur = _score(a)
        for _ in range(10):
            a_ls = a - alpha * step_size * g
            a_ls = np.maximum(a_ls, 0.0)
            s_ls = a_ls.sum()
            if s_ls < 1e-30:
                alpha *= 0.5
                continue
            a_ls = a_ls / s_ls
            sc_ls = _score(a_ls)
            if sc_ls < sc_cur:
                a = a_ls
                found = True
                break
            alpha *= 0.5

        if not found:
            stall += 1
            step_size *= 0.9
            if stall > 300:
                break
        else:
            step_size = min(step_size * 1.005, 2.0)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining LP active set, Fourier projection,
    L-BFGS, Nesterov gradient, and diverse basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        parent = _make_chirp(4096, freq=1.0, eps=0.3)
        try_update(parent, "default_init")

    n_parent = len(parent)

    # Also create diverse initializations at parent's length
    inits = [
        ("chirp1", _make_chirp(n_parent, 1.0, 0.3)),
        ("chirp2", _make_chirp(n_parent, 2.0, 0.25)),
        ("multi_13", _make_multiband(n_parent, [1, 3], 0.25)),
        ("u_shape", _make_u_shape(n_parent, 2.0)),
        ("bimodal", _make_bimodal(n_parent, 0.15)),
    ]
    for name, a_init in inits:
        sc = _score(a_init)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_init, name)

    for r in range(3):
        a_r = _make_random_positive(n_parent, rng)
        sc = _score(a_r)
        print(f"  random_{r}: score={sc:.10f}")
        try_update(a_r, f"rand_{r}")

    if best_seq is None:
        best_seq = parent.copy()
        best_score_val = _score(best_seq)
        best_n = len(best_seq)

    print(f"  Phase 1 done. Best: {best_score_val:.10f}, n={best_n}")

    # === Phase 2: Main L-BFGS on best candidate ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    reserve_p2 = 700.0
    if time.monotonic() < deadline - reserve_p2:
        a_p2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=reserve_p2)
        try_update(a_p2, "main_lbfgs")

    # === Phase 3: LP Active Set ===
    print(f"\n=== Phase 3: LP Active Set (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        reserve = max(300.0, (deadline - time.monotonic()) * 0.4)
        a_lp = _lp_active_set(best_seq, deadline - reserve, max_rounds=30, k_max=100)
        try_update(a_lp, "lp_active")
        # Follow up with L-BFGS
        if time.monotonic() < deadline - reserve:
            a_lp2 = _adaptive_lbfgs(a_lp, deadline, reserve_time=reserve)
            try_update(a_lp2, "lp+lbfgs")

    # === Phase 4: Fourier Peak Clipping ===
    print(f"\n=== Phase 4: Fourier Projection (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 350:
        reserve = max(250.0, (deadline - time.monotonic()) * 0.35)
        a_fp = _fourier_peak_clip(best_seq, deadline - reserve, n_iter=300, alpha_init=0.4)
        try_update(a_fp, "fourier_clip")
        if time.monotonic() < deadline - reserve:
            a_fp2 = _adaptive_lbfgs(a_fp, deadline, reserve_time=reserve)
            try_update(a_fp2, "fp+lbfgs")

    # === Phase 5: Nesterov Gradient ===
    print(f"\n=== Phase 5: Nesterov Gradient (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        reserve = max(200.0, (deadline - time.monotonic()) * 0.3)
        a_nes = _nesterov_grad(best_seq, deadline - reserve, lr_init=0.5)
        try_update(a_nes, "nesterov")

    # === Phase 6: Direct Subgradient ===
    print(f"\n=== Phase 6: Direct Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 250:
        reserve = max(150.0, (deadline - time.monotonic()) * 0.25)
        a_sg = _subgrad_direct(best_seq, deadline - reserve, k_active=3, max_steps=15000)
        try_update(a_sg, "sg_direct")

    # === Phase 7: Basin Hopping ===
    print(f"\n=== Phase 7: Basin Hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        reserve = max(100.0, (deadline - time.monotonic()) * 0.2)
        a_hop = _basin_hop_diverse(best_seq, deadline - reserve, rng, n_hops=10,
                                   reserve=reserve * 0.5)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Multi-n Exploration ===
    print(f"\n=== Phase 8: Multi-n Exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        lengths_to_try = []
        bn = len(best_seq)
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0]:
            target = int(bn * factor)
            target = max(1024, min(target, 65536))
            if target != bn and target not in lengths_to_try:
                lengths_to_try.append(target)

        for target_n in lengths_to_try[:4]:
            if time.monotonic() > deadline - 50:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interp score: {sc_interp:.10f}")
            try_update(a_interp, f"interp_n{target_n}")

            if time.monotonic() < deadline - 40:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=40.0)
                try_update(a_opt, f"lbfgs_n{target_n}")

    # === Phase 9: Second Round of LP + Polish ===
    print(f"\n=== Phase 9: LP Round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        reserve = max(50.0, (deadline - time.monotonic()) * 0.1)
        a_lp2 = _lp_active_set(best_seq, deadline - reserve, max_rounds=20, k_max=150)
        try_update(a_lp2, "lp_r2")
        if time.monotonic() < deadline - reserve:
            a_lp3 = _adaptive_lbfgs(a_lp2, deadline, reserve_time=reserve)
            try_update(a_lp3, "lp_r2+lbfgs")

    # === Phase 10: Final Tight Polishing ===
    print(f"\n=== Phase 10: Final Polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final = _tight_polish(best_seq, deadline - 2)
        try_update(a_final, "tight_polish")

        # One more round of very tight L-BFGS
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            if mb > 1e-30:
                for beta_frac in [1e-7, 5e-8, 2e-8]:
                    if time.monotonic() > deadline - 2:
                        break
                    beta = mb * beta_frac
                    max_iter = min(3000, max(50, int((deadline - 2 - time.monotonic()) / 0.01)))
                    a_t = _lbfgs_smooth(best_seq, beta, n, max_iter)
                    try_update(a_t, f"ubt_{beta_frac}")

    # === Final Validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
