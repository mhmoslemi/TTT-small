Let me just add `b[i] <= B` for all i, where B is a large enough bound. Since we're maximizing sum(b) and the optimal should have sum(b) not much larger than n (when a is normalized to sum 1 and M is the peak), B = n * 2 should be sufficient.

Actually, let me think about this more carefully. If a is normalized to sum 1, then the "average" entry is 1/n. The peak of a*a is at least 1/n (by the averaging argument on the diagonal... wait no, the diagonal isn't well-defined for linear convolution).

For a uniform a (all entries = 1/n), conv(a,a)[i] = (i+1)/n^2 for i < n, and (2n-1-i)/n^2 for i >= n. The max is at i = n-1: n/n^2 = 1/n.

So M >= 1/n for any valid a. And the LP with M = 1/n and uniform a would give b = a (by symmetry), with sum(b) = 1.

For a non-uniform a, M > 1/n, and the LP might find a b with sum(b) > 1.

Let me just add bounds on b to ensure feasibility:
`0 <= b[i] <= M / min_positive_a` for each i.

Or more simply, I'll replace any zero in a with a tiny epsilon (like 1e-10) before building the constraint matrix.

"""
LP-based alternating optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal "blending" directions, with coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _build_conv_matrix(a):
    """Build sparse matrix C where C @ b = conv(a, b)"""
    n = len(a)
    m = 2 * n - 1
    C = lil_matrix((m, n))
    for j in range(n):
        r_start = j
        r_end = min(j + n, m)
        if r_end > r_start:
            C[r_start:r_end, j] = a[j]
    return C.tocsr()

def _solve_lp(a, M, time_limit=5.0):
    """
    Solve: max sum(b) s.t. C*b <= M*1, b >= 0
    where C is the convolution matrix for a.
    Returns (b_opt, sum_b_opt) or (None, 0).
    """
    n = len(a)
    m = 2 * n - 1
    
    # Add small floor to a to ensure LP is bounded
    a_safe = np.maximum(a, 1e-15)
    
    C = _build_conv_matrix(a_safe)
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: C*b <= M*1
    b_ub = M * np.ones(m)
    
    # Bounds: 0 <= b[i] <= M / min(a_safe) (to ensure boundedness)
    b_max = M / a_safe.min()
    bounds = [(0.0, b_max)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x, float(-result.fun)
        elif result.status == 9:  # time limit
            # Return best found so far? HiGHS doesn't easily give this
            return None, 0
        else:
            return None, 0
    except Exception as e:
        print(f"LP error: {e}")
        return None, 0

def _lp_optimize_a(a_init, deadline, max_iters=200, blend_rate=0.5, 
                   lp_time_limit=3.0, verbose=True):
    """
    LP-based alternating optimization.
    a_init should be normalized to sum=1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    t_start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        
        b_opt, sum_b = _solve_lp(a, M, time_limit=lp_time_limit)
        
        if b_opt is None or sum_b <= 1.0 + 1e-10:
            # At (or near) local optimum
            if verbose:
                print(f"  LP iter {it}: sum_b={sum_b:.6f}, stopping (at local opt)")
            break
        
        # Blend
        b_norm = b_opt / sum_b  # normalize to sum 1
        a_new = (1 - blend_rate) * a + blend_rate * b_norm
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc_new = _score_norm(a_new)
        
        if verbose and (it % 5 == 0 or it < 5):
            t_elapsed = time.monotonic() - t_start
            print(f"  LP iter {it}: M={M:.6e}, sum_b={sum_b:.4f}, score={sc_new:.10f}, t={t_elapsed:.1f}s")
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_sc

def _fast_cd_polish(a, deadline, max_rounds=100):
    """Fast coordinate descent: try scaling individual entries to reduce max(conv)."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _fft_convolve(a, a)
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        c_max = float(c.max())
        
        # Find positions where conv is close to max (active set)
        active_tol = c_max * 1.01
        active = np.where(c >= active_tol)[0]
        
        for k in active:
            # Which a[j] affect conv[k]?
            if k < n:
                js = range(max(0, k - n + 1), min(k + 1, n))
            else:
                js = range(k - n + 1, n)
            
            for j in js:
                if a[j] < 1e-15:
                    continue
                for scale in [0.5, 0.7, 0.85, 1.2, 1.5, 0.3]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(2.0 * n * c_cand.max())
                    if sc < best_sc * (1.0 - 1e-10):
                        best_sc = sc
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        c = c_cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_sc

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _adam_log_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                    max_iter=100000, active_k=3, decay=0.6,
                    eval_interval=200, clip_norm=0.5):
    """Adam in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if T < T_start * 0.05:
            g = _compute_active_gradient(a, c, active_k, decay)
        else:
            g = _compute_smooth_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5:
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_smooth_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

def _compute_active_gradient(a, c, top_k=3, decay=0.7):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

LP-based alternating optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Core innovation: solve the LP dual (interval covering problem) to find optimal
descent directions, combined with Adam log-space polishing and CD finishing.

import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

# ... helper functions ...

def _build_dual_matrix(n):
    """Build the n x (2n-1) consecutive-ones constraint matrix for the dual LP."""
    m = 2 * n - 1
    A = lil_matrix((n, m))
    for j in range(n):
        i_end = min(j + n, m)
        A[j, j:i_end] = 1
    return A.tocsr()

def _solve_dual_lp(A, a, M):
    """
    Solve dual LP: min M * sum(y) s.t. A*y >= 1/a, y >= 0
    Returns (y_opt, sum_y_opt) or (None, 0).
    """
    n = a.shape[0]
    m = A.shape[1]
    
    a_safe = np.maximum(a, 1e-15)
    c = M * np.ones(m)
    # A*y >= 1/a  →  -A*y <= -1/a
    A_ub = -A
    b_ub = -1.0 / a_safe
    bounds = [(0, None)] * m
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, float(np.sum(result.x))
    return None, 0
