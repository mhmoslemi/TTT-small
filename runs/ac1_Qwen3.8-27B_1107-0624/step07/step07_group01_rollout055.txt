5. Multi-length exploration and basin hopping
6. Freeze-and-optimize block coordinate refinement
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _smoobj_z(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_z(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth_z(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-14):
    """L-BFGS in z-space (softmax parameterization)."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj_z(z, beta, n)

    def jac(z):
        return _smograd_z(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    """Adaptive L-BFGS with tight beta schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Tighter initial betas
    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                              0.0005, 0.0002, 0.0001, 5e-5, 2e-5,
                              1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, (n / 4096.0) ** 1.1)
        max_iter = min(15000, max(200, int(remaining / (0.003 * scale * 15))))

        a_new = _lbfgs_smooth_z(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.501:
            break

    return current_a


def _proj_subgrad_nesterov(a_init, deadline, k_active=5, max_steps=50000,
                           lr_init=0.02, momentum=0.9):
    """Projected subgradient with Nesterov momentum on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()  # Nesterov extrapolation point
    v = np.zeros(n)  # momentum velocity
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        a = _proj_simplex(y - momentum * v)
        a = a / max(a.sum(), 1e-15)

        c = _full_convolve(a, a)
        c_max = c.max()

        if c_max < 1e-15:
            break

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()  # make orthogonal to all-ones (simplex tangent)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Update
        v = momentum * v + grad / (gnorm + 1e-30)
        y = y - lr * v
        v = v / (np.linalg.norm(v) + 1e-30)

        # Evaluate and track best
        if i % 200 == 0:
            a_check = _proj_simplex(y)
            a_check = a_check / max(a_check.sum(), 1e-15)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.5)
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95

    a_final = _proj_simplep(y) if False else _proj_simplex(y)
    a_final = a_final / max(a_final.sum(), 1e-15)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a, best_sc


def _multiplicative_weights(a_init, deadline, k_active=5, max_steps=30000,
                            lr_init=0.05):
    """Multiplicative weights (row-action) method for the simplex."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        if c_max < 1e-15:
            break

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Multiplicative update: reduce a[i] where it contributes to peaks
        contrib = h  # contribution of each position to the active peaks
        # Normalize contribution
        contrib_max = contrib.max()
        if contrib_max > 1e-15:
            contrib = contrib / contrib_max

        # Update: reduce high-contribution positions, increase low ones
        a_new = a * np.exp(-lr * (contrib - contrib.min()))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            lr *= 0.5
            stall += 1
            if stall > 50:
                break
            continue
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            stall = 0
            lr = min(lr * 1.02, 0.5)
        else:
            stall += 1
            if stall > 20:
                lr *= 0.7
                if lr < 1e-8:
                    break
            else:
                lr *= 0.95

        a = a_new

    return best_a, best_sc


def _deflation_step(a, deadline, n_steps=50):
    """
    Deflation: identify peak-contributing pairs and redistribute mass.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        peak_k = int(np.argmax(c))

        # Find dominant contributing pairs for the peak
        # c[peak_k] = sum_i a[i] * a[peak_k - i]
        valid_i = range(max(0, peak_k - n + 1), min(n, peak_k + 1))
        pairs = []
        for i in valid_i:
            j = peak_k - i
            if 0 <= j < n:
                pairs.append((i, j, a[i] * a[j]))
        pairs.sort(key=lambda x: -x[2])

        if not pairs:
            break

        # Take the top few dominant pairs
        top_pairs = pairs[:3]
        total_pair_contrib = sum(p[2] for p in top_pairs)
        if total_pair_contrib < 1e-15:
            break

        # Try to reduce mass at the dominant positions
        # Move mass from high-contribution positions to low-contribution ones
        contrib = np.zeros(n)
        for i, j, val in pairs:
            contrib[i] += val
            if j != i:
                contrib[j] += val * 0.5

        contrib_max = contrib.max()
        if contrib_max < 1e-15:
            break

        # Find positions with low contribution (safe to add mass)
        low_mask = contrib < 0.2 * contrib_max
        low_positions = np.where(low_mask)[0]

        if len(low_positions) < n // 4:
            low_mask = contrib < 0.5 * contrib_max
            low_positions = np.where(low_mask)[0]

        if len(low_positions) == 0:
            break

        # Compute how much to move
        move_frac = 0.02 + 0.01 * step
        # Reduce at high-contrib positions, increase at low-contrib
        high_mask = contrib > 0.5 * contrib_max
        high_positions = np.where(high_mask)[0]

        if len(high_positions) == 0:
            break

        total_high = a[high_positions].sum()
        if total_high < 1e-15:
            break

        # Tentative move
        a_try = a.copy()
        extracted = total_high * move_frac
        a_try[high_positions] -= extracted * a_try[high_positions] / max(total_high, 1e-15)
        a_try = np.maximum(a_try, 0.0)
        # Distribute to low positions
        low_sum = a_try[low_positions].sum()
        if low_sum > 1e-15:
            a_try[low_positions] += extracted * a_try[low_positions] / low_sum
        else:
            a_try[low_positions] += extracted / len(low_positions)

        s = a_try.sum()
        if s < 1e-30:
            break
        a_try = a_try / s
        a_try = np.maximum(a_try, 0.0)
        a_try = a_try / max(a_try.sum(), 1e-30)

        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def _freeze_optimize(a_init, deadline, block_size=128, n_rounds=5):
    """
    Freeze-and-optimize: split into blocks, optimize each block while
    fixing others.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    n_blocks = max(1, n // block_size)
    block_size_actual = n // n_blocks

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size_actual
            end = min(n, start + block_size_actual)
            if end - start < 4:
                continue

            # Optimize this block using L-BFGS in z-space
            # Fix other blocks
            block_len = end - start
            z_block = np.log(np.maximum(a[start:end], 1e-18))

            fixed_part = np.concatenate([a[:start], a[end:]])
            fixed_sum = float(fixed_part.sum())

            def block_obj(z_b):
                p_b = _softmax(z_b)
                p_full = np.concatenate([fixed_part[:start], p_b, fixed_part[start:]])
                c = _full_convolve(p_full, p_full)
                return c.max()

            try:
                res = minimize(block_obj, z_block, method='L-BFGS-B',
                               options={'maxiter': 200, 'disp': False})
                new_block = _softmax(res.x)
                a_new = a.copy()
                a_new[start:end] = new_block
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                    sc = _score(a_new)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        a = a_new
            except Exception:
                pass

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_oscillatory(n, freq=0.7, mod=0.4):
    """Oscillatory pattern that might reduce center autocorrelation."""
    t = np.arange(n)
    envelope = 0.5 + 0.5 * np.sin(np.pi * t / n)
    osc = 1.0 + mod * np.cos(2.0 * np.pi * freq * t / n * n / (n // max(1, freq * n)))
    a = envelope * osc
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=20.0):
    """Basin hopping with diverse perturbation strategies."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(30.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = min(deadline - reserve, time.monotonic() + time_per_hop * 0.6)

        pert_type = h % 4
        if pert_type == 0:
            # Gaussian in log-space
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier perturbation (low freq)
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(1 + h // 2):
                freq = int(rng.integers(1, max(2, n // 64)))
                amp = 0.015 * rng.uniform(0.5, 3.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Block shuffle
            a_pert = best_a.copy()
            n_blocks = 4 + h
            block_sz = n // n_blocks
            perm = rng.permutation(n_blocks)
            for old_b, new_b in enumerate(perm):
                s1 = old_b * block_sz
                e1 = s1 + block_sz
                s2 = new_b * block_sz
                e2 = s2 + block_sz
                if old_b != new_b:
                    a_pert[s2:e2] = best_a[s1:e1]
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
        else:
            # Random restart blended with current
            a_rand = _make_random_positive(n, rng)
            blend = 0.3 + 0.1 * h
            a_pert = (1 - blend) * best_a + blend * a_rand
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        beta = mc * 0.005
        a_opt = _lbfgs_smooth_z(a_pert, beta, n, 4000)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Hop {h} improved: {best_sc:.10f}")

    return best_a, best_sc


def _final_micro_polish(a_init, deadline):
    """Very tight final polishing with multiple methods."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Ultra-tight L-BFGS
    c_b = _full_convolve(best_a, best_a)
    mb = float(c_b.max())
    for beta_frac in [1e-6, 5e-7, 2e-7, 1e-7, 5e-8]:
        if time.monotonic() > deadline:
            break
        beta = mb * beta_frac
        max_iter = min(8000, max(500, int((deadline - time.monotonic()) / 0.004)))
        a_f = _lbfgs_smooth_z(best_a, beta, n, max_iter)
        sc_f = _score(a_f)
        if sc_f < best_sc:
            best_a = a_f.copy()
            best_sc = sc_f

    # Subgradient with very small steps
    if time.monotonic() < deadline - 1:
        a_s, sc_s = _proj_subgrad_nesterov(best_a, deadline, k_active=3,
                                           max_steps=20000, lr_init=0.01)
        if sc_s < best_sc:
            best_a = a_s.copy()
            best_sc = sc_s

    # Deflation
    if time.monotonic() < deadline - 1:
        a_d, sc_d = _deflation_step(best_a, deadline, n_steps=100)
        if sc_d < best_sc:
            best_a = a_d.copy()
            best_sc = sc_d

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining projected subgradient, L-BFGS,
    multiplicative weights, deflation, and basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization (budget={budget_s}s) ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        print("  No parent available, using structured inits.")
        n_init = 4096
        parent = _make_chirp(n_init, freq=1.0, eps=0.3)
        try_update(parent, "chirp_init")

    # Also try a few diverse initializations
    n_test = len(parent)
    diverse_inits = [
        ("multiband_13", _make_multiband(n_test, bands=[1, 3], eps=0.3)),
        ("tapered", _make_tapered(n_test, power=0.7)),
        ("oscillatory", _make_oscillatory(n_test, freq=0.7, mod=0.4)),
        ("random1", _make_random_positive(n_test, rng)),
        ("chirp2", _make_chirp(n_test, freq=2.0, eps=0.25)),
    ]

    for name, a_init in diverse_inits:
        sc = _score(a_init)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_init, name)

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f} (n={best_n})")

    # === Phase 2: Main L-BFGS ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=500.0)
    try_update(a_p2, "main_lbfgs")

    # === Phase 3: Projected subgradient with Nesterov ===
    print(f"\n=== Phase 3: Projected subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        a_sg, sc_sg = _proj_subgrad_nesterov(best_seq, deadline - 400,
                                             k_active=5, max_steps=30000, lr_init=0.03)
        try_update(a_sg, "proj_sg_k5")

    if time.monotonic() < deadline - 350:
        a_sg2, sc_sg2 = _proj_subgrad_nesterov(best_seq, deadline - 350,
                                               k_active=3, max_steps=20000, lr_init=0.02)
        try_update(a_sg2, "proj_sg_k3")

    # === Phase 4: Multiplicative weights ===
    print(f"\n=== Phase 4: Multiplicative weights (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        a_mw, sc_mw = _multiplicative_weights(best_seq, deadline - 300,
                                              k_active=5, max_steps=20000, lr_init=0.03)
        try_update(a_mw, "mult_weights")

    if time.monotonic() < deadline - 280:
        a_mw2, sc_mw2 = _multiplicative_weights(best_seq, deadline - 280,
                                                k_active=3, max_steps=15000, lr_init=0.02)
        try_update(a_mw2, "mult_weights_k3")

    # === Phase 5: Deflation ===
    print(f"\n=== Phase 5: Deflation (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 250:
        a_def, sc_def = _deflation_step(best_seq, deadline - 250, n_steps=100)
        try_update(a_def, "deflation")

    # === Phase 6: Second L-BFGS round ===
    print(f"\n=== Phase 6: Second L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_p6 = _adaptive_lbfgs(best_seq, deadline, reserve_time=150.0)
        try_update(a_p6, "lbfgs_r2")

    # === Phase 7: Freeze-and-optimize ===
    print(f"\n=== Phase 7: Freeze-optimize (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        a_fo, sc_fo = _freeze_optimize(best_seq, deadline - 150,
                                       block_size=256, n_rounds=3)
        try_update(a_fo, "freeze_opt")

    # === Phase 8: Basin hopping ===
    print(f"\n=== Phase 8: Basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 100, rng, n_hops=5, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        lengths_to_try = [8192, 2048, 16384, 12288]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 40:
                break
            if abs(target_n - best_n) < target_n // 4:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")
            try_update(a_interp, f"interp_n{target_n}")

            if time.monotonic() < deadline - 30:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"opt_n{target_n}")

    # === Phase 10: Final micro-polishing ===
    print(f"\n=== Phase 10: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 2:
        for rnd in range(3):
            if time.monotonic() > deadline - 2:
                break
            a_fin, sc_fin = _final_micro_polish(best_seq, deadline - 1)
            try_update(a_fin, f"final_{rnd}")

            if time.monotonic() < deadline - 2:
                a_d, sc_d = _deflation_step(best_seq, deadline - 1, n_steps=50)
                try_update(a_d, f"final_def_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
