```python
def _solve_lp(a, M, n):
    """Solve: max sum(a') s.t. (a*a')_k <= M for all k, a' >= 0"""
    from scipy.optimize import linprog
    
    # Build constraint matrix A (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k-n+1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            A[k, i] = a[k-i]
    
    # Objective: maximize sum(a') → minimize -sum(a')
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    return result.x
```

```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_step(a, M, n):
    """One step of LP successive approximation.
    Solve: max sum(a') s.t. (a*a')_k <= M for all k, a' >= 0
    Returns the normalized solution.
    """
    # Build sparse constraint matrix A (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: max sum(a') → min -sum(a')
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if result.success:
        a_prime = result.x
        s = a_prime.sum()
        if s > 0:
            return a_prime / s
    return a.copy()

def _lp_iterate(a_init, deadline, max_iters=50, tol=1e-8):
    """Run LP successive approximation for max_iters steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        a_new = _lp_step(a, M, n)
        
        # Check if we improved
        c_new = _full_convolve(a_new, a_new)
        M_new = float(c_new.max())
        
        sc_new = 2.0 * n * M_new  # since sum(a_new) = 1
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Check convergence
        if abs(M_new - M) < tol * M:
            break
        
        a = a_new
    
    return best_a
```

```python
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _build_sparse_A(a, n):
    """Build (2n-1) x n sparse matrix with A[k,i] = a[k-i] if 0<=k-i<=n-1."""
    n_rows = 2 * n - 1
    total_nz = n * n
    data = np.empty(total_nz)
    row_idx = np.empty(total_nz, dtype=np.int32)
    col_idx = np.empty(total_nz, dtype=np.int32)

    idx = 0
    arr = np.arange(n)
    for k in range(n):
        length = k + 1
        row_idx[idx:idx + length] = k
        col_idx[idx:idx + length] = arr[:length]
        data[idx:idx + length] = a[k:-length - 1:-1] if length > 1 else a[k:k+1]
        idx += length

    for k in range(n, 2 * n - 1):
        length = 2 * n - 1 - k
        i_min = k - n + 1
        row_idx[idx:idx + length] = k
        col_idx[idx:idx + length] = arr[i_min:i_min + length]
        data[idx:idx + length] = a[n - 1:n - 1 - length:-1] if length > 1 else a[n - 1:n]
        idx += length

    return csr_matrix((data[:idx], (row_idx[:idx], col_idx[:idx])), shape=(n_rows, n))


def _lp_step(a, M, n):
    """Solve: max sum(a') s.t. (a*a')_k <= M for all k, a' >= 0.
    Returns normalized solution or None if infeasible."""
    A = _build_sparse_A(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                         method='highs', options={'time_limit': 60, 'presolve': True})
        if result.success:
            a_prime = result.x
            s = float(a_prime.sum())
            if s > 1e-15:
                return a_prime / s
    except Exception:
        pass
    return None


def _lp_iterate(a_init, deadline, max_iters=40, tol=1e-9):
    """LP successive approximation. Returns best sequence found."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()

        c = _full_convolve(a, a)
        M = float(c.max())

        a_new = _lp_step(a, M, n)
        if a_new is None:
            break

        c_new = _full_convolve(a_new, a_new)
        M_new = float(c_new.max())
        sc_new = float(2.0 * n * M_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    LP it={it}: M={M:.8f} -> M_new={M_new:.8f} score={sc_new:.10f} "
                  f"(dt={time.monotonic()-t0:.1f}s)")

        if abs(M_new - M) < tol * max(M, 1e-10):
            print(f"    LP converged at it={it}")
            break

        a = a_new

    return best_a


def _active_set_newton(a_init, deadline, max_iter=300, tol_start=0.002, tol_end=0.0002):
    """Active-set Newton with adaptive tolerance (geometric schedule)."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive tolerance: geometric from tol_start to tol_end
        frac = it / max(1, max_iter - 1)
        tol = tol_start * (tol_end / tol_start) ** frac

        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n

        m = n_active
        A_vec = active.astype(np.float64)

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-7 * (c_max / max(m, 1)) + 1e-9

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve
        u = np.zeros(n)
        r = g_t - hess_matvec(u)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            u += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-13 * (1 + gnorm):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        w = np.zeros(n)
        r = np.ones(n) - hess_matvec(w)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(300):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            w += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u - np.mean(-u)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w - np.mean(-u + lam * w)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-11 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            improved = False
            for ls in range(20):
                a_try = np.maximum(a - t * g_t, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-11 * t * gnorm:
                    a = a_try
                    improved = True
                    break
                t *= 0.5
            if not improved:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.000008])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 6))))

        try:
            result = minimize(
                _lsobj, z, args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _symmetric_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    """Active-set Newton respecting a_i = a_{n-1-i} symmetry."""
    n = len(a_init)
    if n % 2 == 0:
        half_n = n // 2
    else:
        half_n = n // 2 + 1

    # Extract half
    a_half = a_init[:half_n].copy()
    # Build full from half
    def _expand(half):
        if n % 2 == 0:
            return np.concatenate([half, half[::-1]])
        else:
            return np.concatenate([half, half[-2::-1]])

    a = _expand(a_half)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    a_half = a[:half_n].copy()

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol_active * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            top_idx = np.argsort(c)[::-1][:2 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 2 * n

        m = n_active
        A_vec = active.astype(np.float64)

        # Gradient wrt a (full)
        g_full = (2.0 / m) * _corr_a_A(a, A_vec, n)
        # Gradient wrt a_half (symmetric reduction)
        if n % 2 == 0:
            g_half = g_full[:half_n] + g_full[half_n:][::-1]
        else:
            g_half = g_full[:half_n].copy()
            g_half += g_full[half_n:][::-1]
            g_half[-1] = g_full[:-1][-1]  # center element only counted once
            # Actually for odd n, the center a[n//2] maps to itself
            # g_full[i] for i < n//2 pairs with g_full[n-1-i]
            # center index n//2 maps to itself
            g_half = np.zeros(half_n)
            for i in range(half_n):
                g_half[i] = g_full[i]
                if n - 1 - i != i:
                    g_half[i] += g_full[n - 1 - i]

        g_half_t = g_half - np.mean(g_half)
        gnorm = np.linalg.norm(g_half_t)
        if gnorm < 1e-14:
            break

        # For simplicity, just do gradient step in symmetric space
        lr = 0.1 / (1 + 0.01 * it)
        a_half_new = a_half - lr * g_half_t
        a_half_new = np.maximum(a_half_new, 0)
        s_new = a_half_new.sum()
        if s_new < 1e-15:
            break
        a_half_new = a_half_new / s_new
        a_new = _expand(a_half_new)
        sc_new = _score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a_half = a_half_new.copy()
            a = a_new.copy()
        else:
            lr *= 0.5

    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shuffle_profile(n, rng, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * t)) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 2 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-50 * (t - 0.3) ** 2) + np.exp(-50 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "m_seq_soft":
        # Softened m-sequence
        period = 63
        # Primitive polynomial x^6 + x + 1
        state = 1
        m_vals = np.zeros(n)
        for i in range(n):
            m_vals[i] = 1 - 2 * (state & 1)
            new_bit = ((state >> 5) ^ state) & 1
            state = ((state >> 1) | (new_bit << 5)) & 0x3F
        a = 0.5 + 0.3 * m_vals
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining LP successive approximation,
    adaptive active-set Newton, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _shuffle_profile(n_parent, rng, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: LP successive approximation at reduced n
    print(f"\n[Phase 1] LP successive approximation (t={time.monotonic() - t_start:.1f}s)")
    n_lp = 2048
    if time.monotonic() < deadline - 600:
        if parent is not None and n_parent != n_lp:
            a_lp = _interpolate_to_n(parent, n_lp)
        else:
            a_lp = _shuffle_profile(n_lp, rng, "uniform")
        print(f"  Starting LP at n={n_lp}, initial score={_score(a_lp):.10f}")
        lp_deadline = min(deadline - 500, time.monotonic() + 300)
        a_lp_best = _lp_iterate(a_lp, lp_deadline, max_iters=30, tol=1e-9)
        sc_lp = _score(a_lp_best)
        print(f"  LP result n={n_lp}: score={sc_lp:.10f}")

        # Upsample and refine
        a_up = _interpolate_to_n(a_lp_best, 8192)
        a_up_opt = _run_lbfgs(a_up, deadline, min_time_remaining=300.0)
        update_best(a_up_opt, _score(a_up_opt), "lp_upsample_8192")

        # Also try n=4096 LP
        if time.monotonic() < deadline - 400:
            n_lp2 = 4096
            if parent is not None and abs(n_parent - n_lp2) < 100:
                a_lp2 = parent.copy()
            else:
                a_lp2 = _interpolate_to_n(a_lp_best, n_lp2)
            print(f"  Starting LP at n={n_lp2}, initial score={_score(a_lp2):.10f}")
            lp_deadline2 = min(deadline - 350, time.monotonic() + 200)
            a_lp2_best = _lp_iterate(a_lp2, lp_deadline2, max_iters=20, tol=1e-9)
            sc_lp2 = _score(a_lp2_best)
            print(f"  LP result n={n_lp2}: score={sc_lp2:.10f}")
            update_best(a_lp2_best, sc_lp2, f"lp_n{n_lp2}")

    # Phase 2: Active-set Newton on best so far
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as_deadline = min(deadline - 150, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=200,
                                         tol_start=0.002, tol_end=0.0003)
        update_best(a_as, sc_as, "as_newton_main")

        # Second pass with even tighter tolerance
        if time.monotonic() < deadline - 100:
            as_dl2 = min(deadline - 80, time.monotonic() + 60)
            a_as2, sc_as2 = _active_set_newton(a_as, as_dl2, max_iter=100,
                                                tol_start=0.001, tol_end=0.0001)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 3: L-BFGS on best
    print(f"\n[Phase 3] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_lb = _run_lbfgs(best_seq, deadline, min_time_remaining=60.0)
        update_best(a_lb, _score(a_lb), "lbfgs_refine")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - 40 - time.monotonic()
        if remaining < 30:
            break

        if best_seq is not None and len(best_seq) != n_try:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _shuffle_profile(n_try, rng, "uniform")

        sc_init = _score(a_init)
        # Quick L-BFGS
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=20.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick Newton
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 20:
            a_opt2, sc_opt2 = _active_set_newton(a_opt,
                                                  min(deadline - 15, time.monotonic() + 40),
                                                  max_iter=80, tol_start=0.001, tol_end=0.0003)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_strategies = [
            ("perturb_003", 0.03),
            ("perturb_008", 0.08),
            ("perturb_015", 0.15),
            ("perturb_025", 0.25),
        ]
        for label, sigma in restart_strategies:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r,
                                                  min(deadline - 5, time.monotonic() + 25),
                                                  max_iter=60, tol_start=0.001, tol_end=0.0005)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 6: Symmetry-constrained optimization
    print(f"\n[Phase 6] Symmetric optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_sym = _symmetric_newton(best_seq, min(deadline - 10, time.monotonic() + 60),
                                   max_iter=150, tol_active=0.001)
        update_best(a_sym[0], a_sym[1], "symmetric")

    # Phase 7: Final ultra-tight polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        a_fin, sc_fin = _active_set_newton(best_seq,
                                            min(deadline - 3, time.monotonic() + 60),
                                            max_iter=200, tol_start=0.0005, tol_end=0.00005)
        update_best(a_fin, sc_fin, "final_as_tight")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n_fin = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 2e-5
            remaining = deadline - 3 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj, z, args=(beta_final, n_fin),
                        jac=_lsgrad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14,
                                 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _shuffle_profile(256, rng, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _shuffle_profile(256, rng, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
