"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_max_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_scaled = beta * c
    c_max_val = np.max(c_scaled)
    if c_max_val > 700:
        c_max_val = 700
    shifted = c_scaled - c_max_val
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(np.max(c))
    return float(c_max_val / beta + np.log(s) / beta)


def _smoothed_max_grad(z, beta, n):

                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100, 'maxfun': max_iter * 2}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgradient_polish(a_init, deadline, n_max_steps=5000, lr_init=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        tol = 0.001 * c.max()
        active = c >= (c.max() - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        N = _next_pow2(4 * n)
        FA = np.fft.rfft(a, N)
        FW = np.fft.rfft(w, N)
        grad_a = np.fft.irfft(np.conj(FW) * FA, N)[:n] + np.fft.irfft(FA * FW, N)[:n]
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_ramp(n, alpha=1.0, p=1.0):
    t = np.linspace(0, 1, n)
    a = (1.0 + alpha * t) ** p
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        return _make_ramp(n, alpha=1.0, p=1.0)
    elif kind == "ramp2":
        return _make_ramp(n, alpha=2.0, p=1.0)
    elif kind == "ramp_sqrt":
        return _make_ramp(n, alpha=1.0, p=0.5)
    elif kind == "cos_skew":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.7 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.05 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")
            return True
        return False

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score(parent)
                print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
                update_best(parent, sc_parent, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_best = len(best_seq) if best_seq is not None else 8192

    # === Phase 1: Try diverse initializations at the best length ===
    print(f"\n[Phase 1] Diverse initializations (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        init_kinds = ["ramp", "ramp2", "ramp_sqrt", "cos_skew", "tapered", "exp_decay"]
        for kind in init_kinds:
            if time.monotonic() > deadline - 500:
                break
            a_init = _make_profile(n_best, kind)
            sc_init = _score(a_init)
            if sc_init < best_score:
                update_best(a_init, sc_init, f"init_{kind}")
                # If good, run continuation
                if time.monotonic() < deadline - 100:
                    a_opt = _run_continuation(a_init, min(deadline - 80, time.monotonic() + 120), min_time_remaining=60)
                    update_best(a_opt, _score(a_opt), f"continue_{kind}")
            else:
                # Still try continuation from this init if time permits
                if time.monotonic() < deadline - 300 and sc_init < best_score * 1.05:
                    a_opt = _run_continuation(a_init, min(deadline - 100, time.monotonic() + 90), min_time_remaining=60)
                    update_best(a_opt, _score(a_opt), f"continue_{kind}")

    # === Phase 2: Main continuation from best ===
    print(f"\n[Phase 2] Main continuation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_main = _run_continuation(best_seq, min(deadline - 300, time.monotonic() + 400), min_time_remaining=200)
        update_best(a_main, _score(a_main), "main_continuation")

    # === Phase 3: Active-set Newton ===
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_newton, sc_newton = _active_set_newton(best_seq, min(deadline - 150, time.monotonic() + 200), max_iter=300)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg, sc_sg = _subgradient_polish(best_seq, min(deadline - 80, time.monotonic() + 150), n_max_steps=5000, lr_init=0.008)
        update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 5: Try different lengths ===
    print(f"\n[Phase 5] Multi-length exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 12288, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_best and best_seq is not None:
            continue

        # Interpolate best to new length
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_profile(n_try, "ramp")

        sc_init = _score(a_init)
        a_opt = _run_continuation(a_init, min(deadline - 40, time.monotonic() + 120), min_time_remaining=30)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 25, time.monotonic() + 60), max_iter=100)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 6: Diverse restarts from best ===
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("log_perturb_03", lambda: _perturb(best_seq, 0.03, rng)),
            ("log_perturb_01", lambda: _perturb(best_seq, 0.01, rng)),
            ("fourier_perturb", lambda: _perturb_fourier(best_seq, rng, n_modes=4)),
            ("ramp_mix", lambda: 0.5 * best_seq + 0.5 * _make_profile(len(best_seq), "ramp")),
        ]
        for name, gen_func in restart_configs:
            if time.monotonic() > deadline - 25:
                break
            a_init = gen_func()
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s > 1e-10:
                a_init = a_init / s
            a_opt = _run_continuation(a_init, min(deadline - 15, time.monotonic() + 40), min_time_remaining=10)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"restart_{name}")

    # === Phase 7: Final combined polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very low beta continuation
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = 1.0 / (mb * 0.00005)
        remaining = deadline - 3 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(50, int(remaining / 0.008)))
            try:
                def fun_final(zz):
                    return _smoothed_max_obj(zz, beta_final, n)
                def jac_final(zz):
                    return _smoothed_max_grad(zz, beta_final, n)
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 40}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra_lbfgs")
            except Exception:
                pass

        # Final Newton
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_newton(best_seq, max(deadline - 2, time.monotonic()), max_iter=80)
            update_best(a_fin2, sc_fin2, "final_newton")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
