LP-inspired multiplicative weight optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses cross-correlation gradients with softmax-smoothed active set,
multiplicative (exponential) updates, adaptive temperature annealing,

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _fft_cross_correlate(a, p):
    """Compute cross-correlation: (a cross p)[j] = sum_m a[m] * p[m+j] for j=0..len(a)-1"""
    n = len(a)
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    cross = np.fft.irfft(np.conj(A) * P, N)
    return cross[:n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1 (or close to it)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

    """Compute softmax weights from convolution values"""
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    return p

                              max_iter=1000000, eval_interval=100):

    Multiplicative weight optimization in the simplex.
    a is maintained with sum=1 and a>=0.
    """

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a /= s
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    max_c = float(c.max())
    
    for t in range(max_iter):

        T = T_start * (T_end / T_start) ** frac

        g = 2.0 * _fft_cross_correlate(a, p)

        if max_g < 1e-15:

        lr = lr_scale / max_g
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a  # revert
        else:

        max_c_new = float(c.max())

            best_score = 2.0 * n * max_c_new
            best_a = a_new.copy()
        
        a = a_new
        max_c = max_c_new
        
        if t % eval_interval == 0:
            pass  # can add logging
    
    return best_a, best_score

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_gradient_from_active(a, active_set):
    """
    Compute (1/|S|) * sum_{k in S} grad conv(a,a)_k
    where grad_j = 2 * a[k-j] for valid indices.
    Returns gradient of (1/|S|) * sum_{k in S} conv(a,a)_k w.r.t. a.
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            m_min = k - j_max
            m_max = k - j_min
            g[j_min:j_max + 1] += 2.0 * a[m_min:m_max + 1][::-1]
    g /= len(active_set)
    return g


def _compute_gradient_softmax(a, c, T):
    """Compute gradient using softmax-weighted active set."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    # g[j] = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is 2 * cross_correlate(a, p)[j]
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    cross = np.fft.irfft(np.conj(A) * P, N)
    g = 2.0 * cross[:n]
    return g


def _optimization_loop(a_init, deadline, T_start_frac=0.01, T_end_frac=0.0001,
                       lr_scale=0.25, max_iter=2000000, eval_interval=200,
                       active_mode='softmax', top_k=5, restart_interval=5000,
                       restart_sigma=0.01):
    """
    Main multiplicative update optimization loop.
    a is kept with sum=1, a>=0.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    c = _fft_convolve(a, a)
    best_maxc = float(c.max())
    best_score = 2.0 * n * best_maxc
    max_c = best_maxc

    patience = 0
    best_patience_score = best_score
    rng = np.random.default_rng(12345)

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, (t + 1) / max_iter)
        T_start = T_start_frac * max_c
        T_end = T_end_frac * max_c
        T = T_start * (T_end / T_start) ** frac

        # Compute gradient
        if active_mode == 'sparse':
            k_indices = np.argsort(c)[::-1][:top_k]
            g = _compute_gradient_from_active(a, k_indices)
        else:
            g = _compute_gradient_softmax(a, c, T)

        max_g = float(g.max())
        if max_g < 1e-15 or not np.isfinite(max_g):
            break

        # Multiplicative update
        lr = lr_scale / max_g
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s_new

        c = _fft_convolve(a_new, a_new)
        max_c_new = float(c.max())

        if max_c_new < best_maxc:
            best_maxc = max_c_new
            best_score = 2.0 * n * max_c_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        max_c = max(max_c_new, max_c * 0.999)  # track running max for T scaling

        # Periodic restart to escape local minima
        if restart_interval > 0 and t > 0 and t % restart_interval == 0:
            if patience > eval_interval * 3:
                sigma = restart_sigma * (1 + patience / (eval_interval * 5))
                z = np.log(np.maximum(a, 1e-15)) + sigma * rng.standard_normal(n)
                z = z - z.mean()
                a_pert = np.exp(z)
                a_pert /= a_pert.sum()
                c_pert = _fft_convolve(a_pert, a_pert)
                if float(c_pert.max()) < best_maxc:
                    best_maxc = float(c_pert.max())
                    best_score = 2.0 * n * best_maxc
                    best_a = a_pert.copy()
                # Don't always accept perturbation; use best as new start
                if patience > eval_interval * 6:
                    a = best_a.copy()
                    c = _fft_convolve(a, a)
                    max_c = float(c.max())
                else:
                    a = a_pert
                    c = c_pert
                    max_c = float(c.max())
                patience = 0

    return best_a, best_score


def _coordinate_descent_polish(a, deadline, max_rounds=100, top_k=5):
    """
    Coordinate descent: try scaling individual a[j] to reduce max(conv).
    Only considers j that appear in the top-k convolution positions.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    max_c = float(c.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        k_indices = np.argsort(c)[::-1][:top_k]

        for k in k_indices:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min > j_max:
                continue
            # Try scaling each j in the contributing range
            js = range(j_min, j_max + 1)
            for j in js:
                for scale in [0.5, 0.7, 0.85, 1.2, 1.5, 0.3]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s_cand = a_cand.sum()
                    if s_cand < 1e-15:
                        continue
                    a_cand /= s_cand
                    c_cand = _fft_convolve(a_cand, a_cand)
                    max_cand = float(c_cand.max())
                    if max_cand < max_c * (1.0 - 1e-10):
                        max_c = max_cand
                        best_score = 2.0 * n * max_c
                        best_a = a_cand.copy()
                        c = c_cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_a, best_score


def _fast_targeted_cd(a, deadline, max_rounds=200):
    """
    Faster CD: only try the most impactful coordinates.
    """
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(best_a, best_a)
    max_c = float(c.max())
    best_score = 2.0 * n * max_c

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        k_star = int(np.argmax(c))
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        # Only try the top-contributing j's
        contributions = np.zeros(n)
        for j in range(j_min, j_max + 1):
            m = k_star - j
            if 0 <= m < n:
                contributions[j] = 2.0 * a[j] * a[m]
        # Get top contributors
        top_js = np.argsort(contributions[j_min:j_max + 1])[::-1][:10] + j_min

        for j in top_js:
            m = k_star - j
            if m < 0 or m >= n:
                continue
            # Optimal scaling: reduce a[j] and a[m] to balance
            # Try reducing the larger one
            for scale in [0.7, 0.8, 0.9, 0.5, 0.6]:
                a_cand = best_a.copy()
                a_cand[j] *= scale
                s_cand = a_cand.sum()
                if s_cand < 1e-15:
                    continue
                a_cand /= s_cand
                c_cand = _fft_convolve(a_cand, a_cand)
                max_cand = float(c_cand.max())
                if max_cand < max_c * (1.0 - 1e-12):
                    max_c = max_cand
                    best_score = 2.0 * n * max_c
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break

        # Also try the second-worst position
        if not improved:
            c_sorted = np.argsort(c)[::-1]
            for k2 in c_sorted[1:3]:
                j_min2 = max(0, k2 - n + 1)
                j_max2 = min(k2, n - 1)
                if j_min2 > j_max2:
                    continue
                contributions2 = np.zeros(n)
                for j in range(j_min2, j_max2 + 1):
                    m2 = k2 - j
                    if 0 <= m2 < n:
                        contributions2[j] = 2.0 * a[j] * a[m2]
                top_js2 = np.argsort(contributions2[j_min2:j_max2 + 1])[::-1][:5] + j_min2
                for j in top_js2:
                    for scale in [0.8, 0.7, 0.9]:
                        a_cand = best_a.copy()
                        a_cand[j] *= scale
                        s_cand = a_cand.sum()
                        if s_cand < 1e-15:
                            continue
                        a_cand /= s_cand
                        c_cand = _fft_convolve(a_cand, a_cand)
                        max_cand = float(c_cand.max())
                        if max_cand < max_c * (1.0 - 1e-12):
                            max_c = max_cand
                            best_score = 2.0 * n * max_c
                            best_a = a_cand.copy()
                            c = c_cand
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def propose_candidate():
