import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)

    for idx in range(len(lags)):

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:

        i_idx = np.arange(i_start, i_end + 1)
        a_idx = k - i_idx
        g[i_start:i_end + 1] += w * a[a_idx]

    if sy <= 1e-15:

    return max(1e-8, min(sy / yy, 1.0))

                accepted = True
                break

        if not accepted:
            prev_a = a.copy()
            prev_g = g.copy()
            lr *= 0.7
            stall += 1
            if lr < lr_min or stall > stall_threshold:
                break

        # Periodic check
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            rate = (it + 1) / max(1e-6, elapsed)
            remaining = deadline - time.monotonic()
            if it > 0:
                print(f"    GD iter={it}, score={best_sc:.10f}, lr={lr:.6f}, rate={rate:.0f}/s, time={elapsed:.1f}s")

    return best_a, best_sc


def _smoobj(z, beta, n):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = e / s
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e2 = np.exp(shifted)
    s2 = e2.sum()
    if s2 < 1e-300 or not np.isfinite(s2):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s2))


def _smograd(z, beta, n):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = e / s
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e2 = np.exp(shifted)
    s2 = e2.sum()
    if s2 < 1e-300 or not np.isfinite(s2):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e2 / s2
    # Gradient of score w.r.t. a: 2n * (conv(w, a_rev)[n-1:2n-1] - W)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * n * (h - W)
    # Chain rule: grad_z = a * grad_a - a * sum(a * grad_a)
    grad_z = a * grad_a - a * float(np.dot(a, grad_a))
    return grad_z


def _lbfgs_smooth(a_init, beta, max_iter=5000):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        z = result.x
        z = z - np.max(z)
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return a_init.copy()
        return e / s
    except Exception:
        return a_init.copy()


def _lbfgs_schedule(a_start, deadline, min_time_remaining=120.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break
        max_iter = min(8000, max(200, int(remaining / 0.005)))
        a_new = _lbfgs_smooth(current_a, beta, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc
            print(f"      L-BFGS beta={beta:.3e}: score={sc:.10f}")
        else:
            # Try a smaller beta directly
            beta2 = beta * 0.5
            a_new2 = _lbfgs_smooth(current_a, beta2, max(100, max_iter // 2))
            sc2 = _score(a_new2)
            if sc2 < current_sc - 1e-14:
                current_a = a_new2
                current_sc = sc2
                print(f"      L-BFGS beta={beta2:.3e}: score={sc2:.10f}")

    return current_a


def _perturb(a, rng, scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return a.copy()
    return e / s


def _perturb_fourier(a, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n, dtype=np.float64)
    for _ in range(3):
        freq = int(rng.integers(1, max(2, n // 10)))
        amp = rng.uniform(0.005, 0.02)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return a.copy()
    return e / s


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: fast structured gradient descent with multi-lag active set,
    periodic L-BFGS refinement, multi-restart, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(4096) / 4096
        print(f"No parent. Using uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init")

    # === Phase 1: Fast gradient descent on parent length ===
    print(f"\n=== Phase 1: Fast gradient descent (n={len(best_seq)}) ===")
    t_phase = time.monotonic()
    gd_deadline = deadline - 800
    a_gd, sc_gd = _gradient_descent(best_seq, gd_deadline, max_iters=500000, n_active_max=50)
    try_update(a_gd, "gd_p1")
    print(f"  Phase 1 done: t={time.monotonic()-t_phase:.1f}s, score={sc_gd:.10f}")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n=== Phase 2: L-BFGS refinement ===")
    t_phase = time.monotonic()
    lbfgs_deadline = deadline - 500
    a_lb = _lbfgs_schedule(best_seq, lbfgs_deadline, min_time_remaining=200.0)
    try_update(a_lb, "lbfgs_p2")
    print(f"  Phase 2 done: t={time.monotonic()-t_phase:.1f}s")

    # === Phase 3: More gradient descent (longer) ===
    print(f"\n=== Phase 3: Extended gradient descent ===")
    t_phase = time.monotonic()
    gd_deadline = deadline - 350
    a_gd2, sc_gd2 = _gradient_descent(best_seq, gd_deadline, max_iters=500000, n_active_max=30)
    try_update(a_gd2, "gd_p3")
    print(f"  Phase 3 done: t={time.monotonic()-t_phase:.1f}s, score={sc_gd2:.10f}")

    # === Phase 4: Short L-BFGS polish ===
    print(f"\n=== Phase 4: L-BFGS polish ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 200:
        a_lb2 = _lbfgs_schedule(best_seq, deadline - 150, min_time_remaining=100.0)
        try_update(a_lb2, "lbfgs_p4")
    print(f"  Phase 4 done: t={time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Multi-restart ===
    print(f"\n=== Phase 5: Multi-restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        ("gauss_01", "gauss", 0.01),
        ("gauss_03", "gauss", 0.03),
        ("gauss_05", "gauss", 0.05),
        ("fourier_1", "fourier", 0.0),
        ("fourier_2", "fourier", 0.0),
    ]
    for name, mode, scale in restart_configs:
        if time.monotonic() > deadline - 100:
            break
        if mode == "gauss":
            p_start = _perturb(best_seq, rng, scale)
        else:
            p_start = _perturb_fourier(best_seq, rng)
        sc_start = _score(p_start)
        print(f"  Restart [{name}]: start={sc_start:.10f}")

        # Quick gradient descent
        r_deadline = min(deadline - 60, time.monotonic() + 120)
        a_r, sc_r = _gradient_descent(p_start, r_deadline, max_iters=100000, n_active_max=40)
        try_update(a_r, f"restart_{name}_gd")

        # Quick L-BFGS
        if time.monotonic() < deadline - 40:
            a_r2 = _lbfgs_schedule(a_r, min(deadline - 30, time.monotonic() + 60),
                                   min_time_remaining=20.0)
            try_update(a_r2, f"restart_{name}_lbfgs")

    print(f"  Phase 5 done: t={time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length exploration ===")
    t_phase = time.monotonic()
    current_n = len(best_seq)
    target_lengths = []
    if current_n < 16384:
        target_lengths.append(16384)
    if current_n != 4096 and current_n > 4096:
        target_lengths.append(4096)

    for target_n in target_lengths:
        if time.monotonic() > deadline - 80:
            break
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"    Interpolated: score={sc_interp:.10f}")
        try_update(a_interp, f"interp_n{target_n}")

        # Quick GD
        r_deadline = min(deadline - 40, time.monotonic() + 60)
        a_r, sc_r = _gradient_descent(a_interp, r_deadline, max_iters=50000, n_active_max=40)
        try_update(a_r, f"gd_n{target_n}")

    print(f"  Phase 6 done: t={time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish ===")
    t_phase = time.monotonic()
    for cycle in range(4):
        if time.monotonic() > deadline - 5:
            break

        # GD
        if time.monotonic() < deadline - 5:
            a_f1, _ = _gradient_descent(best_seq, deadline - 5, max_iters=100000, n_active_max=20)
            try_update(a_f1, f"final_gd_{cycle}")

        # L-BFGS
        if time.monotonic() < deadline - 5 and best_seq is not None:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            if mb > 0:
                beta = mb * 1e-5
                a_f2 = _lbfgs_smooth(best_seq, beta, max_iter=2000)
                try_update(a_f2, f"final_lbfgs_{cycle}")

    print(f"  Phase 7 done: t={time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
