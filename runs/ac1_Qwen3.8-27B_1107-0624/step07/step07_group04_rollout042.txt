"""
Active-set KKT + L-BFGS log-sum-exp optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: exact KKT active-set iterations, dual weight tracking,
support reduction, multi-n exploration, and aggressive final polish.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where H[j,m] = 2*W[j+m]."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cross_corr(W, a, n):
    """Compute (W*a)[j] = sum_k W[k]*a[k-j] for j=0..n-1."""
    a_rev = a[::-1]
    out_len = len(W) + n - 1
    N = _next_pow2(out_len)
    F_W = np.fft.rfft(W, N)
    F_ar = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(F_W * F_ar, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _project_simplex(a):
    """Project onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(a)
    if n <= 1:
        return np.ones(n)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)


def _cg_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs via CG where H[j,m] = 2*W[j+m]. Uses regularization if needed."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            # Regularize
            Hp = Hp + 1e-10 * p
            pHp = float(p @ Hp)
            if abs(pHp) < 1e-30:
                break
        alpha = rs / (pHp + 1e-30 * np.sign(pHp) if pHp != 0 else 1e-30)
        if not np.isfinite(alpha) or abs(alpha) > 1e10:
            break
        x += alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            x -= alpha * p
            break
        x = x  # already updated
        r = r_new
        p = (rs_new / max(rs, 1e-30)) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _cross_corr(w, a, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)

    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
                       0.0078, 0.004, 0.002, 0.001, 0.0005, 0.00025,
                       0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        t_now = time.monotonic()
        if t_now + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - t_now
        if remaining < 2:
            break

        est_time = 0.001 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 20))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass

    best_a = a.copy()

        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with adaptive tolerance
        tol = max(1e-6 * c_max, 1e-15)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Support
        support = a > max(1e-8, 1e-6 * a.max())
        n_support = int(support.sum())
        if n_support == 0:
            break

        support_idx = np.where(support)[0]
        active_idx = np.where(active)[0]

        # Solve for w: need (w*a)[j] = c0 for j in support, sum(w) = 1, w >= 0 on active
        # System: M @ w = c0 * 1, 1^T w = 1
        # where M[j, k] = a[k-j] (valid range)
        
        # Build M (n_support x n_active)
        M_mat = np.zeros((n_support, n_active))
        for i, j in enumerate(support_idx):
            for k_pos, k in enumerate(active_idx):
                diff = k - j
                if 0 <= diff < n:
                    M_mat[i, k_pos] = a[diff]
                # Also the symmetric term: a[j-k] if valid
                # Actually (w*a)[j] = sum_k w[k]*a[k-j], so only one term per k
        
        # Augmented system: [M | 1; 1^T | 0] [w; c0] = [0; 1]
        # M @ w - c0 * 1 = 0
        # 1^T w = 1
        
        # If n_active >= n_support + 1: use least squares
        # Form: A_aug @ x = b_aug where x = [w; c0]
        A_aug = np.zeros((n_support + 1, n_active + 1))
        A_aug[:n_support, :n_active] = M_mat
        A_aug[:n_support, n_active] = -1.0  # -c0
        A_aug[n_support, :n_active] = 1.0   # sum w = 1
        A_aug[n_support, n_active] = 0.0
        b_aug = np.zeros(n_support + 1)
        b_aug[n_support] = 1.0

        # Solve (least squares if overdetermined)
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_aug, b_aug, rcond=1e-15)
            w_vec = sol[:n_active]
            w_vec = np.maximum(w_vec, 0)
            w_sum = w_vec.sum()
            if w_sum < 1e-15:
                # Fallback: uniform on active
                w_vec = np.ones(n_active) / n_active
            else:
                w_vec = w_vec / w_sum
        except Exception:
            w_vec = np.ones(n_active) / n_active

        # Build W vector
        W = np.zeros(2 * n - 1)
        W[active_idx] = w_vec

        # Newton step: minimize a^T H_w a on simplex
        # KKT: H_w @ a_new + lambda * 1 = 0 on support, a_new >= 0
        # Solve H_w @ y = 1 (full), then a_new = y / sum(y), project to simplex
        
        y = _cg_hankel(W, n, np.ones(n), max_iter=500)
        y_sum = float(y.sum())
        
        if y_sum > 1e-10 and np.all(np.isfinite(y)):
            a_candidate = y / y_sum
        else:
            # Fallback: gradient step
            grad = _hankel_matvec(a, W, n)
            grad_centered = grad - grad.mean()
            a_candidate = a - 0.01 * grad_centered

        a_candidate = np.maximum(a_candidate, 0)
        a_candidate = _project_simplex(a_candidate)

        # Line search
        c_cur = c_max
        accepted = False
        alpha_try = 1.0
        for ls in range(30):
            a_try = (1 - alpha_try) * a + alpha_try * a_candidate
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha_try *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_cur - 1e-18:
                a = a_try
                accepted = True
                break
            alpha_try *= 0.5
            if alpha_try < 1e-10:
                break

        if not accepted:
            # Try smaller step
            grad = _hankel_matvec(a, W, n)
            grad_c = grad - grad.mean()
            step = 0.01 / (1 + 0.01 * rnd)
            for ls in range(20):
                a_try = np.maximum(a - step * grad_c, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    step *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_cur:
                    a = a_try
                    break
                step *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            if verbose and rnd % 10 == 0:
                print(f"    KKT round {rnd}: score={sc:.10f}, |K|={n_active}, |S|={n_support}")

        # Check convergence
        if rnd > 5 and abs(sc - best_sc) < 1e-16 * best_sc:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient method with adaptive step and dynamic active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.03
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        h = _cross_corr(w, a, n)
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15 * n:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 10 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.06)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.02
            z = z_new
        else:
            z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _support_reduction(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            # Quick re-optimize
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt

    return best_a, best_sc


def _resample(a, n_new):
    """Resample sequence to new length using interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS + Active Set KKT + Subgradient + Support Reduction
    + Multi-n + Restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 10:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        t = np.linspace(0, 1, n_parent)
        parent = np.sin(np.pi * t) ** 0.5
        parent = np.maximum(parent, 1e-10)
        parent = parent / parent.sum()
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_main = len(parent)

    # Phase 1: L-BFGS main run
    print(f"\n[Phase 1] L-BFGS main (n={n_main}, t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=400.0, verbose=True)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Active Set KKT polish
    print(f"\n[Phase 2] Active Set KKT (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        t2_deadline = min(deadline - 300, time.monotonic() + 250)
        a2, sc2 = _active_set_kKT(best_seq, t2_deadline, max_rounds=300, verbose=True)
        update_best(a2, sc2, "kkt_1")

    # Phase 3: Subgradient polish
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        t3_deadline = min(deadline - 200, time.monotonic() + 80)
        a3, sc3 = _subgrad_polish(best_seq, t3_deadline, max_steps=5000)
        update_best(a3, sc3, "subgrad_1")

    # Phase 4: KKT polish after subgradient
    print(f"\n[Phase 4] KKT polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        t4_deadline = min(deadline - 150, time.monotonic() + 100)
        a4, sc4 = _active_set_kKT(best_seq, t4_deadline, max_rounds=200, verbose=True)
        update_best(a4, sc4, "kkt_2")

    # Phase 5: Second L-BFGS pass with tighter beta
    print(f"\n[Phase 5] L-BFGS pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a5 = _run_lbfgs(best_seq, deadline, min_time_remaining=80.0)
        update_best(a5, _score(a5), "lbfgs_2")

    # Phase 6: KKT final
    print(f"\n[Phase 6] KKT final (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        t6 = min(deadline - 80, time.monotonic() + 60)
        a6, sc6 = _active_set_kKT(best_seq, t6, max_rounds=150)
        update_best(a6, sc6, "kkt_3")

    # Phase 7: Support reduction
    print(f"\n[Phase 7] Support reduction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a7, sc7 = _support_reduction(best_seq, deadline, min_time_remaining=40.0)
        update_best(a7, sc7, "shrink")
        if sc7 < best_score + 0.001 and time.monotonic() < deadline - 40:
            a7b, sc7b = _active_set_kKT(best_seq, min(deadline - 30, time.monotonic() + 30), max_rounds=50)
            update_best(a7b, sc7b, "shrink_kkt")

    # Phase 8: Multi-n exploration
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == n_main:
            continue
        a_init = _resample(best_seq, n_try) if best_seq is not None else _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 35:
            a_np, sc_np = _active_set_kKT(a_opt, min(deadline - 30, time.monotonic() + 25), max_rounds=50)
            update_best(a_np, sc_np, f"n_{n_try}_kkt")

    # Phase 9: Restarts
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 25:
        for i in range(3):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.05, rng)
            else:
                # Shrink and perturb
                n_r = len(best_seq)
                p_shrunk = best_seq.copy()
                p_shrunk[np.argsort(p_shrunk)[:n_r // 5]] = 0
                p_shrunk = p_shrunk / p_shrunk.sum()
                p_start = _perturb(p_shrunk, 0.03, rng)

            if time.monotonic() > deadline - 18:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 12:
                a_rp, sc_rp = _active_set_kKT(a_r, min(deadline - 10, time.monotonic() + 20), max_rounds=40)
                update_best(a_rp, sc_rp, f"restart_{i}_kkt")

    # Phase 10: Ultra-fine final polish
    print(f"\n[Phase 10] Final ultra-fine polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_kKT(best_seq, deadline - 2, max_rounds=100)
        update_best(a_fin, sc_fin, "final_kkt")

        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 5e-8
            remaining = deadline - 2 - time.monotonic()
            if remaining > 1:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        else:
            final = final / s
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
