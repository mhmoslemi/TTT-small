Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:

2. Active-set Newton with adaptive tolerance

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_conv(a, w, n):
    """
    Compute grad[t] = 2 * sum_j a[j] * w[t+j] for t = 0,...,n-1
    where w has length 2n-1.
    """
    w_rev = w[::-1]
    c = _full_convolve(a, w_rev)
    # c has length n + (2n-1) - 1 = 3n - 2
    # g[t] = c[2n-2-t] for t=0,...,n-1
    # indices: 2n-2, 2n-3, ..., n-1
    g = c[n-1:2*n-1][::-1]
    return 2.0 * g

def _topk_subgrad_step(z, k, lr, n):
    """
    One step of top-k subgradient descent in log-softmax parametrization.
    z is the log-amplitude vector.
    Returns updated z and the score.
    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Find top-k positions
    topk_idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[topk_idx] = 1.0 / k
    
    # Gradient of (1/k) * sum_{j in topk} conv[a][j] w.r.t. a
    grad_a = _grad_conv(a, w, n)
    
    # Gradient w.r.t. z: grad_z = a * (grad_a - sum(a * grad_a))
    # (because a = softmax(z), so da/dz = a * (I - a*a^T) in the tangent space)
    # Actually, d/ dz [f(softmax(z))] = diag(a) * (grad_a - sum(a * grad_a) * ones)
    # Wait, more carefully: if a_i = exp(z_i)/sum(exp(z)), then
    # d a_i / d z_j = a_i (delta_ij - a_j)
    # So grad_z_j = sum_i a_i (delta_ij - a_j) * grad_a_i = grad_a_j * a_j - a_j * sum_i a_i * grad_a_i
    # = a_j * (grad_a_j - dot(a, grad_a))
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    
    # Update
    z_new = z - lr * grad_z
    z_new = z_new - z_new.mean()
    
    return z_new, c_max

def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.002):
    """
    Damped Newton method on the active set.
    Minimizes the average of the top conv values.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        # Cap active set size
        if n_active > 3 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:3*n]
            active[top_idx] = True
            n_active = 3 * n

        m = n_active
        A_vec = active.astype(np.float64)
        
        # Objective: f(a) = (1/m) * sum_{k in active} conv[a][k], subject to sum(a)=1, a>=0
        # grad[i] = (2/m) * sum_j a[j] * A_vec[i+j]
        g_raw = _grad_conv(a, A_vec / m, n)  # this gives 2 * sum_j a[j] * (A_vec/m)[j+i]
        # Wait, _grad_conv(a, w, n) computes 2 * sum_j a[j] * w[i+j]
        # So g_raw[i] = 2 * sum_j a[j] * (A_vec/m)[i+j] = (2/m) * sum_j a[j] * A_vec[i+j]
        # This is the gradient of (1/m) * sum_{k in active} conv[a][k] w.r.t. a[i]
        # (since d/d(a[i]) conv[a][k] = 2*a[k-i] when 0<=k-i<n)
        
        # Project to tangent space (sum = 0)
        g_t = g_raw - np.mean(g_raw)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Hessian: H[i,j] = (2/m) * A_vec[i+j] (symmetric, since A_vec is symmetric around n-1)
        # Wait, is A_vec symmetric? A_vec[k] = 1 if k is in active set.
        # H[i,j] = (2/m) * A_vec[i+j] (for the quadratic form a^T H a / 2 = (1/m) sum_k A_vec[k] * conv[a][k])
        # Actually, the Hessian of (1/m) * sum_k A_vec[k] * conv[a][k] is:
        # H[i,j] = (2/m) * A_vec[i+j] (counting both (i,j) and (j,i) contributions)
        # Wait: conv[a][k] = sum_{p+q=k} a[p]*a[q]
        # d^2/d(a[i])d(a[j]) conv[a][k] = 2 if i+j=k (from the a[i]*a[j] and a[j]*a[i] terms)
        # Actually: if i != j, the term a[i]*a[j] appears twice (once as (p=i,q=j) and once as (p=j,q=i)),
        # so the second derivative is 2. If i = j, the term is a[i]^2, second derivative is 2.
        # So H[i,j] = (2/m) * A_vec[i+j] for all i,j (when i+j < 2n-1).
        
        # Hessian-vector product: (Hv)[i] = (2/m) * sum_j A_vec[i+j] * v[j]
        # This is (2/m) * _grad_conv(v, A_vec, n) / 2 * ... 
        # _grad_conv(v, A_vec, n) = 2 * sum_j v[j] * A_vec[i+j]
        # So (Hv)[i] = (2/m) * sum_j A_vec[i+j] * v[j] = _grad_conv(v, A_vec/m, n)
        
        mu = 1e-8 + 1e-6 * c_max / m

        def hess_vec(v):
            return _grad_conv(v, A_vec / m, n) + mu * v

        # CG to solve (H + mu*I) * u = g_t
        # But we also need to handle the constraint sum(a) = 1
        # Use the standard approach: solve the KKT system
        
        # First, solve (H + mu*I) * u = g_t using CG
        def cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - hess_vec(x)
            p = r.copy()
            rs = np.dot(r, r)
            if rs < 1e-30:
                return x
            for cg_it in range(max_cg):
                Ap = hess_vec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = float(np.dot(r_new, r_new))
                if rs_new < 1e-24 * (1.0 + float(np.dot(rhs, rhs))):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w_vec = cg_solve(np.ones(n))
        sum_w = float(w_vec.sum())
        
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_vec
            dx = dx - np.mean(dx)

        # Line search with non-negativity
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99)
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-10:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Gradient fallback
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= f_old + 1e-10:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc

def _topk_descent(a_init, deadline, k_frac=0.05, lr_init=0.1, n_iters=20000):
    """
    Top-k subgradient descent in log-softmax space.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30).astype(np.float64))
    z = z - z.mean()
    best_a = a_init.copy().astype(np.float64)
    best_sc = _score(best_a)
    lr = lr_init
    patience = 0
    k = max(1, int(n * k_frac))
    
    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive k: increase k when the top values are very close
        top_vals = np.sort(c)[::-1]
        if len(top_vals) > k + 1:
            gap = top_vals[k-1] - top_vals[k]
            if gap < 0.01 * c_max:
                k = min(len(c), k + max(1, n // 100))
            elif gap > 0.1 * c_max and k > max(1, n // 100):
                k = max(1, k - max(1, n // 200))
        
        # Compute gradient
        topk_idx = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        w[topk_idx] = 1.0 / k
        grad_a = _grad_conv(a, w, n)
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        
        # Step
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        # Check improvement every 10 steps
        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    # Reset from best
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(0.01, lr * 0.5)
                else:
                    lr = max(0.005, lr * 0.95)
    
    return best_a, best_sc

def _make_asymmetric_profile(n, shift=0.2):
    """Create an asymmetric profile: Gaussian bump shifted from center."""
    t = np.linspace(0, 1, n)
    center = 0.5 + shift
    sigma = 0.3
    a = np.exp(-((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_half_profile(n, ratio=0.7):
    """First half large, second half small."""
    a = np.zeros(n)
    a[:n//2] = 1.0
    a[n//2:] = ratio
    return a / a.sum()

def _make_two_level(n, k, p, q):
    """Two-level sequence: p for first k elements, q for rest."""
    a = np.zeros(n)
    a[:k] = p
    a[k:] = q
    return a / a.sum()

def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _lbfgs_lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lbfgs_lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_a = _grad_conv(a, w, n)
    W = float(np.dot(a, grad_a))
    return a * (grad_a - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy().astype(np.float64)
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _lbfgs_lsobj,
                z,
                args=(beta, n),
                jac=lambda z_local, b, n_local: _lbfgs_lsgrad(z_local, b, n_local),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'f
