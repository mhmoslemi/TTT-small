            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-12:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(12000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _lsobj, z,
                args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _spectral_flatten(a, n_iter=30, alpha=0.4):
    """Iterative spectral equalization to reduce autocorrelation peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    N = _next_pow2(2 * n - 1)
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        mag_sq = np.abs(A) ** 2
        avg_mag_sq = float(mag_sq.mean())
        if avg_mag_sq < 1e-30:
            break
        # Scale to push toward flat |A|^2
        ratio = np.sqrt(np.maximum(avg_mag_sq, 1e-30) / np.maximum(mag_sq, 1e-30))
        ratio = np.clip(ratio, 0.3, 3.0)
        scale = 1.0 - alpha + alpha * ratio
        A_new = A * scale
        a_new = np.real(np.fft.irfft(A_new, N)[:n])
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if np.linalg.norm(a_new - a) < 1e-12:
            a = a_new
            break
        a = a_new
    return a


def _block_cd(a_init, block_size, n_rounds, deadline):
    """Block coordinate descent: optimize blocks sequentially."""
    n = len(a_init)
    a = a_init.copy()
    best_sc = _score(a)
    best_a = a.copy()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        for start in range(0, n, block_size):
            if time.monotonic() > deadline:
                break
            end = min(start + block_size, n)
            blen = end - start
            # Compute gradient contribution for this block
            c = _full_convolve(a, a)
            c_max = float(c.max())
            tol = 0.003 * c_max
            active = c >= (c_max - tol)
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()
            # Gradient for block
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad_full = 4.0 * n * a * (h - W)
            grad_block = grad_full[start:end].copy()
            # Project to tangent (within block, keep sum of block constant)
            grad_block_t = grad_block - np.mean(grad_block)
            # Line search
            f_old = float(c.max())
            lr = 0.01
            for step in range(30):
                a_try = a.copy()
                a_try[start:end] = a[start:end] - lr * grad_block_t
                a_try[start:end] = np.maximum(a_try[start:end], 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    lr *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                lr *= 0.5
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Subgradient method with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                best_patience_sc = sc
            else:
                patience += 1
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr * 0.7
                    if lr < 0.001:
                        break
                else:
                    lr *= 0.93

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / 0.05)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage(a, deadline):
    """Try zeroing smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.25]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Spectral flatten + L-BFGS from parent
    Phase 2: Active-set Newton (tight)
    Phase 3: Block CD
    Phase 4: Subgradient polish
    Phase 5: Shrinkage + re-optimize
    Phase 6: Multi-n exploration
    Phase 7: Basin-hopping restarts
    Phase 8: Final tight Newton
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        best_seq = parent.copy()
        best_score = sc_parent
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Phase 1: Spectral flatten + L-BFGS
    print(f"\n[Phase 1] Spectral + L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_spec = _spectral_flatten(parent, n_iter=20, alpha=0.5)
    sc_spec = _score(a_spec)
    print(f"  After spectral flatten: {sc_spec:.10f}")

    a_main_init = parent.copy()
    if sc_spec < best_score:
        a_main_init = a_spec

    a_lbfgs = _run_lbfgs(a_main_init, deadline, min_time_remaining=400.0)
    sc_lbfgs = _score(a_lbfgs)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")

    # Phase 2: Active-set Newton (tight)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        as_deadline = min(deadline - 350, time.monotonic() + 150)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=400, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton_1")

    if time.monotonic() < deadline - 300:
        as_deadline2 = min(deadline - 300, time.monotonic() + 100)
        a_as2, sc_as2 = _active_set_newton(best_seq, as_deadline2, max_iter=300, tol_active=0.0005)
        update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 3: Block CD
    print(f"\n[Phase 3] Block CD (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        n = len(best_seq)
        bs = max(64, n // 64)
        a_bc, sc_bc = _block_cd(best_seq, bs, n_rounds=4, deadline=min(deadline - 250, time.monotonic() + 80))
        update_best(a_bc, sc_bc, "block_cd")

    # Phase 4: Subgradient polish
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 200, time.monotonic() + 60),
                                       n_max_steps=3000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # Phase 5: Shrinkage + re-optimize
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_shr, sc_shr = _shrinkage(best_seq, min(deadline - 150, time.monotonic() + 30))
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=100.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")
            a_shr3, sc_shr3 = _active_set_newton(a_shr2, min(deadline - 80, time.monotonic() + 40),
                                                  max_iter=150, tol_active=0.0005)
            update_best(a_shr3, sc_shr3, "as_post_shrink")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_tries = [4096, 8192, 16384]
    for n_try in n_tries:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent:
            continue
        if parent is not None:
            a_init_n = _interpolate_to_n(parent, n_try)
        else:
            a_init_n = _make_profile(n_try, "uniform")
        a_opt_n = _run_lbfgs(a_init_n, deadline, min_time_remaining=40.0)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}_lbfgs")
        if sc_opt_n < best_score + 0.02 and time.monotonic() < deadline - 30:
            a_opt_n2, sc_opt_n2 = _active_set_newton(a_opt_n, min(deadline - 25, time.monotonic() + 30),
                                                      max_iter=100, tol_active=0.
