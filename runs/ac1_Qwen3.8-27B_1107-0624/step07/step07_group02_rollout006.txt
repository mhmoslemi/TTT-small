def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _softmax_obj(a, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _smoobj(z, beta, n):

        self.n = n

def _variance_obj(z, beta, n, lam):
    """Objective with variance penalty: smoothed_max + lam * Var(c)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        sm = c_max
    else:
        sm = c_max + beta * np.log(s)
    c_mean = float(c.mean())
    var = float(((c - c_mean) ** 2).mean())
    return sm + lam * var


def _variance_grad(z, beta, n, lam):
    """Gradient of variance-regularized objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    
    # Variance gradient
    c_mean = float(c.mean())
    # d(Var(c))/da = (2/(len(c))) * sum_k (c_k - mean) * dc_k/da
    # dc_k/da_j = 2*a_{k-j} (approximately)
    # d(Var)/da_j = (2/(2n-1)) * sum_k (c_k - mean) * 2*a_{k-j}
    # = (4/(2n-1)) * conv(c - mean, a_rev)[n-1:2n-1][j]
    dvar_da = (4.0 / (2 * n - 1)) * _full_convolve(c - c_mean, a_rev)[n - 1:2 * n - 1]
    
    # Combined gradient w.r.t. a
    grad_a = 2.0 * h + lam * dvar_da
    
    # Convert to gradient w.r.t. z
    W_total = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - W_total)
    return grad_z

        gap = c_sorted[0] * 0.001

        lams = np.zeros(len(betas))
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()

            break

def _multiplicative_update(a, deadline, lr_init=0.01, min_time_remaining=10.0):
    """Mirror descent using multiplicative updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-30)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Active set: top few % of convolution values
        n_active = max(1, int(0.05 * len(c)))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / n_active
        
        # Compute gradient direction
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = h - W  # This is the "direction" in log-space
        
        # Multiplicative update: a_new = a * exp(-lr * grad) / sum(a * exp(-lr * grad))
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        # Clip to prevent overflow
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:
            a = a_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
    
    return best_a, _score(best_a)


def _projected_gradient(a_init, deadline, lr_init=0.01, min_time_remaining=10.0):
    """Projected gradient descent with adaptive step size."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Active set
        n_active = max(1, int(0.03 * len(c)))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / n_active
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = h - W
        
        # Line search
        improved = False
        alpha = lr
        for _ in range(40):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            lr = min(0.05, lr * 1.03)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_a, _score(best_a)

            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.6
            if no_improve >= 300:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.02)
    
    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_flat":
        # Raised cosine with wider plateau
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "broad_taper":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2 * t - 1) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _multi_scale_optimize(n_final, deadline, min_time_remaining=200.0, seed=42):
    """Multi-scale: optimize at small n, upsample, optimize at larger n."""
    rng = np.random.default_rng(seed)
    
    # Start small and grow
    scales = []
    s = 128
    while s < n_final:
        scales.append(s)
        s *= 2
    if scales[-1] != n_final:
        scales.append(n_final)
    # Make sure we have reasonable number of scales
    if len(scales) > 5:
        # Skip some
        step = len(scales) // 5
        scales = scales[::step]
        if scales[-1] != n_final:
            scales.append(n_final)
    elif len(scales) < 3:
        scales = [128, max(256, n_final // 2), n_final]
    
    a = _make_profile(scales[0], "uniform")
    # Add small random perturbation
    a = a + 0.01 * rng.standard_normal(len(a))
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    for i, n_s in enumerate(scales):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if len(a) != n_s:
            a = _resample(a, n_s)
        
        remaining_for_this = min(deadline - time.monotonic() - min_time_remaining, 
                                  max(30, 300 * (n_s / n_final) ** 0.5))
        
        # L-BFGS
        a = _run_lbfgs(a, time.monotonic() + remaining_for_this, min_time_remaining=10.0)
        sc = _score(a)
        print(f"  Scale n={n_s}: score={sc:.10f}")
        
        # Quick projected gradient
        if time.monotonic() + min_time_remaining < deadline:
            a, sc = _projected_gradient(a, time.monotonic() + min(remaining_for_this * 0.3, 30),
                                         lr_init=0.01, min_time_remaining=0)
            print(f"  Scale n={n_s} after PG: score={sc:.10f}")
    
    return a


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    
    return best_a, best_sc


def _spectral_equalization(a, deadline):
    """Try to equalize the power spectrum to flatten the autocorrelation."""
    n = len(a)
    a_work = a.copy()
    s = a_work.sum()
    a_work = a_work / s
    best_a = a_work.copy()
    best_sc = _score(a_work)
    
    # Work in frequency domain
    N = _next_pow2(2 * n - 1)
    
    lr = 0.005
    for it in range(5000):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_work, a_work)
        sc = 2.0 * n * float(c.max())
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_work.copy()
        
        # Target: make |A(f)|^2 proportional to 1/(2n-1) in the "meaningful" band
        # Actually, we want the time-domain autocorrelation to be flat
        # This means |A(f)|^2 should be the DFT of a flat sequence
        # For a flat sequence of length 2n-1, its DFT is a Dirichlet kernel
        # So we want |A(f)| to look like a Dirichlet kernel
        
        # Alternative: just use the variance of c as a proxy
        c_mean = c.mean()
        c_var = ((c - c_mean) ** 2).mean()
        
        # Gradient of Var(c) w.r.t. a
        a_rev = a_work[::-1]
        grad_var = (4.0 / (2 * n - 1)) * _full_convolve(c - c_mean, a_rev)[n - 1:2 * n - 1]
        
        # Also add gradient of max(c)
        n_active = max(1, int(0.02 * len(c)))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_max = 2.0 * (h - W)
        
        grad = grad_max + 0.1 * grad_var
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-15:
            break
        
        alpha = lr / (gnorm + 1e-15)
        a_try = a_work - alpha * grad_t
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        
        if sc_try < sc - 1e-14:
            a_work = a_try
            lr = min(0.02, lr * 1.02)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_a, _score(best_a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining:
    1. Multi-scale L-BFGS optimization
    2. Projected gradient / multiplicative updates
    3. Spectral equalization
    4. Active-set polishing
    5. Shrinkage and multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")
    
    # === Phase 1: Multi-scale optimization on parent ===
    print(f"\n[P1] Multi-scale (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        ms_deadline = deadline - 500
        a_ms = _multi_scale_optimize(n_parent, ms_deadline, min_time_remaining=200.0, seed=seed)
        sc_ms = _score(a_ms)
        if sc_ms < best_score:
            best_seq = a_ms.copy()
            best_score = sc_ms
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [multiscale] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Main L-BFGS with variance penalty ===
    print(f"\n[P2] L-BFGS with variance (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_v = _run_lbfgs(best_seq, deadline - 400, min_time_remaining=200.0,
                         use_var_penalty=True, var_lam=0.001)
        sc_v = _score(a_v)
        if sc_v < best_score:
            best_seq = a_v.copy()
            best_score = sc_v
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_var] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Projected gradient ===
    print(f"\n[P3] Projected gradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pg_deadline = min(deadline - 300, time.monotonic() + 120)
        a_pg, sc_pg = _projected_gradient(best_seq, pg_deadline, lr_init=0.015)
        if sc_pg < best_score:
            best_seq = a_pg.copy()
            best_score = sc_pg
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [PG] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Spectral equalization ===
    print(f"\n[P4] Spectral equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        se_deadline = min(deadline - 250, time.monotonic() + 100)
        a_se, sc_se = _spectral_equalization(best_seq, se_deadline)
        if sc_se < best_score:
            best_seq = a_se.copy()
            best_score = sc_se
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [spectral] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Active-set polish (coarse) ===
    print(f"\n[P5] Active-set coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pol_deadline = min(deadline - 200, time.monotonic() + 100)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.005, tol_frac_end=0.0001,

    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:

    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.1f}s)")

            break

            continue

        remaining = deadline - time.monotonic() - 30

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)

    # === Phase 9: Diverse restarts ===
    print(f"\n[P9] Restarts (t={time.monotonic()-t_start:.1f}s)")
