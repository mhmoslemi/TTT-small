Key ideas: exploit the structure of the autocorrelation peak, use spectral
methods to shape the power spectrum for flatter auto-correlation, and combine
with local search for fine-tuning.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

def _spectral_shaping(a_init, deadline, max_iters=5000):

        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            break
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            lr = min(lr * 1.01, 0.2)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if lr < 1e-7 or stall > 200:
                break

    return best_a, best_sc


def _mirror_optimize(a_half, deadline):
    """Given a half-sequence, optimize it knowing the full sequence is mirrored."""
    n_half = len(a_half)
    n_full = 2 * n_half

    def mirror_and_score(z):
        a_h = _softmax(z)
        a_full = np.concatenate([a_h, a_h[::-1]])
        return _score(a_full), a_full

    def obj(z):
        c = _full_convolve(a_half, a_half)
        a_h = _softmax(z)
        a_full = np.concatenate([a_h, a_h[::-1]])
        c_full = _full_convolve(a_full, a_full)
        s = a_full.sum()
        return 2.0 * n_full * c_full.max() / (s * s)

    z0 = np.log(np.maximum(a_half, 1e-15))
    z0 = z0 - z0.mean()

    try:
        result = minimize(obj, z0, method='L-BFGS-B',
                          options={'maxiter': 5000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxcor': 80})
        a_h = _softmax(result.x)
        a_full = np.concatenate([a_h, a_h[::-1]])
        return a_full
    except Exception:
        return np.concatenate([a_half, a_half[::-1]])


def _coordinate_descent(a_init, deadline, max_rounds=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        # Try a few fractions
        fracs = [0.5, 0.4, 0.6, 0.3, 0.7, 0.2, 0.8, 0.1, 0.9]
        rng.shuffle(fracs)
        for frac in fracs:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * 2 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_z(a, rng, scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=8, amp_scale=0.01):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = amp_scale * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses L-BFGS beta continuation, spectral shaping, peak shaving,
    mirror construction, coordinate descent, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n0 = 4096
        a0 = _make_profile(n0, "uniform")
        print(f"No parent. Uniform n={n0}, score={_score(a0):.10f}")
        try_update(a0, "init")

    # === Phase 1: L-BFGS fine schedule on parent ===
    print(f"\n=== Phase 1: L-BFGS on n={len(best_seq)} ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 600:
        a_p1 = _lbfgs_schedule(best_seq, deadline - 500, reserve_time=400.0)
        try_update(a_p1, "lbfgs_p1")
    print(f"  Phase 1: {time.monotonic()-t_phase:.1f}s")

    # === Phase 2: Spectral shaping ===
    print(f"\n=== Phase 2: Spectral shaping ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 400:
        a_sp, sc_sp = _spectral_shaping(best_seq, deadline - 400, max_iters=3000)
        try_update(a_sp, "spectral")
    print(f"  Phase 2: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 350:
        a_ps, sc_ps = _peak_shaving(best_seq, deadline - 350, max_iters=80000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 3: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Mirror optimization ===
    print(f"\n=== Phase 4: Mirror optimization ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 250:
        n_cur = len(best_seq)
        n_half = n_cur // 2
        a_half = best_seq[:n_half].copy()
        a_half = np.maximum(a_half, 1e-10)
        a_half = a_half / a_half.sum()
        a_mirror = _mirror_optimize(a_half, deadline - 250)
        try_update(a_mirror, "mirror")

        # Also try a truncated mirror (odd length)
        n_half2 = n_cur // 2 + 1
        if n_half2 != n_half:
            a_half2 = best_seq[:n_half2].copy()
            a_half2 = np.maximum(a_half2, 1e-10)
            a_half2 = a_half2 / a_half2.sum()
            a_mirror2 = np.concatenate([a_half2, a_half2[-2::-1]])
            try_update(a_mirror2, "mirror_odd")
    print(f"  Phase 4: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    t_phase = time.monotonic()
    n_cur = len(best_seq)
    target_lengths = [8192, 6144, 12288, 16384]
    for tl in target_lengths:
        if time.monotonic() > deadline - 150:
            break
        if tl == n_cur:
            continue
        print(f"  Trying n={tl}...")
        a_interp = _interpolate_to_n(best_seq, tl)
        sc_interp = _score(a_interp)
        print(f"    Interpolated: score={sc_interp:.10f}")
        try_update(a_interp, f"interp_{tl}")

        if time.monotonic() < deadline - 100:
            a_opt = _lbfgs_schedule(a_interp, deadline - 80, reserve_time=60.0)
            try_update(a_opt, f"n{tl}_lbfgs")

        if time.monotonic() < deadline - 60:
            a_ps2, _ = _peak_shaving(a_opt, deadline - 50, max_iters=20000)
            try_update(a_ps2, f"n{tl}_ps")
    print(f"  Phase 5: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 6: Multi-restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        (0.01, "gauss_s1"),
        (0.03, "gauss_s3"),
        (0.08, "gauss_s8"),
        (0.15, "gauss_s15"),
    ]
    for scale, name in restart_configs:
        if time.monotonic() > deadline - 80:
            break
        a_p = _perturb_z(best_seq, rng, scale)
        a_f = _fourier_perturb(best_seq, rng)
        candidates = [a_p, a_f]
        for ci, cand in enumerate(candidates):
            if time.monotonic() > deadline - 60:
                break
            a_r = _lbfgs_schedule(cand, deadline - 40, reserve_time=30.0)
            label = f"restart_{name}_{ci}"
            try_update(a_r, label)
            if time.monotonic() < deadline - 30:
                a_r2, _ = _peak_shaving(a_r, deadline - 20, max_iters=10000)
                try_update(a_r2, f"{label}_ps")
    print(f"  Phase 6: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Ultra-fine L-BFGS on best ===
    print(f"\n=== Phase 7: Ultra-fine L-BFGS ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 40:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        ultra_betas = mb * np.array([1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8])
        for i, beta in enumerate(ultra_betas):
            if time.monotonic() > deadline - 25:
                break
            remaining = deadline - 25 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(10000, max(200, int(remaining / 0.004)))
            a_ultra, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_ultra, f"ultra_{i}")
    print(f"  Phase 7: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: Coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 15, max_rounds=15000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 8: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Final combined polish cycle ===
    print(f"\n=== Phase 9: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(4):
            if time.monotonic() > deadline - 5:
                break
            # Peak shave
            if time.monotonic() < deadline - 5:
                a_f1, _ = _peak_shaving(best_seq, deadline - 5, max_iters=20000, tol_frac=1e-6)
                try_update(a_f1, f"final_ps_{cycle}")
            # Fine L-BFGS
            if time.monotonic() < deadline - 5:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-7
                remaining = deadline - 5 - time.monotonic()
                if remaining > 3:
                    max_iter = min(5000, max(100, int(remaining / 0.004)))
                    a_f2, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                    try_update(a_f2, f"final_lbfgs_{cycle}")
            # Coord descent
            if time.monotonic() < deadline - 3:
                a_f3, _ = _coordinate_descent(best_seq, deadline - 3, max_rounds=5000)
                try_update(a_f3, f"final_cd_{cycle}")
    print(f"  Phase 9: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
