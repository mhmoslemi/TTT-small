"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Soft active-set with exponential weighting and Nesterov momentum
2. Adaptive lambda (peak focus) schedule with new-peak detection
3. Multi-n exploration (8192, 12288, 16384)
4. Quadratic line search for better step sizes
5. Symmetrization trick to reduce effective dimensionality
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


class _SmoothedProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * self.n * c_max
        return 2.0 * self.n * (c_max + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / self.beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[self.n - 1:2 * self.n - 1]
        W = float(np.dot(w, c))
        return 4.0 * self.n * a * (h - W)


def _lbfgs_optimize(a_init, deadline, min_remaining=30.0, min_time_per_phase=3.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.004)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.031, 0.016, 0.008, 0.004,
                                  0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < min_time_per_phase:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 20))))

        prob = _SmoothedProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-17,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _hard_active_set(a_init, deadline, tol_start=0.005, tol_end=0.0001,

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

            print(f"  n={n_try}: score={sc_opt:.10f} (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts at best n ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    restarts = [
        ("pert2", _perturb(best_seq, 0.02, rng)),
        ("pert5", _perturb(best_seq, 0.05, rng)),
        ("sym", _symmetrize(best_seq)),
    ]
    for label, a_r in restarts:
        if time.monotonic() > deadline - 25:
            break
        a_opt = _lbfgs_optimize(a_r, deadline, min_remaining=15.0)
        sc_opt = _score(a_opt)
        if time.monotonic() < deadline - 10:
            a_opt2, sc_opt2 = _soft_active_set(a_opt, min(deadline - 8, time.monotonic() + 25),
                                                lam_start=8.0, lam_end=60.0,
                                                momentum_start=0.4, momentum_end=0.7,
                                                lr_start=0.008, max_stall=150)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _soft_active_set(best_seq, deadline - 2,
                                          lam_start=30.0, lam_end=200.0,
                                          momentum_start=0.5, momentum_end=0.8,
                                          lr_start=0.003, lr_min=0.0002, max_stall=500)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_soft] t={time.monotonic()-t_start:.1f}s")

    # Final: tiny L-BFGS
    remaining = deadline - 2 - time.monotonic()
    if remaining > 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.00005
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _SmoothedProb(beta_fin, n)
        max_iter = min(3000, max(30, int(remaining / 0.05)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-17,
                                       'gtol': 1e-15, 'maxcor': 80})
            a_fin2 = _softmax(result.x)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
