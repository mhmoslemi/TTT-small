"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA with top-k subgradient on the probability simplex,
efficient coordinate descent polishing with incremental convolution updates,
multi-scale optimization (n=2048 -> 4096 -> 8192), and diverse non-palindromic initializations.
Key insight: optimal sequences should be non-palindromic with edge concentration
to flatten the self-convolution below the triangular shape of uniform sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _compute_topk_gradient(y, c, top_k):
    """Compute smoothed subgradient using top-k averaging."""
    n = len(y)
    k = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(c, -k)[-k:]
    p = np.zeros(2 * n - 1)
    p[top_idx] = 1.0 / k
    g = 2.0 * _fft_convolve(y[::-1], p)[n - 1:2 * n - 1]
    return g


def _compute_softmax_gradient(y, c, T):
    """Compute gradient using softmax smoothing of max."""
    n = len(y)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros(2 * n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    g = 2.0 * _fft_convolve(y[::-1], p)[n - 1:2 * n - 1]
    return g

    best_b = b.copy()

    lr = 1.0 / (4.0 * n)
    iter_count = 0

        iter_count += 1

        if use_softmax:

            g = _compute_softmax_gradient(y, c_y, T)
        else:

            c_y = _fft_convolve(y, y)

        y = y_new
        lr = step

        # Track best
        if func_new < c_best:
            c_best = func_new
            best_b = b.copy()

    return best_b, c_best
