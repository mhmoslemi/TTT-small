            a_mix_sum = a_mix.sum()
            if a_mix_sum > 1e-15:
                a_mix = a_mix / a_mix_sum
                c_mix = _full_convolve(a_mix, a_mix)
                sc_mix = 2.0 * n * float(c_mix.max())
                if sc_mix < best_sc * 1.001:  # Accept if close to best
                    a = a_mix

        # Limit history
        if len(X_hist) > M + 1:
            X_hist.pop(0)
        if len(Y_hist) > M + 1:
            Y_hist.pop(0)

    return best_a, _score(best_a)


def _peak_deflation(a_init, deadline, n_peaks=3):
    """Specifically target the top n_peaks of the autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.01
    stall = 0
    max_stall = 500

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0

        # Select top n_peaks locations
        top_idx = np.argsort(c)[::-1][:n_peaks]
        # Only include peaks that are significant
        threshold = c_max * 0.95
        top_idx = [i for i in top_idx if c[i] >= threshold]
        if len(top_idx) == 0:
            top_idx = [int(np.argmax(c))]

        w = np.zeros_like(c)
        for idx in top_idx:
            w[idx] = 1.0
        w = w / w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.8
            stall += 1
            if stall >= max_stall:
                break
        else:
            lr = min(0.01, lr * 1.02)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

    """Compute steepest feasible descent direction on the probability simplex."""
    # Project -grad onto the tangent space of the simplex
    # The tangent space is {d : sum(d) = 0} intersected with the normal cone at the boundary
    n = len(a)

"""
Improved step-function optimizer using extended L-BFGS beta schedule,
exponentially-weighted active-set gradient descent, dedicated peak deflation,
and multi-n exploration. Targets score below 1.5027 by focusing on
aggressive final polishing with tighter tolerances and more iterations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap * 0.4, max_c * 0.0015)

    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625,
        0.0078125, 0.003906, 0.001953, 0.000977, 0.000488,
        0.000244, 0.000122, 0.000061, 0.0000305, 0.0000152, 0.0000076
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 4096.0)
        if beta < base_beta * 0.01:
            max_iter = min(8000, max(100, int(remaining * 0.7 / (0.003 * scale))))
        elif beta < base_beta * 0.1:
            max_iter = min(5000, max(50, int(remaining * 0.5 / (0.003 * scale))))
        else:
            max_iter = min(3000, max(30, int(remaining * 0.4 / (0.003 * scale))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={current_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00001, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Exponential weighting: peaks closer to max get more weight
        if n_active > 1:
            active_vals = c[active]
            shifts = (c_max - active_vals) / (tol + 1e-30)
            w_active = np.exp(-2.0 * shifts)
            w_sum = w_active.sum()
            if w_sum > 1e-30:
                w_active = w_active / w_sum
            else:
                w_active = np.ones(n_active) / n_active
        else:
            w_active = np.array([1.0])

        w = np.zeros_like(c)
        w[active] = w_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 25 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.75)
                lr *= 0.65
            if no_improve >= 500:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 12:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _peak_deflation(a_init, deadline, n_peaks=3):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.008
    stall = 0
    max_stall = 600

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0

        top_idx = np.argsort(c)[::-1][:n_peaks]
        threshold = c_max * 0.93
        top_idx = np.array([i for i in top_idx if c[i] >= threshold])
        if len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(c))])

        w = np.zeros_like(c)
        w[top_idx] = 1.0
        w = w / w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(70):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.75
            stall += 1
            if stall >= max_stall:
                break
        else:
            lr = min(0.008, lr * 1.02)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _micro_scan(a_init, deadline, rng, n_tries=200):
    """Stochastic local search: tiny random perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for _ in range(n_tries):
        if time.monotonic() > deadline:
            break
        # Random direction in the tangent space
        d = rng.standard_normal(n)
        d = d - np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            continue
        d = d / d_norm

        # Try small steps
        for step in [0.001, 0.0005, 0.0002, 0.0001]:
            a_try = a + step * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break
            a_try = a - step * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: extended L-BFGS + exponential-weighted active-set
    + peak deflation + micro-scan + multi-n exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Extended L-BFGS ===
    print(f"\n[P1] L-BFGS extended (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs_ext(parent, deadline, min_time_remaining=700.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_ext] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish (coarse) ===
    print(f"\n[P2] Active-set polish coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        pol_deadline = min(deadline - 500, time.monotonic() + 200)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.008, tol_frac_end=0.0002,
                                           lr_init=0.04)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [polish_coarse] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak deflation (3 peaks) ===
    print(f"\n[P3] Peak deflation 3 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pd_deadline = min(deadline - 400, time.monotonic() + 150)
        a_pd, sc_pd = _peak_deflation(best_seq, pd_deadline, n_peaks=3)
        if sc_pd < best_score:
            best_seq = a_pd.copy()
            best_score = sc_pd
            print(f"  NEW BEST: {best_score:.10f} [peakdef3] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Active-set polish (fine) ===
    print(f"\n[P4] Active-set polish fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 180)
        a_pol2, sc_pol2 = _active_set_polish(best_seq, pol_deadline,
                                             tol_frac_start=0.002, tol_frac_end=0.00005,
                                             lr_init=0.015)
        if sc_pol2 < best_score:
            best_seq = a_pol2.copy()
            best_score = sc_pol2
            print(f"  NEW BEST: {best_score:.10f} [polish_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak deflation (2 peaks) ===
    print(f"\n[P5] Peak deflation 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pd_deadline = min(deadline - 200, time.monotonic() + 100)
        a_pd2, sc_pd2 = _peak_deflation(best_seq, pd_deadline, n_peaks=2)
        if sc_pd2 < best_score:
            best_seq = a_pd2.copy()
            best_score = sc_pd2
            print(f"  NEW BEST: {best_score:.10f} [peakdef2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 12288]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 40:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs_ext(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 35, time.monotonic() + 30),
                                                 tol_frac_start=0.003, tol_frac_end=0.0001,
                                                 lr_init=0.012)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("bimo", _make_profile(n_best, "bimodal")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 30:
                break
            a_opt = _run_lbfgs_ext(a_r, deadline, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 8,
                                           tol_frac_start=0.0008, tol_frac_end=0.00001,
                                           lr_init=0.005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Peak deflation final
        if time.monotonic() < deadline - 8:
            a_pd3, sc_pd3 = _peak_deflation(best_seq, deadline - 6, n_peaks=2)
            if sc_pd3 < best_score:
                best_seq = a_pd3.copy()
                best_score = sc_pd3
                print(f"  NEW BEST: {best_score:.10f} [final_peakdef] t={time.monotonic()-t_start:.1f}s")

        # Micro-scan
        if time.monotonic() < deadline - 5:
            a_ms, sc_ms = _micro_scan(best_seq, deadline - 4, rng, n_tries=300)
            if sc_ms < best_score:
                best_seq = a_ms.copy()
                best_score = sc_ms
                print(f"  NEW BEST: {best_score:.10f} [micro_scan] t={time.monotonic()-t_start:.1f}s")

        # Final tiny L-BFGS
        remaining = deadline - 4 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(4000, max(50, int(remaining / 0.025)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 80, 'maxls': 50})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One more polish after final L-BFGS
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _active_set_polish(best_seq, deadline - 1,
                                                 tol_frac_start=0.0005, tol_frac_end=0.000005,
                                                 lr_init=0.003)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [final_polish2] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
