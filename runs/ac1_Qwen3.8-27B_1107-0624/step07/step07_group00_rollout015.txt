def water_filling_step(a, S, weights, step_size):
    n = len(a)
    # Compute weighted subgradient
    g = np.zeros(n)
    for k in S:
        w = weights[k]
        # c[k] = sum_j a[j] * a[k-j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            g[j] += w * a[k - j]
    # Multiplicative update
    # We want g[j] to be constant, so multiply a[j] by (target/g[j])^step
    target = np.mean(g[g > 0])
    a_new = a * (target / np.maximum(g, 1e-15)) ** step_size
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new

        T_eff = T * c_max
        w = _softmax_weights(c, T_eff)
        g = _compute_subgradient(w, a)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_weights(z, 1.0)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_weights(z, 1.0)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _water_filling(a_init, deadline, eta_start=0.3, eta_end=0.001,
                   top_frac=0.05, min_top_k=3, max_rounds=50000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score_norm(a)
    a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    round_num = 0

    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        frac = min(1.0, round_num / max_rounds)
        eta = eta_start * (eta_end / eta_start) ** frac

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        threshold = c_max * (1.0 - top_frac)
        active = np.where(c >= threshold)[0]
        if len(active) < min_top_k:
            idx = np.argsort(c)[::-1][:min_top_k]
            active = idx

        k = len(active)
        w = np.zeros(2 * n - 1)
        for i, idx in enumerate(active):
            w[idx] = 1.0 / np.sqrt(c[idx] + 1e-30)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        g = _compute_subgradient(w, a)
        g_pos = np.maximum(g, 1e-15)
        g_max = float(g.max())
        if g_max < 1e-15:
            break

        a_new = a * np.exp(-eta * g / g_max)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score_norm(a_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.5
            if eta < eta_end * 0.1:
                break

    return best_a, best_score


def _peak_chasing_cd(a_init, deadline, max_rounds=2000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        top_k = 3
        k_indices = np.argsort(c)[::-1][:top_k]

        for k in k_indices:
            c_k = float(c[k])
            if c_k < c_max * 0.99:
                continue
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min > j_max:
                continue
            contributions = np.array([a[j] * a[k - j] for j in range(j_min, j_max + 1)])
            js = np.arange(j_min, j_max + 1)
            if len(js) == 0:
                continue
            j_best = js[np.argmax(contributions)]
            for scale in [0.7, 0.5, 0.85, 0.6, 0.3]:
                a_cand = a.copy()
                old_val = a_cand[j_best]
                a_cand[j_best] = old_val * scale
                removed = old_val - a_cand[j_best]
                if removed < 1e-15:
                    continue
                other_js = np.array([j for j in range(n) if j != j_best])
                prod_vals = np.array([a_cand[j] * a_cand[k - j] if 0 <= k - j < n else 0.0
                                      for j in other_js])
                top_others = np.argsort(prod_vals)[::-1][:5]
                for oj in top_others[::-1]:
                    a_cand[oj] += removed * 0.1
                a_cand = np.maximum(a_cand, 0.0)
                s_cand = a_cand.sum()
                if s_cand < 1e-15:
                    continue
                a_cand = a_cand / s_cand
                c_cand = _fft_convolve(a_cand, a_cand)
                sc_cand = float(2.0 * n * c_cand.max())
                if sc_cand < best_score * (1.0 - 1e-10):
                    best_score = sc_cand
                    best_a = a_cand.copy()
                    a = a_cand
                    c = c_cand
                    c_max = float(c.max())
                    improved = True
                    break
            if improved:
                break

        if not improved:
            c = _fft_convolve(a, a)
            c_max = float(c.max())
            if c_max * 2.0 * n < best_score * (1.0 + 1e-8):
                best_score = c_max * 2.0 * n
                best_a = a.copy()
            break

    return best_a, best_score


def _fast_cd(a_init, deadline, max_rounds=5000, top_k_peaks=5):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    tol = c_max * 1.001

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        active_peaks = np.where(c > tol)[0]
        if len(active_peaks) == 0:
            break
        improved = False
        for k in active_peaks[:top_k_peaks]:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min > j_max:
                continue
            for j in range(j_min, j_max + 1):
                for scale in [1.2, 1.5, 0.8, 0.6, 0.4]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    a_cand = np.maximum(a_cand, 0.0)
                    s_cand = a_cand.sum()
                    if s_cand < 1e-15:
                        continue
                    a_cand = a_cand / s_cand
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(2.0 * n * c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.001
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        a = parent / parent.sum()
        print(f"Loaded parent: n={len(a)}, score={_score(a):.10f}")
    except Exception:
        a = np.ones(8192) / 8192
        print("Using uniform initialization, n=8192")

    best_a = a.copy()
    best_score = _score(a)
    n_best = len(best_a)
    print(f"Initial: n={n_best}, score={best_score:.10f}")

    # Phase 1: Adam log-space on parent (0 - 350s)
    phase1_end = min(deadline, t_start + 350.0)
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    a_opt, sc = _adam_logspace(a, phase1_end,
                               T_start=0.004 * max_c, T_end=0.00002 * max_c,
                               lr_start=0.05, lr_end=0.002,
                               max_iter=400000, eval_interval=150, clip_norm=0.4)
    full_sc = _score(a_opt)
    print(f"Phase 1 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 1b: Second Adam pass with different params (350s - 500s)
    phase1b_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase1b_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())

        print(f"Phase 1b Adam2: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 2: Water-filling (500s - 1000s)
    phase2_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_end:
        a_wf, sc_wf = _water_filling(best_a, phase2_end,
                                     eta_start=0.2, eta_end=0.0005,
                                     top_frac=0.02, min_top_k=3,

    # Phase 3: Peak-chasing CD (1000s - 1300s)
