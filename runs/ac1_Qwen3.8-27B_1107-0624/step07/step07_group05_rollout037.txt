import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gaussian_mixture_init(n, n_gaussians=8, seed=42):
    """Initialize using a Gaussian mixture for a good starting point."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    for m in range(n_gaussians):
        mu = rng.uniform(n * 0.1, n * 0.9)
        sigma = rng.uniform(n * 0.05, n * 0.3)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _fourier_perturb(a, rng, n_modes=8, amplitude=0.02):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

    print(f"\n[Phase 2] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        pe_deadline = min(deadline - 350, time.monotonic() + 120)
        a_pe, sc_pe = _pressure_equalize(best_seq, pe_deadline, n_iters=400, gamma=0.25)
        _update(a_pe, sc_pe)
        if best_score < par_sc - 0.001:
            # Follow up with L-BFGS
            a_follow = _run_lbfgs(best_seq, deadline, min_time_remaining=350)
            _update(a_follow, _score(a_follow))

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 60:
            break

        # Initialize by resampling best or using profile
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()
        else:
            a_init = _gaussian_mixture_init(n_try, n_gaussians=10, seed=rng.integers(10000))

        min_rem = max(30, remaining_for_this * 0.25)
        a_opt, sc_opt = _multi_stage_optimize(
            a_init, deadline, min_remaining=min_rem, label=f"n_{n_try}")
        _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts from best ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)

        for i in range(4):
            if time.monotonic() > deadline - 40:
                break
            # Different perturbation strategies
            if i == 0:
                a_pert = _fourier_perturb(best_seq, rng, n_modes=6, amplitude=0.015)
            elif i == 1:
                # Blend with tapered
                a_blend = 0.7 * best_seq + 0.3 * _make_tapered(n_best, power=0.3)
                a_blend = a_blend / a_blend.sum()
                a_pert = _fourier_perturb(a_blend, rng, n_modes=4, amplitude=0.01)
            elif i == 2:
                # Gaussian mixture perturbation
                a_gm = _gaussian_mixture_init(n_best, n_gaussians=12, seed=rng.integers(10000))
                a_blend = 0.6 * best_seq + 0.4 * a_gm
                a_blend = np.maximum(a_blend, 1e-10)
                a_pert = a_blend / a_blend.sum()
            else:
                # Small random perturbation in log domain
                z = np.log(np.maximum(best_seq, 1e-30))
                z += 0.005 * rng.standard_normal(len(z))
                a_pert = _softmax(z)

            a_r, sc_r = _multi_stage_optimize(
                a_pert, deadline, min_remaining=40, label=f"restart_{i}")
            _update(a_r, sc_r)

    # --- Phase 5: Ultra-fine final polish ---
    print(f"\n[Phase 5] Ultra-fine final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Final L-BFGS with smallest beta
        n = len(best_seq)
        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(c_cur.max())
        tiny_beta = max_c * 0.001

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        if time.monotonic() < deadline - 5:
            try:
                max_iter = int((deadline - 5 - time.monotonic()) / 0.005)
                max_iter = min(max_iter, 20000)
                max_iter = max(max_iter, 100)
                result = minimize(
                    _smoobj, z, args=(tiny_beta, n),
                    jac=_smograd,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 100, 'maxls': 300}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                _update(a_fin, sc_fin)
            except Exception as ex:
                print(f"  Final LBFGS error: {ex}")

        # Final subgradient micro-polish
        if time.monotonic() < deadline - 3:
            a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.003)
            _update(a_sg, sc_sg)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(256)
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal_score={sc_final:.10f}, eval_score={eval_sc:.10f}, n={len(final)}")
            print(f"Time used: {time.monotonic()-t_start:.1f}s / {budget_s}s")
            if np.isfinite(eval_sc) and eval_sc < sc_final:
                print(f"  (eval score is lower, using it as reference)")
        except Exception as ex:
            print(f"  eval check error: {ex}")
        print(f"  n={len(final)}, sum={s:.6f}")
        return [float(x) for x in final]

    fallback = _make_tapered(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
