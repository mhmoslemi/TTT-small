    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _H_matvec(v, A_ind, n):
    """(H*v)[i] = sum_j A_ind[i+j]*v[j], computed via FFT."""
    rev_A = A_ind[::-1]
    conv = _full_convolve(v, rev_A)
    return conv[n - 1:2 * n - 1][::-1]


def _newton_kkt_polish(a_init, deadline, max_rounds=100, active_tol=1e-4):
    """
    Newton-style polish using KKT conditions.
    At optimum: (H*a)[i] = C for all i in support, where H depends on active set.
    Uses iterative linear solve with FFT-based mat-vec.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = active_tol * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        A_ind = np.zeros(2 * n - 1)
        A_ind[active] = 1.0

        # Compute h = H*a
        h = _H_matvec(a, A_ind, n)
        C_val = float(np.mean(h))
        
        # Residual: how far is h from being constant
        resid = h - C_val
        resid_norm = float(np.linalg.norm(resid))
        if resid_norm < 1e-14 * n * C_val:
            break

        # Newton step: solve H * delta = -resid + lambda * 1
        # Using CG on the normal equations (H is symmetric)
        # Actually, use a simpler approach: damped Newton via CG
        x = np.zeros(n)
        r = -resid.copy()
        p = r.copy()
        rs = float(r @ r)
        
        # Also need 1^T * delta = 0 (to preserve normalization)
        # Project r and p to be orthogonal to 1
        r = r - np.mean(r)
        p = p - np.mean(p)
        rs = float(r @ r)
        if rs < 1e-30:
            break

        for cg_iter in range(200):
            Hp = _H_matvec(p, A_ind, n)
            Hp = Hp - np.mean(Hp)  # Project
            pHp = float(p @ Hp)
            if abs(pHp) < 1e-30:
                break
            alpha_cg = rs / pHp
            x_new = x + alpha_cg * p
            r_new = r - alpha_cg * Hp
            r_new = r_new - np.mean(r_new)
            rs_new = float(r_new @ r_new)
            if not np.isfinite(rs_new) or rs_new > rs * 1.5:
                break
            x = x_new
            r = r_new
            p = (rs_new / rs) * p + r
            rs = rs_new
            if rs < 1e-24 * float(r @ r) or rs < 1e-30:
                break

        # Line search
        a_try = a + x
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            # Fallback to gradient step
            x = -resid + np.mean(resid)
            x = x - np.mean(x)
            a_try = a + x
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
        a_try = a_try / s_try

        c_try = _full_convolve(a_try, a_try)
        c_max_try = float(c_try.max())
        if c_max_try < c_max - 1e-15:
            a = a_try
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1, 0.05]:
                a_try = a + frac * x
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    sc = _score(a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = a.copy()
                    break

    return best_a, best_sc


def _lbfgs_optimize(z_init, n, deadline, min_time_remaining=30.0):
    """
    L-BFGS with aggressive beta continuation.
    Much more iterations at small beta values.
    """
    current_a = _softmax(z_init)
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)

    # Very aggressive schedule: 30 steps
    ratios = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015, 0.0075, 0.004, 0.002, 0.001,
        0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003, 0.000015, 0.0000075, 0.000004,
        0.000002, 0.000001, 0.0000005, 0.0000002,
        0.0000001, 0.00000005, 0.00000002,
        0.00000001, 0.000000005, 0.000000002, 0.000000001
    ])
    betas = base_beta * ratios

    current_sc = _score(current_a)
    z = z_init.copy()
    z = z - z.mean()

    # Estimate time per iteration
    t0 = time.monotonic()
    _smograd(z, betas[0], n)
    t1 = time.monotonic()
    time_per_iter = max(0.001, t1 - t0)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max iterations based on remaining time and beta index
        # More iterations at small beta (later stages)
        stage_weight = 1.0 + 2.0 * (i / len(betas))
        max_iter = min(5000, max(50, int(remaining / (time_per_iter * 20) * stage_weight)))

        try:
            result = minimize(
                _smoobj_wrapper,
                z,
                args=(beta, n),
                jac=_smograd_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                # Update time estimate
                time_per_iter = max(0.0005, time_per_iter * 0.99)
            print(f"    beta={beta:.2e} iter={result.niter} sc={current_sc:.10f} t={time.monotonic()-t0+time.monotonic():.0f}s")
        except Exception as e:
            print(f"    beta={beta:.2e} failed: {e}")

    return current_a


def _smoobj_wrapper(z, beta, n):
    return _smoobj(z, beta, n)


def _smograd_wrapper(z, beta, n):
    return _smograd(z, beta, n)


def _support_optimize(a, deadline, min_time_remaining=60.0):
    """
    Try zeroing out small elements and re-optimizing.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    fractions = [0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15, 0.20, 0.30]
    a_sorted_idx = np.argsort(a)

    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            print(f"    support shrink {frac*100:.0f}%: score={best_sc:.10f}")
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
    elif kind == "bimodal":
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
    elif kind == "super_tapered":
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
    elif kind == "flat_top":
        a = np.ones(n)
        w = int(n * 0.3)
        a[:w] = np.linspace(0.3, 1.0, w)
        a[-w:] = np.linspace(1.0, 0.3, w)
    else:
        a = np.ones(n)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
