"""
Aggressive active-set Newton optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses KKT-based active set refinement, variable elimination, and multi-scale approach.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*sum_k W_k * delta_{j+m,k}"""
    # H*v = 2 * [sum_m W_{j+m} * v_m] for j = 0,...,n-1
    # This is a convolution-like operation
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs using CG."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    fb = np.fft.rfft(b, N)

        r = r_new

        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            ones = np.ones(n)
            lam = float(np.dot(ones, Hg)) / float(np.dot(ones, H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Damped Newton with very aggressive backtracking
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Accept if we reduce the max or if gradient alignment improves
            if c_max_try < c_max - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.7
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback
            alpha = 0.1
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _kkt_polish(a_init, deadline, max_iters=300):
    """KKT-based active set refinement: verify and tighten the active set using dual conditions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    support = a > 1e-10 * a.max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Very tight active set for KKT verification
        tol = 1e-8 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Uniform weights for KKT check
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()

        g = _hankel_matvec(a, W, n)

        # KKT: on support, g should be constant
        g_supp = g[support]
        g_mean = float(np.mean(g_supp))
        g_resid = float(np.max(np.abs(g_supp - g_mean)))

        # Off-support: g should be >= g_mean (for minimization with a>=0)
        if (~support).any():
            g_off = g[~support]
            g_off_violation = float(np.max(g_mean - g_off)) if len(g_off) > 0 else 0.0
        else:
            g_off_violation = 0.0

        if g_resid < 1e-12 * np.mean(np.abs(g_supp)) and g_off_violation < 1e-10:
            # KKT satisfied
            break

        # If KKT not satisfied, do a Newton step
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20:
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.dot(np.ones(n), Hg)) / float(np.dot(np.ones(n), H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)

        # Backtracking
        alpha = 1.0
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-15:
                a = a_try
                support = a > 1e-10 * a.max()
                break
            alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _waterfilling_polish(a_init, deadline, max_rounds=100):
    """Water-filling: iteratively equalize the active constraints."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_min_active = float(c.min())
        tol = 1e-6 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Compute the "excess" on active set vs inactive
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-13 * n:
            break

        # Newton step with extra regularization for stability
        reg = 1e-10
        Hg = _cg_solve(W, n, g + reg * g_t, max_iter=600)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=600)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20:
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.dot(np.ones(n), Hg)) / float(np.dot(np.ones(n), H1))
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)

        alpha = 1.0
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max - 1e-15:
                a = a_try
                break
            alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_reopt = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: aggressive L-BFGS + KKT-verified Newton + water-filling polish
    + multi-n exploration (up to 16384) + shrinkage + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Aggressive L-BFGS from parent (most time)
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=700.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: KKT-verified Newton polish
    print(f"\n[Phase 2] KKT Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _kkt_polish(best_seq, deadline - 500, max_iters=200)
        update_best(a2, sc2, "kkt_1")

    # Phase 3: Water-filling polish
    print(f"\n[Phase 3] Water-filling (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _waterfilling_polish(best_seq, deadline - 400, max_rounds=150)
        update_best(a3, sc3, "waterfill_1")

    # Phase 4: Progressive Newton (tightest)
    print(f"\n[Phase 4] Progressive Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a4, sc4 = _newton_active_polish(best_seq, deadline - 350, max_rounds=200)
        update_best(a4, sc4, "prog_newton")

    # Phase 5: n=16384 exploration
    print(f"\n[Phase 5] n=16384 exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a_16k = _resample(parent, 16384)
        a_16k_opt = _run_lbfgs(a_16k, deadline, min_time_remaining=200.0)
        sc_16k = _score(a_16k_opt)
        update_best(a_16k_opt, sc_16k, "n_16384")
        if sc_16k < best_score + 0.005 and time.monotonic() < deadline - 150:
            a_16k_np, sc_16k_np = _kkt_polish(a_16k_opt, deadline - 150, max_iters=100)
            update_best(a_16k_np, sc_16k_np, "n_16384_kkt")

    # Phase 6: n=4096 exploration
    print(f"\n[Phase 6] n=4096 exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_4k = _resample(parent, 4096)
        a_4k_opt = _run_lbfgs(a_4k, deadline, min_time_remaining=100.0)
        sc_4k = _score(a_4k_opt)
        update_best(a_4k_opt, sc_4k, "n_4096")
        if sc_4k < best_score + 0.005 and time.monotonic() < deadline - 80:
            a_4k_np, sc_4k_np = _kkt_polish(a_4k_opt, deadline - 80, max_iters=80)
            update_best(a_4k_np, sc_4k_np, "n_4096_kkt")

    # Phase 7: Shrinkage on best
    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a5, sc5 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=60.0)
        update_best(a5, sc5, "shrinkage")

    # Phase 8: Diverse restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restarts = [
            ("perturb_small", _perturb_safe(best_seq, 0.02, rng)),
            ("perturb_med", _perturb_safe(best_seq, 0.08, rng)),
            ("bimodal", _make_profile(n_best, "bimodal")),
        ]
        for i, (label, p_start) in enumerate(restarts):
            if time.monotonic() > deadline - 25:
                break
            remaining_for_this = max(10, deadline - 20 - time.monotonic() - i * 5)
            a_r = _run_lbfgs(p_start, t_start + remaining_for_this + time.monotonic() - (deadline - remaining_for_this - time.monotonic()), min_time_remaining=15.0)
            # Simpler: just give it until deadline - 15
            a_r = _run_lbfgs(p_start, deadline - 15.0 + time.monotonic(), min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")
            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 10:
                a_rp, sc_rp = _kkt_polish(a_r, deadline - 5, max_iters=50)
                update_best(a_rp, sc_rp, f"restart_{label}_kkt")

    # Phase 9: Final ultra-tight polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _kkt_polish(best_seq, deadline - 2, max_iters=200)
        update_best(a_fin, sc_fin, "final_kkt")
        if time.monotonic() < deadline - 2:
            a_wf, sc_wf = _waterfilling_polish(best_seq, deadline - 1, max_rounds=100)
            update_best(a_wf, sc_wf, "final_wf")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]


def _perturb_safe(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
