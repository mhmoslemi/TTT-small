"""
Optimized step function search using asymmetric profiles, aggressive L-BFGS
with log-sum-exp smoothing, Newton active-set refinement, and local search.
Key insight: asymmetric sequences can achieve autocorrelation peaks below the
Cauchy-Schwarz bound for symmetric sequences, enabling scores < 2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.001 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break
        mu = 1e-8
        def _hv(v):
            return _hessian_matvec(v, active, n) + mu * v
        dx = _cg_solve(_hv, -grad_t, n, max_iter=200, tol=1e-10)
        dx = dx - np.mean(dx)
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-10:
                break
            a_new = a
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.arange(1, n + 1, dtype=np.float64)
        a = t / t.sum()
        return a
    elif kind == "ramp2":
        t = np.arange(1, n + 1, dtype=np.float64)
        a = t ** 2
        return a / a.sum()
    elif kind == "ramp_half":
        t = np.arange(1, n + 1, dtype=np.float64)
        a = np.minimum(t, n - t + 1) ** 0.3
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-0.8 * t / n)
        return a / a.sum()
    elif kind == "exp_decay_fast":
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-2.0 * t / n)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "truncated_gauss":
        t = np.linspace(0, 1, n)
        center = 0.4
        a = np.exp(-((t - center) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power3":
        t = np.arange(1, n + 1, dtype=np.float64)
        a = t ** 3
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + 0.5 * np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        est_time_per_iter = 0.002 * max(1, n / 2048.0)
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 6))))
        def _fun_local(z_local):
            return _smoobj(z_local, beta, n)
        def _jac_local(z_local):
            return _smograd(z_local, beta, n)
        try:
            result = minimize(
                _fun_local, z, jac=_jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.001)
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    for frac in [0.01, 0.02, 0.05, 0.10, 0.15, 0.20]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc

    print(f"\n[Phase 2] Deep L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        dl = min(deadline - 150, time.monotonic() + 120)
        a_deep = _run_lbfgs_aggressive(best_seq, dl, min_time_remaining=100)
        _update_best(a_deep, _score(a_deep), "deep_lbfgs")

    # === Phase 3: Newton active-set ===
    print(f"\n[Phase 3] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        dl = min(deadline - 100, time.monotonic() + 200)
        a_newton, sc_newton = _newton_active_set(best_seq, dl, max_iter=300)
        _update_best(a_newton, sc_newton, "newton")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        dl = min(deadline - 80, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(best_seq, dl, n_max_steps=3000, lr_init=0.012)
        _update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 5: Shrinkage + re-optimize ===
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 60)
        _update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score * 1.005 and time.monotonic() < deadline - 60:
            a_shr2 = _run_lbfgs_aggressive(a_shr, deadline, min_time_remaining=40)
            _update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # === Phase 6: Try different n with best shape ===
    print(f"\n[Phase 6] Multi-n refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        n_tries = [n_best // 2, n_best, n_best * 2, n_best + 1024, n_best - 1024]
        n_tries = [x for x in n_tries if 1024 <= x <= 32768]
        n_tries = sorted(set(n_tries))
        for n_try in n_tries:
            if time.monotonic() > deadline - 40:
                break
            if n_try == n_best:
                a_init = best_seq.copy()
            else:
                x_old = np.linspace(0, 1, n_best)
                x_new = np.linspace(0, 1, n_try)
                a_init = np.interp(x_new, x_old, best_seq)
                a_init = np.maximum(a_init, 0)
                a_init = a_init / a_init.sum()
            dl = min(deadline - 30, time.monotonic() + 60)
            a_opt = _run_lbfgs_aggressive(a_init, dl, min_time_remaining=20)
            sc_opt = _score(a_opt)
            _update_best(a_opt, sc_opt, f"n_{n_try}_refine")
            if sc_opt < best_score * 1.003 and time.monotonic() < deadline - 25:
                a_opt2, sc_opt2 = _newton_active_set(a_opt, min(deadline - 20, time.monotonic() + 40), max_iter=80)
                _update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(5):
            if time.monotonic() > deadline - 15:
                break
            if i == 0:
                p_start = _perturb_log(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3)
            elif i == 2:
                p_start = _perturb_log(best_seq, 0.05, rng)
            elif i == 3:
                p_start = _perturb_fourier(best_seq, rng, n_modes=8)
            else:
                p_start = _perturb_log(_perturb_fourier(best_seq, rng, n_modes=2), 0.01, rng)
            dl = min(deadline - 10, time.monotonic() + 30)
            a_r = _run_lbfgs_aggressive(p_start, dl, min_time_remaining=8)
            sc_r = _score(a_r)
            _update_best(a_r, sc_r, f"restart_{i}")

    # === Phase 8: Final combined polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Local search
        dl_ls = min(deadline - 2, time.monotonic() + 60)
        a_ls, sc_ls = _local_search(best_seq, dl_ls, n_trials=500, rng=rng)
        _update_best(a_ls, sc_ls, "local_search")

        # Newton on result
        if time.monotonic() < deadline - 2:
            a_fin, sc_fin = _newton_active_set(best_seq, max(deadline - 1, time.monotonic()), max_iter=80)
            _update_best(a_fin, sc_fin, "final_newton")

        # Ultra-low beta L-BFGS
        if time.monotonic() < deadline - 1:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.000005
            remaining = deadline - 1 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(100, int(remaining / 0.005)))
                def _fun_final(z_local):
                    return _smoobj(z_local, beta_final, n)
                def _jac_final(z_local):
                    return _smograd(z_local, beta_final, n)
                try:
                    result = minimize(
                        _fun_final, z, jac=_jac_final, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100, 'maxls': 200}
                    )
                    a_fin3 = _softmax(result.x)
                    _update_best(a_fin3, _score(a_fin3), "ultra_lbfgs")
                except Exception:
                    pass

        # Final local search
        if time.monotonic() < deadline - 0.5:
            a_ls2, sc_ls2 = _local_search(best_seq, deadline - 0.5, n_trials=200, rng=rng)
            _update_best(a_ls2, sc_ls2, "final_local_search")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence confirms: {sc_eval:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
