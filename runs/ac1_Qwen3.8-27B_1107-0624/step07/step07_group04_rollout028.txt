import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, w, n):
    """Compute B_w * v where (B_w)_{i,j} = w_{i+j}, w has length 2n-1."""
    v_flip = v[::-1]
    N = _next_pow2(n + len(w) - 1)
    Fv = np.fft.rfft(v_flip, N)
    Fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(Fv * Fw, N)[:n + len(w) - 1]
    return conv[n - 1:2 * n - 1]


def _cg_hankel(w, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    """Solve (B_w + reg*I) * x = rhs using conjugate gradients."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, w, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            if reg < 1e-8 * (rs0 / max(n, 1)):
                reg = 1e-8 * (rs0 / max(n, 1))
                Hp = _hankel_matvec(p, w, n) + reg * p
                pHp = float(p @ Hp)
            if abs(pHp) < 1e-30 or not np.isfinite(pHp):
                break
        alpha = rs / pHp
        if not np.isfinite(alpha) or abs(alpha) < 1e-30:
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _inner_qp(w, n, max_active=50, max_cg=300):
    """
    Solve: min a^T B_w a / 2 s.t. sum(a)=1, a>=0.
    Uses CG on the full system with clamping.
    """
    rhs = np.ones(n)
    x = _cg_hankel(w, n, rhs, max_iter=max_cg, reg=1e-12)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _structured_grad(a, w, c, n):
    """
    Compute gradient of 2n * (c_max + beta*log(sum exp((c-c_max)/beta))) w.r.t. a.
    Returns 4*n*a*(B_w*a - w·c).
    """
    h = _hankel_matvec(a, w, n)
    wc = float(np.dot(w, c))
    return 4.0 * n * a * (h - wc)

                                     0.0001, 0.00005, 0.00002, 0.00001]]

def _newton_polish(a_init, deadline, max_rounds=200):
    """
    Active-set Newton method for the final polish.
    Solves the KKT system: B_w * dx = -grad + lambda*1 on the active set.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with adaptive tolerance
        tol = max(1e-6 * c_max, 1e-12)
        active_c = c >= (c_max - tol)
        n_active = int(active_c.sum())
        if n_active == 0:
            active_c[int(np.argmax(c))] = True
            n_active = 1

        # Build weight vector
        w = np.zeros(2 * n - 1)
        w[active_c] = 1.0 / n_active

        # Compute gradient of the QP objective: grad_i = 2*(B_w*a)_i
        h = _hankel_matvec(a, w, n)
        grad = 2.0 * h
        # Project gradient for the sum constraint
        grad_t = grad - np.mean(grad)

        gn = float(np.linalg.norm(grad_t))
        if gn < 1e-14 * n:
            break

        # Newton step: solve B_w * dx = -grad_t (with sum(dx)=0 constraint)
        # Actually solve: B_w * dx = -grad + lambda*1
        # where 1^T dx = 0

        # Use CG to solve B_w * x1 = grad and B_w * x2 = 1
        x1 = _cg_hankel(w, n, grad, max_iter=300, reg=1e-10)
        x2 = _cg_hankel(w, n, np.ones(n), max_iter=300, reg=1e-10)

        # lambda such that 1^T dx = 0: dx = -x1 + lambda*x2, 1^T dx = 0
        # -1^T x1 + lambda * 1^T x2 = 0
        sum_x1 = float(np.sum(x1))
        sum_x2 = float(np.sum(x2))
        if abs(sum_x2) > 1e-20:
            lam = sum_x1 / sum_x2
        else:
            lam = 0.0

        dx = -x1 + lam * x2
        dx = dx - np.mean(dx)  # ensure sum-zero

        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-30 or not np.isfinite(dx_norm):
            # Fallback: gradient step
            dx = -grad_t / (gn + 1e-30)
            dx = dx - np.mean(dx)

        # Damped Newton step with backtracking
        alpha = 1.0
        step_accepted = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())

            # Armijo condition: sufficient decrease in c_max
            if c_max_try <= c_max - 0.01 * alpha * gn * 0.1:
                a = a_try
                step_accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_accepted:
            # Try pure gradient descent step
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * grad_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check if we're at a good local minimum
        if sc < best_sc + 1e-15 and gn / n < 1e-10:
            break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, max_steps=20000):
    """
    Mirror descent (exponential gradient) with structured gradient.
    Uses softmax parameterization.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0

    # Adaptive learning rate
    lr_init = 0.05
    lr = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Weight: uniform on near-active set
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()

        # Structured gradient
        h = _hankel_matvec(a, w, n)
        wc = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - wc)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Step
        step = lr / (1.0 + 0.0001 * i)
        # In z-space: dz = -step * grad / (4*n*a) ... but we just do z -= step * grad_normalized
        # Actually, the gradient w.r.t. z is: 4*n*a*(h-wc) * (I - a*a^T) approximately
        # Simpler: just do z -= step * grad / (a + eps) which gives the "natural gradient" step
        # But that can be unstable. Let's just do z -= step * grad / (n * grad_norm + 1)

        grad_z = grad / (np.sqrt(n) * grad_norm + 1e-10)
        z = z - step * grad_z
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Check every N steps
        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.1)
            else:
                patience += 1
                if patience % 5 == 0:
                    lr *= 0.85
                if patience > 100:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.2) ** 2) + np.exp(-6.0 * (t - 0.8) ** 2) + 0.1
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "zigzag":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.sin(10 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s
