    # Second term: k-i=j, i.e., i=k-j, gives a[k-j] (valid if 0<=k-j<n)
    # 
    # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (if 0<=k-j<n)
    # 
    # Check: k=1, j=0: 2*a[1] ✓
    # Check: k=1, j=1: 2*a[0] ✓
    # 
    # Great! So the formula is: dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    # 
    # But wait, what about k=0? c[0] = a[0]*a[0] = a[0]^2
    # dc[0]/da[0] = 2*a[0]
    # Formula: k=0, j=0: 2*a[0] (since k-j=0, 0<=0<n ✓) ✓
    # 
    # And k=2n-2: c[2n-2] = a[n-1]*a[n-1] = a[n-1]^2
    # dc[2n-2]/da[n-1] = 2*a[n-1]
    # Formula: k=2n-2, j=n-1: k-j = n-1, 0<=n-1<n ✓, so 2*a[n-1] ✓
    
    # So the formula is simply: dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    # This can be computed efficiently using convolution!
    # 
    # grad[j] = sum_{k in active} 2*a[k-j] * I(0<=k-j<n)
    # Let b = reversed a, so b[i] = a[n-1-i]
    # Then a[k-j] = b[n-1-(k-j)] = b[n-1-k+j]
    # 
    # Actually, let's think of it differently.
    # For a fixed j, the valid k values are those with 0 <= k-j < n, i.e., j <= k < j+n.
    # So: grad[j] = 2 * sum_{k in active, j <= k < j+n} a[k-j]
    # Let m = k-j, then: grad[j] = 2 * sum_{m: 0<=m<n, j+m in active} a[m]
    # 
    # This is like a cross-correlation of the indicator of "active shifted by j" with a.
    # 
    # Efficient computation: 
    # Let w[k] = 1 if k in active_idx, else 0 (extended to length 2n-1)
    # Then grad[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    # = 2 * (w * a_rev)[j + n - 1] ... let me think about this with FFT.
    
    # Actually, the cleanest way:
    # grad = 2 * (convolve(w, a))[n-1 : 2n-1] ... no wait.
    
    # Let me define: for each j, grad[j] = 2 * sum_{k in active} a[k-j] * I(0 <= k-j < n)
    # 
    # Let's define a vector w of length 2n-1 where w[k] = 1 if k in active_idx.
    # Then grad[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
    # 
    # The sum over k with w[k]=1 and 0<=k-j<n: let m = k-j, then m ranges over [0,n) and k = j+m.
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # 
    # This is the cross-correlation of w (padded to length 2n-1) with a, evaluated at position j.
    # Specifically: (w ⊛ a)[j] = sum_m w[j+m] * a[m] ... hmm, this is a specific correlation.
    # 
    # Alternatively: grad[j] = 2 * sum_m a[m] * w[j+m]
    # 
    # If we define W = FFT(w) (padded to appropriate length), A = FFT(a) (same length),
    # then this is a "cross-correlation" which can be computed as:
    # ifft(conj(A) * W) ... let me think.
    # 
    # Actually, the cross-correlation (f ⊛ g)[j] = sum_m f[m] * conj(g[m-j])
    # In our case: grad[j]/2 = sum_m a[m] * w[j+m]
    # Let's define h[k] = w[k] (length 2n-1) and a (length n)
    # sum_m a[m] * w[j+m] = sum_m a[m] * h[j+m]
    # This is the "anti-correlation" or cross-correlation: (a ⊛ h)[-j] ... 
    # 
    # Let me just use the direct formula with FFT:
    # Define a padded version: A_pad of length N (power of 2, >= 2n-2+n = 3n-2... actually we need the convolution of a (length n) with w (length 2n-1) which has length n+2n-2 = 3n-2)
    # 
    # conv(a, w) has length n + (2n-1) - 1 = 3n-2
    # conv(a, w)[j] = sum_i a[i] * w[j-i] = sum_i a[i] * w[j-i]
    # We want sum_m a[m] * w[j+m] = sum_m a[m] * w[(j+m)]
    # Let i = m, then sum_i a[i] * w[j+i]
    # This is conv(a, w_reversed)[j + (2n-2)] where w_reversed = w[::-1]
    # Because conv(a, w_rev)[k] = sum_i a[i] * w_rev[k-i] = sum_i a[i] * w[(2n-2)-(k-i)] = sum_i a[i] * w[2n-2-k+i]
    # Setting k = 2n-2-j: sum_i a[i] * w[j+i] ✓
    
    # So: grad[j] = 2 * conv(a, w_rev)[2n-2+j] for j = 0,...,n-1
    # Or equivalently, if we compute the full conv(a, w_rev), we take elements [2n-2 : 2n-2+n]
    
    # Let me verify with a small example:
    # a = [1, 2, 3], n=3
    # active = {1}, so w = [0, 1, 0, 0, 0] (length 5 = 2*3-1)
    # w_rev = [0, 0, 0, 1, 0]
    # conv(a, w_rev) = [0, 0, 0, 1, 2, 3, 0, 0, 0] (length 3+5-1=7)
    # We want elements [2n-2 : 2n-2+n] = [4:7] = [3, 0, 0]
    # grad = 2 * [3, 0, 0] = [6, 0, 0]
    # 
    # Check manually: c = conv(a,a) = [1, 4, 10, 12, 9, 12, 9]
    # dc[1]/da[0] = 2*a[1] = 4
    # dc[1]/da[1] = 2*a[0] = 2  
    # dc[1]/da[2] = 0 (since k-j = 1-2 = -1 < 0)
    # So grad should be [4, 2, 0], not [6, 0, 0]!
    
    # I made an error. Let me redo.
    # w = [0, 1, 0, 0, 0] (length 5)
    # We want: grad[j] = 2 * sum_m a[m] * w[j+m] (where w is defined for indices 0..4)
    # j=0: 2*(a[0]*w[0] + a[1]*w[1] + a[2]*w[2]) = 2*(1*0 + 2*1 + 3*0) = 4 ✓
    # j=1: 2*(a[0]*w[1] + a[1]*w[2] + a[2]*w[3]) = 2*(1*1 + 2*0 + 3*0) = 2 ✓
    # j=2: 2*(a[0]*w[2] + a[1]*w[3] + a[2]*w[4]) = 2*(1*0 + 2*0 + 3*0) = 0 ✓
    # 
    # Now using conv(a, w_rev):
    # a = [1, 2, 3], w_rev = w[::-1] = [0, 0, 0, 1, 0]
    # conv(a, w_rev):
    # [0] = a[0]*w_rev[0] = 0
    # [1] = a[0]*w_rev[1] + a[1]*w_rev[0] = 0
    # [2] = a[0]*w_rev[2] + a[1]*w_rev[1] + a[2]*w_rev[0] = 0
    # [3] = a[0]*w_rev[3] + a[1]*w_rev[2] + a[2]*w_rev[1] = 1
    # [4] = a[0]*w_rev[4] + a[1]*w_rev[3] + a[2]*w_rev[2] = 0 + 2 + 0 = 2
    # [5] = a[1]*w_rev[4] + a[2]*w_rev[3] = 0 + 3 = 3
    # [6] = a[2]*w_rev[4] = 0
    # 
    # So conv(a, w_rev) = [0, 0, 0, 1, 2, 3, 0]
    # 
    # We want grad[j] = 2 * sum_m a[m] * w[j+m]
    # For j=0: sum_m a[m]*w[m] = a[0]*w[0]+a[1]*w[1]+a[2]*w[2] = 0+2+0 = 2
    # conv(a, w_rev)[?] ... Let me find which index gives 2.
    # conv(a, w_rev)[3] = 1, [4] = 2, [5] = 3
    # 
    # Hmm, it seems like conv(a, w_rev)[3+j] gives the answer for j=0,1,2:
    # [3] = 1 = sum a[m]*w[m+0]... wait that's not right either.
    # 
    # Let me try a different indexing. 
    # sum_m a[m]*w[j+m]: this is the cross-correlation of a and w at lag j.
    # In numpy terms: np.correlate(w, a, mode='full') ... let me think.
    # np.correlate(a, w, mode='full')[k] = sum_n a[n] * w[n+k - (len(w)-1)] ... this is confusing.
    # 
    # Let me just use the FFT approach directly.
    # sum_m a[m]*w[j+m] where a has length n, w has length 2n-1, j in [0,n-1], m in [0,n-1]
    # = sum_m a[m] * w[j+m]
    # 
    # Pad a to length N (>= 2n-1+n-1 = 3n-2, say N = next_pow2(3n)):
    # A = fft(a_padded, N)
    # W = fft(w_padded, N)  (w_padded has length 2n-1, padded to N)
    # 
    # The cross-correlation sum_m a[m]*w[j+m] can be computed as:
    # ifft(conj(A) * W)[j + (N - n + 1)] ... I'm getting confused with the indexing.
    
    # Let me just use a simple approach: since we're doing this for a fixed active set,
    # and the active set is small (top k), we can just loop over the active indices.
    
    for k in active_idx:
        # dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n
        # So for each valid j: k-j in [0, n) means j in [k-n+1, k] intersect [0, n-1]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # For j in [j_start, j_end]: grad[j] += 2*a[k-j]
            # k-j ranges from k-j_start to k-j_end
            m_start = k - j_end  # = k - min(n-1, k)
            m_end = k - j_start  # = k - max(0, k-n+1)
            # grad[j_start : j_end+1] += 2 * a[m_end : m_start-1 : -1]
            # i.e., grad[j] += 2*a[k-j], so grad[j_start] += 2*a[k-j_start] = 2*a[m_end]
            # grad[j_start+1] += 2*a[k-j_start-1] = 2*a[m_end-1]
            # ...
            # grad[j_end] += 2*a[k-j_end] = 2*a[m_start]
            grad[j_start:j_end+1] += 2.0 * a[m_start:m_end+1][::-1]
    
    return grad


def _score_and_conv(a):
    """Return (score, convolution)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, np.zeros(2*n-1)
    c = _full_convolve(a, a)
    sc = float(2.0 * n * c.max() / (s * s))
    return sc, c


def _projected_gradient_step(a, c, n, active_idx, step_size):
    """
    Take a projected gradient step on the simplex.
    Minimize sum_{k in active} c[k] (or the max, approximated by the sum of top peaks)
    subject to a >= 0, sum(a) = s (preserve sum).
    """
    s = float(a.sum())
    grad = _grad_conv_active(a, c, n, active_idx)
    # Scale gradient: we're minimizing 2*n*max_c/s^2, but since s is fixed,
    # we're effectively minimizing max_c, so gradient of sum of active peaks is fine.
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-30:
        return a.copy()
    a_new = a_new * (s / s_new)  # preserve sum
    return a_new


def _line_search(a, c, n, active_idx, grad, max_iter=50):
    """Backtracking line search."""
    s = float(a.sum())
    cur_score_obj = float(sum(c[k] for k in active_idx))  # use sum of active as proxy
    alpha = 1.0
    for _ in range(max_iter):
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            alpha *= 0.5
            continue
        a_new = a_new * (s / s_new)
        c_new = _full_convolve(a_new, a_new)
        new_obj = float(sum(c_new[k] for k in active_idx))
        if new_obj < cur_score_obj - 1e-18 * abs(cur_score_obj):
            return a_new, alpha
        alpha *= 0.5
    return a.copy(), 0.0


def _peak_chase(a_init, deadline, reserve=100.0, k_active=1):
    """
    Alternating peak-chasing algorithm:
    1. Find the peak of the convolution
    2. Take a projected gradient step to reduce it
    3. Periodically re-evaluate the active set
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    base_step = 1.0 / (4.0 * _full_convolve(a, a).max() + 1e-10)
    alpha = base_step
    stall = 0
    stall_limit = 200
    
    while time.monotonic() < deadline - reserve:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find active set: top k values
        # Use adaptive k: if there's a clear peak, use k=1; if flat, use more
        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - (sorted_c[k_active] if k_active < len(sorted_c) else sorted_c[-1])
        if gap < c_max * 0.01:
            k_eff = min(k_active * 3, 20)
        else:
            k_eff = k_active
        
        active_idx = np.argsort(c)[::-1][:k_eff]
        
        grad = _grad_conv_active(a, c, n, active_idx)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        # Normalize gradient
        grad_norm = grad / gnorm
        
        # Line search
        a_new, alpha_used = _line_search(a, c, n, active_idx, grad_norm, max_iter=30)
        
        if alpha_used > 0:
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                stall = 0
            a = a_new
            # Adapt step size
            if alpha_used < 1.0:
                alpha = alpha_used * 0.9
            else:
                alpha = min(alpha * 1.1, 1.0)
        else:
            stall += 1
            alpha *= 0.5
            if stall > stall_limit:
                break
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    """L-BFGS with log-sum-exp smoothing in log-space."""
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.mean()

    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_beta_lbfgs(a_start, deadline, reserve_time=100.0):
    """Adaptive L-BFGS with decreasing beta."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # More aggressive beta schedule
    betas = max_c * np.array([0.3, 0.15, 0.08, 0.04, 0.02, 0.01,
                              0.005, 0.002, 0.001, 0.0005,
                              0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        # Scale max_iter based on n
        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(8000, max(200, int(remaining / (0.005 * scale * 15))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.501:
            break

    return current_a


def _spectral_flatten(a_init, deadline, reserve=100.0):
    """
    Try to flatten the power spectrum |FFT(a)|^2.
    A flat spectrum means the autocorrelation is concentrated at lag 0 (delta-like),
    which minimizes the peak of the linear convolution (for non-edge lags).
    
    We minimize the variance of |FFT(a)|^2 using gradient descent in log-space.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    lr = 0.01
    stall = 0
    
    for it in range(10000):
        if time.monotonic() > deadline - reserve:
            break
        
        a = _softmax(z)
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        # Minimize variance of power (or the peak, which is related)
        # Actually, minimize the peak of the autocorrelation = peak of ifft(power)
        # But flattening power is a good proxy
        
        # Gradient of var(power) w.r.t. z:
        # power[k] = |A[k]|^2 = |sum_j a[j] * exp(-2pi i j k / N)|^2
        # This is complicated. Let's use the peak of the convolution directly.
        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argmax(c)
        
        # Subgradient of c_max
        a_rev = a[::-1]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(a[idx] if idx < n else 0)  # This isn't right
        # Actually W = w[0]*c[0] + ... = c[idx]
        W = c[idx]
        grad = 4.0 * n * a * (h - W)
        
        z = z - lr * grad / (n * a + 1e-30)
        z = z - z.mean()
        
        if it % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                lr = min(lr * 1.05, 0.1)
            else:
                stall += 1
                if stall > 10:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95
    
    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):

    3. Spectral flattening heuristic
    4. Multi-phase alternating with adaptive time allocation
    5. Length exploration (n = 4096 to 32768)
    
    Key differences from previous approach:
    - Direct simplex projection (avoids softmax distortion)
