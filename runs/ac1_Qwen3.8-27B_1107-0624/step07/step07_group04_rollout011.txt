import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    L = np.fft.rfft(lam, N)
    conv = np.fft.irfft(V * L, N)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _cg_solve_hankel(lam, n, rhs, max_iter=500, tol=1e-12, reg=0.0):
    """Solve H*x = rhs using CG, where H_{j,m} = lam[j+m] + reg*I_{j,m}."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, lam, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _primal_dual(a_init, deadline, max_iters=200):
    """
    Primal-dual method with exponentiated gradient.
    Minimizes max_k (a*a)_k subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    m = 2 * n - 1
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    # Initialize dual uniformly
    lam = np.ones(m) / m
    
    best_a = a.copy()
    best_sc = _score(a)
    
    # Adaptive step size
    eta = 0.5
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Primal: solve H(a) a = 1 (approximate)
        ones = np.ones(n)
        a_new = _cg_solve_hankel(lam, n, ones, max_iter=200, tol=1e-10)
        
        # Ensure non-negativity and normalization
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check if this is better
        c = _full_convolve(a_new, a_new)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
        
        # Dual update: exponentiated gradient
        # lam_k proportional to exp(eta * c_k)
        c_shifted = c - c.max()
        lam_new = lam * np.exp(eta * c_shifted)
        lam_sum = lam_new.sum()
        if lam_sum < 1e-300 or not np.isfinite(lam_sum):
            lam_new = np.exp(np.clip(c_shifted * eta, -700, 700))
            lam_sum = lam_new.sum()
            if lam_sum < 1e-300:
                break
        lam_new = lam_new / lam_sum
        lam = lam_new
        
        a = a_new
        
        # Decay step size
        eta *= 0.995
    
    return best_a

def _active_set_qp(a_init, lam_weights, deadline, max_iters=1000):
    """
    Solve the QP: min a^T H a s.t. sum(a)=1, a>=0
    where H is determined by lam_weights (dual variables).
    Uses projected gradient with backtracking.
    """
    n = len(a_init)
    m = 2 * n - 1
    
    # Form the effective lambda (zero out non-active)
    lam = np.zeros(m)
    if lam_weights is not None and len(lam_weights) == m:
        lam = lam_weights.copy()
    else:
        lam = np.ones(m) / m
    
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = float(a @ _hankel_matvec(a, lam, n))
    
    lr = 1.0
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Gradient of a^T H a is 2Ha
        g = 2.0 * _hankel_matvec(a, lam, n)
        
        # Projected gradient step
        a_new = _proj_simplex(a - lr * g, s=1.0)
        
        # Backtracking
        obj_new = float(a_new @ _hankel_matvec(a_new, lam, n))
        if obj_new > best_obj:
            # Try smaller step
            lr *= 0.5
            a_new = _proj_simplex(a - lr * g, s=1.0)
            obj_new = float(a_new @ _hankel_matvec(a_new, lam, n))
            if obj_new > best_obj:
                continue
        
        a = a_new
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            lr *= 0.7
    
    return best_a

def _smoobj_and_grad(z, beta, n):
    """Compute smoothed objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    
    # Gradient: 4*n*a*(h - W) where h = conv(w, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    
    return obj, grad

def _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with log-sum-exp smoothing and aggressive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)
    
    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                       0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                       0.00002, 0.00001, 0.000005, 0.000002, 0.000001])
    betas = base_beta * ratios
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 10))))
        
        def obj_fn(z_local):
            val, _ = _smoobj_and_grad(z_local, beta, n)
            return val
        
        def grad_fn(z_local):
            _, g = _smoobj_and_grad(z_local, beta, n)
            return g
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _newton_polish(a_init, deadline, max_rounds=100):
    """CG-Newton polish on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set
        tol = 0.0003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        
        # Gradient
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break
        
        # Newton step via CG
        Hg = _cg_solve_hankel(W, n, g, max_iter=300, tol=1e-12)
        H1 = _cg_solve_hankel(W, n, np.ones(n), max_iter=300, tol=1e-12)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)

            dx = -Hg + lam_val * H1
