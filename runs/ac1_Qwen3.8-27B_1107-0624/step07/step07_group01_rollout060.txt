"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale L-BFGS with log-sum-exp smoothing and adaptive beta
2. Large-active-set subgradient with Nesterov momentum
3. Quantile-based peak weighting for smoother gradients
4. Intelligent multi-start diversification with annealing
5. Coarse-to-fine refinement strategy
6. Direct subgradient polishing with backtracking line search

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs_v2(a_start, deadline, reserve_time=60.0):
    """Improved L-BFGS with better beta schedule and more iterations per level."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Fewer beta levels, more time per level, start from moderate beta
    betas = max_c * np.array([0.05, 0.03, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6,
                              2e-6, 1e-6])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        time_per_iter = 0.003 * (n / 4096.0) ** 0.8

        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta/max_c:.4f}: score={current_sc:.10f} (improved)")
        else:

def _subgrad_momentum(a_init, deadline, k_active=50, max_steps=100000,
                      lr_init=0.08, momentum=0.85):
    """Subgradient descent with large active set and Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    velocity = np.zeros(n)
    stall = 0
    patience = 800

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov look-ahead
        z_lookahead = z + momentum * velocity
        a = _softmax(z_lookahead)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Quantile-based active set: use peaks above 90th percentile
        threshold = np.percentile(c, 90)
        active_mask = c >= threshold
        n_active = active_mask.sum()
        if n_active < k_active:
            idx = np.argsort(c)[::-1][:k_active]
            w = np.zeros(len(c))
            w[idx] = 1.0 / k_active
        else:
            w = np.zeros(len(c))
            w[active_mask] = 1.0 / n_active
            n_active = active_mask.sum()

        # Compute gradient in z-space
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Update velocity and position
        velocity = momentum * velocity - lr * grad / (n * a.sum() + 1e-30)
        z = z + velocity
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        # Periodic score check
        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.03, 0.5)
                stall = 0
            else:
                stall += 1
                if stall > patience:
                    lr *= 0.7
                    velocity *= 0.5
                    if lr < 1e-9:
                        break
                elif stall > patience // 2:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_direct(a_init, deadline, k_active=30, max_steps=50000):
    """Direct subgradient in a-space with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        c_mean = c.mean()

        # Weight peaks by excess over mean
        excess = np.maximum(c - c_mean, 0)
        if excess.sum() < 1e-30:
            break
        w = excess / excess.sum()

        # Limit to top-k for efficiency
        idx = np.argsort(c)[::-1][:k_active]
        w_top = np.zeros(len(c))
        w_top[idx] = 1.0 / k_active

        # Use combined weighting: 50% top-k, 50% quantile
        w_use = 0.5 * w_top + 0.5 * w

        a_rev = a[::-1]
        wc = _full_convolve(w_use, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w_use, c))
        g = h - W
        g = g - g.mean()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        alpha = step_size
        improved = False
        for _ in range(15):
            a_try = a - alpha * g / (gnorm + 1e-30)
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-30:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
            if sc_try < _score(a) - 1e-16:
                a = a_try
                improved = True
                step_size = min(alpha * 1.05, 2.0 / np.sqrt(n))
                break
            alpha *= 0.7

        if not improved:
            stall += 1
            step_size *= 0.85
            if stall > 200:
                break
        else:
            stall = 0

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_spectral(n, rng, n_freqs=20, max_amp=0.3):
    """Create a sequence with a specific spectral shape (random phases)."""
    t = np.arange(n)
    a = np.ones(n)
    for k in range(1, n_freqs + 1):
        amp = max_amp / np.sqrt(n_freqs) * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_squared_chirp(n, freq=1.0, eps=0.5):
    """a_i = (1 + eps*cos(2*pi*freq*t^2/n))^2 - gives different spectral shape."""
    t = np.arange(n)
    base = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = base ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _anneal_restart(a_best, rng, temperature=0.05):
    """Create a perturbed version of the best solution for diversification."""
    n = len(a_best)
    z = np.log(np.maximum(a_best, 1e-18))
    z = z + temperature * rng.standard_normal(n)
    z = z - z.mean()
    z = np.clip(z, -50, 50)
    return _softmax(z)


def _coarse_to_fine(a_init, n, deadline, block_size=8):
    """Optimize a block-averaged version, then use as initialization for full optimization."""
    n_blocks = n // block_size
    if n_blocks < 16:
        return a_init

    # Coarsen
    a_coarse = a_init[:n_blocks * block_size].reshape(n_blocks, block_size).mean(axis=1)

    # Optimize coarse
    c_coarse = _full_convolve(a_coarse, a_coarse)
    max_c = float(c_coarse.max())
    beta = max_c * 0.02
    max_iter = min(5000, int((deadline - time.monotonic()) / (0.001 * n_blocks * 3)))
    max_iter = max(100, max_iter)

    a_coarse_opt = _lbfgs_smooth(a_coarse, beta, n_blocks, max_iter)

    # Upsample
    a_upsampled = np.repeat(a_coarse_opt, block_size)
    if len(a_upsampled) < n:
        a_upsampled = np.concatenate([a_upsampled, a_init[len(a_upsampled):]])
    a_upsampled = a_upsampled[:n]
    a_upsampled = np.maximum(a_upsampled, 1e-15)
    a_upsampled = a_upsampled / a_upsampled.sum()

    # Blend with original
    a_blend = 0.5 * a_upsampled + 0.5 * a_init
    a_blend = a_blend / a_blend.sum()

    return a_blend

        a_pert = _anneal_restart(best_seq, rng, temperature=temp)

        # Also try a "structured" perturbation: modify the spectral content
        if s % 2 == 1:
            # Add random Fourier components
            z = np.log(np.maximum(a_pert, 1e-18))
            n_mods = 3 + s
            for _ in range(n_mods):
                freq = int(rng.integers(1, max(2, n_target // 8)))
                amp = 0.03 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n_target)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n_target + phase)
            z = z - z.mean()
            z = np.clip(z, -50, 50)
            a_pert = _softmax(z)

        # Quick optimization on perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        beta = mc * 0.01
        max_iter = int(remaining_here / (0.003 * (n_target / 4096.0) ** 0.8))
        max_iter = min(5000, max(500, max_iter))
        a_opt = _lbfgs_smooth(a_pert, beta, n_target, max_iter)

        # Follow with subgradient
        if time.monotonic() < phase4_deadline - 30:
            a_opt2, _ = _subgrad_momentum(a_opt, phase4_deadline - 30,
                                           k_active=30, max_steps=10000,
                                           lr_init=0.04, momentum=0.8)
            if _score(a_opt2) < _score(a_opt):
                a_opt = a_opt2

        try_update(a_opt, f"diversify_{s}")

    # === Phase 5: Coarse-to-fine ===
    print(f"\n=== Phase 5: Coarse-to-fine (t={time.monotonic()-t_start:.1f}s) ===")
    phase5_deadline = t_start + 1600
    if time.monotonic() < phase5_deadline - 30:
        a_cf = _coarse_to_fine(best_seq, n_target, phase5_deadline, block_size=8)
        # Optimize the coarse-to-fine result
        c_cf = _full_convolve(a_cf, a_cf)
        mc_cf = float(c_cf.max())
        beta_cf = mc_cf * 0.005
        max_iter = min(5000, int((phase5_deadline - time.monotonic()) / 0.005))
        a_cf_opt = _lbfgs_smooth(a_cf, beta_cf, n_target, max(200, max_iter))
        try_update(a_cf_opt, "coarse_fine")

    # === Phase 6: Second L-BFGS round on best ===
    print(f"\n=== Phase 6: Second L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_p6 = _adaptive_lbfgs_v2(best_seq, deadline - 100, reserve_time=30.0)
        try_update(a_p6, "lbfgs_round2")

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final, sc_final = _final_polish(best_seq, deadline)
        try_update(a_final, "final_polish")

    # === Phase 8: Last-ditch subgradient ===
    print(f"\n=== Phase 8: Last subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_last, sc_last = _subgrad_momentum(best_seq, deadline - 3,
                                             k_active=20, max_steps=5000,
                                             lr_init=0.01, momentum=0.7)
        try_update(a_last, "last_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
