"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp L-BFGS, multi-scale peak shaving with proper line search,
frequency-domain spectral flattening, and multi-restart strategies.
Key improvements: adaptive beta schedule, multi-scale active sets, spectral post-processing,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_weights(c, tol_frac):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    threshold = c_max - tol_frac * c_max
    active = c >= threshold
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _gradient_from_weights(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _peak_shaving_multiscale(a_init, deadline, max_iters=30000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.05
    stall = 0

    scales = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        scale_idx = min(int(i / 2000), len(scales) - 1)
        tol_frac = scales[scale_idx]
        w = _active_set_weights(c, tol_frac)

        g = _gradient_from_weights(a, c, n, w)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_norm = g / g_norm

        best_sc_try = best_sc
        best_a_try = a.copy()
        best_frac = 1.0

        for fr in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * fr
            a_try = a - step * g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_frac = fr

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.3)
            stall = 0
        else:
            stall += 1
            eta *= 0.85
            if eta < 1e-10 or stall > 200:
                break

    return best_a, best_sc


def _lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    min_c = float(c_cur.min())
    spread = max_c - min_c
    if spread < 1e-15:
        spread = max_c * 0.1

    current_a = a_init.copy()
    current_sc = _score(current_a)
    beta = spread * 0.5
    min_beta = spread * 1e-8
    beta_ratio = 0.4

    while time.monotonic() < deadline - min_time_remaining and beta >= min_beta:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        z_current = np.log(np.maximum(current_a, 1e-15))
        z_current = z_current - z_current.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        max_iter = min(8000, max(100, int(remaining / (0.003 * (n / 2048.0)))))

        try:
            result = minimize(fun, z_current, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxcor': 100, 'maxls': 60})

                current_sc = sc_new

            else:
                a_new = _softmax(result.x)
                sc_new2 = _score(a_new)
                if sc_new2 < current_sc:
                    current_a = a_new
                    current_sc = sc_new2
        except Exception:
            pass

        beta *= beta_ratio

    return current_a


def _spectral_flatten(a, deadline, max_iters=500):
    n = len(a)
    current_a = a.copy()
    current_sc = _score(current_a)

    N = _next_pow2(2 * n - 1)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(current_a, N)
        mag_sq = np.abs(A) ** 2
        total_energy = mag_sq.sum()

        c = _full_convolve(current_a, current_a)
        c_max = c.max()
        target = float(current_a.sum()) ** 2 / (2 * n - 1)

        if c_max < target * 1.01:
            break

        w = _active_set_weights(c, 0.05)
        W_spec = np.fft.rfft(w, N)
        desired_flat = np.ones(N // 2 + 1)
        desired_flat[0] = 1.0

        error_spec = W_spec - np.abs(W_spec).mean()

        alpha = 0.01
        A_new = A - alpha * error_spec * np.conj(A) / (np.abs(A) ** 2 + 1e-10)

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc_new

    return current_a, current_sc


def _coordinate_descent(a_init, deadline, max_rounds=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        if rnd % 2 == 0:
            idx = int(rng.integers(0, n))
            jdx = int(rng.integers(0, n))
            if idx == jdx:
                jdx = (jdx + 1) % n
        else:
            idx = int((rnd * 6151 + 7) % n)
            jdx = int((rnd * 6151 + 13) % n)
            if idx == jdx:
                jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        best_local_sc = best_sc
        best_local_frac = 0.5

        for frac in [0.5, 0.35, 0.65, 0.2, 0.8, 0.3, 0.7, 0.15, 0.85]:
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            if sc_try < best_local_sc - 1e-13:
                best_local_sc = sc_try
                best_local_frac = frac

        if best_local_frac != 0.5:
            a[idx] = s_pair * best_local_frac
            a[jdx] = s_pair * (1.0 - best_local_frac)
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = a.copy()

    return best_a, best_sc


def _multi_restart(a_best, deadline, min_time_per_restart=30.0, max_restarts=8):

                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        else:
            block_size = max(16, n // 32)
            for _ in range(6):

                end = min(n, start + block_size)

                best_sc = sc_opt

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
