def _proj_simplex(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v, s=1.0):
    """Project v onto the probability simplex {x: sum(x)=s, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hess_matvec(x, S_arr, n):
    """Compute H*x where H[j,l] = 2/m if j+l in S, else 0.
    S_arr is a boolean array of length 2n-1 indicating the active set.
    """
    m = int(S_arr.sum())
    if m == 0:
        return np.zeros(n)
    x_rev = x[::-1]
    A = S_arr.astype(np.float64)
    out_len = n + len(A) - 1
    N = _next_pow2(out_len)
    fx = np.fft.rfft(x_rev, N)
    fA = np.fft.rfft(A, N)
    result = np.fft.irfft(fx * fA, N)[:out_len]
    y = result[n - 1:2 * n - 1]
    return (2.0 / m) * y


def _grad_active(a, S_arr, n):
    """Compute gradient of (1/|S|) * sum_{k in S} c[k] at point a."""
    return _hess_matvec(a, S_arr, n)


def _est_lipschitz(S_arr, n, rng):
    """Estimate Lipschitz constant (lambda_max of H) via power iteration."""
    v = rng.random(n)
    v /= np.linalg.norm(v)
    L = 0.0
    for _ in range(30):
        Hv = _hess_matvec(v, S_arr, n)
        new_L = float(np.dot(v, Hv))
        norm = np.linalg.norm(Hv)
        if norm < 1e-15:
            break
        v_new = Hv / norm
        L = max(L, new_L)
        v = v_new
    return max(L * 1.05, 1e-10)


def _fista_qp(a_init, S_arr, n, L, max_iter=1000, tol=1e-13):
    """Solve min (1/2)a^T H a s.t. sum(a)=1, a>=0 using FISTA."""
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    obj_curr = float(0.5 * np.dot(a, _hess_matvec(a, S_arr, n)))

    y = a.copy()
    t = 1.0
    a_prev = a.copy()

    for k in range(max_iter):
        Hy = _hess_matvec(y, S_arr, n)
        a_curr = _proj_simplex(y - Hy / L)
        obj_try = float(0.5 * np.dot(a_curr, Hy)) + 0.5 / L * float(np.dot(Hy, Hy))
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_curr + ((t - 1.0) / t_new) * (a_curr - a_prev)
        a_prev = a_curr
        t = t_new

        if k % 100 == 0:
            diff = np.linalg.norm(a_curr - a_prev)
            if diff < tol * max(1.0, np.linalg.norm(a_curr)):
                break

    return a_curr


def _solve_active_set_qp(a_init, tolerance, n, max_outer=50, max_fista=800, deadline=None):
    """Active-set QP: iteratively identify active set and solve QP."""
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    rng = np.random.default_rng(12345)
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        S_arr = c >= (c_max - tolerance * c_max)
        if not S_arr.any():
            S_arr[int(np.argmax(c))] = True

        m = int(S_arr.sum())
        if m > 2 * n:
            top_idx = np.argsort(c)[::-1][:2 * n]
            S_arr = np.zeros(2 * n - 1, dtype=bool)
            S_arr[top_idx] = True
            m = 2 * n

        L = _est_lipschitz(S_arr, n, rng)

        a_new = _fista_qp(a, S_arr, n, L, max_iter=max_fista)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()

        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())

        if c_max_new <= c_max + 1e-12:
            break

        a = a_new
        if tolerance > 1e-6:
            tolerance *= 0.8

    return best_a, best_sc


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_active(a_init, deadline, n_steps=3000, lr_init=0.05):
    """Subgradient method with active-set averaging and Polyak step."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_cmax = _full_convolve(best_a, best_a).max()

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.005 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS warm-start
    Phase 2: Active-set QP with FISTA (FFT-based Hessian matvec)
    Phase 3: Multi-n exploration
    Phase 4: Subgradient polish and restarts
    Phase 5: Final tight active-set refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS warm-start
    print(f"\n[Phase 1] L-BFGS warm-start (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Active-set QP with FISTA
    print(f"\n[Phase 2] Active-set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        n = len(best_seq)
        # Multiple rounds with decreasing tolerance
        tol_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        current_a = best_seq.copy()
        for round_idx, tol in enumerate(tol_schedule):
            remaining = deadline - time.monotonic()
            if remaining < 100:
                break
            alloc = min(80.0, remaining * 0.1)
            round_deadline = time.monotonic() + alloc
            print(f"  Round {round_idx}: tol={tol}, alloc={alloc:.0f}s")
            a_qp, sc_qp = _solve_active_set_qp(current_a, tol, n,
                                                max_outer=30, max_fista=600,
                                                deadline=round_deadline)
            sc_qp = _score(a_qp)
            update_best(a_qp, sc_qp, f"as_qp_r{round_idx}_tol{tol}")
            if sc_qp < best_score + 1e-10:
                current_a = a_qp

    # Phase 3: Multi-n exploration
    print(f"\n[Phase 3] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates, key=lambda x: abs(x - n_parent)):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue

        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        lbfgs_deadline = time.monotonic() + min(60.0, remaining * 0.4)
        a_opt = _run_lbfgs(a_init, lbfgs_deadline, min_time_remaining=5.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"lbfgs_n{n_try}")

        # Active-set QP on this n
        remaining2 = deadline - time.monotonic()
        if remaining2 > 40 and sc_opt < best_score + 0.05:
            qp_deadline = min(deadline - 50, time.monotonic() + 40)
            a_qp2, sc_qp2 = _solve_active_set_qp(a_opt, 0.005, n_try,
                                                  max_outer=20, max_fista=400,
                                                  deadline=qp_deadline)
            update_best(a_qp2, sc_qp2, f"asqp_n{n_try}")

    # Phase 4: Subgradient polish + restarts
    print(f"\n[Phase 4] Polish & restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        # Subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 60:
            sg_deadline = min(deadline - 60, time.monotonic() + 60)
            a_sg, sc_sg = _subgrad_active(best_seq, sg_deadline, n_steps=2000, lr_init=0.03)
            update_best(a_sg, sc_sg, "sg_polish")

        # Diverse restarts
        restarts = [
            ("perturb_0.03", 0.03),
            ("perturb_0.08", 0.08),
            ("perturb_0.15", 0.15),
        ]
        for label, sigma in restarts:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            ll_deadline = min(deadline - 20, time.monotonic() + 30)
            a_r = _run_lbfgs(p_start, ll_deadline, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            # Quick active-set QP on restart
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20 and sc_r < best_score + 0.01:
                qp_dl = min(deadline - 10, time.monotonic() + 20)
                a_r2, sc_r2 = _solve_active_set_qp(a_r, 0.003, len(a_r),
                                                    max_outer=15, max_fista=300,
                                                    deadline=qp_dl)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 5: Final tight refinement
    print(f"\n[Phase 5] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            # Very tight active-set QP
            final_dl = min(deadline - 3, time.monotonic() + min(40.0, remaining * 0.6))
            a_fin, sc_fin = _solve_active_set_qp(best_seq, 0.0001, n,
                                                  max_outer=50, max_fista=500,
                                                  deadline=final_dl)
            update_best(a_fin, sc_fin, "final_as_tight")

            # Tiny beta L-BFGS
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5 and best_seq is not None:
                n = len(best_seq)
                c_best = _full_convolve(best_seq, best_seq)
                mb = float(c_best.max())
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z - z.mean()
                beta_final = mb * 0.00003
                ll_dl = min(deadline - 2, time.monotonic() + min(30.0, remaining2 * 0.7))
                est_iter = max(100, int((ll_dl - time.monotonic()) / 0.03))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': min(est_iter, 3000), 'ftol': 1e-16,
                                 'gtol': 1e-13, 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
