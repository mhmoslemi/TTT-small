```python
"""
Hybrid optimizer using spectral flattening, coordinate descent, and active-set methods.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

# Spectral flattening algorithm
def _spectral_flatten(a, n_iters=1000, alpha=0.5, deadline=None):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        
        # Target: flatter spectrum
        # Use a "soft" flattening: P_new = P^alpha * mean(P)^(1-alpha)
        # This reduces dynamic range while preserving some structure
        mean_P = np.mean(P)
        if mean_P < 1e-30:
            break
        P_flat = P**alpha * mean_P**(1-alpha)
        
        # Keep phase, use new magnitude
        phase_A = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase_A)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project onto non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        sc = _score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a
```

   c. Compute a "flattened" version: P_flat = P / (P + lambda * mean(P)) or P_flat = P^alpha for alpha < 1
   d. Compute a_new = IFFT(sqrt(P_flat) * phase(A))
   e. Project: a_new = max(a_new, 0)
   f. Normalize: a_new = a_new / sum(a_new)
   g. Set a = a_new
3. Return a

The parameter alpha controls the aggressiveness of flattening. alpha=1 means no change; alpha->0 means complete flattening.

After this gives a good approximation, we can refine with local optimization.
