Let me also think about a completely different approach: **iterative water-filling**.

The idea: if we know which positions k are active (c[k] near max), we want to "push down" the convolution at those positions. This means reducing the contributions a[i]*a[k-i] for those k. We can do this by:
- For each active k, the gradient tells us which pairs (i, k-i) contribute most.
- We can reduce a[i] and/or a[k-i] to reduce c[k].
- But reducing a[i] also affects other c[k'] values.

This is essentially what the active-set Newton method does, but we can make it more efficient.

**Alternative: use CVXPY**
Since the problem is convex (minimize max of quadratic functions subject to linear constraints), we can potentially use CVXPY with a suitable solver. The challenge is that for n=8192, the problem has 8192 variables and 2n-1 = 16383 quadratic constraints, which might be too large for most solvers.

However, we could:
1. Fix the active set (say, the top m positions where m is small, like 50 or 100).
2. Solve the QP: minimize sum_{k in S} c[k] subject to sum(a) = 1, a >= 0.
3. Update the active set.
4. Iterate.

The QP in step 2 has n variables and n-2 constraints (sum + non-negativity). The quadratic objective is sum_{k in S} a^T M_k a where M_k are 0-1 matrices. This is a standard QP that CVXPY or other solvers can handle.

For n=8192 and |S| ~ 100, the QP has a 8192x8192 Hessian (sparse, with at most 2|S| non-zero entries per row). This should be solvable.

def _cg_solve(hess_fn, b, n, max_iter=300, tol=1e-12, restart_every=30):

    x = np.zeros(n)

        Hp = hess_fn(p)

        r = r - alpha * Hp

            rs = r_new

            rs = r_new

        if time.monotonic() > deadline:
            break
        grad = _variance_grad(a, n)
        grad_t = grad - np.mean(grad)
        a_new = a - lr * grad_t
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if i % 20 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            lr = lr * 0.995

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    w_rev = w[::-1]
    a_rev = a[::-1]
    N = _next_pow2(4 * n)
    fw = np.fft.rfft(w_rev, N)
    fa = np.fft.rfft(a_rev, N)
    r = np.fft.irfft(fw * fa, N)
    h = r[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, deadline, min_time_remaining=60):
    """L-BFGS on softened max with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0004,
                                  0.00015, 0.00006, 0.00002, 0.000008])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 6))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01):
    """Subgradient descent on active set for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = (c >= c_max - 0.001 * c_max).astype(np.float64)
        if active.sum() == 0:
            active[int(np.argmax(c))] = 1.0
        grad = _hess_vec(a, active, n)
        grad_t = grad - np.mean(grad)

        step = lr / (1.0 + 0.0005 * i)
        z = z - step * (grad_t / a).astype(np.float64)
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.25 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
