"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategies: log-sum-exp L-BFGS continuation, active-set Newton with CG solve,
accelerated subgradient, multi-n exploration (up to 32768), structured init,
and coordinate refinement on the active set.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=400, tol_active=0.001, verbose=False):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        max_active = max(n_active, n)
        if n_active > max_active:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:max_active]
            active[top_idx] = True
            n_active = max_active

        m = n_active
        A_vec = active.astype(np.float64)

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-5 * (c_max / max(m, 1)) + 1e-9

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=300):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs_old = np.dot(r, r)
            if rs_old < 1e-30:
                return x
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs_old / pAp
                x_new = x + alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13 * max(1.0, np.linalg.norm(rhs)):
                    x = x_new
                    break
                p = r_new + (rs_new / rs_old) * p
                x = x_new
                r = r_new
                rs_old = rs_new
            return x

        u = cg_solve(g_t)
        w_vec = cg_solve(np.ones(n))
        sum_w = float(w_vec.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w_vec
        dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-12:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                if verbose and it < 5:
                    print(f"    Newton iter {it}: no step accepted, gnorm={gnorm:.2e}")
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 20 == 0:
                print(f"    Newton iter {it}: score={sc:.10f}")

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - f_old) < 1e-13:
            break

    return best_a, best_sc


def _accelerated_subgrad(a_init, deadline, max_steps=8000, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    best_f = best_sc / (2.0 * n)

    z_prev = z.copy()
    lr = lr_init
    patience = 0
    best_lr = lr

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 49:
            a_check = _softmax(z)
            sc = _score(a_check)
            f = sc / (2.0 * n)
            if f < best_f - 1e-14:
                best_f = f
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                best_lr = lr
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = best_lr * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03,
                                  0.015, 0.0075, 0.0035, 0.0015, 0.0007,
                                  0.0003, 0.00015, 0.00007, 0.00003, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(12000, max(100, int(remaining / (est_time_per_iter * 6))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if verbose:
                print(f"    L-BFGS beta={beta:.2e}: score={current_sc:.10f}, iters={result.nit}")
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "geometric":
        r = 0.995
        a = r ** np.arange(n)
        return a / a.sum()
    elif kind == "power":
        t = np.arange(1, n + 1) / n
        a = t ** 0.3
        return a / a.sum()
    elif kind == "asym_power":
        t = np.arange(1, n + 1) / n
        a = t ** 0.7
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tail_heavy":
        t = np.arange(1, n + 1) / n
        a = 1.0 / (t ** 0.2 + 0.5)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 3)))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _pairwise_swap_refine(a_init, deadline, n_rounds=50):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 0.002 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    m = int(active.sum())

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        top_idx = np.argsort(a)[::-1]
        for i in top_idx[:min(100, n)]:
            if time.monotonic() > deadline:
                break
            g_i = (2.0 / m) * _corr_a_A(a, active.astype(np.float64), n)[i]
            for j in top_idx[:min(50, n)]:
                if i == j:
                    continue
                g_j = (2.0 / m) * _corr_a_A(a, active.astype(np.float64), n)[j]
                if g_i < g_j - 1e-8:
                    delta = min(a[i], 0.01 * a[i]) * 0.5
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try > 1e-15:
                        a_try = a_try / s_try
                        sc_try = _score(a_try)
                        if sc_try < best_sc - 1e-14:
                            a = a_try
                            best_sc = sc_try
                            best_a = a_try.copy()
                            improved = True

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        m = int(active.sum())

        if not improved:
            break

    return best_a, best_sc

        kinds = ["geometric", "power", "asym_power", "tail_heavy", "cos_asym"]
        for kind in kinds:
            if time.monotonic() > deadline - 30:
                break
            a_struct = _make_profile(n_best, kind)
            sc_struct = _score(a_struct)
            if sc_struct < best_score:
                update_best(a_struct, sc_struct)
            a_st_opt = _run_lbfgs(a_struct, deadline, min_time_remaining=15.0)
            sc_st_opt = _score(a_st_opt)
            update_best(a_st_opt, sc_st_opt)

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        restarts = [
            ("small", lambda: _perturb(best_seq, 0.02, rng)),
            ("mid", lambda: _perturb(best_seq, 0.05, rng)),
            ("fourier", lambda: _perturb_fourier(best_seq, rng, 5)),
            ("large", lambda: _perturb(best_seq, 0.1, rng)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 10:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=8.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r)

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_set_newton(a_r, deadline - 2, max_iter=80, tol_active=0.001)
                update_best(a_r2, sc_r2)

    # Phase 8: Final intensive polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        # Multiple rounds of Newton with decreasing tolerance
        for tol_seq in [0.001, 0.0005, 0.0002, 0.0001]:
            if time.monotonic() > deadline - 2:
                break
            a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1, max_iter=100, tol_active=tol_seq)
            update_best(a_fin, sc_fin)

        # Final tiny-beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3))
                except Exception:
                    pass

        # Pairwise swap refinement
        if time.monotonic() < deadline - 2:
            a_sw, sc_sw = _pairwise_swap_refine(best_seq, deadline - 1, n_rounds=20)
            update_best(a_sw, sc_sw)

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
