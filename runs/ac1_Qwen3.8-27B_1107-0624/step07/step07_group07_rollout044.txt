"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp L-BFGS with peak-focused weighting,
multi-length exploration, and momentum-based peak shaving.
Key improvements: dynamic beta adaptation, top-k gradient focusing,
and spectral-informed initialization for larger lengths.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n, top_k=0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_sub = c[idx]
        c_sub_max = c_sub.max()
        shifted = (c_sub - c_sub_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_sub_max + beta * np.log(s))
    else:
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n, top_k=0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_sub = c[idx]
        c_sub_max = c_sub.max()
        shifted = (c_sub - c_sub_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = np.zeros_like(c)
            w_sub = e / s
            w[idx] = w_sub
    else:
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _weighted_peak_grad(z, beta, n, weights):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = np.zeros_like(c)
    for k, wt in weights.items():
        w[k] = wt
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(z0, beta, n, max_iter, top_k=0, ftol=1e-16, gtol=1e-12):
    def fun(z):
        return _smoobj(z, beta, n, top_k)

    def jac(z):
        return _smograd(z, beta, n, top_k)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 120,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit), float(result.fun)
    except Exception:
        return _softmax(z0), False, 0, _smoobj(z0, beta, n, top_k)

def _peak_shaving_momentum(a_init, deadline, max_iters=20000):

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)

    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp
