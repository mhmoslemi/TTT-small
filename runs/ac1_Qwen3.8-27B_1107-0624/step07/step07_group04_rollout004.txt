def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp < 0:
            # Indefinite Hessian: use gradient direction instead
            alpha = rs / max(pHp, 1e-30) * 0.1
            x = x + alpha * p
            r = r - alpha * Hp
            rs_new = float(r @ r)
            if not np.isfinite(rs_new) or rs_new > rs * 1.5:
                break
            p = (rs_new / rs) * p + r
            rs = rs_new
        else:
            alpha = rs / pHp
            x_new = x + alpha * p
            r_new = r - alpha * Hp
            rs_new = float(r_new @ r_new)

            x = x_new
            r = r_new
            p = (rs_new / rs) * p + r
            rs = rs_new

    return x

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Cosine schedule: 30 stages from base_beta to base_beta * 1e-8
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)

    n_stages = 30
    # Cosine annealing: beta_i = min_beta + 0.5*(max_beta - min_beta)*(1 + cos(pi * i / (n_stages-1)))
    min_beta = base_beta * 1e-10
    stages = np.linspace(0, 1, n_stages)
    betas = min_beta + 0.5 * (base_beta - min_beta) * (1 + np.cos(np.pi * stages))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 3:
            break

        # Scale iterations based on remaining time and n
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        # More iterations for smaller beta (later stages)
        stage_factor = 1.0 + 2.0 * (i / n_stages)
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 20 * stage_factor))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, float(beta), n),
                z,
                jac=lambda z_local: _smograd(z_local, float(beta), n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,

                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i < n_stages * 0.3:

def _subgrad_polyak(a_init, deadline, max_steps=20000):
    """Subgradient method with Polyak averaging for the exact max objective."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    s_norm = 1.0  # normalize to sum = 1

    a = a_init.copy()
    a = a / a.sum()

    # Initialize Polyak average
    a_avg = a.copy()
    c = _full_convolve(a, a)
    c_max_init = float(c.max())

    # Learning rate: step_t = c_max_t / (t + 1)
    for t in range(1, max_steps + 1):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Subgradient of max at a: pick the argmax (or average over near-max)
        tol = 0.0001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient of sum_k w_k * c_k with respect to a
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)  # gradient of weighted sum of c_k

        # Projected subgradient step on the simplex
        step = c_max / (t + 1.0)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        # Polyak average
        a_avg = (t * a_avg + a_new) / (t + 1)
        a = a_new

        # Check improvement periodically
        if t % 500 == 0:
            sc = _score(a_avg)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_avg.copy()

    # Final check
    sc = _score(a_avg)
    if sc < best_sc:
        best_sc = sc
        best_a = a_avg.copy()
    return best_a, best_sc


def _support_reduction(a_init, deadline, min_time_remaining=120.0):
    """Iteratively zero out smallest entries and re-optimize."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for trial in range(8):
        if time.monotonic() + min_time_remaining > deadline:
            break
        # Try zeroing different fractions
        frac = 0.01 * (trial + 1)
        if frac > 0.15:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        idx_sort = np.argsort(a_try)
        a_try[idx_sort[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-10 or (trial == 0 and sc_try < best_sc + 0.001):
            # Re-optimize from this point
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_opt = _run_lbfgs_aggressive(a_try, deadline, min_time_remaining=min_time_remaining)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt
            # Newton polish
            if time.monotonic() + 60 > deadline:
                break
            a_np, sc_np = _newton_active_polish(best_a, deadline - 60, max_rounds=50)
            if sc_np < best_sc:
                best_sc = sc_np
                best_a = a_np

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cazacz":
        # Zadoff-Chu inspired: cosine with quadratic phase
        t = np.linspace(0, 1, n)

    elif kind == "wavenumber":

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):

    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _symmetric_optimize(a_init, deadline, min_time_remaining=60.0):
    """Optimize on the symmetric (palindromic) manifold."""
    n = len(a_init)
    # Symmetrize: a_sym[i] = (a[i] + a[n-1-i]) / 2
    a_sym = (a_init + a_init[::-1]) / 2.0
    a_sym = a_sym / a_sym.sum()

    # Optimize only the first half + middle
    half_n = (n + 1) // 2
    a_half = a_sym[:half_n].copy()

    def _obj_half(z_half):
        ah = np.exp(z_half)
        ah = ah / ah.sum()
        # Reconstruct full symmetric sequence
        if n % 2 == 0:
            a_full = np.concatenate([ah, ah[::-1]])
        else:
            a_full = np.concatenate([ah, ah[-2::-1]])
        a_full = a_full / a_full.sum()
        c = _full_convolve(a_full, a_full)
        c_max = float(c.max())
        beta = c_max * 0.01
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    z_half = np.log(np.maximum(a_half, 1e-30))

    try:
        result = minimize(
            _obj_half,
            z_half,
            method='L-BFGS-B',
            options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12}
        )
        ah_opt = np.exp(result.x)
        ah_opt = ah_opt / ah_opt.sum()
        if n % 2 == 0:
            a_full = np.concatenate([ah_opt, ah_opt[::-1]])
        else:
            a_full = np.concatenate([ah_opt, ah_opt[-2::-1]])
        a_full = a_full / a_full.sum()
    except Exception:
        a_full = a_sym

    sc = _score(a_full)
    return a_full, sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS with aggressive 30-stage cosine beta
    print(f"\n[Phase 1] Aggressive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_lbfgs_aggressive(parent, deadline, min_time_remaining=500.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Newton active set polish
    print(f"\n[Phase 2] Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 350, time.monotonic() + 200), max_rounds=80)
        update_best(a2, sc2, "newton_1")

    # Phase 3: Subgradient Polyak averaging
    print(f"\n[Phase 3] Subgradient Polyak (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a3, sc3 = _subgrad_polyak(best_seq, min(deadline - 250, time.monotonic() + 100), max_steps=15000)
        update_best(a3, sc3, "sg_polyak")
        # Newton polish after subgradient
        if sc3 < best_score + 0.005 and time.monotonic() < deadline - 200:
            a3n, sc3n = _newton_active_polish(best_seq, min(deadline - 200, time.monotonic() + 80), max_rounds=60)
            update_best(a3n, sc3n, "sg_newton")

    # Phase 4: Support reduction
    print(f"\n[Phase 4] Support reduction (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a4, sc4 = _support_reduction(best_seq, deadline, min_time_remaining=100.0)
        update_best(a4, sc4, "shrinkage")

    # Phase 5: Symmetric optimization path
    print(f"\n[Phase 5] Symmetric path (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a5, sc5 = _symmetric_optimize(best_seq, deadline, min_time_remaining=60.0)
        update_best(a5, sc5, "symmetric")
        if sc5 < best_score + 0.01:
            a5n, sc5n = _newton_active_polish(a5, deadline - 40, max_rounds=40)
            update_best(a5n, sc5n, "sym_newton")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [16384, 8192, 4096, 12288]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent and n_try == 8192:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 40:
            a_np, sc_np = _newton_active_polish(a_opt, deadline - 20, max_rounds=40)
            update_best(a_np, sc_np, f"n_{n_try}_np")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("perturb_small", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_med", lambda: _perturb(best_seq, 0.08, rng)),
            ("cazacz", lambda: _make_profile(len(best_seq), "cazacz")),
            ("wavenum", lambda: _make_profile(len(best_seq), "wavenumber")),
            ("bimodal", lambda: _make_profile(len(best_seq), "bimodal")),
        ]
        for name, gen in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen()
            if time.monotonic() > deadline - 25:
                break
            a_r = _run_lbfgs_aggressive(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{name}")
            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_active_polish(a_r, deadline - 10, max_rounds=30)
                update_best(a_rp, sc_rp, f"restart_{name}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 2, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-fine L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-9
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(50, int(remaining / 0.02)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
