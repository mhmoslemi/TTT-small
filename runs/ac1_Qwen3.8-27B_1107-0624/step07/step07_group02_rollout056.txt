def _solve_lp_direction(f, M, n):
    """Solve LP: maximize sum(g) s.t. f*g <= M (element-wise), g >= 0"""
    # A is (2n-1) x n, A[k,i] = f[k-i] if 0 <= k-i < n
    # b = M * ones(2n-1)
    # c = -ones(n) (minimize -sum(g))
    
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    
    # Build the constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None

**Key algorithmic idea: "Peak shaving" via convex optimization**

The problem is: min max_k (a*a)[k] s.t. sum(a) = 1, a >= 0.

This is a convex optimization problem (minimize the max of convex functions, subject to linear constraints). Wait, is (a*a)[k] convex in a? (a*a)[k] = sum_i a[i]*a[k-i], which is a quadratic form. For the max to be convex, each quadratic form needs to be convex, which means the Hessian must be PSD.

The Hessian of (a*a)[k] = sum_i a[i]*a[k-i] is 2*Q_k where (Q_k)_{ij} = 1 if j = k-i (i.e., the "anti-diagonal" at position k). This is a rank-1 matrix (outer product of the indicator vector of the anti-diagonal with itself), which is PSD. So each (a*a)[k] is convex in a!

"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP direction finding combined with L-BFGS smoothing and active-set polishing.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix_sparse(f, n):
    """Build sparse convolution matrix A where A[k,i] = f[k-i] if 0<=k-i<n."""
    rows = []
    cols = []
    vals = []
    f_list = f.tolist()
    for i in range(n):
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        for k in range(k_min, k_max + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(f_list[j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
