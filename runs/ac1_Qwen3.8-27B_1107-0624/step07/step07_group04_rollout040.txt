```python
def _project_simplex(v, s=1.0):
    """Project v onto {a >= 0, sum(a) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = v.copy()
    # Find the threshold
    u_sorted = np.sort(u)[::-1]
    cv = np.cumsum(u_sorted)
    rho = np.nonzero(u_sorted * np.arange(1, n+1) > (cv - s))[0]
    if len(rho) == 0:
        return np.full(n, s/n)
    rho = rho[-1]
    theta = (cv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
                a_S = a_S / s
                return a_S, True
    except Exception:
        pass
    return None, False


def _newton_step_full(a, W, n, max_cg_iter=300):
    """
    One Newton step: minimize sum_k W[k]*c_k s.t. sum(a)=1, a>=0.
    Returns the updated a.
    """
    # Gradient: g = H_W @ a
    g = _hankel_matvec(a, W, n)
    # Center the gradient (remove mean for the sum constraint)
    g_centered = g - np.mean(g)
    gn = np.linalg.norm(g_centered)
    if gn < 1e-14:
        return a

    # Newton direction: solve H_W @ d = -g_centered (approximately)
    d = _pcg_solve(-g_centered, W, n, max_iter=max_cg_iter)
    # Ensure sum(d) = 0
    d = d - np.mean(d)
    dn = np.linalg.norm(d)

    if dn < 1e-30:
        # Fall back to gradient step
        d = -g_centered / gn
        dn = 1.0
    else:
        d = d / dn

    # Backtracking line search
    best_a = a
    best_cn = float(2.0 * np.dot(W, _full_convolve(a, a)))
    alpha = 1.0
    for _ in range(40):
        a_try = a + alpha * d
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        a_try = a_try / s
        c_try = _full_convolve(a_try, a_try)
        cn_try = float(2.0 * np.dot(W, c_try))
        if cn_try < best_cn - 1e-15:
            best_a = a_try
            best_cn = cn_try
            break
        alpha *= 0.5
        if alpha < 1e-12:
            break
    return best_a


def _polish_newton(a_init, deadline, max_rounds=100, active_tol=0.0003):
    """Iterative Newton polish with dynamic active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set with adaptive tolerance
        tol = active_tol * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()

        a = _newton_step_full(a, W, n)
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _exact_support_optimize(a_init, deadline, max_support=3500):
    """
    Iterative support reduction with exact KKT solve.
    Repeatedly removes the smallest elements and solves exactly.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    support = np.ones(n, dtype=bool)

    for iteration in range(200):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Determine active set
        tol = 0.0005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / active.sum()

        # Get current support indices
        support_idx = np.where(support)[0]
        m = len(support_idx)

        if m == 0:
            break

        # Try exact solve if support is small enough
        if m <= max_support:
            a_S, feasible = _solve_kkt(W, n, support_idx)
            if feasible:
                # Build full solution
                a_new = np.zeros(n)
                a_new[support_idx] = a_S
                sc_new = _score(a_new)
                if sc_new < best_sc - 1e-13:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new.copy()

        # Try to reduce support: remove smallest elements
        # Remove in batches to be efficient
        current_support_vals = a[support]
        if len(current_support_vals) > 10:
            # Find candidates to remove (smallest 5%)
            n_remove = max(1, int(len(current_support_vals) * 0.05))
            local_idx = np.argsort(current_support_vals)[:n_remove]
            global_idx = support_idx[local_idx]

            # Only remove if they're truly small
            threshold = np.median(current_support_vals) * 0.1
            to_remove = global_idx[current_support_vals[local_idx] < threshold]

            if len(to_remove) > 0:
                support_new = support.copy()
                support_new[to_remove] = False
                n_new = support_new.sum()

                if n_new > 5:
                    # Check if the solution remains good
                    a_trial = a.copy()
                    a_trial[to_remove] = 0
                    s_trial = a_trial.sum()
                    if s_trial > 1e-10:
                        a_trial = a_trial / s_trial
                        sc_trial = _score(a_trial)
                        if sc_trial < best_sc + 0.001:
                            support = support_new
                            a = a_trial

        # Check if we've converged
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_check = 0.001 * c_max
        n_active = int((c >= c_max - tol_check).sum())

        # If active set is very small and we've made progress, continue tightening
        if n_active <= int(len(c) * 0.1):
            # Try even tighter active set
            tol_tight = 0.0001 * c_max
            active_tight = c >= (c_max - tol_tight)
            if active_tight.sum() >= 1:
                W_tight = np.zeros(2 * n - 1)
                W_tight[active_tight] = 1.0 / active_tight.sum()
                support_idx = np.where(support)[0]
                if len(support_idx) <= max_support:
                    a_S, feasible = _solve_kkt(W_tight, n, support_idx)
                    if feasible:
                        a_new = np.zeros(n)
                        a_new[support_idx] = a_S
                        sc_new = _score(a_new)
                        if sc_new < best_sc - 1e-13:
                            best_sc = sc_new
                            best_a = a_new.copy()
                            a = a_new.copy()

    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS optimization with smooth approximation and adaptive beta."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)

    # More aggressive beta reduction schedule
    ratios = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.008,
                       0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001,
                       0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001,
                       0.0000005, 0.0000002])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    remaining = deadline - min_time_remaining - time.monotonic()
    n_betas = min(len(betas), max(5, int(remaining / 30)))

    for i in range(n_betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        beta = betas[i]
        rem = deadline - min_time_remaining - time.monotonic()
        if rem < 3:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(rem / (est_time_per_iter * 20))))

        try:
            z_obj, z_grad = _lbfgs_callback_factory(z, beta, n)
            result = minimize(
                z_obj,
                z,
                jac=z_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lbfgs_callback_factory(z_init, beta, n):
    """Factory to create objective and gradient functions for L-BFGS."""
    def obj(z_local):
        a = _softmax(z_local)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def grad(z_local):
        a = _softmax(z_local)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        h = _hankel_matvec(a, w, n)
        W_val = float(np.dot(w, c))
        return 4.0 * n * a * (h - W_val)

    return obj, grad


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient method with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.015
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        h = _hankel_matvec(a, w, n)
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_val)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.01

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.3) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "steep":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        a = np.ones(n)
        taper = n // 8
        for i in range(taper):
            frac = (i + 1) / taper
            a[i] = frac
            a[n - 1 - i] = frac
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _spectral_init(n, mode="raised_cosine", bandwidth=0.5):
    """Initialize via frequency domain design."""
    N = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(N) * 2  # normalize to [0, 1]

    if mode == "raised_cosine":
        mask = np.zeros_like(freqs)
        center = freqs < bandwidth
        transition = (freqs >= bandwidth - 0.1) & (freqs < bandwidth)
        mask[center] = 1.0
        if transition.any():
            t = (freqs[transition] - (bandwidth - 0.1)) / 0.1
            mask[transition] = 0.5 * (1 + np.cos(np.pi * t))
    elif mode == "gaussian":
        mask = np.exp(-8.0 * (freqs / bandwidth) ** 2)
    elif mode == "flat":
        mask = (freqs < bandwidth).astype(float)
    else:
        mask = np.ones_like(freqs)

    # Compute a as inverse FFT of sqrt(mask) (approximately)
    # We want |A(f)|^2 ~ mask, so |A(f)| ~ sqrt(mask)
    A = np.sqrt(mask).astype(complex)
    # Make it have zero phase (real, non-negative time domain)
    a = np.fft.irfft(A * N, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, exact KKT solve, support peeling,
    and iterative Newton polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Fast L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a1 = _lbfgs_optimize(parent, deadline, min_time_remaining=900.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # === Phase 2: Newton polish ===
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 700:
        a2, sc2 = _polish_newton(best_seq, min(deadline - 700, time.monotonic() + 120),
                                  max_rounds=80, active_tol=0.0005)
        update_best(a2, sc2, "newton_1")

    # === Phase 3: Exact support optimization (KEY NEW PHASE) ===
    print(f"\n[Phase 3] Exact support optimization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a3, sc3 = _exact_support_optimize(best_seq, min(deadline - 500, time.monotonic() + 300),
                                           max_support=3000)
        update_best(a3, sc3, "exact_support")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a4, sc4 = _subgrad_polish(best_seq, min(deadline - 350, time.monotonic() + 100))
        update_best(a4, sc4, "subgrad")

    # === Phase 5: Another Newton polish after subgradient ===
    print(f"\n[Phase 5] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5, sc5 = _polish_newton(best_seq, min(deadline - 250, time.monotonic() + 80),
                                  max_rounds=60, active_tol=0.0003)
        update_best(a5, sc5, "newton_2")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [n_parent, int(n_parent * 0.75), int(n_parent * 1.25), 4096]
    n_candidates = list(set([x for x in n_candidates if 1024 <= x <= 32768]))
    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 150:
            break
        a_init = _resample_to_n(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "tapered")
        a_opt = _lbfgs_optimize(a_init, deadline, min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.003 and time.monotonic() < deadline - 80:
            a_np, sc_np = _polish_newton(a_opt, min(deadline - 80, time.monotonic() + 40),
                                          max_rounds=40, active_tol=0.0004)
            update_best(a_np, sc_np, f"n_{n_try}_newton")

            if sc_np < best_score + 0.002 and time.monotonic() < deadline - 50:
                a_ex, sc_ex = _exact_support_optimize(a_np, min(deadline - 50, time.monotonic() + 30),
                                                       max_support=2500)
                update_best(a_ex, sc_ex, f"n_{n_try}_exact")

    # === Phase 7: Spectral initialization ===
    print(f"\n[Phase 7] Spectral init (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        for mode in ["raised_cosine", "gaussian"]:
            if time.monotonic() > deadline - 60:
                break
            n_spec = n_parent
            a_spec = _spectral_init(n_spec, mode=mode, bandwidth=0.4)
            a_spec_opt = _lbfgs_optimize(a_spec, deadline, min_time_remaining=40.0)
            sc_spec = _score(a_spec_opt)
            update_best(a_spec_opt, sc_spec, f"spec_{mode}")

            if sc_spec < best_score + 0.01 and time.monotonic() < deadline - 30:
                a_spec_np, sc_spec_np = _polish_newton(a_spec_opt, min(deadline - 30, time.monotonic() + 25),
                                                        max_rounds=30, active_tol=0.0004)
                update_best(a_spec_np, sc_spec_np, f"spec_{mode}_newton")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_inits = []
        # Small perturbation
        z_p = np.log(np.maximum(best_seq, 1e-30)) + 0.02 * rng.standard_normal(len(best_seq))
        restart_inits.append(_softmax(z_p - z_p.mean()))
        # Larger perturbation
        z_p2 = np.log(np.maximum(best_seq, 1e-30)) + 0.05 * rng.standard_normal(len(best_seq))
        restart_inits.append(_softmax(z_p2 - z_p2.mean()))
        # Different profile
        restart_inits.append(_make_profile(len(best_seq), "steep"))
        # Flat taper
        restart_inits.append(_make_profile(len(best_seq), "flat_taper"))

        for i, a_r_init in enumerate(restart_inits):
            if time.monotonic() > deadline - 20:
                break
            a_r = _lbfgs_optimize(a_r_init, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.002 and time.monotonic() < deadline - 10:
                a_r_np, sc_r_np = _polish_newton(a_r, min(deadline - 10, time.monotonic() + 15),
                                                  max_rounds=30, active_tol=0.0004)
                update_best(a_r_np, sc_r_np, f"restart_{i}_np")

    # === Phase 9: Final ultra-tight polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight Newton
        a_fin, sc_fin = _polish_newton(best_seq, deadline - 3, max_rounds=200, active_tol=0.0001)
        update_best(a_fin, sc_fin, "final_newton")

        # Exact solve if support is small
        if time.monotonic() < deadline - 3:
            a_ex, sc_ex = _exact_support_optimize(best_seq, deadline - 2, max_support=4000)
            update_best(a_ex, sc_ex, "final_exact")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
