import scipy.optimize
import time

def convolve_matrix(a, n_row=None, n_col=None):
    """Build the matrix C such that C @ b = convolve(a, b)"""
    n = len(a)
    if n_row is None:
        n_row = 2*n - 1
    if n_col is None:
        n_col = n
    C = np.zeros((n_row, n_col))
    for k in range(n_row):
        for j in range(n_col):
            idx = k - j
            if 0 <= idx < n:
                C[k, j] = a[idx]
    return C

**Key insight for cheap evaluation:**

new_max ≈ max(old_max, max_k (c[k] + 2*delta*a[k-m])) (for small delta)

This is much cheaper to evaluate! We just need to find the max of c[k] + 2*delta*a[k-m] over k, which is the max of c + 2*delta*shifted_a.

But this is still O(n) per evaluation, and we need O(log(1/epsilon)) evaluations for golden section. So per coordinate: O(n log(1/epsilon)). For all n coordinates: O(n^2 log(1/epsilon)). For n=8192: O(67M * 13) ≈ 870M. Still too slow.

**Better CD: Only update "active" coordinates**

The max of conv(a,a) is achieved at some positions k*. Only coordinates that contribute to these positions are "active". A coordinate a[m] contributes to conv(a,a)[k] iff 0 <= k-m < n, i.e., m <= k <= m+n-1.

If the max is at position k*, then coordinates a[m] for m in [max(0, k*-n+1), min(k*, n-1)] are "active" for that position.

So we only need to update O(n) coordinates (those active for the current max), not all n.

But that's still O(n) coordinates per CD pass.

**Alternative: Subsampled CD**
- Each pass, randomly sample a subset of coordinates (say 100-1000)
- Do golden section on each
- Repeat many passes

This is stochastic CD and should work well in practice.

"""
Multi-strategy optimizer combining Multiplicative Weights alternating optimization,
exact LP direction finding, FISTA with exact subgradient, subsampled coordinate
descent, and multi-scale exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy import optimize as sci_opt


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _mw_direction(a, M, n_iter=100, eta=1.0, top_k=3):
    """
    Use multiplicative weights to find b approximately maximizing sum(b)
    subject to conv(a,b) <= M, b >= 0.
    Uses top_k constraints per step for stability.
    """
    n = len(a)
    b = np.ones(n) / n
    for it in range(n_iter):
        cb = _fft_convolve(a, b)
        cmax = float(cb.max())
        if cmax < 1e-15:
            break
        # Find top_k violated constraints
        if top_k >= len(cb):
            k_indices = np.argsort(cb)[::-1]
        else:
            k_indices = np.argpartition(cb, -top_k)[-top_k:]
        # Softmax weighting over top_k
        vals = cb[k_indices]
        w_sum = 0.0
        update = np.zeros(n)
        for idx in k_indices:
            # Contribution of this constraint
            lo = max(0, int(idx) - n + 1)
            hi = min(int(idx), n - 1)
            idx_start = int(idx) - lo
            idx_end = int(idx) - hi
            segment = a[idx_start:idx_end - 1:-1] if idx_end >= 0 else a[idx_start::-1]
            # Weight by violation level
            w = cb[idx] / cmax
            update[lo:hi + 1] += w * segment
        # Normalize update
        update_norm = float(update.sum())
        if update_norm < 1e-30:
            break
        update = update / update_norm
        # Exponential update
        effective_eta = eta * (1.0 + it / n_iter)
        b *= np.exp(-effective_eta * update)
        b_sum = float(b.sum())
        if b_sum < 1e-300 or not np.isfinite(b_sum):
            b = np.ones(n) / n
        else:
            b /= b_sum
    return b


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=3, decay=0.6,
                  eval_interval=200, clip_norm=0.8, use_backtrack=True):
    """Adam in log-space with smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        # Choose gradient based on temperature
        if T < T_start * 0.01:
            # Near-exact: use top-k weighted
            k_indices = np.argsort(c)[::-1][:active_k]
            p = np.zeros_like(c)
            for i, idx in enumerate(k_indices):
                p[idx] = decay ** i
            p_sum = p.sum()
            if p_sum < 1e-300:
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p /= p_sum
        else:
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 50 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 4 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fista_exact(a_init, deadline, max_iter=100000, step_init=0.01):
    """
    FISTA-style accelerated method using exact subgradient of max(conv(a,a)).
    Operates on the simplex.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    y = a.copy()
    t_k = 1.0
    step = step_init

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        # Find argmax positions
        k_star = int(np.argmax(c))
        # Compute exact subgradient at y
        # grad[m] for the term 2n * conv(y,y)[k_star]
        # d/d a[m] conv(y,y)[k] = 2*y[k-m] (if 0<=k-m<n and k!=2m) or y[m] (if k=2m)
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        idx_start = k_star - lo
        idx_end = k_star - hi
        segment = y[idx_start:idx_end - 1:-1] if idx_end >= 0 else y[idx_start::-1]
        grad[lo:hi + 1] = 2.0 * segment
        if k_star % 2 == 0:
            j_half = k_star // 2
            if 0 <= j_half < n:
                grad[j_half] = y[j_half]  # not 2*y[j_half]
        grad *= 2.0 * n  # scale by objective coefficient

        # Gradient step with step size
        a_new = y - step * grad
        a_new = _project_simplex(a_new)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new

        # Adaptive step size
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            step = min(step * 1.1, 0.5)
        elif t % 500 == 0 and sc > best_score * 1.001:
            step *= 0.7

    return best_a, best_score

                sc_cand = _score_norm(a_cand)
                if sc_cand < best_local:
                    best_local = sc_cand
                    best_local_a = a_cand.copy()

            if best_local < best_score * (1.0 - 1e-10):
                best_score = best_local
                best_a = best_local_a
                c = _fft_convolve(best_a, best_a)
                c_max = float(c.max())
                improved = True

        if not improved:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-14)
    return p_new / p_new.sum()


def _lp_direction(a, M, max_time=30.0):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the LP solution b (normalized to sum=1) or None if too slow.
    """
    n = len(a)
    t0 = time.monotonic()

    # Build constraint matrix: C[k,j] = a[k-j] if 0<=k-j<n else 0
    # Size: (2n-1) x n
    # Use the structure: column j has a[0..n-1] at rows j..j+n-1
    # For efficiency, keep only "active" constraints
    c = _fft_convolve(a, a)
    threshold = M * 0.5
    active_constraints = np.where(c > threshold)[0]

    if len(active_constraints) == 0:
        return a.copy()

    # Build A matrix (active constraints x n)
    rows_a = []
    rows_b = []
    for k in active_constraints:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx_start = k - lo
        idx_end = k - hi
        segment = a[idx_start:idx_end - 1:-1] if idx_end >= 0 else a[idx_start::-1]
        # This row has non-zeros at columns lo..hi
        for jj in range(hi - lo + 1):
            rows_a.append(k)
            rows_b.append(lo + jj)
            rows_a_val = segment[jj]

    # Actually, let's build it properly using scipy sparse
    from scipy import sparse

    row_idx = []
    col_idx = []
    data = []
    for k in active_constraints:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for jj in range(lo, hi + 1):
            val = a[k - jj]
            if val > 0:
                row_idx.append(k)
                col_idx.append(jj)
                data.append(val)

    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    # Only keep active rows
    A_active = A[active_constraints, :]
    b_vec = np.full(len(active_constraints), M)

    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)

    try:
        result = sci_opt.linprog(
            c_obj,
            A_ub=A_active,
            b_ub=b_vec,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max(5, max_time - (time.monotonic() - t0)), 'presolve': True}
        )
        if result.success and result.x is not None:
            b_sol = np.maximum(result.x, 0)
            b_sum = float(b_sol.sum())
            if b_sum > 1e-15:
                b_sol /= b_sum
                return b_sol
    except Exception:
        pass

    return None


def _mw_alternating(a, deadline, n_outer=50, n_mw=50, eta_init=1.0, t_init=0.3):
    """
    Alternating MW optimization:
    - Fix a, find b via MW
    - Blend a toward b
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    M = float(c.max())

    for outer in range(n_outer):
        if time.monotonic() > deadline:
            break
        # Adaptive parameters
        frac = outer / max(1, n_outer)
        eta = eta_init * (0.5 + 0.5 * (1 - frac))
        t_blend = t_init * (0.3 + 0.7 * (1 - frac))

        # Find direction b
        b = _mw_direction(a, M, n_iter=n_mw, eta=eta, top_k=3)
        b = b / b.sum()

        # Blend
        a_new = (1 - t_blend) * a + t_blend * b
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            c = _fft_convolve(a, a)
            M = float(c.max())
        else:
            # Try smaller blend
            for t_try in [t_blend * 0.5, t_blend * 0.1]:
                a_try = (1 - t_try) * a + t_try * b
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc_try = _score_norm(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    c = _fft_convolve(a, a)
                    M = float(c.max())
                    break

    return best_a, best_score


def _adam_refine(a_init, deadline, active_k=3):
    """Quick Adam refinement with very low temperature."""
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    return _adam_z_space(
        a_init, deadline,
        T_start=0.001 * max_c, T_end=0.00001 * max_c,
        lr_start=0.02, lr_end=0.001,
        max_iter=200000, active_k=active_k,
        eval_interval=100, use_backtrack=True
    )


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. MW alternating optimization
    2. Adam log-space refinement
    3. FISTA exact subgradient
    4. LP direction finding
    5. Coordinate descent polishing
    6. Multi-scale exploration (n=4096, 8192)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n_parent}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: MW Alternating (0-200s) ===
    phase1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < phase1_end:
        print("Phase 1: MW Alternating")
        a_mw, sc_mw = _mw_alternating(best_a, phase1_end,
                                       n_outer=80, n_mw=40,
                                       eta_init=1.5, t_init=0.4)
        full_sc_mw = _score(a_mw)
        print(f"  Phase 1: score={full_sc_mw:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_mw) and full_sc_mw < best_score:
            best_score = full_sc_mw
            best_a = a_mw.copy()

    # === Phase 2: Adam refinement (200-500s) ===
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end:
        print("Phase 2: Adam Refinement")
        a_adam, sc_adam = _adam_refine(best_a, phase2_end)
        full_sc_adam = _score(a_adam)
        print(f"  Phase 2: score={full_sc_adam:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_adam) and full_sc_adam < best_score:
            best_score = full_sc_adam
            best_a = a_adam.copy()

    # === Phase 3: LP Direction (500-700s) ===
    phase3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase3_end:
        print("Phase 3: LP Direction Finding")
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        M_best = float(c_best.max())
        for lp_iter in range(5):
            if time.monotonic() > phase3_end - 10:
                break
            b_lp = _lp_direction(best_a, M_best, max_time=30.0)
            if b_lp is not None:
                # Blend with decreasing step
                for t_blend in [0.5, 0.3, 0.15, 0.05]:
                    a_cand = (1 - t_blend) * best_a + t_blend * b_lp
                    a_cand = np.maximum(a_cand, 0)
                    a_cand = a_cand / a_cand.sum()
                    sc_cand = _score_norm(a_cand)
                    if sc_cand < _score(best_a):
                        best_a = a_cand.copy()
                        break
            else:
                break
        full_sc_lp = _score(best_a)
        print(f"  Phase 3: score={full_sc_lp:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_lp) and full_sc_lp < best_score:
            best_score = full_sc_lp

    # === Phase 4: FISTA exact (700-900s) ===
    phase4_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase4_end:
        print("Phase 4: FISTA Exact Subgradient")
        a_fista, sc_fista = _fista_exact(best_a, phase4_end, max_iter=50000, step_init=0.005)
        full_sc_fista = _score(a_fista)
        print(f"  Phase 4: score={full_sc_fista:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_fista) and full_sc_fista < best_score:
            best_score = full_sc_fista
            best_a = a_fista.copy()

    # === Phase 5: CD Polish (900-1100s) ===
    phase5_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase5_end:
        print("Phase 5: Coordinate Descent Polish")
        a_cd, sc_cd = _cd_polish(best_a, phase5_end,
                                  max_rounds=300, samples_per_round=1000, rng=rng)
        full_sc_cd = _score(a_cd)
        print(f"  Phase 5: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # === Phase 6: Try n=4096 (1100-1400s) ===
    phase6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase6_end:
        print("Phase 6: Exploring n=4096")
        a_4k = _interp_to_length(best_a, 4096)
        sc_4k_init = _score(a_4k)
        print(f"  Interpolated n=4096: score={sc_4k_init:.10f}")

        # MW at 4096
        mid_dl = min(phase6_end, t_start + 1250.0)
        a_4k_mw, sc_4k_mw = _mw_alternating(a_4k, mid_dl,
                                              n_outer=50, n_mw=50,
                                              eta_init=1.5, t_init=0.4)
        full_sc_4k_mw = _score(a_4k_mw)
        print(f"  n=4096 MW: score={full_sc_4k_mw:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_4k_mw) and full_sc_4k_mw < best_score:
            best_score = full_sc_4k_mw
            best_a = a_4k_mw.copy()

        # Adam at 4096
        if time.monotonic() < phase6_end - 30:
            a_4k_adam, sc_4k_adam = _adam_refine(best_a if len(best_a) == 4096 else a_4k_mw,
                                                  phase6_end - 30)
            full_sc_4k_adam = _score(a_4k_adam)
            print(f"  n=4096 Adam: score={full_sc_4k_adam:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_4k_adam) and full_sc_4k_adam < best_score:
                best_score = full_sc_4k_adam
                best_a = a_4k_adam.copy()

        # CD at 4096
        if time.monotonic() < phase6_end - 5:
            a_4k_cd, sc_4k_cd = _cd_polish(best_a, phase6_end - 3,
                                            max_rounds=100, samples_per_round=500, rng=rng)
            full_sc_4k_cd = _score(a_4k_cd)
            if np.isfinite(full_sc_4k_cd) and full_sc_4k_cd < best_score:
                best_score = full_sc_4k_cd
                best_a = a_4k_cd.copy()
                print(f"  n=4096 CD: score={full_sc_4k_cd:.10f}")

    # === Phase 7: Back to n=8192 if it was better (1400-1600s) ===
    phase7_end = min(deadline, t_start + 1600.0)
    n_best = len(best_a)
    if n_best == 4096 and time.monotonic() < phase7_end:
        print("Phase 7: Upsample 4096 to 8192 and refine")
        a_8k = _interp_to_length(best_a, 8192)
        sc_8k_init = _score(a_8k)
        print(f"  Upsampled n=8192: score={sc_8k_init:.10f}")
        if sc_8k_init < best_score:
            best_a = a_8k
            best_score = sc_8k_init

        a_8k_adam, sc_8k_adam = _adam_refine(best_a, phase7_end)
        full_sc_8k = _score(a_8k_adam)
        if full_sc_8k < best_score:
            best_score = full_sc_8k
            best_a = a_8k_adam
        print(f"  Phase 7: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 8: Final perturbation + refine (1600-1750s) ===
    phase8_end = min(deadline, t_start + 1750.0)
    n_fin = len(best_a)
    if time.monotonic() < phase8_end:
        print("Phase 8: Perturbation Restarts")
        for attempt in range(5):
            if time.monotonic() > phase8_end - 30:
                break
            sigma = 0.01 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            # Quick refinement
            time_per = max(15.0, (phase8_end - time.monotonic()) / (5 - attempt))
            sub_dl = min(phase8_end - 5, time.monotonic() + time_per)
            a_pert_opt, sc_pert_opt = _adam_refine(a_pert, sub_dl)
            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc_pert:.10f}")
            if np.isfinite(full_sc_pert) and full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

    # === Phase 9: Ultra-fine final polish (1750-deadline) ===
    if time.monotonic() < deadline - 5:
        print("Phase 9: Final Ultra-fine Polish")
        n_fin = len(best_a)
        a_final_cd, sc_final_cd = _cd_polish(best_a, deadline - 8,
                                              max_rounds=200, samples_per_round=2000, rng=rng)
        full_sc_final_cd = _score(a_final_cd)
        if np.isfinite(full_sc_final_cd) and full_sc_final_cd < best_score:
            best_score = full_sc_final_cd
            best_a = a_final_cd.copy()

        # One more Adam pass with extremely low temperature
        if time.monotonic() < deadline - 3:
            c_fin = _fft_convolve(best_a, best_a)
            max_cf = float(c_fin.max())
            a_final, sc_final = _adam_z_space(
                best_a, deadline - 2,
                T_start=0.0001 * max_cf, T_end=0.000005 * max_cf,
                lr_start=0.003, lr_end=0.0003,
                max_iter=100000, active_k=2,
                eval_interval=50, use_backtrack=True
            )
            full_sc_final = _score(a_final)
            if np.isfinite(full_sc_final) and full_sc_final < best_score:
                best_score = full_sc_final
                best_a = a_final.copy()

    # Final
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
