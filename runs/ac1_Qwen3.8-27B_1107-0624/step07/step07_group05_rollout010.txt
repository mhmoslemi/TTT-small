import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _pressure(a, active):
    """Compute pressure p_i = sum_{k in active} a_{k-i}."""
    n = len(a)
    a_rev = a[::-1]
    p_full = _full_convolve(active.astype(float), a_rev)
    return p_full[n - 1:2 * n - 1]


def _water_fill_update(a, active, n_iter=50, relax=0.5):
    """
    Water-filling pressure equalization.
    Iteratively: compute pressure, apply water-filling, normalize.
    """
    n = len(a)
    a = a.copy().astype(float)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for it in range(n_iter):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        p = _pressure(a, active)
        p = np.maximum(p, 1e-30)

        # Water-filling: a_i proportional to (p_i - lambda)_+
        # Find lambda such that sum max(0, p_i - lambda) = 1
        p_sorted = np.sort(p)[::-1]
        cumsum = np.cumsum(p_sorted)
        n_nonzero = np.searchsorted(-cumsum + p_sorted, -1.0, side='left')
        if n_nonzero == 0:
            n_nonzero = 1
        n_nonzero = min(n_nonzero, n)
        if n_nonzero >= n:
            lambda_wf = 0.0
        else:
            lambda_wf = (cumsum[n_nonzero - 1] - 1.0) / n_nonzero

        a_wf = np.maximum(p - lambda_wf, 0.0)
        s_wf = a_wf.sum()
        if s_wf < 1e-30:
            break
        a_wf = a_wf / s_wf

        # Relaxed update
        a = (1.0 - relax) * a + relax * a_wf

    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _pressure_equalize_v2(a_init, deadline, method="waterfill", gamma=0.4, n_active_frac=0.05, relax=0.5):

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_direct(z, n, top_k=20):
    """Direct gradient using top-k lags (no smoothing)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    # Use top-k lags with equal weight
    idx = np.argsort(c)[-top_k:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / top_k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

            method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-12}
        )
        a_new = np.maximum(result.x, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            return a_new
    except Exception:
        pass
    return a_init


def _subgrad_direct(a_init, deadline, n_max_steps=3000, lr_init=0.02, top_k=15):
    """Subgradient descent using direct top-k gradient (no smoothing)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        grad = _grad_direct(z, n, top_k=top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lp_direction(a, active_idx):
    """
    Compute LP-based search direction.
    Solves: min d s.t. [grad_k, -1] @ [da, d] <= b_k(a*) - M*  (approximately)
    Simplified: compute the direction that most reduces the active lags.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Gradient of b_k at a: grad_k[j] ≈ 2*a[k-j] (for valid k-j)
    # Direction: da_j = -grad_max[j] where grad_max = sum_{k in S} grad_k
    # This is essentially -pressure
    p = _pressure(a, active_idx.astype(float))
    # Center the direction (to maintain sum)
    p_centered = p - p.mean()
    return -p_centered


def _directional_search(a, direction, deadline, n_trials=20):
    """Line search along a direction."""
    best_a = a.copy()
    best_sc = _score_norm(a)
    t_vals = np.geomspace(0.001, 0.5, n_trials)

    for t in t_vals:
        if time.monotonic() > deadline:
            break
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = _score_norm(a_try)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_try.copy()

    return best_a, best_sc


def _make_tapered(n, alpha=0.5):
    """Create a tapered initial sequence."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** alpha
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_modified(n, n_cos=3, amp=0.1):
    """Create a sequence with cosine modulation."""
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for k in range(1, n_cos + 1):
        a += amp / k * np.cos(2 * np.pi * k * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer focusing on:
    1. Water-filling pressure equalization
    2. Aggressive L-BFGS with extended beta continuation
    3. Direct subgradient with top-k gradient
    4. LP-based directional search
    5. Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  ** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 8192
        parent = _make_tapered(n_parent, alpha=0.7)
        print(f"No parent, creating tapered n={n_parent}, score={_score(parent):.10f}")
        best_seq = parent.copy()
        best_score = _score(parent)

    # === PHASE 1: Pressure Equalization (Water-Filling) ===
    print(f"\n[Phase 1] Water-filling pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    a1 = best_seq.copy()
    s1 = a1.sum()
    a1 = a1 / s1

    # Try multiple gamma/fract combinations
    for n_frac in [0.01, 0.02, 0.05, 0.10]:
        for relax in [0.3, 0.5, 0.7]:
            if time.monotonic() > deadline - 400:
                break
            a_try, sc_try = _pressure_equalize_v2(
                a1, min(deadline - 350, time.monotonic() + 30),
                method="waterfill", n_active_frac=n_frac, relax=relax
            )
            _try_update(a_try, sc_try)
            if sc_try < _score(a1):
                a1 = a_try

    # Multiplicative variant
    for gamma in [0.2, 0.35, 0.5, 0.7]:
        if time.monotonic() > deadline - 350:
            break
        a_try, sc_try = _pressure_equalize_v2(
            a1, min(deadline - 350, time.monotonic() + 25),
            method="multiplicative", gamma=gamma, n_active_frac=0.03
        )
        _try_update(a_try, sc_try)
        if sc_try < _score(a1):
            a1 = a_try

    print(f"  Phase 1 best: {best_score:.10f}")

    # === PHASE 2: L-BFGS ===
    print(f"\n[Phase 2] L-BFGS optimization (t={time.monotonic()-t_start:.1f}s)")
    a_start = best_seq.copy()
    s = a_start.sum()
    a_start = a_start / s

    a2 = _run_lbfgs_v2(a_start, deadline, min_remaining=300)
    _try_update(a2, _score(a2))
    print(f"  Phase 2 best: {best_score:.10f}")

    # === PHASE 3: Subgradient with direct top-k ===
    print(f"\n[Phase 3] Direct subgradient (t={time.monotonic()-t_start:.1f}s)")
    a3, sc3 = _subgrad_direct(best_seq, min(deadline - 250, time.monotonic() + 60),
                               n_max_steps=2000, lr_init=0.015, top_k=10)
    _try_update(a3, sc3)

    a3b, sc3b = _subgrad_direct(best_seq, min(deadline - 200, time.monotonic() + 60),
                                 n_max_steps=2000, lr_init=0.01, top_k=20)
    _try_update(a3b, sc3b)
    print(f"  Phase 3 best: {best_score:.10f}")

    # === PHASE 4: LP-based directional search ===
    print(f"\n[Phase 4] LP directional search (t={time.monotonic()-t_start:.1f}s)")
    a_dir = best_seq.copy()
    s = a_dir.sum()
    a_dir = a_dir / s

    for iteration in range(30):
        if time.monotonic() > deadline - 150:
            break
        c = _full_convolve(a_dir, a_dir)
        c_max = float(c.max())
        # Active set: top lags
        n_top = max(5, min(50, int(0.005 * len(c))))
        idx_top = np.argsort(c)[-n_top:]
        active = np.zeros_like(c)
        active[idx_top] = 1.0

        direction = _lp_direction(a_dir, active)
        # Normalize direction
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        a_new, sc_new = _directional_search(a_dir, direction, deadline - 100)
        if sc_new < _score_norm(a_dir) - 1e-10:
            a_dir = a_new
            _try_update(a_dir, sc_new)
        else:
            break

    print(f"  Phase 4 best: {best_score:.10f}")

    # === PHASE 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != len(best_seq)]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - 80 - time.monotonic()
        if remaining < 30:
            break

        # Resample from best
        n_best = len(best_seq)
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, best_seq)
        a_init = np.maximum(a_init, 1e-12)
        a_init = a_init / a_init.sum()

        # Quick pressure equalization
        a_opt, sc_opt = _pressure_equalize_v2(
            a_init, min(deadline - 60, time.monotonic() + 40),
            method="waterfill", n_active_frac=0.03, relax=0.5
        )
        _try_update(a_opt, sc_opt)

        # Quick L-BFGS if improved
        sc_curr = _score(a_opt)
        if sc_curr < best_score + 0.001 and time.monotonic() < deadline - 40:
            a_lb = _run_lbfgs_v2(a_opt, deadline, min_remaining=40)
            _try_update(a_lb, _score(a_lb))

    print(f"  Phase 5 best: {best_score:.10f}")

    # === PHASE 6: Final intensive polish ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 10:
        # L-BFGS with very small betas
        a_final = _run_lbfgs_v2(best_seq, deadline, min_remaining=5)
        _try_update(a_final, _score(a_final))

    if time.monotonic() < deadline - 2:
        # Tiny subgradient steps
        a_fin2, sc_fin2 = _subgrad_direct(best_seq, deadline - 1,
                                          n_max_steps=500, lr_init=0.003, top_k=5)
        _try_update(a_fin2, sc_fin2)

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_tapered(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
