def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, S, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        x = x + alpha * p
        if rs_new < tol * tol * rs0:
            break
        if rs_new >= rs:
            p = r_new
        else:
            p = (rs_new / rs) * p + r_new
        r = r_new
        rs = rs_new
    return x


def _newton_polish(a_init, deadline, max_rounds=200, active_tol_frac=0.001, min_active=1, max_active=80):
    """Newton polish with adaptive active set."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = active_tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active < min_active:
            idx = np.argsort(c)[::-1][:min_active]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = min_active
        if n_active > max_active:
            top_idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = max_active

        W_vec = np.zeros(2 * n - 1)
        W_vec[active] = 1.0 / n_active

        Ha = _hankel_matvec(a, W_vec, n)
        mu = float(np.mean(Ha))
        e = Ha - mu

        # Solve H*dx = -e with regularization if needed
        reg = 0.0
        if n_active <= 3:
            reg = 1e-8 * c_max
        dx = _cg_solve(W_vec, n, -e, max_iter=300, tol=1e-13, reg=reg)

        # Ensure zero mean step
        dx_mean = float(np.mean(dx))
        dx = dx - dx_mean

        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-15 * n:
            break

        # Backtracking line search
        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try <= c_max - 1e-15 or alpha < 1e-14:
                a = a_try
                step_done = True
                break
            alpha *= 0.5

        if not step_done:
            # Gradient step fallback
            grad = 2.0 * Ha - 2.0 * mu
            g_norm = float(np.linalg.norm(grad))
            if g_norm < 1e-15:
                break
            alpha = 0.01 / g_norm
            for ls in range(30):
                a_try = a - alpha * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _newton_fixed_active(a_init, deadline, active_indices, max_rounds=100):
    """Newton polish with a fixed active set."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    W_vec = np.zeros(2 * n - 1)
    W_vec[active_indices] = 1.0 / len(active_indices)

    # Precompute H*1 for constraint handling
    H1 = _cg_solve(W_vec, n, np.ones(n), max_iter=200)
    H1_sum = float(np.sum(H1))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        Ha = _hankel_matvec(a, W_vec, n)
        # Target: H*a = mu*1 where mu chosen so that sum(a)=1 is consistent
        # At optimum: H*a = mu*1, and mu = sum(H*a)/n (since sum(a)=1)
        mu = float(np.sum(Ha)) / n
        e = Ha - mu

        reg = 1e-10 * c_max if len(active_indices) <= 3 else 0.0
        dx = _cg_solve(W_vec, n, -e, max_iter=300, tol=1e-14, reg=reg)

        # Project to zero-sum: dx = dx - (1^T dx / 1^T H1) * H1
        if H1_sum > 1e-20:
            dx = dx - (float(np.sum(dx)) / H1_sum) * H1

        dx_norm = float(np.linalg.norm(dx))
        if dx_norm < 1e-14 * n:
            break

        alpha = 1.0
        step_done = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try <= c_max - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5

        if not step_done and alpha < 1e-14:
            if dx_norm < 1e-12:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgrad_optimize(a_init, deadline, max_steps=2000):
    """Subgradient method in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.03
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax_local(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        W_vec = np.zeros(2 * n - 1)
        W_vec[active] = 1.0 / active.sum()

        Ha = _hankel_matvec(a, W_vec, n)
        W_val = float(np.dot(W_vec, c))
        h = _full_convolve(W_vec, a[::-1])[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W_val)
        grad = grad - np.mean(grad)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 10 == 0:
            a_check = _softmax_local(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.06)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.02
        z = z_new

    a_final = _softmax_local(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _softmax_local(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _support_reduction(a, deadline, min_remaining=60.0):
    """Try zeroing out smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(best_a)

    fractions = [0.01, 0.03, 0.05, 0.10, 0.15, 0.25]
    for frac in fractions:
        if time.monotonic() + min_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_remaining > deadline:
                break
            # Re-optimize
            a_re, sc_re = _newton_polish(best_a, min(deadline - min_remaining, time.monotonic() + 60), max_rounds=50)
            if sc_re < best_sc:
                best_sc = sc_re
                best_a = a_re

    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing of the max."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.01)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]

    for beta_frac in ratios:
        if time.monotonic() + min_remaining > deadline:
            break
        beta = base_beta * beta_frac
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        max_iter = min(2000, max(20, int(remaining / (0.005 * max(1, n / 4096)))))

        try:
            result = minimize(
                _smooth_obj,
                z,
                jac=_smooth_grad,
                args=(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 40}
            )
            a_new = _softmax_local(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _smooth_obj(z, beta, n):
    a = _softmax_local(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax_local(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W_vec = w
    W_val = float(np.dot(W_vec, c))
    h = _full_convolve(W_vec, a[::-1])[n - 1:2 * n - 1]
    return 4.0 * n * a * (h - W_val)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: Newton active-set polish + L-BFGS smoothing +
    subgradient + support reduction + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        t = np.linspace(0, 1, n_parent)
        parent = np.sin(np.pi * t) ** 0.8
        parent = np.maximum(parent, 1e-10)
        parent = parent / parent.sum()
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS smoothing from parent
    print(f"\n[Phase 1] L-BFGS smoothing (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a1 = _lbfgs_smooth(parent, deadline, min_remaining=400.0)
        update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Newton polish with adaptive active set
    print(f"\n[Phase 2] Newton polish adaptive (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _newton_polish(best_seq, min(deadline - 400, time.monotonic() + 200), max_rounds=150)
        update_best(a2, sc2, "newton_adapt_1")

    # Phase 3: Newton with fixed active sets of various sizes
    print(f"\n[Phase 3] Newton fixed active sets (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        for active_size in [1, 3, 5, 10, 20, 50]:
            if time.monotonic() > deadline - 30:
                break
            top_idx = np.argsort(c)[::-1][:active_size]
            a3, sc3 = _newton_fixed_active(best_seq, min(deadline - 60, time.monotonic() + 80), top_idx, max_rounds=80)
            update_best(a3, sc3, f"newton_fixed_{active_size}")

    # Phase 4: More Newton polish after potential improvements
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a4, sc4 = _newton_polish(best_seq, min(deadline - 200, time.monotonic() + 150), max_rounds=200, active_tol_frac=0.0005)
        update_best(a4, sc4, "newton_adapt_2")

    # Phase 5: Subgradient from current best
    print(f"\n[Phase 5] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a5, sc5 = _subgrad_optimize(best_seq, min(deadline - 150, time.monotonic() + 80), max_steps=1000)
        update_best(a5, sc5, "subgrad_1")

    # Phase 6: Support reduction
    print(f"\n[Phase 6] Support reduction (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a6, sc6 = _support_reduction(best_seq, deadline, min_remaining=80.0)
        update_best(a6, sc6, "support_red")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 4096, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _newton_polish(a_init, min(deadline - 40, time.monotonic() + 60), max_rounds=60)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 8: Final tight Newton polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_polish(best_seq, deadline - 2, max_rounds=300, active_tol_frac=0.0001, min_active=1, max_active=20)
        update_best(a_fin, sc_fin, "final_newton")

    # Phase 9: Ultra-final L-BFGS with tiny beta
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_ultra = _lbfgs_smooth(best_seq, deadline, min_remaining=2.0)
        update_best(a_ultra, _score(a_ultra), "ultra_lbfgs")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        # Normalize for good measure (score is scale-invariant)
        sc_final = _score(final)
        val_sc = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: score={sc_final:.10f}, val_score={val_sc:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
