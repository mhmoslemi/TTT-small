    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Multi-scale initialization (1024→16384) for better starting points
- Adam with exact subgradient for fine-tuning
- Aggressive L-BFGS with near-zero beta
- Coordinate descent for exact 1D optimization per element
- Sparsification with re-optimization
- Multi-n exploration with 16384 as primary

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = float(c.max())

    return 4.0 * n * a * (h - W)


def _exact_subgrad(a):
    """Compute exact subgradient of 2n*max(conv(a,a))/sum(a)^2 w.r.t. log(a)."""
    n = len(a)
    s = float(a.sum())
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Subgradient: pick the max element (or average of near-max)
    tol = c_max * 0.001
    active = c >= (c_max - tol)
    n_act = int(active.sum())
    if n_act == 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
    # d/dz[i] of 2n*max(C)/s^2 where a=softmax(z)
    # = 2n * (1/s^2) * d(max C)/da[i] * a[i] - 2n * max(C)/s^3 * s * a[i] (for softmax constraint)

    # Then grad_z = a * grad_a (chain rule for log/softmax)

    wc = _full_convolve(w, a_rev)

    W = float(np.dot(w, c))

    return grad_z, c_max

    max_c = float(c_cur.max())
    if max_c < 1e-15:

        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

def _adam_finetune(a_init, deadline, min_time_remaining=30.0):
    """Adam optimizer with exact subgradient for fine-tuning."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    t = 0
    best_a = a_init.copy()
    best_sc = _score(best_a)
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        # Compute exact subgradient
        a_cur = _softmax(z)
        grad_z, c_max = _exact_subgrad(a_cur)
        
        # Adaptive learning rate
        lr_cur = lr / (t ** 0.5)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()  # Renormalize

        if t % 50 == 0:
            a_new = _softmax(z)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 50
            if no_improve >= max_no_improve:
                # Reduce learning rate
                lr *= 0.5
                no_improve = 0
                if lr < 1e-6:
                    break

    return best_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.005,
                       tol_frac_end=0.0001, lr_init=0.03):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _coordinate_descent(a_init, deadline, min_time_remaining=30.0):
    """Exact coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    C = _full_convolve(a, a)

    # Shuffle coordinates for stochastic coordinate descent
    order = np.arange(n)

    max_rounds = 20
    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        np.random.shuffle(order)
        
        for j in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            old = a[j]
            if old < 1e-15:
                continue

            # Compute the 1D objective: g(x) = 2n * M(x) / (s - old + x)^2
            # M(x) = max of affected C values
            # Affected k's: k in [j, j+n-1] where 0 <= k-j < n
            # C_new[2j] = C[2j] - old^2 + x^2
            # C_new[k] = C[k] - 2*old*a[k-j] + 2*x*a[k-j] for k in [j,j+n-1], k!=2j
            # C_new[k] = C[k] for other k

            k_start = j
            k_end = min(j + n, 2 * n - 1)
            
            # Precompute base values (C with a[j]=0)
            affected = np.arange(k_start, k_end)
            if len(affected) == 0:
                continue
            offsets = affected - j  # 0 to n-1
            a_offsets = a[offsets]
            
            # C_base for affected k's
            C_aff = C[affected].copy()
            # Remove old contribution
            mask_quad = (affected == 2 * j)
            C_aff[mask_quad] -= old * old
            C_aff[~mask_quad] -= 2.0 * old * a_offsets[~mask_quad]
            
            # Unaffected max
            mask_affected = np.zeros(2 * n - 1, dtype=bool)
            mask_affected[affected] = True
            C_far_max = float(C[~mask_affected].max()) if (~mask_affected).any() else 0.0
            
            # Now find x minimizing 2n * max(C_base + 2x*a_off (excluding quad), C_base_quad + x^2, C_far_max) / (s - old + x)^2
            # Use golden section search
            x_lo = 0.0
            x_hi = min(3.0 * old + 0.01, 1.0)  # Upper bound
            
            def g_1d(x):
                denom = s - old + x
                if denom < 1e-15:
                    return np.inf
                # Affected linear terms
                vals = C_aff[~mask_quad] + 2.0 * x * a_offsets[~mask_quad]
                m_aff = float(vals.max()) if len(vals) > 0 else -np.inf
                # Quadratic term
                val_quad = C_aff[mask_quad][0] + x * x if mask_quad.any() else -np.inf
                m = max(m_aff, val_quad, C_far_max)
                return 2.0 * n * m / (denom * denom)

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x1 = x_hi - gr * (x_hi - x_lo)
            x2 = x_lo + gr * (x_hi - x_lo)
            f1 = g_1d(x1)
            f2 = g_1d(x2)
            
            for _ in range(40):
                if f1 < f2:
                    x_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = x_hi - gr * (x_hi - x_lo)
                    f1 = g_1d(x1)
                else:
                    x_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = x_lo + gr * (x_hi - x_lo)
                    f2 = g_1d(x2)
            
            x_opt = (x_lo + x_hi) / 2.0
            f_opt = g_1d(x_opt)
            
            # Also check x=0
            f_zero = g_1d(0.0)
            if f_zero < f_opt:
                x_opt = 0.0
                f_opt = f_zero
            
            if f_opt < best_sc - 1e-14:
                a[j] = x_opt
                s = s - old + x_opt
                best_a = a.copy()
                best_sc = f_opt
            else:
                a[j] = x_opt  # Still update (might help other coordinates)
                s = s - old + x_opt
    
    # Final normalization
    s_final = a.sum()
    if s_final > 1e-15:
        a = a / s_final
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_and_reopt(a, deadline, min_time_remaining=60.0):
    """Zero out small elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline - min_time_remaining:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            # Re-optimize after sparsification
            a_reopt = _run_lbfgs(a_try, deadline, min_time_remaining=min_time_remaining / 2)
            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt.copy()
            else:
                best_sc = sc_try
                best_a = a_try.copy()
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search with multi-scale init, L-BFGS, Adam, coordinate descent, active-set."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        parent = _make_profile(4096, "uniform")
        parent_sc = _score(parent)
        best_seq = parent.copy()
        best_score = parent_sc
        print(f"No parent, uniform n=4096, score={best_score:.10f}")

    # === Phase 1: Multi-scale initialization ===
    print(f"\n[P1] Multi-scale init (t={time.monotonic()-t_start:.1f}s)")
    target_n = 16384
    scales = [1024, 2048, 4096, 8192, 16384]
    
    # Start from parent or uniform, upsample through scales
    current = parent.copy() if parent is not None else _make_profile(1024, "uniform")
    current_n = len(current)
    
    for scale_n in scales:
        if time.monotonic() > deadline - 800:
            break
        if scale_n <= current_n:
            continue
        # Upsample
        current = _resample(current, scale_n)
        # Quick L-BFGS at this scale
        alloc = min(60.0, max(10.0, 800.0 * (scale_n / target_n)))
        scale_deadline = time.monotonic() + alloc
        current = _run_lbfgs(current, scale_deadline, min_time_remaining=5.0)
        current_n = scale_n
        sc = _score(current)
        print(f"  scale={scale_n}, score={sc:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc < best_score:
            best_score = sc
            best_seq = current.copy()

    # Use the multi-scale result as primary candidate
    a_primary = current.copy() if current_n >= 8192 else _resample(best_seq, target_n)
    n_primary = len(a_primary)
    sc_primary = _score(a_primary)
    print(f"  Primary: n={n_primary}, score={sc_primary:.10f}")
    if sc_primary < best_score:
        best_score = sc_primary
        best_seq = a_primary.copy()

    # === Phase 2: Full L-BFGS on primary ===
    print(f"\n[P2] L-BFGS main (n={n_primary}) (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, time.monotonic() + 500)
    a_p2 = _run_lbfgs(a_primary, p2_deadline, min_time_remaining=100.0)
    sc_p2 = _score(a_p2)
    print(f"  L-BFGS result: {sc_p2:.10f}, time={time.monotonic()-t_start:.1f}s")
    if sc_p2 < best_score:
        best_score = sc_p2
        best_seq = a_p2.copy()
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 3: Adam fine-tuning ===
    print(f"\n[P3] Adam fine-tune (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        p3_deadline = min(deadline - 250, time.monotonic() + 150)
        a_p3 = _adam_finetune(best_seq, p3_deadline, min_time_remaining=20.0)
        sc_p3 = _score(a_p3)
        print(f"  Adam result: {sc_p3:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_p3 < best_score:
            best_score = sc_p3
            best_seq = a_p3.copy()
            print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n[P4] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p4_deadline = min(deadline - 150, time.monotonic() + 120)
        # Coordinate descent on a medium-sized version for speed
        if len(best_seq) > 4096:
            a_cd_init = _resample(best_seq, 4096)
        else:
            a_cd_init = best_seq.copy()
        a_cd, sc_cd = _coordinate_descent(a_cd_init, p4_deadline, min_time_remaining=10.0)
        print(f"  CD result (n={len(a_cd)}): {sc_cd:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_cd < best_score:
            # Upsample back and re-optimize
            a_cd_up = _resample(a_cd, len(best_seq))
            a_cd_up = _run_lbfgs(a_cd_up, min(deadline - 100, time.monotonic() + 60), min_time_remaining=10.0)
            sc_cd_up = _score(a_cd_up)
            if sc_cd_up < best_score:
                best_score = sc_cd_up
                best_seq = a_cd_up.copy()
                print(f"  NEW BEST (CD upscaled): {best_score:.10f}")
        if time.monotonic() < deadline - 100:
            # Also try CD directly on the best sequence (fewer rounds)
            a_cd2_deadline = min(deadline - 100, time.monotonic() + 60)
            a_cd2, sc_cd2 = _coordinate_descent(best_seq, a_cd2_deadline, min_time_remaining=5.0)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST (CD direct): {best_score:.10f}")

    # === Phase 5: Active-set polish ===
    print(f"\n[P5] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        p5_deadline = min(deadline - 80, time.monotonic() + 80)
        a_p5, sc_p5 = _active_set_polish(best_seq, p5_deadline,
                                         tol_frac_start=0.003, tol_frac_end=0.00005,
                                         lr_init=0.02)
        if sc_p5 < best_score:
            best_score = sc_p5
            best_seq = a_p5.copy()
            print(f"  NEW BEST: {best_score:.10f} [polish]")

    # === Phase 6: Sparsification ===
    print(f"\n[P6] Sparsification (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_sp, sc_sp = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=30.0)
        if sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()
            print(f"  NEW BEST: {best_score:.10f} [sparsify]")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_current = len(best_seq)
    n_candidates = [8192, 12288, 16384, 20480]
    # Add current n if not in list
    if n_current not in n_candidates:
        n_candidates.append(n_current)
    n_candidates.sort()

    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if n_try == n_current:
            continue
        a_mn = _resample(best_seq, n_try)
        alloc = min(60.0, max(15.0, (deadline - 50 - time.monotonic()) * 0.4))
        a_mn_opt = _run_lbfgs(a_mn, time.monotonic() + alloc, min_time_remaining=5.0)
        sc_mn = _score(a_mn_opt)
        if sc_mn < best_score:
            best_score = sc_mn
            best_seq = a_mn_opt.copy()
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Diverse restarts ===
    print(f"\n[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        restart_starts = [
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("pert10", _perturb(best_seq, 0.10, rng)),
            ("taper", _resample(_make_profile(n_best, "tapered"), n_best)),
            ("cos", _make_profile(n_best, "cos_mod")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 15:
                break
            alloc = min(30.0, deadline - 15 - time.monotonic())
            if alloc < 10:
                break
            a_opt = _run_lbfgs(a_r, time.monotonic() + alloc, min_time_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 2:
        # Active-set with very tight tolerance
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 1,
                                           tol_frac_start=0.001, tol_frac_end=0.00001,
                                           lr_init=0.01)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

        # Tiny L-BFGS
        remaining = deadline - 1 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00003
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(1000, max(20, int(remaining / 0.05)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
