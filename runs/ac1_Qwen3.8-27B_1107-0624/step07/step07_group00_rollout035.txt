            for idx, k in enumerate(active):
                ki = k - j
                if 0 <= ki < n:
                    s_vals[idx] = 2.0 * a_work[ki]
                else:
                    s_vals[idx] = 0.0
                c_vals[idx] = c[k]

            max_c_active = c_vals.max()

            # Try: delta to reduce the max active lag
            # Find which active lags are at the max
            at_max = np.where(c_vals >= max_c_active - 1e-15)[0]

            if len(at_max) == 0:
                continue

            # If only one lag is at max, reduce it
            # If multiple are at max, reduce the one with largest s_k (most sensitive)
            # Candidate 1: reduce the most sensitive max lag to just below current max
            # delta = -(max_c_active - next_max) / s_k  (small negative delta)

            # Sort by c_vals descending
            sort_idx = np.argsort(c_vals)[::-1]
            top1 = sort_idx[0]
            top2 = sort_idx[1] if len(sort_idx) > 1 else None

            candidates = [0.0]

            # Candidate: make top1 equal to top2
            if top2 is not None and abs(s_vals[top1] - s_vals[top2]) > 1e-15:
                d = (c_vals[top2] - c_vals[top1]) / (s_vals[top1] - s_vals[top2])
                if -a_j * 0.9 <= d <= a_j * 0.9:
                    candidates.append(d)

            # Candidate: reduce top1 by half its excess over the mean of others
            if s_vals[top1] > 1e-15:
                others_mean = c_vals[sort_idx[1:]].mean() if len(sort_idx) > 1 else 0
                excess = c_vals[top1] - others_mean
                if excess > 0:
                    d = -0.5 * excess / s_vals[top1]
                    if -a_j * 0.9 <= d <= a_j * 0.9:
                        candidates.append(d)

            # Candidate: small step
            candidates.append(-0.05 * a_j)
            candidates.append(0.05 * a_j)

            # Evaluate
            for d in candidates:
                if d < -a_j or d > a_j:
                    continue
                new_max = (c_vals + d * s_vals).max()
                if new_max < best_local - 1e-12:
                    best_local = new_max
                    best_delta = d

            if best_delta != 0.0:
                a_work[j] = a_j + best_delta
                any_improved = True

        # Periodic score check
        if sweep % recompute_every == 0:
            sc = _score_norm(a_work)
            if sc < best_sc * (1.0 - 1e-10):
                best_sc = sc
                best_a = a_work.copy()
                stagnation = 0
            else:
                stagnation += 1
            if sweep % 100 == 0:
                print(f"    CD sweep {sweep}: score={best_sc:.10f}, "
                      f"cur={sc:.10f}, stagnation={stagnation}")
            if stagnation > 200 and sweep > 200:
                break

    return best_a, best_sc


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, active_k=5, decay=0.6,
                   eval_interval=200, clip_norm=0.5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_conv(a, a)

        # Weighted active gradient
        k_idx = np.argsort(c)[::-1][:active_k]
        p = np.zeros_like(c)
        for i, idx in enumerate(k_idx):
            p[idx] = decay ** i
        p_sum = p.sum()
        if p_sum < 1e-300:
            continue
        p = p / p_sum
        p_rev = p[::-1]
        conv_full = _fft_conv(a, p_rev)
        g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. L-BFGS-B with decreasing smoothing temperature
    2. Active-set coordinate descent with exact 1D line search
    3. Adam in log-space for exploration
    4. Final ultra-fine polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n = len(a)
    best_a = a.copy()
    best_score = _score_full(best_a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # ===== Phase 1: L-BFGS-B with moderate smoothing (0-250s) =====
    phase1_end = min(deadline, t_start + 250.0)
    c_init = _fft_conv(a, a)
    c_max_init = float(c_init.max())
    T1 = 0.005 * c_max_init

    a1, _ = _lbgfb_phase(a, phase1_end, T1, maxiter=1500, print_tag="P1(T=med)")
    sc1 = _score_full(a1)
    print(f"  Phase 1 score: {sc1:.10f}")
    if np.isfinite(sc1) and sc1 < best_score:
        best_score = sc1
        best_a = a1.copy()

    # ===== Phase 2: Active-set CD (250-500s) =====
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end:
        a2, sc2_norm = _active_cd(best_a, phase2_end, top_k=25,
                                  recompute_every=4, max_sweeps=20000)
        sc2 = _score_full(a2)
        print(f"  Phase 2 (CD): score={sc2:.10f}")
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()

    # ===== Phase 3: L-BFGS-B with finer smoothing (500-700s) =====
    phase3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase3_end:
        c_cur = _fft_conv(best_a, best_a)
        c_max_cur = float(c_cur.max())
        T3 = 0.001 * c_max_cur
        a3, _ = _lbgfb_phase(best_a, phase3_end, T3, maxiter=1000,
                             print_tag="P3(T=fine)")
        sc3 = _score_full(a3)
        print(f"  Phase 3 score: {sc3:.10f}")
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    # ===== Phase 4: Active-set CD with tighter top_k (700-1000s) =====
    phase4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase4_end:
        a4, sc4_norm = _active_cd(best_a, phase4_end, top_k=15,
                                  recompute_every=3, max_sweeps=30000)
        sc4 = _score_full(a4)
        print(f"  Phase 4 (CD tight): score={sc4:.10f}")
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()

    # ===== Phase 5: L-BFGS-B with very fine smoothing (1000-1200s) =====
    phase5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase5_end:
        c_cur = _fft_conv(best_a, best_a)
        c_max_cur = float(c_cur.max())
        T5 = 0.0003 * c_max_cur
        a5, _ = _lbgfb_phase(best_a, phase5_end, T5, maxiter=800,
                             print_tag="P5(T=veryfine)")
        sc5 = _score_full(a5)
        print(f"  Phase 5 score: {sc5:.10f}")
        if np.isfinite(sc5) and sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()

    # ===== Phase 6: Adam log-space exploration (1200-1400s) =====
    phase6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase6_end:
        c_cur = _fft_conv(best_a, best_a)
        c_max_cur = float(c_cur.max())
        a6, sc6_norm = _adam_logspace(
            best_a, phase6_end,
            T_start=0.001 * c_max_cur, T_end=0.00005 * c_max_cur,
            lr_start=0.02, lr_end=0.001,
            max_iter=150000, active_k=5, decay=0.5,
            eval_interval=200, clip_norm=0.4)
        sc6 = _score_full(a6)
        print(f"  Phase 6 (Adam): score={sc6:.10f}")
        if np.isfinite(sc6) and sc6 < best_score:
            best_score = sc6
            best_a = a6.copy()

    # ===== Phase 7: Final active-set CD ultra-fine (1400-1650s) =====
    phase7_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase7_end:
        a7, sc7_norm = _active_cd(best_a, phase7_end, top_k=10,
                                  recompute_every=3, max_sweeps=50000)
        sc7 = _score_full(a7)
        print(f"  Phase 7 (CD ultra): score={sc7:.10f}")
        if np.isfinite(sc7) and sc7 < best_score:
            best_score = sc7
            best_a = a7.copy()

    # ===== Phase 8: Perturbation restarts (1650-1750s) =====
    phase8_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase8_end:
        for attempt in range(3):
            if time.monotonic() > phase8_end - 30:
                break
            sigma = 0.01 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_conv(a_pert, a_pert)
            max_cp = float(c_p.max())
            time_per = max(10.0, (phase8_end - time.monotonic()) / (3 - attempt))
            sub_dl = min(phase8_end - 5, time.monotonic() + time_per)
            a_pert_opt, _ = _active_cd(a_pert, sub_dl, top_k=10,
                                       recompute_every=3, max_sweeps=10000)
            sc_pert = _score_full(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={sc_pert:.10f}")
            if np.isfinite(sc_pert) and sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert_opt.copy()

    # ===== Phase 9: Final polish (1750-1785s) =====
    if time.monotonic() < deadline - 5:
        a_final_cd, _ = _active_cd(best_a, deadline - 3, top_k=8,
                                   recompute_every=2, max_sweeps=10000)
        sc_final_cd = _score_full(a_final_cd)
        if np.isfinite(sc_final_cd) and sc_final_cd < best_score:
            best_score = sc_final_cd
            best_a = a_final_cd.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_check = _score_full(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
