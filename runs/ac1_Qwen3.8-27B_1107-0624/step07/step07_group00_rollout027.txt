import numpy as np
import time

def lp_step(f, M, solver='CLARABEL'):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(b))
    # (f*b)_k for k=0..2n-2
    # (f*b)_k = sum_j f_j * b_{k-j} where 0<=j<n and 0<=k-j<n

        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        # (f*b)_k = sum_{j=j_min}^{j_max} f_j * b_{k-j}

        # b_{k-j} where j from j_min to j_max, so k-j from k-j_max to k-j_min
        coeffs = f[j_min:j_max+1]
        indices = np.arange(k - j_max, k - j_min + 1)
        constraints.append(coeffs @ b[indices] <= M)
    
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver)
    if b.value is not None:

    return None

"""
LP-guided optimizer with Adam smoothing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses cvxpy LP for direction finding, Adam in log-space for fine-tuning,
coordinate descent polishing, and multi-n exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy import sparse

def _next_pow2(n):
    k = 1
    while k < 1 and n > k:
        k <<= 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n*max(c)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(a, n):
    """Build sparse convolution matrix C where C[k,j] = a[j] if 0<=k-j<n else 0"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C

def _lp_direction(a, M, solver=None):
    """Solve LP: max sum(b) s.t. C@b <= M, b >= 0. Returns b or None."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(b))
    
    # Build constraints: (a*b)_k <= M for k=0..2n-2
    C = _build_conv_matrix(a, n)
    constraints = [C @ b <= M]
    
    prob = cp.Problem(obj, constraints)
    try:
        if solver:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception as e:
        print(f"  LP solve error: {e}")
        return None
    
    if b.value is None or not np.all(np.isfinite(b.value)):
        return None
    return np.maximum(b.value, 0)

def _lp_optimize(a, deadline, max_iters=50, solver=None):
    """Iteratively apply LP direction finding."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        
        # Solve LP
        b = _lp_direction(a, M, solver=solver)
        if b is None:
            break
        
        sum_b = float(b.sum())
        if sum_b <= 1 + 1e-6:
            # No improvement possible from LP
            break
        
        # Rescale b to have sum 1
        b_rescaled = b / sum_b
        
        # Line search: find best t in (0,1] such that score((1-t)*a + t*b_rescaled) is minimized
        best_t = 0
        best_sc = best_score
        for t in np.concatenate([np.linspace(0.1, 1.0, 10), [0.01, 0.05]]):
            a_cand = (1 - t) * a + t * b_rescaled
            a_cand = np.maximum(a_cand, 0)
            a_cand = a_cand / a_cand.sum()
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0:
            break
        
        a = (1 - best_t) * a + best_t * b_rescaled
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  LP iter {it}: score={sc:.8f}, t={best_t:.3f}, sum_b={sum_b:.4f}")
        else:
            # Revert
            a = best_a.copy()
            # Try smaller steps
            found = False
            for t in [0.01, 0.005, 0.001]:
                a_cand = (1 - t) * a + t * b_rescaled
                a_cand = np.maximum(a_cand, 0)
                a_cand = a_cand / a_cand.sum()
                sc = _score_norm(a_cand)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cand.copy()
                    a = a_cand.copy()
                    found = True
                    break
            if not found:
                break
    
    return best_a, best_score

def _adam_optimize(a, deadline, T_start=0.005, T_end=0.0001, lr_start=0.03, lr_end=0.002,
                   max_iter=200000, eval_interval=200):
    """Adam optimizer in log-space for the simplex."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        
        # Smoothed gradient
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
        
        p_rev = p[::-1]
        conv_full = _fft_convolve(a_cur, p_rev)
        g = 4.0 * n * conv_full[n-1:2*n-1][::-1]
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        
        z_new = z - update
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_eval = _softmax_stable(z)
            sc = _score_norm(a_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5:
                # Reset
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _cd_polish(a, deadline, max_rounds=100):
    """Coordinate descent polishing targeting peak lags."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        # Get top lags
        tol = c_max * 0.999
        active_lags = np.where(c > tol)[0]
        
        for k in active_lags:
            # Find which j values contribute to lag k
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min > j_max:
                continue
            # For each contributing j, try reducing best_a[j]
            for j in range(j_min, j_max + 1):
                if best_a[j] < 1e-15:
                    continue
                for scale in [0.7, 0.85, 0.9]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 0.01:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _build_conv_matrix_vec(a, n):
    """Vectorized construction of convolution matrix."""
    # For each row k (0 to 2n-2), the non-zero entries are at columns j_min to j_max
    # where j_min = max(0, k-n+1) and j_max = min(k, n-1)
    
    k_vals = np.arange(2*n - 1)
    j_min = np.maximum(0, k_vals - n + 1)
    j_max = np.minimum(k_vals, n - 1)
    lengths = j_max - j_min + 1  # number of non-zeros in each row
    
    # Total number of non-zeros
    total_nz = int(lengths.sum())
    
    # Build row indices
    row_idx = np.repeat(k_vals, lengths)

import cvxpy as cp
from scipy import sparse


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_direction_active(a, M, top_k=80, solver=None):
    """Solve LP with only active-set constraints: max sum(b) s.t. (a*b)_k <= M for top-k lags, b>=0."""
    n = len(a)
    c = _fft_convolve(a, a)
    active_lags = np.argsort(c)[::-1][:top_k]
    
    b = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(b))
    constraints = []
    
    for k in active_lags:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        coeffs = a[j_min:j_max + 1]
        indices = np.arange(k - j_max, k - j_min + 1)
        constraints.append(coeffs @ b[indices] <= M)
    
    prob = cp.Problem(obj, constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve(solver=cp.ECOS, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except Exception:
            return None
    
    if b.value is None or not np.all(np.isfinite(b.value)):
        return None
    return np.maximum(b.value, 0)

def _adam_optimize(a, deadline, T_start=0.005, T_end=0.0001, lr_start=0.03, lr_end=0.002,
                   max_iter=200000, eval_interval=200, clip=0.5):
    """Adam optimizer in log-space for the simplex."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        
        # Smoothed gradient
        c_shifted = (c - c.max()) / max(T, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
        
        p_rev = p[::-1]
        conv_full = _fft_convolve(a_cur, p_rev)
        g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > clip:
            update = update * (clip / (update_norm + 1e-15))
        
        z_new = z - update
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_eval = _softmax_stable(z)
            sc = _score_norm(a_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=80):
    """Targeted coordinate descent on peak lags."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        tol = c_max * 0.998
        active_lags = np.where(c > tol)[0]
        if len(active_lags) == 0:
            break
        
        improved = False
        for k in active_lags:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min > j_max:
                continue
            # Sort j by contribution
            contributions = np.array([best_a[j] * best_a[k - j] for j in range(j_min, j_max + 1)])
            j_range = np.arange(j_min, j_max + 1)
            order = np.argsort(contributions)[::-1]
            
            for idx in order[:10]:
                j = int(j_range[idx])
                if best_a[j] < 1e-15:
                    continue
                for scale in [0.6, 0.75, 0.85, 1.15, 1.3]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 0.01:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _gradient_step_with_line_search(a, deadline, max_iter=5000, T_frac=0.01):
    """Projected gradient with exact line search on the simplex."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        T = T_frac * c_max
        
        c_shifted = (c - c_max) / max(T, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
        
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        # Project gradient onto simplex tangent (sum=0)
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Line search
        best_alpha = 0.0
        best_sc = best_score
        for alpha in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            a_cand = a - alpha * g
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 0.01:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        a = a - best_alpha * g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            a = best_a.copy()
            break
        a /= s
        
        if _score_norm(a) < best_score:
            best_score = _score_norm(a)
            best_a = a.copy()
    
    return best_a, best_score

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n_parent}, score={best_score:.10f}")

    # Phase 1: LP optimization (0-180s)
    phase1_end = min(deadline, t_start + 180.0)
    print(f"Phase 1: LP optimization (n={n_parent})...")
    a_lp, sc_lp = _lp_optimize(a, phase1_end, max_iters=40, top_k=100)
    full_sc_lp = _score(a_lp)
    print(f"Phase 1 done: score={full_sc_lp:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc_lp) and full_sc_lp < best_score:
        best_score = full_sc_lp
        best_a = a_lp.copy()

    # Phase 2: Adam + Gradient (180-450s)
    phase2_end = min(deadline, t_start + 450.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Adam (n={len(best_a)})...")
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        
        mid2 = min(phase2_end, time.monotonic() + 130.0)
        a_adam, _ = _adam_optimize(best_a, mid2,
                                    T_start=0.005 * max_c, T_end=0.00005 * max_c,
                                    lr_start=0.03, lr_end=0.002,
                                    max_iter=150000, eval_interval=150)
        sc_adam = _score(a_adam)
        print(f"  Adam: score={sc_adam:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc_adam) and sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
        
        # Gradient with line search
        if time.monotonic() < phase2_end - 20:
            a_grad, _ = _gradient_step_with_line_search(best_a, phase2_end - 5,
                                                         max_iter=3000, T_frac=0.01)
            sc_grad = _score(a_grad)
            print(f"  Grad: score={sc_grad:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_grad) and sc_grad < best_score:
                best_score = sc_grad
                best_a = a_grad.copy()

    # Phase 3: CD polish (450-600s)
    phase3_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: CD polishing...")
        a_cd, _ = _cd_polish(best_a, phase3_end, max_rounds=60)
        sc_cd = _score(a_cd)
        print(f"Phase 3 done: score={sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()

    # Phase 4: Multi-n exploration (600-1050s)
    if time.monotonic() < deadline - 350:
        print(f"Phase 4: Multi-n exploration...")
        n_tries = [4096, 8192, 16384]
        if n_parent in n_tries:
            n_tries.remove(n_parent)
        n_tries.insert(0, n_parent)
        
        for n_try in n_tries:
            if time.monotonic() > deadline - 250:
                break
            time_alloc = min(120.0, (deadline - 250) - time.monotonic())
            if time_alloc < 15:
                break
            
            a_interp = _interp_to_length(best_a, n_try)
            c_try = _fft_convolve(a_interp, a_interp)
            max_try = float(c_try.max())
            sc_try = _score(a_interp)
            
            phase_end = time.monotonic() + time_alloc * 0.6
            a_opt, _ = _adam_optimize(a_interp, phase_end,
                                       T_start=0.003 * max_try, T_end=0.00003 * max_try,
                                       lr_start=0.025, lr_end=0.002,
                                       max_iter=100000, eval_interval=200)
            sc_opt = _score(a_opt)
            
            # Quick CD
            if time.monotonic() < time.monotonic() + (time_alloc * 0.3):
                a_opt_cd, _ = _cd_polish(a_opt, time.monotonic() + time_alloc * 0.3, max_rounds=20)
                sc_opt_cd = _score(a_opt_cd)
                if sc_opt_cd < sc_opt:
                    sc_opt = sc_opt_cd
                    a_opt = a_opt_cd
            
            print(f"  n={n_try}: initial={sc_try:.8f}, optimized={sc_opt:.8f}, t={time.monotonic()-t_start:.0f}s")
            
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()

    # Phase 5: Intensive polish (1050-1650s)
    if time.monotonic() < deadline - 100:
        print(f"Phase 5: Intensive polish (n={len(best_a)})...")
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        
        # More LP
        phase5_lp_end = min(deadline - 100, time.monotonic() + 100.0)
        if n_best <= 8192:
            a_lp2, _ = _lp_optimize(best_a, phase5_lp_end, max_iters=20, top_k=60)
            sc_lp2 = _score(a_lp2)
            print(f"  LP polish: score={sc_lp2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_a = a_lp2.copy()
        
        # Low-temp Adam
        if time.monotonic() < deadline - 50:
            c_b2 = _fft_convolve(best_a, best_a)
            max_b2 = float(c_b2.max())
            phase5_adam_end = min(deadline - 30, time.monotonic() + 150.0)
            a_adam2, _ = _adam_optimize(best_a, phase5_adam_end,
                                         T_start=0.001 * max_b2, T_end=0.00001 * max_b2,
                                         lr_start=0.01, lr_end=0.0005,
                                         max_iter=150000, eval_interval=100)
            sc_a2 = _score(a_adam2)
            print(f"  Adam polish: score={sc_a2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_a2) and sc_a2 < best_score:
                best_score = sc_a2
                best_a = a_adam2.copy()
        
        # Gradient
        if time.monotonic() < deadline - 30:
            a_gr2, _ = _gradient_step_with_line_search(best_a, min(deadline - 10, time.monotonic() + 80),
                                                        max_iter=2000, T_frac=0.005)
            sc_g2 = _score(a_gr2)
            if np.isfinite(sc_g2) and sc_g2 < best_score:
                best_score = sc_g2
                best_a = a_gr2.copy()
        
        # Final CD
        if time.monotonic() < deadline - 10:
            a_cd2, _ = _cd_polish(best_a, deadline - 5, max_rounds=30)
            sc_cd2 = _score(a_cd2)
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_a = a_cd2.copy()

    # Phase 6: Random restarts (last 100s)
    if time.monotonic() < deadline - 20:
        n_fin = len(best_a)
        print(f"Phase 6: Random restarts (n={n_fin})...")
        for attempt in range(3):
            if time.monotonic() > deadline - 25:
                break
            sigma = 0.01 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            time_per = max(15.0, (deadline - 25 - time.monotonic()) / (3 - attempt))
            a_pert_opt, _ = _adam_optimize(a_pert, time.monotonic() + time_per,
                                            T_start=0.001 * max_cp, T_end=0.00001 * max_cp,
                                            lr_start=0.015, lr_end=0.001,
                                            max_iter=50000, eval_interval=100)
            sc_pert = _score(a_pert_opt)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={sc_pert:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_pert) and sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert_opt.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
