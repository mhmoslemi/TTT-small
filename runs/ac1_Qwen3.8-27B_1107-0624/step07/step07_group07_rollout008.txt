        cross = np.fft.ifft(np.conj(W) * A, N).real[:len(a)]

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = next_pow2(out_n)

    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def central_binomial_coeffs(n):
    a = np.ones(n)
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def apply_taper(a, taper_type, frac=0.25):
    n = len(a)
    a = a.copy()
    m = max(1, int(frac * n))
    if taper_type == 'none':
        pass
    elif taper_type == 'cos_end':
        t = np.linspace(0, np.pi / 2, m)
        a[n - m:] *= np.cos(t)
    elif taper_type == 'cos_both':
        t = np.linspace(np.pi / 2, 0, m)
        a[:m] *= np.cos(t)
        t2 = np.linspace(0, np.pi / 2, m)
        a[n - m:] *= np.cos(t2)
    elif taper_type == 'lin_end':
        t = np.linspace(1, 0, m)
        a[n - m:] *= t
    elif taper_type == 'lin_both':
        t = np.linspace(0, 1, m)
        a[:m] *= t
        t2 = np.linspace(1, 0, m)
        a[n - m:] *= t2
    elif taper_type == 'exp_end':
        t = np.linspace(0, 1, m)
        a[n - m:] *= np.exp(-2.0 * t)
    elif taper_type == 'quad_end':
        t = np.linspace(1, 0, m)
        a[n - m:] *= t * t
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def make_initial(n, taper='none', frac=0.25):
    a = central_binomial_coeffs(n)
    a = apply_taper(a, taper, frac)
    return a


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def smoobj(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def smograd(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def lbfgs_refine(a_init, deadline, min_time=60.0):
    n = len(a_init)
    c_cur = full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_init

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7])
    current_a = a_init.copy()
    current_sc = compute_score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def fun(z):
            return smoobj(z, beta, n)

        def jac(z):
            return smograd(z, beta, n)

        est_per_iter = 0.003 * (n / 4096.0) ** 1.3
        max_iter = min(8000, max(100, int(remaining / (est_per_iter * 15))))

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxcor': 80,
                                       'maxls': 60, 'disp': False})
            a_new = softmax(result.x)
            sc = compute_score(a_new)
            if sc < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc
                print(f"    LBFGS beta={beta:.2e}: score={sc:.10f}")
        except Exception:
            pass

    return current_a


def multiplicative_weights_optimize(a_init, deadline, min_time=300.0):
    """Mirror descent on simplex with log-sum-exp smoothed max."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    c = full_convolve(a, a)
    c_max = float(c.max())

    betas = c_max * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                              0.01, 0.005, 0.002, 0.001, 0.0005,
                              0.0002, 0.0001])
    eta = 0.1
    stall = 0

    for beta in betas:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 30:
            break

        max_steps = min(50000, int(remaining / 0.002))
        stall = 0

        for step in range(max_steps):
            if time.monotonic() + min_time > deadline:
                break
            if stall > 200:
                break

            c = full_convolve(a, a)
            c_max = c.max()
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            w = e / e.sum()

            a_rev = a[::-1]
            wc = full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * n * (h - W)
            grad = grad - grad.mean()

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Multiplicative (mirror) update
            z = np.log(np.maximum(a, 1e-30))
            z = z - eta * grad / (grad_norm + 1e-30)
            z = np.clip(z, -80.0, 80.0)
            a_new = softmax(z)
            a_new = a_new / a_new.sum()

            sc = compute_score(a_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new
                eta = min(eta * 1.02, 0.5)
                stall = 0
            else:
                stall += 1
                a = best_a.copy()
                eta *= 0.95
                if eta < 0.001:
                    break

            if step % 5000 == 0 and step > 0:
                print(f"    MW beta={beta:.2e} step={step}: score={best_sc:.10f}")

    return best_a, best_sc


def peak_shaving(a_init, deadline, min_time=60.0, max_iters=30000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() + min_time > deadline:
            break
        if stall > 200:
            break

        c = full_convolve(a, a)
        c_max = c.max()
        tol = max(1e-4 * c_max, c_max - np.sort(c)[::-1][min(len(c)-1, int(0.05*len(c)))])
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        best_sc_try = best_sc
        best_a_try = a
        for frac in [1.0, 0.5, 0.25, 0.1]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = compute_score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.01, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-9:
                break

    return best_a, best_sc


def coordinate_descent(a_init, deadline, min_time=30.0, max_rounds=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(max_rounds):
        if time.monotonic() + min_time > deadline:
            break

        idx = int((rnd * 6151 + 7) % n)
        jdx = int((rnd * 6151 + 13) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        for frac in np.array([0.3, 0.4, 0.5, 0.6, 0.7, 0.2, 0.8]):
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = compute_score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and rnd % 50 == 0:
            idx2 = int((rnd * 397 + 3) % n)
            jdx2 = int((rnd * 397 + 11) % n)
            if idx2 != jdx2:
                s_p2 = a[idx2] + a[jdx2]
                for frac2 in [0.4, 0.5, 0.6]:
                    a_try2 = a.copy()
                    a_try2[idx2] = s_p2 * frac2
                    a_try2[jdx2] = s_p2 * (1.0 - frac2)
                    sc_try2 = compute_score(a_try2)
                    if sc_try2 < best_sc - 1e-13:
                        best_sc = sc_try2
                        best_a = a_try2.copy()
                        a = a_try2
                        improved = True
                        break

    return best_a, best_sc


def perturb_restart(a, rng, mode="gauss", scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(5):
            freq = int(rng.integers(1, max(2, n // 20)))
            amp = 0.005 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(16, n // 32)
        for _ in range(6):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.005, 0.005)
    z = z - z.mean()
    return softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = compute_score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Phase 0: Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={compute_score(parent):.10f}")
        try_update(parent, "parent")

    # Phase 1: Central binomial initialization scan
    print(f"\n=== Phase 1: Initialization scan ===")
    t_phase = time.monotonic()

    lengths_to_try = [4096, 8192, 16384]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            lengths_to_try.append(pn)
        lengths_to_try.sort()

    tapers = [
        ('none', 0.0),
        ('cos_end', 0.15),
        ('cos_end', 0.25),
        ('cos_end', 0.35),
        ('lin_end', 0.2),
        ('exp_end', 0.2),
        ('quad_end', 0.25),
        ('cos_both', 0.15),
        ('cos_both', 0.25),
    ]

    cb_cache = {}
    for n in lengths_to_try:
        if n not in cb_cache:
            a_base = central_binomial_coeffs(n)
            a_base = a_base / a_base.sum()
            cb_cache[n] = a_base

        for taper_name, taper_frac in tapers:
            if time.monotonic() > deadline - 1200:
                break
            a = cb_cache[n].copy()
            a = apply_taper(a, taper_name, taper_frac)
            sc = compute_score(a)
            tag = f"cb_n{n}_{taper_name}_{taper_frac}"
            if sc < best_score_val:
                best_score_val = sc
                best_seq = a.copy()
                print(f"  NEW BEST: {sc:.10f} (n={n}) [{tag}]")
            elif sc < 1.58:
                print(f"  {tag}: {sc:.6f}")

    if parent is not None and best_seq is None:
        best_seq = parent.copy()
        best_score_val = compute_score(parent)

    if best_seq is None:
        best_seq = make_initial(8192, 'cos_end', 0.25)
        best_score_val = compute_score(best_seq)
        print(f"  Fallback init: {best_score_val:.6f}")

    print(f"  Phase 1 done: best={best_score_val:.10f}, time={time.monotonic()-t_phase:.1f}s")

    # Phase 2: Multiplicative weights optimization (main workhorse)
    print(f"\n=== Phase 2: Multiplicative weights ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 600:
        a_mw, sc_mw = multiplicative_weights_optimize(
            best_seq, deadline, min_time=500.0)
        try_update(a_mw, "mw_opt")
    print(f"  Phase 2 done: time={time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # Phase 3: L-BFGS refinement
    print(f"\n=== Phase 3: L-BFGS refinement ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_lb = lbfgs_refine(best_seq, deadline, min_time=250.0)
        try_update(a_lb, "lbfgs_refine")
    print(f"  Phase 3 done: time={time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # Phase 4: Peak shaving
    print(f"\n=== Phase 4: Peak shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 200:
        a_ps, sc_ps = peak_shaving(best_seq, deadline, min_time=150.0, max_iters=50000)
        try_update(a_ps, "peak_shave")
    print(f"  Phase 4 done: time={time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # Phase 5: Multi-restart
    print(f"\n=== Phase 5: Multi-restart ===")
    t_phase = time.monotonic()
    restart_configs = [
        ("gauss_01", "gauss", 0.01),
        ("gauss_03", "gauss", 0.03),
        ("gauss_05", "gauss", 0.05),
        ("fourier", "fourier", 0.0),
        ("block", "block", 0.0),
    ]
    for name, mode, scale in restart_configs:
        if time.monotonic() > deadline - 100:
            break
        p_start = perturb_restart(best_seq, rng, mode, scale)
        sc_start = compute_score(p_start)
        print(f"  Restart [{name}]: start={sc_start:.10f}")

        a_r, _ = multiplicative_weights_optimize(p_start, deadline, min_time=60.0)
        try_update(a_r, f"restart_{name}_mw")

        if time.monotonic() < deadline - 50:
            a_r2 = lbfgs_refine(a_r, deadline, min_time=40.0)
            try_update(a_r2, f"restart_{name}_lbfgs")
    print(f"  Phase 5 done: time={time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # Phase 6: Try different lengths
    print(f"\n=== Phase 6: Length exploration ===")
    t_phase = time.monotonic()
    current_n = len(best_seq)
    for target_n in [4096, 8192, 16384]:
        if time.monotonic() > deadline - 80:
            break
        if target_n == current_n:
            continue
        # Interpolate
        t_old = np.linspace(0, 1, current_n)
        t_new = np.linspace(0, 1, target_n)
        a_interp = np.interp(t_new, t_old, best_seq)
        a_interp = np.maximum(a_interp, 1e-15)
        a_interp = a_interp / a_interp.sum()
        sc_interp = compute_score(a_interp)
        print(f"  n={target_n} interpolated: {sc_interp:.10f}")

        a_opt, _ = multiplicative_weights_optimize(a_interp, deadline, min_time=60.0)
        try_update(a_opt, f"len_{target_n}_mw")

        if time.monotonic() < deadline - 40:
            a_opt2 = lbfgs_refine(a_opt, deadline, min_time=35.0)
            try_update(a_opt2, f"len_{target_n}_lbfgs")
    print(f"  Phase 6 done: time={time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # Phase 7: Final intensive polish
    print(f"\n=== Phase 7: Final polish ===")
    t_phase = time.monotonic()
    for cycle in range(5):
        if time.monotonic() > deadline - 10:
            break

        if time.monotonic() < deadline - 10:
            a_f1, _ = peak_shaving(best_seq, deadline, min_time=8.0, max_iters=10000)
            try_update(a_f1, f"polish_ps_{cycle}")

        if time.monotonic() < deadline - 10 and best_seq is not None:
            n = len(best_seq)
            c_b = full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            z0 = np.log(np.maximum(best_seq, 1e-30))
            z0 = z0 - z0.mean()

            def fun_f(z):
                return smoobj(z, beta, n)

            def jac_f(z):
                return smograd(z, beta, n)

            max_iter = min(3000, max(100, int((deadline - 10 - time.monotonic()) / 0.005)))
            try:
                result = minimize(fun_f, z0, jac=jac_f, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 50,
                                           'maxls': 40, 'disp': False})
                a_f2 = softmax(result.x)
                try_update(a_f2, f"polish_lbfgs_{cycle}")
            except Exception:
                pass

        if time.monotonic() < deadline - 8:
            a_f3, _ = coordinate_descent(best_seq, deadline, min_time=6.0, max_rounds=5000)
            try_update(a_f3, f"polish_cd_{cycle}")

    print(f"  Phase 7 done: time={time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = central_binomial_coeffs(4096)
            final = final / final.sum()
        sc_final = compute_score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")

        return [float(x) for x in final]

    fallback = central_binomial_coeffs(4096)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: score={compute_score(fallback):.10f}")
    return [float(x) for x in fallback]
