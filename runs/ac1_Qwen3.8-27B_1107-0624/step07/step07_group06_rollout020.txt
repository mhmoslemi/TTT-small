delta_c_k = t * (a_{k-j} + a_{k-j}) = 2*t*a_{k-j} (if 0≤k-j<n and k≠2j)
delta_c_k = t^2 (if k=2j)

And base_k is c_k with a_j = 0 (i.e., the sum over all i≠j of a_i * a_{k-i}, with a_{k-i} = 0 if k-i = j).

Actually, it's easier to compute:
base_k = c_k_original - (terms involving a_j)
       = c_k_original - 2*a_j*a_{k-j} (if 0≤k-j<n and k≠2j)
       = c_k_original - a_j^2 (if k=2j)

Then new c_k = base_k + 2*t*a_{k-j} (if 0≤k-j<n and k≠2j) or base_k + t^2 (if k=2j).

The score is f(t) = 2n * max_k c_k(t) / (1 - a_j + t)^2

This is a ratio of a convex piecewise-linear (or quadratic at k=2j) function to a quadratic. It's not convex in general, but it's unimodal in practice (or close to it).

For a 1D optimization, I can use golden section search or just sample t on a grid and then refine.

Actually, a simpler approach: for the subgradient method, the direction to move a_j is determined by which c_k is at the maximum. If c_k* is max and k* - j is in range, then increasing a_j increases c_k* (since a_{k*-j} > 0), so we should decrease a_j. Conversely, if no active c_k involves a_j, we can increase a_j to reduce the relative contribution of the active peaks.

  - If g_j > mean_gradient: move a_j down
  - If g_j < mean_gradient: move a_j up
  - Line search to find optimal step

"""
Multi-strategy optimizer combining structured anti-symmetric initialization,
combined max+variance L-BFGS, exact coordinate descent, and population
recombination to minimize 2n*max(conv(a,a))/(sum a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_normalize(a):
    z = np.log(np.maximum(a, 1e-30))
    return z - z.mean()


def _smoothed_max_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_max_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _variance_obj(a):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _combined_obj(z, beta, n, var_weight):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        base = 2.0 * n * c_max
    else:
        base = 2.0 * n * (c_max + beta * np.log(s))
    mean_c = np.mean(c)
    var_c = np.mean((c - mean_c) ** 2)
    return base + var_weight * var_c / (c_max + 1e-30)


def _combined_grad(z, beta, n, var_weight):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_main = 4.0 * n * a * (h - W)
    
    # Variance gradient
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g_var = 2.0 * dcd_a[n - 1:2 * n - 1]
    g_var = g_var - np.mean(g_var)
    
    return grad_main + (var_weight / (c_max + 1e-30)) * g_var


def _active_grad(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec_fn, b, n, max_iter=200, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec_fn(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _make_profile_antisym(n, alpha=0.5, taper=2.0, phase=0.0):
    """Create an anti-symmetric tapered profile."""
    i = np.arange(n, dtype=np.float64)
    t = i / n
    # Taper to reduce edge effects
    taper_arr = np.sin(np.pi * t) ** taper
    # Anti-symmetric modulation
    mod = 1.0 + alpha * np.sin(2.0 * np.pi * t + phase)
    a = taper_arr * mod
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_profile_varied(n, kind="uniform", **kwargs):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "truncated_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "antisym":
        alpha = kwargs.get('alpha', 0.5)
        taper = kwargs.get('taper', 2.0)
        phase = kwargs.get('phase', 0.0)
        return _make_profile_antisym(n, alpha, taper, phase)
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        p = kwargs.get('p', 1.0)
        a = t ** p
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        sigma = kwargs.get('sigma', 0.15)
        a = np.exp(-((t - 0.3) ** 2) / (2 * sigma ** 2)) + np.exp(-((t - 0.7) ** 2) / (2 * sigma ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs(z_init, n, betas, var_weights, deadline, min_remaining=30.0):
    """L-BFGS with combined objective and beta continuation."""
    current_z = z_init.copy()
    current_a = _softmax(current_z)
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        
        var_w = var_weights[min(i, len(var_weights) - 1)]
        
        est_time = 0.003 * max(1, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 10))))
        
        def fun_local(z_local):
            return _combined_obj(z_local, beta, n, var_w)

        def jac_local(z_local):
            return _combined_grad(z_local, beta, n, var_w)

        try:
            result = minimize(
                fun_local, current_z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 50, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                current_z = _log_normalize(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_sc

        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coordinate_descent(a_init, deadline, n_sweeps=5):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Compute current autocorrelation with a_j removed
            # c_k = base_k + contribution from a_j
            # We'll just do a simple line search
            old_val = a[j]
            # Try a few values
            candidates = [old_val * 0.5, old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5,
                         old_val * 0.25, old_val * 2.0, 0.0, old_val * 0.1]
            candidates = [max(0.0, c) for c in candidates]
            
            best_local_sc = best_sc
            best_local_val = old_val
            
            for t in candidates:
                a[j] = t
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_val
                    continue
                # Quick score computation
                c = _full_convolve(a, a)
                sc = 2.0 * n * float(c.max()) / (s * s)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local_val = t
            a[j] = best_local_val
            
            if best_local_val != old_val:
                improved = True
                best_sc = max(best_local_sc, 0)
                # Renormalize
                a = a / a.sum()
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_sc


def _recombine(a1, a2, alpha=0.5):
    """Geometric mean recombination in log space."""
    z1 = _log_normalize(a1)
    z2 = _log_normalize(a2)
    z = alpha * z1 + (1 - alpha) * z2
    return _softmax(z)


def _perturb_log(a, sigma, rng):
    z = _log_normalize(a)
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return

        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")

                inits.append(a)

    for kind in ["tapered", "cos2", "wheatstone", "bimodal", "ramp"]:
        a = _make_profile_varied(n_try, kind)

    # Evaluate all and keep top 5
    init_scores = []
    for a in inits:
        sc = _score(a)

    init_scores.sort(key=lambda x: x[0])
    top_inits = [a for _, a in init_scores[:8]]
    print(f"  Tried {len(inits)} initializations, best init score: {init_scores[0][0]:.6f}")
