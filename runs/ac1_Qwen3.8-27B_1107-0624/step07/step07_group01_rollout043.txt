def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _solve_min_norm_qp(K, m):
    """Solve min w^T K w s.t. w >= 0, sum(w) = 1.
    Returns optimal w vector."""
    # Use projected gradient with exact projection onto simplex
    w = np.ones(m) / m
    lr = 1.0 / (2.0 * np.abs(K).max() + 1e-15)
    for _ in range(200):
        grad = 2.0 * K.dot(w)
        w_new = w - lr * grad
        # Project onto simplex {w >= 0, sum(w) = 1}
        w_new = _project_simplex(w_new)
        if np.linalg.norm(w_new - w) < 1e-14:
            w = w_new
            break
        w = w_new
    return w


def _project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _active_newton_step(a, c, S, target_frac=0.0):
    """
    Compute the optimal step direction using active-set Newton method.
    
    da = -V^T * (VV^T)^{-1} * (c_S - target * 1)
    
    where target determines how much we want to reduce the active values.
    VV^T[i,j] = 4*c[n-1 + S[j] - S[i]] (if |S[i]-S[j]| < n)
    
    Returns the centered step direction (sum = 0).
    """
    n = len(a)
    m = len(S)
    
    # Build K = VV^T / 4 (the factor of 4 doesn't matter for direction)
    K = np.zeros((m, m))
    for i in range(m):
        for j in range(m):
            d = int(S[j]) - int(S[i])
            if abs(d) < n:
                K[i, j] = c[n - 1 + d]
    
    # Add regularization for numerical stability
    reg = 1e-10 * np.trace(K) / m
    K_reg = K + reg * np.eye(m)
    
    # Target: reduce active values slightly below their minimum
    c_active = c[S]
    c_min = c_active.min()
    c_max_active = c_active.max()
    spread = c_max_active - c_min
    target = c_min - target_frac * spread  # slightly below min
    
    r = c_active - target  # positive vector (amount to reduce)
    
    # Solve K_reg * w = r
    try:
        w = np.linalg.solve(K_reg, r)
    except np.linalg.LinAlgError:
        w = np.linalg.lstsq(K_reg, r, rcond=None)[0]
    
    # If w has negative components, project and re-solve
    if np.any(w < -1e-10):
        w_pos = np.maximum(w, 0)
        if w_pos.sum() < 1e-15:
            w = np.ones(m) / m
        else:
            w = w_pos / w_pos.sum()
    
    # Compute da = V^T * w (using FFT trick)
    v_pad = np.zeros(2 * n - 1)
    v_pad[S] = w
    a_rev = a[::-1]
    da = _full_convolve(v_pad, a_rev)[n - 1:2 * n - 1]
    
    # Center to maintain sum(a) = 1
    da = da - da.mean()
    
    return da, K_reg, w

                if lr < 1e-10:
                    break
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _direct_subgrad_refined(a_init, deadline, M=3, max_steps=50000):
    """Direct subgradient in a-space with adaptive step and multiple M."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    step_size = 0.3 / np.sqrt(n)
    stall = 0
    M_use = M
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:M_use]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        a_try = a - step_size * g / np.sqrt(n)
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-30:
            step_size *= 0.5
            stall += 1
            if stall > 200:
                break
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        sc_cur = _score(a)
        
        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
        
        if sc_try < sc_cur - 1e-15:
            a = a_try
            step_size = min(step_size * 1.01, 0.5)
        else:
            stall += 1
            step_size *= 0.95
            if stall > 500:
                break
    
    return best_a, best_sc


def _fourier_refine(a_init, deadline, max_iter=5000):
    """
    Fourier-domain refinement: adjust phase of DFT to flatten |A(f)|^2.
    Constraint: IFFT(A) must be real and non-negative.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        
        # Target: flatten |A|^2
        # Variance of |A|^2
        mag2 = mag ** 2
        mean_mag2 = mag2.mean()
        
        if mean_mag2 < 1e-30:
            break
        
        # Gradient of variance w.r.t. log(mag)
        # d(var)/d(mag) = 2*mag*(mag2 - mean_mag2)
        # d(var)/d(log(mag)) = mag2 * 2 * (mag2 - mean_mag2)
        grad_log = 2.0 * mag2 * (mag2 - mean_mag2)
        
        # Take a small step in log-magnitude
        eta = 0.01
        mag2_target = mag2 - eta * grad_log
        mag2_target = np.maximum(mag2_target, 1e-30)
        mag_new = np.sqrt(mag2_target)
        
        # Update A with new magnitude, keep phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, n=N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            if it > 50:
                break
    
    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _kkt_refine(a_init, deadline, M=4, max_iter=500):
    """KKT-inspired refinement with line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:M]
        
        # Compute direction to reduce active values
        w = np.zeros(2 * n - 1)
        weights = np.array([1.0 / (k + 1) for k in range(M)])
        weights /= weights.sum()
        w[idx] = weights
        
        a_rev = a
        wc = _full_convolve(w, a_rev[::-1])
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = h - W
        g = g - g.mean()
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a - alpha * g / n
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6
        
        if not improved:
            break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer focusing on active-set Newton refinement:
    Phase 1: Load parent / best initialization
    Phase 2: Coarse L-BFGS optimization
    Phase 3: Active-set Newton (main refinement)
    Phase 4: Subgradient polishing (multiple M values)
    Phase 5: Fourier-domain refinement
    Phase 6: Ultra-fine active-set polishing
    Phase 7: Final L-BFGS with tiny beta
    Phase 8: KKT refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev-best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")
    
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        n_init = 4096
        a_init = _make_chirp(n_init, freq=1.0, eps=0.3)
        try_update(a_init, "chirp_init")
        best_seq = a_init.copy()
        best_score_val = _score(best_seq)
        print(f"  No parent. Using chirp init: score={best_score_val:.10f}")
    
    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s")
    
    # === Phase 2: Coarse L-BFGS ===
    print(f"\n=== Phase 2: Coarse L-BFGS (t={elapsed:.1f}s) ===")
    reserve_p2 = max(600.0, 0.4 * budget_s)
    a_p2 = _adaptive_lbfgs(best_seq, deadline - reserve_p2, reserve_time=reserve_p2 * 0.5)
    try_update(a_p2, "lbfgs_coarse")
    
    # === Phase 3: Active-set Newton (MAIN) ===
    print(f"\n=== Phase 3: Active-set Newton (t={time.monotonic()-t_start:.1f}s) ===")
    elapsed = time.monotonic() - t_start
    reserve_for_rest = max(300.0, 0.25 * budget_s)
    deadline_p3 = deadline - reserve_for_rest
    if deadline_p3 > t_start + 30:
        a_an, sc_an = _active_newton_optimize(
            best_seq, deadline_p3,
            M_list=[3, 5, 8, 12, 20, 30],
            target_frac_list=[0.0, 0.1, 0.3],
            max_iter=500000
        )
        try_update(a_an, "active_newton")
    
    # === Phase 4: More L-BFGS with smaller beta ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_p4 = _adaptive_lbfgs(best_seq, deadline - 180, reserve_time=100.0)
        try_update(a_p4, "lbfgs_fine")
    
    # === Phase 5: Subgradient polishing ===
    print(f"\n=== Phase 5: Subgradient polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        a_sg, _ = _subgrad_polish_refined(best_seq, deadline - 130, M=5, max_steps=30000)
        try_update(a_sg, "sg_M5")
    
    if time.monotonic() < deadline - 100:
        a_sg2, _ = _direct_subgrad_refined(best_seq, deadline - 80, M=3, max_steps=20000)

    # === Phase 6: Fourier refinement ===

    # === Phase 7: Ultra-fine active-set polishing ===

    if time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        for beta_frac in [1e-6, 5e-7, 2e-7]:

                break
            beta = mb * beta_frac
            max_iter = min(3000, max(100, int((deadline - 5 - time.monotonic()) / 0.005)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_beta_{beta_frac}")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
