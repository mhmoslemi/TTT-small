def _hessian_matvec(v, active, n):

    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "truncated_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _cosine_profile(n, freq, alpha):
    t = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _fejer_profile(n, order):
    t = np.arange(n)
    kernel = np.ones(n)
    for k in range(1, min(order, n)):
        kernel += (order - k) / order * np.cos(2.0 * np.pi * k * t / n)
    a = np.maximum(kernel, 1e-10)
    return a / a.sum()


def _chirp_profile(n, chirp_rate):
    t = np.arange(n)
    phase = np.pi * chirp_rate * t * t / n
    a = np.abs(np.exp(1j * phase)) ** 2
    env = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n * 3)
    a = a * env
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)

            best_a = a_try.copy()

def _block_swap_local_search(a_init, deadline, n_trials=100):
    """Try block swaps to escape local minima."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)

    block_sizes = [n // 8, n // 16, n // 32, n // 64, max(1, n // 128)]

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        bs = block_sizes[trial % len(block_sizes)]
        a_try = best_a.copy()
        i = rng.integers(0, n - 2 * bs)
        j = rng.integers(0, n - 2 * bs)
        if abs(i - j) < bs:
            continue
        tmp = a_try[i:i + bs].copy()
        a_try[i:i + bs] = a_try[j:j + bs]
        a_try[j:j + bs] = tmp
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_try.copy()

    return best_a, best_sc


def _cyclic_shift_search(a_init, deadline, n_shifts=50):
    """Try cyclic shifts of the sequence."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    a_norm = a_init / a_init.sum()

    for k in range(1, min(n_shifts, n // 4)):
        if time.monotonic() > deadline:
            break
        a_shifted = np.roll(a_norm, k)
        a_shifted = np.maximum(a_shifted, 0)
        s = a_shifted.sum()
        if s < 1e-10:
            continue
        a_shifted = a_shifted / s
        sc = _score(a_shifted)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_shifted.copy()

    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral shaping, diverse initializations,
    aggressive L-BFGS with beta continuation, Newton active-set, subgradient
    polish, and block/cyclic local search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profile(4096, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # === Phase 1: Diverse initializations ===
    print(f"\n[Phase 1] Diverse initializations (t={time.monotonic() - t_start:.1f}s)")
    init_candidates = []

    n_values = [4096, 8192, 16384]
    # Add parent n
    if n_parent > 0 and n_parent not in n_values:
        n_values.append(n_parent)
    n_values = sorted(set(n_values))

    for n_try in n_values:
        if time.monotonic() > deadline - 600:
            break

        profiles = [
            _make_profile(n_try, "uniform"),
            _make_profile(n_try, "tapered"),
            _make_profile(n_try, "cos2"),
            _make_profile(n_try, "wheatstone"),
            _make_profile(n_try, "truncated_gauss"),
            _cosine_profile(n_try, 1, 0.5),
            _cosine_profile(n_try, 3, 0.4),
            _cosine_profile(n_try, 7, 0.3),
            _cosine_profile(n_try, n_try // 16, 0.6),
            _fejer_profile(n_try, n_try // 8),
            _chirp_profile(n_try, 2.0),
            _spectral_shape_profile(n_try, 0.7),
        ]

        # If we have a parent, also try interpolating it to this n
        if parent is not None and len(parent) != n_try:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, parent)
            a_interp = np.maximum(a_interp, 1e-10)
            a_interp = a_interp / a_interp.sum()
            profiles.append(a_interp)

        for prof in profiles:
            prof = np.maximum(prof, 0)
            s = prof.sum()
            if s < 1e-10:
                continue
            prof = prof / s
            sc = _score(prof)
            if np.isfinite(sc):
                init_candidates.append((n_try, prof.copy(), sc))

    # Sort by score
    init_candidates.sort(key=lambda x: x[2])
    print(f"  Generated {len(init_candidates)} candidates, best init score: {init_candidates[0][2]:.10f}")

    # Quick-optimize the top initializations
    top_inits = init_candidates[:8]
    for n_init, a_init, sc_init in top_inits:
        if time.monotonic() > deadline - 500:
            break
        update_best(a_init, sc_init, f"init_n{n_init}")
        # Quick L-BFGS
        if time.monotonic() < deadline - 400:
            a_opt = _run_lbfgs_aggressive(a_init, min(deadline - 350, time.monotonic() + 60),
                                          min_time_remaining=350)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"quick_n{n_init}")

    # === Phase 2: Main aggressive L-BFGS on best ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_main = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=300)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Newton active-set ===
    print(f"\n[Phase 3] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        n_deadline = min(deadline - 200, time.monotonic() + 200)
        a_newton, sc_newton = _newton_active_set(best_seq, n_deadline, max_iter=300)
        update_best(a_newton, sc_newton, "newton_main")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        sg_deadline = min(deadline - 150, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=5000, lr_init=0.012)
        update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 5: Shrinkage + re-optimize ===
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 150)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score * 1.002 and time.monotonic() < deadline - 100:
            a_shr2 = _run_lbfgs_aggressive(a_shr, min(deadline - 50, time.monotonic() + 80),
                                           min_time_remaining=50)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")
            if time.monotonic() < deadline - 30:
                a_shr3, sc_shr3 = _newton_active_set(a_shr2, min(deadline - 20, time.monotonic() + 60), max_iter=100)
                update_best(a_shr3, sc_shr3, "newton_post_shrink")

    # === Phase 6: Cyclic shift + block swap ===
    print(f"\n[Phase 6] Local search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_cs, sc_cs = _cyclic_shift_search(best_seq, min(deadline - 80, time.monotonic() + 30), n_shifts=100)
        update_best(a_cs, sc_cs, "cyclic_shift")
        if time.monotonic() < deadline - 60:
            a_bs, sc_bs = _block_swap_local_search(best_seq, min(deadline - 40, time.monotonic() + 30), n_trials=200)
            update_best(a_bs, sc_bs, "block_swap")

    # === Phase 7: Diverse restarts from best ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_restarts = 6
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            label = f"restart_{i}"
            if i == 0:
                p_start = _perturb_log(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=8)
            elif i == 2:
                p_start = _perturb_log(best_seq, 0.05, rng)
            elif i == 3:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3)
                p_start = _perturb_log(p_start, 0.01, rng)
            elif i == 4:
                p_start = _perturb_log(best_seq, 0.08, rng)
                p_start = _perturb_fourier(p_start, rng, n_modes=5)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=12)

            remaining = deadline - 25 - time.monotonic()
            if remaining < 5:
                break
            a_r = _run_lbfgs_aggressive(p_start, time.monotonic() + max(5, remaining - 5),
                                        min_time_remaining=5)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, label)

    # === Phase 8: Multi-n exploration (try different n from best) ===
    print(f"\n[Phase 8] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        n_candidates = [n_best]
        if n_best != 8192:
            n_candidates.append(8192)
        if n_best != 16384:
            n_candidates.append(16384)
        if n_best != 4096:
            n_candidates.append(4096)

        for n_try in n_candidates:
            if time.monotonic() > deadline - 15:
                break
            if n_try == n_best:
                a_init = best_seq.copy()
            else:
                x_old = np.linspace(0, 1, n_best)
                x_new = np.linspace(0, 1, n_try)
                a_init = np.interp(x_new, x_old, best_seq)
                a_init = np.maximum(a_init, 1e-10)
                a_init = a_init / a_init.sum()

            remaining = deadline - 10 - time.monotonic()
            if remaining < 3:
                break
            a_opt = _run_lbfgs_aggressive(a_init, time.monotonic() + max(2, remaining - 3),
                                          min_time_remaining=3)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"multin_{n_try}")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        # Very low beta L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.000005
        remaining = deadline - 2 - time.monotonic()
        if remaining > 2:
            max_iter = min(5000, max(100, int(remaining / 0.005)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 100, 'maxls': 300}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_ultra_lbfgs")
            except Exception:
                pass

        # Final subgradient
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 1, n_max_steps=3000, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
