    # Build constraint matrix for dual
    # Dual: min M * sum(lambda) s.t. A @ lambda >= b, lambda >= 0
    # A[j, i] = 1 if position j is covered by peak active_indices[i]
    # b[j] = 1/a_j
    
    # For peak k, it covers positions j where max(0, k-n+1) <= j <= min(k, n-1)
    A_mat = np.zeros((n, m))
    for i, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A_mat[j_min:j_max + 1, i] = 1.0
    
    # Clip a to avoid division by zero
    a_clipped = np.maximum(a, 1e-12)
    b_vec = 1.0 / a_clipped
    
    # Solve: min c^T x s.t. A_ub x <= b_ub, 0 <= x
    # Our constraint: A_mat @ lambda >= b_vec => -A_mat @ lambda <= -b_vec
    c_obj = M * np.ones(m)
    
    try:
        res = linprog(c_obj, A_ub=-A_mat, b_ub=-b_vec, 
                      bounds=[(0, None)] * m, method='highs')
        if not res.success:
            return None
        
        lambda_vals = res.x
        
        # Recover primal g using complementary slackness
        # g_j > 0 only where dual constraint j is tight
        A_lambda = A_mat @ lambda_vals  # shape (n,)
        tight = (a_clipped * A_lambda) >= (1.0 - 1e-6)
        
        if not np.any(tight):
            return None
        
        # For tight positions, set g_j proportional to something
        # Actually, the primal optimum has sum(g) = M * sum(lambda)
        # And g_j > 0 only at tight positions
        # The simplest valid g: set g_j = 1/a_j for tight j
        # Then conv(a, g)[k] = sum_{j tight, j in range_k} a_j * (1/a_j) = count of tight j in range_k
        # This might not equal M, so we need to scale
        
        g = np.zeros(n)
        g[tight] = 1.0 / a_clipped[tight]
        
        # Scale so that max(conv(a,g)[k] for k in active) = M
        c_g = _fft_convolve(a, g)
        max_cg = max(c_g[k] for k in active_indices)
        if max_cg > 1e-15:
            g = g * (M / max_cg)
        
        s_g = g.sum()
        if s_g < 1e-15:
            return None
        
        # Check that sum(g) > 1 (descent direction)
        if s_g <= 1.0 + 1e-9:
            return None
        
        return g / s_g  # Normalize to sum=1
    
    except Exception:
        return None


def _mirror_descent(a_init, deadline, T_start, T_end, lr_start, lr_end,
                    max_iter=200000, eval_interval=200,
                    use_lp=True, lp_interval=10, active_k=5):
    """
    Mirror descent in log-space with optional LP direction updates.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c = _fft_convolve(a_init, a_init)
    
    t = 0
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        
        if t % lp_interval == 0 and use_lp:
            # Use LP direction
            c = _fft_convolve(a, a)
            c_max = float(c.max())
            top_k_idx = np.argsort(c)[::-1][:active_k]
            # Only use peaks that are close to max
            threshold = c_max * 0.95
            active = top_k_idx[c[top_k_idx] >= threshold]
            if len(active) > 0:
                g_lp = _lp_direction(a, c, active, c_max)
                if g_lp is not None:
                    # Mix with gradient direction
                    a_new = (0.5 * a + 0.5 * g_lp)
                    a_new = a_new / a_new.sum()
                    # Update z to match
                    z = np.log(np.maximum(a_new, 1e-15))
                    z = z - z.mean()
                    sc = _score_norm(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        patience = 0
                    else:
                        patience += 1
                    if patience > 10:
                        lr_mult *= 0.7
                        patience = 0
                        z = np.log(np.maximum(best_a, 1e-15))
                        z = z - z.mean()
                    continue
        
        # Gradient step
        if t % 50 == 0:
            c = _fft_convolve(a, a)
        
        g = _smoothed_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        z_new = z - lr * z_grad
        z_new = np.clip(z_new, -70.0, 70.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_a = a_final.copy()
        best_score = sc_final
    
    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=200):
    """Coordinate descent focusing on active peak contributors."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        tol = c_max * 0.98
        active = np.where(c >= tol)[0]
        
        for k in active:
            # Find which j's contribute most to c_k
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue
            
            # Sort by contribution
            contribs = [(j, abs(best_a[j] * best_a[k - j]) if 0 <= k - j < n else 0) for j in js]
            contribs.sort(key=lambda x: -x[1])
            
            # Try adjusting top contributors
            for j, _ in contribs[:10]:
                for scale in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
