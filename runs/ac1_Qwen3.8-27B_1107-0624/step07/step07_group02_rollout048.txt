"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses peak-focused gradient descent, adaptive log-sum-exp L-BFGS,
structural perturbations, and multi-n exploration with aggressive
step sizes to squeeze below 1.5027.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _convg(a):
    return _full_convolve(a, a)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs_v2(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with more aggressive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.0005:
        gap = max_c * 0.0005
    base_beta = max(gap, max_c * 0.001)

    # More aggressive schedule: start at moderate beta, go very small
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _peak_peel_gd(a_init, deadline, top_k=3, max_stall=200):
    """Peak-focused gradient descent: only target the top-k peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.03
    stall = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break

        # Find top-k peaks
        topk_idx = np.argsort(c)[::-1][:top_k]
        # Compute gradient of sum of top-k peaks
        grad = np.zeros(n)
        for k in topk_idx:
            for j in range(n):
                val = 0.0
                if 0 <= k - j < n:
                    val += a[k - j]
                if 0 <= j - k < n:
                    val += a[j - k]
                grad[j] += val

        # Center the gradient (remove mean to preserve sum)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        alpha = min(lr, 0.5 / gnorm)
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.4

        if not improved:
            lr *= 0.7
            if lr < 1e-8:
                break

        # Occasionally adjust top_k
        if stall % 50 == 0 and top_k < 10:
            top_k += 1
        elif stall % 50 == 0 and top_k > 1:
            top_k = max(1, top_k - 1)

    return best_a, best_sc


def _active_set_polish_v2(a_init, deadline, tol_frac_start=0.01,
                          tol_frac_end=0.00002, lr_init=0.04):
    """Improved active-set polish with weighted active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Weighted active set: weight proportional to (c_k - threshold)^2
        tol = tol_frac * c_max
        active_mask = c >= (c_max - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True
            n_active = 1

        # Use squared-excess weights for more focus on peaks
        excess = np.maximum(c - (c_max - tol), 0)
        w = excess ** 2
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[active_mask] = 1.0 / n_active
        else:
            w = w / w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * a * (h - W)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= 300:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.25
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_structural_init(n, kind, rng):
    """Generate various structural initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos1":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos5":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.sin(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "trimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.15) ** 2) + np.exp(-10.0 * (t - 0.5) ** 2) + np.exp(-10.0 * (t - 0.85) ** 2) + 0.01
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-3.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "comb":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 17.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fourier_rich":
        t = np.arange(n)
        a = np.ones(n)
        for freq in [3, 7, 13, 21]:
            amp = 0.08 / freq
            phase = rng.uniform(0, 2 * np.pi)
            a = a + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        # Random low-frequency signal
        k = min(20, n // 4)
        spec = rng.standard_normal(k) + 1j * rng.standard_normal(k)
        a = np.fft.irfft(np.concatenate([spec, spec[1:k//2+1][::-1].conj()]), n)
        a = np.abs(a) + 0.1
        return a / a.sum()
    else:

def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc

            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_peel2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Fine active-set polish ===
    print(f"\n[P5] Active-set polish fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pol2_deadline = min(deadline - 200, time.monotonic() + 100)
        a_pol2, sc_pol2 = _active_set_polish_v2(best_seq, pol2_deadline,
                                                tol_frac_start=0.002, tol_frac_end=0.00003,
                                                lr_init=0.012)
        if sc_pol2 < best_score:
            best_seq = a_pol2.copy()
            best_score = sc_pol2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 150)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 120:
                a_shr2 = _run_lbfgs_v2(best_seq, deadline, min_time_remaining=80.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted([4096, 5120, 6144, 7168, 8192, 9216, 10240, 12288])
    # Ensure we try the parent n
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        remaining_for_n = min(60, deadline - 80 - time.monotonic())
        if remaining_for_n < 15:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)

        # Short L-BFGS
        a_opt = _run_lbfgs_v2(a_init, min(deadline - 60, time.monotonic() + remaining_for_n * 0.6),
                              min_time_remaining=10.0)
        sc_opt = _score(a_opt)

        # Short peak-peel
        if time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _peak_peel_gd(a_opt, min(deadline - 45, time.monotonic() + 20),
                                            top_k=3, max_stall=100)
            if sc_opt2 < sc_opt:
                a_opt, sc_opt = a_opt2, sc_opt2

        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Diverse structural restarts ===
    print(f"\n[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    structural_kinds = ["cos1", "cos5", "bimodal", "trimodal", "asym", "comb", "fourier_rich", "random_smooth"]

    for kind in structural_kinds:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - 40 - time.monotonic()
        if remaining < 20:
            break

        # Mix structural init with parent
        a_struct = _make_structural_init(n_best, kind, rng)
        # Blend with best
        for mix in [0.3, 0.5, 0.7]:
            a_blend = mix * a_struct + (1 - mix) * best_seq
            a_blend = a_blend / a_blend.sum()

            a_opt = _run_lbfgs_v2(a_blend, min(deadline - 30, time.monotonic() + remaining * 0.3),
                                  min_time_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{kind}_m{mix}] t={time.monotonic()-t_start:.1f}s")
                break

    # === Phase 9: Log-perturbation restarts ===
    if time.monotonic() < deadline - 30:
        for sigma in [0.02, 0.05, 0.1]:
            if time.monotonic() > deadline - 25:
                break
            a_pert = _perturb_log(best_seq, sigma, rng)
            remaining = deadline - 25 - time.monotonic()
            if remaining < 10:
                break
            a_opt = _run_lbfgs_v2(a_pert, min(deadline - 20, time.monotonic() + remaining * 0.5),
                                  min_time_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [perturb_{sigma}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Final ultra-fine convergence ===
    print(f"\n[P10] Final convergence (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very fine active-set
        fin_deadline = min(deadline - 3, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_polish_v2(best_seq, fin_deadline,
                                              tol_frac_start=0.0005, tol_frac_end=0.000005,
                                              lr_init=0.004)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny peak-peel
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            a_fin2, sc_fin2 = _peak_peel_gd(best_seq, min(deadline - 2, time.monotonic() + remaining * 0.5),
                                            top_k=1, max_stall=100)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_peak_peel] t={time.monotonic()-t_start:.1f}s")

        # Final L-BFGS with tiny beta
        remaining = deadline - 2 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-16, 'maxcor': 80})
                a_fin3 = _softmax(result.x)
                sc_fin3 = _score(a_fin3)
                if sc_fin3 < best_score:
                    best_seq = a_fin3.copy()
                    best_score = sc_fin3
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs_tiny] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_structural_init(256, "uniform", rng)
            s = final.sum()
        final = final / s
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_structural_init(256, "uniform", rng)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
