def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Active-set LP optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Core innovation: iterative LP-based active set method that directly optimizes
the feasible direction given current convolution peaks, combined with CG-Newton
polishing on Hankel-structured Hessians. Uses adaptive active set management,
dual-based constraint refinement, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _self_conv(a):
    return _fft_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _build_active_matrix(a, active_idx):
    """Build rows of convolution matrix for given active indices."""
    n = len(a)
    active_idx = np.asarray(active_idx, dtype=np.int64)
    m = len(active_idx)
    if m == 0:
        return np.zeros((0, n))
    i_range = np.arange(n)
    k_range = active_idx[:, np.newaxis]
    diff = k_range - i_range[np.newaxis, :]
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((m, n))
    A[valid] = a[diff[valid]]
    return A


def _lp_find_direction(a, c_max, active_idx, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= c_max for k in active, g >= 0."""
    m = len(active_idx)
    if m == 0:
        return a.copy()
    A = _build_active_matrix(a, active_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(m, c_max)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status != 0:
            return None
        g = res.x
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            return None
        # Get dual variables for active set refinement
        duals = None
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            if hasattr(res.ineqlin, 'marginals'):
                duals = res.ineqlin.marginals
        return g, duals
    except Exception:
        return None


def _hankel_matvec(v, w, n):
    """Compute H_W * v where (H_W)_{i,j} = w_{i+j}."""
    v_rev = v[::-1]
    out_len = len(v_rev) + len(w) - 1
    N = _next_pow2(out_len)
    F_v = rfft(v_rev, N)
    F_w = rfft(w, N)
    conv = irfft(F_v * F_w, N)[:out_len]
    return conv[n - 1:2 * n - 1]


def _cg_solve_hankel(w, n, rhs, max_iter=500, tol=1e-14):
    """Solve H_W * x = rhs using CG."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-40:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-40 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new >= rs:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

    a = np.maximum(a_init.copy(), 0).astype(np.float64)

    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = float(2.0 * n * _self_conv(a).max())

        if time.monotonic() > deadline:
            break
        c = _self_conv(a)
        c_max = float(c.max())

        # Adaptive active set: start moderate, tighten over rounds
        frac = max(0.001, 0.02 * (0.95 ** rnd))
        tol = frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Build weight vector
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        # Gradient of sum_{k in S} w_k * c_k
        g = _hankel_matvec(a, w, n)
        g_proj = g - np.mean(g)
        gn = float(np.linalg.norm(g_proj))
        if gn < 1e-15 * n:
            break

        # CG for H_W * x = g (projected)
        Hg = _cg_solve_hankel(w, n, g_proj, max_iter=300)
        H1 = _cg_solve_hankel(w, n, np.ones(n) - np.ones(n).mean() / n, max_iter=200)

        # Newton direction with sum-to-zero constraint
        ones = np.ones(n)
        lam_num = float(ones @ Hg)
        lam_den = float(ones @ H1)
        if abs(lam_den) > 1e-30:
            lam = lam_num / lam_den
        else:
            lam = 0.0
        dx = -Hg + lam * H1
        dx = dx - np.mean(dx)
        dx_norm = float(np.linalg.norm(dx))

        if dx_norm < 1e-20 or not np.isfinite(dx_norm):
            dx = -g_proj / (gn + 1e-30)
            dx = dx - np.mean(dx)

        # Backtracking line search
        accepted = False
        alpha = 1.0
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                alpha *= 0.5

            c_try = _self_conv(a_try)

                accepted = True

        if not accepted:
            # Try smaller gradient step
            alpha = 0.01
            for ls in range(30):
                a_try = a - alpha * g_proj
                a_try = np.maximum(a_try, 0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _self_conv(a_try)
                if float(c_try.max()) < c_max - 1e-14:
                    a = a_try
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break

        sc = float(2.0 * n * _self_conv(a).max())
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _lp_active_set_phase(a_init, deadline, max_iters=200):
    """Main LP-based active set optimization phase."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    max_active = min(2000, 2 * n - 1)
    active_tol = 0.02

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        # Adaptive active set
        tol = active_tol * c_max
        active_mask = c >= (c_max - tol)
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        # Cap active set size
        if len(active_idx) > max_active:
            active_vals = c[active_idx]
            threshold = np.sort(active_vals)[len(active_vals) - max_active]
            keep = active_vals >= threshold
            active_idx = active_idx[keep]

        # Solve LP
        result = _lp_find_direction(a, c_max, active_idx, n)
        if result is None:
            # Relax active set
            active_tol = min(active_tol * 2, 0.5)
            if active_tol >= 0.5:
                break
            continue

        if isinstance(result, tuple):
            g, duals = result
        else:
            g, duals = result, None

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-12:
            # No improvement from LP - tighten or break
            active_tol = max(active_tol * 0.7, 0.001)
            continue

        # Rescale g to sum=1
        g = g / g_sum

        # Line search for step size
        accepted = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc - 1e-12:
                a = a_try
                sc = sc_try
                accepted = True
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    if it % 10 == 0:
                        print(f"    LP iter {it}: score={sc:.10f}, "
                              f"active={len(active_idx)}, tol={active_tol:.4f}")
                break

        if accepted:
            # Tighten active set tolerance slowly
            active_tol = max(active_tol * 0.95, 0.002)
        else:
            # Try relaxing
            active_tol = min(active_tol * 1.5, 0.3)
            if active_tol > 0.3:
                break

        # Optionally refine active set using duals
        if duals is not None and len(duals) == len(active_idx):
            active_dual = np.abs(duals)
            # Remove constraints with very small dual values
            keep_threshold = 1e-6 * max(1.0, np.max(active_dual))
            if np.sum(active_dual < keep_threshold) > 0.3 * len(active_idx):
                keep = active_dual >= keep_threshold
                if np.sum(keep) > 5:
                    active_idx = active_idx[keep]

    return best_a, best_sc


def _continuation_phase(a_init, deadline, max_rounds=30):
    """Continuation method: solve QP with decreasing temperature weights."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = float(2.0 * n * _self_conv(a).max())

    c_cur = _self_conv(a)
    c_max = float(c_max_init if False else c_cur.max())

    # Temperature schedule
    betas = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                      0.015625, 0.0078, 0.0039, 0.00195, 0.000975,
                      0.00049, 0.000245, 0.00012, 0.00006, 0.00003,
                      0.000015, 0.000007, 0.000003, 0.000001])
    base_beta = max(c_max * 0.1, 1e-6)
    betas = base_beta * betas

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        # Compute soft weights
        c = _self_conv(a)
        shifted = (c - c.max()) / beta
        e = np.exp(np.clip(shifted, -500, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        # Solve QP: minimize a^T H_W a s.t. sum(a)=1, a>=0
        # Using projected gradient (simple but effective)
        lr = 1.0 / (2.0 * np.max(np.abs(np.diff(w, prepend=w[0])) + 1e-10) * n + 1e-10)
        lr = min(lr, 0.05)

        for sub in range(50):
            if time.monotonic() > deadline:
                break
            c_sub = _self_conv(a)
            g = _hankel_matvec(a, w, n)
            g_proj = g - np.mean(g)
            g_norm = np.linalg.norm(g_proj)
            if g_norm < 1e-15:
                break
            a_new = a - lr * g_proj
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            a = a_new
            lr = lr * 0.98  # Slow decay

        sc = float(2.0 * n * _self_conv(a).max())
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(t - 0.5) * 4.0)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(np.clip(z - z.max(), -600, 0))
    s = e.sum()
    if s < 1e-300:
        return a.copy()
    return e / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP active set + continuation + CG-Newton polish +
    multi-n exploration. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.3e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
                update_best(parent, _score(parent), "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_parent = len(parent)

    # === Phase 1: LP Active Set (main optimization) ===
    print(f"\n[Phase 1] LP Active Set (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + budget_s * 0.45
    a1, sc1 = _lp_active_set_phase(parent, p1_deadline, max_iters=300)
    update_best(a1, _score(a1), "lp_phase1")

    # === Phase 2: Continuation (soft weights) ===
    print(f"\n[Phase 2] Continuation (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p2_deadline = t_start + budget_s * 0.60
        a2, sc2 = _continuation_phase(best_seq, p2_deadline, max_rounds=30)
        update_best(a2, _score(a2), "continuation")

    # === Phase 3: CG-Newton Polish ===
    print(f"\n[Phase 3] CG-Newton Polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p3_deadline = t_start + budget_s * 0.78
        a3, sc3 = _cg_newton_polish(best_seq, p3_deadline, max_rounds=100)
        update_best(a3, _score(a3), "newton_1")
        # Second round with tighter params
        if time.monotonic() < p3_deadline:
            a3b, sc3b = _cg_newton_polish(a3, p3_deadline, max_rounds=60)
            update_best(a3b, _score(a3b), "newton_2")

    # === Phase 4: LP again from polished point ===
    print(f"\n[Phase 4] LP from polished (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p4_deadline = t_start + budget_s * 0.85
        a4, sc4 = _lp_active_set_phase(best_seq, p4_deadline, max_iters=50)
        update_best(a4, _score(a4), "lp_phase2")

        # Final Newton
        if time.monotonic() < t_start + budget_s * 0.92:
            a5, sc5 = _cg_newton_polish(best_seq, t_start + budget_s * 0.92, max_rounds=80)
            update_best(a5, _score(a5), "newton_final")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p5_deadline = t_start + budget_s * 0.95
        n_tries = [n_parent, n_parent // 2, n_parent * 2]
        n_tries = sorted(set(x for x in n_tries if 1024 <= x <= 32768))
        for n_try in n_tries:
            if time.monotonic() > p5_deadline:
                break
            if n_try == n_parent:
                continue
            a_init = _resample(best_seq, n_try)
            # Quick LP
            a_opt, sc_opt = _lp_active_set_phase(a_init, min(p5_deadline, time.monotonic() + 60), max_iters=30)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}_lp")
            # Quick Newton
            if sc_opt < best_score + 0.003 and time.monotonic() < p5_deadline:
                a_np, sc_np = _cg_newton_polish(a_opt, min(p5_deadline, time.monotonic() + 40), max_rounds=50)
                update_best(a_np, _score(a_np), f"n_{n_try}_np")

    # === Phase 6: Diverse restarts ===
    print(f"\n[Phase 6] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p6_deadline = deadline
        for i in range(3):
            if time.monotonic() > p6_deadline - 10:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.08, rng)
            else:
                p_start = _make_profile(len(best_seq), "edge_heavy")
            a_r, sc_r = _lp_active_set_phase(p_start, min(p6_deadline - 5, time.monotonic() + 40), max_iters=30)
            update_best(a_r, _score(a_r), f"restart_{i}")
            if time.monotonic() < p6_deadline - 5:
                a_rp, sc_rp = _cg_newton_polish(a_r, min(p6_deadline - 2, time.monotonic() + 30), max_rounds=40)
                update_best(a_rp, _score(a_rp), f"restart_{i}_np")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
