This is like saying $a$ is an eigenvector of the "operator" $\sum_{k \in S} \lambda_k \text{Shift}_k$ (with the diagonal correction).

We can use **power iteration**: start with a candidate $a$, compute the "pressure" $p_j = \sum_{k \in S} \lambda_k a_{k-j}$, then update $a_j \propto f(p_j)$ for some function $f$ that pushes $a$ toward equalizing the pressure.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_soft(a, c, n, temp=0.01):

    shifted = (c - c_max) / (temp * c_max)

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _pressure_equalize_momentum(a, deadline, n_iters=300, gamma_init=0.5, momentum_init=0.3):
    """Pressure equalization with momentum and adaptive gamma."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    momentum = np.zeros(n)
    gamma = gamma_init
    mom = momentum_init

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Soft active set: weights proportional to exp(c_k / (temp * c_max))
        temp = 0.005 + 0.01 * (it / n_iters)
        shifted = (c - c_max) / (temp * c_max)
        w = np.exp(np.clip(shifted, -50, 0))
        w = w / w.sum()

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(w, a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Momentum update
        p_mean = np.mean(p)
        ratio = p_mean / p
        target = np.power(ratio, gamma)
        momentum = mom * momentum + (1 - mom) * (target - 1.0)
        a = a * (1.0 + momentum * 0.5)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Adaptive gamma: reduce over time for fine-tuning
        gamma = gamma_init * (0.5 + 0.5 * (1 - it / n_iters))
        mom = min(0.9, momentum_init + 0.01 * it)

        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

def _coordinate_polish_analytical(a, deadline, n_rounds=80):
    """
    Coordinate descent with analytical 1D min-max optimization.
    For each coordinate a_j, the optimal value minimizes
    max_k (2*a_j*a_{k-j} + c_k^{(j)})
    where c_k^{(j)} is the background (without a_j's contribution).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find most influential coordinates
        # Coordinate j contributes a_j * a_{k-j} to c_k for each k
        a_rev = a[::-1]
        # Total peak contribution per coordinate
        peak_indices = np.argsort(c)[-min(20, 2*n-1):]
        contrib = np.zeros(n)
        for k in peak_indices:
            for j in range(n):
                other = k - j
                if 0 <= other < n:
                    contrib[j] += a[j] * a[other]

        # Try top coordinates
        n_try = min(n, max(20, n // 40))
        top_idx = np.argsort(contrib)[-n_try:]

        improved = False
        for j in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue

            c_old = c.max()

            # Compute background: c_k without a_j's contribution
            # c_k = sum_i a_i * a_{k-i}
            # The terms involving a_j are: a_j * a_{k-j} (from i=j) + a_{k-j} * a_j (from i=k-j)
            # If j != k-j: contribution = 2 * a_j * a_{k-j}
            # If j == k-j: contribution = a_j^2
            # So background c_k^{(j)} = c_k - 2*a_j*a_{k-j} (+ a_j^2 if 2j=k)

            # Find optimal a_j to minimize max of (2*a_j*a_{k-j} + bg_k)
            # The peak is where 2*a_j*a_{k-j} + bg_k is maximized
            # We want to find a_j such that the new max is minimized

            # Try a range of values around old_val
            factors = np.array([0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5])
            best_factor = 1.0
            best_new_max = c_old

            for f in factors:
                new_val = old_val * f
                if new_val < 0:
                    continue
                a[j] = new_val
                c_try = _full_convolve(a, a)
                new_max = float(c_try.max())
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_factor = f
                c = c_try  # reuse

            a[j] = old_val * best_factor
            if a[j] < 0:
                a[j] = old_val

            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                improved = True

        if not improved and rnd > 20:
            break

    return best_a, best_sc


def _spectral_anneal(a, deadline, n_steps=200, T_init=0.01, T_final=0.0001):
    """Spectral simulated annealing in Fourier domain."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    log_T = np.linspace(np.log(T_init), np.log(T_final), n_steps)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        T = np.exp(log_T[step])

        # Fourier-domain perturbation
        z = np.log(np.maximum(a, 1e-30))
        z_hat = np.fft.rfft(z)
        n_modes = max(2, n // 100)
        for _ in range(n_modes):
            freq = rng.integers(1, max(2, n // 4))
            if freq < len(z_hat):
                delta = T * rng.standard_normal()
                phase = rng.uniform(0, 2 * np.pi)
                z_hat[freq] += delta * np.exp(1j * phase)
                if freq > 0 and freq < len(z_hat) - 1:
                    z_hat[-freq] = np.conj(z_hat[freq])

        z_new = np.fft.irfft(z_hat, n)
        z_new = z_new - z_new.mean()
        a_try = _softmax(z_new)
        sc_try = _score(a_try)

        # Metropolis acceptance
        if sc_try < best_sc:
            a = a_try
            best_a = a_try.copy()
            best_sc = sc_try
        elif rng.random() < np.exp(-(sc_try - best_sc) / T):
            a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 3 * t * t) + 1.5
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 13 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2 * 8) + 0.15
        return a / a.sum()
    elif kind == "fejer_like":
        # Fejer kernel is (sin(n*x/2)/sin(x/2))^2 / n, always non-negative
        t = np.arange(n)
        a = np.maximum(1.0 - t / (n * 0.8), 0)
        a = a ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_sequence(src, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(src)
    if n_old == n_new:
        return src.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    out = np.interp(x_new, x_old, src)
    out = np.maximum(out, 0)
    s = out.sum()
    if s > 1e-10:
        out = out / s
    else:
        out = _make_profile(n_new, "uniform")
    return out


def _multi_stage_optimize(a_init, deadline, min_remaining=60.0, label=""):
    """Full optimization pipeline: L-BFGS -> pressure -> L-BFGS -> subgrad."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    t0 = time.monotonic()

    # Stage 1: L-BFGS
    if time.monotonic() < deadline - min_remaining - 30:
        a1, sc1 = _run_lbfgs_extended(current_a, deadline, min_time_remaining=min_remaining + 30,
                                       n_beta_steps=15)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 15:
        pe_deadline = min(deadline - min_remaining - 10, time.monotonic() + 120)
        a2, sc2 = _pressure_equalize_momentum(current_a, pe_deadline,
                                              n_iters=200, gamma_init=0.45, momentum_init=0.4)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: L-BFGS again (finer betas)
    if time.monotonic() < deadline - min_remaining - 10:
        a3, sc3 = _run_lbfgs_extended(current_a, deadline, min_time_remaining=min_remaining + 10,
                                       n_beta_steps=10)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        sg_deadline = min(deadline - min_remaining, time.monotonic() + 60)
        a4, sc4 = _subgrad_polish(current_a, sg_deadline, n_max_steps=2000, lr_init=0.015)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Coordinate descent
    if time.monotonic() < deadline - min_remaining + 20:
        cd_deadline = min(deadline - min_remaining + 20, time.monotonic() + 90)
        a5, sc5 = _coordinate_polish_analytical(current_a, cd_deadline, n_rounds=50)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    elapsed = time.monotonic() - t0
    print(f"  [{label}] Final score: {current_sc:.10f} (n={n}, time={elapsed:.1f}s)")
    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search combining L-BFGS, pressure equalization,
    spectral annealing, coordinate descent, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
                _update(parent)
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}, score={_score(parent):.10f}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    min_rem_p1 = 800
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=min_rem_p1, label="main")
    _update(a_main, sc_main)

    # --- Phase 2: Extended pressure equalization on best ---
    print(f"\n[Phase 2] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        pe_deadline = min(deadline - 500, time.monotonic() + 180)
        a_pe, sc_pe = _pressure_equalize_momentum(best_seq, pe_deadline,
                                                  n_iters=400, gamma_init=0.3, momentum_init=0.5)
        _update(a_pe, sc_pe)
        if sc_pe < best_score:
            a_f, sc_f = _run_lbfgs_extended(a_pe, deadline, min_time_remaining=500, n_beta_steps=12)
            _update(a_f, sc_f)

    # --- Phase 3: Spectral annealing for diversity ---
    print(f"\n[Phase 3] Spectral annealing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        sa_deadline = min(deadline - 300, time.monotonic() + 200)
        a_sa, sc_sa = _spectral_anneal(best_seq, sa_deadline, n_steps=300,
                                        T_init=0.008, T_final=0.00005)
        _update(a_sa, sc_sa)
        if sc_sa < best_score + 0.005:
            # Polish the annealed result
            a_sp, sc_sp = _multi_stage_optimize(a_sa, deadline, min_remaining=300, label="anneal_polish")
            _update(a_sp, sc_sp)

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(key=lambda x: -abs(x - len(best_seq)) if best_seq is not None else x)

    for n_try in n_candidates[:4]:  # Try at most 4 different n
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - 150 - time.monotonic()
        if remaining < 40:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        # Initialize by interpolating best
        if best_seq is not None:
            a_init = _interpolate_sequence(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(30, remaining * 0.4)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem, label=f"n={n_try}")
        _update(a_opt, sc_opt)

    # --- Phase 5: Diverse restarts ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        profiles = ["chirp", "double_cos", "gaussian_taper", "fejer_like"]

        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - 40 - time.monotonic()
            if remaining < 30:
                break

            # Blend best with profile
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()

            # Small perturbation
            z = np.log(np.maximum(a_blend, 1e-30))
            z += 0.01 * rng.standard_normal(n_best)
            a_pert = _softmax(z)

            min_rem = max(10, remaining * 0.3)
            a_r, sc_r = _multi_stage_optimize(a_pert, deadline, min_remaining=min_rem,
                                              label=f"restart_{prof}")
            _update(a_r, sc_r)

    # --- Phase 6: Final intensive polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # L-BFGS with very fine betas
        a_fin, sc_fin = _run_lbfgs_extended(best_seq, deadline - 2, min_time_remaining=3,
                                             n_beta_steps=20)
        _update(a_fin, sc_fin)

        if time.monotonic() < deadline - 5:
            # Pressure equalization (final)
            a_pe2, sc_pe2 = _pressure_equalize_momentum(best_seq, deadline - 3,
                                                        n_iters=100, gamma_init=0.2, momentum_init=0.6)
            _update(a_pe2, sc_pe2)

        if time.monotonic() < deadline - 3:
            # Coordinate descent (final)
            a_cd, sc_cd = _coordinate_polish_analytical(best_seq, deadline - 2, n_rounds=30)
            _update(a_cd, sc_cd)

        if time.monotonic() < deadline - 2:
            # One more L-BFGS on the absolutely best
            a_last, sc_last = _run_lbfgs_extended(best_seq, deadline - 1, min_time_remaining=1,
                                                   n_beta_steps=8)
            _update(a_last, sc_last)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
