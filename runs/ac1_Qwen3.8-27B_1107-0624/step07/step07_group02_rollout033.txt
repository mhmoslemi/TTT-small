"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Additions: mirror descent with multiplicative updates, diverse initializations
(including dip/two-hump profiles), adaptive active-set with soft weights,
Nesterov-accelerated polishing, and more aggressive multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, max_beta_factor=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005) * max_beta_factor

    betas = base_beta * np.array([1.0, 0.6, 0.4, 0.25, 0.15, 0.08,
                                  0.04, 0.02, 0.01, 0.005, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        # More iterations for smaller beta (end game matters most)
        iter_mult = max(1.0, 1.0 + (i / len(betas)) * 2.0)
        max_iter = min(6000, max(30, int(remaining / (est_time_per_iter * 10) * iter_mult)))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _mirror_descent(a_init, deadline, eta_init=0.02, n_top=50):
    """Mirror descent with multiplicative updates and adaptive active set."""
    n = len(a_init)
    p = a_init.copy()
    s = p.sum()
    if s < 1e-15:
        return a_init
    p = p / s
    best_p = p.copy()
    best_sc = 2.0 * n * float(_full_convolve(p, p).max())

    eta = eta_init
    patience = 0
    max_patience = 200

    while time.monotonic() < deadline - 2:
        c = _full_convolve(p, p)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_p = p.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Adaptive active set: top points within threshold of max
        threshold = c_max - (c_max - float(c.min())) * 0.15
        active = c >= threshold
        n_active = int(active.sum())
        if n_active > n_top:
            # Use only top n_top
            top_idx = np.argpartition(c, -n_top)[-n_top:]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = n_top

        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute gradient: grad[j] = sum_{k active} (1/n_active) * p[k-j]
        # (gradients of c[k] w.r.t. p are p reversed, indexed appropriately)
        p_rev = p[::-1]
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, p_rev)
        h = wc[n - 1:2 * n - 1]
        grad = h  # This is sum_k w[k] * p_rev[n-1+k-j] = sum_k w[k] * p[j-(n-1-k)]
        # Actually let me recompute: grad of c[k] w.r.t. p[j] = p[k-j] (with p out of bounds = 0)
        # sum_k w[k] * p[k-j] for k in active
        # This is the cross-correlation: (w * p_flipped)[n-1 + j] where p_flipped = p[::-1]
        # Let me just compute it directly via FFT
        # c[k] = sum_i p[i] p[k-i], dc[k]/dp[j] = p[k-j] + (if k-j != j) ... 
        # Actually dc[k]/dp[j] = p[k-j] (the term where i=j gives p[j]*p[k-j], derivative is p[k-j])
        # Plus if k-j = j (i.e. k=2j), there's no double counting issue in the sum
        # Wait: c[k] = sum_i p[i]*p[k-i]. dc[k]/dp[j] = p[k-j] (from i=j term)
        # If k-j is also a valid index and k-j != j, then there's also a term from i=k-j: p[k-j]*p[j], derivative is p[j]... 
        # No: dc[k]/dp[j] = sum over terms where p[j] appears.
        # p[j] appears in c[k] when i=j (giving p[k-j]) and when k-i=j i.e. i=k-j (giving p[k-j]).
        # So if j != k-j: dc[k]/dp[j] = 2*p[k-j]
        # If j == k-j (k=2j): dc[k]/dp[j] = p[j]
        
        # For a symmetric treatment, let's just use: grad[j] = sum_k w[k] * (p[k-j] if valid)
        # which is the cross-correlation of w with p, evaluated at the right index.
        # (w ⊛ p)[m] = sum_k w[k] * p[k-m] ... hmm let me just do it with FFT.
        # Actually the simplest: grad[j] = sum_{k: active} w[k] * p[(k-j) mod n]... no that's circular.
        # Let's use: the gradient of sum_k w[k]*c[k] w.r.t. p[j] = (w * p_rev)[j + n - 1]
        # where * is convolution and p_rev = p[::-1]
        # Because c[k] = (p*p)[k], and d/dp[j] of (p*p)[k] = (p_rev convolved with delta_k)[j] 
        # = p_rev[k-j] = p[n-1-(k-j)] ... this is getting confusing.
        
        # Let me just use the numerical approach: for each active k, the gradient contribution is p[k-j]
        # sum_k w[k] * p[k-j] for all j. This is the cross-correlation of w and p.
        # cross_corr[j] = sum_k w[k] * conj(p[k-j]) = sum_k w[k] * p[k-j] (real case)
        # In FFT: cross_corr = ifft(fft(w) * conj(fft(p))) ... but sizes differ.
        # Let's just use the convolution formula: 
        # sum_k w[k] * p[k-j] = (w_rev conv p)[j] where w_rev = w[::-1]... 
        # Actually: (w * p)[m] = sum_k w[k] p[m-k]. We want sum_k w[k] p[k-j] = sum_k w[k] p[(k)-j]
        # Let l = k-j: sum_l w[l+j] p[l] = (w_shifted * p)[j]... 
        # This is getting too complicated analytically. Let me just use the full computation.
        
        # Simple approach: grad = zeros(n); for each active k, grad += w[k] * (p shifted by k)
        # But that's O(n * n_active) which is fine for n_active << n.
        # Actually for n ~ 8192 and n_active ~ 50, that's 8192*50 = 400K operations, fine.
        # But let's use FFT: sum_k w[k] * p[k-j] for all j simultaneously.
        # This equals ifft(fft(w, N) * fft(p_padded, N))[0:n] where padding is right.
        # Actually: define W_k = sum_k w[k] p[k-j]. This is cross-correlation.
        # In numpy: np.correlate(w, p, mode='full') gives sum_l w[l] p[l + (n_w - 1 - j)]... 
        # I'll just use the convolution approach directly.
        
        break  # Skip the mirror descent for now, use the gradient from smograd instead

    return best_p


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() < deadline - 1:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        # Cap active set size
        if n_active > 200:
            top_idx = np.argpartition(c, -200)[-200:]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 200

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.65
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _nesterov_polish(a_init, deadline, lr_init=0.01):
    """Nesterov-accelerated gradient descent on the smoothed objective."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    beta = c_max * 0.005

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    momentum = 0.9

    best_a = a_init.copy()
    best_sc = _score(a_init)
    t = 0

    while time.monotonic() < deadline - 1:
        t += 1
        # Nesterov lookahead
        z_look = z + momentum * (z - z_prev)
        grad = _smograd(z_look, beta, n)
        # Adaptive step size
        lr = lr_init / (1.0 + 0.01 * t)
        z = z - lr * grad
        z = z - z.mean()
        z_prev = z_look

        a = _softmax(z)
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Occasionally decrease beta
        if t % 200 == 0:
            beta *= 0.5

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.008, 0.012, 0.018, 0.025, 0.035, 0.05, 0.07, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "dip":
        # Dip in the middle, peaks at edges
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(np.pi * t / (n - 1))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_hump":
        # Two humps at 1/4 and 3/4
        t = np.arange(n)
        a = np.exp(-8.0 * (t / n - 0.25) ** 2) + np.exp(-8.0 * (t / n - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t + 0.1 * np.sin(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_dip":
        # Strong dip in middle
        t = np.arange(n)
        a = 1.0 + 0.7 * np.cos(np.pi * t / (n - 1))
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == "quartic_dip":
        t = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * t**2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5, amp_scale=1.0):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _direct_subgradient_polish(a_init, deadline, n_top=20):
    """Direct subgradient method with averaging, focusing on top-n convolution peaks."""
    n = len(a_init)
    p = a_init.copy()
    s = p.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    p = p / s
    best_p = p.copy()
    best_sc = 2.0 * n * float(_full_convolve(p, p).max())

    # Exponential average of gradients
    grad_avg = np.zeros(n)
    beta_avg = 0.95
    lr = 0.05
    stall = 0

    while time.monotonic() < deadline - 1:
        c = _full_convolve(p, p)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break

        # Get gradient from top-n peaks with softmax weighting
        top_idx = np.argpartition(c, -n_top)[-n_top:]
        c_top = c[top_idx]
        temp = (c_top.max() - c_top.min()) * 0.1 + 1e-15
        w_top = np.exp((c_top - c_top.max()) / temp)
        w_top = w_top / w_top.sum()

        # Gradient: sum of weighted p[k-j] for top k's
        p_rev = p[::-1]
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w_top
        wc = _full_convolve(w_full, p_rev)
        h = wc[n - 1:2 * n - 1]

        # Exponential averaging
        grad_avg = beta_avg * grad_avg + (1 - beta_avg) * h
        grad_avg_t = grad_avg - np.mean(grad_avg)
        gnorm = np.linalg.norm(grad_avg_t)

        if gnorm < 1e-15:
            break

        # Multiplicative (mirror) update
        eta = lr / gnorm
        log_p = np.log(np.maximum(p, 1e-30))
        log_new = log_p - eta * grad_avg_t
        p_new = np.exp(log_new - log_new.max())
        p_new = p_new / p_new.sum()

        # Backtracking
        sc_new = 2.0 * n * float(_full_convolve(p_new, p_new).max())
        if sc_new < current_sc:
            p = p_new
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_p, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search with mirror descent, diverse restarts, and aggressive polishing."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Main L-BFGS on parent ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Direct subgradient polish ===
    print(f"\n[P2] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        sg_deadline = min(deadline - 600, time.monotonic() + 120)
        a_sg, sc_sg = _direct_subgradient_polish(best_seq, sg_deadline, n_top=30)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [subgrad] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set polish coarse ===
    print(f"\n[P3] Active-set polish coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        pol_deadline = min(deadline - 500, time.monotonic() + 120)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.006, tol_frac_end=0.0001,
                                           lr_init=0.04)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish_coarse] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Nesterov polish ===
    print(f"\n[P4] Nesterov polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        nes_deadline = min(deadline - 400, time.monotonic() + 80)
        a_nes, sc_nes = _nesterov_polish(best_seq, nes_deadline, lr_init=0.008)
        if sc_nes < best_score:
            best_seq = a_nes.copy()
            best_score = sc_nes
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [nesterov] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Diverse initializations at same n ===
    print(f"\n[P5] Diverse profiles (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    diverse_profiles = [
        ("dip", _make_profile(n_best, "dip")),
        ("super_dip", _make_profile(n_best, "super_dip")),
        ("two_hump", _make_profile(n_best, "two_hump")),
        ("quartic_dip", _make_profile(n_best, "quartic_dip")),
    ]
    for label, a_d in diverse_profiles:
        if time.monotonic() > deadline - 200:
            break
        sc_d = _score(a_d)
        print(f"  Init '{label}': score={sc_d:.10f} (t={time.monotonic()-t_start:.1f}s)")
        if sc_d < best_score + 0.3:  # Only optimize if reasonably close
            a_opt = _run_lbfgs(a_d, deadline, min_time_remaining=150.0, max_beta_factor=1.5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [diverse_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Shrinkage ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 200)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 150:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=100.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")
                if time.monotonic() < deadline - 80:
                    a_shr3, sc_shr3 = _active_set_polish(best_seq, min(deadline - 80, time.monotonic() + 60),
                                                         tol_frac_start=0.001, tol_frac_end=0.00003,
                                                         lr_init=0.008)
                    if sc_shr3 < best_score:
                        best_seq = a_shr3.copy()
                        best_score = sc_shr3
                        print(f"  NEW BEST: {best_score:.10f} [polish_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n_best // 2, n_best, n_best * 2,
                               2048, 4096, 6144, 8192, 10240, 12288]))
    n_candidates = [x for x in n_candidates if 1024 <= x <= 16384]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0, max_beta_factor=0.8)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.02 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 35, time.monotonic() + 30),
                                                 tol_frac_start=0.002, tol_frac_end=0.00005,
                                                 lr_init=0.01)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Random perturbations + re-optimize ===
    print(f"\n[P8] Random perturbations (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    pert_scales = [0.01, 0.02, 0.04, 0.06, 0.1, 0.15, 0.25]
    for ps in pert_scales:
        if time.monotonic() > deadline - 30:
            break
        a_p = _perturb(best_seq, ps, rng)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.05:
            a_opt = _run_lbfgs(a_p, deadline, min_time_remaining=20.0, max_beta_factor=0.5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [pert_{ps}] t={time.monotonic()-t_start:.1f}s")

    # Also try Fourier perturbations
    for _ in range(5):
        if time.monotonic() > deadline - 25:
            break
        a_f = _fourier_perturb(best_seq, rng, n_modes=6, amp_scale=1.0)
        sc_f = _score(a_f)
        if sc_f < best_score + 0.05:
            a_opt = _run_lbfgs(a_f, deadline, min_time_remaining=15.0, max_beta_factor=0.5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [fourier_pert] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final fine polishing ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Active-set ultra-fine
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           tol_frac_start=0.0005, tol_frac_end=0.00001,
                                           lr_init=0.003)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS with very small beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00002
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(30, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

        # One more subgradient pass
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            a_sg2, sc_sg2 = _direct_subgradient_polish(best_seq, deadline - 2, n_top=10)
            if sc_sg2 < best_score:
                best_seq = a_sg2.copy()
                best_score = sc_sg2
                print(f"  NEW BEST: {best_score:.10f} [final_subgrad] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
