def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

  f[n-1 : k-n : -1] would give [f[n-1], f[n-2], ..., f[k-n+1]] if we stop at k-n+1.
  In Python: f[n-1 : k-n : -1] gives elements from index n-1 down to k-n+1 (exclusive of k-n).
  That's [f[n-1], f[n-2], ..., f[k-n+1]], which has length n-1-(k-n+1)+1 = 2n-k-1 = 2n-1-k. ✓

"""
Optimizer using LP-based iterative refinement (Fichter/Weiszfeld style)
combined with gradient methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.

The key innovation is using linear programming to find the optimal direction
at each step, following the approach from the additive combinatorics literature.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    h = wc[n - 1:2 * n - 1][::-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _make_lp_matrix(f):
    """Build the dense constraint matrix for the LP."""
    n = len(f)
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n))
    for k in range(n):
        A[k, :k + 1] = f[k::-1]
    for k in range(n, n_rows):
        i_start = k - n + 1
        length = 2 * n - 1 - k
        A[k, i_start:i_start + length] = f[n - 1:n - 1 - length:-1]
    return A


def _solve_lp(f, M, time_limit=60):
    """Solve: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = _make_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            g = result.x
            g = np.maximum(g, 0)
            return g
    except Exception as e:
        print(f"    LP failed: {e}")
    return None


def _lp_iteration(f, t_values=None, time_limit=60):
    """Perform one LP-based iteration. Returns (new_f, best_score)."""
    n = len(f)
    f_sum = float(f.sum())
    c = _full_convolve(f, f)
    M = float(c.max())

    g = _solve_lp(f, M, time_limit=time_limit)
    if g is None:
        return f, _score(f)

    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return f, _score(f)

    # Rescale g to match f's sum
    g_scaled = g * (f_sum / g_sum)

    if t_values is None:
        t_values = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7]

    best_new = f
    best_sc = _score(f)
    for t in t_values:
        f_new = (1 - t) * f + t * g_scaled
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s  # normalize
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_new = f_new.copy()

    return best_new, best_sc


def _lp_cycle(f_init, deadline, max_iter=100, reserve_time=200):
    """Run multiple LP iterations."""
    n = len(f_init)
    f_cur = f_init.copy()
    f_cur = f_cur / f_cur.sum()
    best_f = f_cur.copy()
    best_sc = _score(f_cur)

    time_limit = 30  # seconds per LP
    if n > 4096:
        time_limit = 60

    for it in range(max_iter):
        if time.monotonic() > deadline - reserve_time:
            break
        if time.monotonic() + time_limit > deadline - reserve_time:
            break

        t0 = time.monotonic()
        f_new, sc_new = _lp_iteration(f_cur, time_limit=time_limit)
        elapsed = time.monotonic() - t0

        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_f = f_new.copy()
            f_cur = f_new.copy()

        if it % 10 == 0 or sc_new < best_sc:
            print(f"    LP iter {it}: score={sc_new:.10f}, "
                  f"improvement={_score(f_init)-sc_new:.2e}, "
                  f"time={elapsed:.1f}s, total={time.monotonic()-t0+ (time.monotonic()-t0)*0:.1f}s")

        # Check for convergence
        if _score(f_cur) >= best_sc - 1e-14:
            # No improvement possible with current approach
            # Try a perturbation
            rng = np.random.default_rng(it * 12345 + 67890)
            z = np.log(np.maximum(f_cur, 1e-18))
            z = z + 0.01 * rng.standard_normal(n)
            z = z - z.mean()
            f_pert = _softmax(z)
            f_cur = f_pert

    return best_f, best_sc


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.2)
        max_iter = min(10000, max(100, int(remaining / (0.004 * scale * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=50000, lr_init=0.06):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    stall_threshold = max(10, k_active * 3)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)

        w_rev = w[::-1]
        wc = _full_convolve(w_rev, a)
        h = wc[n - 1:2 * n - 1][::-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = lr * a.sum()
        z = z - step * grad / (a * n)
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= 0.6
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(t_old, a)
        a_interp = interp(t_new)
    except Exception:
        a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _kkt_refine(a, deadline, k_active=4, max_iter=200):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        w_rev = w[::-1]
        wc = _full_convolve(w_rev, a)
        h = wc[n - 1:2 * n - 1][::-1]
        W = float(np.dot(w, c))

        g = h - W
        g = g - g.mean()

        alpha = 1.0
        improved = False
        for _ in range(20):
            a_try = a - alpha * g / n
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:
            break

    return best_a, best_sc
