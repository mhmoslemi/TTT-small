def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

Coordinate descent optimizer with incremental FFT convolution updates.
Uses exact 1D optimization per coordinate, multi-start initialization,
and random restarts to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Assumes sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _norm(a):

    a = np.maximum(a, 1e-8)

    a = 1.0 + sigma * rng.standard_normal(n)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _block_init(n, num_blocks, rng):

        end = start + block_size if i < num_blocks - 1 else n
