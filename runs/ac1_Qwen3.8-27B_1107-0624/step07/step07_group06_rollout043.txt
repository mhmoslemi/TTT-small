"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses variance minimization warm-start, L-BFGS with log-sum-exp smoothing,
Newton active-set with CG solve, subgradient polish, multi-n exploration,
and diverse restarts. Key improvements: corrected Hessian structure,
adaptive active-set width, momentum-based subgradient, and better time management.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    """Compute H*v where H[i,j] = 2*active[i+j]. Uses FFT."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1].copy()
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=300, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha < 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x

    a = a_init.copy()
    a = np.maximum(a, 0)

    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(active_tol * c_max, 1e-14)
        active = (c >= (c_max - tol)).astype(np.float64)
        if active.sum() == 0:
            active[int(np.argmax(c))] = 1.0

        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:

        mu = 1e-6 * (1.0 + it * 0.1)

        def hess_vec(v):
            return _hessian_matvec(v, active, n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=200, tol=1e-10)
        dx = dx - np.mean(dx)

        f_old = float(c[active > 0.5].sum())
        alpha_ls = 1.0
        a_new = a.copy()
        for ls in range(60):

            a_try = np.maximum(a_try, 1e-30)

            if s_try < 1e-15:

            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active > 0.5].sum())
            if f_try < f_old - 1e-14 * alpha_ls * gnorm:

                break

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 80, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0.01:
                a_new = a_new / s
                sc_new = _score(a_new)
                if sc_new < current_sc - 1e-15:
                    current_a = a_new
                    current_sc = sc_new
                    z = np.log(np.maximum(current_a, 1e-30))
                    z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    momentum = np.zeros(n)
    beta_mom = 0.8

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.003)
        step = lr / (1.0 + 0.0005 * i)
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        z = z - step * momentum
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0 or i == n_max_steps - 1:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _variance_warm_start(a_init, deadline, n_steps=400):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.003
    momentum = np.zeros(n)
    beta_mom = 0.7

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        mean_c = np.mean(c)
        centered = c - mean_c
        dcd_a = _full_convolve(a[::-1], centered)
        g = 2.0 * dcd_a[n - 1:2 * n - 1]
        g = g - np.mean(g)
        momentum = beta_mom * momentum + (1 - beta_mom) * g
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new
        if i % 20 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        lr *= 1.002

    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.04 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_target):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        update_best(parent, _score(parent), "init_uniform")

    # === Phase 1: Variance warm-start ===
    print(f"\n[Phase 1] Variance warm-start (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        v_deadline = min(deadline - 600, time.monotonic() + 100)
        a_v, sc_v = _variance_warm_start(best_seq, v_deadline, n_steps=300)
        update_best(a_v, sc_v, "variance_ws")
        if sc_v < best_score + 0.001:
            a_v2 = _run_lbfgs(a_v, deadline, min_time_remaining=500)
            update_best(a_v2, _score(a_v2), "variance_ws_lbfgs")

    # === Phase 2: Main L-BFGS ===
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_main = _run_lbfgs(best_seq, deadline, min_time_remaining=500)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Newton active-set (multiple widths) ===
    print(f"\n[Phase 3] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        for aw in [0.01, 0.005, 0.002]:
            if time.monotonic() > deadline - 400 - 60 * (1 - aw / 0.01):
                break
            n_deadline = min(deadline - 400, time.monotonic() + 150)
            a_newton, sc_newton = _newton_active_set(
                best_seq, n_deadline, max_iter=200, active_tol=aw)
            update_best(a_newton, sc_newton, f"newton_aw{aw}")
            if sc_newton < best_score:
                # Follow up with L-BFGS from improved point
                a_fn = _run_lbfgs(a_newton, deadline, min_time_remaining=300)
                update_best(a_fn, _score(a_fn), f"lbfgs_after_newton_aw{aw}")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline,
                                       n_max_steps=4000, lr_init=0.018)
        update_best(a_sg, sc_sg, "sg_polish")
        if sc_sg < best_score + 0.0005:
            a_sg2, sc_sg2 = _newton_active_set(a_sg, min(deadline - 300,
                                                         time.monotonic() + 100),
                                                max_iter=100, active_tol=0.003)
            update_best(a_sg2, sc_sg2, "newton_after_sg")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if best_seq is not None and len(best_seq) != n_try:
            a_init = _resample(best_seq, n_try)
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = np.ones(n_try) / n_try

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80,
                           n_beta_stages=8)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if sc_opt < best_score + 0.001 and time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _newton_active_set(
                a_opt, min(deadline - 50, time.monotonic() + 80),
                max_iter=100, active_tol=0.004)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 6: Coarse-to-fine ===
    print(f"\n[Phase 6] Coarse-to-fine (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        if n_best >= 2048:
            n_coarse = max(256, n_best // 8)
            a_coarse = _resample(best_seq, n_coarse)
            a_coarse_opt = _run_lbfgs(a_coarse, deadline, min_time_remaining=60,
                                      n_beta_stages=6)
            a_up = _resample(a_coarse_opt, n_best)
            if time.monotonic() < deadline - 40:
                a_cf = _run_lbfgs(a_up, deadline, min_time_remaining=30,
                                  n_beta_stages=6)
                update_best(a_cf, _score(a_cf), "coarse_to_fine")

    # === Phase 7: Diverse restarts ===
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb_log(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=7)
            elif i == 2:
                p_start = _perturb_log(best_seq, 0.05, rng)
                p_start = _perturb_fourier(p_start, rng, n_modes=3)
            elif i == 3:
                p_start = _perturb_fourier(best_seq, rng, n_modes=15)
            else:
                p_start = _perturb_log(best_seq, 0.01, rng)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15,
                             n_beta_stages=6)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _newton_active_set(
                    a_r, min(deadline - 8, time.monotonic() + 20),
                    max_iter=50, active_tol=0.003)
                update_best(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 8: Final combined polish ===
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Subgradient
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 5,
                                         n_max_steps=2000, lr_init=0.006)
        update_best(a_fin, sc_fin, "final_sg")

        # Newton
        if time.monotonic() < deadline - 4:
            a_fin2, sc_fin2 = _newton_active_set(
                best_seq, max(deadline - 4, time.monotonic()),
                max_iter=60, active_tol=0.002)
            update_best(a_fin2, sc_fin2, "final_newton")

        # Ultra-low beta L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.015)))

                def fun_final(z_local):
                    return _smoobj(z_local, beta_final, n)

                def jac_final(z_local):
                    return _smograd(z_local, beta_final, n)

                try:
                    result = minimize(
                        fun_final, z, jac=jac_final, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-15, 'maxcor': 40}
                    )
                    a_fin3 = _softmax(result.x)
                    a_fin3 = np.maximum(a_fin3, 0)
                    s = a_fin3.sum()
                    if s > 0.01:
                        a_fin3 = a_fin3 / s
                        update_best(a_fin3, _score(a_fin3), "final_ultra_lbfgs")
                except Exception:
                    pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence score: {sc_eval:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
