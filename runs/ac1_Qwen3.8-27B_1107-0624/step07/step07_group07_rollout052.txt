"""

adaptive L-BFGS with log-sum-exp smoothing, FFT-accelerated peak shaving,
Fourier-space simulated annealing, and multi-length exploration.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lso_value(c, beta):
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)


def _lso_weights(c, beta):
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _gradient_lso(a, beta, n):
    """Gradient of 2*n*LSO(conv(a,a)) w.r.t. a (with softmax parameterization)."""
    c = _full_convolve(a, a)
    w = _lso_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _obj_lso(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    val = _lso_value(c, beta)
    return 2.0 * n * val


def _jac_lso(z, beta, n):
    return _gradient_lso(_softmax(z), beta, n)


def _lbfgs_smoothed(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-16):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _obj_lso(z, beta, n)

    def jac(z):
        return _jac_lso(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 100, 'disp': False})
        return _softmax(result.x), result.success, int(result.nit)
    except Exception:
        return a_init.copy(), False, 0

def _fft_peak_gradient(a, c, n, active_mask):

    # Using FFT: corr = ifft(fft(a) * conj(fft(active))) -- not quite
    # Let me just do it directly with the convolution
    # Actually the simplest: the Jacobian of c[k] w.r.t. a[j] is the element a[k-j] 
    # (from the product rule on sum_j a[j]*a[k-j])

def _peak_shaving_fft(a_init, deadline, max_iters=30000, tol_frac=1e-4):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break

        # Build active mask
        threshold = c_max - tol_frac * c_max
        active = (c >= threshold).astype(np.float64)
        
        # Compute gradient direction
        grad_sum = _fft_peak_gradient(a, c, n, active)
        # We want to DECREASE the active peaks, so direction is -grad_sum
        d = -grad_sum
        d -= d.mean()  # preserve sum
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Backtracking line search
        best_alpha = 0
        best_sc_try = best_sc
        for alpha in [eta, eta*0.5, eta*0.25, eta*0.1, eta*0.05]:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_alpha = alpha

        if best_alpha > 0 and best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = (a + best_alpha * d)
            best_a = np.maximum(best_a, 0.0)
            best_a = best_a / best_a.sum()
            a = best_a
            eta = min(eta * 1.02, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.92
            if eta < 1e-10 or stall > 200:
                break

    return best_a, best_sc

        sc_try = _score(a_try)
        
        if sc_try < current_sc:
            coeffs = cf_new
            current_a = a_try
            current_sc = sc_try
            accept_count += 1
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                print(f"      SA new best: {best_sc:.12f} (T={temp:.4f}, step={step_count})")
        else:
            # Accept with probability exp(-(sc_try - current_sc) / temp)
            delta = sc_try - current_sc
            if delta < 1.0:  # don't accept huge jumps
                prob = np.exp(-delta / max(temp, 1e-10))
                if rng.random() < prob:
                    coeffs = cf_new
                    current_a = a_try
                    current_sc = sc_try
                    accept_count += 1
        
        step_count += 1
        temp = max(min_temp, temp * temp_decay)
        
        if step_count % 1000 == 0:
            print(f"      SA: step={step_count}, T={temp:.4f}, current={current_sc:.10f}, best={best_sc:.10f}")
        
        if temp <= min_temp * 1.01:
            # Annealed, do a local polish
            a_polished = _adaptive_lbfgs(current_a, min(time.monotonic()+60, deadline), reserve=10)
            sc_pol = _score(a_polished)
            if sc_pol < best_sc:
                best_sc = sc_pol
                best_a = a_polished.copy()
                print(f"      SA polish: {best_sc:.12f}")
            break
    
    # Final polish
    a_final = _adaptive_lbfgs(best_a, min(time.monotonic()+120, deadline), reserve=10)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _fourier_mode_search(a_init, deadline, max_modes=12):
    """Try adding individual Fourier modes to reduce the peak."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng = np.random.default_rng(777)
    
    for mode in range(1, max_modes + 1):
        if time.monotonic() > deadline:
            break
        
        for sign in [1.0, -1.0]:
            for amp in [0.01, 0.03, 0.05, 0.08]:
                if time.monotonic() > deadline:
                    break
                
                t = np.arange(n)
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z = z + sign * amp * np.cos(2 * np.pi * mode * t / n + rng.uniform(0, 2*np.pi))
                a_try = _softmax(z)
                sc_try = _score(a_try)
                
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    # Polish this direction
                    a_pol = _adaptive_lbfgs(best_a, min(time.monotonic()+120, deadline), reserve=30)
                    sc_pol = _score(a_pol)
                    if sc_pol < best_sc:
                        best_sc = sc_pol
                        best_a = a_pol.copy()
                        print(f"    Mode {mode} {sign:+.0f} {amp}: {best_sc:.12f}")
    
    return best_a, best_sc


def _coordinate_descent_fast(a_init, deadline, max_rounds=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        idx = int((rnd * 6151) % n)
        jdx = int((rnd * 6151 + 13) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        # Golden section search for optimal split
        best_frac = 0.5
        best_sc_local = best_sc
        a_current = a[idx] / s_pair if s_pair > 0 else 0.5
        
        for frac in [a_current, 0.1, 0.3, 0.5, 0.7, 0.9, 0.5 + 0.05, 0.5 - 0.05,
                      a_current + 0.05, a_current - 0.05]:
            frac = np.clip(frac, 0.01, 0.99)
            a_try = a.copy()
            a_try[idx] = s_pair * frac
            a_try[jdx] = s_pair * (1.0 - frac)
            sc_try = _score(a_try)
            if sc_try < best_sc_local:
                best_sc_local = sc_try
                best_frac = frac

        if best_sc_local < best_sc - 1e-14:
            a[idx] = s_pair * best_frac
            a[jdx] = s_pair * (1.0 - best_frac)
            best_sc = best_sc_local
            best_a = a.copy()

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses adaptive L-BFGS, FFT peak shaving, Fourier-space SA,
    mode search, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score_val:.12f} (n={len(p)}) [{label}] t={elapsed:.1f}s ***")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.12f}")
        try_update(parent, "parent")
    else:
        n0 = 8192
        a0 = np.ones(n0) / n0
        print(f"No parent. Using uniform n={n0}, score={_score(a0):.12f}")
        try_update(a0, "init")

    if best_seq is None:
        best_seq = np.ones(4096) / 4096

    n_cur = len(best_seq)
    print(f"\nStarting optimization: n={n_cur}, score={best_score_val:.12f}")

    # === Phase 1: Adaptive L-BFGS fine-tuning ===
    print(f"\n=== Phase 1: Adaptive L-BFGS (n={n_cur}) ===")
    t_phase = time.monotonic()
    a_p1 = _adaptive_lbfgs(best_seq, deadline - 1200, reserve=800)
    try_update(a_p1, "lbfgs_phase1")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.12f}")

    # === Phase 2: FFT peak shaving ===
    print(f"\n=== Phase 2: FFT Peak Shaving ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 800:
        a_ps, sc_ps = _peak_shaving_fft(best_seq, deadline - 800, max_iters=50000, tol_frac=1e-4)
        try_update(a_ps, "peak_shave_fft")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Another L-BFGS pass (warmed up) ===
    print(f"\n=== Phase 3: L-BFGS round 2 ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 600:
        a_p3 = _adaptive_lbfgs(best_seq, deadline - 600, reserve=400)
        try_update(a_p3, "lbfgs_round2")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Fourier mode search ===
    print(f"\n=== Phase 4: Fourier Mode Search ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 500:
        a_fm, sc_fm = _fourier_mode_search(best_seq, deadline - 500, max_modes=10)
        try_update(a_fm, "fourier_mode")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Fourier-space simulated annealing ===
    print(f"\n=== Phase 5: Fourier SA ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 300:
        a_sa, sc_sa = _fourier_sa(best_seq, deadline - 300, M=12, temp_init=0.015,
                                   temp_decay=0.9997, min_temp=1e-5)
        try_update(a_sa, "fourier_sa")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length exploration ===")
    t_phase = time.monotonic()
    target_lengths = set()
    if n_cur == 8192:
        target_lengths.update([4096, 16384])
    elif n_cur == 4096:
        target_lengths.update([8192])
    else:
        target_lengths.update([4096, 8192])

    for target_n in sorted(target_lengths):
        if time.monotonic() > deadline - 200:
            break
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"    Interpolated score: {sc_interp:.12f}")
        try_update(a_interp, f"interp_n{target_n}")

        if time.monotonic() < deadline - 100:
            a_opt = _adaptive_lbfgs(a_interp, deadline - 100, reserve=50)
            try_update(a_opt, f"lbfgs_n{target_n}")

        if time.monotonic() < deadline - 50:
            a_ps2, _ = _peak_shaving_fft(a_opt if best_seq is None or _score(a_opt) < best_score_val
                                         else best_seq, deadline - 50, max_iters=10000)
            try_update(a_ps2, f"ps_n{target_n}")

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Intensive final optimization on best ===
    print(f"\n=== Phase 7: Final intensive optimization ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 80:
        # L-BFGS with very tight settings
        a_f1 = _adaptive_lbfgs(best_seq, deadline - 60, reserve=30)
        try_update(a_f1, "final_lbfgs")

        # Peak shaving
        if time.monotonic() < deadline - 40:
            a_f2, _ = _peak_shaving_fft(best_seq, deadline - 40, max_iters=20000, tol_frac=1e-5)
            try_update(a_f2, "final_ps")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: Coordinate Descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cd, _ = _coordinate_descent_fast(best_seq, deadline - 30, max_rounds=50000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Final micro-polish ===
    print(f"\n=== Phase 9: Micro-polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 10:
        for cycle in range(5):
            if time.monotonic() > deadline - 10:
                break
            a_m1, _ = _peak_shaving_fft(best_seq, deadline - 10, max_iters=5000, tol_frac=1e-6)
            try_update(a_m1, f"micro_ps_{cycle}")
            
            if time.monotonic() < deadline - 10:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-7
                max_iter = min(3000, max(200, int((deadline - 15 - time.monotonic()) / 0.004)))
                a_m2, _, _ = _lbfgs_smoothed(best_seq, beta, n, max_iter)
                try_update(a_m2, f"micro_lbfgs_{cycle}")
    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.12f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
