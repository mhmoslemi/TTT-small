"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key new ideas: coordinate descent with target-level projection (CD-PL),
binary search on the target max level, asymmetric initial profiles,
aggressive multi-n exploration, and refined Newton active set polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _power_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_sum_p = np.sum(c ** p)
    val = c_sum_p ** (1.0 / p)
    return 2.0 * n * val


def _power_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_p = np.sum(c ** p)
    if c_p < 1e-300:
        return np.zeros_like(a)
    cp1 = c ** (p - 1)
    cp1_sum = np.sum(cp1)
    w = cp1 / cp1_sum
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return (2.0 * n * (p - 1.0) / p) * a * (h - W)


def _cd_target_pass(a, c, target, n):
    """One pass of coordinate projection: for each j, reduce a[j] so affected c_k <= target."""
    reduced = 0
    for j in range(n):
        aj = a[j]
        if aj < 1e-30:
            continue
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        if k_end < k_start:
            continue
        ks_offset = np.arange(k_end - k_start + 1)
        a_idx = ks_offset  # a[k - j] indexed by k - j = 0..k_end-j
        a_vals = a[a_idx]
        c_vals = c[k_start:k_end + 1].copy()
        # Compute base: remove current a_j contribution
        base = c_vals - 2.0 * aj * a_vals
        # Handle k = 2j
        if 2 * j >= k_start and 2 * j <= k_end:
            idx2j = 2 * j - k_start
            base[idx2j] = c_vals[idx2j] - aj * aj
        # Find max x such that base + 2*x*a_vals <= target (for a_vals > 0)
        max_x = aj
        mask = a_vals > 1e-30
        if mask.any():
            bounds = (target - base[mask]) / (2.0 * a_vals[mask])
            max_x = min(max_x, float(bounds.min()))
        if 2 * j >= k_start and 2 * j <= k_end:
            idx2j = 2 * j - k_start
            if base[idx2j] < target:
                quad_bound = np.sqrt(target - base[idx2j])
                max_x = min(max_x, float(quad_bound))
        max_x = max(0.0, max_x)
        if max_x < aj - 1e-14:
            delta = max_x - aj
            # Update convolution
            c[k_start:k_end + 1] += 2.0 * delta * a_vals
            if 2 * j >= k_start and 2 * j <= k_end:
                idx2j = 2 * j - k_start
                c[k_start:k_end + 1] -= delta * a_vals[idx2j]
            a[j] = max_x
            reduced += 1
    return a, c, reduced


def _cd_polish_bisection(a_init, deadline, min_time_remaining=30.0):
    """Binary search on target M using coordinate descent for feasibility."""
    n = len(a_init)
    best_a = a_init.copy()
    s = best_a.sum()
    best_a = best_a / s
    c_init = _full_convolve(best_a, best_a)
    cur_max = float(c_init.max())
    best_score = 2.0 * n * cur_max  # since sum=1
    best_a_copy = best_a.copy()

    lo = cur_max * 0.5
    hi = cur_max

    # Try to find smallest feasible M
    for outer in range(30):
        if time.monotonic() > deadline - min_time_remaining:
            break
        target = (lo + hi) / 2.0
        # Feasibility check: do CD passes to see if we can achieve max <= target
        a_test = a_init.copy()
        a_test = a_test / a_test.sum()
        c_test = _full_convolve(a_test, a_test)
        feasible = False
        max_passes = 5
        for p in range(max_passes):
            if time.monotonic() > deadline - min_time_remaining:
                break
            a_test, c_test, red = _cd_target_pass(a_test, c_test, target, n)
            if c_test.max() <= target * (1 + 1e-10):
                feasible = True
                break
            if red == 0:
                break
        if feasible:
            hi = target
            best_a_copy = a_test.copy()
            new_score = 2.0 * n * float(c_test.max())
            if new_score < best_score:
                best_score = new_score
            print(f"    CD bisection: hi={hi:.10f}, score={best_score:.10f} (pass {outer})")
        else:
            lo = target
            print(f"    CD bisection: lo={lo:.10f} (pass {outer})")

    # Final: do CD with the best target to get a polished solution
    a_final = a_init.copy()
    a_final = a_final / a_final.sum()
    c_final = _full_convolve(a_final, a_final)
    for p in range(3):
        if time.monotonic() > deadline - min_time_remaining:
            break
        a_final, c_final, _ = _cd_target_pass(a_final, c_final, hi, n)

    s_final = a_final.sum()
    if s_final > 0.01:
        sc = 2.0 * n * float(c_final.max()) / (s_final * s_final)
        if sc < best_score:
            best_score = sc
            best_a_copy = a_final.copy()

    return best_a_copy, best_score

    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_polish(a_init, deadline, max_rounds=100):
    """CG-Newton polish on the active set with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance: tighten over rounds
        base_tol = 0.001 * c_max
        tol = base_tol * max(0.01, 1.0 - 0.02 * rnd)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        # Ensure minimum active set size
        if n_active < 3:
            idxs = np.argsort(c)[::-1][:5]
            active = np.zeros(len(c), dtype=bool)
            active[idxs] = True
            n_active = 5
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-14 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=200)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=200)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Backtracking line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.05 * alpha * gn * 0.001:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            alpha = 0.01
            for ls in range(30):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_asymmetric_profile(n, rng, kind="default"):
    """Create an asymmetric initial profile to help break symmetry."""
    if kind == "default":
        t = np.linspace(0, 1, n)
        # Asymmetric: weight shifted left
        center = 0.35 + 0.1 * rng.random()
        width = 0.2 + 0.1 * rng.random()
        a = np.exp(-((t - center) / width) ** 2) + 0.05 * np.ones(n)
        # Add asymmetry
        a *= (1.0 + 0.3 * np.sin(np.pi * t * rng.uniform(1, 3)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_bumps":
        t = np.linspace(0, 1, n)
        c1 = rng.uniform(0.15, 0.35)
        c2 = rng.uniform(0.55, 0.85)
        w1 = rng.uniform(0.08, 0.15)
        w2 = rng.uniform(0.08, 0.15)
        a = np.exp(-((t - c1) / w1) ** 2) + 0.7 * np.exp(-((t - c2) / w2) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t ** rng.uniform(0.5, 2.0) + 0.1
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_polish(a_init, deadline, max_steps=3000):
    """Subgradient with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.015

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0005 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()

        if i % 30 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.04)
            else:
                patience += 1
                lr *= 0.85
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.01
            z = z_new
        else:
            z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining:
    1. L-BFGS with log-sum-exp and power smoothing
    2. Coordinate descent with target projection (CD-PL) - NEW
    3. Newton active set polish
    4. Subgradient polish
    5. Multi-n exploration
    6. Iterative refinement cycles
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_asymmetric_profile(n_parent, rng, "default")
        print(f"No parent, using asymmetric n={n_parent}")
        update_best(parent, _score(parent), "init")

    # ===== Phase 1: L-BFGS log-sum-exp from parent =====
    print(f"\n[Phase 1] L-BFGS log-sum-exp (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs_logsumexp(parent, deadline, min_time_remaining=900.0)
    update_best(a1, _score(a1), "lbfgs_lse")

    # ===== Phase 2: L-BFGS power smoothing =====
    print(f"\n[Phase 2] L-BFGS power (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        a2 = _run_lbfgs_power(best_seq, deadline, min_time_remaining=700.0)
        update_best(a2, _score(a2), "lbfgs_power")

    # ===== Phase 3: Newton active set =====
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 650:
        a3, sc3 = _newton_active_polish(best_seq, min(deadline - 650, time.monotonic() + 200))
        update_best(a3, sc3, "newton_1")

    # ===== Phase 4: Coordinate Descent with Target Projection (NEW) =====
    print(f"\n[Phase 4] CD-PL polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a4, sc4 = _cd_polish_bisection(best_seq, min(deadline - 500, time.monotonic() + 300),
                                       min_time_remaining=400.0)
        update_best(a4, sc4, "cd_pl")

    # ===== Phase 5: Newton after CD =====
    print(f"\n[Phase 5] Newton after CD (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a5, sc5 = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 150))
        update_best(a5, sc5, "newton_2")

    # ===== Phase 6: Another CD pass =====
    print(f"\n[Phase 6] CD-PL pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a6, sc6 = _cd_polish_bisection(best_seq, min(deadline - 300, time.monotonic() + 150),
                                       min_time_remaining=200.0)
        update_best(a6, sc6, "cd_pl_2")

    # ===== Phase 7: Subgradient polish =====
    print(f"\n[Phase 7] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a7, sc7 = _subgrad_polish(best_seq, min(deadline - 200, time.monotonic() + 100))
        update_best(a7, sc7, "subgrad")

    # ===== Phase 8: Multi-n exploration =====
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 12288, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        a_resamp = _resample(best_seq, n_try)
        a_opt = _run_lbfgs_logsumexp(a_resamp, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if time.monotonic() < deadline - 60 and best_seq is not None:
            a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 50, time.monotonic() + 40))
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # ===== Phase 9: Asymmetric restarts =====
    print(f"\n[Phase 9] Asymmetric restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_curr = len(best_seq)
        for i in range(3):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _make_asymmetric_profile(n_curr, rng, "two_bumps")
            elif i == 1:
                # Perturb best
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z + 0.05 * rng.standard_normal(n_curr)
                z = z - z.mean()
                p_start = _softmax(z)
            else:
                p_start = _make_asymmetric_profile(n_curr, rng, "ramp")

            if time.monotonic() > deadline - 25:
                break
            a_r = _run_lbfgs_logsumexp(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"asym_{i}")

            if time.monotonic() < deadline - 15:
                a_cd, sc_cd = _cd_polish_bisection(a_r, deadline - 5, min_time_remaining=5.0)
                update_best(a_cd, sc_cd, f"asym_{i}_cd")

    # ===== Phase 10: Final refinement cycle =====
    print(f"\n[Phase 10] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # CD
        if time.monotonic() < deadline - 8:
            a_f1, sc_f1 = _cd_polish_bisection(best_seq, deadline - 4, min_time_remaining=4.0)
            update_best(a_f1, sc_f1, "final_cd")
        # Newton
        if time.monotonic() < deadline - 3:
            a_f2, sc_f2 = _newton_active_polish(best_seq, deadline - 1, max_rounds=50)
            update_best(a_f2, sc_f2, "final_newton")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_asymmetric_profile(4096, rng, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_asymmetric_profile(8192, rng, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
