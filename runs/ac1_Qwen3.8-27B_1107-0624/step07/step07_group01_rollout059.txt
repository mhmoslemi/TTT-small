"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based peak shaving (Frank-Wolfe style with convex combination steps)
2. Adaptive log-sum-exp L-BFGS with asymmetric initializations
3. Nesterov-accelerated proximal gradient for final polishing
4. Multi-scale strategy (optimize at moderate n, upscale, refine)
5. Fourier-domain spectral flattening as auxiliary step
Key insight: exploit sequence asymmetry to reduce convolution peaks below the
Cauchy-Schwarz bound that applies to palindromic sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _full_correlate(a, b):
    """Compute sum_k a[k]*b[k-j] for each j (cross-correlation)."""
    n = len(a)
    m = len(b)
    # Cross-correlation: r[j] = sum_k a[k] * b[k-j]
    # = conv(a, flip(b))[j + m - 1]
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(np.flip(b), N)
    r = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return r[m - 1:m + n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_max_conv(a, k_active=5):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a using top-k active set."""
    n = len(a)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:k_active]
    w = np.zeros_like(c)
    w[idx] = 1.0 / len(idx)
    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * (correlation of w and a at lag j)
    # = 2 * (w * a_rev)[j + n - 1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return h, w, c, idx


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-12):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
              1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, (n / 4096.0) ** 1.1)
        max_iter = min(15000, max(200, int(remaining / (0.003 * scale * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5001:
            break

    return current_a


def _lp_peak_shaving(a, M, deadline, max_iters=200, t_init=0.3):
    """
    LP-based peak shaving: iteratively solve
      max sum(g) s.t. (a*g)[k] <= M, g >= 0
    then update a = (1-t)*a + t*g/sum(g) with backtracking.
    """
    from scipy.optimize import linprog
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    c = _full_convolve(a, a)
    M = float(c.max())
    best_a = a.copy()
    best_sc = _score(a)
    t = t_init

    # Build constraint matrix A: A[k,j] = a[k-j] if 0 <= k-j < n
    # Size: (2n-1) x n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For row k, A[k,j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1, with a[k-j] = a[0..n-1]
        r = np.arange(j, j + n)
        c_idx = np.full(n, j)
        v = a.copy()
        rows.extend(r.tolist())
        cols.extend(c_idx.tolist())
        vals.extend(v.tolist())

    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP: max 1^T g s.t. A*g <= M, g >= 0
        c_obj = -np.ones(n)  # minimize -sum(g)
        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 30, 'presolve': True})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = res.x
        S = float(g.sum())
        if S < 1e-15:
            break

        g_norm = g / S  # Normalize to sum = 1

        # Backtracking: find t such that score decreases
        improved = False
        for _ in range(8):
            a_try = (1 - t) * best_a + t * g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_a = a_try
                best_sc = sc_try
                c = _full_convolve(a_try, a_try)
                M = float(c.max())
                a = a_try.copy()
                improved = True
                t = min(t * 1.1, 0.8)
                break
            else:
                t *= 0.5

        if not improved:
            # Try even smaller step
            t = t * 0.5
            if t < 0.001:
                # Try larger t (maybe we're at a flat spot)
                t = 0.1
                a_try = (1 - t) * best_a + t * g_norm
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        best_a = a_try
                        best_sc = sc_try
                        c = _full_convolve(a_try, a_try)
                        M = float(c.max())
                        a = a_try.copy()
                        improved = True

        if not improved:
            # No progress, try perturbing
            break

        if it % 20 == 0:
            print(f"    LP iter {it}: score={best_sc:.10f}, M={M:.6f}, t={t:.3f}")

    return best_a, best_sc, M


def _lp_peak_shaving_direct(a, deadline, max_iters=500, t_init=0.2):
    """
    Simpler LP peak shaving without rebuilding the matrix each time.
    Rebuilds matrix only when a changes significantly.
    """
    from scipy.optimize import linprog
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    c = _full_convolve(a, a)
    M = float(c.max())
    best_a = a.copy()
    best_sc = _score(a)
    t = t_init
    last_build = -100

    def build_matrix(a_vec):
        rows = []
        cols = []
        vals = []
        for j in range(n):
            r = np.arange(j, j + n)
            c_idx = np.full(n, j)
            rows.extend(r.tolist())
            cols.extend(c_idx.tolist())
            vals.extend(a_vec.tolist())
        return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    A_mat = build_matrix(a)
    last_build = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if it - last_build >= 20:
            A_mat = build_matrix(a)
            last_build = it

        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20, 'presolve': True})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x, 0)
        S = float(g.sum())
        if S < 1e-15 or S <= 1.0 + 1e-10:
            # LP says we can't do better than current sum
            break

        g_norm = g / S

        improved = False
        for _ in range(6):
            a_try = (1 - t) * best_a + t * g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_a = a_try
                best_sc = sc_try
                c = _full_convolve(a_try, a_try)
                M = float(c.max())
                a = a_try.copy()
                improved = True
                t = min(t * 1.05, 0.7)
                break
            else:
                t *= 0.5

        if not improved:
            t *= 0.7
            if t < 0.005:
                t = 0.15

        if it % 50 == 0:
            print(f"    LP iter {it}: score={best_sc:.10f}, M={M:.6f}, S={S:.4f}, t={t:.3f}")

    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, k_active=5, max_steps=20000, lr_init=0.03):
    """Nesterov-accelerated subgradient method in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    momentum = 0.9
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(y)
        h, w, c, idx = _grad_max_conv(a, k_active)
        W = float(np.dot(w, c))
        grad = a * (h - W)
        grad = grad - grad.mean()  # Remove component along all-ones (sum constraint)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Nesterov step
        z_new = y - lr * grad / (n * gnorm) if gnorm > 0 else y
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        y = z_new
        z = momentum * z + (1 - momentum) * y
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(y)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.01
                stall = 0
            else:
                stall += 1
                lr *= 0.85
                if lr < 1e-10:
                    break
                if stall > 50:
                    break

    a_final = _softmax(y)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _fourier_flatten(a, deadline, max_iters=200):
    """Iteratively flatten the spectrum in Fourier domain."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        C = np.fft.irfft(A * A, N)[:2 * n - 1]  # This is wrong for complex
        # Actually: IFFT(|A|^2) gives circular autocorrelation, not convolution
        # For convolution: IFFT(A * A_conj) where A_conj is the "conjugate" in FFT
        # For real a: conv(a,a) in frequency is A * A (not A * conj(A))
        # Circular conv = IFFT(A * A)
        C_circ = np.fft.irfft(A * A, N)[:2 * n - 1]

        c_max = C_circ.max()
        # Reduce high-power bins
        threshold = c_max * 0.9
        C_flat = np.minimum(C_circ, threshold)
        C_flat = np.maximum(C_flat, 0)
        C_flat = C_flat / C_flat.sum() * C_circ.sum()

        # Inverse: want |A_new|^2 to have inverse = C_flat
        # A_new = sqrt(C_flat) * exp(i*phase(A))
        power_new = np.fft.rfft(C_flat, N)
        # This doesn't work directly. Let's use a different approach.
        # Reduce the peak of C by reducing the corresponding frequency components.
        # C = IFFT(A*A), so to reduce C[k], reduce specific A components.
        # Simple approach: multiply A by a "taper" that reduces high-amplitude bins.
        amp = np.abs(A)
        amp_max = amp.max()
        # Soft-clip the amplitude
        alpha = 0.05
        amp_new = np.where(amp > amp_max * (1 - alpha),
                           np.sqrt(amp ** 2 - alpha * amp_max * (amp - amp_max * (1 - alpha))),
                           amp)
        # Actually simpler: just reduce the peak of the power spectrum
        power = np.abs(A) ** 2
        p_max = power.max()
        power_clipped = np.minimum(power, p_max * 0.95)
        scale = np.sqrt(power_clipped / np.maximum(power, 1e-30))
        A_new = A * scale

        # Preserve DC component (sum constraint)
        A_new[0] = A[0]

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3, asym=False):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    if asym:
        a = a * (0.5 + 0.5 * t / n)  # Asymmetric weight
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric_chirp(n, freq=1.0, eps=0.3):
    """Chirp with built-in asymmetry."""
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n + np.pi / 3)
    # Add linear asymmetry
    a = a * (1.0 + 0.2 * (t / n - 0.5))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2, asym_phase=False):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        phase = 0.5 * f if asym_phase else 0
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5, asym_shift=0.0):
    t = np.linspace(0, 1, n)
    t_shifted = np.clip(t - asym_shift, 0, 1)
    a = np.sin(np.pi * t_shifted) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exp_decay(n, rate=0.5):
    t = np.arange(n)
    a = np.exp(-rate * t / n)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=30.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6

        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 16)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 3 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, n - block_sz))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Mirrored perturbation (flip part)
            a_pert = best_a.copy()
            split = n // 2 + int(rng.integers(-n // 8, n // 8))
            a_pert[split:] = best_a[split:][::-1] * 0.9 + 0.1 * best_a[split:]
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        a_opt = _lbfgs_smooth(a_pert, beta, n, 2000)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Basin hop {h}: new best {best_sc:.10f}")

    return best_a, best_sc

    best_score_val = np.inf

        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")
