`c_k = sum_i a_i * a_{k-i}` (with a_m = 0 for m < 0 or m >= n)

`dc_k/da_j = 2 * a_{k-j}` if `0 <= k-j < n`, else `a_{k-j}` if only one valid... 

Wait, I showed earlier that `dc_k/da_j = 2 * a_{k-j}` where we define `a_m = 0` for out-of-range m. Let me verify:

`c_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a_i * a_{k-i}`

`dc_k/da_j`:
- If `j` is in the range `[max(0,k-n+1), min(k,n-1)]`, then `a_j` appears as `a_i`, and the derivative contributes `a_{k-j}`.
- If `k-j` is in the range `[max(0,k-n+1), min(k,n-1)]`, then `a_j` appears as `a_{k-i}` (with `i = k-j`), and the derivative contributes `a_{k-j}`.

These two conditions are the same! `j` is in `[max(0,k-n+1), min(k,n-1)]` iff `k-j` is in `[max(0,k-n+1), min(k,n-1)]` iff `0 <= j < n` and `0 <= k-j < n`.

So `dc_k/da_j = 2 * a_{k-j}` if `0 <= j < n` and `0 <= k-j < n`, else 0.

Great, so the gradient is simply `g_j = 2 * a_{k*-j}` (with out-of-range = 0) for a single active k, or a weighted sum for multiple active k's.

"""
Multi-phase optimizer for min 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral shaping init, mirror descent with Nesterov momentum,
LP-guided refinement, adaptive temperature, and coordinate descent.
"""
import numpy as np
import time
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n = len(a)
    m = len(b) if b is not None else n
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N) if b is not None else fa
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * c.max() / (s * s)

def _score_unit(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _fft_conv(a, a)
    return 2.0 * n * c.max()

def _softmax_stable(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _subgradient(a, c, top_k=5, temperature=0.0):
    """
    Compute a (possibly smoothed) subgradient of max(c).
    Returns gradient in a-space (not z-space).
    """
    n = len(a)
    # Find top-k indices of c
    k_idx = np.argsort(c)[::-1][:top_k]
    
    if temperature > 0:
        # Softmax weights
        c_top = c[k_idx]
        w = np.exp((c_top - c_top[0]) / temperature)
        w = w / w.sum()
    else:
        # Uniform over top-k
        w = np.ones(top_k) / top_k
    
    # g_j = sum_k w_k * 2 * a_{k-j}
    g = np.zeros(n)
    for kk, ww in zip(k_idx, w):
        # a_{kk-j} is valid when 0 <= kk-j < n, i.e., kk-n+1 <= j <= kk
        j_start = max(0, kk - n + 1)
        j_end = min(kk, n - 1)
        if j_start <= j_end:
            g[j_start:j_end+1] += ww * 2.0 * a[kk - j_end:kk - j_start:-1]
    return g

def _mirror_descent_nesterov(a_init, deadline, T_start, T_end, 
                               lr_start, lr_end, top_k=5,
                               max_iter=200000, eval_interval=100):
    """
    Mirror descent in log-space with Nesterov momentum and backtracking.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15)) - np.log(np.maximum(a_init, 1e-15)).mean()
    y = z.copy()  # Nesterov lookahead
    z_old = z.copy()
    t = 0
    momentum = 0.0
    best_z = z.copy()
    best_score = _score_unit(a_init)
    consecutive_bad = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = t / max_iter
        
        # Adaptive schedule
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.5) * lr_mult
        
        # Nesterov update
        momentum_new = (t + 2.0) / (t + 1.0) * momentum  # Nesterov momentum schedule
        # Actually, standard Nesterov: beta_t = (1 - sqrt(1 + 4/t^2)) / 2
        # Or simpler: beta_t = (t-1)/(t+2)
        beta = (t - 1) / (t + 2) if t > 1 else 0.0
        
        y = z + beta * (z - z_old)
        z_old = z.copy()
        
        a = _softmax_stable(y)
        c = _fft_conv(a, a)
        
        # Compute gradient
        g = _subgradient(a, c, top_k, T)
        
        # Transform to z-space gradient
        W = np.dot(a, g)
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        z_new = y - lr * z_grad
        z_new = z_new - z_new.mean()
        z = z_new
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_unit(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_z = z.copy()
                consecutive_bad = 0
            else:
                consecutive_bad += 1
            
            # Adaptive restart
            if consecutive_bad > 10:
                consecutive_bad = 0
                lr_mult *= 0.5
                if lr_mult < 0.01:
                    lr_mult = 0.01
                z = best_z + 0.01 * np.random.randn(n)
                z -= z.mean()
                z_old = z.copy()
    
    a_final = _softmax_stable(best_z)
    return a_final / a_final.sum(), _score_unit(a_final / a_final.sum())

def _spectral_shaping(a_init, max_iter=100):
    """
    Flatten the spectrum of a to reduce peak autocorrelation.
    """
    n = len(a_init)
    a = a_init.copy().astype(float)
    a = a / a.sum()
    
    for _ in range(max_iter):
        A = np.fft.rfft(a, 2 * n - 1)
        # Flatten magnitude
        mag = np.abs(A)
        # Target: flat magnitude (constant)
        # But we need to preserve the DC component (sum = 1)
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # Preserve DC
        
        # Interpolate between current and target
        new_mag = 0.5 * mag + 0.5 * target_mag
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        # Check if improved
        sc_new = _score_unit(a_new)
        sc_old = _score_unit(a)
        if sc_new < sc_old:
            a = a_new
    
    return a

def _lp_refine(a, M, deadline, n_lp=4096):
    """
    LP-based refinement: solve max sum(b) s.t. (a*b)_k <= M, b >= 0.
    Returns b/sum(b) as the new iterate.
    """
    n = len(a)
    num_rows = 2 * n - 1
    
    # Build sparse constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    
    # Maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    b_ub = np.ones(num_rows) * M
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': max(5, deadline - time.monotonic() - 10)})
    
    if result.success:
        b_new = result.x
        s = b_new.sum()
        if s > 1e-10:
            return b_new / s
    return a

def _cd_polish(a, deadline, max_rounds=100):
    """
    Fast coordinate descent: try scaling each coordinate.
    Focus on coordinates that affect the peak convolution entries.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_unit(a)
    c = _fft_conv(a, a)
    c_max = c.max()
    
    # Identify active convolution entries
    tol = c_max * 0.95
    active_k = np.where(c > tol)[0]
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        
        # For each active k, try adjusting the coordinates that contribute most
        for k in active_k[:10]:  # Limit to top 10 active k's
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = range(j_start, j_end + 1)
            
            # Find the j with largest contribution
            contributions = np.array([a[j] * a[k - j] for j in js])
            top_js = [js[i] for i in np.argsort(contributions)[::-1][:5]]
            
            for j in top_js:
                for scale in [0.8, 0.9, 1.1, 1.2, 1.5]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    sc = _score_unit(a_cand)
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        a = a_cand
                        improved = True
                        c = _fft_conv(a, a)
                        c_max = c.max()
                        tol = c_max * 0.95
                        active_k = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return a, best_score
