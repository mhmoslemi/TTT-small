"""
Multi-phase optimizer with coordinate descent polish for minimizing the
autocorrelation peak-to-sum-ratio of non-negative sequences.
Uses L-BFGS with log-sum-exp continuation, Newton active-set, and 
fast incremental coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):

def _smograd(z, beta, n):

    c = _full_convolve(a, a)

def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:

    tol = tol_frac * c_max
    active = c >= (c_max - tol)

        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300):

"""
Multi-phase optimizer for minimizing the autocorrelation peak-to-sum-ratio.
Key innovation: fast incremental coordinate descent as a polish step,
combined with L-BFGS continuation and Newton active-set methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(active, n, b, max_iter=200, tol=1e-10, mu=1e-8):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hessian_matvec(p, active, n) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.001 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break

        dx = _cg_solve(active, n, -grad_t, max_iter=200, tol=1e-10, mu=1e-8)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-10:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5

        return a / a.sum()
    elif kind == "cos2":

        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":

        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)

    else:

def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(10, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001,
                                  0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 2048.0)
        est_time_per_iter = 0.004 * scale
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n), jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _coord_descent(a_init, deadline, max_sweeps=500):
    """Fast coordinate descent with incremental convolution update."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = float(a.sum())
    if S < 1e-15:
        return a_init, _score(a_init)
    a = a / S
    S = 1.0

    c = _full_convolve(a, a)
    best_a = a.copy()
    best_sc = float(2.0 * n * c.max())
    current_sc = best_sc

    stagnation = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved_this_sweep = False

        for j in range(n):
            aj = a[j]
            if aj < 1e-12:
                continue

            k_start = j
            k_end = j + n  # exclusive, so affected indices are [j, j+n-1]

            # Get current max
            c_max = float(c.max())
            k_max = int(np.argmax(c))

            # Determine optimal t
            if k_start <= k_max < k_end:
                # Max is in affected range
                if k_max == 2 * j:
                    # Quadratic case
                    if S - aj > 1e-30:
                        t_opt = (c[k_max] - aj * S) / (S - aj)
                    else:
                        t_opt = 0.0
                else:
                    dk = 2.0 * a[k_max - j]
                    if dk > 1e-30:
                        t_opt = S - c[k_max] / dk
                    else:
                        t_opt = 0.0
            else:
                # Max not in affected range - try to increase t
                t_opt = aj * 0.5  # Try a moderate increase
                # Check if all affected values stay below c_max
                for k in range(k_start, min(k_end, 2 * n - 1)):
                    if k == 2 * j:
                        val = c[k] + 2.0 * aj * t_opt + t_opt * t_opt
                    else:
                        val = c[k] + 2.0 * a[k - j] * t_opt
                    if val > c_max:
                        # Need to reduce t_opt
                        if k == 2 * j:
                            # t^2 + 2*aj*t + c[k] - c_max <= 0
                            # t = -aj + sqrt(aj^2 - c[k] + c_max)
                            disc = aj * aj - c[k] + c_max
                            if disc > 0:
                                t_bound = -aj + np.sqrt(disc)
                                t_opt = min(t_opt, t_bound * 0.99)
                            else:
                                t_opt = 0.0
                                break
                        else:
                            d = 2.0 * a[k - j]
                            if d > 1e-30:
                                t_bound = (c_max - c[k]) / d
                                t_opt = min(t_opt, t_bound * 0.99)
                            else:
                                t_opt = 0.0
                                break

            # Clamp t
            t_opt = max(t_opt, -aj + 1e-12)  # Ensure a[j]+t > 0
            t_opt = min(t_opt, aj)  # Don't more than double

            if abs(t_opt) < 1e-12:
                continue

            # Apply update
            new_aj = aj + t_opt
            new_aj = max(new_aj, 0.0)
            actual_t = new_aj - aj

            if abs(actual_t) < 1e-14:
                continue

            # Update conv incrementally
            for k in range(k_start, min(k_end, 2 * n - 1)):
                if k == 2 * j:
                    c[k] += 2.0 * aj * actual_t + actual_t * actual_t
                else:
                    c[k] += 2.0 * a[k - j] * actual_t

            a[j] = new_aj
            S += actual_t

            # Check score
            new_c_max = float(c.max())
            new_sc = 2.0 * n * new_c_max / (S * S)
            if new_sc < current_sc - 1e-14:
                current_sc = new_sc
                improved_this_sweep = True
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_a = a.copy()

        if improved_this_sweep:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 10:
                break

    return best_a, best_sc

"""
Multi-phase optimizer for minimizing the autocorrelation peak-to-sum-ratio of
non-negative sequences. Key innovation: fast vectorized coordinate descent with
incremental convolution updates, combined with L-BFGS log-sum-exp continuation
and Newton active-set methods. The coordinate descent provides exact 1D
optimization of each coordinate, enabling finer polish than gradient-based methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, n + n + 2)
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(active, n, b, max_iter=200, tol=1e-10, mu=1e-8):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = _hessian_matvec(p, active, n) + mu * p
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if alpha <= 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        dx = _cg_solve(active, n, -grad_t, max_iter=200, tol=1e-12, mu=1e-8)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        accepted = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            if alpha < 1e-10:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    idx = min(10, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    base_beta = max(gap, max_c * 0.008)

    ratios = np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                       0.0075, 0.004, 0.002, 0.001, 0.0005,
                       0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 2048.0)
        est_time_per_iter = 0.005 * scale
        max_iter = min(4000, max(20, int(remaining / (est_time_per_iter * 12))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n), jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 50, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if np.isfinite(sc_new) and sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _coord_descent(a_init, deadline, max_sweeps=2000):
    """Fast vectorized coordinate descent with incremental conv update."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = float(a.sum())
    if S < 1e-15:
        return a_init, _score(a_init)
    a = a / S
    S = 1.0

    c = _full_convolve(a, a).astype(np.float64)
    best_a = a.copy()
    best_sc = 2.0 * n * float(c.max())
    current_sc = best_sc

    stagnation = 0
    sweep = 0

    while sweep < max_sweeps:
        if time.monotonic() > deadline:
            break

        improved_this_sweep = False

        for j in range(n):
            aj = a[j]
            if aj < 1e-14:
                continue

            k_start = j
            k_end = j + n

            c_max = float(c.max())
            k_max = int(np.argmax(c))

            # Determine optimal t
            if k_start <= k_max < k_end:
                if k_max == 2 * j:
                    denom = S - aj
                    if denom > 1e-30:
                        t_opt = (c[k_max] - aj * S) / denom
                    else:
                        t_opt = 0.0
                else:
                    dk = 2.0 * a[k_max - j]
                    if dk > 1e-30:
                        t_opt = S - c[k_max] / dk
                    else:
                        t_opt = 0.0
            else:
                # Max not in affected range - try increasing t
                t_opt = aj * 0.3
                # Check all affected values stay below c_max
                for k in range(k_start, k_end):
                    if k == 2 * j:
                        val = c[k] + 2.0 * aj * t_opt + t_opt * t_opt
                    else:
                        val = c[k] + 2.0 * a[k - j] * t_opt
                    if val > c_max * 1.0001:
                        if k == 2 * j:
                            disc = aj * aj - c[k] + c_max
                            if disc > 0:
                                t_bound = -aj + np.sqrt(disc)
                            else:
                                t_bound = 0.0
                        else:
                            d = 2.0 * a[k - j]
                            if d > 1e-30:
                                t_bound = (c_max - c[k]) / d
                            else:
                                t_bound = 0.0
                        t_opt = min(t_opt, t_bound * 0.99)
                        if t_opt <= 1e-14:
                            break

            # Clamp
            t_opt = max(t_opt, -aj * 0.999)
            t_opt = min(t_opt, aj * 2.0)

            if abs(t_opt) < 1e-14:
                continue

            new_aj = aj + t_opt
            if new_aj < 0:
                new_aj = 0.0
                t_opt = -aj
            actual_t = new_aj - aj

            if abs(actual_t) < 1e-15:
                continue

            # Vectorized conv update (using OLD a values)
            c[k_start:k_end] += 2.0 * actual_t * a
            # Fix the k=2j quadratic term
            c[2 * j] += actual_t * actual_t

            # Update a[j]
            a[j] = new_aj
            S += actual_t

            # Check score (amortized: check every 32 coords for speed, always for improvement)
            new_c_max = float(c.max())
            new_sc = 2.0 * n * new_c_max / (S * S)
            if new_sc < current_sc - 1e-15:
                current_sc = new_sc
                improved_this_sweep = True
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_a = a.copy()
            else:
                # Revert
                a[j] = aj
                S -= actual_t
                c[k_start:k_end] -= 2.0 * actual_t * a[j]  # a[j] is now aj again
                c[2 * j] -= actual_t * actual_t
                # Safety: ensure no negative values from floating point
                c = np.maximum(c, 0.0)

        sweep += 1

        # Periodic full reconv to avoid drift
        if sweep % 200 == 0:
            c = _full_convolve(a, a).astype(np.float64)
            S = float(a.sum())
            current_sc = 2.0 * n * float(c.max()) / (S * S)

        if improved_this_sweep:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 15:
                break

    # Final normalization
    s_final = float(a.sum())
    if s_final > 1e-15:
        a = a / s_final
    final_sc = _score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()

    return best_a, best_sc
