"""
Step sequence optimizer using chirp-based initialization, 
L-BFGS with log-sum-exp smoothing, direct-space peak shaving,

exploits the CAZAC property to start near a flat autocorrelation.

import numpy as np

# All helper functions at top level (no closures, no lambdas)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _chirp_init(n, c=0.01):
    i = np.arange(n)
    a = 1.0 + c * np.cos(2.0 * np.pi * i * i / (2.0 * n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    i = np.arange(n)

    return np.ones(n) / n

def _random_smooth_init(n, rng, n_modes=10):

    a = np.ones(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.01 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        a = a + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _adaptive_active_set(c, tol_frac_base=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(3, int(0.01 * len(c)))
    if n_active_est >= len(c):
        n_active_est = len(c)
    threshold_from_rank = sorted_c[min(n_active_est - 1, len(c) - 1)]
    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac_base * c_max
    tol = max(tol_fixed, min(tol_rank, 0.5 * c_max))
    active = c >= (c_max - tol)

        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w

    z0 = z0 - z0.mean()

        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:

                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac_base * c_max
    tol = max(tol_fixed, min(tol_rank, 0.5 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _gradient_direct(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    try:
        result = minimize(_smoobj, z0, args=(beta, n), jac=_smograd,
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=300.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.3, 0.15, 0.08, 0.04, 0.02, 0.01,
                              0.005, 0.0025, 0.001, 0.0005,
                              0.0002, 0.0001, 5e-5, 2.5e-5,
                              1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7,
                              1e-7, 5e-8, 2e-8, 1e-8, 5e-9])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_per_iter = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(15000, max(300, int(remaining / (est_per_iter * 15))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.3e}: score={sc:.10f} (iter={niter})")
        else:
            print(f"    beta={beta:.3e}: no improvement")

    return current_a


def _peak_shaving(a_init, deadline, max_iters=50000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0
    stall_threshold = 200

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        w = _adaptive_active_set(c, 1e-4)

        g = _gradient_direct(a, c, n, w)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        g_normalized = g / g_norm

        best_a_try = a.copy()
        best_sc_try = best_sc

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g_normalized
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.02, 0.05)
            stall = 0
        else:
            stall += 1
            eta *= 0.9
            if eta < 1e-9:
                break
            if stall > stall_threshold:
                break

    return best_a, best_sc


def _targeted_peak_reduction(a_init, deadline, max_iters=20000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    step_frac = 0.005

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        K = min(5, len(c))
        top_indices = np.argsort(c)[::-1][:K]

        improved = False
        for k_idx in top_indices:
            k = int(k_idx)
            # conv(a,a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
            # d/da[j] = a[k-j]
            valid_js = []
            partner_vals = []
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    valid_js.append(j)
                    partner_vals.append(a[kj])

            if not valid_js:
                continue

            valid_js_arr = np.array(valid_js)
            partner_vals_arr = np.array(partner_vals)

            top_contrib = np.argsort(partner_vals_arr)[::-1][:3]
            bottom_contrib = np.argsort(partner_vals_arr)[:3]

            for tc in top_contrib:
                j_reduce = int(valid_js_arr[tc])
                for bc in bottom_contrib:
                    j_add = int(valid_js_arr[bc])
                    if j_add == j_reduce:
                        continue

                    delta = step_frac * a[j_reduce]
                    if delta < 1e-12:
                        continue

                    a_try = a.copy()
                    a_try[j_reduce] -= delta
                    a_try[j_add] += delta
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)

                    if sc_try < best_sc - 1e-14:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

                if improved:
                    break

            if improved:
                break

        if not improved:
            step_frac *= 0.8
            if step_frac < 1e-6:
                break

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = int((rnd * 6151) % n)
        jdx = int((rnd * 6151 + 13) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        improved = False

        for frac in np.linspace(0.1, 0.9, 17):
            new_i = s_pair * frac
            new_j = s_pair * (1.0 - frac)
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            idx2 = int((rnd * 397 + 5) % n)
            for scale in [0.85, 0.92, 1.08, 1.15]:
                a_try = a.copy()
                a_try[idx2] *= scale
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_for_restart(a, rng, mode="gauss", scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        n_modes = 8
        for _ in range(n_modes):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.005 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "chirp_mix":
        t = np.arange(n)
        f1 = rng.uniform(0.5, 2.0)
        amp = 0.01 * rng.uniform(0.5, 2.0)
        z = z + amp * np.cos(2.0 * np.pi * f1 * t * t / (2.0 * n) + rng.uniform(0
        , 2 * np.pi))
    z = z - z.mean()
    return _softmax(z)


def _try_update(state, p, label, t_start):
    """Updates state[0] (best_seq) and state[1] (best_score) if p is better."""
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    sc = _score(p)
    if np.isfinite(sc) and sc < state[1]:
        state[0] = p.copy()
        state[1] = float(sc)
        elapsed = time.monotonic() - t_start
        print(f"  NEW BEST: {state[1]:.10f} (n={len(p)}) [{label}] "
              f"t={elapsed:.1f}s")

    """

    t_start = time.monotonic()

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

    except Exception:
        parent = None

    n_base = 8192

    sc = _score(a_chirp2)
    print(f"  Chirp2 c=0.01: score={sc:.10f}")
    _try_update(state, a_chirp2, "chirp2", t_start)
    init_list.append((a_chirp2, n_base))

    # Additional chirp variants
    for c in [0.01, 0.03]:
        a_c3 = _chirp_init2(n_base, c)
