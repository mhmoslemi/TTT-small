"""
Advanced optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based smoothed gradient descent with momentum, adaptive beta cooling,
local window perturbations, sparsity moves, and multi-n exploration.
Key innovation: direct linear-space optimization with Nesterov momentum and
careful backtracking, plus "peak peeling" for final refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve_self(a):
    return _full_convolve(a, a)


def _smoothed_grad_fft(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    w = np.exp(np.clip(shifted, -700, 0.0))
    w_sum = w.sum()
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    w_flip = w[::-1]
    conv_aw = _full_convolve(a, w_flip)
    grad = conv_aw[n - 1:2 * n - 1][::-1]
    return w, c_max, grad


def _lse_score(a, beta):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0.0))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        return 2.0 * n * c_max / (s * s)
    lse = c_max + beta * np.log(s_e)
    return 2.0 * n * lse / (s * s)


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    elif kind == "super_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 4) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_momentum_gd(a_init, deadline, beta_start_frac=0.005, beta_end_frac=0.00005,
                     lr_init=0.01, momentum_init=0.9, min_time_remaining=30.0):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    if c_max0 < 1e-15:
        return a, _score(a)

    best_a = a.copy()
    best_sc = _score(a)

    beta = c_max0 * beta_start_frac
    n_betas_red = 50
    beta_ratio = (beta_end_frac / beta_start_frac) ** (1.0 / n_betas_red)

    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    no_improve = 0
    max_stall = 200

    t_start = time.monotonic()
    beta_count = 0

    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        if beta < c_max * beta_end_frac:
            break

        _, _, grad = _smoothed_grad_fft(a, beta)
        grad_c = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_c)
        if gnorm < 1e-16:
            break

        # Nesterov: compute gradient at lookahead point
        a_look = a + momentum * v / (gnorm + 1e-30)
        a_look = np.maximum(a_look, 0.0)
        a_look = _normalize(a_look)
        _, _, grad_look = _smoothed_grad_fft(a_look, beta)
        grad_look_c = grad_look - np.mean(grad_look)

        v = momentum * v + grad_look_c
        a_new = a - lr * v / (np.linalg.norm(v) + 1e-30)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:
            a = a_new
            no_improve = 0
            if no_improve == 0 and lr < lr_init:
                lr = min(lr_init, lr * 1.05)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve % 20 == 0:
                beta *= beta_ratio
                beta_count += 1
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                beta *= beta_ratio
                beta_count += 1
                lr = lr_init * 0.5
                no_improve = 0
                if beta_count > n_betas_red * 3:
                    break

    sc_final = _score(best_a)
    return best_a, sc_final


def _run_direct_gd(a_init, deadline, beta_start_frac=0.003, beta_end_frac=0.00002,
                   lr_init=0.005, min_time_remaining=10.0):
    """Simpler gradient descent without momentum, with more careful step control."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)

    c0 = _full_convolve(a, a)
    c_max0 = float(c0.max())
    beta = c_max0 * beta_start_frac
    beta_ratio = (beta_end_frac / beta_start_frac) ** (1.0 / 60)
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        if beta < c_max * beta_end_frac:
            break

        _, _, grad = _smoothed_grad_fft(a, beta)
        grad_c = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_c)
        if gnorm < 1e-16:
            break

        alpha = lr / (gnorm + 1e-30)
        a_old_sc = current_sc
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_c
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                beta *= beta_ratio
                lr *= 0.7
            if no_improve >= 300:
                a = best_a.copy()
                beta *= beta_ratio
                lr = lr_init * 0.3
                no_improve = 0
                if beta < c_max0 * beta_end_frac * 0.5:
                    break
        else:
            if no_improve == 0 and lr < lr_init:
                lr = min(lr_init, lr * 1.03)
            if no_improve > 0 and no_improve < 10:
                lr = min(lr_init, lr * 1.01)

    sc_final = _score(best_a)
    return best_a, sc_final


def _peak_peel_refine(a_init, deadline, min_time_remaining=5.0):
    """Identify active peaks and do targeted optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)

    tol = 1e-6
    lr = 0.002
    no_improve = 0

    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Find active set (peaks within tol of max)
        active_thresh = c_max - c_max * tol
        active = c >= active_thresh
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Uniform weight on active peaks
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Compute gradient
        w_flip = w[::-1]
        conv_aw = _full_convolve(a, w_flip)
        grad = conv_aw[n - 1:2 * n - 1][::-1]
        grad_c = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_c)

        if gnorm < 1e-16:
            break

        alpha = lr / (gnorm + 1e-30)
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_c
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            tol *= 0.8
            lr *= 0.6
            if no_improve > 100:
                break
        else:
            lr = min(0.005, lr * 1.02)
            if no_improve == 0:
                tol = max(1e-8, tol * 0.95)

    return best_a, _score(best_a)


def _local_window_optimize(a_init, deadline, window_sizes=None, min_time_remaining=30.0):
    """Optimize random windows of elements while holding others fixed."""
    n = len(a_init)
    if window_sizes is None:
        window_sizes = [32, 64, 128]

    rng = np.random.default_rng(12345)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    trials = 0
    max_trials = 5000
    while time.monotonic() < deadline - min_time_remaining and trials < max_trials:
        trials += 1
        m = window_sizes[rng.integers(0, len(window_sizes))]
        m = min(m, n)
        i0 = rng.integers(0, n - m + 1)

        # Save the window
        window = a[i0:i0 + m].copy()

        # Optimize this window using local gradient steps
        lr_local = 0.01
        for step in range(50):
            c = _full_convolve(a, a)
            c_max = float(c.max())
            current_sc = 2.0 * n * c_max / (a.sum()) ** 2

            # Gradient for the window elements only
            _, _, full_grad = _smoothed_grad_fft(a, c_max * 0.001)
            w_grad = full_grad[i0:i0 + m].copy()
            # Center within the window
            w_grad_c = w_grad - np.mean(w_grad)
            w_gnorm = np.linalg.norm(w_grad_c)
            if w_gnorm < 1e-14:
                break

            alpha = lr_local / (w_gnorm + 1e-30)
            improved = False
            for _ in range(20):
                new_window = np.maximum(window - alpha * w_grad_c, 0.0)
                a_try = a.copy()
                a_try[i0:i0 + m] = new_window
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    window = new_window
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                break

        sc_after = _score(a)
        if sc_after < best_sc - 1e-12:
            best_sc = sc_after
            best_a = a.copy()

    return best_a, best_sc


def _sparsity_search(a_init, deadline, min_time_remaining=20.0):
    """Try zeroing out small elements and re-optimizing."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline - min_time_remaining:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _lbfgs_polish(a_init, deadline, min_time_remaining=5.0):
    """Quick L-BFGS polish using log-sum-exp parameterization."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    if c_max < 1e-15:
        return a_init, _score(a_init)

    beta = c_max * 0.001
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    def obj(z_vec):
        a_z = _softmax_safe(z_vec)
        c_z = _full_convolve(a_z, a_z)
        cm = float(c_z.max())
        shifted = (c_z - cm) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            return 2.0 * n * cm
        lse = cm + beta * np.log(s_e)
        return 2.0 * n * lse

    def grad_fn(z_vec):
        a_z = _softmax_safe(z_vec)
        c_z = _full_convolve(a_z, a_z)
        cm = float(c_z.max())
        shifted = (c_z - cm) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c_z)
            w[int(np.argmax(c_z))] = 1.0
        else:
            w = e / s_e
        w_flip = w[::-1]
        conv_w = _full_convolve(w, a_z[::-1])
        h = conv_w[n - 1:2 * n - 1]
        W = float(np.dot(w, c_z))
        return 4.0 * n * a_z * (h - W)

    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 3:
        return a_init, _score(a_init)
    max_iter = min(3000, max(20, int(remaining / 0.03)))

    try:
        result = minimize(obj, z, jac=grad_fn, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxcor': 50})
        a_new = _softmax_safe(result.x)
        return a_new, _score(a_new)
    except Exception:
        return a_init, _score(a_init)


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fourier_perturb(a, rng, n_modes=5, amp_range=(0.01, 0.08)):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax_safe(z)


def _gaussian_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax_safe(z)

    a2, sc2 = _run_direct_gd(best_seq, p2_deadline,

                             lr_init=0.004, min_time_remaining=10.0)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [direct_gd] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak peel refine ===
    print(f"\n[P3] Peak peel (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 400, time.monotonic() + 180)
    a3, sc3 = _peak_peel_refine(best_seq, p3_deadline, min_time_remaining=10.0)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [peak_peel] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Local window optimization ===
    print(f"\n[P4] Local window opt (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = min(deadline - 250, time.monotonic() + 200)
    a4, sc4 = _local_window_optimize(best_seq, p4_deadline,
                                     window_sizes=[16, 32, 64, 128],
                                     min_time_remaining=10.0)
    if sc4 < best_score:
        best_seq = a4.copy()
        best_score = sc4
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [local_window] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Sparsity search ===
    print(f"\n[P5] Sparsity (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = min(deadline - 200, time.monotonic() + 60)
    a5, sc5 = _sparsity_search(best_seq, p5_deadline, min_time_remaining=5.0)
    if sc5 < best_score:
        best_seq = a5.copy()
        best_score = sc5
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [sparsity] t={time.monotonic()-t_start:.1f}s")
    else:
        # Try L-BFGS polish after sparsity attempt
        if time.monotonic() < deadline - 180:
            a5b, sc5b = _lbfgs_polish(best_seq, deadline - 170, min_time_remaining=10.0)
            if sc5b < best_score:
                best_seq = a5b.copy()
                best_score = sc5b
                print(f"  NEW BEST: {best_score:.10f} [lbfgs_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    restarts = [
        ("gauss_pert2", _gaussian_perturb(best_seq, 0.02, rng)),
        ("gauss_pert5", _gaussian_perturb(best_seq, 0.05, rng)),
        ("fourier4", _fourier_perturb(best_seq, rng, n_modes=4, amp_range=(0.01, 0.05))),
        ("fourier7", _fourier_perturb(best_seq, rng, n_modes=7, amp_range=(0.02, 0.08))),
        ("tapered", _make_profile(n_best, "tapered")),
        ("cos_mod", _make_profile(n_best, "cos_mod")),
        ("gauss_prof", _make_profile(n_best, "gaussian")),
        ("plateau", _make_profile(n_best, "plateau")),
    ]
    for label, a_r in restarts:
        if time.monotonic() > deadline - 80:
            break
        a_opt, sc_opt = _run_momentum_gd(a_r, min(deadline - 70, time.monotonic() + 40),
                                         beta_start_frac=0.003, beta_end_frac=0.0001,
                                         lr_init=0.006, momentum_init=0.9,
                                         min_time_remaining=5.0)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([2048, 3072, 4096, 6144, 8192, 10240, 12288, 16384]))
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        a_opt, sc_opt = _run_direct_gd(a_init, min(deadline - 50, time.monotonic() + 40),
                                       beta_start_frac=0.002, beta_end_frac=0.00005,
                                       lr_init=0.003, min_time_remaining=5.0)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    final_deadline = deadline - 3
    if time.monotonic() < final_deadline - 30:
        a_fin, sc_fin = _peak_peel_refine(best_seq, final_deadline - 30, min_time_remaining=5.0)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_peel] t={time.monotonic()-t_start:.1f}s")

    if time.monotonic() < final_deadline - 10:
        a_fin2, sc_fin2 = _lbfgs_polish(best_seq, final_deadline - 5, min_time_remaining=3.0)
        if sc_fin2 < best_score:
            best_seq = a_fin2.copy()
            best_score = sc_fin2
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # One more momentum GD pass with very tight parameters
    if time.monotonic() < final_deadline - 5:
        a_fin3, sc_fin3 = _run_momentum_gd(best_seq, final_deadline - 2,
                                           beta_start_frac=0.0005, beta_end_frac=0.00001,
                                           lr_init=0.002, momentum_init=0.85,
                                           min_time_remaining=2.0)
        if sc_fin3 < best_score:
            best_seq = a_fin3.copy()
            best_score = sc_fin3
            print(f"  NEW BEST: {best_score:.10f} [final_momentum] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Scale to ensure sum is reasonable (avoid overflow/underflow in evaluator)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, sum={s:.4f}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
