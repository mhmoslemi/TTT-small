```python
    N = _next_pow2(3 * n)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    # We want c[i] = sum_k w[k] * a_pad[k-i] for i = 0, ..., 2n-2
    # This is cross-correlation of w and a_pad
    # = IFT(FFT(w) * conj(FFT(a_pad)))
    # But actually it's sum_k w[k] * a[k-i] which is the (n-1-i)th element of conv(w, a)
    # Let's just compute conv(w, a) and take the right slice
    fa = np.fft.rfft(a_pad, N)
    fw = np.fft.rfft(w_pad, N)
    conv_wa = np.fft.irfft(fw * fa, N)  # This is w * a (linear conv)
    # conv_wa[k] = sum_j w[j] * a[k-j] for k = 0, ..., 3n-3
    # We want pressure[i] = sum_k w[k] * a[k-i]
    # = sum_k w[k] * a[k-i] = conv_wa[i] ... no
    # conv_wa[m] = sum_j w[j] * a[m-j]
    # We want: sum_k w[k] * a[k-i] = sum_k w[k] * a[k-i]
    # Let m = k-i, so k = m+i: sum_m w[m+i] * a[m] = ... 
    # This is the cross-correlation! = (w * a_rev)[n-1+i] approximately
    # Let me just use: pressure = conv(w, a_rev) shifted
    a_rev = a[::-1]
    # conv(w, a_rev)[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-k+j]
    # We want sum_k w[k] * a[k-i]. Let's substitute: let j = k, then sum_j w[j] * a[j-i]
    # = sum_j w[j] * a[n-1-(n-1-j+i)] = sum_j w[j] * a_rev[n-1-j+i]
    # Let m = n-1-j+i, so j = n-1-m+i:
    # = sum_m w[n-1-m+i] * a_rev[m] = conv(w_shifted, a_rev)[m] where w_shifted[m] = w[n-1-m+i]
    # This is getting too complicated. Let me just do it directly with numpy.
    
    # Direct approach (fast enough for our purposes):
    pressure = np.zeros(n)
    for k in peak_idx:
        # a[k-i] for i = 0, ..., n-1, where a[j] = 0 for j < 0 or j >= n
        # i ranges such that 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # a[k-i] for i in [i_lo, i_hi]
            # k-i ranges from k-i_hi to k-i_lo
            src = a[k-i_hi:k-i_lo+1][::-1]
            pressure[i_lo:i_hi+1] += src
    return pressure
```

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast IAA (Iterative Amplitude Adjustment) with adaptive active peaks and backtracking
2. BFGS continuation on log-sum-exp smoothed objective for fine-tuning
3. Multi-start with diverse structured initializations
4. Cross-length interpolation and re-optimization
5. Newton-style equiripple equalization on the active peak set

Key differences from previous SOTA:
- IAA operates directly on a (not log-space), exploring different basins
- Adaptive peak count that increases over iterations
- Backtracking line search ensures monotone improvement
- Final very-tight BFGS continuation (beta as small as 1e-8 * max_c)
- Equiripple Newton solver for the top-k active peaks
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_pressure(a, peak_idx):
    """Compute the pressure vector: pressure[i] = sum_{k in peak_idx} a[k-i] for valid indices."""
    n = len(a)
    pressure = np.zeros(n)
    for k in peak_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            src = a[k - i_hi:k - i_lo + 1][::-1]
            pressure[i_lo:i_hi + 1] += src
    return pressure


def _iaa_optimize(a_init, deadline, max_steps=200000, start_k=5, end_k=20):
    """
    Fast IAA with adaptive peak count and backtracking.
    Works directly on a (non-negative, sum-normalized).
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())
    best_sc = 2.0 * n * best_cmax

    c = _full_convolve(a, a)
    cmax = float(c.max())
    lr = 2.0 / n

    stall = 0
    stall_limit = 2000
    k_current = start_k

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Gradually increase the number of active peaks
        frac = min(1.0, step / (max_steps * 0.7))
        k_cur = int(start_k + (end_k - start_k) * frac)
        if k_cur != k_current:
            k_current = k_cur

        c = _full_convolve(a, a)
        cmax = float(c.max())
        sc = 2.0 * n * cmax

        if sc < best_sc - 1e-16:
            best_sc = sc
            best_a = a.copy()
            best_cmax = cmax
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            break

        # Get active peaks
        idx = np.argsort(c)[::-1][:k_cur]

        # Compute pressure
        pressure = _compute_pressure(a, idx)

        # Backtracking line search
        alpha = lr
        improved = False
        for _bt in range(15):
            a_try = a - alpha * pressure
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            cmax_try = float(c_try.max())
            if cmax_try < cmax - 1e-18:
                a = a_try
                cmax = cmax_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller alpha
            alpha = lr * 0.1
            for _bt in range(5):
                a_try = a - alpha * pressure
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    break
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < cmax - 1e-18:
                    a = a_try
                    break

            # If still no improvement, try a different peak set
            if not improved:
                # Perturb: randomly select a different subset of peaks
                all_peaks = np.argsort(c)[::-1][:max(3 * k_cur, 30)]
                subset = np.random.choice(all_peaks, size=k_cur, replace=False)
                pressure = _compute_pressure(a, subset)
                a_try = a - lr * 0.5 * pressure
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    a = a_try

        # Adapt learning rate
        if improved:
            lr = min(lr * 1.01, 10.0 / n)
        else:
            lr *= 0.8
            if lr < 1e-8:
                lr = 2.0 / n  # reset

    c_final = _full_convolve(a, a)
    sc_final = 2.0 * n * float(c_final.max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _iaa_multiplicative(a_init, deadline, max_steps=100000, k_peaks=8):
    """
    Multiplicative IAA variant that naturally preserves non-negativity.
    a_new[i] = a[i] * (1 - lr * pressure[i] / a[i]) clipped to [0, inf]
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = 1.0
    stall = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cmax = float(c.max())
        sc = 2.0 * n * cmax

        if sc < best_sc - 1e-16:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break

        idx = np.argsort(c)[::-1][:k_peaks]
        pressure = _compute_pressure(a, idx)
        pmax = pressure.max()
        if pmax < 1e-20:
            break

        # Multiplicative update
        ratio = np.ones(n)
        mask = a > 1e-20
        ratio[mask] = 1.0 - lr * pressure[mask] / (a.sum() * pmax)
        ratio = np.maximum(ratio, 0.01)
        a_new = a * ratio
        a_new = a_new / a_new.sum()

        c_new = _full_convolve(a_new, a_new)
        if float(c_new.max()) < cmax - 1e-18:
            a = a_new
            lr = min(lr * 1.005, 5.0)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    c_final = _full_convolve(a, a)
    sc_final = 2.0 * n * float(c_final.max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _bfgs_smoothed(a_init, beta, n, max_iter):
    """Run L-BFGS-B on the log-sum-exp smoothed objective in log-space."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def jac(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxcor': 80, 'maxls': 100})
        a_out = _softmax_fn(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _continuation_bfgs(a_start, deadline, reserve_time=50.0):
    """Adaptive continuation BFGS with aggressive beta reduction."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = 2.0 * n * max_c

    # Start with moderate smoothing and decrease
    betas = max_c * np.array([
        0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001, 5e-5,
        2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8
    ])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter based on remaining time
        time_per_iter = 0.003 * (n / 4096.0) ** 1.0
        max_iter = min(20000, max(100, int(remaining / (time_per_iter * 30))))

        a_new = _bfgs_smoothed(current_a, beta, n, max_iter)
        sc = 2.0 * n * float(_full_convolve(a_new, a_new).max())
        s = a_new.sum()
        if s < 1e-15:
            continue
        sc_norm = 2.0 * n * float(_full_convolve(a_new, a_new).max()) / (s * s)

        if sc_norm < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_norm

        if current_sc < 1.5025:
            break

    return current_a


def _equiripple_newton(a_init, deadline, k=8, max_iter=500):
    """
    Newton-style equalization of the top-k convolution peaks.
    Parameterize a = softmax(z) and solve for z such that
    the top-k convolution values are as equal as possible.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = 2.0 * n * float(_full_convolve(best_a, best_a).max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        idx = np.argsort(c)[::-1][:k]
        active_vals = c[idx]
        mean_val = active_vals.mean()
        spread = active_vals.max() - active_vals.min()

        if spread < 1e-18:
            break

        # Gradient of spread w.r.t. z
        # d(c[k])/d(z[i]) = a[i] * (d(c[k])/d(a[i]))
        # d(c[k])/d(a[i]) = sum of contributions (use pressure-like computation)
        # For equalization, we want to reduce the spread
        # Use the gradient of (max - min) as a proxy

        # Compute which indices contribute to max and min
        k_max = idx[0]
        k_min = idx[-1]

        # Direction to reduce c[k_max] and increase c[k_min]
        pressure_max = _compute_pressure(a, [k_max])
        pressure_min = _compute_pressure(a, [k_min])

        # We want to decrease c[k_max] and increase c[k_min]
        # d(c[k])/d(a[i]) ~ pressure[i] (for the dominant term)
        # To decrease c[k_max]: decrease a where pressure_max is high
        # To increase c[k_min]: increase a where pressure_min is low (relatively)
        direction = pressure_max - pressure_min
        direction = direction - direction.mean()

        norm_dir = np.linalg.norm(direction)
        if norm_dir < 1e-20:
            break

        # Newton step with line search
        alpha = 1.0
        improved = False
        for _ in range(20):
            z_try = z - alpha * direction / (n * norm_dir)
            a_try = _softmax_fn(z_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                z = z_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try a smaller step in the gradient direction of the smoothed max
            c = _full_convolve(a, a)
            beta = float(c.max()) * 0.01
            idx_all = np.argsort(c)[::-1][:k]
            w = np.zeros_like(c)
            w[idx_all] = 1.0 / len(idx_all)
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            z = z - 0.1 * grad / max(1.0, np.linalg.norm(grad) / n)
            z = z - z.mean()

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands, eps=0.2):
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric_chirp(n, freq=1.0, eps=0.3, skew=0.5):
    """Asymmetric chirp: different frequency sweep at start vs end."""
    t = np.arange(n) / n
    # Modulate the chirp rate
    rate = freq * (1.0 + skew * (t - 0.5))
    phase = 2.0 * np.pi * freq * t * t
    a = 1.0 + eps * np.cos(phase)
    # Add asymmetry
    a = a * (1.0 + skew * 0.3 * np.sin(np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_windowed_binary(n, rng):
    """Random binary-like pattern with windowing."""
    a = rng.choice([0.3, 1.0], size=n)
    # Apply a smooth window
    t = np.linspace(0, 1, n)
    window = 0.5 * (1 + np.cos(np.pi * t))
    a = a * window
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _subgrad_polish_fast(a_init, deadline, k_active=5, max_steps=30000):
    """Fast subgradient polishing in z-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = 2.0 * n * float(_full_convolve(best_a, best_a).max())
    lr = 0.03 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        z = z - lr * grad / max(1e-20, np.abs(a).max())
        z = z - z.mean()
        z = np.clip(z, -80.0, 80.0)

        if i % 200 == 0:
            a_check = _softmax_fn(z)
            sc = 2.0 * n * float(_full_convolve(a_check, a_check).max())
            s = a_check.sum()
            sc = 2.0 * n * float(_full_convolve(a_check, a_check).max()) / (s * s)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    lr *= 0.7
                    if lr < 1e-10:
                        break

    a_final = _softmax_fn(z)
    s_f = a_final.sum()
    sc_final = 2.0 * n * float(_full_convolve(a_final, a_final).max()) / (s_f * s_f)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining IAA, BFGS continuation, and Newton equiripple.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    n_main = 16384
    candidates = []

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Structured inits at n_main
    structured = [
        ("chirp1", _make_chirp(n_main, freq=1.0, eps=0.3)),
        ("chirp2", _make_chirp(n_main, freq=2.0, eps=0.25)),
        ("chirp1_large", _make_chirp(n_main, freq=1.0, eps=0.5)),
        ("asym_chirp", _make_asymmetric_chirp(n_main, freq=1.0, eps=0.3, skew=0.5)),
        ("asym_chirp2", _make_asymmetric_chirp(n_main, freq=2.0, eps=0.2, skew=0.3)),
        ("multi_13", _make_multiband(n_main, bands=[1, 3], eps=0.25)),
        ("multi_124", _make_multiband(n_main, bands=[1, 2, 4], eps=0.2)),
        ("tapered", _make_tapered(n_main, power=0.5)),
    ]

    for name, a_init in structured:
        candidates.append((name, a_init.copy()))
        sc = _score(a_init)
        print(f"  {name}: score={sc:.10f}")
        try_update(a_init, name)

    for r in range(2):
        a_r = _make_random_positive(n_main, rng)
        candidates.append((f"rand_{r}", a_r))
        try_update(a_r, f"rand_{r}")

    if best_seq is None:
        best_seq = candidates[0][1].copy()
        best_score_val = _score(best_seq)
        best_n = len(best_seq)

    print(f"  Phase 1 done. Best: {best_score_val:.10f} (n={best_n})")

    # === Phase 2: IAA on best candidate ===
    print(f"\n=== Phase 2: IAA optimization (t={time.monotonic()-t_start:.1f}s) ===")
    iaa_deadline = min(deadline - 300, t_start + 400)
    a_iaa, sc_iaa = _iaa_optimize(best_seq, iaa_deadline, max_steps=150000, start_k=5, end_k=25)
    try_update(a_iaa, "iaa_main")

    # IAA multiplicative variant
    if time.monotonic() < deadline - 200:
        a_iaa2, sc_iaa2 = _iaa_multiplicative(best_seq, deadline - 200, max_steps=50000, k_peaks=10)
        try_update(a_iaa2, "iaa_mult")

    # === Phase 3: BFGS continuation ===
    print(f"\n=== Phase 3: BFGS continuation (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_bfgs, sc_bfgs = None, np.inf
        remaining = deadline - time.monotonic() - 80
        if remaining > 30:
            a_bfgs = _continuation_bfgs(best_seq, t_start + time.monotonic() - t_start + remaining, reserve_time=30)
            try_update(a_bfgs, "bfgs_cont")

    # === Phase 4: Equiripple Newton ===
    print(f"\n=== Phase 4: Equiripple Newton (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for k_nr in [6, 10, 15]:
            if time.monotonic() > deadline - 30:
                break
            a_ne, sc_ne = _equiripple_newton(best_seq, deadline - 30, k=k_nr, max_iter=200)
            try_update(a_ne, f"newton_k{k_nr}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for k_sg in [5, 10, 20]:
            if time.monotonic() > deadline - 20:
                break
            a_sg, sc_sg = _subgrad_polish_fast(best_seq, deadline - 20, k_active=k_sg, max_steps=20000)
            try_update(a_sg, f"sg_k{k_sg}")

    # === Phase 6: IAA + BFGS combined (alternate) ===
    print(f"\n=== Phase 6: Alternating IAA+BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cur = best_seq.copy()
        for round_num in range(4):
            if time.monotonic() > deadline - 20:
                break
            # IAA step
            a_tmp, _ = _iaa_optimize(a_cur, deadline - 60, max_steps=30000, start_k=8, end_k=20)
            if _score(a_tmp) < _score(a_cur):
                a_cur = a_tmp
            # BFGS step
            if time.monotonic() < deadline - 25:
                c_tmp = _full_convolve(a_cur, a_cur)
                mc = float(c_tmp.max())
                beta = mc * 1e-5
                n_cur = len(a_cur)
                time_per_it = 0.004 * (n_cur / 4096.0)
                max_it = max(50, int((deadline - 30 - time.monotonic()) / (time_per_it * 20)))
                a_b = _bfgs_smoothed(a_cur, beta, n_cur, max_it)
                if _score(a_b) < _score(a_cur):
                    a_cur = a_b
            try_update(a_cur, f"alt_round{round_num}")

    # === Phase 7: Cross-length exploration ===
    print(f"\n=== Phase 7: Cross-length (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Try slightly different lengths
        n_cur = len(best_seq)
        for target_n in [n_cur * 2, n_cur // 2, n_cur + 1024, n_cur - 1024]:
            if time.monotonic() > deadline - 15:
                break
            if target_n < 128 or target_n > 32768 or target_n == n_cur:
                continue
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_int = _score(a_interp)
            print(f"  n={target_n}: interp score={sc_int:.10f}")
            # Quick IAA
            if time.monotonic() < deadline - 10:
                a_opt, sc_opt = _iaa_optimize(a_interp, deadline - 10, max_steps=20000, start_k=5, end_k=15)
                try_update(a_opt, f"n_{target_n}_iaa")
            # Quick BFGS
            if time.monotonic() < deadline - 10:
                c_t = _full_convolve(a_interp, a_interp)
                mc_t = float(c_t.max())
                a_b = _bfgs_smoothed(a_interp, mc_t * 1e-4, target_n, 2000)
                try_update(a_b, f"n_{target_n}_bfgs")

    # === Phase 8: Final aggressive micro-polish ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 2:
        for rnd in range(8):
            if time.monotonic() > deadline - 2:
                break
            n_f = len(best_seq)
            c_f = _full_convolve(best_seq, best_seq)
            mf = float(c_f.max())

            # Very tight BFGS
            beta_t = mf * (5e-7 / (rnd + 1))
            time_rem = deadline - 2 - time.monotonic()
            max_it = max(50, int(time_rem / 0.005))
            a_f = _bfgs_smoothed(best_seq, beta_t, n_f, max_it)
            try_update(a_f, f"final_bfgs_{rnd}")

            # IAA micro-step
            if time.monotonic() < deadline - 2:
                a_m, _ = _iaa_optimize(best_seq, deadline - 2, max_steps=10000, start_k=10, end_k=30)
                try_update(a_m, f"final_iaa_{rnd}")

            # Equiripple
            if time.monotonic() < deadline - 2:
                a_e, _ = _equiripple_newton(best_seq, deadline - 2, k=12, max_iter=100)
                try_update(a_e, f"final_eq_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
