def _active_set_lp(a, M, top_k=50, deadline=None):
    """
    Solve: max sum(b) s.t. conv(a,b)[i] <= M for all i, b >= 0.
    Use only top_k most constraining lags for speed.
    """
    n = len(a)
    c = _fft_convolve(a, a)
    # Most constraining lags are those where c is closest to M (from below)
    # Actually, all lags are constraints. Let's use the top_k lags with highest c values.
    k_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A_eq, b_eq, A_ub, b_ub
    # Variables: b[0], ..., b[n-1]
    # Constraints: sum_i a[i]*b[k-i] <= M for k in k_indices
    # In matrix form: A_ub @ b <= b_ub
    
    # A_ub[k_idx, j] = coefficient of b[j] in constraint k_idx
    # conv(a, b)[k] = sum_j a[k-j] * b[j] (where valid indices)
    # So A_ub[row, j] = a[k - j] if 0 <= k-j < n, else 0
    
    A_ub = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[i, j] = a[idx]
    
    b_ub = np.full(len(k_indices), M)
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        b = result.x
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 0:
            b = b / s
        return b
    return None

def _lp_refinement(a, deadline, max_iters=20, top_k=50):
    """Iteratively refine using LP direction finding."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        M = float(c.max())
        
        b = _active_set_lp(best_a, M, top_k=top_k)
        if b is None:
            break
        
        # Try different step sizes
        best_step_sc = best_sc
        best_step_a = best_a
        for t_frac in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1 - t_frac) * best_a + t_frac * b
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand = a_cand / s
            sc = _score_norm(a_cand)
            if sc < best_step_sc:
                best_step_sc = sc
                best_step_a = a_cand
        
        if best_step_sc < best_sc * (1 - 1e-10):
            best_sc = best_step_sc
            best_a = best_step_a
        else:
            break
    
    return best_a, best_sc
