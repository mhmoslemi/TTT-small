def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

def _interpolate_to_n(a, n_new):

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with adaptive log-sum-exp continuation
2. Active-set Newton with adaptive active set tracking and CG solve
3. Multi-round refinement with progressively tighter active sets
4. Coordinate-wise polish for final precision
Key improvements: adaptive active set management, tighter tolerances,
cubic line search, and multiple refinement passes.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute g[i] = sum_j a[j] * A_vec[i+j+n-1] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _cg_solve_hess(rhs, A_vec, m, mu, n, max_cg=500, x0=None):
    """Solve (H + mu*I) x = rhs where H is the Hessian defined by A_vec."""
    if x0 is None:
        x = np.zeros(n)
    else:
        x = x0.copy()
    r = rhs - ((2.0 / m) * _corr_a_A(x, A_vec, n) + mu * x)
    p = r.copy()
    rs = np.dot(r, r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for cg_it in range(max_cg):
        if rs < 1e-28 * rs0:
            break
        Ap = (2.0 / m) * _corr_a_A(p, A_vec, n) + mu * p
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if rs_new >= rs * 0.999:
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x

def _lbfgs_adaptive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    # Aggressive beta schedule with more stages at the end
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002, 0.00001, 0.000005
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 4096.0)
        max_iter = min(12000, max(50, int(remaining / (est_time_per_iter * 12))))

        try:
            res = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(res.x)
            a_new = np.maximum(a_new, 0)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:

                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))

def _two_side_transfer(a_init, deadline, max_iter=2000, active_tol=0.005):

    n = len(a_init)

            a_new[min_idx] = max(0.0, a[min_idx] - t_try)

                best_t = t_try

        if best_t <= 0:
            # Try reverse direction
            min_idx2 = int(np.argmax(g_t))
            max_idx2 = int(np.argmin(g_t))
            t_hi2 = a[max_idx2] * 0.99

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:

        a = a / a.sum()

        sc = _score(a)
        if sc < best_sc - 1e-14:

    best_score = np.inf

    def _update(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s_val = float(p.sum())
        if s_val < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_val = float(parent_raw.sum())
            if s_val > 0.01:
                parent = parent_raw / s_val
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        _update(parent, _score(parent), "init")

    # Phase 1: L-BFGS
    print(f"\n[Phase 1] L-BFGS adaptive (t={time.monotonic()-t_start:.1f}s)")
    a_main = _lbfgs_adaptive(parent, deadline, min_time_remaining=700.0)
    _update(a_main, _score(a_main), "lbfgs_1")

    # Phase 2: Active-set Newton (moderate tolerance)
    print(f"\n[Phase 2] Active-set Newton moderate (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 450:
        dl = min(deadline - 450, time.monotonic() + 200)
        a_as1, sc_as1 = _active_set_newton_precise(best_seq, dl, max_iter=300, active_tol_frac=0.005)
        _update(a_as1, sc_as1, "as_mod")

    # Phase 3: L-BFGS round 2
    if best_seq is not None and time.monotonic() < deadline - 350:
        print(f"\n[Phase 3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
        a_main2 = _lbfgs_adaptive(best_seq, deadline, min_time_remaining=300.0)
        _update(a_main2, _score(a_main2), "lbfgs_2")

    # Phase 4: Active-set Newton (tight)
    if best_seq is not None and time.monotonic() < deadline - 250:
        print(f"\n[Phase 4] Active-set Newton tight (t={time.monotonic()-t_start:.1f}s)")
        dl2 = min(deadline - 250, time.monotonic() + 150)
        a_as2, sc_as2 = _active_set_newton_precise(best_seq, dl2, max_iter=400, active_tol_frac=0.002)
        _update(a_as2, sc_as2, "as_tight")

    # Phase 5: Subgradient refinement
    if best_seq is not None and time.monotonic() < deadline - 150:
        print(f"\n[Phase 5] Subgradient (t={time.monotonic()-t_start:.1f}s)")
        dl3 = min(deadline - 150, time.monotonic() + 80)
        a_sg, sc_sg = _subgrad_refine(best_seq, dl3, n_steps=5000, lr_init=0.01)
        _update(a_sg, sc_sg, "subgrad")

    # Phase 6: Active-set Newton (very tight)
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"\n[Phase 6] Active-set Newton vtight (t={time.monotonic()-t_start:.1f}s)")
        dl4 = min(deadline - 100, time.monotonic() + 100)
        a_as3, sc_as3 = _active_set_newton_precise(best_seq, dl4, max_iter=500, active_tol_frac=0.001)
        _update(a_as3, sc_as3, "as_vtight")

    # Phase 7: Coordinate transfer
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"\n[Phase 7] Coordinate transfer (t={time.monotonic()-t_start:.1f}s)")
        dl5 = min(deadline - 50, time.monotonic() + 50)
        a_ct, sc_ct = _two_side_transfer(best_seq, dl5, max_iter=1000, active_tol=0.003)
        _update(a_ct, sc_ct, "coord_transfer")

    # Phase 8: Final active-set Newton
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n[Phase 8] Final AS Newton (t={time.monotonic()-t_start:.1f}s)")
        dl6 = min(deadline - 10, time.monotonic() + 50)
        a_as4, sc_as4 = _active_set_newton_precise(best_seq, dl6, max_iter=500, active_tol_frac=0.0005)
        _update(a_as4, sc_as4, "as_final")

    # Phase 9: Ultra-tight final
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n[Phase 9] Ultra-tight (t={time.monotonic()-t_start:.1f}s)")
        dl7 = min(deadline - 3, time.monotonic() + 30)
        a_as5, sc_as5 = _active_set_newton_precise(best_seq, dl7, max_iter=300, active_tol_frac=0.0002)
        _update(a_as5, sc_as5, "as_ultra")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_val = float(final.sum())
        if s_val < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
