"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_fast(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _kkt_pressure_iteration(a, deadline, max_iters=5000, gamma_start=0.8, gamma_end=0.3, tol=1e-10):
    """
    KKT fixed-point: at optimum, for active a[i]>0:
      P[i] * a[i] = const
    where P[i] = sum_{k in S} w[k] * a[k-i]
    We iterate: a[i] <- a[i] * (const/P[i])^gamma, renormalize.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    best_a = a.copy()
    best_sc = _score_fast(a)
    a_rev = a[::-1]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set
        frac = min(0.1, 0.01 + 0.02 * (it / max_iters))
        tol_active = c_max * (1.0 - frac)
        active = c >= tol_active
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-max(3, n // 100):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = active.sum()

        # Use softmax-like weights on active set
        active_vals = c[active]
        active_range = active_vals.max() - active_vals.min()
        if active_range > 1e-15:
            w_active = np.exp((active_vals - active_vals.max()) / (active_range + 1e-30))
        else:
            w_active = np.ones(n_active)
        w = np.zeros_like(c)
        w[active] = w_active / w_active.sum()

        # Compute pressure P[i] = sum_k w[k] * a[k-i]
        # P = conv(w, a_reversed)[n-1:2n-1]
        p = _full_convolve(w, a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Damped gamma: start aggressive, end conservative
        progress = min(1.0, it / (max_iters * 0.7))
        gamma = gamma_start - (gamma_start - gamma_end) * progress

        # KKT update: a[i] <- a[i] * (mean_p / P[i])^gamma
        p_mean = np.mean(p)
        ratio = p_mean / p
        update = np.power(ratio, gamma)
        a = a * update
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        a_rev = a[::-1]

        # Check improvement
        if it % 10 == 0:
            sc = _score_fast(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
            elif it > 200 and (sc - best_sc) / best_sc < tol:
                # converged
                if it > 500:
                    break

    sc_final = _score_fast(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, _score(best_a)


def _run_lbfgs_improved(a_init, deadline, min_remaining=30.0):
    """L-BFGS with aggressive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    # Estimate good beta from the spread of top conv values
    c_sorted = np.sort(c_cur)[::-1]
    n_top = min(50, len(c_sorted))
    spread = c_sorted[0] - c_sorted[n_top - 1] if n_top > 1 else c_sorted[0] * 0.1
    base_beta = max(spread * 0.5, max_c * 0.01)

    # More aggressive schedule
    betas = base_beta * np.array([1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
                                  0.005, 0.002, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time = 0.008 * max(1.0, n / 4096.0)
        max_iter = min(20000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _peak_reduction(a, deadline, n_rounds=200):
    """
    Directly reduce the peak by finding which a[i] contribute most
    to the max convolution value and reducing them.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score_fast(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find a[i] that contributes most to c[peak_idx]
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # Contribution of a[j] to c[peak_idx]: a[j] * a[peak_idx - j] (if valid)
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        if j_lo >= j_hi:
            break
        j_range = np.arange(j_lo, j_hi + 1)
        contributions = a[j_range] * a[peak_idx - j_range]
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break

        # Reduce top contributors with largest a[j]*a[peak_idx-j]
        # but a[j] large means reducing it helps most
        # Score: contribution / (something) -> reduce elements with high product
        top_k = min(len(j_range), max(5, n // 100))
        top_local = np.argsort(contributions)[-top_k:]
        top_global = j_range[top_local]

        # Try reducing each
        improved = False
        for j in top_global:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            # Try reducing by small factors
            for factor in [0.8, 0.9, 0.95, 1.05, 1.1]:
                a[j] = old_val * factor
                if a[j] < 0:
                    a[j] = old_val
                    break
                s_try = a.sum()
                if s_try < 1e-15:
                    a[j] = old_val
                    break
                a_norm = a / s_try
                c_try = _full_convolve(a_norm, a_norm).max()
                c_old_normalized = c_max / (s / s_try) ** 2
                if c_try < c_max / (s * s) - 1e-15:
                    improved = True
                    break
                else:
                    a[j] = old_val
                    break

        if improved:
            a = a / a.sum()
            sc = _score_fast(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, _score(best_a)


def _coordinate_descent(a, deadline, n_rounds=500):
    """

    # This is complex, so use a simpler approach: evaluate a few candidate values

            old_val = a[i]
            if old_val < 1e-10:

            best_c = c_old_max

def _fourier_blend(a1, a2, rng, weight=0.5):
    """Blend two sequences in Fourier domain for diversity."""
    n = len(a1)
    if len(a2) != n:
        a2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a2)), a2)
    z1 = np.log(np.maximum(a1, 1e-30))
    z2 = np.log(np.maximum(a2, 1e-30))
    z = weight * z1 + (1 - weight) * z2
    # Small random perturbation
    z += rng.standard_normal(n) * 0.01
    return _softmax(z)

        phase3_deadline = t_start + min(1200, deadline - 400)
        a3, sc3 = _kkt_pressure_iteration(best_seq, phase3_deadline,
                                           max_iters=20000, gamma_start=0.6, gamma_end=0.1)
        _update(a3, sc3)
        print(f"  Phase 3 done: score={best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 4: Peak reduction ===
    print(f"\n[Phase 4] Peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a4, sc4 = _peak_reduction(best_seq, t_start + min(1450, deadline - 200), n_rounds=200)
        _update(a4, sc4)

    # === Phase 5: Coordinate descent ===
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a5, sc5 = _coordinate_descent(best_seq, t_start + min(1550, deadline - 150), n_rounds=300)
        _update(a5, sc5)

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [12288, 16384, 6144]
    n_candidates = [x for x in n_candidates if x != n0]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - 100 - time.monotonic()
        if remaining < 60:
            break

        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")

        # Initialize by interpolation
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_uniform(n_try)
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_uniform(n_try)

        # Quick optimization: L-BFGS + KKT
        sub_deadline = min(deadline - 100, time.monotonic() + min(remaining * 0.4, 120))
        a_nr, sc_nr = _run_lbfgs_improved(a_init, sub_deadline, min_remaining=30)
        _update(a_nr, sc_nr)

        sub_deadline2 = min(deadline - 80, time.monotonic() + min(remaining * 0.4, 120))
        a_nr2, sc_nr2 = _kkt_pressure_iteration(a_nr, sub_deadline2,
                                                  max_iters=10000, gamma_start=0.8, gamma_end=0.2)
        _update(a_nr2, sc_nr2)

    # === Phase 7: Final fine-tuning on best ===
    print(f"\n[Phase 7] Final polishing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # L-BFGS with very small beta
        a7, sc7 = _run_lbfgs_improved(best_seq, deadline - 10, min_remaining=5)
        _update(a7, sc7)

        # One more KKT pass
        if time.monotonic() < deadline - 5:
            a7b, sc7b = _kkt_pressure_iteration(best_seq, deadline - 5,
                                                 max_iters=5000, gamma_start=0.4, gamma_end=0.05)
            _update(a7b, sc7b)

    # === Final ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
