"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Nesterov-accelerated subgradient with adaptive active set
- L-BFGS with log-sum-exp smoothing
- Active-set Newton refinement
- Frequency-domain spectral flattening
- Multi-size exploration and diverse restarts
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute g[i] = sum_j a[j] * A_vec[i+j] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _nesterov_subgrad(a_init, deadline, max_iter=10000, lr_init=0.05, momentum_init=0.9):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)  # momentum
    momentum = momentum_init
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        z_lookahead = z - momentum * v
        a = _softmax(z_lookahead)
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set: use top percentile
        # Start with wide band, narrow over time
        progress = min(1.0, i / max(1, max_iter * 0.5))
        tol = (0.01 * (1 - progress) + 0.001 * progress) * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())
        
        # Uniform weight on active set
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        
        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Update with Nesterov momentum
        v = momentum * v + grad
        lr_eff = lr / (1.0 + 0.1 * i / max_iter)
        z = z - lr_eff * v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Adapt momentum (Nesterov schedule)
        if i > 0:
            momentum = (1 + 0.5 / (i + 1)) / (2 + 0.5 / (i + 1))
            momentum = min(momentum, 0.999)
        
        # Periodic evaluation
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                v = np.zeros(n)
            elif i % 500 == 0:
                lr *= 0.8
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


class _ObjFunc:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lsobj(z, self.beta, self.n)

    def grad(self, z):
        return _lsgrad(z, self.beta, self.n)


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            n_active = 2 * n
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:n_active]
            active[top_idx] = True

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def _hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def _cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - _hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = _hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = _cg_solve(g_t)
        w = _cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-12 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-12:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 6))))

        try:
            obj_fn = _ObjFunc(beta, n)
            result = minimize(
                obj_fn,
                z,
                jac=obj_fn.grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_subgrad(a_init, deadline, max_iter=20000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    mu = 0.9

    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        z_la = z - mu * v
        a = _softmax(z_la)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        progress = min(1.0, i / (max_iter * 0.4))
        frac = 0.015 * (1.0 - progress) + 0.001 * progress
        tol = frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        w = np.zeros_like(c)
        w[active] = 1.0 / n_act

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        v = mu * v + grad
        lr_eff = lr / (1.0 + 0.05 * i / max(1, max_iter))
        z = z - lr_eff * v
        z = np.clip(z, -55.0, 55.0)
        z = z - z.mean()

        if i > 0:
            mu = (1.0 + 0.5 / (i + 1.0)) / (2.0 + 0.5 / (i + 1.0))
            mu = min(mu, 0.999)

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    lr = max(lr * 0.7, 0.001)
                    stall_count = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _spectral_flatten(a_init, deadline):
    """Try to flatten the convolution via frequency domain manipulation."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    N = _next_pow2(2 * n - 1)
    spectrum = np.fft.rfft(a, N)
    power = np.abs(spectrum) ** 2

    # Try different smoothing windows
    n_f = len(power)
    for win_frac in [0.01, 0.02, 0.05, 0.1]:
        if time.monotonic() > deadline:
            break
        win_size = max(1, int(n_f * win_frac))
        kernel = np.hanning(win_size)
        kernel /= kernel.sum()

        # Smooth the power spectrum
        power_smooth = np.convolve(power, kernel, mode='same')
        # Ensure non-negativity and preserve DC
        power_smooth = np.maximum(power_smooth, 0)
        # Preserve the DC component (squared sum)
        power_smooth[0] = power[0]

        # Reconstruct: we want a new spectrum with this power
        # Take the phase from the original spectrum
        phase = np.angle(spectrum)
        new_mag = np.sqrt(power_smooth)
        new_spectrum = new_mag * np.exp(1j * phase)

        # Inverse transform
        a_new = np.fft.irfft(new_spectrum, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            spectrum = np.fft.rfft(a, N)
            power = np.abs(spectrum) ** 2

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _cosine_param_optimize(n, deadline, rng):
    """Optimize a sequence parameterized as 1 + sum of cosines."""
    best_a = np.ones(n) / n
    best_sc = _score(best_a)

    # Try different numbers of modes
    for n_modes in [1, 2, 3, 5, 8]:
        if time.monotonic() > deadline:
            break
        # Random initialization of frequencies and amplitudes
        for trial in range(5):
            if time.monotonic() > deadline:
                break
            freqs = rng.integers(1, max(2, n // 6), size=n_modes)
            amps = rng.uniform(0.1, 0.8, size=n_modes)
            phases = rng.uniform(0, 2 * np.pi, size=n_modes)

            t = np.arange(n)
            a = np.ones(n)
            for k in range(n_modes):
                a += amps[k] * np.cos(2 * np.pi * freqs[k] * t / n + phases[k])
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _coordinate_pair_refine(a_init, deadline, n_sweeps=5):
    """Try swapping small amounts between coordinate pairs to reduce max convolution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    peak_indices = np.where(c >= c_max * 0.999)[0]

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        # Pick a peak index and try to reduce it
        if len(peak_indices) == 0:
            break
        k = peak_indices[sweep % len(peak_indices)]
        # c[k] = sum_i a[i]*a[k-i]
        # Try reducing a[i] and increasing a[j] for some pair
        # This is a heuristic: reduce the large elements contributing to c[k]
        # and spread to smaller elements

        # Find the largest contributors to c[k]
        valid_i = range(max(0, k - (n - 1)), min(n, k + 1))
        contribs = []
        for i in valid_i:
            j = k - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        if len(contribs) < 2:
            continue

        # Try moving mass from the largest contributor to a random small element
        top_contrib, i_top, j_top = contribs[0]
        # Find a small element far from the peak
        small_idx = np.argsort(a)[:n // 4]
        if len(small_idx) == 0:
            continue
        target = small_idx[len(small_idx) // 2]

        # Try different step sizes
        for frac in [0.01, 0.05, 0.1]:
            amt = frac * a[i_top]
            a_try = a.copy()
            a_try[i_top] -= amt
            a_try[target] += amt
            if np.min(a_try) < -1e-10:
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                c = c_try
                c_max = float(c.max())
                peak_indices = np.where(c >= c_max * 0.999)[0]
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with adaptive log-sum-exp,
    Nesterov subgradient, active-set Newton, spectral flattening,
    cosine-parameterized search, coordinate pair refinement,
    and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Nesterov subgradient
    print(f"\n[Phase 2] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        sg_deadline = min(deadline - 500, time.monotonic() + 300)
        a_sg, sc_sg = _nesterov_subgrad(best_seq, sg_deadline, max_iter=15000, lr_init=0.04)
        update_best(a_sg, sc_sg, "nesterov_sg")

    # Phase 3: Active-set Newton
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        as_deadline = min(deadline - 400, time.monotonic() + 150)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "active_set_newton")

        if time.monotonic() < deadline - 370:
            as_deadline2 = min(deadline - 370, time.monotonic() + 80)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_deadline2, max_iter=100, tol_active=0.0003)
            update_best(a_as2, sc_as2, "newton_tight")

    # Phase 4: Spectral flattening
    print(f"\n[Phase 4] Spectral flattening (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        sf_deadline = min(deadline - 350, time.monotonic() + 60)
        a_sf, sc_sf = _spectral_flatten(best_seq, sf_deadline)
        update_best(a_sf, sc_sf, "spectral")
        if sc_sf < best_score:
            a_sf2 = _run_lbfgs(a_sf, deadline, min_time_remaining=200.0)
            update_best(a_sf2, _score(a_sf2), "lbfgs_post_spectral")

    # Phase 5: Cosine-parameterized search
    print(f"\n[Phase 5] Cosine param search (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        n_cos = n_parent if best_seq is not None else 8192
        cp_deadline = min(deadline - 300, time.monotonic() + 60)
        a_cp, sc_cp = _cosine_param_optimize(n_cos, cp_deadline, rng)
        if sc_cp < best_score + 0.1:
            update_best(a_cp, sc_cp, "cosine_init")
            a_cp2 = _run_lbfgs(a_cp, deadline, min_time_remaining=150.0)
            update_best(a_cp2, _score(a_cp2), "cosine_optimized")

    # Phase 6: Coordinate pair refinement
    print(f"\n[Phase 6] Coordinate pair refine (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        cpr_deadline = min(deadline - 250, time.monotonic() + 60)
        a_cpr, sc_cpr = _coordinate_pair_refine(best_seq, cpr_deadline, n_sweeps=20)
        update_best(a_cpr, sc_cpr, "coord_pair")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
        update_best(a_opt, _score(a_opt), f"n_{n_try}")

        if time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 40, time.monotonic() + 40),
                                                  max_iter=80, tol_active=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 8: Diverse restarts
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_strategies = [
            ("perturb_003", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_008", lambda: _perturb(best_seq, 0.08, rng)),
            ("perturb_015", lambda: _perturb(best_seq, 0.15, rng)),
            ("fourier_3", lambda: _perturb_fourier(best_seq, rng, n_modes=3)),
            ("fourier_8", lambda: _perturb_fourier(best_seq, rng, n_modes=8)),
        ]
        for label, gen_fn in restart_strategies:
            if time.monotonic() > deadline - 20:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 8, time.monotonic() + 25),
                                                  max_iter=60, tol_active=0.0005)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 9: Final intensive polish
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very tight active-set Newton
        as_final = min(deadline - 2, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=300, tol_active=0.0002)
        update_best(a_fin, sc_fin, "final_newton")

        # Final Nesterov with very small lr
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _nesterov_subgrad(best_seq, deadline - 2, max_iter=5000, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_nesterov")

        # One more L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 3 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    obj_fn = _ObjFunc(beta_final, n)
                    result = minimize(
                        obj_fn,
                        z,
                        jac=obj_fn.grad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    sc_fin3 = _score(a_fin3)
                    update_best(a_fin3, sc_fin3, "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
