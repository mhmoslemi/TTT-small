        # Use weights proportional to exp((c_k - c_max) / (tol*0.5)) for emphasis on peaks
        if n_active > 1 and tol > 0:
            weights = np.exp((c[active] - c_max) / (tol * 0.5))
            weights = weights / weights.sum()
            W[active] = weights
        else:
            W[active] = 1.0 / n_active

        # Compute gradient: g = H*a where H is the weighted Hankel
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-15 * n:
            break

        # Newton step via CG
        Hg = _cg_solve(W, n, g, max_iter=300, tol=1e-15)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300, tol=1e-15)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Damped Newton step with backtracking
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.01 * alpha * gn:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            # Gradient fallback
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max * (1.0 - 1e-15):
                    a = a_try
                    break
                alpha *= 0.5

        iter_count += 1
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # KKT check: if gradient is nearly flat on support, we're close to optimum
        if iter_count % 10 == 0:
            c_check = _full_convolve(a, a)
            c_max_check = float(c_check.max())
            tol_check = 1e-6 * c_max_check
            active_check = c_check >= (c_max_check - tol_check)
            W_check = np.zeros(2 * n - 1)
            W_check[active_check] = 1.0 / active_check.sum()
            g_check = _hankel_matvec(a, W_check, n)
            g_check_t = g_check - np.mean(g_check)
            kkt_residual = float(np.linalg.norm(g_check_t)) / (n * c_max_check)
            if kkt_residual < 1e-8:
                if verbose_check(rnd):
                    print(f"    KKT satisfied at round {rnd}, residual={kkt_residual:.2e}")
                break

    return best_a, best_sc


def verbose_check(rnd):
    return rnd % 50 == 0


def _support_prune_and_reopt(a, deadline, min_time_remaining=120.0):
    """Iteratively prune support and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    idx = np.argsort(a)

    for frac in [0.01, 0.03, 0.05, 0.10, 0.15, 0.20, 0.30]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            idx = np.argsort(best_a)
            # Quick re-optimize
            if time.monotonic() + min_time_remaining > deadline:
                a_re = _run_lbfgs(best_a, deadline, min_time_remaining=min_time_remaining)
                sc_re = _score(a_re)
                if sc_re < best_sc:
                    best_sc = sc_re
                    best_a = a_re

    return best_a, best_sc


def _make_profile(n, kind="ramp"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        # Exponential ramp: a_i = exp(c * i / n)
        t = np.linspace(0, 1, n)
        a = np.exp(2.0 * t)
        return a / a.sum()
    elif kind == "ramp_quadratic":
        t = np.linspace(0, 1, n)
        a = (t + 0.1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 1.0))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_ramp":
        t = np.linspace(0, 1, n)
        a = (t + 0.05) ** 3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining L-BFGS beta continuation, active set Newton,
    support pruning, and multi-n exploration. Uses asymmetric ramp profiles
    for better initial conditions.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "ramp")
        print(f"No parent, using ramp n={n_parent}")
        update_best(parent, _score(parent), "init_ramp")

    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 1] Main L-BFGS from parent (t={elapsed:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Newton polish
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 2] Newton polish 1 (t={elapsed:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 300))
        update_best(a2, sc2, "newton_1")

    # Phase 3: Second L-BFGS pass (refined)
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 3] L-BFGS pass 2 (t={elapsed:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a3 = _run_lbfgs(best_seq, deadline, min_time_remaining=250.0)
        update_best(a3, _score(a3), "lbfgs_2")

    # Phase 4: Newton polish 2
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 4] Newton polish 2 (t={elapsed:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 250, time.monotonic() + 150),
                                         max_rounds=150, tight_tol=0.00005)
        update_best(a4, sc4, "newton_2")

    # Phase 5: Support pruning
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 5] Support pruning (t={elapsed:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _support_prune_and_reopt(best_seq, deadline, min_time_remaining=100.0)
        update_best(a5, sc5, "pruning")

    # Phase 6: Multi-n exploration
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 6] Multi-n exploration (t={elapsed:.1f}s)")
    n_candidates = [16384, 4096, 2048, 32768]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        # Try both ramp and resampled
        a_init_ramp = _make_profile(n_try, "ramp")

        # Use better of resampled and ramp
        sc_res = _score(a_init)
        sc_ramp = _score(a_init_ramp)
        a_start = a_init if sc_res < sc_ramp else a_init_ramp

        a_opt = _run_lbfgs(a_start, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.001 and time.monotonic() < deadline - 50:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 40, time.monotonic() + 50),
                                                 max_rounds=80, tight_tol=0.00005)
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 7: Diverse restarts from asymmetric profiles
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 7] Diverse restarts (t={elapsed:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_profiles = ["ramp_quadratic", "power_ramp", "cos_ramp"]
        for i, prof_name in enumerate(restart_profiles):
            if time.monotonic() > deadline - 20:
                break
            n_r = len(best_seq)
            p_start = _make_profile(n_r, prof_name)
            # Blend with current best
            blend = 0.5 * p_start + 0.5 * best_seq
            blend = blend / blend.sum()

            if time.monotonic() > deadline - 15:
                break
            a_r = _run_lbfgs(blend, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{prof_name}")

    # Phase 8: Final ultra-precise polish
    elapsed = time.monotonic() - t_start
    print(f"\n[Phase 8] Final ultra-precise polish (t={elapsed:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very tight Newton
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1,
                                               max_rounds=500, tight_tol=0.00001)
        update_best(a_fin, sc_fin, "final_newton_tight")

        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-10
            remaining = deadline - 2 - time.monotonic()
            if remaining > 1:
                max_iter = min(5000, max(50, int(remaining / 0.005)))
                try:
                    result = minimize(
                        _smoobj, z, args=(beta_fin, n),
                        jac=_smograd,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-16,
                                 'maxcor': 80, 'maxls': 300}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "ramp")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "ramp")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
