        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_grad_elem(c, k, n):
    """Gradient of c[k] = sum_j a[j]*a[k-j] w.r.t. a[m] is 2*a[k-m] (reversed/shifted)."""
    a = _S  # global scratch
    out = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        m = k - j
        out[m] += 2.0 * a[j] if j != m else a[j]
    # Actually simpler: grad of c[k] w.r.t. a[m] = a[k-m] + a[k-m] = 2*a[k-m] if 0<=k-m<n
    out = np.zeros(n)
    if 0 <= k < 2 * n - 1:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        for m in range(start, end + 1):
            jm = k - m
            if 0 <= jm < n:
                out[m] += a[jm]
    # The gradient of c[k] = sum_{j+m=k} a[j]*a[m] is: for each m, d/d(a[m]) gives a[k-m]
    # But since c[k] is symmetric in the two a's, and we're differentiating a*a,
    # d(c[k])/d(a[m]) = sum_j [delta_{j,m} a[k-j] + a[j] delta_{k-j,m}]
    # = a[k-m] + a[k-m] = 2*a[k-m] if 0<=k-m<n, else a[k-m] if m==k-m (i.e. k=2m)
    # Actually for the full convolution of a with itself:
    # c[k] = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} a[j] * a[k-j]
    # d(c[k])/d(a[m]) = a[k-m] (from j=m term) + a[k-m] (from k-j=m term) if m != k-m
    #                   = a[k-m] (if m == k-m, i.e., k=2m)
    # So: d(c[k])/d(a[m]) = 2*a[k-m] if m != k-m (and 0<=k-m<n), else a[k-m]
    out = np.zeros(n)
    if 0 <= k < 2 * n - 1:
        km = k - m if False else None
        for m in range(n):
            jm = k - m
            if 0 <= jm < n:
                if m == jm:
                    out[m] = a[jm]
                else:
                    out[m] = 2.0 * a[jm]
    return out


def _conv_grad_vec(a, k):
    """Vectorized: gradient of c[k] = full_conv(a,a)[k] w.r.t. a."""
    n = len(a)
    if k < 0 or k >= 2 * n - 1:
        return np.zeros(n)
    # d(c[k])/d(a[m]) for each m
    # c[k] = sum_j a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
    # For each m: contribution from j=m (gives a[k-m]) and from k-j=m i.e. j=k-m (gives a[k-m])
    # So d(c[k])/d(a[m]) = a[k-m] + a[k-m] = 2*a[k-m] if m != k-m
    #                      = a[k-m] if m == k-m (i.e., 2m=k)
    out = np.zeros(n)
    valid_m = np.arange(n)
    valid_jm = k - valid_m
    valid = (valid_jm >= 0) & (valid_jm < n)
    if not np.any(valid):
        return out
    out[valid] = 2.0 * a[valid_jm[valid]]
    # Fix the diagonal case where m == k-m, i.e., m = k/2
    if k % 2 == 0:
        mid = k // 2
        if 0 <= mid < n:
            out[mid] = a[mid]  # not 2*a[mid]
    return out


def _smoobj_grad(a, beta, n):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. a (not z)."""
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Objective: 2n*(c_max + beta*log(s)) = 2n*sum(w*c)
    # Gradient w.r.t. a: 2n * sum_k w[k] * grad(c[k])
    W = float(np.dot(w, c))
    # Compute weighted gradient
    # grad_W = sum_k w[k] * grad(c[k])
    # grad(c[k])[m] = 2*a[k-m] (or a[k-m] if k=2m)
    # grad_W[m] = sum_k w[k] * 2*a[k-m] (with correction)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    # h[m] = sum_k w[k] * a[k-m+n-1] ... hmm let me think
    # Actually _full_convolve(w, a_rev) has length (2n-1)+(n-1)-1 = 3n-3
    # (w * a_rev)[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-(k-j)] = sum_j w[j] * a[n-1-k+j]
    # We want sum_k w[k] * a[k-m] for k-m in [0,n-1]
    # = sum_k w[k] * a_rev[n-1-(k-m)] = sum_k w[k] * a_rev[n-1-m-k]
    # Let l = n-1-m-k, then k = n-1-m-l
    # = sum_l a_rev[l] * w[n-1-m-l]
    # This is the cross-correlation of w and a_rev evaluated at n-1-m
    # In convolution terms: (w * a_rev)[n-1-m] = sum_j w[j]*a_rev[n-1-m-j]
    # Let j = n-1-m-l: = sum_l w[n-1-m-l]*a_rev[l]
    # Yes! So sum_k w[k]*a[k-m] = (w * a_rev)[n-1-m] ... wait, let me verify with indices.
    # (w * a_rev)[p] = sum_j w[j] * a_rev[p-j] where a_rev[q] = a[n-1-q]
    # = sum_j w[j] * a[n-1-(p-j)] = sum_j w[j] * a[n-1-p+j]
    # We want: sum_k w[k] * a[k-m]
    # Set n-1-p+j = k-m => j = k-m-n+1+p... this is getting confusing.
    # Let me just use the formula: grad_W[m] = 2*h[m] - h[m] (diagonal correction)
    # Actually the previous code used: h = wc[n-1:2*n-1] and grad = 4*n*a*(h-W)
    # But that's for the softmax parametrization. Here I want grad w.r.t. a directly.
    
    # Let me just compute it directly for correctness:
    # grad_W = sum_k w[k] * grad_c_k where grad_c_k is vector of length n
    # For efficiency, use the convolution trick:
    # grad_W[m] = sum_k w[k] * (2*a[k-m] if k!=2m else a[k-m])
    # = 2*sum_k w[k]*a[k-m] (approximately, ignoring diagonal)
    # sum_k w[k]*a[k-m] can be computed via convolution
    
    # (w convolved with reversed a):
    # Let's define: g[m] = sum_k w[k] * a[(k-m) mod n]? No, that's circular.
    # For full convolution: (w * a_flipped)[p] where a_flipped[q] = a[n-1-q]
    # = sum_j w[j] * a_flipped[p-j] = sum_j w[j] * a[n-1-p+j]
    # We want: for each m, S[m] = sum_k w[k] * a[k-m] where 0<=k-m<n
    # = sum_k w[k] * a_flipped[n-1-(k-m)] = sum_k w[k] * a_flipped[n-1-m-k]
    # Let p = n-1-m, then S[m] = sum_k w[k] * a_flipped[p-k]
    # But (w * a_flipped)[p] = sum_j w[j] * a_flipped[p-j] (standard convolution)
    # So S[m] = (w * a_flipped)[n-1-m]
    
    a_flipped = a[::-1]
    wc2 = _full_convolve(w, a_flipped)
    # wc2 has length (2n-1)+(n-1)-1 = 3n-3
    # S[m] = wc2[n-1-m] for m = 0..n-1
    S = wc2[n - 1: 2 * n - 1][::-1]
    # grad_W[m] = 2*S[m] - S[m] (diagonal correction where k=2m)
    # The diagonal correction: for m where 2m is in [0,2n-2], subtract S[m] (one copy of a[m]*w[2m])
    # Actually: grad_W[m] = sum_k w[k]*(2*a[k-m] if k!=2m else a[k-m])
    # = 2*S[m] - sum_{k=2m} w[k]*a[k-m] = 2*S[m] - w[2*m]*a[m] (if 2m < 2n-1)
    diag_correction = np.zeros(n)
    for m in range(n):
        k_diag = 2 * m
        if 0 <= k_diag < 2 * n - 1:
            diag_correction[m] = w[k_diag] * a[m]
    grad_W = 2.0 * S - diag_correction
    
    obj = 2.0 * n * W
    grad = 2.0 * n * grad_W
    return obj, grad


def _run_continuation(a_init, deadline, min_remaining=60):
    """Aggressive continuation method with L-BFGS on softmax parametrization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    c0 = _full_convolve(a, a)
    c_max0 = c0.max()
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_max0 * 0.001:
        gap = c_max0 * 0.001
    base_beta = max(gap * 2, c_max0 * 0.01)
    
    # More aggressive beta schedule with many more steps at small beta
    mults = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
             0.004, 0.002, 0.001, 0.0005, 0.00025, 0.00012, 0.00006,
             0.00003, 0.000015, 0.000007]
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
    
    for i, mult in enumerate(mults):
        beta = base_beta * mult
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        
        # More iterations at small betas
        if mult > 0.01:
            max_iter = min(2000, max(20, int(remaining / (est_time_per_iter * 8))))
        elif mult > 0.001:
            max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 6))))
        else:
            max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 4))))
        
        # Time budget for this beta stage
        time_budget = remaining * (0.7 if mult < 0.001 else 0.5)
        stage_deadline = time.monotonic() + min(time_budget, remaining - 1)
        
        def smooth_obj(z_val):
            a_s = _softmax(z_val)
            c_s = _full_convolve(a_s, a_s)
            cm = c_s.max()
            shifted = (c_s - cm) / beta
            e_s = np.exp(np.clip(shifted, -700, 0))
            s_s = e_s.sum()
            if s_s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s_s))
        
        def smooth_grad(z_val):
            a_s = _softmax(z_val)
            c_s = _full_convolve(a_s, a_s)
            cm = c_s.max()
            shifted = (c_s - cm) / beta
            e_s = np.exp(np.clip(shifted, -700, 0))
            s_s = e_s.sum()
            if s_s < 1e-300:
                w_s = np.zeros_like(c_s)
                w_s[int(np.argmax(c_s))] = 1.0
            else:
                w_s = e_s / s_s
            a_rev = a_s[::-1]
            wc_s = _full_convolve(w_s, a_rev)
            h_s = wc_s[n - 1:2 * n - 1]
            W_s = float(np.dot(w_s, c_s))
            return 4.0 * n * a_s * (h_s - W_s)
        
        try:
            result = minimize(
                smooth_obj, z, jac=smooth_grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
                print(f"    beta={beta:.2e}: score={best_sc:.10f} (improved)")
        except Exception as ex:
            print(f"    beta={beta:.2e}: error {ex}")
    
    return best_a


def _active_set_polish_v2(a_init, deadline, tol_start=0.005, tol_end=0.00002):
    """Improved active-set gradient descent with adaptive step size and restart."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.03
    tol_frac = tol_start
    no_improve = 0
    max_stall = 300
    stall_resets = 0
    max_stall_resets = 12
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Weighted active set: weights proportional to how close to max
        w = np.zeros_like(c)
        active_vals = c[active]
        w_active = (active_vals - (c_max - tol)) / (tol + 1e-30)
        w_active = np.maximum(w_active, 0.1)
        w[active] = w_active / w_active.sum()
        
        # Compute gradient of weighted max
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * h - W * np.ones(n)  # gradient of sum(w*c) w.r.t. a
        # Project to sum(a)=1 constraint
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-16:
            break
        
        # Backtracking line search
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        a_old_sc = current_sc
        for _ in range(50):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.8
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = 0.02
                no_improve = 0
                stall_resets += 1
                if stall_resets > max_stall_resets:
                    break
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.98)
            lr = min(0.05, lr * 1.01)
    
    return best_a, _score(best_a)


def _peak_shaving(a_init, deadline, n_peaks=10):
    """Novel peak-shaving: directly reduce contributions to top peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for iteration in range(100):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        peak_indices = peak_indices[peak_indices < 2 * n - 1]
        
        # For each peak, compute which a elements contribute most
        # and reduce them
        total_reduction = np.zeros(n)
        for k in peak_indices:
            if k < 0 or k >= 2 * n - 1:
                continue
            # c[k] = sum_j a[j]*a[k-j], gradient w.r.t. a[m] = 2*a[k-m] (approx)
            # Reduce a[m] proportionally to a[k-m] (the "other factor")
            m_range = range(max(0, k - (n - 1)), min(n - 1, k) + 1)
            for m in m_range:
                jm = k - m
                if 0 <= jm < n:
                    total_reduction[m] += a[jm]
        
        # Normalize reduction
        tr_max = total_reduction.max()
        if tr_max < 1e-15:
            break
        total_reduction = total_reduction / tr_max
        
        # Apply reduction with varying strength
        for eta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_try = a * (1.0 - eta * total_reduction)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                a = a_try
                best_sc = sc_try
                best_a = a.copy()
                break
            elif sc_try < current_sc - 1e-12:
                a = a_try
                break
        else:
            # No improvement, try angle perturbation
            break
    
    return best_a, _score(best_a)


def _sa_polish(a_init, deadline, n_steps=5000):
    """Simulated annealing for final polish."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    rng = np.random.default_rng(12345)
    T_start = 0.05
    T_end = 0.0001
    n_iter = min(n_steps, max(100, int((deadline - time.monotonic()) * 10)))
    
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        T = T_start * (T_end / T_start) ** (i / max(1, n_iter - 1))
        
        # Random perturbation: pick a random block and shift energy
        n_block = rng.integers(max(1, n // 100), max(2, n // 10))
        start = rng.integers(0, n - n_block + 1)
        
        a_try = a.copy()
        # Move some energy from random positions to others
        src = start + rng.integers(0, n_block)
        dst = (src + rng.integers(-n_block, n_block + 1)) % n
        amount = T * a.mean() * rng.uniform(0.5, 2.0)
        
        a_try[src] -= amount
        a_try[dst] += amount
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        
        sc_try = _score(a_try)
        sc_cur = 2.0 * n * _full_convolve(a, a).max()
        
        if sc_try < sc_cur:
            a = a_try
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a.copy()
        else:
            # Accept with probability exp(-(sc_try - sc_cur)/T)
            delta = (sc_try - sc_cur) / (T + 1e-15)
            if np.log(rng.random()) > -delta:
                a = a_try
    
    return best_a, _score(best_a)


def _multiplicative_update(a_init, deadline, n_iter=200):
    """Multiplicative (ratio) update that preserves non-negativity."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        
        # Compute "pressure" at each position: how much this element contributes to peaks
        pressure = np.zeros(n)
        c_sorted_idx = np.argsort(c)[::-1][:20]
        for k in c_sorted_idx:
            if k < 0 or k >= 2 * n - 1:
                continue
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            for m in range(m_lo, m_hi + 1):
                jm = k - m
                if 0 <= jm < n:
                    pressure[m] += c[k] * a[jm]
        
        # Normalized pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # Multiplicative update: reduce where pressure is high
        eta = 0.01 * (1.0 + 0.5 * np.cos(i / float(n_iter) * np.pi))
        a_try = a * np.exp(-eta * pressure)
        s_try = a_try.sum()
        if s_try < 1e-15:
            break
        a_try = a_try / s_try
        a = a_try
    
    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 17.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_mix":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.3)**2) + 0.8*np.exp(-15*(t-0.6)**2) + 0.5*np.exp(-10*(t-0.85)**2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _shrinkage(a, deadline, fracs=None):
    """Try zeroing out smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    if fracs is None:
        fracs = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    
    a_sorted_idx = np.argsort(a)
    for frac in fracs:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining aggressive continuation L-BFGS,
    active-set GD, peak shaving, multiplicative updates, and SA polish.
    Explores multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Aggressive continuation on parent n ===
    print(f"\n[P1] Continuation L-BFGS on n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + (deadline - t_start) * 0.35
    a1 = _run_continuation(parent, p1_deadline, min_remaining=120)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} (n={n_parent}) [continuation] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish ===
    print(f"\n[P2] Active-set polish (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline - 400, time.monotonic() + 200)
    a2, sc2 = _active_set_polish_v2(best_seq, p2_deadline, tol_start=0.004, tol_end=0.00003)
    if sc2 < best_score:
        best_seq = a2.copy()
        best_score = sc2
        print(f"  NEW BEST: {best_score:.10f} [active_set] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Peak shaving ===
    print(f"\n[P3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = min(deadline - 300, time.monotonic() + 120)
    a3, sc3 = _peak_shaving(best_seq, p3_deadline, n_peaks=15)
    if sc3 < best_score:
        best_seq = a3.copy()
        best_score = sc3
        print(f"  NEW BEST: {best_score:.10f} [peak_shave] t={time.monotonic()-t_start:.1f}s")
    # Re-polish after peak shaving
    if time.monotonic() < deadline - 250:
        p3b_deadline = min(deadline - 250, time.monotonic() + 80)
        a3b, sc3b = _active_set_polish_v2(best_seq, p3b_deadline, tol_start=0.002, tol_end=0.00002)
        if sc3b < best_score:
            best_seq = a3b.copy()
            best_score = sc3b
            print(f"  NEW BEST: {best_score:.10f} [polish_post_shave] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multiplicative update ===
    print(f"\n[P4] Multiplicative update (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, time.monotonic() + 80)
        a4, sc4 = _multiplicative_update(best_seq, p4_deadline, n_iter=100)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [mult_update] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Shrinkage ===
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a5, sc5 = _shrinkage(best_seq, min(deadline - 100, time.monotonic() + 60))
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 80:
                a5b = _run_continuation(best_seq, min(deadline - 30, time.monotonic() + 50), min_remaining=20)
                sc5b = _score(a5b)
                if sc5b < best_score:
                    best_seq = a5b.copy()
                    best_score = sc5b
                    print(f"  NEW BEST: {best_score:.10f} [post_shrink_opt] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    n_candidates = sorted(set([n_best, n_best // 2, n_best * 2,
                                4096, 6144, 10240, 12288]))
    # Filter reasonable sizes
    n_candidates = [n for n in n_candidates if 1024 <= n <= 16384]
    
    explored = set()
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try in explored:
            continue
        explored.add(n_try)
        if n_try == n_best:
            continue
        
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        
        # Run a shorter continuation
        sub_deadline = min(deadline - 30, time.monotonic() + 60)
        a_opt = _run_continuation(a_init, sub_deadline, min_remaining=5)
        sc_opt = _score(a_opt)
        
        # Quick active-set polish
        if time.monotonic() < deadline - 20 and sc_opt < best_score + 0.01:
            a_pol, sc_pol = _active_set_polish_v2(a_opt, min(deadline - 15, time.monotonic() + 25),
                                                  tol_start=0.003, tol_end=0.0001)
            if sc_pol < sc_opt:
                sc_opt = sc_pol
                a_opt = a_pol
        
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Diverse restarts on best n ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restarts = [
            ("taper", _make_profile(n_best, "tapered")),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("bimo", _make_profile(n_best, "bimodal")),
            ("peri", _make_profile(n_best, "periodic")),
            ("gauss", _make_profile(n_best, "gauss_mix")),
            ("perturb", _softmax(np.log(np.maximum(best_seq, 1e-30)) + 0.05 * rng.standard_normal(n_best))),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 10:
                break
            sc_r = _score(a_r)
            if sc_r < best_score:
                # This structured init is already better, optimize it
                a_opt = _run_continuation(a_r, min(deadline - 5, time.monotonic() + 30), min_remaining=3)
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")
            elif sc_r < best_score + 0.05:
                # Close enough, try to optimize
                a_opt = _run_continuation(a_r, min(deadline - 5, time.monotonic() + 20), min_remaining=3)
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very tight active-set
        a_fin, sc_fin = _active_set_polish_v2(best_seq, deadline - 3,
                                              tol_start=0.0005, tol_end=0.000005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_active] t={time.monotonic()-t_start:.1f}s")
        
        # SA polish
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            a_sa, sc_sa = _sa_polish(best_seq, deadline - 2, n_steps=3000)
            if sc_sa < best_score:
                best_seq = a_sa.copy()
                best_score = sc_sa
                print(f"  NEW BEST: {best_score:.10f} [final_sa] t={time.monotonic()-t_start:.1f}s")
        
        # One more tiny continuation
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            a_fin2 = _run_continuation(best_seq, deadline - 1, min_remaining=0)
            sc_fin2 = _score(a_fin2)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_cont] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Also verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal={sc_final:.10f}, evaluated={eval_score:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        except Exception:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    else:
        fallback = _make_profile(256, "uniform")
        print(f"FALLBACK: score={_score(fallback):.10f}")
        return [float(x) for x in fallback]
