"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines spectral pressure shaping, Adam in log-space with adaptive temperature,
pairwise coordinate descent, multi-scale exploration (n=4096, 8192, 16384),
and diverse restart strategies.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_full(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * np.max(c))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_grad(a, c, T):
    n = len(a)
    c_shift = (c - c.max()) / T
    e = np.exp(c_shift)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_f = _fft_convolve(a, p_rev)
    return 2.0 * conv_f[n - 1:2 * n - 1][::-1]


def _spectral_pressure(a, n_steps, dt_limit=None, t_start=None):
    """
    New method: iteratively apply 'pressure' from peak positions of C
    to redistribute mass in a, using exponential tilting.
    """
    n = len(a)
    a = a.copy()
    for step in range(n_steps):
        if dt_limit is not None and time.monotonic() > dt_limit:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        c_avg = float(c.sum()) / len(c)
        # Temperature: start high, decrease
        T = c_max * (0.05 + 0.005 * (step / max(1, n_steps)))
        # Peak weights
        w = np.exp((c - c_max) / T)
        w = w / w.sum()
        # Compute pressure on each a[j]
        # pressure[j] = sum_m w[m] * a[m-j] (for valid m-j)
        # = cross-correlation at lag j (from the formula derived above)
        w_rev = w[::-1]
        conv_f = _fft_convolve(a, w_rev)
        pressure = conv_f[n - 1:2 * n - 1][::-1]
        # Update: exponential tilting
        lr = 0.3
        a = a * np.exp(-lr * pressure / (pressure.max() + 1e-30))
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
    return a


def _adam_log(a_init, deadline, T_start=0.01, T_end=0.001,
              lr_start=0.05, lr_end=0.003, max_iter=500000,
              eval_interval=200, clip_norm=0.5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _smoothed_grad(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _pairwise_cd(a, deadline, max_rounds=500):
    """
    Pairwise coordinate descent: for the peak position m, find the pair
    (i, m-i) that contributes most, and optimize their values jointly.
    """
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(best_a, best_a)
    best_score = float(c.max())  # normalized score (sum=1)
    tol = 1e-12

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        # Find top active positions
        active = np.where(c > 0.9 * c_max)[0]
        improved = False

        for m in active:
            if m >= 2 * n - 1:
                continue
            # Valid pairs (i, m-i) with 0 <= i < n, 0 <= m-i < n
            i_lo = max(0, m - n + 1)
            i_hi = min(m, n - 1)
            if i_lo > i_hi:
                continue

            # Find the pair with maximum contribution
            best_pair = None
            best_contrib = 0
            for i in range(i_lo, i_hi + 1):
                j = m - i
                contrib = best_a[i] * best_a[j]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_pair = (i, j)

            if best_pair is None:
                continue
            i, j = best_pair
            if i == j:
                # C[m] contains a[i]^2; reducing a[i] reduces C[m] by 2*a[i]*delta
                # but also affects other C[k]
                # Try reducing a[i] and adding to a random other position
                for scale in [0.7, 0.8, 0.5, 0.9]:
                    a_cand = best_a.copy()
                    old = a_cand[i]
                    a_cand[i] = old * scale
                    # Add the freed mass to a position far from m
                    # Find position k that minimizes contribution to C[m]
                    a_cand = a_cand / a_cand.sum()
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max())
                    if sc < best_score - tol:
                        best_score = sc
                        best_a = a_cand
                        improved = True
                        break
            else:
                # C[m] contains a[i]*a[j]; try to reduce this product
                # while maintaining sum
                # Strategy: reduce both a[i] and a[j] by some amount, add to others
                for fi in [0.7, 0.8, 0.5]:
                    for fj in [0.7, 0.8, 0.5]:
                        a_cand = best_a.copy()
                        a_cand[i] *= fi
                        a_cand[j] *= fj
                        a_cand = a_cand / a_cand.sum()
                        c_cand = _fft_convolve(a_cand, a_cand)
                        sc = float(c_cand.max())
                        if sc < best_score - tol:
                            best_score = sc
                            best_a = a_cand
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break  # restart the active set computation

        if not improved:
            # Try single-coordinate moves as fallback
            any_improved = False
            # Focus on high-mass elements that contribute to peaks
            for m in active[:5]:
                i_lo = max(0, m - n + 1)
                i_hi = min(m, n - 1)
                candidates = np.random.choice(range(i_lo, i_hi + 1),
                                           size=min(20, i_hi - i_lo + 1), replace=False)
                for i in candidates:
                    for scale in [0.5, 0.7, 0.85, 0.3]:
                        a_cand = best_a.copy()
                        a_cand[i] *= scale
                        a_cand = a_cand / a_cand.sum()
                        c_cand = _fft_convolve(a_cand, a_cand)
                        sc = float(c_cand.max())
                        if sc < best_score - tol:
                            best_score = sc
                            best_a = a_cand
                            any_improved = True
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not any_improved:
                break

    return best_a, best_score

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _cosine_init(n):
    j = np.arange(n)
    a = np.cos(np.pi * j / (n + 1)) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _exponential_init(n, rate=1.0):
    j = np.arange(n)
    a = np.exp(-rate * j / n)
    a = a + np.exp(-rate * (n - 1 - j) / n)  # symmetric
    return a / a.sum()


def _two_hump_init(n):
    j = np.arange(n)
    center1 = n * 0.25
    center2 = n * 0.75
    sigma = n * 0.1
    a = np.exp(-0.5 * ((j - center1) / sigma) ** 2) + np.exp(-0.5 * ((j - center2) / sigma) ** 2)
    return a / a.sum()


def _random_init(n, rng):
    a = np.exp(rng.standard_normal(n) * 0.5)
    return a / a.sum()
