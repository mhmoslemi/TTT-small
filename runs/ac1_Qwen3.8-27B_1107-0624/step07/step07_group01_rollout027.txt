import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    if target_sum <= 0:
        return np.zeros(n)
    v_scaled = v / target_sum
    v_sorted = np.sort(v_scaled)[::-1]
    cumsum = np.cumsum(v_sorted)
    w = np.arange(1, n + 1, dtype=np.float64)
    condition = v_sorted * w > (cumsum - 1.0)
    rho_vals = np.where(condition)[0]
    if len(rho_vals) == 0:
        return np.full(n, target_sum / n)
    rho = rho_vals[-1] + 1
    theta = (cumsum[rho - 1] - 1.0) / rho
    result = np.maximum(v_scaled - theta, 0.0)
    return result * target_sum


def _grad_pnorm(a, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) where c = a*a (convolution)."""
    n = len(a)

        return np.zeros(n)

    if sum_cp < 1e-300:
        return np.zeros(n)
    w = c ** (p - 1)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    factor = 2.0 * n * (sum_cp) ** (1.0 / p - 1.0)
