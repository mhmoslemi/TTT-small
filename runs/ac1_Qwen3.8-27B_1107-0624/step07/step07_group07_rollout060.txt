"""
Step sequence optimizer using LP-based alternating optimization combined
with gradient-based refinement. Key innovation: uses CVXPY LP to find
optimal direction at each step, combined with log-sum-exp L-BFGS for
polishing. Multi-length exploration and adaptive active-set methods.
"""
import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import linalg as sparse_linalg


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_weights(c, tol_frac):
    """Compute weights for active lags using adaptive tolerance."""
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    # Try multiple tolerance levels and pick the one that gives "balanced" weights
    best_w = None
    for frac in [tol_frac, tol_frac * 0.5, tol_frac * 2.0]:
        tol = frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        elif active.sum() > 0.1 * len(c):
            # Too many active; use top fraction
            k = max(1, int(0.05 * len(c)))
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:k]
            active[top_idx] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        if best_w is None:
            best_w = w
    return best_w


def _gradient_active(a, c, n, w):
    """Compute gradient of the active-set approximation."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 150,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False

def _lp_direction(a, deadline_step=None):
    """
    Solve LP: minimize t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns optimal b or None if LP fails.
    Only for n <= 2048 for efficiency.
    """
    n = len(a)
    if n > 2048:
        return None

    # Build convolution matrix T (2n-1) x n
    # T[k, j] = a[k-j] if 0 <= k-j < n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    b_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        T @ b_var <= t_var,
        b_var >= 0,
        cp.sum(b_var) == 1
    ]
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate'] and b_var.value is not None:
            b_val = np.maximum(b_var.value, 0)
            s = b_val.sum()
            if s > 1e-10:
                b_val = b_val / s
                return b_val
    except Exception:
        pass
    return None

        if b_opt is None:

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()

def _fourier_perturb(a, rng, n_modes=8, max_amp=0.05):
    """Add structured Fourier perturbations to escape local minima."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = max_amp * rng.uniform(0.2, 1.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _gauss_perturb(a, rng, scale=0.03):
    """Gaussian perturbation in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    # === Phase 4: Multi-restart with perturbations (350s) ===
    print(f"\n=== Phase 4: Multi-restart ===")
    t_phase4_end = t_start + 1200
    if best_seq is not None:
        n_restart = len(best_seq)
        restart_budget = max(20, (t_phase4_end - time.monotonic()) / 8)

        configs = [
            ("gauss_002", "gauss", 0.02),
            ("gauss_005", "gauss", 0.05),
            ("gauss_010", "gauss", 0.10),
            ("gauss_020", "gauss", 0.20),
            ("fourier_1", "fourier", 0.03),
            ("fourier_2", "fourier", 0.06),
            ("fourier_3", "fourier", 0.10),
            ("fourier_4", "fourier", 0.15),
        ]

        for name, mode, scale in configs:
            if time.monotonic() > t_phase4_end:
                break
            if mode == "gauss":
                a_start = _gauss_perturb(best_seq, rng, scale)
            else:
                a_start = _fourier_perturb(best_seq, rng, n_modes=6, max_amp=scale)

            sc_start = _score(a_start)
            # Quick optimization
            a_opt = _lbfgs_continuation(a_start, time.monotonic() + restart_budget,
                                        reserve=10)
            try_update(a_opt, f"restart_{name}")

            if time.monotonic() < t_phase4_end - 30 and restart_budget > 30:
                a_ps, _ = _peak_shaving_exact(a_opt, time.monotonic() + restart_budget / 3)
                try_update(a_ps, f"restart_{name}_ps")

    # === Phase 5: Multi-length exploration (200s) ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    t_phase5_end = t_start + 1400
    if best_seq is not None:
        current_n = len(best_seq)
        target_lengths = []
        for tn in [2048, 4096, 8192, 12288, 16384]:
            if tn != current_n and tn <= 16384:
                target_lengths.append(tn)

        # Sort by proximity to current
        target_lengths.sort(key=lambda x: abs(x - current_n))

        for target_n in target_lengths[:3]:
            if time.monotonic() > t_phase5_end:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            try_update(a_interp, f"interp_n{target_n}")

            # Optimize at this length
            time_for_this = min(60, max(15, (t_phase5_end - time.monotonic()) / 2))
            a_opt = _lbfgs_continuation(a_interp, time.monotonic() + time_for_this,
                                        reserve=5)
            try_update(a_opt, f"opt_n{target_n}")

            if time.monotonic() < t_phase5_end - 10:
                a_ps, _ = _peak_shaving_exact(a_opt, time.monotonic() + 20)
                try_update(a_ps, f"ps_n{target_n}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None:
        # Cycle: peak shave -> L-BFGS -> multi-scale -> coordinate
        for cycle in range(4):
            if time.monotonic() > deadline:
                break

            a_f1, _ = _peak_shaving_exact(best_seq, min(deadline - 5, time.monotonic() + 40))
            try_update(a_f1, f"polish_ps_{cycle}")

            if time.monotonic() < deadline - 10:
                a_f2 = _lbfgs_continuation(best_seq, min(deadline - 5, time.monotonic() + 50),
                                           reserve=5)
                try_update(a_f2, f"polish_lbfgs_{cycle}")

            if time.monotonic() < deadline - 10:
                a_f3, _ = _multi_scale_active_optimize(best_seq,
                                                       min(deadline - 5, time.monotonic() + 30))
                try_update(a_f3, f"polish_ms_{cycle}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(2048, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
