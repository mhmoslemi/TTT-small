So in all cases: $\frac{\partial c_k}{\partial a_m} = 2a_{k-m}$ if $0 \leq k-m < n$, else 0.

This means the gradient of $\sum_{k \in S} c_k$ with respect to $a_m$ is:
$\sum_{k \in S} 2a_{k-m} \cdot \mathbb{1}[0 \leq k-m < n]$

This can be computed efficiently using FFT if $S$ is a large set.

"""
Multi-scale top-k peak reduction optimizer using Adam with FFT-based gradients.
Exploits coarse-to-fine resolution and adaptive peak weighting to minimize
the normalized autocorrelation peak of non-negative sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto the probability simplex {x: x_i >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _compute_topk_gradient(a, c, k, n):
    """
    Compute gradient of sum of top-k autocorrelation values.
    
    grad[m] = sum_{k in S} 2 * a[k-m] for valid (k-m) in [0, n-1]
    where S is the set of top-k indices in c.
    """
    # Get top-k indices
    if k >= len(c):
        S = np.ones(len(c), dtype=np.float64)
    else:
        topk_idx = np.argpartition(c, -k)[-k:]
        S = np.zeros(len(c), dtype=np.float64)
        S[topk_idx] = 1.0 / k
    
    # grad[m] = 2 * sum_j S[j] * a[j-m] for valid j-m
    # = 2 * (a_reversed * S)[m + n - 1] ... let me think
    # sum_j S[j] * a[j-m]: this is the cross-correlation of a and S
    # = (a_rev * S)[n-1 + m] where a_rev = a[::-1]
    # Because (a_rev * S)[p] = sum_i a_rev[i] * S[p-i] = sum_i a[n-1-i] * S[p-i]
    # Let j = n-1-i, so i = n-1-j:
    # = sum_j a[j] * S[p-n+1+j]
    # So if p = n-1+m: = sum_j a[j] * S[m+j]
    # But we want sum_j S[j] * a[j-m] = sum_j a[j-m] * S[j]
    # Let i = j-m: = sum_i a[i] * S[i+m]
    # = (a_rev * S)[n-1 + (2n-1-m-1)] ... this is getting confusing.
    # Let me use a different approach.
    # sum_j S[j] * a[j-m] = (flip(S) * a)[m + len(S) - 1]
    # where flip(S)[i] = S[len(S)-1-i]
    # (flip(S) * a)[p] = sum_i flip(S)[i] * a[p-i] = sum_i S[len(S)-1-i] * a[p-i]
    # Let j = len(S)-1-i, i = len(S)-1-j:
    # = sum_j S[j] * a[p-len(S)+1+j]
    # If p = len(S)-1+m: = sum_j S[j] * a[m+j]
    # Hmm, that gives sum_j S[j] * a[m+j], not sum_j S[j] * a[j-m].
    
    # Let me just compute it directly using FFT.
    # We want: g[m] = sum_j S[j] * a[j-m] for m = 0, ..., n-1
    # Note: j-m must be in [0, n-1], so j must be in [m, m+n-1]
    # Also j must be in [0, 2n-2] (the length of S)
    # So: g[m] = sum_{j=max(m,0)}^{min(m+n-1, 2n-2)} S[j] * a[j-m]
    
    # This is the "valid" cross-correlation of a and S.
    # g = cross_correlate(a, S) [valid mode]
    # In terms of convolution: g[m] = sum_j a[j] * S[m+j]
    # Wait no, that's different.
    # g[m] = sum_j S[j] * a[j-m]
    # Let i = j-m: g[m] = sum_i S[i+m] * a[i]
    # = sum_i a[i] * S[i+m]
    # This is a "generalized correlation": for each m, dot product of a with S shifted by m.
    # = (a * flip(S))[m + len(S) - 1] ... let me verify
    # (a * flip(S))[p] = sum_i a[i] * flip(S)[p-i] = sum_i a[i] * S[len(S)-1-(p-i)]
    # = sum_i a[i] * S[len(S)-1-p+i]
    # If p = len(S)-1-m: = sum_i a[i] * S[m+i]
    # Yes! So g[m] = (a * flip(S))[len(S)-1-m]
    # But we want g for m = 0, ..., n-1, and S has length 2n-1.
    # len(S)-1 = 2n-2
    # g[m] = (a * flip(S))[2n-2-m]
    
    S_flip = S[::-1]
    conv = _fft_convolve(a, S_flip)  # length n + (2n-1) - 1 = 3n-1
    # g[m] = conv[2n-2-m] for m = 0, ..., n-1

    # [::-1] reverses: 2n-2, 2n-3, ..., n-1
    # So g[0] = conv[2n-2], g[1] = conv[2n-3], ..., g[n-1] = conv[n-1]. Correct!
    
    return 2.0 * g


def _compute_soft_gradient(a, c, beta, n):
    """Compute gradient of log-sum-exp approximation of max."""
    c_max = c.max()
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    w = np.exp(shifted)
    w = w / w.sum()
    
    S_flip = w[::-1]
    conv = _fft_convolve(a, S_flip)
    g = conv[n-1:2*n-1][::-1]
    return 2.0 * g


def _adam_update(z, grad, m, v, t, lr, beta1, beta2, eps):
    """Adam update step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v


def _optimize_scale(a_init, deadline, min_time_remaining, n,

    """
    Optimize a sequence at a given scale using Adam with top-k/soft gradient.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    if n_iters is None:

        # Each iteration is about 3 FFTs of size ~4n, so ~3*4n*log(4n) ops
        # For n=8192, that's ~3*32768*15 ≈ 1.5M ops per iter
        # At ~100M ops/sec, that's ~15ms per iter
        time_per_iter = 0.015 * max(1, n / 4096.0)
        n_iters = int((deadline - min_time_remaining - time.monotonic()) / time_per_iter)
        n_iters = max(100, min(n_iters, 500000))
    
    if beta_start is None:
        c = _fft_convolve(a, a)
        c_max = c.max()
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
        beta_start = max(gap * 0.1, c_max * 0.01)
    
    if beta_end is None:
        beta_end = beta_start * 0.001
    
    if k_end is None:
        k_end = max(k_start, int(n * 0.1))
    
    # Use log-space parameterization
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    patience = 0
    max_patience = 500
    
    for it in range(n_iters):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Current a from softmax
        z_max = z.max()
        e = np.exp(z - z_max)
        a = e / e.sum()
        
        # Compute autocorrelation
        c = _fft_convolve(a, a)
        
        # Compute score
        sc = float(2.0 * n * c.max())
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Adaptive k: start small, increase over time
        progress = it / max(1, n_iters - 1)
        k_cur = int(k_start + (k_end - k_start) * progress)
        k_cur = max(1, k_cur)
        
        # Adaptive beta: decrease over time
        beta_cur = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient (soft top-k)
        if k_cur >= len(c):
            grad = _compute_soft_gradient(a, c, beta_cur, n)
        else:
            # Hybrid: use soft weights on top-k
            topk_idx = np.argpartition(c, -k_cur)[-k_cur:]
            topk_vals = c[topk_idx]
            topk_max = topk_vals.max()
            soft_w = np.exp((topk_vals - topk_max) / beta_cur)
            soft_w = soft_w / soft_w.sum()
            
            # Full weight vector
            w = np.zeros(len(c))
            w[topk_idx] = soft_w
            
            S_flip = w[::-1]
            conv = _fft_convolve(a, S_flip)
            grad = 2.0 * conv[n-1:2*n-1][::-1]
        
        # Project gradient to tangent space (sum = 0)
        grad = grad - grad.mean()
        
        # Adam update
        lr = lr_init / (1.0 + 0.01 * it)

        # Keep z bounded
        z = np.clip(z, -50, 50)
        z = z - z.mean()
    
    # Final check
    z_max = z.max()
    e = np.exp(z - z_max)
    a_final = e / e.sum()
    sc_final = float(2.0 * n * _fft_convolve(a_final, a_final).max())
    if sc_final < best_sc:

def _upscale(a, n_target):
    """Upscale a sequence to a new length using linear interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n_target) / n_target
    else:
        a_new = a_new / s
    return a_new


def _downscale(a, n_target):
    """Downscale a sequence to a new length."""
    return _upscale(a, n_target)


def _newton_polish(a_init, deadline, max_iter=50):
    """Newton's method on the active set for final polish."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: indices where c is within 1% of max
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Gradient of sum of active c_k
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        S_flip = w[::-1]
        conv = _fft_convolve(a, S_flip)
        grad = 2.0 * conv[n-1:2*n-1][::-1]
        grad = grad - grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Simple gradient step with line search (Newton would need Hessian)
        alpha = 0.01
        f_old = float(c[active].sum())
        for ls in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old:
                a = a_try
                break
            alpha *= 0.5
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _shrink_and_redistribute(a, deadline):
    """Try zeroing small elements and redistributing their mass."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        mass = a_try[a_sorted_idx[:n_zero]].sum()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        # Redistribute mass to remaining elements proportionally
        a_try[a_sorted_idx[n_zero:]] += mass / (n - n_zero)
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc

"""
Multi-scale top-k peak reduction optimizer using Adam with FFT-based gradients.
Uses coarse-to-fine resolution, adaptive peak weighting, Newton polish,
and diversified restarts to minimize the normalized autocorrelation peak.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _autocorr(a):
    return _fft_convolve(a, a)


def _compute_gradient_from_weights(a, w, n):
    """Compute gradient given weight vector w over autocorrelation lags.
    grad[m] = 2 * sum_j w[j] * a[j-m] for valid indices.
    Uses: grad[m] = 2 * (a * flip(w))[n-1 : 2n-1][::-1]
    """
    w_flip = w[::-1]
    conv = _fft_convolve(a, w_flip)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_step(z, grad, m, v, t, lr, b1, b2, eps):
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

    n = len(a)
    best_a = a.copy()

    idx = np.argsort(a)

        if time.monotonic() > deadline:

        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        mass = a_try[idx[:n_zero]].sum()
        a_try[idx[:n_zero]] = 0.0
        s_rem = a_try[idx[n_zero:]].sum()
        if s_rem < 1e-10:
            continue
        a_try[idx[n_zero:]] = a_try[idx[n_zero:]] + mass * a_try[idx[n_zero:]] / s_rem
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _upscale(a, n_target):

    x_new = np.linspace(0, 1, n_target)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a_new / s

def _perturb_block_swap(a, rng, n_blocks=3):
    n = len(a)
    p = a.copy()
    block_size = max(1, n // (2 * n_blocks * 2))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)

            p[i:i + block_size] = p[j:j + block_size]
            p[j:j + block_size] = tmp
    s = p.sum()
    if s < 1e-15:
        return a
    return p / s

    total_scale_weight = 0.0

    for ns in scales:
        w = (ns / max(scales)) ** 0.8
        scale_weights.append(w)

        remaining = deadline - time.monotonic()

        time_alloc = max(5.0, min(time_alloc, remaining - 20))

            if sc_n < sc_opt:
