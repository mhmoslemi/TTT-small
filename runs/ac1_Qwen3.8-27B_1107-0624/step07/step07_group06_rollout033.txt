- We want to find a new a' (close to a) such that b'_{k_1} = b'_{k_2} = ... = b'_{k_m} (all equal to the minimum possible common value)
- This is a system of m equations in n unknowns (with the constraint sum a' = 1)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "truncated_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_grad(a, c, n, K=10):
    """Compute gradient of mean of top-K convolution values."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    # Get top K indices
    K_actual = min(K, len(c))
    top_idx = np.argsort(c)[-K_actual:]
    S_indicator = np.zeros(len(c))
    S_indicator[top_idx] = 1.0
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, S_indicator)
    grad = (2.0 / K_actual) * conv[n - 1:2 * n - 1]
    return grad


def _run_lbfgs_relative(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with relative beta scheduling (beta as fraction of current c_max)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    # Relative beta schedule: fraction of c_max at each stage
    fractions = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, frac in enumerate(fractions):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        # Recompute c_max for adaptive beta
        c_now = _full_convolve(current_a, current_a)
        c_max_now = float(c_now.max())
        beta = max(c_max_now * frac, 1e-12)

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(6000, max(100, int(remaining / (est_time_per_iter * 8))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 120, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Tried but no improvement; continue to next beta
                pass
        except Exception:
            pass

    return current_a, current_sc


def _active_set_gradient_descent(a_init, deadline, K=10, max_iter=50000):
    """
    Gradient descent on the mean of top-K convolution values.
    Uses multiplicative updates in log-space for stability.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    alpha = 0.05
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive K: increase as peaks get closer
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > K + 1:
            gap_top = c_sorted[0] - c_sorted[K]
            gap_ref = c_sorted[0] - c_sorted[min(2 * K, len(c_sorted) - 1)]
            if gap_ref > 1e-15 and gap_top / gap_ref < 0.1:
                K = min(K * 2, 40)
            elif gap_ref > 1e-15 and gap_top / gap_ref > 0.5:
                K = max(K // 2, 2)

        grad = _active_set_grad(a, c, n, K)

        # Convert to log-space gradient: grad_z = a * (grad - mean(grad * a))
        G = float(np.dot(a, grad))
        grad_z = a * (grad - G)

        gnorm = float(np.linalg.norm(grad_z))
        if gnorm < 1e-15:
            break

        # Adaptive step size with backtracking
        f_old = c_max
        a_new = a
        alpha_try = alpha
        for ls in range(50):
            z_try = z - alpha_try * grad_z
            a_try = _softmax(z_try)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * alpha_try * gnorm:
                a_new = a_try
                z = z_try
                break
            alpha_try *= 0.5
        else:
            if alpha_try < 1e-15:
                break
            a_new = a
            z = z

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            patience = 0
            alpha = min(alpha * 1.05, 0.1)
        else:
            patience += 1
            alpha *= 0.9
            if patience > 200:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                patience = 0
                alpha = 0.05

    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _peak_equalization(a_init, deadline, n_steps=2000):
    """
    Iteratively equalize the top peaks by reducing contributions
    from the most loaded coordinates.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Top K peaks
        K = min(20, len(c))
        top_idx = np.argsort(c)[-K:]
        active_level = c[top_idx].mean()

        # For each coordinate i, compute its "load": how much it contributes to active peaks
        # load[i] = sum_{k in top_idx} a[k-i] (with bounds)
        S_indicator = np.zeros(len(c))
        S_indicator[top_idx] = 1.0
        a_rev = a[::-1]
        load_conv = _full_convolve(a_rev, S_indicator)
        load = load_conv[n - 1:2 * n - 1]

        # Multiplicative update: reduce coordinates with high load
        eta = 0.01 / (1.0 + 0.01 * step)
        load_max = load.max()
        if load_max < 1e-15:
            break
        factor = np.exp(-eta * load / load_max)
        a = a * factor
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=20000, lr_init=0.01):
    """Subgradient descent on the max using top-K active set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    K = 5

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 2:
            ratio = (c_sorted[0] - c_sorted[K]) / max(c_sorted[0] - c_sorted[min(2*K, len(c_sorted)-1)], 1e-15)
            if ratio < 0.1:
                K = min(K * 2, 30)
            elif ratio > 0.5:
                K = max(K // 2, 2)

        grad = _active_set_grad(a, c, n, K)
        G = float(np.dot(a, grad))
        grad_z = a * (grad - G)
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad_z
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _interpolate_sequence(a_src, n_target):
    """Interpolate a sequence to a different length."""
    x_old = np.linspace(0, 1, len(a_src))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_src)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements to reduce effective n."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.01, 0.02, 0.05, 0.08, 0.12, 0.15, 0.20, 0.30]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            print(f"    Shrinkage {frac*100:.0f}%: score={sc_try:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: refined L-BFGS with relative beta + active-set
    gradient descent + peak equalization + subgradient polish + multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")
            return True
        return False

    # Load parent
    parent = None
    n_parent = 8192
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent)
    else:
        parent = _make_profile(8192, "uniform")
        print(f"No parent, using uniform n=8192")
        update_best(parent, _score(parent))

    # === Main optimization loop over n values ===
    # Prioritize the parent n, then neighbors
    n_values = [n_parent]
    for n_try in [4096, 6144, 12288, 16384]:
        if n_try != n_parent:
            n_values.append(n_try)
    n_values = sorted(set(n_values))

    # Allocate time budget: give more time to the parent n
    total_time = deadline - t_start
    # Parent gets 50%, others split the rest
    n_parent_time = total_time * 0.45
    n_other_time_each = (total_time * 0.5) / max(1, len(n_values) - 1)
    # Reserve 60s for final polish
    n_parent_time = min(n_parent_time, total_time - 60)
    n_other_time_each = min(n_other_time_each, (total_time - 60 - n_parent_time) / max(1, len(n_values) - 1))

    for practice_n in n_values:
        t_phase_start = time.monotonic()
        if practice_n == n_parent:
            phase_deadline = t_phase_start + n_parent_time
        else:
            phase_deadline = t_phase_start + max(n_other_time_each, 30)

        if time.monotonic() > deadline - 80:
            break

        print(f"\n=== Optimizing n={practice_n} (budget={phase_deadline - t_start - t_phase_start:.0f}s) ===")

        # Initialize
        if practice_n == n_parent and parent is not None:
            a_work = parent.copy()
        else:
            a_work = _interpolate_sequence(best_seq, practice_n)

        # Phase A: L-BFGS with relative beta continues
        t_A = time.monotonic() + (phase_deadline - t_phase_start) * 0.35
        if time.monotonic() < min(phase_deadline, deadline - 60):
            a_next, sc_A = _run_lbfgs_relative(a_work, min(phase_deadline, deadline - 60), min_time_remaining=30)
            if update_best(a_next, sc_A):
                a_work = a_next

        # Phase B: Active-set gradient descent
        t_B = time.monotonic() + (phase_deadline - t_phase_start) * 0.25
        if time.monotonic() < min(phase_deadline, deadline - 50):
            a_next, sc_B = _active_set_gradient_descent(a_work, min(phase_deadline, deadline - 50), K=10, max_iter=30000)
            if update_best(a_next, sc_B):
                a_work = a_next

        # Phase C: Peak equalization
        if time.monotonic() < min(phase_deadline, deadline - 40):
            a_next, sc_C = _peak_equalization(a_work, min(phase_deadline, deadline - 40), n_steps=3000)
            if update_best(a_next, sc_C):
                a_work = a_next
            # Short L-BFGS after peak equalization
            if time.monotonic() < min(phase_deadline, deadline - 30):
                a_next2, sc_C2 = _run_lbfgs_relative(a_work, min(phase_deadline, deadline - 30), min_time_remaining=15)
                if update_best(a_next2, sc_C2):
                    a_work = a_next2

        # Phase D: Subgradient polish
        if time.monotonic() < min(phase_deadline, deadline - 25):
            a_next, sc_D = _subgrad_polish(a_work, min(phase_deadline, deadline - 25), n_max_steps=20000, lr_init=0.008)
            if update_best(a_next, sc_D):
                a_work = a_next

        # Phase E: Shrinkage + re-optimize
        if time.monotonic() < min(phase_deadline, deadline - 20):
            a_next, sc_E = _shrinkage(a_work, min(phase_deadline, deadline - 20))
            if update_best(a_next, sc_E):
                a_work = a_next
            if time.monotonic() < min(phase_deadline, deadline - 15):
                a_next2, sc_E2 = _active_set_gradient_descent(a_work, min(phase_deadline, deadline - 15), K=5, max_iter=10000)
                if update_best(a_next2, sc_E2):
                    a_work = a_next2

        print(f"  Phase done for n={practice_n}, best={best_score:.10f}")

    # === Final polish on the best found ===
    print(f"\n=== Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # One more round of active-set GD
        a_fin, sc_fin = _active_set_gradient_descent(best_seq, deadline - 3, K=5, max_iter=10000)
        update_best(a_fin, sc_fin)

        # Peak equalization
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _peak_equalization(best_seq, deadline - 2, n_steps=2000)
            update_best(a_fin2, sc_fin2)

        # Ultra-low beta L-BFGS
        if time.monotonic() < deadline - 2:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n_fin = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = max(mb * 1e-7, 1e-14)
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(3000, max(50, int(remaining / 0.005)))

                def fun_final(z_local):
                    return _smoobj(z_local, beta_final, n_fin)

                def jac_final(z_local):
                    return _smograd(z_local, beta_final, n_fin)

                try:
                    result = minimize(
                        fun_final, z, jac=jac_final, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3))
                except Exception:
                    pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        sc_eval = evaluate_sequence([float(x) for x in final])
        if np.isfinite(sc_eval):
            sc_final = sc_eval
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
