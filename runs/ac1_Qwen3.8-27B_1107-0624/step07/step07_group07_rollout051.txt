    """Solve LP: max sum(b) s.t. (a*b)(k) <= M for k in active_indices, b >= 0.
    If too many constraints, subsample to max_constraints.
    """
    n = len(a)
    sel = active_indices
    if len(sel) > max_constraints:
        # Subsample: keep the most "important" constraints
        sel = sel[np.linspace(0, len(sel)-1, max_constraints, dtype=int)]
    
    # Build constraint matrix for selected constraints
    A_rows = []
    A_cols = []
    A_vals = []
    for idx, k in enumerate(sel):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_rows.append(idx)
            A_cols.append(j)
            A_vals.append(a[k - j])
    
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(len(sel), n))
    b_vec = np.full(len(sel), M)
    c_vec = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n,
                  method='highs', options={'maxiter': 10000})
    
    if res.success:
        return res.x
    return None

    threshold = sorted_c[n_active - 1]
    tol = c_max - threshold
    active = np.where(c >= c_max - tol)[0]
    return active

def _solve_lp(a, M, active_idx, deadline, max_constraints=2000):
    """Solve LP: max sum(b) s.t. (a*b)(k) <= M for k in active_idx, b >= 0."""
    n = len(a)
    sel = active_idx
    if len(sel) > max_constraints:
        # Subsample while keeping the most important (largest c values)
        sel = sel[np.linspace(0, len(sel) - 1, max_constraints, dtype=int)]
    
    # Build sparse constraint matrix
    A_rows = []
    A_cols = []
    A_vals = []
    for idx, k in enumerate(sel):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        js = range(j_min, j_max + 1)
        A_rows.extend([idx] * len(js))
        A_cols.extend(list(js))
        A_vals.extend(a[k - j] for j in js)
    
    if not A_rows:
        return None
    
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(len(sel), n))
    b_vec = np.full(len(sel), M)
    c_vec = -np.ones(n)
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _lp_iteration(a, deadline, max_iters=200, step_init=0.3):
    """Run LP-based iterations to reduce the peak."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    best_a = a.copy()
    best_score = _score(a)
    t = step_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        # Get active set
        active = _get_active_set(c, n_frac=0.1, min_count=20)
        
        # Solve LP
        t_lp = time.monotonic()
        b = _solve_lp(a, M, active, deadline, max_constraints=2000)
        lp_time = time.monotonic() - t_lp
        
        if b is None:
            break
        
        b = np.maximum(b, 0)
        sb = b.sum()
        if sb < 1e-10:
            break
        b = b / sb  # normalize b to sum=1
        
        # Line search for step size
        best_t = 0
        best_a_try = a
        best_sc_try = _score(a)
        
        for t_try in [t, t*0.5, t*0.25, t*0.1, t*0.05]:
            a_try = (1 - t_try) * a + t_try * b
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_t = t_try
        
        if best_t > 0:
            a = best_a_try
            if best_sc_try < best_score:
                best_score = best_sc_try
                best_a = a.copy()
            t = min(t * 1.05, 0.8)
        else:
            t *= 0.7
            if t < 0.001:
                break
        
        # Periodic re-normalization
        if it % 10 == 0:
            s = a.sum()
            if s > 0:
                a = a / s
    
    return best_a, best_score

def _lbfgs_refine(a_init, deadline, min_time=10):
    """Refine using L-BFGS with log-sum-exp."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - min_time:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        def fun(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = c.max()
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def jac(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = c.max()
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)
        
        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': 5000, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxcor': 100})
            a_out = _softmax(result.x)
            sc = _score(a_out)
            if sc < current_sc - 1e-15:
                current_a = a_out
                current_sc = sc
                z0 = np.log(np.maximum(a_out, 1e-15))
                z0 = z0 - z0.mean()
        except Exception:
            pass
    
    return current_a, current_sc

def _interpolate(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    t_old = np.linspace(0, n - 1, n)
    t_new = np.linspace(0, target_n - 1, target_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new

def _gradient_descent_direct(a_init, deadline, max_iters=10000):
    """Direct gradient descent in the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Active set
        n_act = max(5, int(0.02 * len(c)))
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:n_act]
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W * a)  # gradient of 2n*(w·c) for sum(a)=1
        
        # Project gradient onto tangent space of simplex
        grad = grad - a * np.dot(grad, a)
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        
        a_new = a - lr * grad / g_norm
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
    
    return best_a, best_sc
