```python
from scipy.sparse import lil_matrix

def conv_matrix(a, n):
    """Build the matrix A such that A @ g = conv(a, g)."""
    out_len = 2 * n - 1
    A = lil_matrix((out_len, n), dtype=np.float64)
    for k in range(out_len):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A.tocsr()
```

For n=8192, building this matrix takes O(n^2) = 67M operations, which is fast. The LP solve should take a few seconds.

```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

# ... (smoobj, smograd, etc. from previous code)

def _build_conv_matrix(a, n):
    """Build sparse matrix A where A @ g = conv(a, g)."""
    out_len = 2 * n - 1
    A = lil_matrix((out_len, n), dtype=np.float64)
    for k in range(out_len):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def _lp_refine_step(a, n, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    A = _build_conv_matrix(a, n)
    c_vec = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_vec = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
```

```python
    c = _full_convolve(a, a)
```

"""
Multi-phase optimizer with LP-based (Frank-Wolfe) refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Algorithm:
1. L-BFGS with adaptive log-sum-exp smoothing for initial optimization
2. LP-based conditional gradient (Frank-Wolfe) for near-exact refinement
3. Nesterov-accelerated active-set gradient descent for final polish
4. Shrinkage exploration for sparser solutions

The LP step solves: max sum(g) s.t. conv(a,g)[k] <= M for active k, g >= 0
This gives the exact optimal direction in the linearized problem.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
