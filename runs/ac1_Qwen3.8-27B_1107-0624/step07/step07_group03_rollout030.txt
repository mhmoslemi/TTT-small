                    'maxiter': max_iter,

    elif kind == "sine":

        a = np.maximum(a, 1e-10)
        return a / a.sum()

        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    elif kind == "asym_bump":
        t = np.arange(n)
        a = 1.0 + 0.6 * np.exp(-((t - 0.3 * n) / (0.15 * n))**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.25 * n) / (0.1 * n))**2) + 0.3 * np.exp(-((t - 0.75 * n) / (0.1 * n))**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)**0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:

    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.005 * max(1, n / 2048.0)
        max_iter = min(10000, max(50, int(remaining / (est_time * 8))))
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient with active-set tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.88
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _variance_aware_optimize(a_init, deadline, n_beta_steps=8):
    """First flatten variance, then minimize max."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    c_cur = _full_convolve(current_a, current_a)
    c_mean = float(c_cur.mean())
    c_var = float(np.var(c_cur))
    c_max = float(c_cur.max())

    beta_var = c_var / max(c_max, 1e-15) + c_max * 0.1
    for i in range(n_beta_steps):
        remaining = deadline - 30 - time.monotonic()
        if remaining < 5:
            break
        beta = beta_var * (0.5 ** i)
        max_iter = min(5000, max(50, int(remaining / 0.05)))
        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine_odd":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * t / n + np.pi / 3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t / n - 0.7)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * t
        return a / a.sum()
    elif kind == "bump_left":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.35 * n) / (0.2 * n)) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump_right":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.exp(-((t - 0.65 * n) / (0.2 * n)) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_freq":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t / n) + 0.2 * np.cos(4.0 * np.pi * t / n + 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skew_gauss":
        t = np.linspace(0, 1, n)
        base = 1.0 + 0.8 * (t - 0.5)
        bump = 0.4 * np.exp(-((t - 0.4) / 0.15) ** 2)
        a = base + bump
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=3, amp=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 3)))
        a_ = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + a_ * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Load parent or create diverse initializations
    2. Multi-scale: optimize at smaller n, upscale to target n
    3. L-BFGS with log-sum-exp smoothing
    4. Active-set Newton refinement
    5. Subgradient polish
    6. Diverse restarts from best
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Optimize parent at its n
    print(f"\n[Phase 1] L-BFGS on parent n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    remaining_budget = deadline - 400 - time.monotonic()
    if remaining_budget > 20:
        a_opt1 = _run_lbfgs(parent, t_start + 400 + remaining_budget, min_time_remaining=200)
        update_best(a_opt1, _score(a_opt1), f"lbfgs_n{n_parent}")

    # Phase 2: Active-set Newton on best so far
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as_dl = min(deadline - 200, time.monotonic() + 120)
        a_as, sc_as = _active_set_newton(best_seq, as_dl, max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton_1")
        if time.monotonic() < deadline - 180:
            as_dl2 = min(deadline - 180, time.monotonic() + 80)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_dl2, max_iter=150, tol_active=0.0005)
            update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: Try different n values
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_targets = [2048, 4096, 8192, 16384]
    for n_try in n_targets:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue
        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")
        remaining = deadline - 100 - time.monotonic()
        if remaining < 15:
            break
        max_time = min(remaining, 80)
        a_opt_n = _run_lbfgs(a_init, time.monotonic() + max_time, min_time_remaining=5)
        sc_opt_n = _score(a_opt_n)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")
        if sc_opt_n < best_score + 0.01 and time.monotonic() < deadline - 60:
            a_opt_n2, sc_n2 = _active_set_newton(a_opt_n, min(deadline - 50, time.monotonic() + 40),
                                                  max_iter=80, tol_active=0.001)
            update_best(a_opt_n2, sc_n2, f"n_{n_try}_as")

    # Phase 4: Diverse initializations at best n
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        profiles = ["sine_odd", "cos_asym", "ramp", "bump_left", "bump_right",
                    "two_freq", "skew_gauss", "tapered"]
        for prof in profiles:
            if time.monotonic() > deadline - 30:
                break
            a_p = _make_profile(n_best, prof)
            remaining = deadline - 25 - time.monotonic()
            if remaining < 10:
                break
            max_t = min(remaining, 25)
            a_opt_p = _run_lbfgs(a_p, time.monotonic() + max_t, min_time_remaining=3)
            sc_p = _score(a_opt_p)
            update_best(a_opt_p, sc_p, f"profile_{prof}")

    # Phase 5: Perturbation restarts from best
    print(f"\n[Phase 5] Perturbation restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        perturbations = [
            ("small", 0.02, 0),
            ("med", 0.05, 0),
            ("large", 0.10, 0),
            ("fourier3", 0, 3),
            ("fourier5", 0, 5),
        ]
        for label, sigma, n_modes in perturbations:
            if time.monotonic() > deadline - 20:
                break
            if n_modes > 0:
                p_start = _fourier_perturb(best_seq, rng, n_modes=n_modes)
            else:
                p_start = _perturb(best_seq, sigma, rng)
            remaining = deadline - 15 - time.monotonic()
            if remaining < 5:
                break
            max_t = min(remaining, 20)
            a_r = _run_lbfgs(p_start, time.monotonic() + max_t, min_time_remaining=3)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"pert_{label}")
            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 12:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 8, time.monotonic() + 20),
                                                  max_iter=60, tol_active=0.001)
                update_best(a_r2, sc_r2, f"pert_{label}_as")

    # Phase 6: Subgradient polish
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        sg_dl = min(deadline - 5, time.monotonic() + 60)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_dl, n_max_steps=3000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # Phase 7: Final active-set Newton with tight tolerance
    print(f"\n[Phase 7] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        fin_dl = min(deadline - 2, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_newton(best_seq, fin_dl, max_iter=300, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_newton(best_seq, deadline - 2, max_iter=200, tol_active=0.0001)
            update_best(a_fin2, sc_fin2, "final_as_tight")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
