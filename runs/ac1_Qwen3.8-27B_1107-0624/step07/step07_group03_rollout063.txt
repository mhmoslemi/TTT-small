- Set `a_i = 0`, renormalize, compute the new `max_b`.
- The importance is `max_b_new - max_b_old`.

"""
Multi-phase optimizer using mirror descent, greedy shrinkage, and active-set Newton.
Novel contributions: entropy mirror descent with adaptive step size, 
greedy coordinate removal, and two-scale refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_a_A(a, w, n):
    """Compute g[i] = sum_j w[j] * a[j+i-n+1] via FFT (cross-correlation)."""
    # g[i] = sum_k w[k] * a[k-i] where valid indices are 0<=k-i<n
    # = cross-correlation of w and a
    flip_a = a[::-1]
    out_len = n + n - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(flip_a, N)
    result = np.fft.irfft(fw * fa, N)[:out_len]
    return result[n-1:2*n-1]

def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _mirror_descent(a_init, deadline, max_iter=20000, verbose=True):
    """Entropy mirror descent with adaptive step size and temperature."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.5  # initial step size
    beta = 0.05  # initial temperature
    patience = 0
    max_patience = 500
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Softmax weights on autocorrelation
        w = np.exp((c - c_max) / beta)
        w_sum = w.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        
        # Subgradient
        g = 2.0 * _corr_a_A(a, w, n)
        
        # Project to tangent space
        g_t = g - g.mean()
        g_norm = float(np.abs(g_t).max())
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        eta = 0.5 / g_norm
        
        # Mirror descent update
        z = np.log(np.maximum(a, 1e-30))
        z = z - eta * g_t  # In log space, the update is multiplicative
        z = z - z.max()
        a_new = np.exp(z)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300 or not np.isfinite(a_new_sum):
            eta *= 0.5
            continue
        a_new /= a_new_sum
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Slowly decrease temperature
            if it > 0 and it % 200 == 0:
                beta = max(0.005, beta * 0.97)
        else:
            patience += 1
            eta *= 0.7
            if patience > max_patience:
                break
        
        if verbose and it % 5000 == 0:
            print(f"    MD iter {it}: score={best_sc:.10f}, eta={eta:.4f}, beta={beta:.4f}")
    
    return best_a, best_sc

def _greedy_shrink(a_init, deadline, max_rounds=100):
    """Greedy coordinate removal: remove the coordinate whose removal least hurts."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        n_current = len(a)
        if n_current < 8:
            break
        
        # Find the smallest non-zero element
        # Try removing it
        min_idx = int(np.argmin(a))
        if a[min_idx] < 1e-10:
            break
        
        # Remove this element
        a_try = np.delete(a, min_idx)
        sc_try = _score(a_try)
        
        if sc_try < best_sc + 0.001:  # Allow small increase
            a = a_try
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a.copy()
        else:
            break
    
    return best_a, best_sc

def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    """Damped Newton on active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:2*n]
            active[top_idx] = True
            n_active = 2 * n
        
        A_vec = active.astype(np.float64)
        m = n_active
        
        # Gradient
        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break
        
        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8
        
        def hvp(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v
        
        # CG solve
        x = np.zeros(n)
        r = g_t - hvp(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(200):
            Ap = hvp(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        
        u = x
        w = np.zeros(n)
        r = np.ones(n) - hvp(w)
        p = r.copy()
        rs = np.dot(r, r)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses mirror descent with variance-regularized objectives, greedy shrinkage,
active-set Newton polishing, and diverse multi-n exploration.
Key innovations: entropy mirror descent, spectral flatness penalty,
aggressive coordinate removal, and two-scale refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global state to avoid lambdas/closures for scipy
_OPT_STATE = {'beta': 0.0, 'n': 0, 'lambda_var': 0.0}


def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_a_w(a, w, n):
    """Compute h[i] = sum_k w[k] * a[k-i] (valid index handling) via FFT."""
    flip_a = a[::-1]
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(flip_a, N)
    result = np.fft.irfft(fw * fa, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _smooth_obj_and_grad(z, beta, n, lambda_var):
    """Compute smoothed objective and gradient."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

    # Base objective: 2n * (c_max + beta * log(s)) / S^2 (S=1 since softmax)
    base_obj = 2.0 * n * (c_max + beta * np.log(s))
    base_grad = 4.0 * n * a * (h - W)

    # Variance penalty
    if lambda_var > 0:
        c_var = float(np.mean((c - c_mean) ** 2))
        var_obj = lambda_var * c_var
        # Gradient of variance: d/dc_k = 2*(c_k - c_mean)/(2n-1)
        # Then chain rule through a
        dc_var = 2.0 * (c - c_mean) / len(c) * lambda_var
        a_rev2 = a[::-1]
        wc_v = _full_convolve(dc_var, a_rev2)
        h_v = wc_v[n - 1:2 * n - 1]
        W_v = float(np.dot(dc_var, c))
        var_grad = 2.0 * n * a * (h_v - W_v)
        obj = base_obj + var_obj
        grad = base_grad + var_grad
    else:
        obj = base_obj
        grad = base_grad

    return float(obj), grad


def _obj_wrapper(z):
    obj, _ = _smooth_obj_and_grad(z, _OPT_STATE['beta'], _OPT_STATE['n'], _OPT_STATE['lambda_var'])
    return obj


def _grad_wrapper(z):
    _, grad = _smooth_obj_and_grad(z, _OPT_STATE['beta'], _OPT_STATE['n'], _OPT_STATE['lambda_var'])
    return grad


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, lambda_var=0.0):
    """L-BFGS with beta continuation and optional variance penalty."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    _OPT_STATE['n'] = n
    _OPT_STATE['lambda_var'] = lambda_var

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        _OPT_STATE['beta'] = beta
        est_time = 0.005 * max(1, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _obj_wrapper,
                z,
                jac=_grad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _mirror_descent(a_init, deadline, max_iter=50000, use_var_penalty=True):
    """Entropy mirror descent with adaptive step size and temperature schedule."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    eta = 1.0
    beta = 0.05
    patience = 0
    max_patience = 2000
    lambda_var = 0.1 if use_var_penalty else 0.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Softmax weights
        w = np.exp((c - c_max) / beta)
        w_sum = w.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        # Subgradient of peak term
        g_peak = 2.0 * _corr_a_w(a, w, n)

        # Variance subgradient
        if lambda_var > 0:
            dc_var = 2.0 * (c - c_mean) / len(c) * lambda_var
            g_var = 2.0 * _corr_a_w(a, dc_var, n)
            g = g_peak + g_var
        else:
            g = g_peak

        # Project to tangent space
        g_t = g - g.mean()
        g_norm = float(np.abs(g_t).max())
        if g_norm < 1e-15:
            break

        # Adaptive step size
        eta = 0.5 / g_norm

        # Mirror descent update in log space
        z = np.log(np.maximum(a, 1e-30))
        z = z - eta * g_t
        z = z - z.max()
        a_new = np.exp(z)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300 or not np.isfinite(a_new_sum):
            eta *= 0.5
            patience += 1
            continue
        a_new /= a_new_sum

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
            # Slowly decrease temperature
            if it % 500 == 0 and it > 0:
                beta = max(0.005, beta * 0.97)
                if use_var_penalty and it > 5000:
                    lambda_var *= 0.99
        else:
            patience += 1
            eta *= 0.8
            if patience > max_patience:
                break

    return best_a, best_sc


def _greedy_shrink(a_init, deadline, max_rounds=500):
    """Iteratively remove the smallest element if it doesn't hurt the score."""
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    removed = 0

    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        n_current = len(a)
        if n_current < 16:
            break

        # Find smallest element
        min_idx = int(np.argmin(a))
        if a[min_idx] < 1e-12:
            a = np.delete(a, min_idx)
            removed += 1
            continue

        # Try removing it
        a_try = np.delete(a, min_idx)
        sc_try = _score(a_try)

        if sc_try <= best_sc + 1e-6:
            a = a_try
            removed += 1
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a.copy()
                # After a good removal, do a quick local re-optimization
                if time.monotonic() < deadline - 5:
                    a_quick = _run_lbfgs(a, min(deadline - 3, time.monotonic() + 2), min_time_remaining=1)
                    sc_quick = _score(a_quick)
                    if sc_quick < best_sc:
                        best_sc = sc_quick
                        best_a = a_quick.copy()
                        a = a_quick.copy()
        else:
            # Try a few other small elements
            sorted_idx = np.argsort(a)
            found = False
            for trial in range(min(10, n_current)):
                idx = sorted_idx[trial]
                a_try2 = np.delete(a, idx)
                sc_try2 = _score(a_try2)
                if sc_try2 <= best_sc + 1e-6:
                    a = a_try2
                    removed += 1
                    if sc_try2 < best_sc:
                        best_sc = sc_try2
                        best_a = a.copy()
                    found = True
                    break
            if not found:
                break

    return best_a, best_sc, removed


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):

    n = len(a_init)
    a = a_init.copy()

            n_active = 1

        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        def _hvp(v):
            return (2.0 / m) * _corr_a_w(v, A_vec, n) + mu * v

            if np.sqrt(rs_new) < 1e-12:

            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        # CG solve for (H+mu*I) w = ones
        w = np.zeros(n)
        r = np.ones(n) - _hvp(w)

            w += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:

            rs = rs_new

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for _ in range(50):

            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-10:
                a = a_try
                accepted = True

        sc = _score(a)

def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * t / 2.0))
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "oscillatory":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t / 7.0)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.uniform(0.5, 1.5, n)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb_log(a, sigma, rng):
    """Perturb in log space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_fn(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Diverse exploration with mirror descent across n values and profiles
    Phase 2: Refined L-BFGS + mirror descent on best candidates
    Phase 3: Greedy shrinkage
    Phase 4: Active-set Newton polish
    Phase 5: Final multi-n check and ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init_uniform")

    # === Phase 1: Diverse exploration ===
    print(f"\n[Phase 1] Diverse exploration (t={time.monotonic() - t_start:.1f}s)")
    n_options = []
    base_n = n_parent if n_parent > 0 else 8192
    for n_try in [4096, 8192, 16384]:
        if n_try != base_n:
            n_options.append(n_try)
    n_options.append(base_n)

    profiles = ["uniform", "cosine", "oscillatory", "tapered"]
    candidates = []

    for n_try in n_options:
        if time.monotonic() > deadline - 600:
            break
        for prof in profiles:
            if time.monotonic() > deadline - 580:
                break
            if parent is not None and n_parent != n_try:
                a_init = _interpolate_to_n(parent, n_try)
                # Blend with profile
                a_prof = _make_profile(n_try, prof)
                a_init = 0.7 * a_init + 0.3 * a_prof
                a_init = np.maximum(a_init, 0)
                a_init /= a_init.sum()
            elif parent is not None and n_parent == n_try:
                a_init = parent.copy()
            else:
                a_init = _make_profile(n_try, prof)

            sc_init = _score(a_init)
            candidates.append((sc_init, a_init, f"n{n_try}_{'blend' if parent is not None and n_parent != n_try else prof}"))

    # Sort candidates by initial score
    candidates.sort(key=lambda x: x[0])

    # Run mirror descent on top candidates
    for sc_init, a_init, label in candidates[:8]:
        if time.monotonic() > deadline - 400:
            break
        print(f"    MD on {label} (init_sc={sc_init:.6f}, n={len(a_init)})...")
        md_deadline = min(deadline - 380, time.monotonic() + 60)
        a_md, sc_md = _mirror_descent(a_init, md_deadline, max_iter=20000)
        update_best(a_md, sc_md, f"md_{label}")

    # === Phase 2: Refined optimization on best ===
    print(f"\n[Phase 2] Refined L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        # L-BFGS with variance penalty
        lb_deadline = min(deadline - 280, time.monotonic() + 120)
        a_lb = _run_lbfgs(best_seq, lb_deadline, min_time_remaining=100, lambda_var=0.05)
        update_best(a_lb, _score(a_lb), "lbfgs_var")

        # L-BFGS without variance penalty
        if time.monotonic() < deadline - 200:
            lb_deadline2 = min(deadline - 180, time.monotonic() + 80)
            a_lb2 = _run_lbfgs(best_seq, lb_deadline2, min_time_remaining=80, lambda_var=0.0)
            update_best(a_lb2, _score(a_lb2), "lbfgs_plain")

        # Mirror descent with lower beta
        if time.monotonic() < deadline - 150:
            md_deadline2 = min(deadline - 130, time.monotonic() + 60)
            a_md2, sc_md2 = _mirror_descent(best_seq, md_deadline2, max_iter=30000, use_var_penalty=True)
            update_best(a_md2, sc_md2, "md_refined")

    # === Phase 3: Greedy shrinkage ===
    print(f"\n[Phase 3] Greedy shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        shr_deadline = min(deadline - 100, time.monotonic() + 80)
        a_shr, sc_shr, n_rem = _greedy_shrink(best_seq, shr_deadline, max_rounds=300)
        update_best(a_shr, sc_shr, f"shrink_{n_rem}removed")

        # Re-optimize after shrinkage
        if sc_shr < best_score and time.monotonic() < deadline - 80:
            a_reopt = _run_lbfgs(a_shr, min(deadline - 60, time.monotonic() + 30), min_time_remaining=30)
            update_best(a_reopt, _score(a_reopt), "shrink_reopt")
