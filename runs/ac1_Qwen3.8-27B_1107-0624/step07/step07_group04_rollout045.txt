"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with tight log-sum-exp, active-set Newton, coordinate descent polish,
peak equalization, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _newton_active_polish(a_init, deadline, max_rounds=100):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Very tight active set
        tol = 0.0001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active

        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-15 * n:
            break

        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try < c_max - 0.05 * alpha * gn * 0.01:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            alpha = 0.005
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _smoobj_factory(beta, n):
    # Will be used with minimize, can't use lambda
    class SMOObj:
        def __init__(self, beta, n):
            self.beta = beta
            self.n = n
        def __call__(self, z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / self.beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * self.n * c_max
            return 2.0 * self.n * (c_max + self.beta * np.log(s))
    return SMOObj(beta, n)


def _smograd_factory(beta, n):
    class SMOGrad:
        def __init__(self, beta, n):
            self.beta = beta
            self.n = n
        def __call__(self, z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / self.beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[self.n - 1:2 * self.n - 1]
            W = float(np.dot(w, c))
            return 4.0 * self.n * a * (h - W)
    return SMOGrad(beta, n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_mult=1.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.003) * beta_mult

    # Very fine beta schedule - more stages, goes much tighter
    ratios = [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.035,
              0.025, 0.018, 0.012, 0.008, 0.005, 0.0035, 0.0025, 0.0015,
              0.001, 0.0007, 0.0005, 0.00035, 0.00025, 0.00015, 0.0001,
              0.00007, 0.00005, 0.000035, 0.000025, 0.000015]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # More iterations at tighter betas
        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        if beta < base_beta * 0.01:
            max_iter = min(5000, max(100, int(remaining / (est_iter_time * 8))))
        else:
            max_iter = min(3000, max(50, int(remaining / (est_iter_time * 15))))

        try:
            obj_fn = _smoobj_factory(float(beta), n)
            grad_fn = _smograd_factory(float(beta), n)
            result = minimize(
                obj_fn,
                z,
                jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i < len(betas) - 1:
                # Try a smaller step
                z = z * 0.99  # Damp
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, max_passes=50):
    """Coordinate descent polish: optimize one coordinate at a time."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Process coordinates in order of impact (those near the max peaks)
    # The peak of c is at certain locations; coordinates that contribute most
    # to those peaks should be optimized first
    peak_idx = int(np.argmax(c))
    # Coordinates j that contribute to c[peak_idx]: a[j]*a[peak_idx-j]
    # So j and peak_idx-j are both important
    
    # Use random order to avoid bias
    rng = np.random.default_rng(12345)
    
    for pass_idx in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        # Random permutation of coordinates
        order = rng.permutation(n)
        
        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            
            # Optimize a[j] by 1D search
            # c[k] as function of a[j]:
            # c[k] = 2*a[j]*a[k-j] (for k-j != j and both in range)
            #      + a[j]^2 (for k = 2j)
            #      + (terms not involving a[j])
            
            a_old = a[j]
            # The contribution of a[j] to c:
            # For k in [0, 2n-2]:
            #   c[k] includes a[j]*a[k-j] if 0 <= k-j <= n-1
            #   and a[k-j]*a[j] if 0 <= k-j <= n-1 (same term unless k-j==j)
            
            # Let's just do a simple golden section search on a[j] in [0, 3*a_old + 0.01/n]
            lo = 0.0
            hi = 3.0 * a_old + 0.01 / n
            if hi <= lo:
                hi = 0.01 / n
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            def eval_coord(val):
                a[j] = val
                s_new = a.sum()
                if s_new < 1e-15:
                    return np.inf
                c_new = _full_convolve(a, a)
                return float(c_new.max() / (s_new * s_new))
            
            f1 = eval_coord(c1)
            f2 = eval_coord(c2)
            
            for _ in range(30):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_coord(c2)
            
            best_val = (lo + hi) / 2.0
            a[j] = best_val
            # Restore or keep?
            # Check if it improved
            s_final = a.sum()
            if s_final > 1e-15:
                c_final = _full_convolve(a, a)
                sc_final = float(c_final.max() / (s_final * s_final))
                if sc_final < best_sc - 1e-16:
                    best_sc = sc_final
                    best_a = a.copy()
                    c = c_final
                    c_max = float(c.max())
                    improved = True
        
        if not improved:
            break
    
    a = best_a / best_a.sum()
    return a


def _peak_equalization(a_init, deadline):
    """Try to equalize the top peaks of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for iteration in range(50):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        top_k = min(20, len(c))
        top_indices = np.argsort(c)[-top_k:]
        top_values = c[top_indices]
        
        # If already well equalized, stop
        if top_values.max() - top_values.min() < 1e-8 * c_max:
            break
        
        # Compute subgradient of "spread" of top peaks
        # We want to reduce the max while not increasing others
        # Use the highest peak's subgradient
        peak_idx = int(np.argmax(c))
        
        # Gradient of c[peak_idx] with respect to a:
        # d(c[k])/da[j] = a[k-j] + (1 if k-j==j else 0) * a[j]... 
        # Actually d(c[k])/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if k-j != j
        # and = 2*a[j] if k-j == j (i.e., k = 2j)
        # Wait: c[k] = sum_i a[i]*a[k-i]
        # d(c[k])/da[j] = a[k-j] + (k-j) term: if j != k-j, d/d a[j] of a[j]*a[k-j] = a[k-j]
        # and d/d a[j] of a[k-j]*a[j] = a[k-j] (same, it's the same term)
        # So d(c[k])/da[j] = 2*a[k-j] (for k-j in [0,n-1])
        # If k-j == j (k=2j): d/d a[j] of a[j]^2 = 2*a[j]
        
        grad = np.zeros(n)
        k = peak_idx
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if kj == j:
                    grad[j] = 2.0 * a[j]
                else:
                    grad[j] = 2.0 * a[kj]
        
        # Also need to account for the 1/sum(a)^2 factor in the score
        # Score = 2n * max(c) / s^2
        # d(score)/da[j] = 2n * (d(max_c)/da[j] * s^2 - max_c * 2*s * da[j]) / s^4
        #               = 2n * (grad[j] - max_c * 2/s * ... ) / s^2  -- this gets complicated
        # For simplicity, just use the gradient of c[peak_idx] and project onto sum=0
        
        grad = grad - np.mean(grad)  # Project to preserve sum
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Take a small step to reduce the peak
        step_size = 0.01 * a.std() / grad_norm
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-16:
            a = a_new
            best_sc = sc_new
            best_a = a.copy()
        else:
            # Try smaller step
            step_size *= 0.1
            a_try = a - step_size * grad
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                else:
                    break
    
    return best_a


def _subgrad_polish(a_init, deadline, max_steps=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    lr = 0.03

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Dynamic active set - very tight
        tol = 0.0005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 15 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.06)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.02
            z = z_new
        else:
            z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "symmetric_tapered":
        # Explicitly symmetric
        half = n // 2
        t = np.linspace(0, 1, half + 1)
        a_half = np.sin(np.pi * t) ** 0.6
        a_half = np.maximum(a_half, 1e-10)
        if n % 2 == 0:
            a = np.concatenate([a_half[:-1][::-1], a_half])
        else:
            a = np.concatenate([a_half[:-1][::-1], a_half])
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with ultra-tight log-sum-exp continuation,
    active-set CG-Newton polish, coordinate descent, peak equalization,
    subgradient, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS from parent with tight beta
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=400.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Newton active set polish
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 300, time.monotonic() + 150))
        update_best(a2, sc2, "newton_1")

    # Phase 3: Peak equalization
    print(f"\n[Phase 3] Peak equalization (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a3 = _peak_equalization(best_seq, min(deadline - 200, time.monotonic() + 60))
        update_best(a3, _score(a3), "peak_eq")

    # Phase 4: More Newton after peak eq
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 150, time.monotonic() + 80))
        update_best(a4, sc4, "newton_2")

    # Phase 5: Coordinate descent
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a5 = _coordinate_descent(best_seq, min(deadline - 100, time.monotonic() + 60), max_passes=10)
        update_best(a5, _score(a5), "coord_desc")

    # Phase 6: Subgradient polish
    print(f"\n[Phase 6] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a6, sc6 = _subgrad_polish(best_seq, min(deadline - 80, time.monotonic() + 40))
        update_best(a6, sc6, "subgrad")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [16384, 4096]:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_np, sc_np = _newton_active_polish(a_opt, min(deadline - 25, time.monotonic() + 30))
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 8: Perturbed restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(3):
            if time.monotonic() > deadline - 15:
                break
            sigma = [0.02, 0.05, 0.1][i]
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")
