fine log-sum-exp L-BFGS, tight active-set peak shaving with FFT-based

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(2.0 * n * c_max)
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # FFT-based gradient: grad[j] = sum_k w[k] * a[k-j] (valid)
    # = (linear_convolve(reversed(w), a))[n-1:2n-1][::-1]
    w_rev = w[::-1]
    N = _next_pow2(len(w_rev) + len(a) - 1)
    Wf = np.fft.rfft(w_rev, N)
    Af = np.fft.rfft(a, N)
    conv = np.fft.irfft(Wf * Af, N)
    grad = conv[n - 1:2 * n - 1][::-1]
    return 4.0 * n * a * grad


def _gradient_exact(a, w, n):
    """Compute grad[j] = sum_k w[k] * a[k-j] (valid) via FFT."""
    w_rev = w[::-1]
    N = _next_pow2(len(w_rev) + len(a) - 1)
    Wf = np.fft.rfft(w_rev, N)
    Af = np.fft.rfft(a, N)
    conv = np.fft.irfft(Wf * Af, N)
    return conv[n - 1:2 * n - 1][::-1]


def _top_k_weights(c, k=3):
    """Create weights concentrated on top-k peaks."""
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    # Use top-k with exponential weighting
    sorted_idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    for rank, idx in enumerate(sorted_idx):
        w[idx] = 1.0 / (rank + 1)
    w = w / w.sum()
    return w


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-12):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',

                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _pairwise_nudging(a_init, deadline, max_iters=30000):

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()

        if lag == 0:

            # Use gradient of c_0 = sum a_i^2

            a = a_try

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-5
            max_iter = min(5000, max(500, int((deadline - 200 - time.monotonic()) / 0.005)))
            a_alt, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_alt, f"alt_lbfgs_{cycle}")

        # Peak shaving
        if time.monotonic() < deadline - 150:
            a_ps_alt, _ = _peak_shaving_tight(best_seq, deadline - 150, max_iters=20000, top_k=2)
            try_update(a_ps_alt, f"alt_ps_{cycle}")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Try n=12288 ===
    print(f"\n=== Phase 6: Larger n exploration ===")
    t_phase = time.monotonic()
    if start_n < 12288 and time.monotonic() < deadline - 200:
        print("  Trying n=12288...")
        a_interp = _interpolate_to_n(best_seq, 12288)
        sc_interp = _score(a_interp)
        print(f"    Interpolated score: {sc_interp:.10f}")
        a_opt12 = _lbfgs_schedule(a_interp, deadline - 150, min_time_remaining=100.0)
        try_update(a_opt12, "n12288_lbfgs")
        if time.monotonic() < deadline - 100:
            a_ps12, _ = _peak_shaving_tight(a_opt12, deadline - 100, max_iters=20000, top_k=3)
            try_update(a_ps12, "n12288_ps")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Coordinate descent on best ===
    print(f"\n=== Phase 7: Coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 60, max_rounds=30000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Multi-restart ===
    print(f"\n=== Phase 8: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("gauss_s1", "gauss", 0.01),
            ("gauss_s3", "gauss", 0.03),
            ("fourier", "fourier", 0.0),
            ("block", "block", 0.0),
            ("gauss_s5", "gauss", 0.05),
        ]
        for name, mode, scale in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb_restart(best_seq, rng, mode, scale)
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            if time.monotonic() < deadline - 25:
                a_r = _lbfgs_schedule(p_start, deadline - 20, min_time_remaining=15.0)
                try_update(a_r, f"restart_{name}")

            if time.monotonic() < deadline - 15:
                a_r2, _ = _peak_shaving_tight(a_r, deadline - 10, max_iters=10000, top_k=3)
                try_update(a_r2, f"restart_{name}_ps")
    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(3):
            if time.monotonic() > deadline - 5:
                break
            # Ultra-fine L-BFGS
            if time.monotonic() < deadline - 5 and cycle == 0:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 3e-8
                max_iter = min(5000, max(200, int((deadline - 5 - time.monotonic()) / 0.004)))
                a_ultra, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                try_update(a_ultra, f"ultra_{cycle}")

            # Peak shave
            if time.monotonic() < deadline - 5:
                a_f1, _ = _peak_shaving_tight(best_seq, deadline - 5, max_iters=15000, top_k=2)
                try_update(a_f1, f"final_ps_{cycle}")

            # Coordinate descent
            if time.monotonic() < deadline - 5:
                a_f2, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=10000)
                try_update(a_f2, f"final_cd_{cycle}")

            # Nudge
            if time.monotonic() < deadline - 5:
                a_f3, _ = _pairwise_nudging(best_seq, deadline - 5, max_iters=10000)
                try_update(a_f3, f"final_nudge_{cycle}")

    print(f"  Phase 9 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(8192) / 8192.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(8192) / 8192.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
