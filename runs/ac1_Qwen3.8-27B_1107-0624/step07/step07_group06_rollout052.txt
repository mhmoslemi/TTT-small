"""

adaptive active-set selection, aggressive L-BFGS beta continuation,
Newton active-set refinement, multi-n exploration, and diverse restarts.
Key improvements over previous: (1) exponentiated gradient as main workhorse

(3) adaptive active-set size, (4) periodic shrinkage, (5) multi-n search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk(a, c, n, K, beta_w=0.0):
    """Gradient of weighted sum of top-K autocorrelation values."""
    c_max = float(c.max())
    if beta_w <= 0:
        # Hard top-K
        if K >= len(c):
            active = np.ones_like(c, dtype=bool)
        else:
            threshold = np.partition(c, len(c) - K)[len(c) - K]
            active = c >= threshold
            if active.sum() == 0:
                active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
    else:
        # Soft top-K with temperature
        shifted = (c - c_max) / beta_w
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hessian_matvec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=100, tol=1e-10):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _mirror_descent(a_init, deadline, min_time_remaining=100.0):
    """Exponentiated gradient (mirror descent) with adaptive active-set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.05
    patience = 0
    K_frac = 0.1  # top 10% active set
    beta_w = 0.0  # hard top-K initially

    elapsed_per_step = 0.001
    step_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive K: start broad, narrow down
        progress = min(1.0, step_count / 50000.0)
        K = max(1, int(len(c) * (K_frac * (1.0 - 0.5 * progress))))

        # Adaptive beta_w: start soft, become harder
        beta_w = c_max * 0.01 * (1.0 - progress) + c_max * 0.001

        grad = _gradient_topk(a, c, n, K, beta_w)

        # Step in log-domain
        z = z - lr * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        step_count += 1

        # Periodic evaluation
        if step_count % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.2)
            else:
                patience += 1
                lr *= 0.97
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    lr = 0.05
                    patience = 0

        # Periodic shrinkage: zero out smallest elements
        if step_count % 5000 == 0:
            a_try = _softmax(z).copy()
            threshold_val = np.percentile(a_try, 5)
            a_try[a_try < threshold_val] = 0
            s_try = a_try.sum()
            if s_try > 0.01:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-10:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_aggressive(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with very aggressive beta continuation (more stages, lower final)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    # 20 stages, geometric from 1x to 1e-8
    n_stages = 20
    betas = base_beta * (1e-8) ** (np.arange(n_stages) / (n_stages - 1))
    # Ensure monotonically decreasing
    for i in range(1, n_stages):
        if betas[i] > betas[i-1]:
            betas[i] = betas[i-1] * 0.5

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate iterations: each iter ~ 2-10ms for n=8192
        est_ms_per_iter = 0.5 + 0.3 * max(1, n / 4096.0) * 10
        max_iter = min(20000, max(200, int(remaining * 1000 / (est_ms_per_iter * 5))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 80, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 5 == 0:
                    print(f"    L-BFGS stage {i}/{n_stages}: beta={beta:.3e}, score={current_sc:.10f}")
        except Exception as ex:
            print(f"    L-BFGS stage {i} error: {ex}")

    return current_a


def _newton_active_set(a_init, deadline, max_iter=200):
    """Newton's method on the active set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _hessian_matvec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-13:
            break

        mu = 1e-7

        def hess_vec(v):
            return _hessian_matvec(v, active, n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=100, tol=1e-10)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        stepped = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                stepped = True
                break
            alpha *= 0.5

        if not stepped and alpha < 1e-12:
            break

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _shrinkage_optimize(a, deadline, min_time_remaining=60.0):
    """Try zeroing out smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.0, 0.02, 0.05, 0.10, 0.15, 0.25]:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = int(n * frac)
        if n_zero == 0:
            continue
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 0.01:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            # Re-optimize from shrunk point
            if time.monotonic() + min_time_remaining > deadline:
                a_opt = _lbfgs_aggressive(best_a, deadline, min_time_remaining=min_time_remaining + 30)
                sc_opt = _score(a_opt)
                if sc_opt < best_sc:
                    best_sc = sc_opt
                    best_a = a_opt

    return best_a, best_sc


def _perturb(a, rng, mode=0):
    n = len(a)
    if mode == 0:
        # Log perturbation
        z = np.log(np.maximum(a, 1e-30))
        z = z + 0.02 * rng.standard_normal(n)
        z = z - z.mean()
        return _softmax(z)
    elif mode == 1:
        # Fourier perturbation
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        t = np.arange(n)
        for _ in range(3):
            freq = rng.integers(1, max(2, n // 5))
            amp = 0.02 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        z = z - z.mean()
        return _softmax(z)
    elif mode == 2:
        # Block swap
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        n_blocks = 3
        block_size = max(1, n // (2 * n_blocks))
        for _ in range(n_blocks):
            i = rng.integers(0, max(1, n - 2 * block_size))
            j = rng.integers(0, max(1, n - 2 * block_size))
            if abs(i - j) < block_size:
                continue
            tmp = z[i:i + block_size].copy()
            z[i:i + block_size] = z[j:j + block_size]
            z[j:j + block_size] = tmp
        return _softmax(z)
    elif mode == 3:
        # Mirror + noise
        mirrored = a.copy()
        mirrored[n // 2:] = a[:n // 2][::-1]
        z = np.log(np.maximum(mirrored, 1e-30))
        z = z + 0.03 * rng.standard_normal(n)
        z = z - z.mean()
        return _softmax(z)
    else:
        # Random reweight
        z = np.log(np.maximum(a, 1e-30))
        z = z + 0.05 * rng.standard_normal(n)
        z = z - z.mean()
        return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: mirror descent + aggressive L-BFGS + Newton
    + shrinkage + multi-n + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Mirror descent (peak shaving) ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 1] Mirror descent (remaining={remaining:.0f}s)")
    if remaining > 500:
        md_time = min(400, remaining - 500)
        md_deadline = time.monotonic() + md_time
        a_md, sc_md = _mirror_descent(best_seq, md_deadline, min_time_remaining=10)
        update_best(a_md, sc_md, "mirror_descent")

    # === Phase 2: Aggressive L-BFGS ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 2] Aggressive L-BFGS (remaining={remaining:.0f}s)")
    if remaining > 400:
        lbfgs_time = min(450, remaining - 400)
        lbfgs_deadline = time.monotonic() + lbfgs_time
        a_lb = _lbfgs_aggressive(best_seq, lbfgs_deadline, min_time_remaining=30)
        sc_lb = _score(a_lb)
        update_best(a_lb, sc_lb, "lbfgs_main")

    # === Phase 3: Newton active-set ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 3] Newton active-set (remaining={remaining:.0f}s)")
    if best_seq is not None and remaining > 300:
        n_deadline = time.monotonic() + min(200, remaining - 300)
        a_newton, sc_newton = _newton_active_set(best_seq, n_deadline, max_iter=150)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 4: Mirror descent again (from improved point) ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 4] Mirror descent round 2 (remaining={remaining:.0f}s)")
    if remaining > 300:
        md2_time = min(200, remaining - 300)
        md2_deadline = time.monotonic() + md2_time
        a_md2, sc_md2 = _mirror_descent(best_seq, md2_deadline, min_time_remaining=10)
        update_best(a_md2, sc_md2, "mirror_descent_2")
        if sc_md2 < best_score * 1.001:
            a_md2b = _lbfgs_aggressive(a_md2, deadline, min_time_remaining=150)
            update_best(a_md2b, _score(a_md2b), "md2_lbfgs")

    # === Phase 5: Shrinkage ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 5] Shrinkage (remaining={remaining:.0f}s)")
    if best_seq is not None and remaining > 200:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline, min_time_remaining=100)
        update_best(a_shr, sc_shr, "shrinkage")

    # === Phase 6: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 6] Multi-n exploration (remaining={remaining:.0f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 0.01:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "uniform")
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        update_best(a_init, sc_init, f"n_{n_try}_init")

        # Quick L-BFGS
        if time.monotonic() > deadline - 80:
            break
        a_opt = _lbfgs_aggressive(a_init, deadline, min_time_remaining=60)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        # Quick mirror descent
        if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 50:
            a_md_n, sc_md_n = _mirror_descent(a_opt, min(deadline - 30, time.monotonic() + 80), min_time_remaining=10)
            update_best(a_md_n, sc_md_n, f"n_{n_try}_md")

    # === Phase 7: Diverse restarts ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 7] Diverse restarts (remaining={remaining:.0f}s)")
    if best_seq is not None and remaining > 80:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            mode = i % 4
            p_start = _perturb(best_seq, rng, mode=mode)
            # Quick optimization
            a_r = _lbfgs_aggressive(p_start, deadline, min_time_remaining=15)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}_lbfgs")

            if sc_r < best_score * 1.003 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _mirror_descent(a_r, min(deadline - 10, time.monotonic() + 40), min_time_remaining=5)
                update_best(a_r2, sc_r2, f"restart_{i}_md")

    # === Phase 8: Final ultra-polish ===
    remaining = deadline - time.monotonic()
    print(f"\n[Phase 8] Final polish (remaining={remaining:.0f}s)")
    if best_seq is not None and remaining > 20:
        # Newton
        n_fin = time.monotonic() + min(60, remaining - 10)
        a_fin, sc_fin = _newton_active_set(best_seq, n_fin, max_iter=100)
        update_best(a_fin, sc_fin, "final_newton")

        # Ultra-low beta L-BFGS
        if time.monotonic() < deadline - 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_ult = mb * 1e-6
            remaining_ultra = deadline - 10 - time.monotonic()
            if remaining_ultra > 5:
                max_iter = min(5000, max(100, int(remaining_ultra / 0.01)))

                def fun_ult(z_local):
                    return _smoobj(z_local, beta_ult, n)

                def jac_ult(z_local):
                    return _smograd(z_local, beta_ult, n)

                try:
                    result = minimize(
                        fun_ult, z, jac=jac_ult, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 60}
                    )
                    a_ult = _softmax(result.x)
                    update_best(a_ult, _score(a_ult), "final_ultra_lbfgs")
                except Exception:
                    pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
