import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_weights(c, beta):
    shifted = (c - c.max()) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _power_weights(c, p):
    """Weights proportional to c_k^(p-1), normalized."""
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** (p - 1)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    return w


def _autocorr_gradient(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = convolve(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _hybrid_weights(c, k_active, beta, alpha):
    """Combine top-k onehot with softmax weighting."""
    n_c = len(c)
    top_k = np.argsort(c)[::-1][:k_active]
    w_active = np.zeros(n_c)
    w_active[top_k] = 1.0 / k_active
    w_smooth = _logsumexp_weights(c, beta)
    w = alpha * w_active + (1.0 - alpha) * w_smooth
    s = w.sum()
    if s < 1e-300:
        w = w_active
    else:
        w = w / s
    return w

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    eta = 1.0 / np.sqrt(n)
    momentum = 0.9
    g_prev = np.zeros(n)
    stall = 0
    stall_limit = 200
    check_interval = max(100, n // 100)

    for t in range(1, max_steps + 1):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k = min(k_active + t // 10000, 20)
        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / k

        g = _autocorr_gradient(a, w)
        g = g - g.mean()

        # Nesterov: use extrapolated gradient
        g_nest = g + momentum * (g - g_prev)
        g_prev = g.copy()

        # Multiplicative update in log-space
        z = np.log(np.maximum(a, 1e-18))
        z = z - eta * g_nest / (a.sum() / n)
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        a_new = _softmax(z)

            if sc < best_sc:

                best_a = a_new.copy()

            else:
                stall += 1
                eta *= 0.95
                if stall > stall_limit:
                    break

def _kkt_newton_refine(a_init, deadline, k_active=4, max_iter=200):
    """
    Newton-type refinement of KKT conditions.
    At optimum: for a[i] > 0, sum_k lambda_k * (shift_k(a))[i] = constant.
    We iteratively adjust a to equalize the active constraint gradients.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        top_k = np.argsort(c)[::-1][:k_active]

        # Compute the "autocorrelation gradient" at active lags
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / k_active
        g = _autocorr_gradient(a, w)

        # The KKT condition says g should be constant on the support of a.
        # Compute the deviation from constancy:
        support = a > 1e-10
        if support.sum() < 2:
            break
        g_support = g[support]
        g_mean = g_support.mean()
        deviation = g - g_mean

        # Newton-like step: reduce the deviation
        # Hessian approximation: use the variance of g as scaling
        g_var = np.var(g)
        if g_var < 1e-20:
            break

        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(15):
            a_try = a * np.exp(-alpha * deviation / (g_var + 1e-10))
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            # Try direct gradient descent step
            alpha = 0.1 / np.sqrt(n)
            a_try = a - alpha * deviation
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
            else:
                break

    return best_a, best_sc


def _multigrid_optimize(a_init, deadline, n_coarse=None):
    """
    Coarse-to-fine optimization:
    1. Downsample to coarse grid
    2. Optimize on coarse grid
    3. Upsample and refine on fine grid
    """
    n = len(a_init)
    if n_coarse is None:
        n_coarse = max(64, n // 16)

    # Downsample

        t_fine = np.linspace(0, 1, n)
        t_coarse = np.linspace(0, 1, n_coarse)
        a_coarse = np.interp(t_coarse, t_fine, a_init)
        a_coarse = np.maximum(a_coarse, 1e-12)
        a_coarse = a_coarse / a_coarse.sum()

        # Optimize on coarse grid

    else:

def _spectral_basin_hop(a_current, deadline, rng, n_hops=5, reserve=30.0):
    """
    Diversification using spectral perturbations followed by quick optimization.
    """
    n = len(a_current)
    best_a = a_current.copy()

    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6

        # Generate perturbation in spectral domain
        A = np.fft.rfft(a_current, _next_pow2(2 * n - 1))
        amp = np.abs(A)
        phase = np.angle(A)

        # Perturb amplitude
        n_pert = 5 + 3 * h
        for _ in range(n_pert):
            freq = int(rng.integers(1, max(2, len(A) // 4)))
            delta = rng.uniform(-0.1, 0.1) * (1 + h * 0.3)
            amp[freq] = np.maximum(amp[freq] * np.exp(delta), 1e-10)
            if freq + 1 < len(A):
                amp[freq + 1] = np.maximum(amp[freq + 1] * np.exp(-delta * 0.5), 1e-10)

        A_new = amp * np.exp(1j * phase)
        a_pert = np.fft.irfft(A_new, _next_pow2(2 * n - 1))[:n]
        a_pert = np.maximum(a_pert, 0.0)

        if s < 1e-15:

        # Quick optimization from perturbation

            a_pert, hop_deadline, lr_init=0.03,
            k_active_init=5, max_steps=10000)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc

        c_mean = c.mean()
        c_std = c.std()

        # above a threshold
        threshold = c_max - 3 * c_std
        active_mask = c > threshold
        k = max(3, int(active_mask.sum()))
        k = min(k, 20)  # cap at 20

        top_k = np.argsort(c)[::-1][:k]
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / k

        g = _autocorr_gradient(a, w)
        g = g - g.mean()

        # Line search with backtracking
        alpha = 1.0
        improved = False
        for _ in range(12):
            a_try = a * np.exp(-alpha * g / (n * c_max + 1e-10))
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.65

        if not improved:
            if alpha < 1e-10:
                break
            # Try small direct step
            a_try = a - 0.01 * g / n
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
            else:
                break

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining Adam, multiplicative updates,
    KKT refinement, multigrid, and spectral diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        # Fallback initialization
        n_init = 4096
        parent = _make_chirp(n_init, freq=1.0, eps=0.3)
        try_update(parent, "chirp_fallback")
        print(f"  Fallback chirp: n={len(parent)}, score={_score(parent):.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: Adam smoothing optimization (main) ===
    print(f"\n=== Phase 2: Adam smoothing (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        a_adam, sc_adam = _adam_smooth_optimize(
            best_seq, deadline - 300, lr_init=0.05,
            k_active_init=7, max_steps=200000)
        try_update(a_adam, "adam_main")

    # === Phase 3: Nesterov multiplicative ===
    print(f"\n=== Phase 3: Nesterov multiplicative (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 250:
        a_nest, sc_nest = _nesterov_multiplicative(
            best_seq, deadline - 250, k_active=5, max_steps=50000)
        try_update(a_nest, "nesterov")

    # === Phase 4: Adaptive KKT ===
    print(f"\n=== Phase 4: Adaptive KKT (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_kkt, sc_kkt = _topk_adaptive_kkt(best_seq, deadline - 200, max_iter=1000)
        try_update(a_kkt, "kkt_adaptive")

    # === Phase 5: Second Adam round with tighter params ===
    print(f"\n=== Phase 5: Adam round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_adam2, sc_adam2 = _adam_smooth_optimize(
            best_seq, deadline - 200, lr_init=0.02,
            k_active_init=4, max_steps=100000)
        try_update(a_adam2, "adam_r2")

    # === Phase 6: Newton KKT refinement ===
    print(f"\n=== Phase 6: Newton KKT (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        a_newton, sc_newton = _kkt_newton_refine(
            best_seq, deadline - 150, k_active=5, max_iter=500)
        try_update(a_newton, "newton_kkt")

    # === Phase 7: Spectral basin hopping ===
    print(f"\n=== Phase 7: Spectral basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_hop, sc_hop = _spectral_basin_hop(
            best_seq, deadline - 100, rng, n_hops=6, reserve=40.0)
        try_update(a_hop, "spectral_hop")

    # === Phase 8: Multigrid ===
    print(f"\n=== Phase 8: Multigrid (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        a_mg, sc_mg = _multigrid_optimize(best_seq, deadline - 80)
        try_update(a_mg, "multigrid")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [8192, 32768, 10240, 6144]:
            if time.monotonic() > deadline - 40:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")
            if sc_interp < best_score_val + 0.05 and time.monotonic() < deadline - 30:
                a_opt, _ = _adam_smooth_optimize(
                    a_interp, deadline - 20, lr_init=0.03, max_steps=30000)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 10: Final polishing ===
    print(f"\n=== Phase 10: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(8):
            if time.monotonic() > deadline - 5:
                break

            # Adam with very tight smoothing
            a_f, _ = _adam_smooth_optimize(
                best_seq, deadline - 5, lr_init=0.01,
                k_active_init=3, max_steps=20000)
            try_update(a_f, f"final_adam_{rnd}")

            # Quick KKT
            if time.monotonic() < deadline - 5:
                a_k, _ = _topk_adaptive_kkt(best_seq, deadline - 5, max_iter=200)
                try_update(a_k, f"final_kkt_{rnd}")

            # Nesterov
            if time.monotonic() < deadline - 5:
                a_ns, _ = _nesterov_multiplicative(
                    best_seq, deadline - 5, k_active=3, max_steps=10000)
                try_update(a_ns, f"final_nest_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
