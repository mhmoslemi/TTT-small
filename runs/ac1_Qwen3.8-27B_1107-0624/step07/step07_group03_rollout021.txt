  2. Optimize over $a_i \geq 0$ the objective $\max_k c_k(a)$ (without worrying about the sum)
  3. The optimal $a_i$ is the same whether or not we normalize

  3. Taking the best

"""
Optimized step function minimizer using direct projected gradient with adaptive 
active set, Newton refinement, coordinate descent polish, and multi-start diversity.
Key innovation: works directly in a-space (no log-sum-exp bias), uses exact 
backtracking line search, and coordinate descent for final precision.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    total = v.sum()
    if total < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient(a, active_idx):
    n = len(a)
    m = len(active_idx)
    if m == 0:
        return np.zeros(n)
    w = np.zeros(2 * n - 1)
    w[active_idx] = 1.0 / m
    flip_a = a[::-1]
    wc = _convolve(w, flip_a)
    return 2.0 * wc[n - 1:2 * n - 1]


def _projected_gradient(a_init, deadline, max_iter=50000, min_time=60.0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    best_c = _convolve(a, a)
    best_max = float(best_c.max())

    alpha = 0.01
    stall = 0
    last_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline - min_time:
            break

        c = _convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: top peaks within tolerance
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[np.argmax(c)] = True
            n_active = 1
        if n_active > 4 * n:
            top = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top] = True
            n_active = 4 * n

        grad = _subgradient(a, np.where(active)[0])
        # Project gradient onto sum=0 subspace
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        f_old = float(2.0 * n * c_max)  # proportional to score since sum=1
        step_accepted = False
        step = alpha
        for ls in range(60):
            a_try = a - step * grad
            # Non-negativity: clamp
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                step *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-14 * step * gnorm:
                a = a_try
                step_accepted = True
                break
            step *= 0.5

        if not step_accepted:
            stall += 1
            # Try smaller adaptive step
            step = max(1e-6, alpha * 0.1)
            a_try = np.maximum(a - step * grad, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try / s_try
            c_try = _convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old:
                a = a_try
            else:
                if stall > 100:
                    break
                # Add small perturbation
                rng_local = np.random.default_rng(it)
                a = a + 0.001 * rng_local.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / max(a.sum(), 1e-15)
                stall = 0
        else:
            if step < alpha * 0.5:
                alpha = max(step, 1e-8)
            elif step > alpha * 1.5:
                alpha = min(step * 1.2, 0.1)

        # Track best
        c_cur = _convolve(a, a)
        cur_max = float(c_cur.max())
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_sc = float(2.0 * n * best_max)
            best_a = a.copy()
            if it - last_improve > 0:
                alpha = min(alpha * 1.05, 0.1)
            last_improve = it

    return best_a, best_sc


def _newton_refine(a_init, deadline, max_iter=200, tol_active=0.001, min_time=30.0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = np.maximum(a, 0)
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline - min_time:
            break

        c = _convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[np.argmax(c)] = True
            n_active = 1
        if n_active > 3 * n:
            top = np.argsort(c)[::-1][:3 * n]
            tmp = np.zeros_like(c, dtype=bool)
            tmp[top] = True
            active = tmp
            n_active = 3 * n

        m = n_active
        A_idx = np.where(active)[0]

        # Gradient of average active: (2/m) * sum_{k in A} a[k-i]
        g = _subgradient(a, A_idx)
        # Project to tangent space
        g_t = g - g.mean()
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Newton direction: solve (H + mu*I) dx = -g_t in tangent space
        # H = (2/m) * convolution operator with active indicator
        mu = 1e-6 * c_max / max(m, 1) + 1e-8

        def hess_matvec(v):
            w = np.zeros(2 * n - 1)
            w[A_idx] = 1.0 / m
            flip_v = v[::-1]
            wc = _convolve(w, flip_v)
            return 2.0 * wc[n - 1:2 * n - 1] + mu * v

        # CG solve
        x = np.zeros(n)
        r = -g_t - hess_matvec(x)
        p = r.copy()
        rs = float(np.dot(r, r))
        for cg_it in range(100):
            Ap = hess_matvec(p)
            pAp = float(np.dot(p, Ap))
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            x = x + alpha_cg * p
            r = r - alpha_cg * Ap
            rs_new = float(np.dot(r, r))
            if np.sqrt(rs_new) < 1e-12 * (1.0 + gnorm):
                rs = rs_new
                break
            p = r + (rs_new / rs) * p
            rs = rs_new
        dx = x
        # Ensure tangent space
        dx = dx - dx.mean()

        # Line search with non-negativity
        neg_mask = dx < -1e-14
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.95
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-12)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.any(a_try < -1e-12):
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * t * gnorm:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # Gradient fallback
            t = min(0.05, t_max)
            for ls in range(20):
                a_try = np.maximum(a - t * g_t, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, n_sweeps=5, min_time=10.0):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = np.maximum(a, 0)
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    order = np.arange(n)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - min_time:
            break
        np.random.shuffle(order)

        for i in order:
            if time.monotonic() > deadline - min_time:
                break

            # Compute c for current a
            # c_k as function of a[i]: c_k = B_k + coeff * a[i] + quad * a[i]^2
            # For each k: 
            #   if k == 2*i: c_k = B_k + a[i]^2
            #   elif 0 <= k-i < n and k-i != i: c_k = B_k + 2*a[k-i]*a[i]
            #   else: c_k = B_k
            # B_k = c_k(current) - contribution of a[i]
            c_cur = _convolve(a, a)

            # Contribution of a[i] to c_k:
            # a[i] * a[k-i] if 0 <= k-i < n
            # Plus if k-i == i (k=2i): the a[i]^2 term is already counted once
            # Actually c_k = sum_j a[j]*a[k-j]. The terms involving a[i] are:
            #   j=i: a[i]*a[k-i] (if 0<=k-i<n)
            #   k-j=i => j=k-i: a[k-i]*a[i] (if 0<=k-i<n and k-i != i)
            # So total contribution = a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] if k!=2i
            #                      = a[i]^2 if k==2i
            # B_k = c_k - (2*a[i]*a[k-i] if 0<=k-i<n and k!=2i else a[i]^2 if k==2i else 0)

            ai = a[i]
            coeff = np.zeros(2 * n - 1)
            quad = np.zeros(2 * n - 1)
            B = c_cur.copy()

            # For k where 0 <= k-i < n:
            k_range = np.arange(max(0, i), min(2 * n - 1, n + i))
            for k in k_range:
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        quad[k] = 1.0
                        B[k] = c_cur[k] - ai * ai
                    else:
                        coeff[k] = 2.0 * a[ki]
                        B[k] = c_cur[k] - 2.0 * ai * a[ki]

            # Also for k where k-i >= n or k-i < 0: no contribution, B[k] = c_k
            # (already handled since we only modified valid k)

            # Now minimize max_k (B[k] + coeff[k]*x + quad[k]*x^2) over x >= 0
            # subject to: the new a[i] = x, and we renormalize
            # But since score is scale-invariant, we just minimize max_k(...) / (sum)^2
            # which after renormalization is 2n * max_k(...) / (sum_new)^2
            # where sum_new = (S - ai + x), S = sum(a)
            
            S = float(a.sum())
            
            # We want to minimize: 2n * max_k(B[k] + coeff[k]*x + quad[k]*x^2) / (S - ai + x)^2
            # Let's just minimize max_k(B[k] + coeff[k]*x + quad[k]*x^2) (the sum part 
            # is a smooth factor, and the max part dominates the optimization)
            
            # Find critical points: where two constraints are equal
            # This is complex, so use a practical approach: sample + refine
            
            if ai < 1e-15:
                # Try a small positive value
                candidates = [1e-6, 1e-5]
            else:
                # Explore around current value
                span = max(ai * 2.0, 0.01 / n)
                candidates = np.linspace(0.0, ai + span, 20)
                # Add some finer points around the current value
                candidates = np.concatenate([candidates, 
                                              np.linspace(max(0, ai * 0.5), ai * 1.5, 10)])

            best_x = ai
            best_val = np.inf
            for x in candidates:
                x = max(x, 0.0)
                c_new = B + coeff * x + quad * x * x
                val = float(c_new.max())
                if val < best_val:
                    best_val = val
                    best_x = x

            # Refine with binary search around best_x
            lo = max(0.0, best_x * 0.8 if best_x > 0 else 0.0)
            hi = best_x * 1.2 if best_x > 0 else 0.1 / n
            for _ in range(30):
                mid1 = lo + (hi - lo) / 3.0
                mid2 = hi - (hi - lo) / 3.0
                v1 = float((B + coeff * mid1 + quad * mid1 * mid1).max())
                v2 = float((B + coeff * mid2 + quad * mid2 * mid2).max())
                if v1 < v2:
                    hi = mid2
                else:
                    lo = mid1
            best_x = (lo + hi) / 2.0

            # Apply if improvement
            if abs(best_x - ai) > 1e-15:
                a[i] = best_x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = ai
                    continue
                # Check actual score
                c_check = _convolve(a, a)
                sc_check = float(2.0 * n * c_check.max() / (s_new * s_new))
                if sc_check < best_sc - 1e-14:
                    best_sc = sc_check
                    best_a = a.copy()
                else:
                    a[i] = ai  # Revert

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_wave":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 2.0 * t) + 0.2 * np.cos(2.0 * np.pi * 5.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):

    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

    return a_new / s

            a_fast2, sc_fast2 = _newton_refine(a_fast, d_fast2, max_iter=100, tol_active=0.001, min_time=10.0)
            update_best(a_fast2, sc_fast2)

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("perturb_003", 0.03),
            ("perturb_010", 0.10),
            ("perturb_020", 0.20),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            if sc_start < best_score + 0.05:
                d_r = min(deadline - 15, time.monotonic() + 60)
                a_r, sc_r = _projected_gradient(p_start, d_r, max_iter=10000, min_time=10.0)
                update_best(a_r, sc_r)

                if sc_r < best_score + 0.01 and time.monotonic() < deadline - 10:
                    d_r2 = min(deadline - 5, time.monotonic() + 40)
                    a_r2, sc_r2 = _newton_refine(a_r, d_r2, max_iter=100, tol_active=0.0005, min_time=5.0)
                    update_best(a_r2, sc_r2)

    # Phase 6: Structured seeds (different profiles)
    print(f"\n[Phase 6] Structured seeds (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        seeds = ["tapered", "cos_wave", "asym_cos"]
        n_seed = n_parent if n_parent in [4096, 8192] else 8192
        for kind in seeds:
            if time.monotonic() > deadline - 10:
                break
            a_seed = _make_profile(n_seed, kind)
            # Quick optimization
            d_s = min(deadline - 5, time.monotonic() + 30)
            a_s, sc_s = _projected_gradient(a_seed, d_s, max_iter=3000, min_time=5.0)
            update_best(a_s, sc_s)

    # Phase 7: Final polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Final Newton
        d_fin = min(deadline - 2, time.monotonic() + 30)
        a_fin, sc_fin = _newton_refine(best_seq, d_fin, max_iter=100, tol_active=0.0001, min_time=2.0)
        update_best(a_fin, sc_fin)

        # Final coordinate descent
        if time.monotonic() < deadline - 2:
            d_cd = min(deadline - 1, time.monotonic() + 60)
            a_cd, sc_cd = _coordinate_descent(best_seq, d_cd, n_sweeps=3, min_time=1.0)
            update_best(a_cd, sc_cd)

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
