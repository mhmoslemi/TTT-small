"""
Active-set optimization with tilt initialization for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses analytical gradients, simplex projection,
momentum, L-BFGS with power-mean smoothing, and multi-phase search strategy.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_z(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _tilt_init(n, alpha, rng=None):
    """Create a tilted uniform sequence."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * t - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _scurve_init(n, alpha):
    """Create an S-curve shaped sequence."""
    t = np.linspace(0, 1, n)
    # Cubic S-curve
    a = 1.0 + alpha * (3.0 * t**2 - 2.0 * t**3 - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _gaussian_tilt_init(n, alpha, center=0.6):
    """Gaussian-weighted tilt."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.exp(-20.0 * (t - center) ** 2) * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _active_set_gradient(a, active_indices):
    """
    Compute gradient of max_{k in active} c[k] subject to sum(a)=1.
    d(c[k])/d(a[i]) = 2*a[k-i] (with a[j]=0 for j out of range).
    Returns gradient vector (length n) and the current active peak value.
    """
    n = len(a)
    S = len(active_indices)
    # h[i] = (1/S) * sum_{k in S} a[k-i] for valid (k-i) in [0, n-1]
    h = np.zeros(n)
    for k in active_indices:
        # a[k-i] is valid when k-i in [0, n-1], i.e., i in [max(0,k-n+1), min(k, n-1)]
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            h[i_lo:i_hi + 1] += a[i_lo:i_hi + 1][::-1]
    h /= S
    # Gradient of the smoothed objective (with Lagrange multiplier for sum=1)
    grad = 2.0 * h - 2.0 / n
    return grad


def _active_set_gd(a, deadline, max_iter=500000, tol_peak=0.002, eta_init=0.1,
                   momentum_init=0.85, shrink_active_every=50, verbose=True):
    """
    Active-set gradient descent with simplex projection and momentum.
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)
    eta = eta_init
    momentum = momentum_init
    tol = tol_peak
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find active set
        threshold = c_max - tol * c_max
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[idx] = True
            n_active = 1

        active_indices = np.where(active_mask)[0]

        # Compute gradient
        grad = _active_set_gradient(a, active_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Gradient step with momentum (inline Nesterov-style)
        # Simple: a_new = proj_simplex(a - eta * grad)
        a_try = a - eta * grad
        a_try = _proj_simplex(a_try)

        # Check improvement
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < current_sc - 1e-14:
            a = a_try
            # Increaseeta slightly
            eta = min(0.5, eta * 1.02)
            if verbose and it % 100000 == 0:
                print(f"    AS-GD it={it}, sc={sc_try:.8f}, n_act={n_active}, eta={eta:.4f}")
        else:
            # Backtrack
            eta *= 0.5
            if eta < 1e-6:
                # No step possible, tighten active set
                tol *= 0.7
                eta = eta_init * 0.1
                if tol < 1e-6:
                    break

        # Periodically tighten active set
        if no_improve % shrink_active_every == 0 and no_improve > 0:
            tol = max(1e-5, tol * 0.8)
            momentum = max(0.5, momentum * 0.99)

    return best_a, _score(best_a)

    M = (c_p.sum()) ** (1.0 / p)
    return 2.0 * n * M

    # = a_j * (grad_f_j - dot(grad_f, a))
    
    # Hmm that gives a diagonal result which doesn't seem right. Let me verify with a simple case.
    # Actually I think the correct formula is:
    # grad_z = a * (grad_a - a * np.dot(a, grad_a) / np.dot(a, a))
    # No, since sum(a) = 1, dot(a,a) = sum(a^2) ≠ 1.
    
    # Let me just use the correct formula:
    # grad_z_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    # Wait no. Let me be very explicit.
    # z and a are related by a_i = exp(z_i) / sum_j exp(z_j)
    # da_i/dz_j = a_i * delta_ij - a_i * a_j = a_i (delta_ij - a_j)
    # dL/dz_j = sum_i (dL/da_i) (da_i/dz_j) = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
    # = a_j * (grad_a_j - dot(grad_a, a))
    
    # So grad_z = a * (grad_a - a * dot(grad_a, a))
    # Wait: a_j * (grad_a_j - dot(grad_a, a)) means:
    # grad_z = a * (grad_a - a * dot(grad_a, a))... no
    # grad_z_j = a_j * grad_a_j - a_j * dot(grad_a, a)
    # = a_j * (grad_a_j - dot(grad_a, a))
    # In vector form: grad_z = a * (grad_a - a * dot(grad_a, a))... 
    # No! a_j * (grad_a_j - dot(grad_a,a)) is element-wise:
    # grad_z = a * (grad_a - a * scalar) where scalar = dot(grad_a, a)
    # = a * grad_a - a * a * dot(grad_a, a)
    # Hmm that doesn't look standard. Let me just verify numerically later.
    # For now, use the simpler approach: just do optimization in a-space directly.
    
    # Actually, let me just return the a-space gradient and do optimization in a-space.
    return grad_a, a, c


def _power_mean_lbfgs(a_init, deadline, p=20, min_time_remaining=30):
    """L-BFGS optimization using power mean smoothing in a-space (via log parameterization)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    def obj(zv):
        a = _softmax_z(zv)
        c = _full_convolve(a, a)
        c_pos = np.maximum(c, 1e-30)
        M = (c_pos ** p).sum() ** (1.0 / p)
        return 2.0 * n * M

    def grad(zv):
        a = _softmax_z(zv)
        c = _full_convolve(a, a)
        c_pos = np.maximum(c, 1e-30)
        c_p = c_pos ** p
        M_p = c_p.sum()
        w = c_pos ** (p - 1)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        coef = 2.0 * n * M_p ** (1.0 / p - 1.0)
        grad_a = coef * h
        # Convert to z-space
        # grad_z_j = a_j * (grad_a_j - dot(grad_a, a))
        scalar = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - a * scalar)
        return grad_z

    current_sc = _score(a_init)
    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 5:
        return a_init, current_sc

    max_iter = min(3000, max(20, int(remaining / 0.05)))
    try:
        result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxcor': 50})
        a_new = _softmax_z(result.x)
        sc_new = _score(a_new)
        if sc_new < current_sc:
            return a_new, sc_new
    except Exception:
        pass
    return a_init, current_sc


def _logsumexp_lbfgs(a_init, deadline, beta_init=None, min_time_remaining=20):
    """L-BFGS with log-sum-exp smoothing, multi-beta annealing."""
    n = len(a_init)
    if beta_init is None:
        c = _full_convolve(a_init, a_init)
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        beta_init = max(gap, c_sorted[0] * 0.005)

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    betas = [beta_init, beta_init * 0.3, beta_init * 0.1, beta_init * 0.03, beta_init * 0.01]
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        def obj_s(zv):
            a = _softmax_z(zv)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))

        def grad_s(zv):
            a = _softmax_z(zv)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad_a = 4.0 * n * a * (h - W)
            scalar = float(np.dot(grad_a, a))
            return a * (grad_a - a * scalar)

        max_iter = min(2000, max(20, int(remaining / 0.05)))
        try:
            result = minimize(obj_s, z, jac=grad_s, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15,
                                       'gtol': 1e-13, 'maxcor': 50})
            a_new = _softmax_z(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, _score(current_a)


def _shrinkage(a, deadline):
    """Zero out smallest elements, re-optimize."""
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    idx_sorted = np.argsort(a)
    fractions = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[idx_sorted[:n_zero]] = 0.0
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_try.copy()
            a = a_try
            idx_sorted = np.argsort(a)
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new
