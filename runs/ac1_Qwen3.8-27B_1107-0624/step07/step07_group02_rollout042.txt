Newton-CG Active Set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS for bulk optimization, then a structured Newton-CG method exploiting

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

"""
Newton-CG Active Set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing for bulk optimization, then exploits
the Hankel structure of the active-set Hessian via conjugate gradient for
quadratically-convergent final polishing. Also employs mirror descent
(exponential weights) as an alternative refinement path.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

    c_cur = _full_convolve(a, a)
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c_sorted[0] * 0.001:
        gap = c_sorted[0] * 0.001
    base_beta = max(gap, c_sorted[0] * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002])

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 8))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxcor': 100, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_sc = sc_new
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return a


def _active_hessian_matvec(v, k_lo, k_hi, n):
    """Compute H_A @ v for contiguous active set [k_lo, k_hi].
    H_A[i,j] = 2 if i+j in [k_lo,k_hi] and i!=j; 1 if i=j, 2i in [k_lo,k_hi]; 0 else.
    """
    result = np.zeros(n)
    if k_hi < 0 or k_lo >= 2 * n - 1:
        return result
    k_lo = max(0, k_lo)
    k_hi = min(2 * n - 2, k_hi)
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(v)
    for i in range(n):
        j_lo = max(0, k_lo - i)
        j_hi = min(n - 1, k_hi - i)
        if j_lo > j_hi:
            continue
        s = prefix[j_hi + 1] - prefix[j_lo]
        if j_lo <= i <= j_hi:
            s -= v[i]
        result[i] = 2.0 * s
        if k_lo <= 2 * i <= k_hi:
            result[i] += v[i]
    return result


def _active_gradient(b, k_lo, k_hi, n):
    """Gradient of (1/|A|)*sum_{k in A} c_k, for contiguous A."""
    n_active = k_hi - k_lo + 1
    if n_active <= 0:
        return np.zeros(n)
    grad = np.zeros(n)
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(b)
    k_lo = max(0, k_lo)
    k_hi = min(2 * n - 2, k_hi)
    for i in range(n):
        j_lo = max(0, k_lo - i)
        j_hi = min(n - 1, k_hi - i)
        if j_lo > j_hi:
            continue
        s = 2.0 * (prefix[j_hi + 1] - prefix[j_lo])
        if k_lo <= 2 * i <= k_hi:
            s -= b[i]
        grad[i] = s / n_active
    return grad


def _newton_cg_polish(b_init, deadline, max_newton_iters=500):
    """Newton-CG active set method exploiting Hankel Hessian structure."""
    n = len(b_init)
    a = b_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    tol_frac = 0.01
    reg = 1e-6
    no_improve = 0

    for it in range(max_newton_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            k_lo = int(np.argmax(c))
            k_hi = k_lo
        else:
            k_lo = int(active_idx[0])
            k_hi = int(active_idx[-1])

        g = _active_gradient(a, k_lo, k_hi, n)
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # CG to solve (H + reg*I) d = -g
        r = -g.copy()
        p = r.copy()
        r0 = np.dot(r, r)
        d = np.zeros(n)
        max_cg = 200
        for _ in range(max_cg):
            Hp = _active_hessian_matvec(p, k_lo, k_hi, n)
            Ap = Hp + reg * p
            pAp = np.dot(p, Ap)
            if abs(pAp) < 1e-30:
                break
            alpha_cg = r0 / pAp
            d += alpha_cg * p
            r -= alpha_cg * Ap
            r_new = np.dot(r, r)
            if np.sqrt(r_new) < 1e-10 * np.sqrt(max(r0, 1e-30)):
                break
            p = r + (r_new / max(r0, 1e-30)) * p
            r0 = r_new

        d = d - np.mean(d)

        # Check descent direction
        gd = np.dot(g, d)
        if gd >= 0:
            # Fall back to steepest descent
            d = -g
            d = d - np.mean(d)
            gd = np.dot(g, d)
            if gd >= 0:
                break

        # Line search with nonnegativity constraint
        max_alpha = 1e10
        pos_mask = d < 0
        if np.any(pos_mask):
            max_alpha = float(np.min(a[pos_mask] / (-d[pos_mask])))
            max_alpha *= 0.999

        alpha = min(1.0 / (gnorm + 1e-15), max_alpha)
        improved = False
        for _ in range(50):
            if alpha < 1e-20:
                break
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Increase regularization
            reg *= 10.0
            if reg > 1.0:
                # Fall back to plain gradient step
                d = -g
                d = d - np.mean(d)
                alpha = min(0.01 / (gnorm + 1e-15), max_alpha)
                a_try = a + alpha * d
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a = a_try / s_try
                    c_try = _full_convolve(a, a)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-13:
                        improved = True
            if not improved:
                no_improve += 5
                tol_frac = max(0.0001, tol_frac * 0.8)
                if no_improve > 100:
                    a = best_a.copy()
                    reg = 1e-6
                    tol_frac = 0.005
                    no_improve = 0
                    if tol_frac < 0.0001:
                        break
        else:
            reg = max(1e-10, reg * 0.8)
            if no_improve == 0 and tol_frac > 0.0001:
                tol_frac = max(0.0001, tol_frac * 0.95)

    return best_a, _score(best_a)


def _mirror_descent_polish(b_init, deadline, lr_init=0.05):
    """Exponential weights / mirror descent on the simplex."""
    n = len(b_init)
    a = b_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Gradient of smoothed objective
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Multiplicative update: a_i <- a_i * exp(-lr * g_i), then normalize
        step = lr * g / (gnorm + 1e-30)
        step = np.clip(step, -5.0, 5.0)
        a_new = a * np.exp(-step)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-13:
            a = a_new
            lr = min(lr_init * 2, lr * 1.05)
        else:
            lr *= 0.5
            no_improve += 1
            if no_improve > 50:
                a_log = np.log(np.maximum(a, 1e-30))
                a_log = a_log - a_log.mean()
                a = _softmax(a_log)
                lr = lr_init * 0.5
                no_improve = 0
                if lr < lr_init * 0.01:
                    break

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tilted":
        t = np.linspace(-1, 1, n)
        a = 1.0 + 0.3 * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tilted2":
        t = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * t + 0.1 * t * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skew_gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.4) ** 2) + 0.1
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS -> Newton-CG Active Set -> Mirror Descent.
    Exploits the Hankel structure of the convolution Hessian for fast final convergence.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "tilted")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tilted n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main optimization ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 400, t_start + 1000)
    a_main = _run_lbfgs(parent, lbfgs_deadline, min_time_remaining=300.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  L-BFGS: {best_score:.10f} (n={len(best_seq)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Newton-CG Active Set ===
    print(f"\n[P2] Newton-CG Active Set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        nc_deadline = min(deadline - 200, time.monotonic() + 300)
        a_nc, sc_nc = _newton_cg_polish(best_seq, nc_deadline, max_newton_iters=1000)
        if sc_nc < best_score:
            best_seq = a_nc.copy()
            best_score = sc_nc
            print(f"  Newton-CG: {best_score:.10f} (n={len(best_seq)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Mirror descent ===
    print(f"\n[P3] Mirror Descent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        md_deadline = min(deadline - 150, time.monotonic() + 120)
        a_md, sc_md = _mirror_descent_polish(best_seq, md_deadline, lr_init=0.03)
        if sc_md < best_score:
            best_seq = a_md.copy()
            best_score = sc_md
            print(f"  MirrorDescent: {best_score:.10f} (n={len(best_seq)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second Newton-CG pass (tighter) ===
    print(f"\n[P4] Newton-CG pass 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        nc_deadline2 = min(deadline - 80, time.monotonic() + 100)
        a_nc2, sc_nc2 = _newton_cg_polish(best_seq, nc_deadline2, max_newton_iters=500)
        if sc_nc2 < best_score:
            best_seq = a_nc2.copy()
            best_score = sc_nc2
            print(f"  Newton-CG2: {best_score:.10f} (n={len(best_seq)}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240, 12288]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick L-BFGS
        lb_deadline = min(deadline - 50, time.monotonic() + 80)
        a_opt = _run_lbfgs(a_init, lb_deadline, min_time_remaining=20.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  Multi-n ({n_try}): {best_score:.10f} t={time.monotonic()-t_start:.1f}s")
        # Quick Newton polish
        if time.monotonic() < deadline - 40:
            nc_d = min(deadline - 35, time.monotonic() + 30)
            a_nc3, sc_nc3 = _newton_cg_polish(a_opt, nc_d, max_newton_iters=200)
            if sc_nc3 < best_score:
                best_seq = a_nc3.copy()
                best_score = sc_nc3
                print(f"  Multi-n+NC ({n_try}): {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restarts = [
            _perturb(best_seq, 0.02, rng),
            _perturb(best_seq, 0.05, rng),
            _make_profile(n_best, "tilted2"),
            _make_profile(n_best, "skew_gauss"),
        ]
        for i, a_r in enumerate(restarts):
            if time.monotonic() > deadline - 20:
                break
            lb_d = min(deadline - 15, time.monotonic() + 40)
            a_opt = _run_lbfgs(a_r, lb_d, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  Restart {i}: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine Newton ===
    print(f"\n[P7] Final Newton polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_cg_polish(best_seq, deadline - 3, max_newton_iters=200)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  Final Newton: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None and len(best_seq) > 0:
        final = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
