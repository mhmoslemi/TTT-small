"""
Optimized sequence search combining L-BFGS with log-sum-exp smoothing,
Nesterov-accelerated active-set gradient descent, spectral shaping,
peak shaving refinement, and multi-n exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)

        return a_init

    base_beta = max(gap, max_c * 0.002)

def _nesterov_active_set(a_init, deadline, tol_frac_start=0.005,

    lr = lr_init
    tol_frac = tol_frac_start

    max_stall = 600

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]

        a_try = a + v  # Nesterov: evaluate at lookahead point

        # Project: non-negative and sum-normalized

        s_try = a_try.sum()
        if s_try < 1e-15:

        a_try = a_try / s_try

            a = a_try
            # Occasionally boost momentum

            # Reduce learning rate and reset momentum sometimes

                no_improve = 0

        # The time-domain convolution is the inverse DFT of power
        # We want to make it flatter. In frequency domain, this means
        # making power more "spread out" (less concentrated at DC)
        # But A[0] is fixed (= sum(a) = 1)

        # Strategy: apply a gentle equalization to the power spectrum
        # Reduce high-power bins, boost low-power bins
        power_non_dc = power[1:]
        mean_power = np.mean(power_non_dc)

        # Gentle compression: power_new = power^alpha for alpha < 1
        alpha = 0.7 + 0.3 * (it / n_iter)  # start aggressive, end gentle
        A_magnitude = np.sqrt(power)
        A_magnitude_new = A_magnitude.copy()
        A_magnitude_new[1:] = np.sign(A_magnitude_new[1:]) * (
            A_magnitude_new[1:] ** (0.95 + 0.05 * it / n_iter)
        )

        # Reconstruct with modified magnitude, keeping phase
        A_new = A.copy()
        phase = np.angle(A)
        A_new = A_magnitude_new * np.exp(1j * phase)

        # Ensure A[0] is real and positive (DC)
        A_new[0] = float(A[0].real)

        # Inverse DFT
        a_rec = np.fft.irfft(A_new, N)[:n]

        # Clamp to non-negative
        a_rec = np.maximum(a_rec, 0)
        s_rec = a_rec.sum()
        if s_rec < 1e-15:
            break
        a_rec = a_rec / s_rec

        sc_rec = _score(a_rec)
        if sc_rec < best_sc - 1e-15:
            best_sc = sc_rec
            best_a = a_rec.copy()
            a = a_rec
        else:
            # If no improvement, try a smaller perturbation
            blend = 0.1
            a_blend = (1 - blend) * a + blend * a_rec
            a_blend = np.maximum(a_blend, 0)
            s_blend = a_blend.sum()
            if s_blend > 1e-15:
                a_blend = a_blend / s_blend
                sc_blend = _score(a_blend)
                if sc_blend < best_sc:
                    best_sc = sc_blend
                    best_a = a_blend.copy()
                    a = a_blend
                else:
                    break

    return best_a, _score(best_a)


def _coord_descent_refine(a_init, deadline, n_sweeps=5):
    """Coordinate descent: optimize each element individually."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Precompute: for each i, the contribution of a[i] to each c[k]
    # c[k] = sum_j a[j] * a[k-j]
    # d c[k] / d a[i] = a[k-i] + a[i-k] (for valid indices, with special case k=i)
    # Actually: c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # d c[k] / d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n and i-k != k-i, i.e., k != 2i)
    # Wait, more carefully: c[k] = sum_j a[j] * a[k-j]
    # If j = i: term is a[i] * a[k-i]
    # If k-j = i, i.e., j = k-i: term is a[k-i] * a[i]
    # These are the same term if i = k-i, i.e., k = 2i
    # So: d c[k] / d a[i] = a[k-i] + a[i-k] for k != 2i
    # And: d c[2i] / d a[i] = 2 * a[i] (the term appears twice) ... no
    # Actually c[k] = sum_j a[j]*a[k-j]. The terms involving a[i] are:
    #   j=i: a[i]*a[k-i] (requires 0<=k-i<n)
    #   k-j=i => j=k-i: a[k-i]*a[i] (requires 0<=k-i<n)
    # If i != k-i (i.e., k != 2i), these are two separate terms, each = a[i]*a[k-i]
    # If i = k-i (i.e., k = 2i), it's one term = a[i]^2
    # So: d c[k]/d a[i] = 2*a[k-i] if 0<=k-i<n and k != 2i
    #                    = 2*a[i] if k = 2i
    #                    = a[k-i] if 0<=k-i<n and k = 2i (which means a[i])
    # Hmm, let me just compute it numerically.

    # For efficiency, the gradient of c[k] w.r.t. a[i] is:
    # g[i] = sum_k w[k] * (a[k-i] + a[i-k]) where we only count valid indices
    # This is equivalent to: g = convolve(w, a)[n-1:2n-1] + convolve(w, a_rev)[n-1:2n-1]
    # (approximately, ignoring boundary effects)

    # For coordinate descent, we need: given all a[j] for j != i, what's the optimal a[i]?
    # score = 2*n*max_k(c[k]) / (sum a)^2
    # c[k] = c_base[k] + a[i]*a[k-i] + a[k-i]*a[i] - correction
    # This is complex. Let's just use a simple local search.

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() > deadline:
                break
            # Try a few perturbations of a[i]
            base_val = a[i]
            if base_val < 1e-10:
                continue
            for mult in [0.9, 0.8, 1.1, 1.2]:
                a[i] = base_val * mult
                s_try = a.sum()
                if s_try < 1e-15:
                    a[i] = base_val
                    continue
                a_normalized = a / s_try
                sc_try = _score(a_normalized)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_normalized.copy()
                    a = a_normalized
                    break
            a[i] = base_val  # restore if no improvement

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization:
    1. Load parent and score it
    2. L-BFGS with aggressive beta annealing
    3. Nesterov active-set GD
    4. Spectral shaping
    5. Peak shaving
    6. Multi-n exploration
    7. Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}", flush=True)
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}", flush=True)

    # === Phase 1: L-BFGS main optimization ===
    print(f"\n[P1] L-BFGS refined (t={time.monotonic()-t_start:.1f}s)", flush=True)
    a_main = _run_lbfgs_refined(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 2: Nesterov active-set ===
    print(f"\n[P2] Nesterov active-set (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 250)
        a_pol, sc_pol = _nesterov_active_set(best_seq, pol_deadline,
                                             tol_frac_start=0.004, tol_frac_end=0.00003,
                                             lr_init=0.025, momentum_init=0.4)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [nesterov] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 3: L-BFGS again with smaller beta ===
    print(f"\n[P3] L-BFGS second pass (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 300:
        a_main2 = _run_lbfgs_refined(best_seq, min(deadline - 300, time.monotonic() + 150),
                                      min_time_remaining=100.0)
        sc_main2 = _score(a_main2)
        if sc_main2 < best_score:
            best_seq = a_main2.copy()
            best_score = sc_main2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_2nd] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 4: Spectral shaping ===
    print(f"\n[P4] Spectral shaping (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 250:
        a_spec, sc_spec = _spectral_shaping(best_seq, min(deadline - 250, time.monotonic() + 80), n_iter=50)
        if sc_spec < best_score:
            best_seq = a_spec.copy()
            best_score = sc_spec
            print(f"  NEW BEST: {best_score:.10f} [spectral] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 5: Peak shaving ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 200:
        a_peak, sc_peak = _peak_shaving(best_seq, min(deadline - 200, time.monotonic() + 60), top_k=20)
        if sc_peak < best_score:
            best_seq = a_peak.copy()
            best_score = sc_peak
            print(f"  NEW BEST: {best_score:.10f} [peak_shave] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_base = len(best_seq)
    n_candidates = sorted(set([n_base // 2, n_base, n_base * 3 // 2, n_base * 2,
                                n_base - 256, n_base + 256, n_base + 512]))
    n_candidates = [n for n in n_candidates if 1024 <= n <= 32768]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        if n_try == n_base:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}", flush=True)
        a_opt = _run_lbfgs_refined(a_init, deadline, min_time_remaining=150.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s", flush=True)
        else:
            print(f"  n={n_try}: best score={sc_opt:.10f} (not better)", flush=True)

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("pert10", _perturb(best_seq, 0.10, rng)),
            ("cos2", _make_profile(n_best, "cos2")),
            ("raised", _make_profile(n_best, "raised_cos")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 40:
                break
            a_opt = _run_lbfgs_refined(a_r, deadline, min_time_remaining=20.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final ultra-fine polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 5:
        # Very tight Nesterov
        a_fin1, sc_fin1 = _nesterov_active_set(best_seq, deadline - 3,
                                               tol_frac_start=0.001, tol_frac_end=0.00001,
                                               lr_init=0.008, momentum_init=0.5)
        if sc_fin1 < best_score:
            best_seq = a_fin1.copy()
            best_score = sc_fin1
            print(f"  NEW BEST: {best_score:.10f} [final_nesterov] t={time.monotonic()-t_start:.1f}s", flush=True)

        # Tiny L-BFGS with smallest beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 1e-5
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-16, 'maxcor': 100, 'maxls': 50})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s", flush=True)
            except Exception:
                pass

        # One more Nesterov pass
        remaining = deadline - 2 - time.monotonic()
        if remaining > 10:
            a_fin3, sc_fin3 = _nesterov_active_set(best_seq, deadline - 1,
                                                   tol_frac_start=0.0005, tol_frac_end=0.000005,
                                                   lr_init=0.004, momentum_init=0.6)
            if sc_fin3 < best_score:
                best_seq = a_fin3.copy()
                best_score = sc_fin3
                print(f"  NEW BEST: {best_score:.10f} [final_nesterov2] t={time.monotonic()-t_start:.1f}s", flush=True)

    # Final validation
    print(f"\n{'='*60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
