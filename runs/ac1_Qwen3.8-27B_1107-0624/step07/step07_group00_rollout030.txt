The MWU approach:
- Initialize g = ones(n) / n
- Initialize dual prices y = ones(K) / K
- For each iteration:

  - Update g[j] = g[j] * exp(-eta * (A^T y)[j])
  - Normalize: g = g / sum(g)
  - Update y[k] = y[k] * exp(eta * (A g)[k] - eta * M) (or similar)
  - Normalize y

This is O(K * n) per iteration, which is very fast for K=50, n=8192.

- Normalize g_hat to sum = 1
- Update f = (1-t) * f + t * g_hat

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines MWU-LP guided direction finding, Nesterov mirror descent with 
exact active-set subgradients, coordinate descent polishing, and multi-scale
exploration. Key innovation: uses the LP direction (solved via multiplicative
weight updates on the reduced active constraint set) to guide the iterates,
combined with Nesterov-accelerated mirror descent for fast local convergence.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _subgradient_active(a, c, top_k=5, decay=0.85):
    """Compute weighted subgradient at top-k active lags of conv(a,a)."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    weights = decay ** np.arange(top_k)
    w_sum = weights.sum()
    weights = weights / w_sum
    grad = np.zeros(n)
    for i in range(top_k):
        k = k_indices[i]
        w = weights[i]
        if k < n:
            grad[:k + 1] += w * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] += w * a[n - 1 : n - 1 - (n - 1 - start) : -1]
    return 2.0 * grad


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _nesterov_mirror_descent(a, deadline, max_iter=200000, eta0=0.5, eta_decay=0.999,
                             top_k=5, decay=0.85, eval_interval=200,
                             use_backtrack=True, patience_limit=8):
    """Nesterov-accelerated mirror descent with exact active-set subgradient."""
    n = len(a)
    a = a.copy() / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    t = 0
    t_k = 1.0
    eta = eta0
    patience = 0
    y = a.copy()

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        a_prev = a.copy()
        t_k = t_new

        # Compute subgradient at y
        c_y = _fft_convolve(y, y)
        g = _subgradient_active(y, c_y, top_k, decay)

        # Mirror descent update in log space
        z = z - eta * g
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        a_new = _softmax_stable(z)

        # Optional backtracking
        if use_backtrack and t % 100 == 0:
            sc_cand = _score_norm(a_new)
            if sc_cand > best_score * 1.5:
                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                eta *= 0.5
                a_new = a.copy()
            else:
                # Try to increase eta
                if sc_cand < best_score:
                    eta = min(eta * 1.1, 2.0)
        else:
            a_new = a_new

        a = a_new
        eta *= eta_decay
        eta = max(eta, eta0 * 0.01)

        if t % eval_interval == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_limit:
                # Reset to best and reduce eta
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                a = _softmax_stable(z)
                a_prev = a.copy()
                y = a.copy()
                t_k = 1.0
                eta = max(eta * 0.5, eta0 * 0.005)
                patience = 0

    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _mwu_lp_direction(f, M, K=50, T_mwu=200, eta_mwu=0.3):
    """
    Compute LP direction using MWU on the reduced problem.
    LP: max sum(g) s.t. (f*g)[k] <= M for k in top-K lags, g >= 0.
    Returns normalized g (sum=1).
    """
    n = len(f)
    c = _fft_convolve(f, f)
    active_lags = np.argsort(c)[::-1][:K]

    # Build constraint matrix A (K x n): A[i, j] = f[active_lags[i] - j] if valid
    # (f*g)[k] = sum_j f[j] * g[k-j], so the constraint row for lag k is:
    # sum_j f[k-j] * g[j] <= M, i.e., the row is [f[k], f[k-1], ..., f[k-n+1]] (with 0-padding)
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = f[idx]

    # MWU for: max 1^T g s.t. A g <= M*1, g >= 0
    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    # MWU: maintain primal g and dual y
    g = np.ones(n) / n
    y = np.ones(K) / K
    eta = eta_mwu

    for t in range(T_mwu):
        # Update primal: g_j <- g_j * exp(-eta * (A^T y)_j)
        Ay = A.T @ y  # (A^T y)[j] = sum_i A[i,j] * y[i]
        log_g = np.log(np.maximum(g, 1e-300)) - eta * Ay
        log_g = log_g - log_g.max()
        g = np.exp(log_g)
        s = g.sum()
        if s < 1e-300:
            g = np.ones(n) / n
        else:
            g = g / s

        # Update dual: y_i <- y_i * exp(eta * ((A g)_i - M))
        Ag = A @ g
        log_y = np.log(np.maximum(y, 1e-300)) + eta * (Ag - M)
        log_y = log_y - log_y.max()
        y = np.exp(log_y)
        s2 = y.sum()
        if s2 < 1e-300:
            y = np.ones(K) / K
        else:
            y = y / s2

    return g


def _mwu_lp_iteration(a, deadline, K=50, T_mwu=100, mixing=0.3, max_outer=50,
                      eta_mwu=0.2):
    """
    Iterative LP-guided optimization:
    1. Solve LP (via MWU) to get direction g
    2. Update: a = (1-mixing)*a + mixing*g
    """
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        g = _mwu_lp_direction(a, M, K=K, T_mwu=T_mwu, eta_mwu=eta_mwu)
        g = g / g.sum()
        a_new = (1.0 - mixing) * a + mixing * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        # Decay mixing slightly
        mixing = max(mixing * 0.98, 0.05)

    return best_a, best_score


def _coordinate_descent_polish(a, deadline, max_rounds=200):
    """Fast coordinate descent: adjust individual elements to reduce max convolution."""
    n = len(a)
    best_a = a.copy() / a.sum()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]

    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for k in active:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue
            for j in js:
                for scale in [1.2, 1.1, 0.9, 0.8, 0.7, 1.5, 0.5]:
                    a_cand = best_a.copy()
                    old_val = a_cand[j]
                    a_cand[j] = old_val * scale
                    if a_cand[j] < 0:
                        continue
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand = a_cand / s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _adam_log_space(a, deadline, T_start=0.003, T_end=0.0001,
                    lr_start=0.03, lr_end=0.002, max_iter=200000,
                    top_k=3, decay=0.7, eval_interval=200):
    """Adam optimizer in log-space with smoothed gradient (fallback method)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)

        if T < T_start * 0.02:
            g = _subgradient_active(a_cur, c, top_k, decay)
        else:
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s
            p_rev = p[::-1]
            conv_full = _fft_convolve(a_cur, p_rev)
            g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]

        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new

        if t % eval_interval == 0:
            a_eval = _softmax_stable(z)
            sc = _score_norm(a_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
