"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: asymmetric tilting, multi-scale active set Newton,
n=16384 exploration, FISTA acceleration, and adaptive beta continuation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        rs = rs_new
        p = (rs_new / rs) * p + r
        if rs < tol * tol * rs0:
            break
    return x

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _fista_polish(a_init, deadline, max_steps=3000):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))

    z_old = z.copy()

        if time.monotonic() > deadline:
            break
        beta = base_beta * (0.5 ** beta_idx)
        lr = 0.5 / max(_smograd(z, beta, n).max(), 1e-10)
        lr = min(lr, 0.1)

            z_new = np.clip(z_new, -60.0, 60.0)
            z_new = z_new - z_new.mean()

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_ik * t_ik))
            zk = z_new + ((t_ik - 1.0) / t_new) * (z_new - z_old)
            zk = np.clip(zk, -60.0, 60.0)
            zk = zk - zk.mean()

            z_old = z
            z = zk
            t_ik = t_new

            if i % 50 == 0:
                a_check = _softmax(z)
                sc = _score(a_check)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp_up":
        t = np.linspace(0, 1, n)
        a = (0.1 + 0.9 * t) ** 1.5
        return a / a.sum()
    elif kind == "ramp_down":
        t = np.linspace(0, 1, n)
        a = (0.1 + 0.9 * (1 - t)) ** 1.5
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_peak":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.35) ** 2) + 0.05
        return a / a.sum()
    elif kind == "s_skew":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10.0 * (t - 0.6)))
        a = a + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _tilt(a, alpha):
    """Apply exponential tilt to sequence."""
    n = len(a)
    t = np.linspace(0, 1, n)
    tilted = a * np.exp(alpha * (t - 0.5))
    tilted = np.maximum(tilted, 0)
    s = tilted.sum()
    if s < 1e-15:
        return a.copy()
    return tilted / s


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent_polish(a_init, deadline, max_sweeps=5):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, max(1, n // 2000)):
            if time.monotonic() > deadline:
                break
            # Compute convolution without a[i]
            a_zero = a.copy()
            old_val = a_zero[i]
            a_zero[i] = 0.0
            if old_val < 1e-30:
                continue
            c_base = _full_convolve(a_zero, a_zero)
            # conv(k) = c_base[k] + t * (a_zero[k-i] + a_zero[i-k]) for appropriate k
            # Actually: conv_new[k] = c_base[k] + t * a_zero[k-i] + t * a_zero[i-(k-i)]... 
            # More carefully: conv_new[k] = sum_j a_new[j]*a_new[k-j]
            # = c_base[k] + t*a_zero[k-i] + t*a_zero[i-(k-i)] ... no
            # = c_base[k] + t*a_zero[k-i] (from j=i) + t*a_zero[k-i] (from j=k-i, if k-i != i)
            # Actually if k-i != i: conv_new[k] = c_base[k] + t*a_zero[k-i] + t*a_zero[k-i]
            # Wait no. a_new[i] = t, a_new[j] = a_zero[j] for j != i.
            # conv_new[k] = sum_{j != i} a_zero[j] * a_new[k-j] + a_new[i] * a_new[k-i]
            # = sum_{j != i} a_zero[j] * a_new[k-j] + t * a_new[k-i]
            # For k-j != i: a_new[k-j] = a_zero[k-j]
            # For k-j == i: a_new[k-j] = t
            # So conv_new[k] = c_base[k] + t * (a_zero[k-i] + a_zero[i-(k-i)])
            # Wait, let me be more careful.
            # c_base[k] = sum_{j != i} a_zero[j] * a_zero[k-j] (missing terms involving a[i])
            # The full conv with a_new:
            # conv_new[k] = sum_j a_new[j] * a_new[k-j]
            # = sum_{j != i} a_zero[j] * a_new[k-j] + t * a_new[k-i]
            # For j != i and k-j != i: a_new[j]*a_new[k-j] = a_zero[j]*a_zero[k-j]
            # For j != i and k-j == i (i.e., j = k-i): a_new[j]*a_new[k-j] = a_zero[k-i]*t
            # For j == i and k-i != i: a_new[i]*a_new[k-i] = t*a_zero[k-i]
            # For j == i and k-i == i (i.e., k = 2i): a_new[i]*a_new[k-i] = t*t
            #
            # So conv_new[k] = c_base[k] + t*a_zero[k-i] + t*a_zero[k-i] + (t^2 - t*a_zero[i]) if k=2i
            # Hmm this is getting complicated. Let me just evaluate directly.
            
            # Simpler approach: try a few values of t and pick the best
            # t ranges from 0 to 2*old_val
            best_t = old_val
            best_cmax = c_max
            for frac in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.7, 2.0]:
                t_try = old_val * frac
                a_try = a.copy()
                a_try[i] = t_try
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                cm = float(c_try.max())
                if cm < best_cmax:
                    best_cmax = cm
                    best_t = t_try

            if best_t != old_val:
                a[i] = best_t
                a = a / a.sum()
                improved = True
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search with tilting, FISTA, multi-scale Newton, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Tilted L-BFGS runs
    print(f"\n[Phase 1] Tilted L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    tilt_values = [0.0, 0.5, -0.5, 1.0, -1.0, 2.0, -2.0, 3.0]
    tilt_budget = (deadline - t_start) * 0.25
    tilt_deadline = t_start + tilt_budget

    for alpha in tilt_values:
        if time.monotonic() > tilt_deadline - 10:
            break
        a_tilted = _tilt(parent, alpha)
        a_opt = _run_lbfgs(a_tilted, tilt_deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"tilt_{alpha}")

    # Phase 2: Newton polish on best
    print(f"\n[Phase 2] Newton polish (t={time.monotonic() - t_start:.1f}s}")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a2, sc2 = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 300))
        update_best(a2, sc2, "newton_1")

    # Phase 3: FISTA polish
    print(f"\n[Phase 3] FISTA polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a3, sc3 = _fista_polish(best_seq, min(deadline - 300, time.monotonic() + 150))
        update_best(a3, sc3, "fista_1")

    # Phase 4: Another Newton round
    print(f"\n[Phase 4] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a4, sc4 = _newton_active_polish(best_seq, min(deadline - 250, time.monotonic() + 120))
        update_best(a4, sc4, "newton_2")

    # Phase 5: Coordinate descent polish
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _coordinate_descent_polish(best_seq, min(deadline - 200, time.monotonic() + 100))
        update_best(a5, sc5, "coord_descent")

    # Phase 6: Multi-n exploration (including 16384)
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 8192, 4096, 2048]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent and n_try == 8192 and best_score < 1.51:
            # Already well-optimized at this size
            continue
        if time.monotonic() > deadline - 60:
            break
        a_init = _resample(parent, n_try)
        # Also try tilted version
        a_tilted = _tilt(a_init, 0.5)
        
        # Run L-BFGS on both
        remaining = deadline - time.monotonic() - 50
        if remaining > 30:
            a_opt1 = _run_lbfgs(a_init, t_start + t_start * 0 + time.monotonic() + remaining / 2, min_time_remaining=20.0)
            sc1 = _score(a_opt1)
            update_best(a_opt1, sc1, f"n_{n_try}")
        
        if time.monotonic() < deadline - 30:
            a_opt2 = _run_lbfgs(a_tilted, deadline, min_time_remaining=20.0)
            sc2 = _score(a_opt2)
            update_best(a_opt2, sc2, f"n_{n_try}_tilt")

        # Newton polish on best of the two
        if time.monotonic() < deadline - 20:
            a_np, sc_np = _newton_active_polish(best_seq if best_seq is not None and len(best_seq) == n_try else (a_opt1 if sc1 < sc2 else a_opt2),
                                                 min(deadline - 10, time.monotonic() + 30), max_rounds=30)
            update_best(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_profiles = [
            ("ramp_up", 0.0),
            ("ramp_down", 0.0),
            ("asym_peak", 0.0),
            ("s_skew", 0.0),
        ]
        for i, (prof, alpha) in enumerate(restart_profiles):
            if time.monotonic() > deadline - 25:
                break
            p_start = _make_profile(len(best_seq), prof)
            if alpha != 0:
                p_start = _tilt(p_start, alpha)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{prof}")
            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 8, time.monotonic() + 20), max_rounds=20)
                update_best(a_rp, sc_rp, f"restart_{prof}_np")

        # Perturbation restarts
        for i in range(3):
            if time.monotonic() > deadline - 15:
                break
            sigma = 0.05 + 0.1 * i
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"perturb_{i}")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=150)
        update_best(a_fin, sc_fin, "final_newton")

        if time.monotonic() < deadline - 2:
            # One more FISTA with very small beta
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 5e-7
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(20, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 50}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
