import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):

    e = np.exp(np.clip(z, -700, 700))

    if s < 1e-300:

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: L-BFGS with log-sum-exp smoothing, minimum-norm subgradient active-set
descent with momentum, adaptive top-k weighting, multi-n exploration, and
iterative peak-shaving refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _grad_c_at_k(a, k):
    """Compute gradient of c[k] = sum_j a[j]*a[k-j] w.r.t. a.
    Result[i] = 2*a[k-i] if 0<=k-i<len(a), else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # Valid i: 0 <= i < n and 0 <= k-i < n
    # i in [max(0, k-n+1), min(n-1, k)]
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        grad[i_start:i_end + 1] = 2.0 * a[k - i_end:k - i_start - 1:-1]
    return grad


def _active_set_grad(a, c, top_k=5, tol_frac=0.001):
    """Compute weighted active-set gradient using top-k peaks.
    Uses geometric decreasing weights for the top peaks.
    """
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)

    # Find peaks above threshold
    threshold = c_max * (1.0 - tol_frac)
    active = c >= threshold
    active_idx = np.where(active)[0]

    if len(active_idx) == 0:
        idx = int(np.argmax(c))
        active_idx = np.array([idx])

    # Sort by c value descending, take top_k
    if len(active_idx) > top_k:
        order = np.argsort(c[active_idx])[::-1][:top_k]
        active_idx = active_idx[order]

    # Geometric weights
    m = len(active_idx)
    weights = np.array([0.85 ** i for i in range(m)])
    weights /= weights.sum()

    # Compute weighted gradient
    grad = np.zeros(n)
    for ii in range(m):
        k = int(active_idx[ii])
        grad += weights[ii] * _grad_c_at_k(a, k)

    return grad


def _min_norm_subgrad(a, c, active_indices):
    """Compute minimum-norm subgradient using QP over the active set."""
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return np.zeros(n)
    if m == 1:
        return _grad_c_at_k(a, int(active_indices[0]))

    # Compute Gram matrix: G[i,j] = grad_c[k_i] . grad_c[k_j]
    # = 4 * sum_p a[k_i - p] * a[k_j - p] for valid p
    G = np.zeros((m, m))
    for ii in range(m):
        for jj in range(ii, m):
            k1 = int(active_indices[ii])
            k2 = int(active_indices[jj])
            # sum_p a[k1-p]*a[k2-p] for p where both indices valid
            p_start = max(0, k1 - n + 1, k2 - n + 1)
            p_end = min(n - 1, k1, k2)
            s = 0.0
            if p_start <= p_end:
                idx1 = k1 - np.arange(p_end + 1)
                idx2 = k2 - np.arange(p_end + 1)
                mask = (idx1 >= 0) & (idx1 < n) & (idx2 >= 0) & (idx2 < n)
                s = np.dot(a[idx1[mask]], a[idx2[mask]])
            G[ii, jj] = 4.0 * s
            G[jj, ii] = G[ii, jj]

    # Add small regularization for numerical stability
    G += 1e-12 * np.eye(m)

    # Solve QP: min lambda^T G lambda s.t. sum(lambda)=1, lambda>=0
    # Use projective gradient for small m
    lam = np.ones(m) / m
    lr_qp = 1.0 / (2.0 * np.linalg.norm(G, ord=2) + 1e-15)
    for _ in range(200):
        grad_qp = 2.0 * G @ lam
        lam = lam - lr_qp * grad_qp
        # Project onto simplex {lambda >= 0, sum(lambda) = 1}
        # Use the standard projection algorithm
        nu = 0.0
        for _ in range(50):
            lam_new = np.maximum(lam - nu, 0.0)
            nu_new = nu + (lam_new.sum() - 1.0) / m
            if abs(nu_new - nu) < 1e-14:
                nu = nu_new
                break
            nu = nu_new
        lam = np.maximum(lam - nu, 0.0)
        lam_sum = lam.sum()
        if lam_sum > 1e-15:
            lam = lam / lam_sum

    # Compute subgradient
    grad = np.zeros(n)
    for ii in range(m):
        k = int(active_indices[ii])
        grad += lam[ii] * _grad_c_at_k(a, k)

    return grad


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, n_betas=None):
    """Run L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    if n_betas is None:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                      0.03, 0.015, 0.0075, 0.004,
                                      0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                      0.00005, 0.00002])
    else:
        betas = n_betas

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(float(beta), n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_descent(a_init, deadline, momentum=0.3, top_k=5,
                        tol_frac_start=0.01, tol_frac_end=0.00005,
                        lr_init=0.03, use_min_norm=False):
    """Active-set gradient descent with momentum and adaptive parameters."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0
    v = np.zeros(n)  # momentum

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        active_idx = np.where(active)[0]

        if len(active_idx) == 0:
            idx = int(np.argmax(c))
            active_idx = np.array([idx])

        # Limit active set size
        if len(active_idx) > 20:
            order = np.argsort(c[active_idx])[::-1][:20]
            active_idx = active_idx[order]
            top_active = active_idx[:min(top_k, len(active_idx))]
        else:
            top_active = active_idx[:min(top_k, len(active_idx))]

        # Compute gradient direction
        if use_min_norm:
            grad = _min_norm_subgrad(a, c, top_active)
        else:
            grad = _active_set_grad(a, c, top_k=min(top_k, len(active_idx)),
                                    tol_frac=tol_frac)

        # Center the gradient (remove mean to preserve sum)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad_t
        v_t = v - np.mean(v)
        vnorm = np.linalg.norm(v_t)

        if vnorm < 1e-15:
            break

        # Line search with backtracking
        alpha = min(lr, 0.5 / (vnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * v_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            v *= 0.5  # dampen momentum on failure
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.2
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init * 2.0, lr * 1.03)

    return best_a, _score(best_a)


def _peak_shave(a_init, deadline, n_shaves=100):
    """Targeted peak shaving: find the max peak and apply a local correction."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    for iteration in range(n_shaves):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Find the peak
        k_star = int(np.argmax(c))

        # Compute the "contribution" of each a[i] to c[k_star]
        # c[k_star] = sum_j a[j]*a[k_star-j]
        # d c[k_star] / d a[i] = 2*a[k_star-i]
        contrib = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            contrib[i_start:i_end + 1] = 2.0 * a[k_star - i_end:k_star - i_start - 1:-1]

        # We want to decrease a where contrib is large, increase where it's small
        # But maintain sum = 1
        contrib_centered = contrib - np.mean(contrib)

        # Take a small step in the -contrib_centered direction
        lr = 0.01 / (np.linalg.norm(contrib_centered) + 1e-15)
        lr = min(lr, 0.005)
        a_try = a - lr * contrib_centered
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        c_try = _full_convolve(a_try, a_try)
        sc_try = 2.0 * n * float(c_try.max())

        if sc_try < current_sc - 1e-13:
            a = a_try
        else:
            # Try smaller step
            lr *= 0.3
            a_try = a - lr * contrib_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-13:
                    a = a_try

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.25) ** 2) + np.exp(-15.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 17.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fourier_rand":
        rng = np.random.default_rng(12345)
        n_freq = max(2, n // 4)
        freqs = rng.integers(1, n // 2, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        amps = rng.uniform(0.05, 0.2, size=n_freq)
        t = np.arange(n)
        a = np.ones(n)
        for f, p, am in zip(freqs, phases, amps):
            a += am * np.cos(2.0 * np.pi * f * t / n + p)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try zeroing out the smallest elements to improve the score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    a_sorted_idx = np.argsort(a)

    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc

        print(f"  P1: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 2: Active-set descent with momentum (coarse) ===
    print(f"\n[P2] Active-set momentum coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        pol_deadline = min(deadline - 600, time.monotonic() + 200)
        a2, sc2 = _active_set_descent(best_seq, pol_deadline,
                                       momentum=0.3, top_k=5,
                                       tol_frac_start=0.01, tol_frac_end=0.001,
                                       lr_init=0.03, use_min_norm=False)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  P2: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 3: L-BFGS with small beta (refinement) ===
    print(f"\n[P3] L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)]
        base_beta = max(gap * 0.5, mb * 0.002)
        betas_refine = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                             0.03, 0.01, 0.005, 0.002, 0.001,
                                             0.0005, 0.0002])
        a3 = _run_lbfgs(best_seq, deadline, min_time_remaining=500.0,
                        n_betas=betas_refine)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  P3: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 4: Active-set descent with momentum (fine) ===
    print(f"\n[P4] Active-set momentum fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        pol_deadline = min(deadline - 400, time.monotonic() + 180)
        a4, sc4 = _active_set_descent(best_seq, pol_deadline,
                                       momentum=0.2, top_k=3,
                                       tol_frac_start=0.003, tol_frac_end=0.0001,
                                       lr_init=0.01, use_min_norm=False)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  P4: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 5: Peak shaving ===
    print(f"\n[P5] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        shave_deadline = min(deadline - 350, time.monotonic() + 80)
        a5, sc5 = _peak_shave(best_seq, shave_deadline, n_shaves=200)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  P5: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 6: Active-set with min-norm subgradient ===
    print(f"\n[P6] Min-norm active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 120)
        a6, sc6 = _active_set_descent(best_seq, pol_deadline,
                                       momentum=0.25, top_k=5,
                                       tol_frac_start=0.002, tol_frac_end=0.00005,
                                       lr_init=0.015, use_min_norm=True)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  P6: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 7: Shrinkage ===
    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a7, sc7 = _shrinkage_optimize(best_seq, deadline - 250)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  P7: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s}")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 200:
                a7b = _run_lbfgs(best_seq, deadline, min_time_remaining=200.0)
                sc7b = _score(a7b)
                if sc7b < best_score:
                    best_seq = a7b.copy()
                    best_score = sc7b
                    print(f"  P7b: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s}")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([target_n // 2, int(target_n * 0.75), target_n,
                               int(target_n * 1.25), target_n * 2]))
    n_candidates = [x for x in n_candidates if 512 <= x <= 32768]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0,
                          n_betas=None)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  P8 n={n_try}: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s}")
        elif sc_opt < best_score + 0.005 and time.monotonic() < deadline - 50:
            a_pol, sc_pol = _active_set_descent(a_opt, min(deadline - 45, time.monotonic() + 30),
                                                 momentum=0.2, top_k=3,
                                                 tol_frac_start=0.003, tol_frac_end=0.00005,
                                                 lr_init=0.008)
            if sc_pol < best_score:
                best_seq = a_pol.copy()
                best_score = sc_pol
                print(f"  P8 n={n_try} polish: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # === Phase 9: Diverse restarts ===
    print(f"\n[P9] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier", _make_profile(n_best, "fourier_rand")),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("taper", _make_profile(n_best, "tapered")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 20:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=15.0,
                              n_betas=None)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  P9 [{label}]: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s}")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        # Tight active-set
        a_fin, sc_fin = _active_set_descent(best_seq, deadline - 2,
                                             momentum=0.15, top_k=2,
                                             tol_frac_start=0.0005, tol_frac_end=0.00001,
                                             lr_init=0.004, use_min_norm=False)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  P10a: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")

        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 6:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            c_sorted = np.sort(c_best)[::-1]
            gap = c_sorted[0] - c_sorted[min(5, len(c_sorted)-1)]
            beta_fin = max(gap * 0.3, mb * 0.0001)
            betas_fin = beta_fin * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                             0.03, 0.01, 0.005])
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(float(beta_fin), n)
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 80, 'maxls': 50})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  P10b: {best_score:.10f} (t={time.monotonic()-t_start:.1f}s)")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = float(final.sum())
        # Scale so sum is reasonable (the evaluator clips to [0, 1000] and requires sum >= 0.01)
        # Since the score is scale-invariant, any positive scaling works
        # But we need sum >= 0.01, so ensure that
        if s < 0.01:
            final = final * (0.1 / s)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, sum={s:.4f}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
