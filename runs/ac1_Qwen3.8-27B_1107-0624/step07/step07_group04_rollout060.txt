"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines active-set Newton with structured Hessian, dual subgradient,
L-BFGS with log-sum-exp smoothing, mirror descent, and multi-restart.
Key improvements: exact Hessian-vector products via FFT, adaptive active set

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=200, tol=1e-12, reg=0.0):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        if reg > 0:
            Hp = Hp + reg * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        x = x_new
        r = r_new
        rs = rs_new
        p = (rs / rs0) * p + r
        rs0_save = rs0
        rs0 = rs0_save
        if rs < tol * tol * max(rs0_save, 1e-30):
            break
    return x

    a_rev = a[::-1]
    wc = _full_convolve(W, a_rev)
    h = wc[n - 1:2 * n - 1]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, betas=None):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.003)

    if betas is None:
        ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075,
                  0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 0.00005,
                  0.00002, 0.00001]
        betas = [base_beta * r for r in ratios]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-300))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if time.monotonic() > deadline:
            break

        est_iter_time = 0.002 * max(1.0, n / 8192.0)
        max_iter = min(2000, max(30, int((deadline - min_time_remaining - time.monotonic()) / (est_iter_time * 20))))
        if max_iter < 10:
            break

        try:
            result = minimize(
                _smoobj_factory(z, beta, n),
                z,
                jac=_smograd_factory(z, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-300))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _smoobj_factory(z0, beta, n):
    return lambda z_local: _smoobj(z_local, beta, n)


def _smograd_factory(z0, beta, n):
    return lambda z_local: _smograd(z_local, beta, n)


def _mirror_descent(a_init, deadline, max_steps=20000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.05
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use top-m active set (m adapts)
        m = min(10, n)
        top_idx = np.argsort(c)[::-1][:m]
        W = np.zeros(2 * n - 1)
        for k in top_idx:
            W[k] = 1.0 / m

        g = _grad_f(a, W, n)
        g_t = g - np.mean(g)
        gn = np.linalg.norm(g_t)
        if gn < 1e-15:
            break

        # Mirror descent update in log-space
        step = lr / (1.0 + 0.0001 * i)
        a_safe = np.maximum(a, 1e-300)
        z_new = z - step * g_t / a_safe
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 50 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.1)
            else:
                patience += 1
                lr *= 0.9
                if patience > 100:
                    z = np.log(np.maximum(best_a, 1e-300))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.03
        z = z_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _dual_subgradient(a_init, deadline, max_steps=10000):
    """Dual method: optimize multipliers w for active constraints."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Initialize dual variables at the argmax
    c = _full_convolve(a, a)
    c_max = float(c.max())
    W = np.zeros(2 * n - 1)
    W[int(np.argmax(c))] = 1.0

    lr_w = 0.01
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Solve primal: min sum_k W[k]*c[k] s.t. a>=0, sum(a)=1
        # Using projected gradient on the smoothed problem
        g = _grad_f(a, W, n)
        g_t = g - np.mean(g)

        step = lr_w / (1.0 + 0.001 * i)
        a_try = a - step * g_t
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 1e-10:
            a_try = a_try / s_try
            a = a_try

        # Update dual: increase W at positions where c[k] > current max
        c = _full_convolve(a, a)
        c_max_new = float(c.max())

        # Subgradient update for W
        W_new = W.copy()
        for k in range(2 * n - 1):
            if c[k] > c_max_new * 0.999:
                W_new[k] += lr_w * 0.1
        W_new = np.maximum(W_new, 0)
        W_sum = W_new.sum()
        if W_sum > 1e-10:
            W = W_new / W_sum

        # Track best
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                patience = 0
                lr_w *= 0.7

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "flat_mined":
        a = np.ones(n) * 0.3
        center = n // 2
        width = n // 6
        a[center - width:center + width] = 0.1
        return a / a.sum()
    elif kind == "raised":
        t = np.linspace(0, 1, n)
        a = 0.7 + 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-300))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage_reopt(a, deadline, min_time_remaining=60.0):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)
            if time.monotonic() + min_time_remaining < deadline:
                a_reopt = _active_set_newton(best_a, deadline - min_time_remaining, max_rounds=50)
                sc_reopt = _score(a_reopt)
                if sc_reopt < best_sc:
                    best_sc = sc_reopt
                    best_a = a_reopt

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.12f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.12f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Quick L-BFGS with small beta (120s budget)
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    update_best(a1, _score(a1), "lbfgs_1")

    # Phase 2: Active set Newton (200s budget)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _active_set_newton(best_seq, min(deadline - 500, time.monotonic() + 250), max_rounds=100, verbose=True)
        update_best(a2, sc2, "asnewton_1")

    # Phase 3: Mirror descent (150s budget)
    print(f"\n[Phase 3] Mirror descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _mirror_descent(best_seq, min(deadline - 400, time.monotonic() + 150))
        update_best(a3, sc3, "mirror_1")

    # Phase 4: Active set Newton again (150s budget)
    print(f"\n[Phase 4] Active-set Newton 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4, sc4 = _active_set_newton(best_seq, min(deadline - 300, time.monotonic() + 150), max_rounds=100)
        update_best(a4, sc4, "asnewton_2")

    # Phase 5: Shrinkage + re-optimize (100s budget)
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a5, sc5 = _shrinkage_reopt(best_seq, deadline, min_time_remaining=100.0)
        update_best(a5, sc5, "shrinkage")

    # Phase 6: More L-BFGS with very small beta (150s budget)
    print(f"\n[Phase 6] L-BFGS fine (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        tiny_betas = [mb * r for r in [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]]
        a6 = _run_lbfgs(best_seq, deadline, min_time_remaining=60.0, betas=tiny_betas)
        update_best(a6, _score(a6), "lbfgs_fine")

    # Phase 7: Multi-n exploration
    print(f"\n[Phase 7] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        a_init = _resample(parent, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 30:
            a_np, sc_np = _active_set_newton(a_opt, min(deadline - 20, time.monotonic() + 40), max_rounds=40)
            update_best(a_np, sc_np, f"n_{n_try}_as")

    # Phase 8: Diverse restarts
    print(f"\n[Phase 8] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            (0.02, "perturb_small"),
            (0.05, "perturb_med"),
            (0.1, "perturb_large"),
        ]
        for sigma, label in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _active_set_newton(p_start, min(deadline - 10, time.monotonic() + 40), max_rounds=50)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 9: Different profile restarts
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        for kind in ["cos", "bimodal", "raised"]:
            if time.monotonic() > deadline - 10:
                break
            p_start = _make_profile(n_best, kind)
            # Blend with best
            p_blend = 0.7 * p_start + 0.3 * best_seq
            p_blend = np.maximum(p_blend, 0)
            p_blend = p_blend / p_blend.sum()
            a_r = _active_set_newton(p_blend, min(deadline - 5, time.monotonic() + 30), max_rounds=40)
            update_best(a_r, _score(a_r), f"profile_{kind}")

    # Phase 10: Final ultra-fine polish
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _active_set_newton(best_seq, deadline - 1, max_rounds=200)
        update_best(a_fin, sc_fin, "final_as")

        if time.monotonic() < deadline - 1:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-300))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 1 - time.monotonic()
            if remaining > 1:
                max_iter = min(500, max(10, int(remaining / 0.01)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_fin, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 50}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
