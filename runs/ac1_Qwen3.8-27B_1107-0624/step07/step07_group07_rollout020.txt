from scipy.optimize import linprog
import time

def solve_lp(a, c, t_init):
    n = len(a)
    # Variables: d[0..n-1], t
    # Objective: minimize t
    # Constraints:
    #   sum(d) = 0
    #   c[k] + 2*conv(a,d)[k] <= t for k=0..2n-2
    #   a[j] + d[j] >= 0 for j=0..n-1  =>  -d[j] <= a[j]
    
    # conv(a,d)[k] = sum_j a[j] * d[k-j] (valid j where 0<=j<n and 0<=k-j<n)
    
    # For the inequality constraints: c[k] + 2*conv(a,d)[k] - t <= 0
    # In standard form: -2*conv(a,d)[k] + t >= c[k]
    # Or: -2*sum_j a[j]*d[k-j] + t >= c[k]
    
    # For linprog: minimize c^T x subject to A_ub x <= b_ub, A_eq x = b_eq
    # x = [d[0..n-1], t]
    
    # Objective: [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Inequality: c[k] + 2*conv(a,d)[k] - t <= 0
    # => 2*conv(a,d)[k] - t <= -c[k]
    # conv(a,d)[k] = sum_j a[j] * d[k-j]
    # For a given k, the valid j are: max(0, k-(n-1)) <= j <= min(n-1, k)
    # So d index = k - j, which ranges from k - min(n-1,k) to k - max(0, k-(n-1))
    
    A_ub = np.zeros((2*n-1, n + 1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            d_idx = k - j
            A_ub[k, d_idx] += 2.0 * a[j]
        A_ub[k, n] = -1.0
        b_ub[k] = -c[k]
    
    # Bounds: a[j] + d[j] >= 0 => d[j] >= -a[j]
    bounds = [(-a[j], np.inf) for j in range(n)] + [(0, np.inf)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    if result.success:
        d = result.x[:n]
        return d, result.x[-1]
    return None, None

import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

Step sequence optimizer using FFT-based CG Newton method for the 'flatten h'

Key innovation: O(n log n) Newton steps via Hankel matrix structure.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(w, v, n):
    """Compute J @ v where J[j,l] = w[j+l] for j,l in 0..n-1.
    w has length 2n-1."""
    v_rev = v[::-1].copy()
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    V = np.fft.rfft(v_rev, N)
    conv_result = np.fft.irfft(W * V, N)
    return conv_result[n-1 : 2*n-1]


def _cg_solve_hankel(w, rhs, n, max_iter=100, tol=1e-10):
    """Solve J x = rhs using CG, where J[j,l] = w[j+l].
    J is symmetric positive semi-definite (if w >= 0)."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = np.dot(r, r)
    rs0 = rs
    
    for _ in range(max_iter):
        Ap = _hankel_matvec(w, p, n)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * np.sqrt(rs0):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    
    return x


def _flatten_h_newton(a, S_indices, w, n, max_cg_iter=50, cg_tol=1e-10):
    """
    Compute the Newton direction for the 'flatten h' subproblem.
    
    We want to find delta such that h_new ≈ constant, where
    h[j] = sum_k w[k] * 1_S[k] * a[k-j]
    
    Linearization: h_new[j] ≈ h[j] + (J @ delta)[j]
    We want (J @ delta)[j] = -(h[j] - mean(h)) for all j.
    So J @ delta = -(h - mean(h)).
    
    Also need sum(delta) = 0.
    
    Solve the augmented system using CG with a perturbation for the sum constraint.
    """
    # Build w_vec: w_vec[k] = w[k] if k in S, else 0
    w_vec = np.zeros(2 * n - 1)
    w_vec[S_indices] = w[S_indices] if len(w) > n else w
    # Actually, w is already the weight vector for the active set
    # Let's redefine: w_vec[k] = weight for lag k
    
    # Compute current h
    h = _hankel_matvec(w_vec, a, n)
    h_mean = h.mean()
    rhs = -(h - h_mean)
    
    # Solve J @ delta = rhs using CG
    # J[j,l] = w_vec[j+l]
    delta = _cg_solve_hankel(w_vec, rhs, n, max_cg_iter, cg_tol)
    
    # Enforce sum(delta) = 0
    delta -= delta.mean()
    
    return delta


def _line_search(a, direction, score_current, max_step=1.0):
    """Backtracking line search. Returns (best_a, best_score, step)."""
    n = len(a)
    best_a = a.copy()
    best_sc = score_current
    s = a.sum()
    
    for frac in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
        step = max_step * frac
        a_try = a + step * direction
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc
