"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. U-shaped/asymmetric initializations exploiting non-symmetry of self-convolution
2. Aggressive log-sum-exp L-BFGS with beta down to 1e-11
3. Active-set Newton polishing with adaptive tolerance
4. Peak-peeling: explicitly target and reduce top convolution peaks
5. Multi-length exploration (n=4096 to 32768)
6. Mirror descent / exponential weighting on the simplex
7. Basin hopping with structured perturbations
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 120, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=120.0, target_score=1.499):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)
    print(f"    Starting LBFGS: score={current_sc:.10f}, max_c={max_c:.6f}")

    betas = max_c * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.00025,
                              0.0001, 5e-5, 2.5e-5, 1e-5, 5e-6,
                              2.5e-6, 1e-6, 5e-7, 2.5e-7, 1e-7,
                              5e-8, 2.5e-8, 1e-8, 5e-9, 2.5e-9,
                              1e-9, 5e-10, 2.5e-10, 1e-10, 5e-11])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        scale = max(1.0, (n / 4096.0) ** 1.1)
        max_iter = min(15000, max(200, int(remaining / (0.003 * scale * 25))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            if i % 5 == 0:
                print(f"    beta={beta:.2e}: score={current_sc:.10f}")

        if current_sc < target_score:
            print(f"    Reached target score {target_score} at beta={beta:.2e}")
            break

    return current_a


def _active_set_newton(a_init, deadline, k_active=5, max_iter=500):
    """Newton-style polishing on the active set of constraints."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]
        active_vals = c[idx]
        active_mean = active_vals.mean()

        # Check if active set is well-equalized
        spread = (active_vals - active_mean).max() - (active_vals - active_mean).min()
        if spread < 1e-12 * c_max:
            break

        # Compute gradient for the active set
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)

        # Newton-like step with line search
        alpha = 1.0 / np.sqrt(n)
        improved = False
        for _ in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
            alpha *= 0.7

        if not improved:
            # Try smaller active set
            k_active = max(2, k_active - 1)
            if k_active < 2:
                break

    return best_a, best_sc


def _peak_peeling(a_init, deadline, max_iter=1000):
    """Explicitly target and reduce the highest convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        peak_idx = int(np.argmax(c))

        # Find which a[i] contribute most to c[peak_idx]
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] += a[i] * a[j]
                if i != j:
                    contributions[j] += a[i] * a[j]

        # Reduce the elements that contribute most to the peak
        # Use a "push down" step: reduce high-contributing elements,
        # redistribute mass to low-contributing elements
        top_k = min(10, n // 10)
        top_idx = np.argsort(contributions)[::-1][:top_k]
        bot_idx = np.argsort(contributions)[:top_k]

        total_top = a[top_idx].sum()
        total_bot = a[bot_idx].sum()

        if total_top < 1e-15:
            break

        # Move a small fraction of mass from top to bottom
        frac = 0.01 + 0.02 * (it / max_iter)
        delta = frac * total_top / top_k
        a_try = a.copy()
        a_try[top_idx] -= delta
        a_try[bot_idx] += delta * (total_top / max(total_bot, 1e-15)) / top_k
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            break
        a_try = a_try / s

        sc_try = _score(a_try)
        sc_cur = _score(a)

        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try.copy()
        elif sc_try < sc_cur - 1e-15:
            a = a_try.copy()
        else:
            # Try a different redistribution
            alt_top = np.argsort(contributions)[::-1][:max(3, top_k // 2)]
            a_try2 = a.copy()
            a_try2[alt_top] *= (1 - frac * 2)
            a_try2 = np.maximum(a_try2, 0.0)
            s2 = a_try2.sum()
            if s2 > 1e-15:
                a_try2 = a_try2 / s2
                sc_try2 = _score(a_try2)
                if sc_try2 < best_sc - 1e-15:
                    best_sc = sc_try2
                    best_a = a_try2.copy()
                    a = a_try2.copy()

    return best_a, best_sc


def _mirror_descent(a_init, deadline, lr_init=0.1, max_iter=50000):
    """Mirror descent (exponential weighting) on the simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set gradient
        k_active = min(5, 2 * n - 1)
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)

        # Mirror descent step: z -= lr * grad (in the simplex geometry)
        z = z - lr * grad / n
        z = z - z.mean()

        if i % 1000 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.01, 1.0)
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a, best_sc


def _make_ushape(n, sharpness=2.0):
    """U-shaped: high at ends, low in middle."""
    t = np.linspace(0, 1, n)
    a = t**sharpness + (1 - t)**sharpness
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_hump(n, pos1=0.15, pos2=0.85, width=0.08):
    """Two-hump: peaks at two specific positions."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - pos1) / width)**2) + np.exp(-((t - pos2) / width)**2)
    a += 0.01 * np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric_chirp(n, freq=1.0, eps=0.4):
    """Asymmetric chirp with linear amplitude modulation."""
    t = np.arange(n) / n
    a = (1.0 + t) * (1.0 + eps * np.cos(2.0 * np.pi * freq * t * n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.25):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exponential_asym(n, rate=2.0):
    """Exponentially decaying (asymmetric)."""
    t = np.arange(n)
    a = np.exp(-rate * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=4, reserve=30.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.7

        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-30))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier perturbation
            z = np.log(np.maximum(best_a, 1e-30))
            t = np.arange(n)
            for _ in range(3):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.01 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # U-shape perturbation
            a_pert = best_a * (1 + 0.05 * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
        else:
            # Asymmetric shift perturbation
            shift = int(rng.integers(1, max(2, n // 50)))
            a_shifted = np.zeros(n)
            a_shifted[shift:] = best_a[:-shift]
            a_shifted[:shift] = best_a[-shift:] * 0.5
            a_pert = 0.7 * best_a + 0.3 * a_shifted
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()

        c_p = _full_convolve(a_pert, a_pert)
        mc = float(c_p.max())
        beta = max(mc * 0.005, 1e-15)

        if sc_opt < best_sc:
            best_sc = sc_opt

            print(f"    Hop {h}: new best {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer exploiting non-symmetry of self-convolution:
    Phase 1: Diverse initialization (U-shaped, two-hump, asymmetric chirps)
    Phase 2: Aggressive adaptive L-BFGS with log-sum-exp smoothing
    Phase 3: Active-set Newton + peak peeling
    Phase 4: Mirror descent polishing
    Phase 5: Basin hopping
    Phase 6: Length exploration
    Phase 7: Final aggressive polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except NameError:
        pass
    except Exception:
        pass

    candidates = []

    if parent is not None:
        candidates.append(("parent", parent))
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    n_init = 4096
    structured_inits = [
        ("ushape_2", _make_ushape(n_init, sharpness=2.0)),
        ("ushape_4", _make_ushape(n_init, sharpness=4.0)),
        ("ushape_1", _make_ushape(n_init, sharpness=1.0)),
        ("two_hump", _make_two_hump(n_init, pos1=0.15, pos2=0.85, width=0.1)),
        ("two_hump_w", _make_two_hump(n_init, pos1=0.1, pos2=0.9, width=0.15)),
        ("asym_chirp", _make_asymmetric_chirp(n_init, freq=1.0, eps=0.4)),
        ("chirp_f1", _make_chirp(n_init, freq=1.0, eps=0.3)),
        ("chirp_f2", _make_chirp(n_init, freq=2.0, eps=0.25)),
        ("multi_13", _make_multiband(n_init, bands=[1, 3], eps=0.2)),
        ("tapered", _make_tapered(n_init, power=0.5)),
        ("exp_asym", _make_exponential_asym(n_init, rate=3.0)),
    ]

    for name, a_init in structured_inits:
        candidates.append((name, a_init))
        sc = _score(a_init)
        try_update(a_init, name)

    print(f"  Best after init: {best_score_val:.10f}")
    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s")

    # === Phase 2: Main L-BFGS on best candidate ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        a_p2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=500.0)
        try_update(a_p2, "main_lbfgs")

    # Also try L-BFGS on a few other good initializations
    for name, a_init in candidates[:5]:
        if time.monotonic() > deadline - 500:
            break
        if a_init is best_seq:
            continue
        if _score(a_init) > best_score_val * 1.05:
            continue
        print(f"  Trying L-BFGS from {name}...")
        a_opt = _adaptive_lbfgs(a_init, deadline, reserve_time=500.0)
        try_update(a_opt, f"lbfgs_{name}")

    # === Phase 3: Active-set Newton + Peak peeling ===
    print(f"\n=== Phase 3: Active-set & Peak peeling (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_an, sc_an = _active_set_newton(best_seq, deadline - 300, k_active=5, max_iter=300)
        try_update(a_an, "active_newton")

    if best_seq is not None and time.monotonic() < deadline - 250:
        a_pp, sc_pp = _peak_peeling(best_seq, deadline - 250, max_iter=500)
        try_update(a_pp, "peak_peel")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror descent (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_md, sc_md = _mirror_descent(best_seq, deadline - 200, lr_init=0.05, max_iter=30000)
        try_update(a_md, "mirror_descent")

    # === Phase 5: More L-BFGS (second round) ===
    print(f"\n=== Phase 5: Second L-BFGS round (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_p5 = _adaptive_lbfgs(best_seq, deadline, reserve_time=150.0)
        try_update(a_p5, "lbfgs_round2")

    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 80, rng, n_hops=4, reserve=40.0)
        try_update(a_hop, "basin_hop")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        lengths_to_try = [8192, 16384, 32768, 2048]
        if best_n not in lengths_to_try:
            lengths_to_try.append(best_n)

        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 40:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            if time.monotonic() < deadline - 30:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 8: Final aggressive polishing ===
    print(f"\n=== Phase 8: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 3:
        for rnd in range(3):
            if time.monotonic() > deadline - 3:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = max(mb * 1e-10, 1e-20)
            max_iter = min(8000, max(300, int((deadline - 3 - time.monotonic()) / 0.004)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            if time.monotonic() < deadline - 3:
                a_an, _ = _active_set_newton(best_seq, deadline - 3, k_active=3, max_iter=100)
                try_update(a_an, f"final_newton_{rnd}")

            if time.monotonic() < deadline - 3:
                a_pp, _ = _peak_peeling(best_seq, deadline - 3, max_iter=200)
                try_update(a_pp, f"final_peel_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
