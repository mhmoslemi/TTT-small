Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set Newton with direct solve, adaptive constraint management,
and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _conv_gradient(a, W, n):
    """Compute the gradient of sum_k W[k]*c[k] with respect to a.
    grad[j] = 2 * sum_k W[k] * a[k-j] (with a[i]=0 for i outside [0,n-1])
    This equals 2 * (W_conv * a_flipped)[n-1 : 2n-1] where W_conv has length 2n-1.
    """
    a_flip = a[::-1]
    WC = _full_convolve(W, a_flip)
    return 2.0 * WC[n-1:2*n-1]

def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1.
    H*v = 2 * (W * v_flipped)[n-1 : 2n-1]
    """
    v_flip = v[::-1]
    conv = _full_convolve(W, v_flip)
    return 2.0 * conv[n-1:2*n-1]

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """H*v where H[j,m] = 2*W[j+m]. Computed via FFT."""
    v_flip = v[::-1]
    conv = _full_convolve(W, v_flip)
    return 2.0 * conv[n-1:2*n-1]


def _conv_grad(a, W, n):
    """Gradient of sum_k W[k]*c[k] w.r.t. a, where c = conv(a,a).
    grad[j] = 2*sum_k W[k]*a[k-j] = 2*(W*a_flipped)[n-1:2n-1]

    a_flip = a[::-1]
    WC = _full_convolve(W, a_flip)
    return 2.0 * WC[n-1:2*n-1]

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))
