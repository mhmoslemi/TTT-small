Key innovation: FFT-based Hessian-vector products enable fast Newton iterations
with quadratic convergence, combined with L-BFGS warm-up and coordinate descent.
The Hessian of the smoothed max-convolution objective has a banded structure
(anti-diagonals corresponding to active convolution indices), enabling O(n log n)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    if s < 1e-300:

    return e / s

    Here w is the normalized weight vector of length 2n-1."""

def _compute_grad_newton(w, a, n, N_hmat):
    """Compute gradient: g[i] = 4n * sum_k w[k]*a[k-i] for i=0..n-1."""
    W = np.fft.rfft(w, N_hmat)
    A = np.fft.rfft(a, N_hmat)
    R = np.fft.irfft(W * np.conj(A), N_hmat)
    return 4.0 * n * R[:n]


def _hmatvec(w, v, n, N_hmat):
    """Compute (H*v)[i] = 2 * sum_k w[k]*v[k-i] for i=0..n-1."""
    W = np.fft.rfft(w, N_hmat)
    V = np.fft.rfft(v, N_hmat)
    R = np.fft.irfft(W * np.conj(V), N_hmat)
    return 2.0 * R[:n]


def _cg_solve(w, rhs, n, N_hmat, lam, max_iter=80, tol=1e-10):
    """Solve (H + lam*I) x = rhs using conjugate gradient."""
    def matvec(v):
        return _hmatvec(w, v, n, N_hmat) + lam * v

    x0 = np.zeros(n)
    r = rhs - matvec(x0)
    p = r.copy()
    rs = r @ r
    ns = np.sqrt(rs)
    if ns < 1e-300:
        return np.zeros(n)
    r = r / ns
    p = p / ns
    rs_new = r @ r

    for _ in range(max_iter):
        Ap = matvec(p)
        alpha = rs_new / (p @ Ap)
        x0 += alpha * p
        r -= alpha * Ap
        rs_new_new = r @ r
        if np.sqrt(rs_new_new) < tol * ns:
            break
        beta = rs_new_new / rs_new
        p = r + beta * p
        rs_new = rs_new_new

    return x0


def _newton_optimize(a_init, deadline, min_time_remaining=100.0):
    """Newton optimization with FFT-based Hessian-vector products."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    N_hmat = _next_pow2(3 * n - 2)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Beta continuation schedule
    betas = c_max * np.array([0.2, 0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])

    lam = 1e-8

    for beta_idx, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break

        # Recompute beta relative to current max
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = c_max * betas[beta_idx] / (c_max * 0.2)  # rescale
        # Actually just use the original schedule fraction
        beta = c_max * np.array([0.2, 0.1, 0.05, 0.02, 0.01, 0.005,
                                 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])[beta_idx]

        max_iters = 200
        n_iter = 0
        for it in range(max_iters):
            if time.monotonic() + min_time_remaining > deadline:
                break

            # Compute weights
            c = _full_convolve(a, a)
            c_max = c.max()
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            w = e / e.sum()

            # Compute gradient
            g = _compute_grad_newton(w, a, n, N_hmat)
            # Center the gradient (remove mean to enforce sum(d)=0)
            g = g - g.mean()

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break

            # Newton step: solve (H + lam*I) d + mu*1 = -g, 1^T d = 0
            # From H x = g: x = (H+lam*I)^{-1} g
            # From H y = 1: y = (H+lam*I)^{-1} 1
            # mu = (1^T x) / (1^T y) (with sign adjustment)
            # d = -x + mu * y

            x = _cg_solve(w, g, n, N_hmat, lam)
            y = _cg_solve(w, np.ones(n), n, N_hmat, lam)

            num = np.dot(np.ones(n), x)
            den = np.dot(np.ones(n), y)
            if abs(den) < 1e-30:
                mu = 0.0
            else:
                mu = num / den

            d = -x + mu * y
            # Ensure sum(d) = 0 (numerical correction)
            d = d - d.mean()

            d_norm = np.linalg.norm(d)
            if d_norm < 1e-16:
                break

            # Line search
            best_t = 0.0
            best_f = None
            f_cur = 2.0 * n * (c_max + beta * np.log(e.sum()))

            for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
                a_try = a + t * d
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                cm_try = c_try.max()
                shifted_try = (c_try - cm_try) / beta
                e_try = np.exp(shifted_try)
                s_e = e_try.sum()
                if s_e < 1e-300:
                    f_try = 2.0 * n * cm_try
                else:
                    f_try = 2.0 * n * (cm_try + beta * np.log(s_e))

                if f_try < f_cur - 1e-10 * t * g_norm:
                    best_t = t
                    best_f = a_try.copy()
                    break
                elif best_f is None and f_try < f_cur:
                    best_t = t
                    best_f = a_try.copy()

            if best_f is not None and best_t > 0:
                a = best_f
                sc = _score(a)
                n_iter += 1
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    if n_iter % 50 == 0:
                        print(f"      Newton beta_idx={beta_idx}, iter={n_iter}: score={best_score:.10f}")
            else:
                break

            # Occasionally increase regularization if struggling
            if best_t < 0.01 and lam < 1e-4:
                lam *= 10
        else:
            pass

        # Decay lam as we refine
        lam = max(1e-10, lam * 0.5)

    return best_a, best_score


def _lbfgs_run(a_init, beta, n, max_iter):
    """L-BFGS on log-sum-exp smoothed objective in z-space."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_time_remaining=100.0):
    """L-BFGS with beta continuation."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    fractions = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        beta = max_c * frac
        est_per_iter = 0.003 * (n / 4096.0) ** 1.5
        max_iter = min(8000, max(200, int(remaining / (est_per_iter * 20))))

        a_new, success = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc

    return current_a


def _peak_shaving_newton(a_init, deadline, max_iters=2000):
    """Peak shaving using Newton steps on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    N_hmat = _next_pow2(3 * n - 2)
    lam = 1e-8

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set: top 1% of convolution values
        n_active = max(5, int(0.01 * len(c)))
        idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros_like(c)
        w[idx] = 1.0 / n_active

        # Gradient
        g = _compute_grad_newton(w, a, n, N_hmat)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        # Newton step
        x = _cg_solve(w, g, n, N_hmat, lam)
        y = _cg_solve(w, np.ones(n), n, N_hmat, lam)
        num = np.dot(np.ones(n), x)
        den = np.dot(np.ones(n), y)
        mu = num / den if abs(den) > 1e-30 else 0.0
        d = -x + mu * y
        d = d - d.mean()

        # Line search
        improved = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = np.maximum(a + t * d, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            lam *= 5
            if lam > 1.0:
                break

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=20000):
    """Pairwise coordinate descent."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = int((rnd * 6151) % n)
        jdx = int((rnd * 6151 + 13) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = a[idx] + a[jdx]
        if s_pair < 1e-15:
            continue

        improved = False
        for frac in np.linspace(0.1, 0.9, 9):
            a_try = a.copy()
            a_try[idx] = s_pair * frac
            a_try[jdx] = s_pair * (1.0 - frac)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "increasing":
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, rng, scale=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS warm-up, Newton polishing with FFT Hessian-VP,
    peak shaving, coordinate descent, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(8192, "uniform")
        print(f"No parent. Using uniform n=8192.")
        try_update(a0, "init")

    # === Phase 1: L-BFGS warm-up ===
    print(f"\n=== Phase 1: L-BFGS warm-up ===")
    t_phase = time.monotonic()
    a_p1 = _lbfgs_schedule(best_seq, deadline - 1200, min_time_remaining=700.0)
    try_update(a_p1, "lbfgs")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: Newton optimization (main work) ===
    print(f"\n=== Phase 2: Newton optimization ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_nt, sc_nt = _newton_optimize(best_seq, deadline - 500, min_time_remaining=300.0)
        try_update(a_nt, "newton")
        print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_nt):.10f}")

    # === Phase 3: Peak shaving with Newton ===
    print(f"\n=== Phase 3: Peak shaving (Newton) ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_ps, sc_ps = _peak_shaving_newton(best_seq, deadline - 300, max_iters=3000)
        try_update(a_ps, "peak_shave_newton")
        print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: More Newton (refined) ===
    print(f"\n=== Phase 4: Refined Newton ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_nt2, sc_nt2 = _newton_optimize(best_seq, deadline - 200, min_time_remaining=100.0)
        try_update(a_nt2, "newton2")
        print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 50, max_rounds=50000)
        try_update(a_cd, "coord_desc")
        print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_scales = [0.01, 0.02, 0.05, 0.03, 0.04]
        for i, scale in enumerate(restart_scales):
            if time.monotonic() > deadline - 40:
                break
            p_start = _perturb(best_seq, rng, scale)
            sc_start = _score(p_start)
            print(f"  Restart {i} (scale={scale}): start={sc_start:.10f}")
            remaining = deadline - 40 - time.monotonic()
            if remaining < 30:
                break

            # Quick L-BFGS + Newton
            a_r = _lbfgs_schedule(p_start, deadline - 20, min_time_remaining=15.0)
            try_update(a_r, f"restart_{i}_lbfgs")

            if time.monotonic() < deadline - 20:
                a_r2, _ = _newton_optimize(a_r, deadline - 20, min_time_remaining=10.0)
                try_update(a_r2, f"restart_{i}_newton")

        print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 10:
        for cycle in range(2):
            if time.monotonic() > deadline - 10:
                break
            # Newton
            if time.monotonic() < deadline - 10:
                a_f1, _ = _newton_optimize(best_seq, deadline - 10, min_time_remaining=5.0)
                try_update(a_f1, f"final_newton_{cycle}")
            # Coord descent
            if time.monotonic() < deadline - 5:
                a_f2, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=20000)
                try_update(a_f2, f"final_cd_{cycle}")
            # Peak shave
            if time.monotonic() < deadline - 5:
                a_f3, _ = _peak_shaving_newton(best_seq, deadline - 5, max_iters=2000)
                try_update(a_f3, f"final_ps_{cycle}")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "uniform")
        else:
            final = final / s
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
