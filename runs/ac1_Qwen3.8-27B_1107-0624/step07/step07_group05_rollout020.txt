"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive pressure equalization (multiplicative weights with hysteresis active set)
2. L-BFGS continuation on log-sum-exp smoothed objective
3. Multi-n exploration with structural resampling
4. Targeted coordinate adjustments based on KKT pressure analysis

import numpy as np

def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = 1
    while N < n_out:
        N <<= 1
    N = max(N, 2)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s

def _smoobj(z, beta, n):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return float(n)
    a = e / s
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    ee = np.exp(np.clip(shifted, -700, 0))
    ss = ee.sum()
    if ss < 1e-300 or not np.isfinite(ss):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(ss))


def _smograd(z, beta, n):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    a = e / s
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    ee = np.exp(np.clip(shifted, -700, 0))
    ss = ee.sum()
    if ss < 1e-300 or not np.isfinite(ss):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ee / ss
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _resample_to_n(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)


def _make_structured(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5 + 0.1
        return _normalize(np.maximum(a, 1e-10))
    elif kind == "edge_conc":
        t = np.linspace(0, 1, n)
        a = 1.0 / (0.3 + np.abs(t - 0.5) * 4.0)
        return _normalize(a)
    elif kind == "periodic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 7 * t)
        return _normalize(np.maximum(a, 1e-10))
    elif kind == "random_pos":
        a = np.abs(rng.standard_normal(n))
        return _normalize(a)
    elif kind == "gaussian_mix":
        centers = rng.uniform(0, 1, size=3)
        widths = rng.uniform(0.05, 0.2, size=3)
        t = np.linspace(0, 1, n)
        a = np.zeros(n)
        for cc, ww in zip(centers, widths):
            a += np.exp(-((t - cc) / ww) ** 2)
        return _normalize(a)
    else:
        return np.ones(n) / n

        a_fin = _lbfgs_optimize(best_seq, deadline, min_time_remaining=3.0)
        record(a_fin, _score(a_fin))

    # Phase 9: Final pressure
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fp, sc_fp = _pressure_eql(best_seq, deadline - 2, n_steps=2000, gamma_init=0.2, tol_init=0.03, tol_min=0.001)
        record(a_fp, sc_fp)

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
