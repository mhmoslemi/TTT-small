"""
Saddle-point / mirror descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the von Neumann minimax formulation: min_a max_k (a*a)[k] = max_w min_a sum_k w_k (a*a)[k].
Alternates between:
  - w update: mirror ascent on simplex (w_k proportional to (a*a)[k] or softmax thereof)
  - a update: mirror descent (multiplicative) or projected gradient (water-filling) on simplex
Combined with geometric initializations, L-BFGS polishing, and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _hankel_mv(w, a):
    """Compute h_i = sum_j w[i+j] * a_j for i=0..n-1 using FFT."""
    n = len(a)
    # a_pad = a zero-padded to length 2n-1, then reversed
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    a_rev = a_pad[::-1]
    t = _full_convolve(w, a_rev)
    return t[n - 1:2 * n - 1]


def _softmax(z, temp=1.0):
    z = z / temp
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _water_filling(a, grad, eta):
    """Additive projected gradient step: a_new = max(0, a - eta*grad + lambda), sum=1."""
    n = len(a)
    b = a - eta * grad
    # Find lambda such that sum(max(0, b_i + lambda)) = 1
    # Sort b in decreasing order
    b_sorted = np.sort(b)[::-1]
    # Try each possible support size
    cs = np.cumsum(b_sorted)
    # lambda for support of size k: (1 - cs[k-1]) / k
    # Valid if b_sorted[k-1] + lambda > 0 and (k==n or b_sorted[k] + lambda <= 0)
    lam = (1.0 - cs) / np.arange(1, n + 1)
    # Valid condition: b_sorted[k-1] + lam[k-1] > 0  (i.e., the k-th element is positive)
    # i.e., b_sorted[k-1] + (1 - cs[k-1])/k > 0
    # i.e., k*b_sorted[k-1] + 1 - cs[k-1] > 0
    # i.e., 1 + k*b_sorted[k-1] - cs[k-1] > 0
    valid = np.zeros(n, dtype=bool)
    valid[0] = True
    for k in range(1, n):
        valid[k] = (b_sorted[k - 1] + lam[k - 1]) > 0 and (k == n - 1 or (b_sorted[k] + lam[k]) <= 0)
    # Handle the last element
    valid[n - 1] = (b_sorted[n - 1] + lam[n - 1]) > 0
    # Find the valid k (should be exactly one)
    valid_indices = np.where(valid)[0]
    if len(valid_indices) == 0:
        # Fallback: uniform
        return np.ones(n) / n
    k_star = valid_indices[0]
    lam_star = lam[k_star]
    a_new = np.maximum(b + lam_star, 0.0)
    a_new = a_new / a_new.sum()
    return a_new


def _saddle_mirror(a_init, deadline, alpha_a=0.8, alpha_w=1.0, use_softmax_w=False,
                   w_temp=0.1, max_iter=200000, check_every=500):
    """Saddle point optimization with multiplicative (mirror descent) updates."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    s_len = 2 * n - 1
    w = np.ones(s_len) / s_len

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute c = a * a
        c = _full_convolve(a, a)

        # Update w (maximization)
        if use_softmax_w:
            w_new = _softmax(c, w_temp)
        else:
            w_new = c / c.sum()

        # Blend for stability
        w = (1 - alpha_w) * w + alpha_w * w_new
        w = np.maximum(w, 1e-300)
        w = w / w.sum()

        # Compute gradient for a: h_i = sum_j w[i+j] a_j
        h = _hankel_mv(w, a)

        # Update a (minimization) - multiplicative
        h_max = np.max(np.abs(h))
        if h_max < 1e-30:
            break
        eta = alpha_a / h_max
        log_factor = -eta * h
        log_factor -= np.max(log_factor)  # for numerical stability
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-300)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-150:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        a = a_new

        # Check score
        if (it + 1) % check_every == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                if best_sc < 1.5005:
                    break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _saddle_waterfill(a_init, deadline, alpha_a=1.0, alpha_w=1.0,
                      max_iter=200000, check_every=500, momentum=0.9):
    """Saddle point optimization with water-filling (additive) updates."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    s_len = 2 * n - 1
    w = np.ones(s_len) / s_len

    best_a = a.copy()
    best_sc = _score(a)

    # Momentum
    v = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute c = a * a
        c = _full_convolve(a, a)

        # Update w
        w_new = c / c.sum()
        w = (1 - alpha_w) * w + alpha_w * w_new
        w = np.maximum(w, 1e-300)
        w = w / w.sum()

        # Compute gradient
        h = _hankel_mv(w, a)
        grad = 2.0 * h

        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        grad_eff = v

        # Water-filling step
        g_max = np.max(np.abs(grad_eff))
        if g_max < 1e-30:
            break
        eta = alpha_a / g_max
        a_new = _water_filling(a, grad_eff, eta)
        a = a_new

        # Check score
        if (it + 1) % check_every == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _lbfgs_smooth(a_init, beta, n, max_iter):
    """L-BFGS with log-sum-exp smoothing."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def _smoobj(z):
        a = _softmax(z, 1.0)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def _smograd(z):
        a = _softmax(z, 1.0)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(_smoobj, z0, jac=_smograd, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x, 1.0)
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=30.0):
    """Adaptive L-BFGS with decreasing beta."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5])

    for beta in betas:
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break
        scale = max(1.0, (n / 4096.0) ** 1.2)
        max_iter = min(8000, max(100, int(remaining / (0.005 * scale * 20))))
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
        if current_sc < 1.5005:
            break

    return current_a


def _make_geometric(n, r=0.95):
    """Geometric sequence: a_i = r^i / sum."""
    t = np.arange(n)
    a = r ** t
    a = a / a.sum()
    return a


def _make_exponential(n, alpha=0.5):
    """Exponential decay: a_i = exp(-alpha*i/(n-1)) / sum."""
    t = np.arange(n)
    a = np.exp(-alpha * t / max(1, n - 1))
    a = a / a.sum()
    return a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_geometric(n, r=0.8, power=0.5):
    """Tapered geometric: combine geometric with sinusoidal taper."""
    t = np.arange(n)
    geo = r ** t
    taper = np.sin(np.pi * t / (n - 1)) ** power
    a = geo * taper
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_ramp(n, slope=1.0):
    """Linear ramp with floor."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * t / n
    a = a / a.sum()
    return a


def _make_bimodal(n, ratio=0.7):
    """Bimodal: two Gaussian bumps."""
    t = np.linspace(0, 1, n)
    a = np.exp(-20 * (t - 0.3) ** 2) + ratio * np.exp(-20 * (t - 0.7) ** 2)
    a = a / a.sum()
    return a


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _active_subgrad(a_init, deadline, k_active=5, max_steps=20000, lr_init=0.04):
    """Active-set subgradient with adaptive learning rate."""

                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=15.0)
                try_update(a_opt, f"lbfgs_n{target_n}")

    # === Phase 8: Final aggressive polishing ===
    print(f"\n=== Phase 8: Final polishing (best={best_score_val:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(4):
            if time.monotonic() > deadline - 3:
                break
            # Saddle with very slow steps
            a_f, _ = _saddle_mirror(best_seq, deadline - 3, alpha_a=0.3, alpha_w=0.2,
                                    use_softmax_w=True, w_temp=0.01,
                                    max_iter=50000, check_every=1000)
            try_update(a_f, f"final_saddle_{rnd}")

            # L-BFGS micro
            if time.monotonic() < deadline - 3:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-6
                max_iter = min(3000, max(100, int((deadline - 3 - time.monotonic()) / 0.005)))
                a_f2 = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_f2, f"final_lbfgs_{rnd}")

    # === Final ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
