Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using central binomial initialization
and L-BFGS with log-sum-exp smoothing.
Key insight: The central binomial coefficient sequence binom(2k,k)/4^k has the property

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _central_binomial(n):
    """Generate the central binomial coefficient sequence truncated to length n."""
    # a_k = binom(2k, k) / 4^k
    # Using recurrence: a_0 = 1, a_k = a_{k-1} * (2k-1)/(2k)
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n, score_mode=True):
    """Smoothed objective: 2n * logsumexp((c - c_max)/beta) + 2n*c_max, normalized by sum^2."""
    a = _softmax(z)
    s = a.sum()  # should be 1
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    esum = e.sum()
    if esum < 1e-300 or not np.isfinite(esum):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(esum))


def _smograd(z, beta, n):
    """Gradient of the smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    esum = e.sum()
    if esum < 1e-300 or not np.isfinite(esum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / esum
    # Gradient of 2n * logsumexp(c/beta) w.r.t. a, then chain rule through softmax
    # d(d_obj)/d(a_i) = 2n * sum_k w_k * a_{k-i} (for valid k-i)
    # This is 2n * (w * reverse(a))_{n-1:2n-1} ... let me think more carefully.
    # c_k = sum_j a_j * a_{k-j}
    # dc_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (for valid indices)
    # Actually: dc_k/da_i = sum of terms where a_i appears
    # c_k = sum_j a_j * a_{k-j}
    # da_i appears as a_j when j=i: contributes a_{k-i}
    # da_i appears as a_{k-j} when k-j=i, i.e., j=k-i: contributes a_{k-i}
    # So dc_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (when both i and k-i are in [0,n-1])
    # But we need to be careful about the factor of 2 and the softmax chain rule.
    
    # d_obj/da_i = 2n * sum_k w_k * dc_k/da_i = 2n * sum_k w_k * 2*a_{k-i}
    # = 4n * sum_k w_k * a_{k-i}
    # = 4n * (conv(w, reverse(a)))_{i + n - 1 - (n-1)} ... 
    
    # Let me just compute it directly:
    # g_i = 4n * sum_k w_k * a_{k-i} where the sum is over valid k-i
    # This is 4n * (w * a_reversed) cross-correlated at appropriate lag
    
    # Actually, sum_k w_k * a_{k-i} = (w * a)_{...} hmm
    # Let's think: (w * a)_m = sum_k w_k * a_{m-k}
    # We want sum_k w_k * a_{k-i} = sum_k w_k * a_{(k-i)}
    # Let j = k-i: sum_j w_{j+i} * a_j = (w shifted by i) * a at lag 0
    # This is the cross-correlation of w and a at lag i.
    
    # Alternatively: sum_k w_k * a_{k-i} = [conv(w, reverse(a))]_{n-1+i - (n-1)} 
    # Hmm, let me just use the formula:
    # h_i = sum_{k} w_k * a_{k-i} (with a_j = 0 for j < 0 or j >= n)
    # = conv(w, a_reversed)[i + n - 1] where a_reversed[j] = a[n-1-j]
    
    # Actually the cleanest way:
    # h = cross-correlation of w with a, evaluated at lags 0 to n-1
    # h_i = sum_k w_k * a_{k-i}
    # = sum_j a_j * w_{j+i}
    # This is the cross-correlation (a ⋆ w)_i = sum_j a_j * w_{j+i}
    
    # In terms of convolution: (a ⋆ w)_i = conv(a, reverse(w))_{n-1+i-(n-1)}... 
    
    # Let me just compute it via FFT:
    # h_i = sum_j a_j * w_{j+i} for j = max(0, -i) to min(n-1, n-1-i)
    # Wait, w has length 2n-1 and a has length n.
    # h_i = sum_{j} a_j * w_{j+i} where j+i must be in [0, 2n-2]
    # = sum_{j=max(0,-i)}^{min(n-1, 2n-2-i)} a_j * w_{j+i}
    
    # This is the cross-correlation. In FFT terms:
    # H = ifft(conj(FFT(a)) * FFT(w)) gives the circular cross-correlation
    # But we want linear cross-correlation, so we need zero-padding.
    
    # Actually, let me use a simpler approach:
    # h_i = sum_k w_k * a_{k-i}
    # If we define b_k = w_k (padded to length 2n-1) and a has length n:
    # h_i = (b * a_reversed)_{i} where * is cross-correlation
    # = conv(b, a_reversed)_{i + (n-1) - (n-1)}... 
    
    # I'll just compute h directly using the convolution:
    # h_i = sum_k w_k * a_{k-i}
    # Let's define v_k = w_k for k = 0,...,2n-2
    # h_i = sum_{k=i}^{i+n-1} v_k * a_{k-i} (where a_j is valid for 0<=j<n)
    #     = sum_{j=0}^{n-1} v_{i+j} * a_j
    #     = (v cross-correlated with a) at lag i
    #     = conv(v, reverse(a))_{(2n-2) - (n-1) + i} = conv(v, reverse(a))_{n-1+i}
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)  # length (2n-1) + n - 1 = 3n-2... that's too long
    # Wait, _full_convolve(w, a_rev) where w has length 2n-1 and a_rev has length n
    # gives output of length (2n-1) + n - 1 = 3n-2
    # h_i = wc[n-1+i] for i = 0, ..., n-1
    h = wc[n-1:2*n-1]
    
    # Now the gradient w.r.t. a is: 4n * a * h (element-wise, since dc/da = 2*a_{k-i})
    # Wait, I need to be more careful.
    # obj = 2n * LSE(c/beta) where c_k = sum_j a_j a_{k-j}
    # d(obj)/da_i = 2n * (1/beta) * sum_k c_k exp(c_k/beta - LSE) * dc_k/da_i
    #            = 2n * (1/beta) * sum_k (c_k * exp(...)) * 2*a_{k-i} / esum
    # Hmm, actually with the logsumexp formulation:
    # obj = 2n * (c_max + beta * log(sum exp((c_k - c_max)/beta)))
    # d(obj)/da_i = 2n * beta * (1/esum) * sum_k exp((c_k-c_max)/beta) * (1/beta) * dc_k/da_i
    #             = 2n * (1/esum) * sum_k e_k * 2*a_{k-i}
    #             = 4n * (1/esum) * sum_k e_k * a_{k-i}
    #             = 4n * h_i / esum  (where h_i = sum_k w_k * a_{k-i} and w = e/esum)
    # Wait, w_k = e_k/esum, so sum_k w_k * a_{k-i} = h_i (as defined above)
    # So d(obj)/da_i = 4n * h_i / esum... no wait.
    # sum_k e_k * a_{k-i} = esum * sum_k (e_k/esum) * a_{k-i} = esum * h_i
    # So d(obj)/da_i = 2n * (1/esum) * esum * h_i = 2n * h_i
    
    # Hmm let me redo this more carefully.
    # LSE = log(sum_k exp((c_k - c_max)/beta))
    # obj = 2n * (c_max + beta * LSE)
    # d(obj)/da_i = 2n * (dc_max/da_i + beta * d(LSE)/da_i)
    # 
    # If c_max = c_{k*} for some fixed k*, then dc_max/da_i = 2*a_{k*-i}
    # d(LSE)/da_i = (1/beta) * sum_k exp((c_k-c_max)/beta) * dc_k/da_i / sum_k exp((c_k-c_max)/beta)
    #             = (1/beta) * sum_k w_k * 2*a_{k-i}
    #             = (2/beta) * h_i  (where h_i = sum_k w_k * a_{k-i})
    # 
    # So d(obj)/da_i = 2n * (2*a_{k*-i} + 2*h_i)  [if c_max term is active]
    # 
    # But in the logsumexp formulation, the c_max term is absorbed:
    # When beta is small, w concentrates on the argmax, and:
    # d(obj)/da_i ≈ 2n * (2*a_{k*-i}/1 + 2*h_i) where h_i = a_{k*-i} (just the one term)
    # = 2n * 4*a_{k*-i}
    # 
    # Hmm, this is getting confusing. Let me just use the standard formula:
    # For f(a) = 2n * logsumexp(c(a)/beta), where c_k = sum_j a_j a_{k-j}:
    # df/da_i = (2n/beta) * sum_k exp(c_k/beta) * dc_k/da_i / sum_k exp(c_k/beta)
    #         = (2n/beta) * sum_k w_k * 2*a_{k-i}
    #         = (4n/beta) * h_i
    #         = (4n/beta) * sum_k w_k * a_{k-i}
    
    # But my objective is 2n*(c_max + beta*log(sum exp((c-c_max)/beta)))
    # = 2n*c_max + 2n*beta*log(sum exp((c-c_max)/beta))
    # d/da_i = 2n*dc_max/da_i + 2n*beta*(1/(beta*esum))*sum_k e_k*dc_k/da_i
    #        = 2n*dc_max/da_i + (2n/esum)*sum_k e_k*2*a_{k-i}
    #        = 2n*dc_max/da_i + (4n/esum)*sum_k e_k*a_{k-i}
    
    # The issue is dc_max/da_i which depends on which k achieves the max.
    # In the logsumexp trick, this is handled by the c_max term.
    # When we use the full logsumexp (c_max + beta*log(sum exp((c-c_max)/beta))),
    # the gradient of the c_max part is subgradient-like.
    
    # For simplicity, let me just use: grad_i = 4n * h_i (dropping the c_max subgradient)
    # This is an approximation that works well when beta is small.
    
    # Now chain through softmax: a = softmax(z)
    # da_i/dz_j = a_i*(delta_ij - a_j)
    # d(obj)/dz_j = sum_i d(obj)/da_i * da_i/dz_j = sum_i g_i * a_i*(delta_ij - a_j)
    #             = g_j*a_j - a_j*sum_i g_i*a_i
    #             = a_j*(g_j - sum_i g_i*a_i)
    
    g = 4.0 * n * h  # gradient w.r.t. a (approximate, dropping c_max subgradient)
    W = float(np.dot(g, a))  # sum_i g_i * a_i
    return a * (g - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with beta continuation on log-space parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Estimate appropriate beta from the spread of c
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))
        
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _pressure_equalize(a, deadline, n_iters=200, gamma=0.4):
    """KKT-motivated pressure equalization."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 3

        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 20 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())
