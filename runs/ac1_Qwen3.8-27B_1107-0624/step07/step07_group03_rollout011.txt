"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-scale log-sum-exp L-BFGS on softmax parameters
2. Gauss-Newton LP refinement on the active set
3. Exact active-set Newton with KKT solving
4. Larger-n exploration (16384, 32768)
5. Variance-guided reinitialization
6. Final tight polish
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.linalg import solve


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _build_toeplitz_row(a, k, n):
    """Build the row of the Toeplitz operator for lag k: M[k,:] where (M*da)[k] = sum_i M[k,i]*da[i]."""
    row = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        row[i] = a[k - i]
    return row


def _gauss_newton_lp(a, active_idx, max_step=1.0):
    """Solve the Gauss-Newton LP step for the given active set. Returns (da, t_new)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_vals = c[active_idx]
    m = len(active_idx)

    # Build constraint matrix: 2*M[k,:] * da - t <= -c[k]
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for j, k in enumerate(active_idx):
        row = _build_toeplitz_row(a, k, n)
        A_ub[j, :n] = 2.0 * row
        A_ub[j, n] = -1.0
        b_ub[j] = -c[k]

    # Equality: sum(da) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: da_i >= -a_i (to keep a + da >= 0), no upper bound on t
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            da = res.x[:n]
            t_new = res.x[n]
            return da, t_new
    except Exception:
        pass
    return None, None


def _gauss_newton_refine(a, deadline, max_iters=20):
    """Iterative Gauss-Newton with LP subproblems."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        # Limit active set size
        if len(active_idx) > 10:
            top = np.argsort(c)[::-1][:10]
            active_idx = top

        da, t_new = _gauss_newton_lp(a, active_idx)
        if da is None:
            break

        # Line search
        f_old = _score(a)
        neg_mask = da < -1e-15
        if neg_mask.any():
            alpha_max = np.min(a[neg_mask] / (-da[neg_mask])) * 0.95
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max, 1.0)

        alpha = alpha_max
        accepted = False
        for ls in range(20):
            a_try = a + alpha * da
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            f_try = _score(a_try)
            if f_try < f_old - 1e-13 or ls == 19:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            alpha *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Check convergence
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    """Damped Newton on the active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 2 * n:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:2 * n]
            active[top_idx] = True
            n_active = 2 * n

        A_vec = active.astype(np.float64)
        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        def cg_solve(rhs, max_cg=200):
            x = np.zeros_like(rhs)
            r = rhs - hess_matvec(x)
            p = r.copy()
            rs = np.dot(r, r)
            for cg_it in range(max_cg):
                Ap = hess_matvec(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha_cg = rs / pAp
                x += alpha_cg * p
                r_new = r - alpha_cg * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-12 * (1 + np.linalg.norm(rhs)):
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            return x

        u = cg_solve(g_t)
        w = cg_solve(np.ones(n))
        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u - np.mean(-u)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * t * gnorm or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-12 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation, extended to very small betas."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1, n / 2048.0)
        est_time_per_iter = 0.005 * scale
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _variance_objective(z, n):
    """Objective: 2n*max(c)/S^2 + gamma * var(c), for guidance."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    c_var = float(np.mean((c - c_mean) ** 2))
    gamma = 0.01
    return 2.0 * n * c_max + gamma * c_var


def _variance_grad(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    # Gradient of 2n*c_max w.r.t. a (use subgradient at argmax)
    idx_max = int(np.argmax(c))
    # grad of c[k] w.r.t. a[i] = 2*a[k-i]
    grad_max = np.zeros(n)
    lo = max(0, idx_max - (n - 1))
    hi = min(n - 1, idx_max)
    for i in range(lo, hi + 1):
        grad_max[i] = 2.0 * a[idx_max - i]
    grad_max = 2.0 * n * grad_max

    # Gradient of var(c) w.r.t. a
    # var(c) = mean(c^2) - mean(c)^2
    # d(var)/da = 2/(2n-1) * sum_k (c[k] - c_mean) * grad(c[k])
    c_dev = c - c_mean
    gamma = 0.01
    # grad_var[i] = 2/(2n-1) * sum_k c_dev[k] * 2*a[k-i]
    # = 4/(2n-1) * _corr_a_A(a, c_dev, n) ... let me check
    # sum_k c_dev[k] * 2*a[k-i] = 2 * sum_k a[k-i] * c_dev[k]
    # This is 2 * (reverse correlation of a and c_dev)
    # = 2 * _corr_a_A(a, c_dev, n)[i] (up to indexing)
    grad_var = (4.0 / (2 * n - 1)) * _corr_a_A(a, c_dev, n)

    # Total gradient w.r.t. a, then chain rule through softmax
    grad_a = grad_max + gamma * grad_var
    # Softmax gradient: grad_z = a * (grad_a - sum(a * grad_a))
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient method with active-set tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _fourier_init(n, rng, n_basis=20):
    """Initialize using a sum of random Fourier modes."""
    t = np.arange(n)
    a = np.ones(n)
    for _ in range(n_basis):
        freq = rng.integers(1, max(2, n // 3))
        amp = rng.uniform(-0.3, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _kkt_refine(a_init, deadline, max_iter=100):
    """
    Refine using KKT conditions on the identified active set.
    At optimum: sum_k alpha_k * grad(c[k]) = lambda * 1, with c[k]=T for active k.
    We solve for a that satisfies c[k] = T for active k and the gradient condition.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        m = len(active_idx)
        if m > 20:
            top = np.argsort(c)[::-1][:20]
            active_idx = top
            m = 20

        # Compute gradients for active lags: grad_k[i] = 2*a[k-i]
        # We want: sum_j alpha_j * grad_{k_j} = lambda * 1
        # This means: for each i, sum_j alpha_j * 2*a[k_j - i] = lambda
        # Let v[i] = sum_j alpha_j * a[k_j - i] (for valid indices)
        # Then: 2*v[i] = lambda for all i -> v is constant.
        # So we need: sum_j alpha_j * a[k_j - i] = const for all i.

        # This is equivalent to: the "blended" reversal v = sum_j alpha_j * shift(a, k_j)
        # should be constant.

        # Newton step: we adjust a to make v more constant.
        # Let's use the current active set weights (uniform for simplicity) and
        # compute the "effective gradient" and do a Newton step.

        # Effective gradient of the active average:
        g_eff = np.zeros(n)
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                g_eff[i] += 2.0 * a[k - i]
        g_eff /= m

        # Project to tangent: g_t = g_eff - mean(g_eff)
        g_t = g_eff - np.mean(g_eff)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Hessian approximation: use the same as active-set Newton
        A_vec = np.zeros(2 * n - 1)
        A_vec[active_idx] = 1.0 / m
        mu = 1e-8 + 1e-6 * c_max / m

        def hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve
        x = np.zeros(n)
        r = g_t - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(100):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            x += alpha_cg * p
            r_new = r - alpha_cg * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        # Account for sum constraint
        w = np.zeros(n)
        r = np.ones(n) - hess_matvec(np.zeros(n))
        p = r.copy()
        rs = np.dot(r, r)
        for cg_it in range(100):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = rs / pAp
            w += alpha_cg * p
            r_new = r - alpha_cg * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -x - np.mean(-x)
        else:
            lam = float(x.sum()) / sum_w
            dx = -x + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.95
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = _score(a)
        accepted = False
        t = min(1.0, t_max)
        for ls in range(30):
            a_try = a + t * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            f_try = _score(a_try)
            if f_try < f_old - 1e-14:
                a = a_try
                accepted = True
                break
            t *= 0.5

        if not accepted:
            break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc
