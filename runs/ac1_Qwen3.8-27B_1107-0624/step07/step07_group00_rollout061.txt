```
def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    rho = np.max(ind[cond])
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0)
```
