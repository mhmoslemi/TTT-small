```python
def _simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Adam-based optimizer with adaptive soft-max peak weighting for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Explores multiple sequence lengths.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    """Compute soft-max of c using log-sum-exp trick."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    """Compute weights proportional to exp((c - c_max)/beta)."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
```

```python
"""
Adam-based optimizer with adaptive log-sum-exp temperature for minimizing
the step function autocorrelation peak score.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

        beta_init = max(gap * 10, c_max_cur * 0.01)
```

```python
    a_cur = _softmax(z)
    c_cur = _full_convolve(a_cur, a_cur)
    c_max_init = float(c_cur.max())
    beta = beta_frac * c_max_init
    best_a = a_cur.copy()
    best_sc = 2.0 * n * c_max_init

    stall = 0
    iter_count = 0

    while time.monotonic() < deadline:
        t += 1
        iter_count += 1
        a_cur = _softmax(z)
        c_cur = _full_convolve(a_cur, a_cur)
        c_max = float(c_cur.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_cur.copy()
            stall = 0
        else:
            stall += 1

        if stall >= stall_limit:
            break

        shifted = (c_cur - c_max) / beta
        e = np.exp(np.clip(shifted, -700.0, 0.0))
        w_sum = e.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_cur)
            w[int(np.argmax(c_cur))] = 1.0
        else:
            w = e / w_sum

        a_rev = a_cur[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c_cur))
        grad = 4.0 * n * a_cur * (h - W)

        m1 = adam_b1 * m1 + (1.0 - adam_b1) * grad
        v1 = adam_b2 * v1 + (1.0 - adam_b2) * grad * grad
        m_hat = m1 / (1.0 - adam_b1 ** t)
        v_hat = v1 / (1.0 - adam_b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - z.mean()

        beta = max(c_max_init * 1e-8, beta * beta_decay)

    return best_a, best_sc


def _adam_multi_temp(a_init, deadline, label=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < c_max * 0.001:
        gap = c_max * 0.001

    temp_schedule = [
        (gap * 5.0, 0.02, 0.9999),
        (gap * 2.0, 0.01, 0.9999),
        (gap * 1.0, 0.005, 0.99995),
        (gap * 0.5, 0.003, 0.99995),
        (gap * 0.2, 0.001, 0.99999),
    ]

    total_budget = deadline - time.monotonic()
    if total_budget <= 10:
        return best_a, best_sc

    for i, (beta_init, lr, decay) in enumerate(temp_schedule):
        remaining = deadline - time.monotonic()
        frac = max(0.1, 0.7 - 0.1 * i)
        phase_deadline = time.monotonic() + remaining * frac
        if phase_deadline > deadline:
            phase_deadline = deadline
        if phase_deadline - time.monotonic() < 5:
            break

        a_res, sc_res = _adam_logopt(best_a, phase_deadline, lr=lr,
                                     beta_frac=beta_init / max(c_max, 1e-30),
                                     beta_decay=decay, stall_limit=2000,
                                     label=f"{label}_t{i}")
        if sc_res < best_sc:
            best_a = a_res.copy()
            best_sc = sc_res
            c = _full_convolve(best_a, best_a)
            c_max = float(c.max())
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
            if gap < c_max * 0.001:
                gap = c_max * 0.001
            temp_schedule[i + 1] = (
                gap * temp_schedule[i + 1][0] / (gap * 5.0) if i + 1 < len(temp_schedule) else 0)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, tol_frac=0.001, lr_start=0.002):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-30)).astype(np.float64)
    z = z - z.mean()

    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    lr = lr_start
    no_improve = 0

    while time.monotonic() < deadline:
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.95
            if no_improve > 3000 or lr < lr_start * 1e-4:
                break

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = max(1, int(active.sum()))
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step_size = lr * lr / (gnorm + 1e-15)
        z = z - step_size * grad
        z = z - z.mean()

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase Adam-based search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses log-domain parameterization with adaptive soft-max temperature,
    multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                best_seq = parent.copy()
                best_score = _score(parent)
                print(f"Parent: n={n_parent}, score={best_score:.10f}")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Main Adam with multi-temperature on parent n ===
    print(f"\n[P1] Adam multi-temp on n={n_parent} (t={time.monotonic() - t_start:.1f}s)")
    p1_dl = min(deadline - 500, t_start + 550)
    a1, sc1 = _adam_multi_temp(parent, p1_dl, label="p1")
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [P1] t={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Second pass (finer) ===
    print(f"\n[P2] Finer Adam n={len(best_seq)} (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p2_dl = min(deadline - 350, time.monotonic() + 250)
        a2, sc2 = _adam_multi_temp(best_seq, p2_dl, label="p2")
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [P2] t={time.monotonic() - t_start:.1f}s")

    # === Phase 3: Larger n exploration ===
    print(f"\n[P3] Larger n (t={time.monotonic() - t_start:.1f}s)")
    larger_ns = [n_parent * 2]
    if n_parent <= 4096:
        larger_ns.append(n_parent * 4)
    for n_try in larger_ns:
        if n_try > 32768 or time.monotonic() > deadline - 200:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}, init={sc_init:.10f}")
        p3_dl = min(deadline - 200, time.monotonic() + 180)
        a3, sc3 = _adam_multi_temp(a_init, p3_dl, label=f"p3_{n_try}")
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [P3 n={n_try}] t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: Smaller n (sometimes better) ===
    if n_parent > 2048 and time.monotonic() < deadline - 150:
        n_small = n_parent // 2
        a_init = _resample(best_seq, n_small)
        sc_init = _score(a_init)
        print(f"\n[P4] Smaller n={n_small}, init={sc_init:.10f} t={time.monotonic() - t_start:.1f}s")
        p4_dl = min(deadline - 150, time.monotonic() + 100)
        a4, sc4 = _adam_multi_temp(a_init, p4_dl, label=f"p4_{n_small}")
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [P4 n={n_small}] t={time.monotonic() - t_start:.1f}s")

    # === Phase 5: Multi-start diverse initializations ===
    print(f"\n[P5] Multi-start (t={time.monotonic() - t_start:.1f}s)")
    n_best = len(best_seq)
    starts = [
        ("pert3", _perturb_log(best_seq, 0.03, rng)),
        ("pert7", _perturb_log(best_seq, 0.07, rng)),
        ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
        ("taper", _make_profile(n_best, "tapered")),
        ("cosm", _make_profile(n_best, "cos_mod")),
    ]
    for label, a_s in starts:
        if time.monotonic() > deadline - 40:
            break
        p5_dl = min(deadline - 40, time.monotonic() + 50)
        a5, sc5 = _adam_logopt(a_s, p5_dl, lr=0.008, beta_frac=0.03,
                               beta_decay=0.9999, stall_limit=1500)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [P5 {label}] t={time.monotonic() - t_start:.1f}s")

    # === Phase 6: Subgradient polish ===
    print(f"\n[P6] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        p6_dl = min(deadline - 18, time.monotonic() + 80)
        a6, sc6 = _subgradient_polish(best_seq, p6_dl, tol_frac=0.0008,
                                      lr_start=0.001)
        if sc6 < best_score:
            best_seq = a6.copy()
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [P6] t={time.monotonic() - t_start:.1f}s")

    # === Phase 7: Final tiny Adam ===
    if time.monotonic() < deadline - 5:
        p7_dl = deadline - 4
        a7, sc7 = _adam_logopt(best_seq, p7_dl, lr=0.0003, beta_frac=0.005,
                               beta_decay=0.99999, stall_limit=5000)
        if sc7 < best_score:
            best_seq = a7.copy()
            best_score = sc7
            print(f"  NEW BEST: {best_score:.10f} [P7] t={time.monotonic() - t_start:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
