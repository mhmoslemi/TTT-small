"""
Step sequence optimizer using LP-based iteration (active-set convolution constraints),
frequency-domain phase optimization, and L-BFGS polishing.
Novel approach: exploits the LP structure of the cross-convolution constraint

import numpy as np
import time

from scipy.optimize import linprog
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(f, n):
    """Build sparse matrix A such that A @ g = conv(f, g)."""
    # A has shape (2n-1, n)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, M, timeout=30):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)  # minimize -sum(g)
    A_ub = A
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x
            if g.sum() > 1e-10:
                return g
        return None
    except Exception:
        return None

FFT-accelerated adaptive peak shaving with dual-guided gradients,

import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.optimize import minimize


def _next_pow2(n):

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)

    r = np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f, n):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])

def _lp_direction(f, max_time=15.0):

def _adaptive_weights(c, frac=0.02, min_count=5):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active = max(min_count, int(frac * len(c)))
    n_active = min(n_active, len(c))
    threshold = sorted_c[n_active - 1]
    tol = max(1e-6 * c_max, 0.5 * (c_max - threshold))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _gradient_from_weights(w, a, n):
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * n * cross[:n]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _interpolate_to_n(a, target_n):

    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

        print(f"  Interpolated to n={n_lp}, score={_score(a_lp):.10f}")
        a_lp_out, sc_lp = _lp_phase(a_lp, lp_budget, max_iters=150)
        print(f"  LP phase done: score={sc_lp:.10f}, time={time.monotonic()-t_phase:.1f}s")
        a_lp_up = _interpolate_to_n(a_lp_out, len(best_seq))
        _try_update(a_lp_up, "lp_interp")

    # === Phase 2: Peak shaving with FFT-accelerated gradient ===
    print("\n=== Phase 2: FFT peak shaving ===")
    t_phase = time.monotonic()
    ps_budget = deadline - 600
    if time.monotonic() < ps_budget:
        a_ps, sc_ps = _peak_shaving_fft(best_seq, ps_budget, max_iters=50000)
        print(f"  Peak shaving: score={sc_ps:.10f}, time={time.monotonic()-t_phase:.1f}s")
        _try_update(a_ps, "peak_shave")

    # === Phase 3: Frequency-domain magnitude flattening ===
    print("\n=== Phase 3: Frequency flattening ===")
    t_phase = time.monotonic()
    ff_budget = deadline - 400
    if time.monotonic() < ff_budget:
        a_ff, sc_ff = _freq_flatten(best_seq, ff_budget, max_iters=300)
        print(f"  Freq flatten: score={sc_ff:.10f}, time={time.monotonic()-t_phase:.1f}s")
        _try_update(a_ff, "freq_flat")

    # === Phase 4: L-BFGS log-sum-exp polishing ===
    print("\n=== Phase 4: L-BFGS fine schedule ===")
    t_phase = time.monotonic()
    lb_budget = deadline - 200
    if time.monotonic() < lb_budget:
        a_lb = _lbfgs_schedule(best_seq, lb_budget, min_remaining=50.0)
        print(f"  L-BFGS: score={_score(a_lb):.10f}, time={time.monotonic()-t_phase:.1f}s")
        _try_update(a_lb, "lbfgs")

    # === Phase 5: Peak shaving again ===
    print("\n=== Phase 5: Second peak shaving ===")
    t_phase = time.monotonic()
    ps2_budget = deadline - 100
    if time.monotonic() < ps2_budget:
        a_ps2, sc_ps2 = _peak_shaving_fft(best_seq, ps2_budget, max_iters=30000)
        print(f"  PS2: score={sc_ps2:.10f}, time={time.monotonic()-t_phase:.1f}s")
        _try_update(a_ps2, "ps2")

    # === Phase 6: Multi-restart with perturbations ===
    print("\n=== Phase 6: Multi-restart ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 60:
        for rname, scale in [("s1", 0.01), ("s3", 0.03), ("s5", 0.05)]:
            if time.monotonic() > deadline - 40:
                break
            p_start = _perturb(best_seq, rng, scale)
            sc_start = _score(p_start)
            a_r = _lbfgs_schedule(p_start, deadline - 30, min_remaining=10.0)
            _try_update(a_r, f"restart_{rname}")
            if time.monotonic() < deadline - 20:
                a_r2, _ = _peak_shaving_fft(a_r, deadline - 15, max_iters=5000)
                _try_update(a_r2, f"restart_{rname}_ps")
    print(f"  Phase 6 done: time={time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Coordinate descent polish ===
    print("\n=== Phase 7: Coordinate descent ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 10:
        a_cd, sc_cd = _coord_descent(best_seq, deadline - 10, max_rounds=15000)
        print(f"  CD: score={sc_cd:.10f}, time={time.monotonic()-t_phase:.1f}s")
        _try_update(a_cd, "coord_desc")

    # === Phase 8: Final L-BFGS ultra-fine ===
    print("\n=== Phase 8: Ultra-fine L-BFGS ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        for beta_frac in [1e-6, 5e-7, 2e-7]:
            if time.monotonic() > deadline - 5:
                break
            beta = mb * beta_frac
            max_iter = min(4000, max(200, int((deadline - 5 - time.monotonic()) / 0.004)))
            a_ultra = _lbfgs_beta(best_seq, beta, n, max_iter)
            _try_update(a_ultra, f"ultra_{beta_frac}")
    print(f"  Phase 8 done: time={time.monotonic()-t_phase:.1f}s")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(8192) / 8192
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]
    fallback = np.ones(8192) / 8192
    return [float(x) for x in fallback]
