**Algorithm: Iterative Convolution Peak Reduction (ICPR)**

1. Initialize a (from parent or uniform)
2. For each iteration:
   a. Compute c = conv(a, a)
   b. Identify active set: indices where c > (1-tol)*max(c)
   c. Compute the "reducing direction": for each active peak k, the gradient d(c[k])/da
      points in the direction that increases c[k]. We want to move AGAINST this.
   d. Form a combined reducing direction: r = sum_{k in active} w_k * grad_k
      where w_k proportional to (c[k] - threshold)
   e. Take a step: a_new = a - step_size * r
   f. Project: a_new = max(a_new, 0), then normalize to sum=1
   g. Check if max(c) decreased; if not, reduce step size
3. Return best a found

The key challenge is computing the gradient efficiently. For a single peak at index k:
g[i] = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)

This is O(n) to compute for one k. For multiple peaks, it's O(|active| * n).

For n=8192 and |active| up to a few hundred, this is fast enough (a few million operations per iteration).

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Compute d(conv(a,a)[k])/da for lag k."""
    g = np.zeros(n)
    # For each i, d(c[k])/da[i] = a[k-i] + a[i-k] (if indices valid)
    # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1

    i_max1 = min(n - 1, k)
    if i_min1 <= i_max1:
        indices = np.arange(i_min1, i_max1 + 1)
        g[indices] += a[k - indices]
    
    i_min2 = max(0, k)
    i_max2 = min(n - 1, k + n - 1)
    if i_min2 <= i_max2:
        indices = np.arange(i_min2, i_max2 + 1)
        g[indices] += a[indices - k]
    
    return g

    stall_limit = 200

        c_max = c.max()

        # d(c[k])/da in the direction that increases c[k]

        r_norm = np.linalg.norm(r)
        if r_norm < 1e-15:

        r_norm = np.linalg.norm(r)
        if r_norm < 1e-15:

            break

def _lbfgs_smoothed(a_init, deadline, min_remaining=60.0):
    """L-BFGS optimization using log-sum-exp smoothing of the max function."""
    n = len(a_init)

    def _softmax(z):
        z = z - np.max(z)
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return np.ones_like(z) / len(z)
        return e / s

    def _smoobj(z, beta):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def _smograd(z, beta):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_init

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5])
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        max_iter = min(5000, max(100, int(remaining / 0.02)))

        def fun(z_in):
            return _smoobj(z_in, beta)

        def jac(z_in):
            return _smograd(z_in, beta)

        try:
            result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15,
                                       'gtol': 1e-12, 'maxcor': 80,
                                       'maxls': 50, 'disp': False})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = result.x
                z = z - z.mean()
                print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 2 * t) + 0.05 * np.cos(2.0 * np.pi * 5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, rng, mode="gauss", scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if mode == "gauss":
        z = z + scale * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(4):
            freq = int(rng.integers(1, max(2, n // 20)))
            amp = 0.005 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(8, n // 32)
        for _ in range(6):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.005, 0.005)
    z = z - z.mean()
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return e / s


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # === Initialization ===
    print(f"{'='*60}")
    print(f"Starting optimization. Budget: {budget_s}s")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(8192, "uniform")
        print(f"No parent. Uniform n=8192, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    start_n = len(best_seq) if best_seq is not None else 8192

    # === Phase 1: Peak-directed reduction (main workhorse) ===
    print(f"\n=== Phase 1: Peak reduction on n={start_n} ===")
    t_phase = time.monotonic()
    p1_deadline = t_start + 0.40 * (budget_s - 15)
    a_p1, sc_p1 = _peak_reduction(best_seq, p1_deadline, max_iters=200000,
                                  tol_frac=0.03, max_active=25)
    try_update(a_p1, "peak_red_1")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={sc_p1:.10f}")

    # === Phase 2: L-BFGS smoothing ===
    print(f"\n=== Phase 2: L-BFGS polishing ===")
    t_phase = time.monotonic()
    p2_deadline = t_start + 0.60 * (budget_s - 15)
    if time.monotonic() < p2_deadline:
        a_p2 = _lbfgs_smoothed(best_seq, p2_deadline, min_remaining=120.0)
        try_update(a_p2, "lbfgs")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 3: Second peak reduction pass ===
    print(f"\n=== Phase 3: Peak reduction round 2 ===")
    t_phase = time.monotonic()
    p3_deadline = t_start + 0.75 * (budget_s - 15)
    if time.monotonic() < p3_deadline:
        a_p3, sc_p3 = _peak_reduction(best_seq, p3_deadline, max_iters=100000,
                                      tol_frac=0.01, max_active=40)
        try_update(a_p3, "peak_red_2")
        # Also try with different tol
        a_p3b, sc_p3b = _peak_reduction(best_seq, p3_deadline, max_iters=50000,
                                        tol_frac=0.05, max_active=15)
        try_update(a_p3b, "peak_red_2b")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    t_phase = time.monotonic()
    p4_deadline = t_start + 0.88 * (budget_s - 15)
    target_lengths = []
    if start_n < 16384:
        target_lengths.append(16384)
    if start_n > 4096:
        target_lengths.append(4096)
    # Also try a prime-ish length for diversity
    target_lengths.append(10000)

    for target_n in target_lengths:
        if time.monotonic() > p4_deadline:
            break
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"    Interpolated: score={sc_interp:.10f}")
        try_update(a_interp, f"interp_n{target_n}")

        # Quick peak reduction
        sub_deadline = min(p4_deadline, time.monotonic() + 60)
        a_opt, _ = _peak_reduction(a_interp, sub_deadline, max_iters=30000,
                                   tol_frac=0.03, max_active=20)
        try_update(a_opt, f"n{target_n}_peak")

        # Quick L-BFGS
        if time.monotonic() < p4_deadline - 20:
            a_lb = _lbfgs_smoothed(a_opt, p4_deadline, min_remaining=30.0)
            try_update(a_lb, f"n{target_n}_lbfgs")

    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 5: Multi-restart with structured perturbations ===
    print(f"\n=== Phase 5: Multi-restart ===")
    t_phase = time.monotonic()
    p5_deadline = deadline - 30.0
    restart_configs = [
        ("gauss_1", "gauss", 0.01),
        ("gauss_3", "gauss", 0.03),
        ("gauss_5", "gauss", 0.05),
        ("fourier", "fourier", 0.0),
        ("block", "block", 0.0),
        ("gauss_10", "gauss", 0.10),
    ]
    for name, mode, scale in restart_configs:
        if time.monotonic() > p5_deadline:
            break
        p_start = _perturb(best_seq, rng, mode, scale)
        sc_start = _score(p_start)
        print(f"  Restart [{name}]: start={sc_start:.10f}")

        # Quick peak reduction on perturbed
        sub_deadline = min(p5_deadline, time.monotonic() + 30)
        a_r, _ = _peak_reduction(p_start, sub_deadline, max_iters=20000,
                                 tol_frac=0.03, max_active=20)
        try_update(a_r, f"restart_{name}")

    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final polish ===")
    t_phase = time.monotonic()
    for cycle in range(4):
        if time.monotonic() > deadline - 5:
            break

        # Peak reduction with tight tolerance
        if time.monotonic() < deadline - 5:
            a_f1, _ = _peak_reduction(best_seq, deadline - 5, max_iters=50000,
                                      tol_frac=0.005, max_active=50)
            try_update(a_f1, f"final_ps_{cycle}")

        # L-BFGS with very small beta
        if time.monotonic() < deadline - 5 and cycle < 2:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-6
            z0 = np.log(np.maximum(best_seq, 1e-15))
            z0 = z0 - z0.mean()

            def fun_final(z_in):
                a_sm = z_in - np.max(z_in)
                e = np.exp(a_sm)
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * mb
                a_sm = e / s_e
                c_sm = _full_convolve(a_sm, a_sm)
                c_mx = c_sm.max()
                shifted = (c_sm - c_mx) / beta
                e2 = np.exp(shifted)
                s2 = e2.sum()
                if s2 < 1e-300:
                    return 2.0 * n * c_mx
                return 2.0 * n * (c_mx + beta * np.log(s2))

            def jac_final(z_in):
                a_sm = z_in - np.max(z_in)
                e = np.exp(a_sm)
                s_e = e.sum()
                if s_e < 1e-300:
                    return np.zeros_like(z_in)
                a_sm = e / s_e
                c_sm = _full_convolve(a_sm, a_sm)
                c_mx = c_sm.max()
                shifted = (c_sm - c_mx) / beta
                e2 = np.exp(shifted)
                s2 = e2.sum()
                if s2 < 1e-300:
                    w = np.zeros_like(c_sm)
                    w[int(np.argmax(c_sm))] = 1.0
                else:
                    w = e2 / s2
                a_rev = a_sm[::-1]
                wc = _full_convolve(w, a_rev)
                h = wc[n - 1:2 * n - 1]
                W = float(np.dot(w, c_sm))
                return 4.0 * n * a_sm * (h - W)

            max_iter = min(3000, max(200, int((deadline - 5 - time.monotonic()) / 0.03)))
            try:
                res = minimize(fun_final, z0, jac=jac_final, method='L-BFGS-B',
                               options={'maxiter': max_iter, 'ftol': 1e-16,
                                        'gtol': 1e-13, 'maxcor': 100,
                                        'maxls': 60, 'disp': False})
                a_z = res.x - np.max(res.x)
                e_z = np.exp(a_z)
                a_final_lb = e_z / e_z.sum()
                try_update(a_final_lb, f"final_lbfgs_{cycle}")
            except Exception:
                pass

    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
