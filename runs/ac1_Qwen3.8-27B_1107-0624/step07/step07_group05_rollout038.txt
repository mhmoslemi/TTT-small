import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _pressure_loop(a, deadline, max_frac_start=0.5, max_frac_end=0.05, 
                   gamma_start=1.0, gamma_end=0.1, min_gap=0.001):
    """
    Pressure equalization with cooling active set.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)
    
    it = 0
    while time.monotonic() < deadline:
        progress = (time.monotonic() - t0) / total_time
        progress = min(1.0, progress)
        
        # Cooling: active set fraction
        max_frac = max_frac_start + (max_frac_end - max_frac_start) * progress
        # Step size cooling
        gamma = gamma_start + (gamma_end - gamma_start) * progress
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[max(1, int(n * 2 * (1 - max_frac))) - 1] if n * 2 * (1 - max_frac) > 1 else c_sorted[-1]
        # Actually let me just use a fraction of the convolution length
        c_len = len(c)  # 2n-1
        n_active_target = max(1, int(c_len * max_frac))
        threshold = c_sorted[min(n_active_target - 1, c_len - 1)]
        
        active = c >= threshold
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 5
        
        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n-1:2*n-1]
        p = np.maximum(p, 1e-30)
        
        # Update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        it += 1
        if it % 500 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc

"""
Pressure-equalization optimizer with cooling active set and aggressive L-BFGS polishing.
Key innovations:
1. FFT-cached pressure equalization with cooling active set (top 50% → top 5%)
2. Very aggressive L-BFGS beta continuation (30+ halvings)
3. Subgradient polish with Polyak step size
4. Multi-n exploration (4096, 8192, 16384)
5. Adaptive restarts on stagnation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_same(a, N_cache=None):
    """Compute a * a using FFT. Returns the full convolution."""
    n = len(a)
    out_n = 2 * n - 1
    N = N_cache if N_cache is not None else _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out_n]
    return c


def _cross_corr(a, b, N_cache=None):
    """Compute cross-correlation: result[i] = sum_k a[k] * b[k-i] (for valid i).
    Returns the 'center' part of length n_a."""
    n_a = len(a)
    out_n = len(a) + len(b) - 1
    N = N_cache if N_cache is not None else _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b[::-1], N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r[len(b)-1: len(b)-1+n_a]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_same(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    c = _convolve_same(a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _pressure_loop(a_init, deadline, 
                   frac_start=0.5, frac_end=0.05,
                   gamma_start=0.8, gamma_end=0.05,
                   check_interval=100):
    """
    Pressure equalization with cooling active set.
    Uses FFT caching for speed.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)
    
    t0 = time.monotonic()
    total_budget = max(1.0, deadline - t0)
    it = 0
    stall_count = 0
    best_stall_sc = best_sc
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_budget)
        
        # Cooling schedule
        frac = frac_start + (frac_end - frac_start) * progress
        gamma = gamma_start + (gamma_end - gamma_start) * progress
        
        # Compute convolution
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2*n-1]
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        # Determine active set
        c_len = 2 * n - 1
        n_active = max(3, int(c_len * frac))
        # Use partition for efficiency
        threshold = np.partition(c, c_len - n_active)[c_len - n_active]
        active = (c >= threshold).astype(np.float64)
        
        # Compute pressure: p_i = sum_{k in active} a_{|k-i|} (approximately)
        # More precisely: p_i = sum_k active[k] * a[k-i] (with bounds)
        # This is: (active * a_rev)[n-1 : 2n-1] where a_rev = a[::-1]
        fa_rev = np.fft.rfft(a[::-1], N)
        f_act = np.fft.rfft(active, N)
        p_conv = np.fft.irfft(f_act * fa_rev, N)[:2*n-1]
        p = p_conv[n-1:2*n-1]
        p = np.maximum(p, 1e-30)
        
        # Multiplicative update
        p_mean = float(np.mean(p))
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        it += 1
        
        # Periodic check
        if it % check_interval == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 20:
                    # Reset to best and perturb
                    a = best_a.copy()
                    stall_count = 0
                    # Slightly perturb
                    a += 0.01 * np.random.default_rng(it).standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc


def _lbfgs_phase(a_init, deadline, min_remaining=5):
    """
    L-BFGS with aggressive beta continuation.
    """
    n = len(a_init)
    c_cur = _convolve_same(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    # Determine initial beta
    c_sorted = np.sort(c_cur)[::-1]
    n_c = len(c_sorted)
    gap = c_sorted[0] - c_sorted[min(n_c // 4, n_c - 1)]
    base_beta = max(gap, max_c * 0.1)
    
    # Very aggressive beta schedule
    n_betas = 30
    betas = base_beta * (0.5 ** np.arange(n_betas))
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    N = _next_pow2(2 * n - 1)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 10))))
        
        # Smoothing objective and gradient
        def obj(z_val):
            a_val = _softmax(z_val)
            fa = np.fft.rfft(a_val, N)
            c_val = np.fft.irfft(fa * fa, N)[:2*n-1]
            c_m = float(c_val.max())
            shifted = (c_val - c_m) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_m
            return 2.0 * n * (c_m + beta * np.log(s))
        
        def grad(z_val):
            a_val = _softmax(z_val)
            fa = np.fft.rfft(a_val, N)
            c_val = np.fft.irfft(fa * fa, N)[:2*n-1]
            c_m = float(c_val.max())
            shifted = (c_val - c_m) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros(2*n-1)
                w[int(np.argmax(c_val))] = 1.0
            else:
                w = e / s
            fa_rev = np.fft.rfft(a_val[::-1], N)
            f_w = np.fft.rfft(w, N)
            wc = np.fft.irfft(f_w * fa_rev, N)[:2*n-1]
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c_val))
            return 4.0 * n * a_val * (h - W)
        
        try:
            result = minimize(
                obj, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _subgrad_polish(a_init, deadline, n_steps=5000):
    """Subgradient polish with adaptive step size."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.05
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _convolve_same(a, N)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        # Active set: top 5%
        c_len = 2 * n - 1
        n_act = max(3, c_len // 20)
        threshold = np.partition(c, c_len - n_act)[c_len - n_act]
        active = c >= threshold
        w = active.astype(float) / active.sum()
        
        # Gradient
        fa_rev = np.fft.rfft(a[::-1], N)
        f_w = np.fft.rfft(w, N)
        wc = np.fft.irfft(f_w * fa_rev, N)[:2*n-1]
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -30, 30)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_chk = _softmax(z)
            sc = _score(a_chk)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_chk.copy()
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.9
                if lr < 0.001:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    lr = 0.02
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5 + 0.1
    elif kind == "bell":
        a = np.exp(-2 * ((t - 0.5) / 0.3) ** 2) + 0.05
    elif kind == "double_hump":
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

"""
Pressure-equalization optimizer with cooling active set and aggressive L-BFGS polishing.
Key innovations over previous:
1. FFT-cached pressure equalization with cooling active set (50% -> 5%)
2. Very aggressive L-BFGS beta continuation (30 halvings)
3. Subgradient polish with adaptive step size
4. Multi-n exploration (4096, 8192, 16384)
5. Adaptive restarts on stagnation
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_same(a, N_cache=None):
    n = len(a)
    out_n = 2 * n - 1
    N = N_cache if N_cache is not None else _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:out_n]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_same(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n, N):
    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c_m = float(c.max())
    shifted = (c - c_m) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_m
    return 2.0 * n * (c_m + beta * np.log(s))


def _smooth_grad(z, beta, n, N):
    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c_m = float(c.max())
    shifted = (c - c_m) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    fa_rev = np.fft.rfft(a[::-1], N)
    f_w = np.fft.rfft(w, N)
    wc = np.fft.irfft(f_w * fa_rev, N)[:2 * n - 1]
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _pressure_loop(a_init, deadline, frac_start=0.5, frac_end=0.05,
                   gamma_start=0.8, gamma_end=0.05, check_interval=100):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    t0 = time.monotonic()
    total_budget = max(1.0, deadline - t0)
    it = 0
    stall_count = 0
    rng = np.random.default_rng(42 + n)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_budget)

        frac = frac_start + (frac_end - frac_start) * progress
        gamma = gamma_start + (gamma_end - gamma_start) * progress

        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        c_len = 2 * n - 1
        n_active = max(3, int(c_len * frac))
        threshold = np.partition(c, c_len - n_active)[c_len - n_active]
        active = (c >= threshold).astype(np.float64)

        fa_rev = np.fft.rfft(a[::-1], N)
        f_act = np.fft.rfft(active, N)
        p_conv = np.fft.irfft(f_act * fa_rev, N)[:2 * n - 1]
        p = p_conv[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        p_mean = float(np.mean(p))
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        it += 1

        if it % check_interval == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 30:
                    a = best_a.copy()
                    stall_count = 0
                    pert = 0.02 * rng.standard_normal(n)
                    a = a + pert
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _lbfgs_phase(a_init, deadline, min_remaining=5):
    n = len(a_init)
    c_cur = _convolve_same(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    n_c = len(c_sorted)
    gap = c_sorted[0] - c_sorted[min(n_c // 4, n_c - 1)]
    base_beta = max(gap, max_c * 0.1)

    n_betas = 30
    betas = base_beta * (0.5 ** np.arange(n_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    N = _next_pow2(2 * n - 1)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _smooth_obj, z, args=(beta, n, N),
                jac=_smooth_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_steps=5000):
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.05

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        c_len = 2 * n - 1
        n_act = max(3, c_len // 20)
        threshold = np.partition(c, c_len - n_act)[c_len - n_act]
        active = c >= threshold
        w = active.astype(float) / float(active.sum())

        fa_rev = np.fft.rfft(a[::-1], N)
        f_w = np.fft.rfft(w, N)
        wc = np.fft.irfft(f_w * fa_rev, N)[:2 * n - 1]
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -30, 30)
        z = z - z.mean()

        if i % 100 == 0:
            a_chk = _softmax(z)
            sc = _score(a_chk)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_chk.copy()
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.9
                if lr < 0.001:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    lr = 0.02

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5 + 0.1
    elif kind == "bell":
        a = np.exp(-2.0 * ((t - 0.5) / 0.3) ** 2) + 0.05
    elif kind == "double_hump":
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_sequence(src, n_target):
    x_old = np.linspace(0, 1, len(src))
    x_new = np.linspace(0, 1, n_target)
    a = np.interp(x_new, x_old, src)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = _make_profile(n_target, "tapered")
    return a
