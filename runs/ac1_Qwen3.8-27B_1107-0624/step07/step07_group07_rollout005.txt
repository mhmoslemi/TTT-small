import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

# Chirp-based initialization
def _make_chirp(n, freq_start=0, freq_end=1, n_harmonics=1):
    t = np.arange(n) / n
    # Linear chirp: frequency sweeps from freq_start to freq_end
    # Phase = 2*pi*(freq_start*t + (freq_end-freq_start)*t^2/2)
    phase = 2 * np.pi * (freq_start * t + (freq_end - freq_start) * t**2 / 2)
    # Non-negative version: 1 + cos(phase)
    a = 1.0 + np.cos(phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()

    phase = np.pi * chirp_rate * t**2
    a = 1.0 + np.cos(phase)
    a = np.maximum(a, 0.0)

def _nesterov_peak_shaving(a_init, deadline, max_iters=50000):
    """
    Nesterov-accelerated gradient descent with adaptive active set.
    Optimizes the max of convolution directly.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_sc = _score(a)
    
    # Adaptive parameters
    tol = 0.1  # active set tolerance (fraction of max)
    tol_decay = 0.995
    lr = 0.01
    momentum = 0.9
    stall = 0
    max_stall = 200
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Determine active set
        threshold = c_max * (1.0 - tol)
        active = c >= threshold
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Compute gradient: d/d a[i] of sum_k w[k] * conv(a,a)[k]
        # = sum_k w[k] * a[k-i] (with proper bounds)
        # This is equivalent to convolving w with reversed a, then extracting the right part
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = wc[n-1:2*n-1]  # shape (n,)
        
        # Center the gradient (maintain constant sum)
        grad = grad - grad.mean()
        
        # Nesterov update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_sc - 1e-16:
            best_sc = sc
            best_a = a_new.copy()
            stall = 0
            # Increase learning rate slightly
            lr = min(lr * 1.01, 0.05)
        else:
            a_new = a.copy()  # revert
            stall += 1
            lr *= 0.95
            momentum = min(momentum * 1.001, 0.99)
        
        a = a_new
        
        # Decay the active set tolerance
        tol = max(tol * tol_decay, 1e-6)
        
        if lr < 1e-10:
            break
        if stall > max_stall:
            if tol > 1e-5:
                tol = max(1e-5, tol * 0.5)  # widen active set when stuck
                lr = 0.01
                v = np.zeros(n)
                stall = 0
            else:
                break
    
    return best_a, best_sc

def _lbfgs_smoothed(a_init, deadline, betas=None):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    
    if betas is None:
        betas = c_max * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    current_a = a_init.copy()
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        def fun(z):
            a_local = _softmax(z)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = c_local.max()
            shifted = (c_local - c_max_local) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max_local
            return 2.0 * n * (c_max_local + beta * np.log(s))
        
        def jac(z):
            a_local = _softmax(z)
            c_local = _full_convolve(a_local, a_local)
            c_max_local = c_local.max()
            shifted = (c_local - c_max_local) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w_local = np.zeros_like(c_local)
                w_local[int(np.argmax(c_local))] = 1.0
            else:
                w_local = e / s
            a_rev = a_local[::-1]
            wc = _full_convolve(w_local, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w_local, c_local))
            return 4.0 * n * a_local * (h - W)
        
        try:
            max_iter = min(5000, max(100, int(remaining / 0.01)))
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-16,
                                      'gtol': 1e-12, 'maxcor': 50,
                                      'maxls': 50, 'disp': False})
            a_new = _softmax(result.x)
            sc = _score(a_new)
            sc_cur = _score(current_a)
            if sc < sc_cur:
                current_a = a_new
                z0 = np.log(np.maximum(a_new, 1e-15))
                z0 = z0 - z0.mean()
        except Exception:
            pass
    
    return current_a

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()
