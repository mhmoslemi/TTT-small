"""
Active-set QP optimizer for autocorrelation flattening.
Uses direct QP subproblem solving with CG for linear systems,
adaptive active set management, and multi-scale search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hess_vec(v, active, n):
    """Compute H@v where H[i,j] = 2*[i+j in active].
    (H@v)[i] = 2 * sum_j v[j] * active[i+j]
    = 2 * (v_rev * active)[n-1+i]
    """
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N)
    fa = np.fft.rfft(active, N)
    r = np.fft.irfft(fv * fa, N)
    return 2.0 * r[n - 1:2 * n - 1]


def _grad_active(a, active, n):
    """Gradient of sum_{k in active} c[k] = 2 * (a correlated with active indicator)."""
    return _hess_vec(a, active, n)


def _solve_qp_active_set(a_init, active, n, deadline, max_inner_iter=5000):
    """Solve QP: min a^T M a s.t. 1^T a = 1, a >= 0.
    Using active-set method for non-negativity + CG for KKT system.
    
    M is defined by (M@v)[i] = sum_j v[j] * [i+j in active] (so Hessian = 2M).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    # KKT system: [2M  1] [da]   [0]
    #             [1^T  0] [lam] = [-g]  (where g = 2M@a is the gradient)
    # Actually: minimize a^T M a s.t. 1^T a = 1
    # KKT: 2M@a + lam*1 = 0, 1^T a = 1
    # So: [2M  1; 1^T  0] [a; lam] = [0; 1]
    
    # With non-negativity: [2M  1; 1^T  0] [a; lam] = [mu; 1] where mu = 2M@a + lam*1 >= 0
    # and mu_i = 0 if a_i > 0.
    
    # Active-set method: maintain set of "free" indices (where a_i > 0).
    # Solve the reduced KKT system for free indices.
    
    free = a > 1e-12
    lam = 0.0
    
    for outer it in range(max_inner_iter):
        if time.monotonic() > deadline:
            break
        
        n_free = free.sum()
        if n_free == 0:
            a = np.ones(n) / n
            break
        if n_free == n:
            # Full system
            pass
        
        # Solve reduced KKT: [2M_ff  1_f; 1_f^T  0] [a_f; lam] = [0; 1]
        # Where _f denotes the free indices.
        # This gives: 2M_ff @ a_f + lam * 1_f = 0
        #             1_f^T @ a_f = 1
        # (indices not in free are set to 0)
        
        # Using the matrix-free Hessian:
        # We need to solve (2M_ff) @ a_f = -lam * 1_f
        # with the constraint 1_f^T a_f = 1.
        
        # Eliminate lam: from the constraint, lam = -2 * (1_f^T M_ff a_f) / (1_f^T 1_f) ... no.
        # From 2M_ff a_f + lam * 1 = 0: a_f = -(lam/2) (M_ff)^{-1} 1_f
        # Constraint: 1_f^T a_f = -(lam/2) * 1_f^T (M_ff)^{-1} 1_f = 1
        # So lam = -2 / (1_f^T (M_ff)^{-1} 1_f)
        # And a_f = (M_ff)^{-1} 1_f / (1_f^T (M_ff)^{-1} 1_f)
        
        # This requires solving M_ff x = 1_f (restricted to free indices).
        # We can do this with CG.
        
        # Define the matrix-free operator on the free subspace:
        # (M_free @ v)_i = sum_{j in free} v_j * [i+j in active] for i in free
        # (v is defined on free indices only, but we can embed it in full space)
        
        b = np.zeros(n)
        b[free] = 1.0
        
        # Solve (2M) x = b with x fixed to 0 on inactive indices
        # Using projected CG
        
        x = np.zeros(n)
        r = b - _hess_vec(x, active, n)  # = b since x=0
        # Project r onto free subspace
        r[~free] = 0
        p = r.copy()
        rs = float(np.dot(r, r))
        
        for cg_iter in range(min(200, n_free + 10)):
            if rs < 1e-30:
                break
            Hp = _hess_vec(p, active, n)
            Hp[~free] = 0  # Project
            pH = float(np.dot(p, Hp))
            if abs(pH) < 1e-30:
                break
            alpha = rs / pH
            x = x + alpha * p
            x[~free] = 0
            r = b - _hess_vec(x, active, n)
            r[~free] = 0
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-12:
                break
            beta_cg = r_new / rs
            p = r + beta_cg * p
            p[~free] = 0
            rs = r_new
        
        # Now x = M^{-1} b (approximately), restricted to free indices
        # a_f = x_f / (1^T x_f)
        x_f_sum = float(x[free].sum())
        if x_f_sum < 1e-15:
            # Fallback
            a = np.zeros(n)
            a[free] = 1.0 / n_free
            break
        
        a_new = x / x_f_sum
        a_new[~free] = 0
        
        # Check KKT conditions: mu = 2M@a + lam*1 >= 0
        # lam from: 2M@a + lam*1 = 0 on free set
        # lam = -2 * (M@a)_free[0] / 1 = ... 
        # Actually: 2M@a = -lam*1 on free set
        # (2M@a)[i] = -lam for i in free
        # So lam = -(2M@a)[free[0]] (if n_free >= 1)
        
        grad_full = _hess_vec(a_new, active, n)
        if n_free >= 1:
            lam = -grad_full[free][0]  # This should be the same for all free indices
        else:
            break
        
        # Check multipliers for inactive indices
        mu_inactive = grad_full[~free] + lam  # = (2M@a)[inactive] + lam
        # Wait: KKT says 2M@a + lam*1 - mu = 0, so mu = 2M@a + lam*1
        # For inactive: mu_i = (2M@a)_i + lam
        # For feasibility: mu_i >= 0 for inactive, mu_i = 0 for active
        
        if (~free).any():
            min_mu = float(mu_inactive.min())
            if min_mu < -1e-10:
                # Infeasible: add the most violating index to the free set
                worst = int(np.where(~free)[0][np.argmin(mu_inactive)])
                free[worst] = True
                continue
        
        # Check if any free index should be removed (a_i < 0)
        if (a_new[free] < -1e-12).any():
            # Remove the most negative
            worst_free = int(np.where(free)[0][np.argmin(a_new[free])])
            free[worst_free] = False
            continue
        
        # Converged
        a = np.maximum(a_new, 0)
        a = a / a.sum()
        break
    
    return a


def _active_set_qp_loop(a_init, deadline, tol_frac=0.001, min_active=3, max_outer=200):
    """Outer active-set loop: repeatedly solve QP with updated active set."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Identify active set
        tol = tol_frac * c_max
        active_bool = c >= (c_max - tol)
        n_active = active_bool.sum()
        
        if n_active < min_active:
            # Include top min_active
            top_idx = np.argsort(c)[::-1][:min_active]
            active_bool = np.zeros_like(c, dtype=bool)
            active_bool[top_idx] = True
            n_active = min_active
        
        # Solve QP
        a_new = _solve_qp_active_set(a, active_bool, n, deadline)
        
        # Check improvement
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
        
        # Check convergence
        if np.allclose(a, a_new, atol=1e-10):
            break
        
        a = a_new
    
    return best_a, best_sc
