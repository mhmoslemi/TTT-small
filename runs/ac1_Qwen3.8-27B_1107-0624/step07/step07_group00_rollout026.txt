def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _conv_and_max(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return c, float(c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoothed_objective(z, T):
    """Compute smoothed max of conv(a,a) where a = softmax(z)."""
    a = _softmax_stable(z)
    c = _fft_convolve(a, a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c.max()
    return T * np.log(s + c.max() / T * 0)  # This is wrong, let me fix

# Actually, the smoothed max is: T * log(sum_k exp(c_k / T))
# = T * log(sum_k exp((c_k - c_max)/T)) + c_max
# = c_max + T * log(sum_k exp((c_k - c_max)/T))

def _smoothed_objective_value(z, T):
    a = _softmax_stable(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = e.sum()
    return c_max + T * np.log(s)

def _smoothed_gradient_z(z, T):
    """Gradient of smoothed objective w.r.t. z (log-space)."""
    a = _softmax_stable(z)
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = e.sum()
    p = e / s  # This is the "Boltzmann" distribution over convolution positions
    
    # Gradient of max_k conv(a,a)[k] w.r.t. a_j is approximately:
    # sum_k p_k * d(conv[a,k])/da_j = sum_k p_k * 2*a_{k-j} (when valid)
    # This is 2 * (p_rev * a)[j] where the convolution is a cross-correlation.
    
    # Let me compute: grad_a[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # = 2 * sum_{l=0}^{n-1} a[l] * p[l+j]  (where l = k-j)
    # This is a cross-correlation of a and p.
    
    # Using FFT: cross-correlation of a (length n) and p (length 2n-1)
    # corr[j] = sum_l a[l] * p[l+j]
    # We can compute this via: IFFT(conj(FFT(a)) * FFT(p)) shifted appropriately
    # Or more simply: it's a convolution of a_rev with p.
    
    a_rev = a[::-1]
    # conv(a_rev, p) has length n + (2n-1) - 1 = 3n - 2
    # The cross-correlation corr[j] = sum_l a[l]*p[l+j] = conv(a_rev, p)[n-1+j]
    # Let me verify: conv(a_rev, p)[m] = sum_i a_rev[i] * p[m-i] = sum_i a[n-1-i] * p[m-i]
    # Let l = n-1-i, so i = n-1-l:
    # = sum_l a[l] * p[m-n+1+l]
    # We want sum_l a[l] * p[l+j], so m-n+1 = j, m = n-1+j.
    # So corr[j] = conv(a_rev, p)[n-1+j].
    
    full_conv = _fft_convolve(a_rev, p)
    grad_a = 2.0 * full_conv[n-1:2*n-1]  # indices n-1 to 2n-2, that's n elements
    
    # Wait, full_conv has length n + (2n-1) - 1 = 3n-2.
    # We want indices n-1 to n-1+n-1 = 2n-2. That's n elements.
    # grad_a[j] = full_conv[n-1+j] for j=0,...,n-1.
    
    # Now convert to z-gradient:
    # a = softmax(z), so da_j/dz_k = a_j * (delta_jk - a_k)
    # grad_z = a * (grad_a - sum(a * grad_a))
    
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    return grad_z, a, c

def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, eval_interval=200):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        grad_z, a, c = _smoothed_gradient_z(z, T)
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and lr_mult > 0.1:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_smoothed(a_init, deadline, T, max_iter=5000, max_fun=2000):
    """Use L-BFGS on the smoothed objective in log-space."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    def objective(z):
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        c_shifted = (c - c_max) / T
        e = np.exp(c_shifted)
        s = e.sum()
        val = c_max + T * np.log(s)
        # Check time
        return val
    
    def gradient(z):
        grad_z, a, c = _smoothed_gradient_z(z, T)
        # Scale: the objective is in "conv" units, but we want to minimize
        # the smoothed max which is in the same units.
        # Actually, our objective is the smoothed max of conv, and the gradient
        # we computed is for max_k conv. The gradient of the smoothed version
        # is sum_k p_k * grad_k, which is what we computed.
        # But we need to account for the T factor... actually no, the gradient
        # of c_max + T*log(sum exp((c-c_max)/T)) w.r.t. z is:
        # sum_k p_k * grad_k where p_k = exp((c_k-c_max)/T)/sum exp((c_j-c_max)/T)
        # which is exactly what we computed in _smoothed_gradient_z.
        return grad_z
    
    start_time = time.monotonic()
    try:
        result = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'maxfun': max_fun,
                                  'ftol': 1e-15, 'gtol': 1e-12})
    except Exception as e:
        print(f"L-BFGS error: {e}")
        return best_a, best_score
    
    a_opt = _softmax_stable(result.x)
    sc = _score_norm(a_opt)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_opt.copy()
    
    return best_a, best_score

def _lbfgs_adaptive(a_init, deadline, T_start, T_end, n_phases=5):
    """Run L-BFGS with decreasing temperature."""
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for i in range(n_phases):
        if time.monotonic() > deadline:
            break
        frac = i / max(1, n_phases - 1)
        T = T_start * (T_end / T_start) ** frac
        phase_deadline = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) / (n_phases - i)))
        
        a_opt, sc = _lbfgs_smoothed(best_a, phase_deadline, T=T,
                                     max_iter=2000, max_fun=1000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        print(f"  L-BFGS phase {i}: T={T:.2e}, score={_score(best_a):.10f}, t={time.monotonic()-t_start:.0f}s")
    
    return best_a, best_score

def _coord_descent(a_init, deadline, max_rounds=100):
    """Fast coordinate descent on the max convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        # Focus on indices where conv is near max
        threshold = c_max * 0.999
        active_k = np.where(c >= threshold)[0]
        
        for k in active_k:
            # For each k, the indices j that affect conv[k] are those where
            # conv[k] includes a_j * a_{k-j}
            if k < n:
                js = range(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = range(k - (n - 1), n)
            
            for j in js:
                # Try reducing a[j]
                for scale in [0.7, 0.5, 0.8]:
                    a_cand = a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    a_cand /= s
                    c_cand_max = _fft_convolve(a_cand, a_cand).max()
                    sc = 2.0 * n * float(c_cand_max)
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)

def _block_descent(a_init, deadline, block_size=8, max_rounds=200):
    """Block coordinate descent: optimize groups of adjacent elements."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(best_a)
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            # Optimize a[start:end] while keeping the rest fixed
            # Simple approach: try perturbing the block
            best_block = a[start:end].copy()
            best_sc = best_score
            
            for _ in range(3):
                # Try random perturbation of the block
                pert = best_block * (1.0 + 0.1 * np.random.uniform(-1, 1, end - start))
                pert = np.maximum(pert, 0)
                a_cand = a.copy()
                a_cand[start:end] = pert
                s = a_cand.sum()
                a_cand /= s
                sc = 2.0 * n * float(_fft_convolve(a_cand, a_cand).max())
                if sc < best_sc:
                    best_sc = sc
                    best_block = pert.copy()
            
            if best_sc < best_score * (1.0 - 1e-12):
                a[start:end] = best_block
                a = a / a.sum()
                best_a = a.copy()
                best_score = best_sc
                improved = True
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
