Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set Adam, LP-based direction finding, multi-scale exploration,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoothed_gradient(a, c, T):
    """Smoothed (softmax) gradient of max(c)."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        W = np.zeros_like(c)
        W[np.argmax(c)] = 1.0
    else:
        W = e / s
    W_rev = W[::-1]
    conv_full = _fft_convolve(a, W_rev)
    g = 4.0 * n * conv_full[n-1:2*n-1][::-1]
    return g

def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, top_k=5, eval_interval=200, 
                   clip_norm=0.5, use_backtrack=True, mode='smooth'):
    """Adam optimizer in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if mode == 'active':
            g = _active_gradient(a, c, top_k)
        else:
            frac = min(1.0, t / max_iter)
            T = T_start * (T_end / T_start) ** frac
            g = _smoothed_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr_start * (lr_end / lr_start) ** min(1.0, t/max_iter) * lr_mult * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lp_direction(f, M, n, max_lp_iters=50, time_limit=30):
    """
    Find direction g that maximizes sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Uses constraint generation (only active constraints).
    Returns the optimal g vector.
    """
    t_start = time.monotonic()
    
    # Initialize active set with the most violating constraints
    c = _fft_convolve(f, f)
    active = list(np.where(c > M * 0.95)[0])
    
    g_opt = np.zeros(n)
    
    for lp_iter in range(max_lp_iters):
        if time.monotonic() - t_start > time_limit:
            break
        if len(active) == 0:
            active = list(range(2*n-1))
        
        # Build constraint matrix (sparse)
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        # We want: A @ g <= M for all k in active
        # Maximize: sum(g) subject to A @ g <= M, g >= 0
        
        rows = []
        cols = []
        vals = []
        for k in active:
            for j in range(max(0, k-n+1), min(k+1, n)):
                i = k - j  # This is the index into f
                if 0 <= i < n:
                    rows.append(k)  # constraint index (local)
                    cols.append(j)
                    vals.append(f[i])
        
        if len(rows) == 0:
            g_opt = np.ones(n) / n
            return g_opt
        
        # Re-index constraints
        k_to_local = {k: i for i, k in enumerate(active)}
        local_rows = [k_to_local[r] for r in rows]
        
        A = csr_matrix((vals, (local_rows, cols)), shape=(len(active), n))
        b = np.full(len(active), M)
        c_obj = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g_opt = res.x
        
        # Check all constraints
        c_new = _fft_convolve(f, g_opt)
        violated = np.where(c_new > M + 1e-8)[0]
        
        if len(violated) == 0:
            break
        
        # Add violated constraints to active set
        active = list(set(active))  # deduplicate
        for v in violated:
            if v not in active:
                active.append(v)
    
    return g_opt

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _cd_polish(a, deadline, max_rounds=100):
    """Coordinate descent polish."""
    n = len(a)
    best_a = a.astype(float).copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = c.max()
    tol = c_max * 1.0005
    active = np.where(c > tol)[0]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for k in active:
            if time.monotonic() > deadline:
                break
            # Find which a[j] affect c[k]
            # c[k] = sum_i a[i]*a[k-i], so a[j] appears in c[k] when k-j is in [0,n-1]
            # i.e., j is in [k-(n-1), k] intersect [0, n-1]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            
            for j in range(j_start, j_end + 1):
                for scale in [0.7, 0.85, 1.15, 1.3, 0.5]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-10):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = c.max()
                        tol = c_max * 1.0005
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_a, _score_norm(best_a)

                f_cand /= f_cand.sum()
                sc_cand = _score_norm(f_cand)
                if sc_cand < best_sc_lp:
                    best_sc_lp = sc_cand
                    best_t = t_step
            
            if best_t > 0:
                f_new = (1 - best_t) * f + best_t * g_scaled
                f_new = np.maximum(f_new, 0)
                f_new /= f_new.sum()
                f = f_new
                M = float(_fft_convolve(f, f).max())
                n_lp_success += 1
                full_sc = _score(f)
                if full_sc < best_score:
                    best_score = full_sc
                    best_a = f.copy()
                print(f"  LP iter {n_lp_success}: M={M:.10f}, score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            else:
                # LP didn't help, break
                break
        
        print(f"Phase 2 LP done: {n_lp_success} iterations, score={best_score:.10f}")
    
    # Phase 3: Active-set Adam (700s - 1200s)
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_act, sc_act = _adam_optimize(best_a, phase3_end,
                                        T_start=0.001 * max_c, T_end=0.00001 * max_c,
                                        lr_start=0.02, lr_end=0.001,
                                        max_iter=200000, top_k=5,
                                        eval_interval=150, mode='active')
        full_sc = _score(a_act)
        print(f"Phase 3 Active Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_act.copy()
    
    # Phase 4: Try different n (1200s - 1500s)
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end:
        # Try n=4096 if current is larger
        target_ns = [4096, 8192]
        n_cur = len(best_a)
        for target_n in target_ns:
            if target_n == n_cur or time.monotonic() > phase4_end - 60:
                continue
            a_target = _interp_to_length(best_a, target_n)
            c_t = _fft_convolve(a_target, a_target)
            max_t = float(c_t.max())
            mid_end = min(phase4_end - 30, time.monotonic() + max(60, (phase4_end - time.monotonic()) / 2))
            a_t_opt, sc_t = _adam_optimize(a_target, mid_end,
                                            T_start=0.002 * max_t, T_end=0.00002 * max_t,
                                            lr_start=0.02, lr_end=0.001,
                                            max_iter=100000, top_k=8,
                                            eval_interval=200, mode='smooth')
            full_sc_t = _score(a_t_opt)
            print(f"Phase 4 n={target_n}: score={full_sc_t:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_t) and full_sc_t < best_score:
                best_score = full_sc_t
                best_a = a_t_opt.copy()
    
    # Phase 5: Fine polish (1500s - 1700s)
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_pol, sc_pol = _adam_optimize(best_a, phase5_end,
                                        T_start=0.0005 * max_cf, T_end=0.000005 * max_cf,
                                        lr_start=0.01, lr_end=0.0003,
                                        max_iter=200000, top_k=3,
                                        eval_interval=100, mode='active')
        full_sc_pol = _score(a_pol)
        print(f"Phase 5 Polish: score={full_sc_pol:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_pol) and full_sc_pol < best_score:
            best_score = full_sc_pol
            best_a = a_pol.copy()
