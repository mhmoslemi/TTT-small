import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _hankel_matvec(w, v, n):
    """Compute H*v where H_{ij} = w[i+j] for 0<=i<n, 0<=j<n, 0<=i+j<2n-1.
    w has length 2n-1, v has length n, result has length n.
    """
    u = v[::-1]  # reverse v
    N = _next_pow2(n + len(w))
    fu = np.fft.rfft(u, N)
    fw = np.fft.rfft(w, N)
    r = np.fft.irfft(fu * fw, N)
    return r[n-1 : 2*n-1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Multi-resolution power-sum optimization with Newton active-set refinement
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences a.
Uses FFT-based Hankel Hessian-vector products, power-sum objectives with
continuation, and a multi-scale strategy from n=1024 to n=8192.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _hankel_matvec(w, v, n):
    """Compute H*v where H_{ij} = w[i+j] for 0<=i<n,0<=j<n. w has length 2n-1."""
    u = v[::-1]
    N = _next_pow2(n + len(w) - 1)
    fu = np.fft.rfft(u, N)
    fw = np.fft.rfft(w, N)
    r = np.fft.irfft(fu * fw, N)
    return r[n - 1:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _power_sum_obj(z, p, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return 2.0 * n * c_max
    normalized = c / c_max
    log_sum = np.log(np.clip(np.sum(normalized ** p), 1e-300, None))
    fs = c_max * np.exp(log_sum / p)
    return 2.0 * n * fs


def _power_sum_grad(z, p, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(z)
    normalized = c / c_max
    w = (normalized ** (p - 1)) / np.clip(np.sum(normalized ** p), 1e-300, None)
    hw = _hankel_matvec(w, a, n)
    fs = c_max * np.exp(np.log(np.clip(np.sum(normalized ** p), 1e-300, None)) / p)
    g_a = 2.0 * fs * hw
    f_val = 2.0 * n * fs
    grad_z = a * (g_a - (g_a @ a))
    return grad_z


def _run_power_sum_lbfgs(a_init, p_list, deadline, min_time_remaining=30.0):
    """Run L-BFGS with power-sum continuation."""
    n = len(a_init)
    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    est_time_per_iter = 0.002 * max(1, n / 2048.0)

    for p in p_list:
        t_now = time.monotonic()
        if t_now + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - t_now
        if remaining < 5:
            break
        max_iter = min(20000, max(100, int(remaining / (est_time_per_iter * 6))))

        def fun_local(z_local):
            return _power_sum_obj(z_local, p, n)

        def jac_local(z_local):
            return _power_sum_grad(z_local, p, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 100}

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            sc_cur = _score(current_a)

def _newton_active_set(a_init, deadline, max_iter=200, tol_frac=0.001):
    """Newton's method on active-set subproblem with FFT-based Hankel Hessian."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-14)
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        grad = 2.0 * _hankel_matvec(w, a, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break

        mu = 1e-6
        b = -grad_t

        x = np.zeros(n)
        r = b.copy()
        p_vec = r.copy()
        rs = float(np.dot(r, r))
        b_norm = float(np.linalg.norm(b))
        if b_norm < 1e-30:
            break

        for _ in range(150):
            Hv = 2.0 * _hankel_matvec(w, p_vec, n) + mu * p_vec
            pH = float(np.dot(p_vec, Hv))
            if abs(pH) < 1e-30:
                break
            alpha = rs / pH
            x = x + alpha * p_vec
            r = r - alpha * Hv
            r_new = float(np.dot(r, r))
            if np.sqrt(r_new) < 1e-10 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p_vec = r + beta_cg * p_vec
            rs = r_new

        dx = x - np.mean(x)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        converged_ls = False
        for _ in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-10 * alpha * gnorm:
                a_new = a_try
                converged_ls = True
                break
            alpha *= 0.5
        if not converged_ls:
            break

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _upscale_by_repeat(a_coarse, n_target):
    """Upscale by repeating elements, then smooth slightly."""
    m = len(a_coarse)
    r = n_target // m
    if r < 2:
        a = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, m), a_coarse)
        a = np.maximum(a, 0)
        return a / a.sum() if a.sum() > 0 else a_coarse
    a = np.repeat(a_coarse, r)
    if len(a) > n_target:
        a = a[:n_target]
    elif len(a) < n_target:
        leftover = n_target - len(a)
        a = np.concatenate([a, a_coarse[:leftover]])
    # Light smoothing to help optimization
    kernel = np.ones(5) / 5.0
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0)
    if a_smooth.sum() < 1e-10:
        a_smooth = a
    return a_smooth / a_smooth.sum()

            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-resolution power-sum optimizer with Newton active-set refinement.
    Strategy:
      1. Start at n=1024, optimize with power-sum L-BFGS + Newton
      2. Upscale to n=4096, optimize
      3. Upscale to n=8192, optimize
      4. Final Newton active-set polish with tight tolerance
      5. Subgradient polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = np.ones(1024) / 1024.0
        update_best(parent, _score(parent), "init")

    # === Phase 1: Coarse n=1024 ===
    print(f"\n[Phase 1] Coarse n=1024 (t={time.monotonic()-t_start:.1f}s)")
    n1 = 1024
    if parent is not None and n_parent != n1:
        a1 = np.interp(np.linspace(0, 1, n1), np.linspace(0, 1, n_parent), parent)
        a1 = np.maximum(a1, 0)
        a1 = a1 / a1.sum()
    else:
        a1 = parent.copy() if parent is not None else np.ones(n1) / n1

    if time.monotonic() < deadline - 600:
        p_list1 = [8, 16, 32, 64]
        a1 = _run_power_sum_lbfgs(a1, p_list1, deadline, min_time_remaining=500)
        update_best(a1, _score(a1), "ps_1024")

        if time.monotonic() < deadline - 500:
            a1n, scn = _newton_active_set(a1, min(deadline - 400, time.monotonic() + 60), max_iter=100)
            update_best(a1n, scn, "newton_1024")

    # === Phase 2: Medium n=4096 ===
    print(f"\n[Phase 2] Medium n=4096 (t={time.monotonic()-t_start:.1f}s)")
    n2 = 4096
    base = best_seq if best_seq is not None else a1
    if len(base) != n2:
        a2 = _upscale_by_repeat(base, n2)
    else:
        a2 = base.copy()

    if time.monotonic() < deadline - 300:
        p_list2 = [16, 32, 64, 128]
        a2 = _run_power_sum_lbfgs(a2, p_list2, deadline, min_time_remaining=200)
        update_best(a2, _score(a2), "ps_4096")

        if time.monotonic() < deadline - 200:
            a2n, scn2 = _newton_active_set(a2, min(deadline - 150, time.monotonic() + 80), max_iter=100)
            update_best(a2n, scn2, "newton_4096")

    # === Phase 3: Fine n=8192 ===
    print(f"\n[Phase 3] Fine n=8192 (t={time.monotonic()-t_start:.1f}s)")
    n3 = 8192
    base = best_seq if best_seq is not None else a2
    if len(base) != n3:
        a3 = _upscale_by_repeat(base, n3)
    else:
        a3 = base.copy()

    if time.monotonic() < deadline - 150:
        p_list3 = [32, 64, 128, 256]
        a3 = _run_power_sum_lbfgs(a3, p_list3, deadline, min_time_remaining=80)
        update_best(a3, _score(a3), "ps_8192")

        if time.monotonic() < deadline - 80:
            a3n, scn3 = _newton_active_set(a3, min(deadline - 50, time.monotonic() + 60), max_iter=150)
            update_best(a3n, scn3, "newton_8192")

    # === Phase 4: Tight Newton refinement ===
    print(f"\n[Phase 4] Tight Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_tight, sc_tight = _newton_active_set(
            best_seq, min(deadline - 30, time.monotonic() + 80),
            max_iter=300, tol_frac=0.0005)
        update_best(a_tight, sc_tight, "tight_newton")

        if sc_tight < best_score * 1.001 and time.monotonic() < deadline - 30:
            a_tight2, sc_tight2 = _newton_active_set(
                a_tight, min(deadline - 20, time.monotonic() + 50),
                max_iter=200, tol_frac=0.0002)
            update_best(a_tight2, sc_tight2, "tighter_newton")

    # === Phase 5: Subgradient polish ===
    print(f"\n[Phase 5] Subgrad polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_sg, sc_sg = _subgrad_polish(
            best_seq, min(deadline - 10, time.monotonic() + 60),
            n_max_steps=10000, lr_init=0.005)
        update_best(a_sg, sc_sg, "subgrad")

        if time.monotonic() < deadline - 8:
            a_sg2, sc_sg2 = _newton_active_set(
                best_seq, min(deadline - 5, time.monotonic() + 30),
                max_iter=100, tol_frac=0.0003)
            update_best(a_sg2, sc_sg2, "final_newton")

    # === Phase 6: Try different n values ===
    print(f"\n[Phase 6] Alternative n (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        for n_alt in [6144, 12288]:
            if time.monotonic() > deadline - 10:
                break
            if len(best_seq) != n_alt:
                a_alt = _upscale_by_repeat(best_seq, n_alt)
            else:
                a_alt = best_seq.copy()
            p_list_alt = [64, 128, 256]
            a_alt = _run_power_sum_lbfgs(a_alt, p_list_alt, deadline, min_time_remaining=5)
            update_best(a_alt, _score(a_alt), f"ps_{n_alt}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
