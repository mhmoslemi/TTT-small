**Multiplicative Weight Descent (MWD)**:
- State: a (normalized, sum = 1, a >= 0)
- Compute conv = conv(a, a)
- Compute weights w_k = softmax(conv / beta) (or top-k indicator)
- Gradient: grad[i] = sum_k w_k * 2*a[k-i] (with validity check)
- Update: a_new[i] = a[i] * (1 - eta * grad[i]) (additive) or a[i] * exp(-eta * grad[i]) (multiplicative)
- Renormalize

The multiplicative update is:

This is exactly what the log-space gradient descent does (with step size eta). The advantage of MWD is that it naturally maintains non-negativity.

With sum(a) = 1 and a >= 0, max(a) <= 1 (actually much less for well-distributed a). So L ≈ 2*max(a) which is small.

"""
Advanced multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses L-BFGS with adaptive log-sum-exp smoothing, multiplicative weight descent
with momentum, QP-based subgradient steps, and fine active-set polishing.
Explores multiple sequence lengths and diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_gradient(a, w):
    """Compute gradient of sum(w_k * conv(a,a)_k) w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return _conv_gradient(a, w)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose_prefix=""):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta schedule based on convolution spread
    c_sorted = np.sort(c_cur)[::-1]
    top_k = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[top_k - 1]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    # More aggressive geometric schedule with more steps
    ratios = [1.0, 0.5, 0.3, 0.18, 0.1, 0.055, 0.03, 0.016, 0.009,
              0.005, 0.0028, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max iterations based on remaining time and n
        est_time_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 15))))

        def obj(z_in):
            return _smoobj(z_in, beta, n)

        def grad(z_in):
            return _smograd(z_in, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose_prefix:
                    print(f"{verbose_prefix} beta={beta:.2e} sc={sc_new:.10f} iter={i}")
        except Exception:
            pass

    return current_a


def _mwd_optimize(a_init, deadline, min_time_remaining=60.0, verbose_prefix=""):
    """Multiplicative Weight Descent with momentum and adaptive parameters."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Momentum buffer
    momentum = np.zeros(n)
    beta_m = 0.9  # momentum decay

    # Step size (in log space)
    eta = 0.3
    eta_decay = 0.995

    # Active set threshold (fraction of max)
    top_frac = 0.1  # top 10% of convolution values

    no_improve = 0
    max_stall = 200

    step_count = 0
    while time.monotonic() + min_time_remaining < deadline:
        step_count += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_stall:
                break

        # Compute active set weights (top fraction of conv values)
        threshold_idx = max(1, int(len(c) * top_frac))
        top_vals = np.sort(c)[::-1][:threshold_idx]
        threshold = top_vals[-1]
        active = c >= threshold
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient
        grad = _conv_gradient(a, w)

        # Normalize gradient (make it zero-mean in the "effective" sense)
        # Use the constraint sum(a)=1: the gradient direction should be orthogonal to 1
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        # Momentum update
        momentum = beta_m * momentum + (1 - beta_m) * (grad_t / gnorm)

        # Adaptive step size: reduce if we overshoot
        # In log-space: z_new = z - eta * momentum_normalized
        # a_new = a * exp(-eta * momentum) / sum(a * exp(-eta * momentum))
        z_update = -eta * momentum / (np.max(np.abs(momentum)) + 1e-15)
        scale = a * np.exp(z_update * np.max(np.abs(momentum)))
        s_scale = scale.sum()
        if s_scale < 1e-300:
            eta *= 0.5
            continue
        a_try = scale / s_scale
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            eta *= 0.5
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-15:
            a = a_try
            eta = min(0.5, eta * 1.01)
        else:
            # Try smaller step
            improved = False
            for shrink in [0.5, 0.25, 0.1]:
                z_update_small = -eta * shrink * momentum / (np.max(np.abs(momentum)) + 1e-15)
                scale_small = a * np.exp(z_update_small * np.max(np.abs(momentum)))
                s_small = scale_small.sum()
                if s_small < 1e-300:
                    continue
                a_small = scale_small / s_small
                a_small = np.maximum(a_small, 0.0) / max(a_small.sum(), 1e-15)
                sc_small = _score(a_small)
                if sc_small < current_sc - 1e-15:
                    a = a_small
                    eta = min(eta, eta * shrink * 1.5)
                    improved = True
                    break
            if not improved:
                eta *= 0.8
                if eta < 0.01:
                    break

        # Decay eta slowly
        eta *= eta_decay
        # Increase momentum over time
        beta_m = min(0.98, beta_m + 0.0005)

    return best_a


def _active_set_polish(a_init, deadline, min_time_remaining=10.0,
                       tol_frac_start=0.005, tol_frac_end=0.00003,
                       lr_init=0.03, verbose_prefix=""):
    """Fine active-set gradient descent with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Adaptive active set size
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Use softmax weights within active set for smoother gradient
        c_active = c[active] - (c_max - tol)
        w_active = np.exp(np.clip(c_active / max(tol * 0.5, 1e-15), -300, 300))
        w_active = w_active / w_active.sum()
        w = np.zeros_like(c)
        w[active] = w_active

        grad = _conv_gradient(a, w)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Line search with backtracking
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_fraс_end, tol_frac * 0.5)
                lr = lr_init * 0.4
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.08
        return a / a.sum()
    elif kind == "triangular_inv":
        # Inverse triangular: peaks at edges
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        # Flat in center, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n) * 0.8 + 0.4 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 10))
        amp = 0.04 * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _qp_subgradient_step(a, active_mask, target_t, step_size):
    """
    Take a QP-inspired step: minimize (1/2)||a - a_cur||^2_D + t
    subject to linearized conv constraints and a >= 0, sum(a) = 1.
    Done via projected gradient in the dual.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    # For active positions, compute the linearized constraint gradient
    # conv_k(a) linearized: 2 * (a_cur * a)[k] - conv_k(a_cur)
    # We want to push a in the direction that decreases the active conv values

    w = np.zeros_like(c)
    w[active_mask] = 1.0 / active_mask.sum()

    # Gradient of the "violation" at active positions
    grad = _conv_gradient(a, w)
    grad_t = grad - np.mean(grad)

    # Take a step
    alpha = step_size
    a_try = a - alpha * grad_t
    a_try = np.maximum(a_try, 0.0)
    s_try = a_try.sum()
    if s_try < 1e-15:
        return a
    a_try = a_try / s_try
    return a_try


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining L-BFGS, MWD with momentum, active-set polishing,
    QP-inspired steps, and diverse restarts with multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: Main L-BFGS on parent ===
    print(f"\n[P1] L-BFGS main n={n_best} (t={time.monotonic()-t_start:.1f}s)")
    remain_after_p1 = 900.0
    if deadline - time.monotonic() > remain_after_p1:
        a_main = _run_lbfgs(parent, deadline, min_time_remaining=remain_after_p1,
                            verbose_prefix="  [P1]")
        sc_main = _score(a_main)
        if sc_main < best_score:
            best_seq = a_main.copy()
            best_score = sc_main
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [lbfgs_main]")

    # === Phase 2: MWD with momentum ===
    print(f"\n[P2] MWD momentum (t={time.monotonic()-t_start:.1f}s)")
    remain_after_p2 = 600.0
    if deadline - time.monotonic() > remain_after_p2 + 60:
        mwd_deadline = deadline - remain_after_p2
        a_mwd = _mwd_optimize(best_seq, mwd_deadline, min_time_remaining=60.0,
                              verbose_prefix="  [P2]")
        sc_mwd = _score(a_mwd)
        if sc_mwd < best_score:
            best_seq = a_mwd.copy()
            best_score = sc_mwd
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [mwd]")

    # === Phase 3: L-BFGS again (refined starting point) ===
    print(f"\n[P3] L-BFGS refine (t={time.monotonic()-t_start:.1f}s)")
    remain_after_p3 = 450.0
    if deadline - time.monotonic() > remain_after_p3 + 30:
        lbfgs_deadline = deadline - remain_after_p3
        a_ref = _run_lbfgs(best_seq, lbfgs_deadline, min_time_remaining=30.0,
                           verbose_prefix="  [P3]")
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_seq = a_ref.copy()
            best_score = sc_ref
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [lbfgs_refine]")

    # === Phase 4: Active-set polish (coarse) ===
    print(f"\n[P4] Active-set coarse (t={time.monotonic()-t_start:.1f}s)")
    remain_after_p4 = 350.0
    if deadline - time.monotonic() > remain_after_p4 + 60:
        pol_deadline = min(deadline - remain_after_p4, time.monotonic() + 200)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           min_time_remaining=60.0,
                                           tol_frac_start=0.006, tol_frac_end=0.0001,
                                           lr_init=0.03, verbose_prefix="  [P4]")
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [polish_coarse]")

    # === Phase 5: MWD round 2 ===
    print(f"\n[P5] MWD round 2 (t={time.monotonic()-t_start:.1f}s)")
    remain_after_p5 = 250.0
    if deadline - time.monotonic() > remain_after_p5 + 60:
        mwd2_deadline = deadline - remain_after_p5
        a_mwd2 = _mwd_optimize(best_seq, mwd2_deadline, min_time_remaining=60.0)
        sc_mwd2 = _score(a_mwd2)
        if sc_mwd2 < best_score:
            best_seq = a_mwd2.copy()
            best_score = sc_mwd2
            print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [mwd2]")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 8192, 10240]
    if n_best not in n_candidates:
        n_candidates.insert(0, n_best)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == n_best:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)

        if time.monotonic() > deadline - 60:
            continue

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            n_best = n_try
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")

        elif sc_opt < best_score + 0.01 and time.monotonic() < deadline - 50:
            a_pol, sc_pol = _active_set_polish(a_opt, min(deadline - 45, time.monotonic() + 40),
                                               min_time_remaining=10.0,
                                               tol_frac_start=0.003, tol_frac_end=0.00008,
                                               lr_init=0.015)
            if sc_pol < best_score:
                best_seq = a_pol.copy()
                best_score = sc_pol
                n_best = n_try
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish]")

    # === Phase 7: Diverse restarts ===
    print(f"\n[P7] Diverse restarts (t={time.monotonic()-t_start:.1f;s}")
    if time.monotonic() < deadline - 40:
        restart_starts = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("tri_inv", _resample(_make_profile(n_best, "triangular_inv"), n_best)),
            ("bimo", _resample(_make_profile(n_best, "bimodal"), n_best)),
            ("flat_c", _resample(_make_profile(n_best, "flat_center"), n_best)),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 20:
                break
            if len(a_r) != n_best:
                a_r = _resample(a_r, n_best)
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=15.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}]")

    # === Phase 8: Ultra-fine final polish ===
    print(f"\n[P8] Final ultra-fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Very tight active-set
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           min_time_remaining=3.0,
                                           tol_frac_start=0.0005, tol_frac_end=0.00001,
                                           lr_init=0.008)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish_tight]")

        # Tiny L-BFGS with smallest beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.00005, 1e-10)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()

            def obj_final(z_in):
                return _smoobj(z_in, beta_fin, n)

            def grad_final(z_in):
                return _smograd(z_in, beta_fin, n)

            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(obj_final, z, jac=grad_final, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-17,
                                           'gtol': 1e-15, 'maxcor': 100, 'maxls': 60})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs_tiny]")
            except Exception:
                pass

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
