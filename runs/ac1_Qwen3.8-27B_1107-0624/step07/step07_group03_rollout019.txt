import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, args):
    beta, n = args
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, args):
    beta, n = args
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    betas = [base_beta * (0.4 ** i) for i in range(20)]
    # Ensure minimum beta
    min_beta = max_c * 0.00002
    betas = [max(b, min_beta) for b in betas]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    prev_sc = current_sc
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(6000, max(100, int(remaining / (est_time_per_iter * 6))))

        args = (beta, n)
        try:
            result = minimize(
                _lsobj, z,
                jac=_lsgrad,
                args=args,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()

                # Adaptive: if improvement was small, skip to smaller beta faster
                improvement = prev_sc - sc_new
                if improvement < 1e-8 and i < len(betas) - 2:
                    # Skip ahead
                    continue
            prev_sc = current_sc
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_lr = lr

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
                best_patience_lr = lr
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = max(best_patience_lr * 0.4, lr_init * 0.1)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.40]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "spectral":
        # Initialize based on random Fourier features
        rng_local = np.random.default_rng(12345)
        a = np.ones(n)
        for _ in range(5):
            freq = rng_local.integers(1, max(2, n // 8))
            amp = 0.1 * rng_local.uniform(0.5, 1.5)
            phase = rng_local.uniform(0, 2 * np.pi)
            a = a + amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _local_perturb_search(a_best, deadline, rng, n_trials=20, sigma=0.05):
    """Systematic local search with decreasing perturbation."""
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        sigma_try = sigma * (0.7 ** (trial // 5))
        a_pert = _perturb(best_a, max(sigma_try, 0.005), rng)
        # Quick L-BFGS refinement
        c_cur = _full_convolve(a_pert, a_pert)
        max_c = float(c_cur.max())
        beta = max(max_c * 0.01, 1e-10)
        z = np.log(np.maximum(a_pert, 1e-30))
        z = z - z.mean()
        args = (beta, n)
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(500, max(50, int(remaining / 2.0)))
        try:
            result = minimize(
                _lsobj, z,
                jac=_lsgrad,
                args=args,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-13:
                best_a = a_new
                best_sc = sc_new
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    total_budget = max(1.0, float(budget_s))
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main adaptive L-BFGS from parent
    print(f"\n[Phase 1] Main adaptive L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + min(450.0, total_budget * 0.25)
    a_main = _adaptive_lbfgs(parent, phase1_deadline, min_time_remaining=30.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Active-set Newton with KKT
    print(f"\n[Phase 2] Active-set Newton + KKT (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_deadline = min(deadline, time.monotonic() + 150)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=300, tol_active=0.001, kkt_check=True)
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 200:
            as_deadline2 = min(deadline - 200, time.monotonic() + 80)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_deadline2, max_iter=200, tol_active=0.0005, kkt_check=True)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=4000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_polish_1")

    # Phase 3: Shrinkage
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 250)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score and time.monotonic() < deadline - 200:
            a_shr2 = _adaptive_lbfgs(a_shr, min(deadline - 150, time.monotonic() + 100), min_time_remaining=50.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # Phase 4: Multi-n exploration (including non-power-of-2)
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192, 12000, 16384, 20000]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    # Prioritize n values around the parent
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None and n_parent != n_try:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        remaining_for_this = min(deadline - 60 - time.monotonic(), 200.0)
        if remaining_for_this < 20:
            continue

        a_opt = _adaptive_lbfgs(a_init, time.monotonic() + remaining_for_this, min_time_remaining=10.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 40, time.monotonic() + 50),
                                                  max_iter=100, tol_active=0.001, kkt_check=True)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 5: Diverse restarts from best
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("perturb_small", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_mid", lambda: _perturb(best_seq, 0.06, rng)),
            ("perturb_large", lambda: _perturb(best_seq, 0.12, rng)),
            ("fourier_3", lambda: _perturb_fourier(best_seq, rng, n_modes=3)),
            ("fourier_5", lambda: _perturb_fourier(best_seq, rng, n_modes=5)),
            ("fourier_8", lambda: _perturb_fourier(best_seq, rng, n_modes=8)),
        ]
        per_restart_time = min(60.0, (deadline - time.monotonic() - 20) / max(1, len(restart_configs)))
        for label, gen_fn in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            p_start = gen_fn()
            a_r = _adaptive_lbfgs(p_start, min(deadline - 15, time.monotonic() + per_restart_time),
                                   min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

    # Phase 6: Local perturb search around best
    print(f"\n[Phase 6] Local perturb search (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        local_deadline = min(deadline - 15, time.monotonic() + 120)
        a_local, sc_local = _local_perturb_search(best_seq, local_deadline, rng, n_trials=30, sigma=0.04)
        update_best(a_local, sc_local, "local_search")

    # Phase 7: Final intensive polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tight Newton
        as_deadline_final = min(deadline - 3, time.monotonic() + 80)
        a_fin, sc_fin = _active_set_newton(best_seq, as_deadline_final, max_iter=300, tol_active=0.0003, kkt_check=True)
        update_best(a_fin, sc_fin, "final_newton_tight")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n_best = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = max(mb * 0.00001, 1e-12)
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.05)))
                args = (beta_final, n_best)
                try:
                    result = minimize(
                        _lsobj, z,
                        jac=_lsgrad,
                        args=args,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 100, 'maxls': 300}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny_beta")
                except Exception:
                    pass

        # One more Newton after L-BFGS
        if time.monotonic() < deadline - 2:
            a_fin4, sc_fin4 = _active_set_newton(best_seq, min(deadline - 1, time.monotonic() + 30),
                                                  max_iter=100, tol_active=0.0002, kkt_check=True)
            update_best(a_fin4, sc_fin4, "final_newton_ultimate")

        # Final subgradient with very small lr
        if time.monotonic() < deadline - 2:
            a_fin5, sc_fin5 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=2000, lr_init=0.003)
            update_best(a_fin5, sc_fin5, "final_sg")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
