"""
Improved optimizer using Adam in log-space, adaptive log-sum-exp smoothing,
multi-scale initialization, and block coordinate refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score for sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lsmax_grad(z, beta, n):
    """Log-sum-exp objective and gradient in log-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        val = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        val = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return val, grad

    """Adam optimizer in log-space with log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    c_max = float(c_cur.max())
    if c_max < 1e-15:
        return a_init, _score(a_init)
    
    # Adaptive beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    beta_start = max(gap * 2, c_max * 0.1)
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Run Adam with decreasing beta
    n_phases = 8
    betas = []
    for i in range(n_phases):
        frac = i / n_phases
        beta = beta_start * (0.05 ** (i / max(1, n_phases - 1)))
        betas.append(max(beta, c_max * 0.0001))
    
    for phase, beta in enumerate(betas):
        t_phase_start = time.monotonic()
        max_time_phase = 60  # seconds per phase

def _pressure_equalize(a, deadline, n_iters=300, gamma=0.3):
    """KKT-motivated pressure equalization."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c_max = float(c.max())
        if c_max < 1e-30:
            break

        active = c >= (c_max - tol)
        if active.sum() < 2:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)
        
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        if it % 30 == 0:

                best_sc = sc
                best_a = a.copy()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-50 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "ripple":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.sin(2 * np.pi * 7 * t) + 0.05 * np.sin(2 * np.pi * 19 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
