import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(x - theta, 0)


def _grad_conv(a, d):
    """Gradient of (1/2)||conv(a,a) - d||^2 w.r.t. a.
    Returns vector of length n."""
    n = len(a)
    a_rev = a[::-1]
    dc = _full_convolve(d, a_rev)
    return dc[n - 1:2 * n - 1]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _grad_conv(a, w)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=200, tol=1e-14):
    """Solve H*x = rhs using CG."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)

        # Backtracking line search
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            if c_max_try <= c_max - 1e-15:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-14:
                break

        if not step_done:
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max - 1e-15:
                    a = a_try
                    break
                alpha *= 0.5

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, max_steps=20000):
    """Nesterov-accelerated subgradient method with dynamic active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    base_lr = 0.05
    gamma = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(y)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Dynamic active set with wider tolerance
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        h = _grad_conv(a, w)
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        t_k = 2.0 / (np.sqrt(i + 3.0) + 1.0)
        lr = base_lr / (np.sqrt(i + 1.0) + 1.0)
        y_new = y - lr * grad
        y_new = np.clip(y_new, -60.0, 60.0)
        z_new = y_new + (t_k - 1.0) * (y_new - y)
        z_new = z_new - z_new.mean()

        if i % 50 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    y = z.copy()
                    patience = 0
                    base_lr = max(base_lr * 0.7, 0.005)

        z = z_new
        y = y_new

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.02
        return a / a.sum()
    elif kind == "sinc_like":
        t = np.linspace(-1, 1, n)
        a = np.sinc(1.5 * t) ** 2
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining ADMM, L-BFGS, Nesterov subgradient,
    and active set Newton for minimizing the convolution peak ratio.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: ADMM from parent (major time allocation)
    print(f"\n[Phase 1] ADMM solver (t={time.monotonic() - t_start:.1f}s)")
    admm_deadline = deadline - 600.0
    a_admm, sc_admm = _admm_solve(parent, admm_deadline, rho_init=5.0,
                                  max_iter=100000, inner_K=40)
    update_best(a_admm, sc_admm, "admm_main")

    # Phase 2: L-BFGS refinement on ADMM result
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_lb = _run_lbfgs(a_admm, deadline, min_time_remaining=300.0)
        update_best(a_lb, _score(a_lb), "lbfgs_ref")

    # Phase 3: Active set Newton polish
    print(f"\n[Phase 3] Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_np, sc_np = _newton_active_polish(best_seq, min(deadline - 200, time.monotonic() + 200))
        update_best(a_np, sc_np, "newton_1")

    # Phase 4: Nesterov subgradient
    print(f"\n[Phase 4] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_ns, sc_ns = _nesterov_subgrad(best_seq, min(deadline - 150, time.monotonic() + 100))
        update_best(a_ns, sc_ns, "nesterov_1")

    # Phase 5: Second Newton polish
    print(f"\n[Phase 5] Newton polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_np2, sc_np2 = _newton_active_polish(best_seq, min(deadline - 100, time.monotonic() + 80), max_rounds=50)
        update_best(a_np2, sc_np2, "newton_2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [16384, 32768, 4096]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        # Quick ADMM
        if time.monotonic() < deadline - 60:
            a_ad, sc_ad = _admm_solve(a_init, min(deadline - 50, time.monotonic() + 120),
                                      rho_init=5.0, max_iter=5000, inner_K=20)
            update_best(a_ad, sc_ad, f"admm_n{n_try}")
            # L-BFGS refinement
            if time.monotonic() < deadline - 40:
                a_lb = _run_lbfgs(a_ad, deadline, min_time_remaining=40.0)
                update_best(a_lb, _score(a_lb), f"lbfgs_n{n_try}")
                # Quick Newton
                if time.monotonic() < deadline - 20:
                    a_np = _newton_active_polish(a_lb, min(deadline - 10, time.monotonic() + 30), max_rounds=20)
                    update_best(a_np, _score(a_np), f"np_n{n_try}")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        restart_configs = [
            (0.05, "tapered"),
            (0.15, "sinc_like"),
            (0.10, "gauss"),
        ]
        for i, (sigma, kind) in enumerate(restart_configs):
            if time.monotonic() > deadline - 20:
                break
            if i < 2:
                p_start = _perturb(best_seq, sigma, rng)
            else:
                p_start = _make_profile(n_best, kind)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}_{kind}")

            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 10:
                a_rp, _ = _newton_active_polish(a_r, min(deadline - 5, time.monotonic() + 20), max_rounds=30)
                update_best(a_rp, _score(a_rp), f"restart_{i}_np")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _newton_active_polish(best_seq, deadline - 1, max_rounds=200)
        update_best(a_fin, sc_fin, "final_newton")

        # Very tight L-BFGS
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 2 - time.monotonic()
            if remaining > 1:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    result = minimize(
                        _smoobj_wrapper_factory(z, beta_fin, n),
                        z,
                        jac=_smograd_wrapper_factory(z, beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 100}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
