    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

So the peak of the circular conv would be 1/n, same as uniform! And the linear conv would have the same peak (since the delta is well within the linear conv range).

So this doesn't help. The flat-magnitude FFT gives the same peak as uniform for the circular conv.

But for the LINEAR conv, the peak might be different! The linear conv "cuts off" the circular wrap-around. If the delta in the circular conv is at position p, then in the linear conv, it appears at position p (if p < n) or wraps around differently.

Actually, I think for linear convolution, having a flat |FFT| spectrum means the linear conv is approximately flat (with edge effects). The peak would be approximately 1/n (the average), potentially slightly higher due to edge effects.

This doesn't give us a better starting point than uniform. The key to getting below 2 is ASYMMETRY in the time domain, which corresponds to COMPLEX phase in the frequency domain.

Let me try a different frequency-domain approach:
- Choose FFT(a) to have a specific phase pattern that, when squared (for the conv), gives a spectrum whose inverse is "anti-peaked"
- Specifically, we want IFFT(FFT(a)²) to have its maximum BELOW 1/n
- This means FFT(a)² should have a spectrum that, when inverted, gives values below 1/n everywhere
- Equivalently, FFT(a)² should be "anti-flat" - concentrated away from DC

But FFT(a)² has DC component = (sum a)² = 1 (if normalized). And the rest of the spectrum must compensate. If we make the non-DC components negative (which is possible since FFT(a)² can be complex), we can reduce the peak of the inverse.

This is getting too theoretical. Let me just go with the optimization approach.

"""
Nesterov subgradient optimizer with peak targeting for minimum peak self-convolution.
Different from previous: uses exact subgradients, simplex projection, 
adaptive active set, and targeted coordinate descent.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n + m - 1]


def _conv2(a):
    return _fft_conv(a, a)


def _max_conv(a):
    c = _conv2(a)
    return float(c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * _max_conv(a) / (s * s))


def _subgrad_active(a, active_lags):
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= max(1, len(active_lags))
    return g


def _clip_norm(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s


def _fista(a_init, deadline, max_iter=300000, lr_start=0.04, lr_end=0.002,

    n = len(a_init)
    x = a_init.copy()
    x = x / x.sum()
    x_prev = x.copy()
    y = x.copy()
    t_k = 1.0
    best_x = x.copy()
    best_f = _max_conv(x)
    stall = 0
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Periodic learning rate decay
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        c = _conv2(y)
        f_y = float(c.max())
        
        # Active set
        top = np.argsort(c)[::-1][:active_k]
        thresh = f_y * thresh_frac
        active = top[c[top] >= thresh]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        g = _subgrad_active(y, active)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new
        
        y_new = x + beta * (x - x_prev)
        x_new = _clip_norm(y_new - lr * g)
        
        x_prev = x
        x = x_new
        t_k = t_new
        y = x + beta * (x - x_prev)
        
        # Periodic evaluation
        if t % 100 == 0 or t < 20:
            f_x = _max_conv(x)
            if f_x < best_f * (1 - 1e-10):
                best_f = f_x
                best_x = x.copy()
                stall = 0
            else:
                stall += 1
        
        # Restart if stalled
        if stall > restart_every // 100:
            rng_restart = np.random.default_rng(t)
            x = best_x + 0.01 * rng_restart.standard_normal(n)
            x = _clip_norm(x)
            x_prev = x.copy()
            y = x.copy()
            t_k = 1.0
            lr_start *= 0.8
            stall = 0
    
    f_x = _max_conv(x)
    if f_x < best_f:
        best_f = f_x
        best_x = x.copy()
    return best_x, best_f
