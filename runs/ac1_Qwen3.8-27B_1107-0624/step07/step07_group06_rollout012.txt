    from scipy.optimize import linprog
    
    # Variables: [a_0, a_1, ..., a_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    A_ub_rows = []
    b_ub_vals = []
    for k in active_indices:
        kernel = _get_kernel(a_cur, k, n)
        row = np.zeros(n + 1)
        row[:n] = 2.0 * kernel
        row[n] = -1.0  # -t

        b_ub_vals.append(c_cur[k])

        return None

    b_ub = np.array(b_ub_vals)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

"""
Multi-phase optimizer using:
1. L-BFGS with log-sum-exp smoothing for global optimization
2. Cutting-plane LP method for local refinement (new approach)
3. Multi-scale exploration across different sequence lengths
4. Diverse restarts and final polish
The key new idea is the cutting-plane LP method: at each step, linearize the
quadratic convolution constraints around the current point, select the most
violated constraints (active set), and solve the resulting LP to get a new
candidate. A line search ensures monotone improvement.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _get_kernel(a_cur, k, n):
    """Get the linearization kernel for convolution entry k.
    kernel[i] = a_cur[k-i] if 0 <= k-i < n, else 0."""
    kernel = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    if i_min <= i_max:
        kernel[i_min:i_max + 1] = a_cur[k - i_max:k - i_min + 1][::-1]
    return kernel


def _solve_cutting_lp(a_cur, c_cur, active_idx, n):
    """Solve the cutting-plane LP.
    Minimize t subject to: 2*kernel_k @ a - t <= c_cur[k] for k in active_idx
                           sum(a) = 1, a >= 0, t >= 0
    """
    if len(active_idx) == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_rows = []
    b_ub_vals = []
    for k in active_idx:
        kernel = _get_kernel(a_cur, k, n)
        row = np.zeros(n + 1)
        row[:n] = 2.0 * kernel
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(c_cur[k])
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.status != 0:
            return None
        a_new = result.x[:n].copy()
    except Exception:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 0.01:
        return None
    a_new = a_new / s
    return a_new


def _cutting_plane_optimize(a_init, deadline, max_iters=300):
    """Cutting-plane LP optimization with line search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    no_improve = 0
    K = 100

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive K: start small, grow if stuck
        K_cur = min(K, 2 * n - 1)
        active_idx = np.argsort(c)[-K_cur:]

        a_lp = _solve_cutting_lp(a, c, active_idx, n)
        if a_lp is None:
            # Try with fewer constraints
            K_cur = min(50, 2 * n - 1)
            active_idx = np.argsort(c)[-K_cur:]
            a_lp = _solve_cutting_lp(a, c, active_idx, n)
            if a_lp is None:
                no_improve += 1
                continue

        # Line search
        sc_best_ls = best_sc
        a_best_ls = best_a
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04]:
            a_try = (1.0 - alpha) * a + alpha * a_lp
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < sc_best_ls:
                sc_best_ls = sc_try
                a_best_ls = a_try.copy()

        if sc_best_ls < best_sc - 1e-12:
            best_sc = sc_best_ls
            best_a = a_best_ls.copy()
            a = a_best_ls.copy()
            no_improve = 0
            # Grow K slowly when making progress
            K = min(K + 10, 500)
        else:
            no_improve += 1
            # If stuck, try larger active set or different approach
            if no_improve % 5 == 0:
                K = min(K + 50, 800)
            if no_improve > 30:
                # Reset and try a perturbation
                a = a + 0.01 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                K = 100

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    def fun_local(z_local):
        return _smoobj(z_local, betas[0], n)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break
        est_time = 0.002 * max(1, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 6))))

        def fun_b(z_local):
            return _smoobj(z_local, beta, n)

        def jac_b(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(fun_b, z, jac=jac_b, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxcor': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _newton_active_set(a_init, deadline, max_iter=200):
    """Newton's method on the active set subproblem."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.002 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient: grad[i] = 2 * sum_j a[j] * active[i+j] (for valid i+j)
        a_rev = a[::-1]
        active_padded = np.zeros(2 * n - 1)
        active_padded[:len(active)] = active
        wc = _full_convolve(a_rev, active_padded)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Project to sum=0
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break

        # Simple gradient step with backtracking (avoids Hessian computation)
        alpha = 0.01
        a_new = a.copy()
        for _ls in range(30):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].mean())
            f_old = float(c[active].mean())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent on the smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    rng = np.random.default_rng(123)

    def _grad_active(a, c, n, tol_frac):
        c_max = float(c.max())
        if c_max < 1e-30:
            return np.zeros_like(a)
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _grad_active(a, c, n, 0.001)
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.25 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _down_up(a, n_coarse, n_fine):
    """Downsample then upsample."""
    x_f = np.linspace(0, 1, len(a))
    x_c = np.linspace(0, 1, n_coarse)
    a_c = np.interp(x_c, x_f, a)
    a_c = np.maximum(a_c, 0)
    a_c = a_c / a_c.sum()
    x_n = np.linspace(0, 1, n_fine)
    a_n = np.interp(x_n, x_c, a_c)
    a_n = np.maximum(a_n, 0)
    a_n = a_n / a_n.sum()
    return a_n

                print(f"  evaluate_sequence confirms: {sc_eval:.10f}")
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
