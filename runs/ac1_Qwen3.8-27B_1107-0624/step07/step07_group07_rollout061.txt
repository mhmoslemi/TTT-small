Uses multi-profile initialization, aggressive L-BFGS beta continuation,

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _adaptive_active_set(c, tol_frac):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(3, int(0.01 * len(c)))
    if n_active_est >= len(c):
        n_active_est = len(c)
    threshold_from_rank = sorted_c[min(n_active_est - 1, len(c) - 1)]
    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac * c_max
    tol = max(tol_fixed, min(tol_rank, 0.5 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w

def _gradient_active_direct(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 150,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        # U-shaped: large at edges, small in middle
        a = 1.0 + 2.0 * (2.0 * t - 1.0) ** 2
        return a / a.sum()
    elif kind == "u_shape_strong":
        t = np.linspace(0, 1, n)
        a = 1.0 + 5.0 * (2.0 * t - 1.0) ** 4
        return a / a.sum()
    elif kind == "inverse_gauss":
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.3
        a = 1.0 / (1.0 + 3.0 * ((t - center) / sigma) ** 2)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_u":
        # Cosine U-shape
        t = np.linspace(0, 1, n)
        a = 0.5 - 0.4 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_u":
        t = np.linspace(0, 1, n)
        # a[i] proportional to (i+1)^(-0.3) at each end
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
        a = 1.0 / (1.0 + 0.5 * dist / n) ** 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2
Uses multi-profile initialization (uniform, U-shaped, composite), aggressive
L-BFGS beta continuation, adaptive peak shaving with tightening active sets,
random restarts, and ultra-fine final polishing.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adaptive_active_set(c, tol_frac):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(3, int(0.005 * len(c)))
    if n_active_est >= len(c):
        n_active_est = len(c)
    threshold_from_rank = sorted_c[min(n_active_est - 1, len(c) - 1)]
    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac * c_max
    tol = max(tol_fixed, min(tol_rank, 0.3 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _gradient_active_direct(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 120, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_continuation(a_start, deadline, min_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.08, 0.04, 0.02, 0.01, 0.005, 0.0025,
                              0.0012, 0.0006, 0.0003, 0.00015,
                              7e-5, 3e-5, 1.5e-5, 7e-6, 3e-6,
                              1.5e-6, 7e-7, 3e-7, 1.5e-7, 7e-8,
                              3e-8, 1.5e-8, 7e-9, 3e-9])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 8:
            break

        est_per_iter = 0.002 * (n / 4096.0) ** 1.3
        max_iter = min(20000, max(200, int(remaining / (est_per_iter * 12))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.2e}: score={sc:.10f} (iter={niter})")

    return current_a


def _peak_shaving(a_init, deadline, max_iters=100000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.008
    stall = 0
    tol_frac_current = 1e-4

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        tol_frac_current = max(5e-7, 1e-4 * (0.9995 ** i))
        w = _adaptive_active_set(c, tol_frac_current)

        g = _gradient_active_direct(a, c, n, w)
        g = g - g.mean()

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        g_normd = g / g_norm

        best_eta = eta
        best_a_try = a.copy()
        best_sc_try = best_sc

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = eta * frac
            a_try = a - step * g_normd
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-16:
                best_sc_try = sc_try
                best_a_try = a_try.copy()
                best_eta = step

        if best_sc_try < best_sc - 1e-15:
            best_sc = best_sc_try
            best_a = best_a_try.copy()
            a = best_a_try
            eta = min(eta * 1.03, 0.04)
            stall = 0
        else:
            stall += 1
            eta *= 0.93
            if eta < 1e-10:
                break
            if stall > 300:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * (2.0 * t - 1.0) ** 2
        return a / a.sum()
    elif kind == "u_shape_strong":
        t = np.linspace(0, 1, n)
        a = 1.0 + 4.0 * (2.0 * t - 1.0) ** 4
        return a / a.sum()
    elif kind == "cosine_u":
        t = np.linspace(0, 1, n)
        a = 0.6 - 0.35 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "inverse_gauss":
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.25
        a = 1.0 / (1.0 + 4.0 * ((t - center) / sigma) ** 2)
        return a / a.sum()
    elif kind == "power_u":
        dist = np.minimum(np.arange(n).astype(float), (n - 1) - np.arange(n).astype(float))
        a = 1.0 / (1.0 + 0.8 * dist / n) ** 0.6
        return a / a.sum()
    elif kind == "composite":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * (2.0 * t - 1.0) ** 2 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_u":
        t = np.linspace(0, 1, n)
        a = 1.0 + 3.0 * (2.0 * t - 1.0) ** 6
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, rng, scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + scale * rng.standard_normal(n)
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n, dtype=float)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = rng.uniform(0.002, 0.015)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    return _softmax(z)


def _ultra_polish(a_init, deadline):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    max_c = float(c.max())

    ultra_betas = max_c * np.array([5e-8, 2e-8, 1e-8, 5e-9, 2e-9, 1e-9, 5e-10])

    for beta in ultra_betas:
        if time.monotonic() > deadline - 3:
            break
        remaining = deadline - 3 - time.monotonic()
        if remaining < 4:
            break
        max_iter = min(8000, max(200, int(remaining / 0.0015)))
        a_new, _, _ = _lbfgs_run(a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < best_sc - 1e-16:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            print(f"    Ultra beta={beta:.2e}: score={sc:.10f}")

    if time.monotonic() < deadline - 2:
        a_ps, sc_ps = _peak_shaving(a, deadline - 1, max_iters=50000)
        if sc_ps < best_sc:
            best_sc = sc_ps
            best_a = a_ps.copy()

    return best_a, best_sc


def _trial_optimize(a0, deadline, label=""):
    n = len(a0)
    sc0 = _score(a0)
    print(f"  [{label}] Initial: score={sc0:.10f}")

    a_lbfgs = _lbfgs_continuation(a0, deadline, min_remaining=10.0)
    sc_lbfgs = _score(a_lbfgs)
    print(f"  [{label}] After L-BFGS: score={sc_lbfgs:.10f}")

    a_best = a_lbfgs.copy()
    sc_best = sc_lbfgs

    if time.monotonic() < deadline - 5:
        a_ps, sc_ps = _peak_shaving(a_lbfgs, deadline, max_iters=30000)
        if sc_ps < sc_best:
            a_best = a_ps
            sc_best = sc_ps
            print(f"  [{label}] After peak shave: score={sc_ps:.10f}")

    return a_best, sc_best


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-profile optimizer with aggressive beta continuation,
    adaptive peak shaving, and ultra-fine final polishing.
    """
    t_start = time.monotonic()
    total_budget = max(1.0, float(budget_s) - 3.0)
    deadline = t_start + total_budget
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n_default = 8192
        a0 = _make_profile(n_default, "uniform")
        print(f"No parent. Using uniform n={n_default}, score={_score(a0):.10f}")
        try_update(a0, "init")

    n = len(best_seq) if best_seq is not None else 8192

    # === Phase 1: Aggressive L-BFGS + peak shave on parent (45% time) ===
    print(f"\n=== Phase 1: Parent refinement (n={n}) ===")
    t_phase = time.monotonic()
    phase1_deadline = t_start + total_budget * 0.40
    a_p1, sc_p1 = _trial_optimize(best_seq.copy(), phase1_deadline, "parent")
    try_update(a_p1, "phase1")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 2: Multi-profile exploration (25% time) ===
    print(f"\n=== Phase 2: Multi-profile exploration ===")
    t_phase = time.monotonic()
    phase2_deadline = t_start + total_budget * 0.65

    profiles_to_try = ["u_shape", "cosine_u", "inverse_gauss", "composite", "power_u"]
    time_per_profile = max(15.0, (phase2_deadline - time.monotonic()) / len(profiles_to_try))

    for prof in profiles_to_try:
        if time.monotonic() > phase2_deadline - 10:
            break
        a_init = _make_profile(n, prof)
        prof_deadline = min(phase2_deadline - 5, time.monotonic() + time_per_profile)
        a_res, sc_res = _trial_optimize(a_init, prof_deadline, prof)
        try_update(a_res, f"profile_{prof}")

    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 3: Random restarts from best (15% time) ===
    print(f"\n=== Phase 3: Random restarts ===")
    t_phase = time.monotonic()
    phase3_deadline = t_start + total_budget * 0.80
    if best_seq is not None:
        restart_configs = [
            ("gauss_002", "gauss", 0.002),
            ("gauss_005", "gauss", 0.005),
            ("gauss_010", "gauss", 0.010),
            ("gauss_020", "gauss", 0.020),
            ("gauss_050", "gauss", 0.050),
            ("fourier", "fourier", 0.0),
        ]
        time_per_restart = max(10.0, (phase3_deadline - time.monotonic()) / len(restart_configs))

        for name, mode, scale in restart_configs:
            if time.monotonic() > phase3_deadline - 10:
                break
            if mode == "gauss":
                a_pert = _perturb(best_seq, rng, scale)
            else:
                a_pert = _perturb_fourier(best_seq, rng)
            r_deadline = min(phase3_deadline - 5, time.monotonic() + time_per_restart)
            a_r, sc_r = _trial_optimize(a_pert, r_deadline, f"restart_{name}")
            try_update(a_r, f"restart_{name}")

    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 4: Ultra-fine polishing (15% time) ===
    print(f"\n=== Phase 4: Ultra-fine polishing ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_ultra, sc_ultra = _ultra_polish(best_seq, deadline - 1)
        try_update(a_ultra, "ultra_polish")

        # One more round of peak shaving
        if time.monotonic() < deadline - 3 and best_seq is not None:
            a_final_ps, sc_final_ps = _peak_shaving(best_seq, deadline - 1, max_iters=60000)
            try_update(a_final_ps, "final_peak_shave")

    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(n) / n
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(8192) / 8192
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
