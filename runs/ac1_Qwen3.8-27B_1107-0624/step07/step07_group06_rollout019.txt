    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)

        rs = r @ r

            pAp = p @ Ap
            if abs(pAp) < 1e-30:

            if pAp <= 0:  # not positive definite, add regularization

            alpha = rs / pAp
            x = x + alpha * p
            r = r - alpha * Ap
            rs_new = r @ r
            if np.sqrt(rs_new) < 1e-12:
                break
            beta_cg = rs_new / rs if rs > 1e-30 else 0

            a = a_new

                a = a_try

  active is length 2n-1
  convolve result has length n + (2n-1) - 1 = 3n-2
  We need indices n-1 to 2n-2 (inclusive), which are n values.
  
  Using FFT: pad both to length N >= 3n-2.
  result = ifft(fft(v_rev, N) * fft(active, N))[:3n-2]
  matvec_result = result[n-1:2n-1]

With N = next_pow2(3n-2) ≈ 4n for n power of 2, this is O(n log n).

Let me now write the complete solution. I'll keep it focused and efficient.

Actually, one more consideration: the active set. In the initial iterations, the active set might be just the single peak. As we iterate, it should grow to include more positions. The tolerance for "active" should be adaptive:
- Start with a generous tolerance (e.g., 50% of max) to include many positions.
- Gradually reduce to a tight tolerance (e.g., 1% of max).

This is similar to the "_beta continuation" idea but for the active set.

Also, I should handle the case where the solution has some a[i] = 0 (sparse support). In that case, the KKT conditions only apply to the support, and the off-support condition is g[i] >= λ. If this is violated, we need to add i to the support.

For simplicity, I'll first assume full support (a[i] > 0 for all i) and see if that works. If some a[i] become negative after the Newton step, I'll clip them to 0 and remove them from the support.

"""
Optimized step-function coefficient search using:
1. FFT-based Newton active-set method (primary innovation)
2. Adaptive L-BFGS with log-sum-exp smoothing (warm-start)
3. Coordinate descent polish
4. Multi-start with diverse initializations
Target: beat 1.5027

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "cos2":
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
    elif kind == "quadratic":
        a = t * (1 - t) + 0.01
    elif kind == "semicircle":
        a = np.sqrt(np.maximum(1 - (2*t - 1)**2, 0.01))
    elif kind == "flat_top":
        a = np.ones(n)
        # flatten the middle more
        mid = n // 2
        width = n // 4
        a[mid-width:mid+width] = 1.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _hessian_matvec(v, active, N_fft):
    """Compute M*v where (M*v)[i] = sum_j v[j]*active[i+j]."""
    n = len(v)
    v_rev = v[::-1]
    fv = np.fft.rfft(v_rev, N_fft)
    fa = np.fft.rfft(active, N_fft)
    r = np.fft.irfft(fv * fa, N_fft)
    out_len = 2 * n - 1
    return r[n - 1:2 * n - 1]


def _cg_solve_matvec(matvec_fn, b, x0, max_iter=300, tol=1e-10, reg=0.0):
    """Conjugate gradient to solve Ax = b using matvec_fn."""
    n = len(b)
    x = x0.copy()
    r = b - matvec_fn(x)
    if reg > 0:
        r += reg * x  # Hmm, this isn't right for regularized system
        # Actually, for (A + reg*I)x = b: r = b - (A+reg*I)x = b - Ax - reg*x
        # Let me handle this differently
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x0
    if rs < 1e-30:
        return x
    
    for _ in range(max_iter):
        Ap = matvec_fn(p)
        if reg > 0:
            Ap = Ap + reg * p
        pH = float(np.dot(p, Ap))
        if abs(pH) < 1e-30:
            break
        if pH <= 0:
            # Not PD, break
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Ap
        if reg > 0:
            r = r - alpha * reg * p
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_fft(a_init, deadline, max_iter=100, active_tol_init=0.1, active_tol_final=0.001):
    """
    Newton/CG method for the active-set KKT system.
    Solves M*a = mu*1, 1^T*a = 1, where M[i,j] = active[i+j].
    """
    n = len(a_init)
    N_fft = _next_pow2(4 * n)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive tolerance
        frac = it / max(1, max_iter - 1)
        tol_frac = active_tol_init + frac * (active_tol_final - active_tol_init)
        tol = tol_frac * c_max
        active = (c >= (c_max - tol)).astype(np.float64)
        n_active = float(active.sum())
        if n_active < 2:
            active[int(np.argmax(c))] = 1.0
            n_active = 1.0
        
        # Solve M * x = 1 using CG
        def matvec(v):
            return _hessian_matvec(v, active, N_fft)
        
        b = np.ones(n)
        x_sol = _cg_solve_matvec(matvec, b, a, max_iter=200, tol=1e-10, reg=1e-8)
        
        # Normalize
        x_sol = np.maximum(x_sol, 0)
        s_sol = x_sol.sum()
        if s_sol < 1e-15:
            break
        a_new = x_sol / s_sol
        
        # Check improvement
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            for damp in [0.5, 0.3, 0.1]:
                a_try = (1 - damp) * a + damp * a_new
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                break
    
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs_smoothed(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation on softmax parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.05)
    
    # Adaptive beta schedule
    n_stages = 16
    betas = base_beta * (0.7 ** np.arange(n_stages))
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations based on remaining time
        # Each iteration: 2 convolutions (obj + grad) ≈ 2 * O(n log n)
        # For n=8192, each convolution takes ~0.5ms, so ~1ms per iter
        iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (iter_time * 10))))
        
        def fun_local(z_local):
            return _smoobj(z_local, beta, n)
        
        def jac_local(z_local):
            return _smograd(z_local, beta, n)
        
        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 60, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _coord_descent_polish(a_init, deadline, n_sweeps=50):
    """Coordinate descent: optimize each coordinate one at a time."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize coordinate order each sweep
        order = np.random.default_rng(sweep).permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            # Save current value
            old_val = a[i]
            # Try a few different values
            c = _full_convolve(a, a)
            # The optimal a[i] minimizes max_k c_k
            # c_k depends on a[i] through: c_{i+j} += a[i]*a[j] and c_{i+j} += a[j]*a[i] for j != i
            # Actually c_k = sum_j a[j]*a[k-j], so a[i] appears in c_{i+j} as a[i]*a[j] and in c_{j+i} as a[j]*a[i]
            # So dc_k/da_i = a[k-i] + a[i-k] (for valid indices)
            # For the active set, we want to reduce the max.
            
            # Simple approach: try a[i] = old_val * factor for various factors
            best_local = old_val
            best_local_sc = _full_convolve(a, a).max()
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3]:
                a[i] = old_val * factor
                sc_c = _full_convolve(a, a).max()
                if sc_c < best_local_sc - 1e-15:
                    best_local_sc = sc_c
                    best_local = a[i]
            
            a[i] = best_local
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            improved = True
    
    return best_a, best_sc


def _perturb_log(a, sigma, rng):

    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
