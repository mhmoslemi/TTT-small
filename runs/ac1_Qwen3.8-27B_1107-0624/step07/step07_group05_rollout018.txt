"""

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ls_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _ls_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _topk_grad(a, n, k=10):

        active = np.zeros_like(c, dtype=bool)

    w[active] = 1.0 / n_act
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

def _flat_init(n):
    """
    Generate sequence with (approximately) flat linear autocorrelation for lags 0..n-1
    using the recursive formula: a[k] = (1 - sum_{i=1}^{k-1} a[i]*a[k-i]) / (2*a[0])
    with target autocorrelation height = 1.
    """
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            a[k] = 0.5
        else:
            s = np.dot(a[1:k], a[k - 1:0:-1])
            a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def _resample(a_old, n_new):
    """Resample a sequence to a new length using linear interpolation."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        a_flat = _flat_init(n_try)
        sc_flat = _score(a_flat)
        print(f"    flat score: {sc_flat:.10f}")
        _update(a_flat, sc_flat)

        if remaining > 40:
            min_rem = max(15, remaining * 0.25)
            a_opt, sc_opt = _run_lbfgs_ls(a_flat, deadline, min_remaining=min_rem)
            _update(a_opt, sc_opt)
            if sc_opt < best_score and remaining > 60:
                a_pe, sc_pe = _pressure_equalize(a_opt, deadline, n_iters=100,
                                                 gamma=0.3, min_remaining=min_rem)
                _update(a_pe, sc_pe)

    # Phase 4: Subgradient polish on best
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a4, sc4 = _subgrad_polish(best_seq, deadline, n_steps=3000, lr_init=0.01,
                                  min_remaining=40)
        _update(a4, sc4)

    # Phase 5: More L-BFGS on best (fine)
    print(f"\n[Phase 5] Fine L-BFGS on best (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a5, sc5 = _run_lbfgs_ls(best_seq, deadline, min_remaining=20)
        _update(a5, sc5)

    # Phase 6: Coordinate polish
    print(f"\n[Phase 6] Coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a6, sc6 = _coordinate_polish(best_seq, deadline, n_rounds=40, min_remaining=10)
        _update(a6, sc6)

    # Phase 7: Final pressure + subgradient
    print(f"\n[Phase 7] Final refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a7, sc7 = _pressure_equalize(best_seq, deadline, n_iters=80, gamma=0.2,
                                     min_remaining=5)
        _update(a7, sc7)
        if time.monotonic() < deadline - 2:
            a8, sc8 = _subgrad_polish(best_seq, deadline, n_steps=500, lr_init=0.003,
                                      min_remaining=2)
            _update(a8, sc8)

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _flat_init(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _flat_init(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
