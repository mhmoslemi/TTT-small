Actually, one more thought: the **proximal operator** for the constraint `a >= 0, sum(a) = 1` in the softmax parameterization is just the softmax itself (since we're always in the simplex). So the proximal gradient method is:

```
z_{k+1} = argmin [f(softmax(z)) + (1/(2*eta)) * ||z - z_k||^2]
```

This doesn't have a closed form, so we use gradient descent on the inner problem, which just gives us a gradient step:
`z_{k+1} = z_k - eta * grad f(softmax(z_k))`

With momentum (FISTA):
`y_k = z_k + ((t_{k-1} - 1) / t_k) * (z_k - z_{k-1})`
`z_{k+1} = y_k - eta * grad f(softmax(y_k))`

The gradient of `f_beta(softmax(z))` w.r.t. `z` is:
`grad_z = D * grad_a` where `D = diag(a) - outer(a, a)` (Jacobian of softmax) and `grad_a` is the gradient w.r.t. `a`.

But there's a simpler formula: `grad_z = a * (grad_a - a^T grad_a)` (the centered version).

For the log-sum-exp objective, `grad_a` is `4n * (H(a) - W(a)) * a` where:
- `H(a)_i = sum_k w_k * a[k-i]` (cross-correlation with weights)
- `W(a) = sum_k w_k * conv[k]` (weighted mean of conv)
- `w_k = exp((conv[k] - max)/beta) / sum(exp((conv - max)/beta))`

This is what the previous code already computes. So my gradient computation is the same.

"""
Step function optimizer using FISTA with log-sum-exp smoothing,
active-set peak shaving, and multi-scale Fourier exploration.
Improvements over previous: proper FISTA momentum, adaptive backtracking,
direct peak-shaving coordinate updates, and structured n-exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize
import scipy.fft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = scipy.fft.rfft(a, N)
    fb = scipy.fft.rfft(b, N)
    return scipy.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score1(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _conv(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsexp_obj(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    if beta < 1e-15:
        return 2.0 * n * cm
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _lsexp_grad(z, beta, n):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    if beta < 1e-15:
        # Hard max gradient
        idx = np.argmax(c)
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        sh = (c - cm) / beta
        e = np.exp(np.clip(sh, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    # grad of sum_k w_k * conv[k] wrt a
    # d(conv[k])/d(a_i) = 2 * a[k-i] (with bounds)
    # So grad_i = 2 * sum_k w_k * a[k-i]
    # This is 2 * conv(w, reverse(a))[i] ... let me think
    # conv(w, a)[i] = sum_j w[j] * a[i-j]
    # We want sum_k w[k] * a[k-i] = sum_k w[k] * a[k-i]
    # Let j = k-i, then k = j+i: sum_j w[j+i] * a[j] = cross-correlation
    # Actually: sum_k w[k] * a[k-i] is the cross-correlation of w and a at lag i
    # = conv(w, reverse(a))[n-1+i] ... hmm, let me just compute it directly
    # Using the formula: grad_i = 4n * a_i * (h_i - W)
    # where h = conv(w, reverse(a))[n-1:2n-1] and W = dot(w, c)
    a_rev = a[::-1]
    wc = _conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_optimize(a_init, deadline, min_time=60.0):
    """FISTA on log-sum-exp smoothed objective with aggressive beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    c = _conv(a, a)
    cm0 = float(c.max())
    
    # Beta schedule
    betas = cm0 * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                            0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    z_prev = z.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_sc = _score1(a)
    
    for beta in betas:
        if time.monotonic() > deadline - min_time:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate max iterations
        # Each iteration costs ~O(n log n) for the FFT
        # Rough estimate: 0.5ms per iteration for n=8192
        est_per_iter = 0.001 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(20, int(remaining / est_per_iter)))
        
        t_mom = 1.0
        z_prev = z.copy()
        lr = 1.0 / (8.0 * n)  # Initial learning rate
        
        for it in range(max_iter):
            if time.monotonic() > deadline - min_time:
                break
            
            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            theta = (t_mom - 1.0) / t_new
            y = z + theta * (z - z_prev)
            
            # Gradient at y
            a_y = _softmax(y)
            grad_y = _lsexp_grad(y, beta, n)
            
            # Step
            z_new = y - lr * grad_y
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            
            # Backtracking: check if objective decreased
            a_cur = _softmax(z)
            a_new = _softmax(z_new)
            val_cur = _lsexp_obj(z, beta, n)
            val_new = _lsexp_obj(z_new, beta, n)
            
            # Accept if improved, otherwise reduce lr
            if val_new < val_cur + 1e-15 * abs(val_cur) + 1e-15:
                z_prev = z
                z = z_new
                t_mom = t_new
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break
            
            # Check score periodically
            if it % 50 == 0:
                a_chk = _softmax(z)
                sc = _score1(a_chk)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_chk.copy()
        
        # Update best after each beta stage
        a_stage = _softmax(z)
        sc_stage = _score1(a_stage)
        if sc_stage < best_sc:
            best_sc = sc_stage
            best_a = a_stage.copy()
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()
    
    return best_a


def _peak_shave(a, deadline, n_rounds=200, min_time=30.0):
    """Direct peak shaving: adjust individual coordinates to reduce the top peaks."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score1(a)
    rng = np.random.default_rng(42)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline - min_time:
            break
        
        c = _conv(a, a)
        cm = float(c.max())
        
        # Find top peaks
        top_k = min(5, len(c))
        peak_idx = np.argsort(c)[-top_k:]
        
        # For each peak, find the coordinate that contributes most
        improved = False
        for pk in peak_idx:
            if time.monotonic() > deadline - min_time:
                break
            # Contribution of a_i to conv[pk]: a_i * a[pk-i] (for valid i)
            # Total contribution: sum over valid i of a_i * a[pk-i] ... no
            # conv[pk] = sum_i a[i] * a[pk-i]
            # d(conv[pk])/d(a_j) = 2 * a[pk-j] (for j such that pk-j is in [0,n-1])
            # To reduce conv[pk], decrease a_j where a[pk-j] is large
            # But this might increase other conv values
            
            # Simple strategy: find j that maximizes a[j]*a[pk-j] and reduce a[j]
            valid_j = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_j) == 0:
                continue
            contributions = a[valid_j] * a[pk - valid_j]
            j_star = valid_j[np.argmax(contributions)]
            
            # Try reducing a[j_star]
            old = a[j_star]
            for factor in [0.7, 0.8, 0.9, 0.5, 0.3]:
                a[j_star] = old * factor
                c_try = _conv(a, a)
                cm_try = float(c_try.max())
                if cm_try < cm - 1e-15:
                    improved = True
                    cm = cm_try
                    break
                a[j_star] = old
            
            # Also try increasing a[j_star] (if it was too small)
            for factor in [1.2, 1.5, 2.0]:
                a[j_star] = old * factor
                c_try = _conv(a, a)
                cm_try = float(c_try.max())
                if cm_try < cm - 1e-15:
                    improved = True
                    cm = cm_try
                    break
                a[j_star] = old
        
        if improved:
            s = a.sum()
            a = a / s
            sc = _score1(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # Try random coordinate perturbations
            n_try = min(20, n // 100)
            for _ in range(n_try):
                if time.monotonic() > deadline - min_time:
                    break
                j = rng.integers(0, n)
                old = a[j]
                factor = rng.uniform(0.5, 1.5)
                a[j] = old * factor
                c_try = _conv(a, a)
                cm_try = float(c_try.max())
                a_chk = a / a.sum()
                sc_try = _score1(a_chk)
                if sc_try < best_sc:
                    a = a_chk
                    best_sc = sc_try
                    best_a = a.copy()
                else:
                    a[j] = old
    
    return best_a


def _lbfgs_polish(a_init, deadline, min_time=30.0):
    """L-BFGS-B on the log-sum-exp objective."""
    n = len(a_init)
    c = _conv(a_init, a_init)
    cm = float(c.max())
    if cm < 1e-15:
        return a_init
    
    # Start with moderate beta
    betas = [cm * 0.05, cm * 0.02, cm * 0.01, cm * 0.005, cm * 0.002, cm * 0.001]
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score1(a_init)
    
    for beta in betas:
        if time.monotonic() > deadline - min_time:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 2:
            break
        
        est_per_iter = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(20, int(remaining / est_per_iter)))
        
        try:
            res = minimize(
                _lsexp_obj, z, args=(beta, n),
                jac=_lsexp_grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxcor': 30, 'maxls': 50}
            )
            a_new = _softmax(res.x)
            sc_new = _score1(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a


def _fourier_explore(a_best, deadline, min_time=30.0, n_trials=15):
    """Explore in Fourier domain for diversity."""
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score1(a_best)
    rng = np.random.default_rng(123)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline - min_time:
            break
        
        # Perturb in frequency domain
        z = np.log(np.maximum(a_best, 1e-30))
        z_hat = scipy.fft.rfft(z)
        
        n_modes = rng.integers(3, 10)
        for _ in range(n_modes):
            freq = rng.integers(1, max(2, n // 4))
            if freq < len(z_hat):
                amp = rng.uniform(0.01, 0.05)
                delta = amp * rng.standard_normal()
                phase = rng.uniform(0, 2 * np.pi)
                z_hat[freq] += delta * np.exp(1j * phase)
                if freq < len(z_hat) - 1:
                    z_hat[-freq] = np.conj(z_hat[freq])
        
        z_new = scipy.fft.irfft(z_hat, n)
        z_new = z_new - z_new.mean()
        a_try = _softmax(z_new)
        
        # Brief FISTA
        a_opt = _fista_optimize(a_try, min(deadline - min_time, time.monotonic() + 20), min_time=10)
        sc = _score1(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
    
    return best_a


def _subgrad_polish(a_init, deadline, n_steps=500, lr_init=0.01):
    """Subgradient descent with momentum in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    momentum = np.zeros_like(z)
    beta_mom = 0.9
    best_a = a_init.copy()
    best_sc = _score1(a_init)
    lr = lr_init
    no_improve = 0
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _conv(a, a)
        cm = float(c.max())
        
        # Active set: top 3 peaks
        top_k = min(3, len(c))
        active_idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        w[active_idx] = 1.0 / top_k
        
        # Gradient
        a_rev = a[::-1]
        wc = _conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Momentum update
        momentum = beta_mom * momentum - lr * grad
        z = z + momentum
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if i % 20 == 0:
            a_chk = _softmax(z)
            sc = _score1(a_chk)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_chk.copy()
                no_improve = 0
                lr = min(lr * 1.01, lr_init)
            else:
                no_improve += 1
                lr *= 0.85
                if no_improve > 15:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros_like(z)
                    no_improve = 0
                    lr = lr_init * 0.5
                    beta_mom = min(beta_mom + 0.02, 0.95)
    
    a_final = _softmax(z)
    sc_final = _score1(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-5 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = (t > 0.1) & (t < 0.9)
        a = a.astype(float)
        a += 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n
