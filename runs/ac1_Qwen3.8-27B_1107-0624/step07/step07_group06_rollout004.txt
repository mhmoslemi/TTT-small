"""
Multi-scale optimizer for step function coefficient minimization.
Strategy: Start from parent, optimize at multiple scales using
ADAM + L-BFGS with aggressive beta continuation, Newton active-set,
and population-based exploration. Targets larger n for better asymptotic scores.

import numpy as np
import time
import sys
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logsumexp_smooth(c, beta):
    """Compute max(c) + beta*log(sum(exp((c-max)/beta)))"""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _smooth_val(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _logsumexp_smooth(c, beta)

def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _adam_optimize(z_init, beta, n, deadline, lr=0.01, max_iter=50000):
    """ADAM optimization on the smoothed objective."""
    z = z_init.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_z = z.copy()
    best_val = _smooth_val(z, beta, n)
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        grad = _smooth_grad(z, beta, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Keep z bounded
        z = np.clip(z, -100, 100)
        z = z - z.mean()  # Center to avoid numerical issues
        
        if i % 1000 == 0:
            val = _smooth_val(z, beta, n)
            if val < best_val:
                best_val = val
                best_z = z.copy()
    
    return best_z

def _lbfgs_continue(a_init, deadline, min_remaining=30, max_stages=20):

    n = len(a_init)

def _newton_active(a_init, deadline, max_iter=200):
    """Newton method on active set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(4 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(0.0005 * c_max, 1e-14)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Gradient of sum_{k in active} c_k with respect to a
        # d c_k / d a_i = a[k-i] for valid k-i (0<=k-i<n)
        # grad[i] = sum_{k in active} a[k-i] (for valid)
        # = sum_j a[j] * active[i+j] (substituting k=i+j)
        # This is the cross-correlation of a with the active indicator
        
        a_rev = a[::-1]
        N_act = _next_pow2(3 * n)
        fa = np.fft.rfft(a_rev, N_act)
        f_act = np.fft.rfft(active.astype(float), N_act)
        r = np.fft.irfft(fa * f_act, N_act)
        grad = r[n-1:2*n-1]
        grad = 2.0 * grad
        
        # Project to sum=0 (tangent to simplex)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break
        
        # Hessian-vector product: H[v]_i = 2 * sum_j v[j] * active[i+j]
        def hess_vec(v):
            v_rev = v[::-1]
            fv = np.fft.rfft(v_rev, N_act)
            r2 = np.fft.irfft(fv * f_act, N_act)
            return 2.0 * r2[n-1:2*n-1]
        
        # Damped Newton: solve (H + mu*I) dx = -grad_t
        mu = 1e-6
        # Use a few steps of preconditioned CG
        x = np.zeros(n)
        r_vec = -grad_t.copy()
        p = r_vec.copy()
        rs = float(np.dot(r_vec, r_vec))
        mu_vec = mu * p
        
        b_norm = float(np.linalg.norm(r_vec))
        for cg_iter in range(100):
            Hp = hess_vec(p) + mu * p
            pH = float(np.dot(p, Hp))
            if pH <= 0:
                # H not positive definite, use gradient step
                x = -grad_t / gnorm
                break
            alpha = rs / pH
            x = x + alpha * p
            r_vec = r_vec - alpha * Hp
            r_new = float(np.dot(r_vec, r_vec))
            if np.sqrt(r_new) < 1e-10 * b_norm:
                break
            beta_cg = r_new / rs if rs > 1e-30 else 0.0
            p = r_vec + beta_cg * p
            rs = r_new
        else:
            x = -grad_t / gnorm
        
        x = x - np.mean(x)
        
        # Line search
        alpha = 1.0
        a_new = a.copy()
        f_old = float(c[active].sum())
        for ls in range(30):
            a_try = a + alpha * x
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try <= f_old - 1e-10:
                a_new = a_try
                break
            alpha *= 0.5
        
        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc

def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.01):
    """Subgradient descent with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    
    beta = _full_convolve(a_init, a_init).max() * 0.01
    beta = max(beta, 1e-6)
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _smooth_grad(z, beta, n)
        step = lr / (1.0 + 0.01 * i)
        z = z - step * grad
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                beta = max(beta * 0.5, 1e-8)
                lr = min(lr * 1.1, lr_init)
            else:
                lr *= 0.85
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc
