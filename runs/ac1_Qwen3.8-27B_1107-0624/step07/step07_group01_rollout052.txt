    N = _next_pow2(2*n-1)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_shaves):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        k_star = int(np.argmax(c))
        
        # In frequency domain, the peak at lag k* is due to the component
        # at frequency k*/N. We reduce this component.
        A = np.fft.rfft(a, N)
        # The frequency bin corresponding to lag k* 
        # Actually, this is more subtle. The peak of IFFT(|A|^2) at lag k*
        # means that |A|^2 has a component at frequency k*/N.
        # |A|^2 = A * conj(A) in the freq domain... no, that's not right.
        
        # Actually: c = IFFT(|A|^2) where A = FFT(a, N).
        # |A|^2 is a real, non-negative sequence of length N.
        # c_k = (1/N) * sum_f |A_f|^2 * exp(2πi f k / N)
        # So c_k is the DFT of |A|^2 (up to scaling).
        # A peak in c at lag k* means |A|^2 has a large DFT coefficient at frequency k*/N.
        
        # To reduce c_{k*}, we want to reduce the "energy" of |A|^2 at frequency k*.
        # This means we want to reduce |A|^2 in a way that reduces its DFT at bin k*.
        # The DFT of |A|^2 at bin k* is sum_f |A_f|^2 * exp(-2πi f k* / N).
        # To reduce this, we can reduce |A_f|^2 for frequencies f where exp(-2πi f k*/N) 
        # has a positive real part (i.e., where the contribution is in phase).
        
        # Simpler approach: just reduce |A|^2 proportionally to its "contribution" to the peak.
        # The contribution of frequency bin f to c_{k*} is |A_f|^2 * exp(2πi f k* / N) / N.
        # The real part is |A_f|^2 * cos(2π f k* / N) / N.
        
        # We want to reduce the sum of positive contributions.
        # So we reduce |A_f|^2 more at frequencies where cos(2π f k* / N) > 0.
        
        P = np.abs(A)**2  # Power spectrum, length N//2+1
        # Create a "shaping" mask
        f = np.arange(len(P))
        mask = np.cos(2 * np.pi * f * k_star / N)
        mask = np.maximum(mask, 0)  # Only reduce where it helps
        mask = mask / mask.sum() if mask.sum() > 0 else np.ones_like(mask)

        reduction = 0.1 * c_max / P.sum() * N  # Scale the reduction
        P_new = P * (1 - reduction * mask)
        P_new = np.maximum(P_new, 0)

        # This is an inverse problem. Simple approach: scale A by sqrt(P_new/P)

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parent initialization with quick L-BFGS warmup
2. Aggressive multi-scale log-sum-exp L-BFGS (extremely tight betas)
3. Adaptive active-set mirror descent with Barzilai-Borwein step size
4. Spectral domain peak equalization (tone-reservation inspired)
5. Multi-length exploration (8192 to 32768)
6. Remez-like peak equalization on active lags
7. Final micro-polishing with ultra-tight smoothing
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Much tighter beta schedule
    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.00025,
                              0.0001, 5e-5, 2.5e-5, 1e-5, 5e-6,
                              2.5e-6, 1e-6, 5e-7, 2.5e-7,
                              1e-7, 5e-8, 2.5e-8, 1e-8])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:

        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:

        # Scale iterations with problem size and remaining time

        max_iter = min(20000, max(200, int(remaining / (0.005 * scale * 10))))

        if sc < current_sc - 1e-15:

            current_sc = sc
            if current_sc < 1.5010:

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    z_prev = z.copy()
    g_prev = None
    eta = 0.05 / np.sqrt(n)
    stall = 0
    check_interval = max(10, n // 50)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Adaptive active set: lags within active_frac% of max
        threshold = c_max * (1.0 - active_frac / 100.0)
        active = np.where(c >= threshold)[0]
        k_act = max(1, len(active))

        w = np.zeros_like(c)
        w[active] = 1.0 / k_act
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 4.0 * n * a * (h - W)

        # BB step size
        if g_prev is not None:
            s_vec = z - z_prev
            y_vec = g - g_prev
            sy = float(np.dot(s_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            yy = float(np.dot(y_vec, y_vec))
            if sy > 1e-30 and ss > 1e-30:
                eta_bb = ss / sy
                eta_bb = np.clip(eta_bb, 0.1 * eta, 10.0 * eta)
                eta = 0.5 * eta + 0.5 * eta_bb

        z_prev = z.copy()
        g_prev = g.copy()

        # Gradient step in z-space
        z = z - eta * g
        z = z - z.mean()
        z = np.clip(z, -120.0, 120.0)

        # Periodic score check
        if it % check_interval == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    eta *= 0.7
                    if eta < 1e-12:
                        break
                elif stall > 15:
                    eta *= 0.85

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a, best_sc


def _spectral_equalize(a_init, deadline, n_iters=500):
    """Spectral domain peak shaving using tone-reservation inspired method."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    reduction_rate = 0.15

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        k_star = int(np.argmax(c))

        # Get power spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2

        # The peak at lag k_star in c corresponds to energy at frequency k_star/N in P.
        # We want to reduce P at frequencies that contribute positively to c_{k_star}.
        f = np.arange(len(P))
        # Contribution of bin f to c_{k_star} is proportional to P[f] * cos(2*pi*f*k_star/N)
        contrib = np.cos(2.0 * np.pi * f * k_star / N)
        positive_mask = np.maximum(contrib, 0)
        total_pos = positive_mask.sum()
        if total_pos < 1e-10:
            break
        positive_mask = positive_mask / total_pos

        # Scale reduction by how dominant the peak is
        second_max = np.sort(c)[::-1]
        if len(second_max) > 1:
            dominance = c_max / max(second_max[1], 1e-30)
        else:
            dominance = 2.0
        reduction = reduction_rate / dominance

        P_new = P * (1.0 - reduction * positive_mask)
        P_new = np.maximum(P_new, 0)

        # Convert back: scale A by sqrt(P_new/P)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc * (1 + 1e-12):
            best_sc = min(best_sc, sc_new)
            if sc_new < best_sc or sc_new < _score(best_a):
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new
        else:
            reduction_rate *= 0.8
            if reduction_rate < 1e-6:
                break

    return best_a, best_sc


def _remez_equalize(a_init, deadline, k_active=8, max_iter=2000):
    """Remez-like algorithm: equalize the top-K convolution values."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    alpha = 1.0 / n

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        # Compute the equalization gradient
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * (h - W)
        g = g - g.mean()

        # Line search
        improved = False
        for _ in range(15):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                alpha = min(alpha * 1.02, 2.0 / n)
                improved = True
                break
            alpha *= 0.7

        if not improved:
            alpha *= 0.5
            if alpha < 1e-15:
                break
            # Check spread of active values
            active_vals = c[idx]
            spread = active_vals.max() - active_vals.min()
            if spread < 1e-14 * c_max:
                break

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _subgrad_polish(a_init, deadline, k_active=5, max_steps=30000, lr_init=0.05):
    """Standard subgradient polishing in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    stall_threshold = 30

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros_like(c)
        w[idx] = 1.0 / k_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        z = z - lr * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall = 0
            else:
                stall += 1
                if stall > stall_threshold:
                    lr *= 0.6
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final
    return best_a, best_sc


def _basin_hop(a_current, deadline, rng, n_hops=5, reserve=30.0):
    """Basin hopping for diversification."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.6

        # Create perturbation
        pert_type = h % 3
        if pert_type == 0:
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 16)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 3 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.04, 0.04)
            z = z - z.mean()
            a_pert = _softmax(z)

        # Quick optimization
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        a_opt = _lbfgs_smooth(a_pert, beta, n, 2000)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, mirror descent, spectral methods,
    Remez equalization, and length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={best_score_val:.10f}")
    else:
        # Fallback initialization
        n_init = 16384
        parent = _make_chirp(n_init, freq=1.0, eps=0.3)
        try_update(parent, "chirp_fallback")
        print(f"  Fallback chirp: score={best_score_val:.10f}")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s")

    # === Phase 2: Quick L-BFGS warmup ===
    print(f"\n=== Phase 2: L-BFGS Warmup (t={time.monotonic()-t_start:.1f}s) ===")
    a_p2 = _adaptive_lbfgs_aggressive(best_seq, deadline, reserve_time=800.0)
    try_update(a_p2, "lbfgs_warm")

    # === Phase 3: Mirror descent with BB step ===
    print(f"\n=== Phase 3: Mirror Descent BB (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_md, sc_md = _bb_mirror_descent(best_seq, deadline - 500,
                                         active_frac=0.5, max_iter=50000)
        try_update(a_md, "mirror_bb_0.5%")

    if best_seq is not None and time.monotonic() < deadline - 400:
        a_md2, sc_md2 = _bb_mirror_descent(best_seq, deadline - 400,
                                           active_frac=1.0, max_iter=30000)
        try_update(a_md2, "mirror_bb_1%")

    # === Phase 4: Spectral equalization ===
    print(f"\n=== Phase 4: Spectral Equalization (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_spec, sc_spec = _spectral_equalize(best_seq, deadline - 400, n_iters=300)
        try_update(a_spec, "spectral")

    # === Phase 5: Remez equalization ===
    print(f"\n=== Phase 5: Remez Equalization (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_remez, sc_remez = _remez_equalize(best_seq, deadline - 350,
                                            k_active=8, max_iter=3000)
        try_update(a_remez, "remez_k8")

    if best_seq is not None and time.monotonic() < deadline - 300:
        a_remez2, sc_remez2 = _remez_equalize(best_seq, deadline - 300,
                                              k_active=12, max_iter=2000)
        try_update(a_remez2, "remez_k12")

    # === Phase 6: More L-BFGS (tighter betas) ===
    print(f"\n=== Phase 6: Tight L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_p6 = _adaptive_lbfgs_aggressive(best_seq, deadline, reserve_time=100.0)
        try_update(a_p6, "lbfgs_tight")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        current_n = len(best_seq)
        lengths_to_try = [l for l in [8192, 12288, 24576, 32768] if l != current_n]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 100:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            # Quick optimization at new length
            if time.monotonic() < deadline - 80:
                a_opt = _adaptive_lbfgs_aggressive(a_interp, deadline,
                                                   reserve_time=50.0)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 8: Basin hopping ===
    print(f"\n=== Phase 8: Basin Hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 80, rng,
                                   n_hops=5, reserve=30.0)
        try_update(a_hop, "basin_hop")

    # === Phase 9: Final micro-polishing ===
    print(f"\n=== Phase 9: Final Polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(8):
            if time.monotonic() > deadline - 5:
                break

            # Ultra-tight L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-9
            remaining_for_lbfgs = deadline - 5 - time.monotonic()
            if remaining_for_lbfgs > 5:
                max_iter = min(8000, max(500, int(remaining_for_lbfgs / 0.006)))
                a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 10:
                a_s, _ = _subgrad_polish(best_seq, deadline - 10,
                                         k_active=3, max_steps=5000, lr_init=0.02)
                try_update(a_s, f"final_sg_k3_{rnd}")

            # Remez on top-5
            if time.monotonic() < deadline - 10:
                a_r, _ = _remez_equalize(best_seq, deadline - 10,
                                         k_active=5, max_iter=1000)
                try_update(a_r, f"final_remez_{rnd}")

            # Spectral
            if time.monotonic() < deadline - 10:
                a_sp, _ = _spectral_equalize(best_seq, deadline - 10, n_iters=100)
                try_update(a_sp, f"final_spec_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
