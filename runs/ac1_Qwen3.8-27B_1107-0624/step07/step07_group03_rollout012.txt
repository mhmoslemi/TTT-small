import numpy as np

def _next_fft_size(n):
    """Find next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_fft_size(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    h = wc[n - 1:2 * n - 1]

    flip_a = a[::-1]

    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind - (cssv - 1.0) > 0
    rho = np.argmax(cond)
    if not cond[rho]:
        rho = 0
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - tau, 0.0)

                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent with backtracking
            t = min(0.1, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-10 * t * gnorm:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-14:
            break

    return best_a, best_sc


def _alternating_lambda_newton(a_init, deadline, max_outer=20, tol_active=0.002):
    """Alternating optimization of lambda (KKT multipliers) and a."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        active_idx = np.where(active)[0]
        n_active = len(active_idx)
        if n_active == 0:
            active_idx = np.array([np.argmax(c)])
            n_active = 1
        if n_active > 200:
            top_idx = np.argsort(c)[::-1][:200]
            active_idx = top_idx
            n_active = 200

        # Solve LP for optimal lambda
        # Minimize r s.t. M*lambda - mu <= r/2, -M*lambda + mu <= r/2
        # subject to sum(lambda)=1, lambda>=0
        # M[i, k] = 2 * c_rev[active_idx[k] - i] for i in 0..n-1
        # To keep LP small, only use a subset of constraints
        # Use all n constraints but limit active set

        if n_active <= 100:
            # Build M matrix (n x n_active)
            c_rev = c[::-1]
            M = np.zeros((n, n_active))
            for j, k in enumerate(active_idx):
                # M[i, j] = 2 * c_rev[k - i] if 0 <= k-i < len(c_rev)
                for i in range(n):
                    idx = k - i
                    if 0 <= idx < len(c_rev):
                        M[i, j] = 2.0 * c_rev[idx]

            # LP: minimize r
            # vars: lambda_0..lambda_{m-1}, mu, r  (total m+2 vars)
            # constraints:
            # M*lambda - mu - r/2 <= 0 for each i (n constraints)
            # -M*lambda + mu - r/2 <= 0 for each i (n constraints)
            # sum(lambda) = 1
            # lambda >= 0, r >= 0
            m = n_active
            # c vector (objective): minimize r -> c = [0]*m + [0, 1]
            c_lp = np.zeros(m + 2)
            c_lp[-1] = 1.0

            # A_ub: (M - [0..0, 1, 0.5]) for first set, (-M + [0..0, 1, 0.5]) for second
            # Row i: M[i]*lambda - mu - 0.5*r <= 0
            A1 = np.hstack([M, -np.ones((n, 1)), -0.5 * np.ones((n, 1))])
            # Row i: -M[i]*lambda + mu - 0.5*r <= 0
            A2 = np.hstack([-M, np.ones((n, 1)), -0.5 * np.ones((n, 1))])
            A_ub = np.vstack([A1, A2])
            b_ub = np.zeros(2 * n)

            # A_eq: sum(lambda) = 1
            A_eq = np.zeros((1, m + 2))
            A_eq[0, :m] = 1.0
            b_eq = np.array([1.0])

            # Bounds: lambda >= 0, mu free, r >= 0
            bounds = [(0, None)] * m + [(None, None), (0, None)]

            try:
                res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs')
                if res.success:
                    lambda_opt = res.x[:m]
                    lambda_opt = np.maximum(lambda_opt, 0)
                    lambda_opt = lambda_opt / lambda_opt.sum()
                else:
                    lambda_opt = np.ones(n_active) / n_active
            except Exception:
                lambda_opt = np.ones(n_active) / n_active
        else:
            lambda_opt = np.ones(n_active) / n_active

        # Now optimize a with fixed lambda
        # Build effective A_vec in full convolution space
        A_vec = np.zeros(len(c))
        for j, k in enumerate(active_idx):
            A_vec[k] = lambda_opt[j]

        # Run a few Newton steps with this fixed active set
        for newton_iter in range(30):
            if time.monotonic() > deadline:
                break

            c_cur = _full_convolve(a, a)
            g = (2.0 / 1.0) * _corr_a_A(a, A_vec, n)  # sum(lambda)=1 so factor is 2
            g_t = g - np.mean(g)
            gnorm = np.linalg.norm(g_t)
            if gnorm < 1e-14:
                break

            mu_reg = 1e-8

            def hvp(v):
                return 2.0 * _corr_a_A(v, A_vec, n) + mu_reg * v

            # CG
            x_cg = np.zeros(n)
            r = g_t - hvp(x_cg)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(200):
                Ap = hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x_cg += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13:
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            u = x_cg

            x_w = np.zeros(n)
            r = np.ones(n) - hvp(x_w)
            p = r.copy()
            rs = np.dot(r, r)
            for _ in range(200):
                Ap = hvp(p)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rs / pAp
                x_w += alpha * p
                r_new = r - alpha * Ap
                rs_new = np.dot(r_new, r_new)
                if np.sqrt(rs_new) < 1e-13:
                    break
                p = r_new + (rs_new / rs) * p
                r = r_new
                rs = rs_new
            w_sol = x_w

            sum_w = float(w_sol.sum())
            if abs(sum_w) < 1e-15:
                dx = -u - np.mean(u)
            else:
                lam = float(u.sum()) / sum_w
                dx = -u + lam * w_sol
                dx = dx - np.mean(dx)

            neg_mask = dx < -1e-15
            if neg_mask.any():
                t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.999
            else:
                t_max = 2.0
            t_max = max(t_max, 1e-12)

            f_old = float(c_cur.max())
            slope = float(np.dot(g_t, dx))
            t = min(1.0, t_max)
            accepted = False
            for ls in range(40):
                a_try = a + t * dx
                if np.min(a_try) < -1e-12:
                    t *= 0.5
                    continue
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= f_old + 1e-4 * t * slope:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5

            if not accepted:
                t = min(0.05, t_max)
                for ls in range(20):
                    a_try = a - t * g_t
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        t *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    f_try = float(c_try.max())
                    if f_try < f_old:
                        a = a_try
                        break
                    t *= 0.5
                else:
                    break

            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

            c_check = _full_convolve(a, a)
            if abs(float(c_check.max()) - f_old) < 1e-14:
                break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _support_redu

ction(a_init, deadline, max_rounds=10):
    """Iteratively zero out small elements and re-optimize."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_sorted = np.sort(a)
        # Try different thresholds
        for frac in [0.05, 0.10, 0.15, 0.20, 0.30]:
            if time.monotonic() > deadline:
                break
            thresh = a_sorted[max(0, int(n * frac))]
            if thresh < 1e-10:
                continue
            a_try = a.copy()
            a_try[a_try < thresh * 0.01] = 0.0
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-10:
                best_sc = sc_try
                best_a = a_try.copy()

        # Re-optimize on the reduced support
        if best_a.sum() > 0:
            a = best_a
            # Quick Newton refinement
            if time.monotonic() < deadline - 10:
                a_new, sc_new = _active_set_newton(a, min(deadline, time.monotonic() + 30),
                                                   max_iter=50, tol_active=0.002)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new
                    a = a_new

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.1 ** 2))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, active-set Newton,
    alternating lambda optimization, support reduction, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 400, time.monotonic() + 180),
                                         max_iter=300, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 350:
            a_as2, sc_as2 = _active_set_newton(best_seq, min(deadline - 350, time.monotonic() + 120),
                                               max_iter=200, tol_active=0.0005)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # Phase 3: Alternating lambda Newton
    print(f"\n[Phase 3] Alternating lambda (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_alt, sc_alt = _alternating_lambda_newton(best_seq, min(deadline - 250, time.monotonic() + 200),
                                                   max_outer=15, tol_active=0.002)
        update_best(a_alt, sc_alt, "alt_lambda")

        if time.monotonic() < deadline - 220:
            a_alt2, sc_alt2 = _alternating_lambda_newton(best_seq, min(deadline - 220, time.monotonic() + 100),
                                                         max_outer=10, tol_active=0.001)
            update_best(a_alt2, sc_alt2, "alt_lambda_tight")

    # Phase 4: Support reduction
    print(f"\n[Phase 4] Support reduction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_spr, sc_spr = _support_reduction(best_seq, deadline - 200, max_rounds=5)
        update_best(a_spr, sc_spr, "support_red")

        # Re-optimize after support reduction
        if sc_spr < best_score + 0.001 and time.monotonic() < deadline - 150:
            a_spr2 = _run_lbfgs(a_spr, deadline, min_time_remaining=100.0)
            update_best(a_spr2, _score(a_spr2), "lbfgs_post_support")
            if time.monotonic() < deadline - 80:
                a_spr3, sc_spr3 = _active_set_newton(a_spr2, min(deadline - 80, time.monotonic() + 60),
                                                     max_iter=100, tol_active=0.001)
                update_best(a_spr3, sc_spr3, "as_post_support")

    # Phase 5: Multi-n search
    print(f"\n[Phase 5] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_parent:
            continue

        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 30, time.monotonic() + 40),
                                                 max_iter=80, tol_active=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_final if False else time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        restart_configs = [
            (0.02, "small"),
            (0.05, "medium"),
            (0.10, "large"),
        ]
        for sigma, label in restart_configs:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 8, time.monotonic() + 20),
                                                 max_iter=60, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 7: Final ultra-tight polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_newton(best_seq, min(deadline - 2, time.monotonic() + 30),
                                           max_iter=200, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as_tight")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 3:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 3 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

        # One more active-set Newton after tiny beta
        if best_seq is not None and time.monotonic() < deadline - 1:
            a_fin4, sc_fin4 = _active_set_newton(best_seq, deadline - 0.5,
                                                  max_iter=100, tol_active=0.0002)
            update_best(a_fin4, sc_fin4, "final_final_as")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
