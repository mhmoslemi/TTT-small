import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _adaptive_beta(c, c_max, n_betas=12):

    c_sorted = np.sort(c)[::-1]

    gap_100 = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    base = max(gap_100, c_max * 0.01)
    # Start large, go small
    betas = base * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,

def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3):

    best_a = a.copy()

    v = np.zeros(n)
    mu = 0.8  # momentum coefficient

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())

        if active.sum() < 2:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Target pressure (geometric mean for log-space)
        p_target = np.exp(np.mean(np.log(p)))

        # Try different gammas via line search
        best_sc_try = _score_normalized(a)
        best_log_a = log_a.copy()
        
        for g in [0.1, 0.2, 0.4, 0.7, 1.0]:
            log_a_try = log_a + g * (np.log(p_target) - np.log(p))
            log_a_try = log_a_try - log_a_try.mean()
            a_try = _softmax(log_a_try)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_log_a = log_a_try.copy()
                best_gamma = g

        # Update with momentum
        if best_sc_try < _score_normalized(a):
            v = mu * v + (best_log_a - log_a)
            log_a = best_log_a
            a = _softmax(log_a)
            sc = _score_normalized(a)
            if sc < best_sc:

                best_a = a.copy()

def _resample(a, n_new):
    """Resample sequence a to length n_new using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new

def _make_smooth_profile(n, alpha=0.5, beta=0.3):
    """Create a smooth positive profile with tunable shape."""
    t = np.linspace(0, 1, n)
    # Bump function with tunable width
    a = np.exp(-alpha * (t - 0.5)**2 / max(beta**2, 1e-6))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

"""
Optimized step function search: adaptive L-BFGS with iterative active-set refinement,
pressure equalization with momentum, and multi-length exploration.
Target: minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _compute_betas(c, c_max):
    """Adaptive beta schedule from current convolution."""
    c_sorted = np.sort(c)[::-1]
    k = min(50, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[k]
    base = max(gap, c_max * 0.005)
    multipliers = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
                            0.015625, 0.0078125, 0.00390625, 0.001953125,
                            0.0009765625, 0.00048828125, 0.000244140625, 0.0001220703125])
    return base * multipliers

def _run_lbfgs(a_init, deadline, min_remaining=30.0):
    """Adaptive L-BFGS with log-sum-exp smoothing and beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    betas = _compute_betas(c_cur, max_c)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(15000, max(100, int(remaining / (est_time * 12))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxcor': 80, 'maxls': 300}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _pressure_equalize(a, deadline, n_iters=300, gamma=0.3):
    """Pressure equalization with adaptive active set and momentum."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)

    log_a = np.log(np.maximum(a, 1e-30))
    v = np.zeros(n)
    mu = 0.7

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive active set
        tol_frac = 0.003 * (1.0 + 0.5 * it / n_iters)
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 3:
            idx = np.argsort(c)[-max(3, int(0.005 * len(c))):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Geometric mean target
        log_p_mean = np.mean(np.log(p))

        # Try multiple step sizes
        best_sc_try = _score_normalized(a)
        best_update = None

        for g in [0.05, 0.1, 0.2, 0.4, 0.7]:
            update = g * (log_p_mean - np.log(p))
            log_a_try = log_a + update
            log_a_try = log_a_try - log_a_try.mean()
            a_try = _softmax(log_a_try)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_update = update

        if best_update is not None:
            v = mu * v + best_update
            log_a = log_a + best_update
            a = _softmax(log_a)
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            # No improvement: take a small step with momentum
            if np.linalg.norm(v) > 1e-10:
                log_a = log_a + 0.1 * v / np.linalg.norm(v) * np.mean(np.abs(v))
                log_a = log_a - log_a.mean()
                a = _softmax(log_a)
            v *= 0.5

def _coord_descent(a, deadline, n_rounds=200):
    """Coordinate descent focusing on peak-contributing coordinates."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Get top peaks
        n_peaks = min(3, len(c))
        peak_indices = np.argsort(c)[-n_peaks:]

        improved = False
        for pk in peak_indices:
            if time.monotonic() > deadline:
                break

            # Find the coordinate pair that contributes most to this peak
            j_max = min(n - 1, pk)
            j_min = max(0, pk - (n - 1))

            if j_max < j_min:
                continue

            best_j = None
            best_contrib = -1
            for j in range(j_min, j_max + 1):
                contrib = a[j] * a[pk - j]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_j = j

            if best_j is None:
                continue

            # Try reducing a[best_j]
            old_val = a[best_j]
            c_prev = c_max
            for factor in [0.5, 0.7, 0.85, 0.95, 1.1, 1.2]:
                a[best_j] = old_val * factor
                if a[best_j] < 0:
                    a[best_j] = old_val
                    break
                c_try = _full_convolve(a, a).max()
                if c_try < c_prev - 1e-14:
                    c_prev = c_try
                    improved = True
                else:
                    a[best_j] = old_val
                    break

            # Try the paired coordinate
            j_pair = pk - best_j
            if j_pair != best_j and 0 <= j_pair < n:
                old_val2 = a[j_pair]
                for factor in [0.5, 0.7, 0.85]:
                    a[j_pair] = old_val2 * factor
                    if a[j_pair] < 0:
                        a[j_pair] = old_val2
                        break
                    c_try = _full_convolve(a, a).max()
                    if c_try < c_prev - 1e-14:
                        c_prev = c_try
                        improved = True
                    else:
                        a[j_pair] = old_val2
                        break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new

def _fourier_perturb(a, rng, n_modes=5, amplitude=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 2)))
        if freq < len(z_hat):
            delta = amplitude * float(rng.uniform(0.5, 2.0)) * float(rng.standard_normal())
            phase = float(rng.uniform(0, 2 * np.pi))
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)

def _pipeline(a_init, deadline, min_remaining=30.0):
    """Full optimization pipeline: L-BFGS -> pressure eq -> L-BFGS -> coord descent."""
    n = len(a_init)
    cur_a = a_init.copy()
    cur_sc = _score(cur_a)

    if time.monotonic() < deadline - min_remaining - 60:
        a1 = _run_lbfgs(cur_a, deadline, min_remaining + 60)
        sc1 = _score(a1)
