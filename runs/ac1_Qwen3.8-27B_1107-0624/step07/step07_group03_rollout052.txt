- The Hessian H is circulant with first row determined by the active set.
- The eigenvalues are λ_m = (1/m_active) * sum_{k in active} 2*cos(2π*m*k/n) for m = 0, 1, ..., n-1.

- λ_m for m = 0, 1, ..., n//2
- The rest are determined by symmetry: λ_{n-m} = λ_m

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based active-set Newton with circulant Hessian approximation,
adaptive active-set selection, coordinate-wise exact polish, and multi-n search.
Key innovations: direct FFT solve for the KKT system (O(n log n) per iteration),
percentile-based active set, and analytical 1D coordinate optimization for final polish.

import numpy as np

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _fft_convolve(a):
    n = len(a)
    out_len = 2 * n - 1
    N = 1
    while N < out_len:
        N <<= 1
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_len]
    return r

def _coord_polish(a_init, deadline, n_sweeps=30):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    n_active = max(2, int(0.4 * (2 * n - 1)))
    part_idx = np.argpartition(c, -n_active)[-n_active:]
    active = np.zeros(2 * n - 1, dtype=bool)
    active[part_idx] = True
    m = n_active

    for sw in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        changed = False
        for i in range(n):
            A_val = (1.0 / m) if active[2 * i] else 0.0
            B_val = 0.0
            for k in part_idx:
                if k == 2 * i:
                    continue
                if 0 <= k - i < n:
                    B_val += a[k - i]
                if 0 <= i - k < n:
                    B_val += a[i - k]
            B_val = (2.0 / m) * B_val

            if A_val > 1e-14:
                a_i_new = max(0.0, -B_val / (2.0 * A_val))
            else:
                if B_val > 1e-14:
                    a_i_new = 0.0
                else:
                    a_i_new = a[i]
            if abs(a_i_new - a[i]) > 1e-14:
                a[i] = a_i_new
                changed = True

        s = a.sum()
        if s > 1e-15:
            a = a / s
        c = _fft_convolve(a)
        f_cur = float(c.max())
        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
        if not changed:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return e / s

        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main Newton from parent
    print(f"\n[Phase 1] Active-set Newton from parent (t={time.monotonic()-t_start:.1f}s)")
    as_deadline = min(deadline, t_start + 500.0)
    a_main = _active_set_newton_fft(parent, as_deadline, max_iter=600, top_frac_start=0.5, top_frac_end=0.25)
    update_best(a_main, _score(a_main), "newton_main")

    # Phase 2: Coordinate polish
    print(f"[Phase 2] Coordinate polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_cp = _coord_polish(best_seq, min(deadline - 200, t_start + 600.0), n_sweeps=20)
        update_best(a_cp, _score(a_cp), "coord_polish_1")

    # Phase 3: Second Newton round with tighter params
    print(f"[Phase 3] Newton round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_n2 = _active_set_newton_fft(best_seq, min(deadline - 150, t_start + 800.0),
                                       max_iter=400, top_frac_start=0.35, top_frac_end=0.15)
        update_best(a_n2, _score(a_n2), "newton_round2")

    # Phase 4: Coordinate polish round 2
    if time.monotonic() < deadline - 150:
        a_cp2 = _coord_polish(best_seq, min(deadline - 100, t_start + 900.0), n_sweeps=30)
        update_best(a_cp2, _score(a_cp2), "coord_polish_2")

    # Phase 5: Multi-n
    print(f"[Phase 5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent in n_candidates:
        n_candidates.remove(n_parent)
    else:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")
        sc_init = _score(a_init)
        a_opt = _active_set_newton_fft(a_init, min(deadline - 60, time.monotonic() + 120.0),
                                        max_iter=300, top_frac_start=0.45, top_frac_end=0.2)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2 = _coord_polish(a_opt, min(deadline - 30, time.monotonic() + 40.0), n_sweeps=10)
            update_best(a_opt2, _score(a_opt2), f"n_{n_try}_coord")

    # Phase 6: Diverse restarts
    print(f"[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for label, sigma in [("small", 0.03), ("mid", 0.08), ("large", 0.15)]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _active_set_newton_fft(p_start, min(deadline - 10, time.monotonic() + 40.0),
                                          max_iter=150, top_frac_start=0.4, top_frac_end=0.2)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # Phase 7: Final polish
    print(f"[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = _coord_polish(best_seq, min(deadline - 2, time.monotonic() + 20.0), n_sweeps=30)
        update_best(a_fin, _score(a_fin), "final_coord")
        a_fin2 = _active_set_newton_fft(best_seq, min(deadline - 2, time.monotonic() + 20.0),
                                         max_iter=200, top_frac_start=0.25, top_frac_end=0.1)
        update_best(a_fin2, _score(a_fin2), "final_newton")
        a_fin3 = _coord_polish(best_seq, min(deadline - 1, time.monotonic() + 10.0), n_sweeps=20)
        update_best(a_fin3, _score(a_fin3), "final_coord2")

    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
