def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_and_grad_smooth(z, beta, n):
    """Compute smoothed objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)

    e = np.exp(shifted)

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:

        obj = 2.0 * n * (c_max + beta * np.log(s))

    h = wc[n - 1:2 * n - 1]

    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _basin_hop(a_current, deadline, rng, n_hops=8, reserve=30.0):
    """Diverse basin hopping with multi-scale perturbations."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / max(1, n_hops))

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.8

        pert_type = h % 4
        z = np.log(np.maximum(best_a, 1e-20))
        z = z - z.mean()

        if pert_type == 0:
            # Gaussian in log-space (small)
            sigma = 0.03 + 0.02 * (h % 4)
            z = z + sigma * rng.standard_normal(n)
        elif pert_type == 1:
            # Gaussian in log-space (large)
            sigma = 0.08 + 0.04 * (h % 4)
            z = z + sigma * rng.standard_normal(n)
        elif pert_type == 2:
            # Fourier perturbation
            for _ in range(3 + h % 3):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.03 * rng.uniform(0.5, 3.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t_arr = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
        else:
            # Anti-symmetric perturbation (exploits the fact that a and a_rev have same score)
            z_rev = z[::-1]
            z = 0.5 * (z + z_rev) + 0.1 * rng.uniform(-1, 1, size=n)
            z = (z + z[::-1]) / 2  # enforce symmetry

        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        a_opt, _ = _lbfgs_smooth(np.log(np.maximum(a_pert, 1e-20)), beta, n, 2000)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Basin hop {h}: new best {best_sc:.10f}")

    return best_a, best_sc


def _final_aggressive_polish(a_start, deadline):
    """Final round of very careful polishing with tiny beta L-BFGS and Newton."""
    n = len(a_start)
    best_a = a_start.copy()
    best_sc = _score(a_start)
    z = np.log(np.maximum(best_a, 1e-20))
    z = z - z.mean()

    # Very small betas for L-BFGS
    c = _full_convolve(best_a, best_a)
    mc = float(c.max())
    for beta_frac in [1e-6, 5e-7, 2e-7, 1e-7, 5e-8]:
        if time.monotonic() > deadline:
            break
        beta = mc * beta_frac
        remaining = max(5.0, deadline - time.monotonic())
        max_iter = min(10000, max(500, int(remaining / 0.005)))
        a_new, z = _lbfgs_smooth(z, beta, n, max_iter)
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    # Nesterov polish
    if time.monotonic() < deadline - 5:
        a_ne, sc_ne = _nesterov_polish(best_a, deadline, beta_frac=1e-7, max_iter=20000)
        if sc_ne < best_sc:
            best_sc = sc_ne
            best_a = a_ne.copy()

    # Active set Newton
    if time.monotonic() < deadline - 2:
        a_nt, sc_nt = _active_set_newton(best_a, deadline, k_active=5, max_iter=100)
        if sc_nt < best_sc:
            best_sc = sc_nt
            best_a = a_nt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS smoothing, Nesterov acceleration,
    KKT Newton polishing, Fourier parameterization, and basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        parent = _make_chirp(4096, freq=1.0, eps=0.3)
        try_update(parent, "default_chirp")

    # Quick L-BFGS warmup on parent
    n_p = len(parent)
    print(f"  Quick L-BFGS warmup on parent (n={n_p})...")
    a_warm = _adaptive_lbfgs(parent, deadline, reserve_time=1500.0)
    try_update(a_warm, "warmup_lbfgs")

    # === Phase 2: Main optimization on best ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 600:
        a_main = _adaptive_lbfgs(best_seq, deadline, reserve_time=600.0)
        try_update(a_main, "main_lbfgs")

    # === Phase 3: Nesterov-accelerated polishing ===
    print(f"\n=== Phase 3: Nesterov polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        a_ne, sc_ne = _nesterov_polish(best_seq, deadline - 400, beta_frac=1e-6, max_iter=30000)
        try_update(a_ne, "nesterov")

    # === Phase 4: Active-set Newton ===
    print(f"\n=== Phase 4: KKT Newton (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_nt, sc_nt = _active_set_newton(best_seq, deadline - 200, k_active=5, max_iter=200)
        try_update(a_nt, "kkt_newton_k5")
    if time.monotonic() < deadline - 150:
        a_nt3, sc_nt3 = _active_set_newton(best_seq, deadline - 150, k_active=3, max_iter=200)
        try_update(a_nt3, "kkt_newton_k3")

    # === Phase 5: More L-BFGS refinement ===
    print(f"\n=== Phase 5: L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_r2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=200.0)
        try_update(a_r2, "lbfgs_r2")

    # === Phase 6: Multiplicative updates ===
    print(f"\n=== Phase 6: Multiplicative updates (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_mult, sc_mult = _multiplicative_update(best_seq, n_iter=200,
                                                 deadline=deadline - 100, deadline_check=50)
        try_update(a_mult, "multiplicative")

    # === Phase 7: Basin hopping ===
    print(f"\n=== Phase 7: Basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 80, rng,
                                   n_hops=10, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        lengths_to_try = [32768, 8192, 16384]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 40:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")
            try_update(a_interp, f"interp_n{target_n}")

            if time.monotonic() < deadline - 35:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30.0)
                try_update(a_opt, f"opt_n{target_n}")

    # === Phase 9: Final aggressive polishing ===
    print(f"\n=== Phase 9: Final aggressive polish (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _final_aggressive_polish(best_seq, deadline)
        try_update(a_fin, "final_polish")

    # === Phase 10: One more basin hop from final best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_hop2, _ = _basin_hop(best_seq, deadline, rng, n_hops=3, reserve=15.0)
        try_update(a_hop2, "basin_hop_final")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
