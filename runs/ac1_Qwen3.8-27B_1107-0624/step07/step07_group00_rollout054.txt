import time

    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):

    c = _fft_convolve(a, a)

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_smoothed_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / max(1e-10, (1.0 - plateau_frac))
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Use active-set gradient when temperature is low
        if T < max_c * 0.01 if 'max_c' in dir() else T < T_start * 0.1:
            # Weighted active gradient
            k_indices = np.argsort(c)[::-1][:active_k]
            p = np.zeros_like(c)
            for i, idx in enumerate(k_indices):
                p[idx] = decay ** i
            p_sum = p.sum()
            if p_sum < 1e-300:
                g = np.zeros(n)
            else:
                p = p / p_sum
                p_rev = p[::-1]
                conv_full = _fft_convolve(a, p_rev)
                g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        else:
            g = _compute_smoothed_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 30 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.02:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _binary_search_cd(a, deadline, max_rounds=100, top_k_constraints=10):
    """Coordinate descent with binary search on each coordinate.
    For each coordinate j, we find the optimal value by binary search,
    given that we're primarily concerned with the top-k convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    # Identify active constraints (lags where conv is close to max)
    tol = c_max * 0.999
    active_lags = np.where(c >= tol)[0]
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        for k in active_lags:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            
            if not js:
                continue
            
            for j in js:
                # Binary search for optimal a[j]
                # Current value
                lo = 0.0
                hi = best_a[j] * 3 + 1e-6
                
                # We want to find the value of a[j] that minimizes the score
                # while keeping sum = 1 (we renormalize after)
                best_val = best_a[j]
                best_local_score = best_score
                
                # Quick check: try a few values
                for _ in range(20):  # Binary search iterations
                    mid = (lo + hi) / 2
                    a_cand = best_a.copy()
                    a_cand[j] = mid
                    s = a_cand.sum()
                    if s < 1e-15:
                        break
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    # Check only the active lags
                    max_active = float(c_cand[active_lags].max()) if len(active_lags) > 0 else float(c_cand.max())
                    sc = 2.0 * n * max_active
                    if sc < best_local_score:
                        best_local_score = sc
                        best_val = mid
                        hi = mid
                    else:
                        lo = mid
                
                if best_local_score < best_score * (1.0 - 1e-12):
                    best_a[j] = best_val
                    s = best_a.sum()
                    if s > 1e-15:
                        best_a /= s
                    best_score = best_local_score
                    c = _fft_convolve(best_a, best_a)
                    c_max = float(c.max())
                    tol = c_max * 0.999
                    active_lags = np.where(c >= tol)[0]
                    improved = True
        
        if not improved:
            break
    
    return best_a, _score_norm(best_a)

def _freq_domain_perturb(a, rng, n_perturb_fracs=None, phase_scale=0.1):
    """Add random phase perturbations in frequency domain, then project back to non-negative."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    
    if n_perturb_fracs is None:
        n_perturb_fracs = np.random.default_rng().uniform(0.05, 0.3, size=5)
    
    # Perturb a random subset of frequency bins
    n_bins = len(A)
    for frac in n_perturb_fracs:
        # Choose random bins to perturb
        n_bins_to_perturb = max(1, int(n_bins * frac))
        idx = rng.choice(n_bins, size=n_bins_to_perturb, replace=False)
        # Add random phase
        phases = rng.uniform(-phase_scale, phase_scale, size=n_bins_to_perturb)
        A[idx] *= np.exp(1j * phases)
    
    # Also slightly perturb magnitudes
    mag_perturb = 1.0 + phase_scale * 0.5 * rng.standard_normal(n_bins)
    mag_perturb = np.maximum(mag_perturb, 0.5)
    A *= mag_perturb
    
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        # For n=8192, the gradient computation is O(n^2) which is too slow

        # Alternative: use the fact that the gradient of (a*a)[k] w.r.t. a[j] is a[k-j] (if valid)
        # So the gradient of sum_k lambda_k * (a*a)[k] w.r.t. a[j] is sum_k lambda_k * a[k-j] (if valid)
        # = (lambda_windowed * a)[j] where we take a window of lambda centered appropriately
        # 
        # This IS a convolution! Specifically:
        # grad[j] = sum_{k} lambda_k * a[k-j] * 1[0 <= k-j <= n-1]

        # = sum_{m=0}^{n-1} lambda_{j+m} * a[m] (with lambda_k = 0 for k out of range)
        # This is a "cross-correlation" of a with lambda (with appropriate shifts)
        # 
        # Using FFT: if we pad lambda to length 2n-1 and compute IFFT(FFT(a) * FFT(lambda_reversed)),
        # we get the cross-correlation

        # This is what we want: grad[j] = sum_m a[m] * lambda[m+j]
        # To compute via FFT: IFFT(FFT(a) * conj(FFT(lambda)))
        # Or: IFFT(FFT(a) * FFT(lambda_reversed))[shifted]

        # Cross-correlation: (a ⋆ lambda)[j] = grad_quad[j + n - 1] (or similar shift)
        # Actually for rfft, the indexing is different. Let me just use the forward approach:
        # grad[j] = sum_m a[m] * lambda[m+j] for m in [0, n-1] and m+j in [0, 2n-2]
        # This equals the "anti-diagonal" sum which is related to the convolution a * lambda_rev

        # d/d a[j] of (a*a)[k] = a[k-j] + a[j-k] (when both are in range, else just one term)
        # So the total gradient is: for each j, sum_k lambda_k * (a[k-j] if 0<=k-j<n else 0) + (a[j-k] if 0<=j-k<n else 0)
        # = sum_k lambda_k * a[k-j] * 1[0<=k-j<n] + sum_k lambda_k * a[j-k] * 1[0<=j-k<n]
        # First term: let m = k-j, k = j+m: sum_m lambda_{j+m} * a[m] * 1[0<=m<n] * 1[0<=j+m<2n-1]
        # Second term: let m = j-k, k = j-m: sum_m lambda_{j-m} * a[m] * 1[0<=m<n] * 1[0<=j-m<2n-1]

        # First term is (a ⋆ lambda)[j] (cross-correlation)
        # Second term is (a ⋆ lambda)[-j + 2n-2] or something...

        # sum_m lambda_{j-m} * a[m] = (a ⋆ lambda_rev_shifted)[j]
        # 
        # For simplicity and correctness, let me just compute it as:
        # grad[j] = (a * lambda_padded)[j + n - 1] ... no this is getting too complicated
        # 
        # Let me just use a different approach: compute the gradient by doing 2 FFT convolutions
        
        # Method: The gradient of sum_k lambda_k * (a*a)[k] w.r.t. a is:
        # 2 * (the part of a * lambda that falls in positions [n-1, 2n-2]) reversed

        conv_full = _fft_convolve(a, lambda_rev)
        g = 2.0 * conv_full[n-1:2*n-1][::-1]

        g_centered = g - g_mean  # Center to preserve sum

        a_new = a - lr * g_centered
        a_new = _project_simplex(a_new)
        a = a_new
        
        # Evaluate
        c_new = _fft_convolve(a, a)
        t_new = float(c_new.max())
        sc = 2.0 * n * t_new
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: (1) binary-search coordinate descent for precise polishing,
(2) swap-based moves between coordinate pairs for active constraint reduction,
(3) frequency-domain perturbation restarts for diversification,
(4) multi-n exploration (4096, 8192, 16384), (5) dual-ascent启发 for active set focus.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smoothed_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=5, decay=0.5,
                  eval_interval=100, clip_norm=0.5, use_backtrack=True,
                  plateau_frac=0.2):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0
    initial_max_c = _score_norm(a_init)

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / max(1e-10, (1.0 - plateau_frac))
            T = T_start * (T_end / max(1e-30, T_start)) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        c_max_cur = float(c.max())

        if T < initial_max_c * 0.005:
            k_indices = np.argsort(c)[::-1][:active_k]
            p = np.zeros_like(c)
            for i, idx in enumerate(k_indices):
                p[idx] = decay ** i
            p_sum = p.sum()
            if p_sum < 1e-300:
                g = np.zeros(n)
            else:
                p = p / p_sum
                p_rev = p[::-1]
                conv_full = _fft_convolve(a, p_rev)
                g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        else:
            g = _compute_smoothed_gradient(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 30 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.02:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and current_lr_mult > 0.05:
                current_lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _binary_search_cd(a, deadline, max_rounds=50, active_tol=0.999, max_coords_per_lag=15):
    """Coordinate descent with binary search. For each active lag, find
    bottleneck coordinates and optimize them precisely."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        tol = c_max * active_tol
        active_lags = np.where(c >= tol)[0]

        for k in active_lags:
            if k < n:
                js = list(range(max(0, k - n + 1), min(k, n - 1) + 1))
            else:
                js = list(range(k - (n - 1), n))
            if not js:
                continue

            # Select top coordinates by value
            js_arr = np.array(js)
            vals = best_a[js_arr]
            top_mask = np.argsort(vals)[::-1][:max_coords_per_lag]
            js_top = js_arr[top_mask].tolist()

            for j in js_top:
                lo = 0.0
                hi = best_a[j] * 2.5 + 1e-8
                best_val = best_a[j]
                best_local = best_score

                for _ in range(18):
                    mid = (lo + hi) / 2.0
                    a_cand = best_a.copy()
                    a_cand[j] = mid
                    s = a_cand.sum()
                    if s < 1e-15:
                        break
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_local:
                        best_local = sc
                        best_val = mid
                        hi = mid
                    else:
                        lo = mid

                if best_local < best_score * (1.0 - 1e-14):
                    best_a[j] = best_val
                    s = best_a.sum()
                    if s > 1e-15:
                        best_a /= s
                    best_score = best_local
                    c = _fft_convolve(best_a, best_a)
                    c_max = float(c.max())
                    improved = True

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _swap_cd(a, deadline, max_rounds=30, active_tol=0.999, max_swaps_per_round=200):
    """Swap-based coordinate descent: move mass from one coordinate to another
    to reduce active convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    rng = np.random.default_rng(123)

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False

        tol = c_max * active_tol
        active_lags = np.where(c >= tol)[0]

        # For each active lag, find the pair that contributes most
        for k in active_lags:
            if k < n:
                js = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = np.arange(k - (n - 1), n)
            if len(js) == 0:
                continue

            # Contribution of each j to lag k is a[j] * a[k-j]
            partners = k - js
            contrib = best_a[js] * best_a[partners]
            # Find the top contributor
            top_idx = np.argmax(contrib)
            j_hi = js[top_idx]
            j_lo = partners[top_idx]

            if j_hi == j_lo:
                # Diagonal: just reduce a[j]
                for frac_out in [0.3, 0.5, 0.7]:
                    amount = best_a[j_hi] * frac_out
                    # Distribute to random low-value positions
                    simple_mask = best_a < np.median(best_a)
                    if not np.any(simple_mask):
                        simple_mask = np.ones(n, dtype=bool)
                    candidates = np.where(simple_mask & (np.arange(n) != j_hi))[0]
                    if len(candidates) == 0:
                        continue
                    dest = rng.choice(candidates, size=min(5, len(candidates)), replace=False)
                    for d in dest:
                        a_cand = best_a.copy()
                        a_cand[j_hi] -= amount / len(dest)
                        a_cand[d] += amount / len(dest)
                        a_cand = np.maximum(a_cand, 0)
                        s = a_cand.sum()
                        if s < 1e-15:
                            continue
                        a_cand /= s
                        c_cand = _fft_convolve(a_cand, a_cand)
                        sc = 2.0 * n * float(c_cand.max())
                        if sc < best_score * (1.0 - 1e-14):
                            best_score = sc
                            best_a = a_cand
                            c = c_cand
                            c_max = float(c.max())
                            improved =
                            break
                    if improved:
                        break
            else:
                # Off-diagonal: try moving mass from j_hi to a different position
                # Reduce a[j_hi] and increase a[d] where d doesn't contribute to active lags
                simple_mask = best_a < np.percentile(best_a, 25)
                simple_mask[j_hi] = False
                simple_mask[j_lo] = False
                candidates = np.where(simple_mask)[0]
                if len(candidates) < 3:
                    continue
                n_try = min(10, len(candidates))
                dests = rng.choice(candidates, size=n_try, replace=False)
                for d in dests:
                    for frac_out in [0.2, 0.4, 0.6]:
                        amount = best_a[j_hi] * frac_out
                        a_cand = best_a.copy()
                        a_cand[j_hi] -= amount
                        a_cand[d] += amount
                        a_cand = np.maximum(a_cand, 0)
                        s = a_cand.sum()
                        if s < 1e-15:
                            continue
                        a_cand /= s
                        c_cand = _fft_convolve(a_cand, a_cand)
                        sc = 2.0 * n * float(c_cand.max())
                        if sc < best_score * (1.0 - 1e-14):
                            best_score = sc
                            best_a = a_cand
                            c = c_cand
                            c_max = float(c.max())
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved and max_swaps_per_round <= 0:
                break
        if not improved:
            break

    return best_a, _score_norm(best_a)


def _freq_domain_perturb(a, rng, phase_scale=0.1, mag_scale=0.05, frac_perturb=0.15):
    """Add random phase and magnitude perturbations in frequency domain."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    n_bins = len(A)
    n_perturb = max(1, int(n_bins * frac_perturb))
    idx = rng.choice(n_bins, size=n_perturb, replace=False)
    # Skip DC bin (index 0)
    idx = idx[idx != 0] if len(idx) > 1 else idx[1:] if len(idx) > 0 else np.array([], dtype=int)
    if len(idx) > 0:
        phases = rng.uniform(-phase_scale, phase_scale, size=len(idx))
        mags = 1.0 + mag_scale * rng.standard_normal(len(idx))
        mags = np.maximum(mags, 0.5)
        A[idx] *= np.exp(1j * phases) * mags
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _dual_polish(a_init, deadline, max_iter=2000):
    """Subgradient dual method focused on active constraints."""
    n = len(a_init)
    a = a_init.copy()
    c = _fft_convolve(a, a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        tol = c_max * 0.9995
        active = c >= tol
        n_active = active.sum()
        if n_active == 0:
            active[np.argmax(c)] = True

        # Subgradient of max at active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / n_active
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]

        # Mirror descent step in log-space
        z = np.log(np.maximum(a, 1e-15))
        lr = 0.1 / (np.linalg.norm(g) + 1e-10)
        lr = min(lr, 0.5)
        z -= lr * g
        z = np.clip(z, -50, 50)
        z -= z.mean()
        a_new = _softmax_stable(z)

        # Accept if better
        sc = _score_norm(a_new)
        if sc < best_score * (1.0 + 1e-10):
            a = a_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    return best_a, best_score

            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n_parent}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: Adam warmup on parent (0-100s) ===
    phase1_end = min(deadline, t_start + 100.0)
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    a_opt, sc = _adam_z_space(a, phase1_end,
                              T_start=0.003 * max_c, T_end=0.00003 * max_c,
                              lr_start=0.03, lr_end=0.002,
                              max_iter=200000, active_k=5,
                              eval_interval=80, use_backtrack=True,
                              plateau_frac=0.3)
    full_sc = _score(a_opt)
    print(f"Phase 1 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # === Phase 2: Binary search CD (100s-350s) ===
    phase2_end = min(deadline, t_start + 350.0)
    if time.monotonic() < phase2_end:
        a_cd, sc_cd = _binary_search_cd(best_a, phase2_end, max_rounds=40, active_tol=0.999)
        full_sc_cd = _score(a_cd)
        print(f"Phase 2 BS-CD: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # === Phase 3: Swap-based CD (350s-500s) ===
    phase3_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase3_end:
        a_swap, sc_swap = _swap_cd(best_a, phase3_end, max_rounds=25, active_tol=0.999)
        full_sc_swap = _score(a_swap)
        print(f"Phase 3 Swap-CD: score={full_sc_swap:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_swap) and full_sc_swap < best_score:
            best_score = full_sc_swap
            best_a = a_swap.copy()

    # === Phase 4: Dual polish (500s-600s) ===
    phase4_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase4_end:
        a_dual, sc_dual = _dual_polish(best_a, phase4_end, max_iter=3000)
        full_sc_dual = _score(a_dual)
        print(f"Phase 4 Dual: score={full_sc_dual:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_dual) and full_sc_dual < best_score:
            best_score = full_sc_dual
            best_a = a_dual.copy()

    # === Phase 5: Try n=4096 (600s-850s) ===
    phase5_end = min(deadline, t_start + 850.0)
    n_current = len(best_a)
    if time.monotonic() < phase5_end:
        a_4k = _interp_to_length(best_a, 4096)
        c_4k = _fft_convolve(a_4k, a_4k)
        max_4k = float(c_4k.max())
        mid5 = min(phase5_end, t_start + 750.0)
        a_4k_opt, sc_4k = _adam_z_space(a_4k, mid5,
                                         T_start=0.002 * max_4k, T_end=0.00002 * max_4k,
                                         lr_start=0.025, lr_end=0.0015,
                                         max_iter=150000, active_k=5,
                                         eval_interval=100, use_backtrack=True)
        # Polish at 4096
        if time.monotonic() < phase5_end - 30:
            a_4k_cd, _ = _binary_search_cd(a_4k_opt, phase5_end - 10, max_rounds=20, active_tol=0.999)
            a_4k_opt = a_4k_cd
        full_sc_4k = _score(a_4k_opt)
        print(f"Phase 5 n=4096: score={full_sc_4k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_4k) and full_sc_4k < best_score:
            best_score = full_sc_4k
            best_a = a_4k_opt.copy()

    # === Phase 6: Try n=16384 (850s-1100s) ===
    phase6_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase6_end:
        a_16k = _interp_to_length(best_a, 16384)
        c_16k = _fft_convolve(a_16k, a_16k)
        max_16k = float(c_16k.max())
        a_16k_opt, sc_16k = _adam_z_space(a_16k, phase6_end - 10,
                                           T_start=0.002 * max_16k, T_end=0.00002 * max_16k,
                                           lr_start=0.02, lr_end=0.001,
                                           max_iter=100000, active_k=5,
                                           eval_interval=200, use_backtrack=True)
        full_sc_16k = _score(a_16k_opt)
        print(f"Phase 6 n=16384: score={full_sc_16k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_16k) and full_sc_16k < best_score:
            best_score = full_sc_16k
            best_a = a_16k_opt.copy()

    # === Phase 7: Frequency-domain perturbation restarts (1100s-1500s) ===
    phase7_end = min(deadline, t_start + 1500.0)
    n_best = len(best_a)
    n_restarts = 6
    for attempt in range(n_restarts):
        if time.monotonic() > phase7_end - 80:
            break
        if attempt == 0:
            a_pert = best_a + 0.008 * rng.standard_normal(n_best)
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
        elif attempt == 1:
            a_pert = _freq_domain_perturb(best_a, rng, phase_scale=0.08, mag_scale=0.03, frac_perturb=0.1)
        elif attempt == 2:
            a_pert = _freq_domain_perturb(best_a, rng, phase_scale=0.2, mag_scale=0.08, frac_perturb=0.2)
        elif attempt == 3:
            a_pert = best_a + 0.02 * rng.standard_normal(n_best)
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
        elif attempt == 4:
            # Low-frequency modulation
            freq = rng.uniform(0.5, 3.0)
            mod = 1.0 + 0.08 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best + rng.uniform(0, 2*np.pi))
            a_pert = best_a * mod
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
        else:
            # Multi-scale: downsample, perturb, upsample
            a_down = _interp_to_length(best_a, n_best // 4)
            a_down = a_down + 0.03 * rng.standard_normal(len(a_down))
            a_down = np.maximum(a_down, 0)
            a_down /= a_down.sum()
            a_pert = _interp_to_length(a_down, n_best)

        c_p = _fft_convolve(a_pert, a_pert)
        max_cp = float(c_p.max())
        time_per = max(50.0, (phase7_end - time.monotonic()) / max(1, (n_restarts - attempt)))
        sub_dl = min(phase7_end - 15, time.monotonic() + time_per)
        a_pert_opt, sc_pert = _adam_z_space(a_pert, sub_dl,
                                             T_start=0.0005 * max_cp, T_end=0.000005 * max_cp,
                                             lr_start=0.015, lr_end=0.0008,
                                             max_iter=80000, active_k=5,
                                             eval_interval=60, use_backtrack=True)
        full_sc_pert = _score(a_pert_opt)
        print(f"  Restart[{attempt}]: score={full_sc_pert:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_pert) and full_sc_pert < best_score:
            best_score = full_sc_pert
            best_a = a_pert_opt.copy()

    # === Phase 8: Final intensive polish (1500s-deadline) ===
    if time.monotonic() < deadline - 30:
        # Binary search CD
        cd_end = min(deadline - 60, time.monotonic() + 200)
        a_cd_f, sc_cd_f = _binary_search_cd(best_a, cd_end, max_rounds=30, active_tol=0.9995)
        full_sc_cd_f = _score(a_cd_f)
        print(f"Phase 8 BS-CD final: score={full_sc_cd_f:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd_f) and full_sc_cd_f < best_score:
            best_score = full_sc_cd_f
            best_a = a_cd_f.copy()

        # Swap CD
        if time.monotonic() < deadline - 40:
            sw_end = min(deadline - 40, time.monotonic() + 100)
            a_sw_f, sc_sw_f = _swap_cd(best_a, sw_end, max_rounds=15, active_tol=0.9995)
            full_sc_sw_f = _score(a_sw_f)
            print(f"Phase 8 Swap final: score={full_sc_sw_f:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_sw_f) and full_sc_sw_f < best_score:
                best_score = full_sc_sw_f
                best_a = a_sw_f.copy()

        # Ultra-low temperature Adam
        if time.monotonic() < deadline - 25:
            c_fin = _fft_convolve(best_a, best_a)
            max_cf = float(c_fin.max())
            a_final, sc_final = _adam_z_space(best_a, deadline - 20,
                                               T_start=0.00003 * max_cf, T_end=0.000001 * max_cf,
                                               lr_start=0.004, lr_end=0.0002,
                                               max_iter=100000, active_k=5,
                                               eval_interval=40, use_backtrack=True,
                                               plateau_frac=0.0)
            full_sc_final = _score(a_final)
            print(f"Phase 8 Adam final: score={full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_final) and full_sc_final < best_score:
                best_score = full_sc_final
                best_a = a_final.copy()

        # Last second of CD
        if time.monotonic() < deadline - 5:
            a_cd_last, _ = _binary_search_cd(best_a, deadline - 3, max_rounds=5, active_tol=0.9998)
            sc_cd_last = _score(a_cd_last)
            if sc_cd_last < best_score:
                best_score = sc_cd_last
                best_a = a_cd_last.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
