            break
        t += 1
        frac = min(1.0, t / max_iter)
        # Temperature: polynomial cooling (slower than geometric)
        T = T_start * (T_end / T_start) ** (frac ** 0.7)
        # Learning rate: polynomial decay
        lr = lr_start * (lr_end / lr_start) ** (frac ** 0.8) * current_lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Choose gradient method based on phase
        if T < T_start * 0.01 and frac > 0.7:
            # Use active-set gradient
            c_max = float(c.max())
            tol = c_max * 0.995
            active = np.where(c >= tol)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            elif len(active) > active_k:
                top_idx = np.argsort(c[active])[::-1][:active_k]
                active = active[top_idx]
            g = _compute_gradient_at_lags(a, c, active)
        else:
            # Smoothed gradient via FFT
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s
            # Compute g[j] = 2 * sum_k p[k] * a[k-j]
            # Using the direct formula with bounds
            g = np.zeros(n)
            # For efficiency, use the top few peaks
            c_sorted = np.argsort(c)[::-1]
            num_active = min(active_k + 3, len(c_sorted))
            for idx in range(num_active):
                k = c_sorted[idx]
                if p[k] < c[-1] * 2:  # if weight is negligible, skip
                    break
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo <= hi:
                    g[lo:hi+1] += p[k] * 2.0 * a[k-hi:k-lo+1][::-1]
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        # Center the gradient (for log-space parameterization)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        
        # Backtracking
        if t % backtrack_interval == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.005:
                # Revert
                z_new = z.copy()
                m_buf *= 0.5
                v_buf *= 0.5
            z = z_new
        else:
            z = z_new
        
        # Evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                last_improve_t = t
            else:
                patience += 1
            
            # Adaptive LR reduction
            if patience > 8 and t - last_improve_t > 500:
                current_lr_mult *= 0.7
                patience = 0
                if current_lr_mult < 0.03:
                    # Restart from best with perturbation
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z += 0.01 * np.random.randn(n)
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    current_lr_mult = 0.3
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _subgradient_polish(a, deadline, max_iter=50000, top_k=3, initial_lr=0.005):
    """Direct subgradient method on the max convolution peak."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = initial_lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Get top-k peaks (equalized)
        sorted_idx = np.argsort(c)[::-1]
        # Only include peaks that are very close to max
        active = [sorted_idx[0]]
        for idx in sorted_idx[1:top_k]:
            if c[idx] > c_max * 0.999:
                active.append(idx)
        
        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        g /= len(active)
        
        # Project gradient to the tangent space of the sum constraint
        # (subtract the mean to ensure the step preserves the sum)
        g_tangent = g - np.sum(g) * a  # approximate projection
        
        g_norm = np.linalg.norm(g_tangent)
        if g_norm < 1e-15:
            break
        
        # Take step
        step = lr * g_tangent / g_norm
        a_new = a - step
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score * (1 - 1e-12):
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            lr *= 0.9995
        else:
            lr *= 0.98
            if lr < 1e-8:
                break
    
    return best_a, _score_norm(best_a)


def _active_set_equalization(a, deadline, max_iter=5000, tol=0.001):
    """Try to equalize the top peaks by solving a linear system."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Find active peaks
        tol_val = c_max * (1 - tol)
        active = np.where(c >= tol_val)[0]
        if len(active) < 2:
            break
        if len(active) > 15:
            sorted_a = active[np.argsort(c[active])[::-1][:15]]
            active = sorted_a
        
        # Build the linear system:
        # For each active peak k: sum_j M[k,j] * d[j] = -1
        # where M[k,j] = 2*a[k-j] if 0 <= k-j < n, else 0
        # Plus constraint: sum_j d[j] = 0
        # Solve minimum-norm: d = M^T (M M^T)^{-1} b
        
        K = len(active)
        M = np.zeros((K, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                M[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Add sum constraint
        M_full = np.vstack([M, np.ones(n)])
        b = np.concatenate([-np.ones(K), [0.0]])
        
        # Minimum norm solution: d = M_full^T @ pinv(M_full @ M_full^T) @ b
        # But M_full is (K+1) x n with n >> K+1, so:
        # d = M_full^T @ inv(M_full @ M_full^T) @ b
        G = M_full @ M_full.T  # (K+1) x (K+1)
        try:
            G_inv = np.linalg.inv(G + 1e-12 * np.eye(G.shape[0]))
            d = M_full.T @ (G_inv @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(M_full, b, rcond=None)[0]
        
        # Scale the step
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Try multiple step sizes
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_cand = a + alpha * d
            a_cand = np.maximum(0, a_cand)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand /= s
            sc = _score_norm(a_cand)
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
        
        if not improved:
            # Try small random perturbation in the null space
            break
    
    return best_a, best_score


def _coordinate_descent_polish(a, deadline, max_rounds=200):
    """Targeted coordinate descent on elements that affect the peak most."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    
    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        
        # Find the peak lag
        k_max = int(np.argmax(c))
        
        # Elements that contribute to this peak
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        
        # Try modifying each contributing element
        for j in range(lo, hi + 1):
            old_val = best_a[j]
            if old_val < 1e-15:
                continue
            for scale in [0.7, 0.85, 0.95, 1.1, 1.2]:
                a_cand = best_a.copy()
                a_cand[j] = old_val * scale
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = float(c_cand.max()) * 2.0 * n
                if sc < best_score * (1 - 1e-12):
                    best_score = sc
                    best_a = a_cand.copy()
                    c = c_cand
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Escalate: try all coordinates
            any_improve = False
            for j in range(0, n, max(1, n // 200)):
                old_val = best_a[j]
                if old_val < 1e-15:
                    continue
                for scale in [0.5, 0.8, 1.25, 1.5]:
                    a_cand = best_a.copy()
                    a_cand[j] = old_val * scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = float(c_cand.max()) * 2.0 * n
                    if sc < best_score * (1 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        c = c_cand
                        any_improve = True
                        improved = True
                        break
            if not any_improve:
                break
    
    return best_a, best_score


def _multi_temperature_adam(a_init, deadline, n_temps=5):
    """Run Adam at multiple temperatures in succession, keeping the best."""
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    temps = [max_c * f for f in [0.01, 0.005, 0.002, 0.001, 0.0005]]
    lrs = [0.03, 0.02, 0.01, 0.005, 0.003]
    
    time_per = max(10.0, (deadline - time.monotonic()) / n_temps)
    
    for i in range(n_temps):
        sub_deadline = min(deadline, time.monotonic() + time_per)
        a_opt, sc = _adam_optimize(
            a_init=best_a,
            deadline=sub_deadline,
            T_start=temps[i],
            T_end=temps[i] * 0.01,
            lr_start=lrs[i],
            lr_end=lrs[i] * 0.1,
            max_iter=200000,
            eval_interval=100,
            active_k=5,
            backtrack_interval=150
        )
        full_sc = _score_norm(a_opt)
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()
        print(f"  Multi-T[{i}] T={temps[i]:.6f}: score={full_sc:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer focused on n=8192.

    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")
    
    n = len(a)
    best_a = a.copy()

    phase1_end = min(deadline, t_start + 500.0)

    a_opt, sc = _multi_temperature_adam(a, phase1_end, n_temps=5)
    if sc < best_score:

        print(f"Phase 4: Refined Adam (lower T)")
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a_opt2, sc2 = _adam_optimize(

        # Subgradient with very small steps

                                                 initial_lr=0.001)
        full_sc_final = _score(a_final)
        print(f"Phase 7a: score={full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()
        
        # Quick coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd_f, sc_cd_f = _coordinate_descent_polish(best_a, deadline - 3, max_rounds=50)
            full_sc_cd_f = _score(a_cd_f)
            if full_sc_cd_f < best_score:
                best_score = full_sc_cd_f
                best_a = a_cd_f.copy()
    
    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
