"""
Optimized step function sequence minimizer using soft pressure equalization
with beta-continuation, momentum-based updates, and multi-stage L-BFGS polishing.
Key improvements over previous: soft active set weighting, momentum, 
adaptive gamma scheduling, and better time allocation.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_pressure(a, beta, n):
    """Compute soft-weighted pressure using exp(c/beta) weighting."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n), c
    shifted = (c - c_max) / beta
    w = np.exp(np.clip(shifted, -700, 0))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    a_rev = a[::-1]
    p = _full_convolve(w, a_rev)
    p = p[n - 1:2 * n - 1]
    return np.maximum(p, 1e-30), c


def _hard_pressure(a, c, n, tol_frac=0.02):
    """Compute hard-active-set pressure."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    p = _full_convolve(w, a_rev)
    p = p[n - 1:2 * n - 1]
    return np.maximum(p, 1e-30)


def _pressure_equalize_soft(a_init, deadline, n_betas=20, gamma_start=0.5,
                            gamma_end=0.15, momentum=0.7, n_iters_per_beta=30,
                            beta_start=None, beta_end=None):
    """
    Soft pressure equalization with beta continuation and momentum.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score_normalized(a)
    a = a / s

    best_a = a.copy()
    best_sc = _score_normalized(a)

    # Compute initial beta range
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if beta_start is None:
        beta_start = max(gap * 2.0, c_max * 0.1)
    if beta_end is None:
        beta_end = max(gap * 0.1, c_max * 0.001)

    beta_schedule = np.geomspace(beta_start, beta_end, n_betas)

    # Momentum buffer for log updates
    log_momentum = np.zeros(n)

    total_iters = 0

    for bi in range(n_betas):
        if time.monotonic() > deadline:
            break
        beta = beta_schedule[bi]
        # Linearly interpolate gamma
        progress = bi / max(1, n_betas - 1)
        gamma = gamma_start + (gamma_end - gamma_start) * progress

        for it in range(n_iters_per_beta):
            if time.monotonic() > deadline:
                break
            total_iters += 1

            c = _full_convolve(a, a)
            c_max = float(c.max())
            if c_max < 1e-30:
                break

            # Soft pressure
            shifted = (c - c_max) / beta
            w = np.exp(np.clip(shifted, -700, 0))
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / w_sum

            a_rev = a[::-1]
            p = _full_convolve(w, a_rev)
            p = p[n - 1:2 * n - 1]
            p = np.maximum(p, 1e-30)

            # Compute target update ratio
            p_mean = float(np.mean(p))
            log_ratio = np.log(p_mean / p)
            log_ratio = np.clip(log_ratio, -3.0, 3.0)

            # Apply momentum
            log_momentum = momentum * log_momentum + (1 - momentum) * log_ratio

            # Update
            log_update = gamma * log_momentum
            log_update = np.clip(log_update, -1.0, 1.0)
            a = a * np.exp(log_update)
            a = np.maximum(a, 1e-30)
            s = a.sum()
            a = a / s

        # Check score at end of each beta stage
        sc = _score_normalized(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _pressure_equalize_with_newton(a_init, deadline, n_iters=200, gamma=0.3):
    """Pressure equalization with Newton correction step."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, _score_normalized(a)
    a = a / s

    best_a = a.copy()
    best_sc = _score_normalized(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive active set
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() < 3:
            idx = np.argsort(c)[-5:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Multiplicative update
        p_mean = float(np.mean(p))
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        a = a / s

        # Newton correction: try a small line search step
        # Direction: reduce a_i where p_i is high, increase where p_i is low
        if it % 10 == 5:
            c_new = _full_convolve(a, a)
            c_new_max = float(c_new.max())
            # If improvement is small, try a perturbation in the direction of reducing peak

                sc = _score_normalized(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                  0.005, 0.002, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 4096.0)
        est_time = 0.003 * scale
        max_iter = min(5000, max(50, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement; try smaller maxiter for next beta
                continue
        except Exception:
            pass

    return current_a

    best_a = a_init.copy()
    best_sc = _score(a_init)

        c_max = float(c.max())

        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.92
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "broad_taper":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _coordinate_polish(a, deadline, n_rounds=30):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(99999)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Find which elements contribute most to the peak
        peak_idx = int(np.argmax(c))
        # Contribution of a_i to c[peak_idx] is a_i * a[peak_idx - i] (when valid)
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
            j2 = peak_idx - (n - 1 - i)
            if 0 <= j2 < n:
                # This handles the other side of the convolution
                pass
        # Simpler: contribution to c[peak_idx] from a[i] is a[i]*a[peak_idx-i] + a[i]*a[peak_idx-(i-n+1)]...
        # Actually for full conv: c[k] = sum_i a[i]*a[k-i], so contribution of a[i] to c[k] is a[i]*a[k-i] (if 0<=k-i<n)
        # Multiple peaks
        tol = 0.03 * c_max
        active_peaks = np.where(c >= c_max - tol)[0]

        contrib = np.zeros(n)
        for pk in active_peaks[:20]:  # Limit to top 20 peaks
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contrib[i] += a[i] * a[j]

        n_try = min(n, max(5, n // 100))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = top_idx[rng.permutation(len(top_idx))[:min(len(top_idx), n_try)]]

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            c_old = c.max()

            for factor in [0.6, 0.8, 0.9, 1.05, 1.2]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[i] = new_val
                s = a.sum()
                a_norm = a / s
                c_try = _full_convolve(a_norm, a_norm).max()
                if c_try < c_old / s * s - 1e-15:  # This comparison is tricky, just check
                    # Simplified: just check if reducing helped
                    sc_try = _score_normalized(a_norm)
                    if sc_try < best_sc - 1e-12:
                        best_sc = sc_try
                        best_a = a_norm.copy()
                        improved = True
                        break
                a[i] = old_val

        if improved:
            s = a.sum()
            a = a / s

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0, label=""):
    """Full optimization pipeline: soft pressure -> L-BFGS -> pressure -> L-BFGS -> subgrad."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: Soft pressure equalization (main workhorse)
    time_budget = (deadline - time.monotonic() - min_remaining) * 0.35
    if time_budget > 10:
        a1, sc1_norm = _pressure_equalize_soft(
            current_a, time.monotonic() + time_budget,
            n_betas=25, gamma_start=0.5, gamma_end=0.1,
            momentum=0.75, n_iters_per_beta=20
        )
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: L-BFGS
    time_budget = (deadline - time.monotonic() - min_remaining) * 0.3
    if time_budget > 10:
        a2 = _run_lbfgs(current_a, time.monotonic() + time_budget,
                        min_time_remaining=5)
        sc2 = _score(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: More soft pressure with smaller betas
    time_budget = (deadline - time.monotonic() - min_remaining) * 0.2
    if time_budget > 8:
        a3, sc3_norm = _pressure_equalize_soft(
            current_a, time.monotonic() + time_budget,
            n_betas=15, gamma_start=0.25, gamma_end=0.05,
            momentum=0.8, n_iters_per_beta=25
        )
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Final L-BFGS
    time_budget = (deadline - time.monotonic() - min_remaining) * 0.15
    if time_budget > 8:
        a4 = _run_lbfgs(current_a, time.monotonic() + time_budget,
                        min_time_remaining=3)
        sc4 = _score(a4)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    # Stage 5: Subgradient polish
    time_budget = (deadline - time.monotonic() - min_remaining) * 0.1
    if time_budget > 5:
        a5, sc5 = _subgrad_polish(current_a, time.monotonic() + time_budget,
                                  n_max_steps=1500, lr_init=0.01)
        if sc5 < current_sc:
            current_a = a5
            current_sc = sc5

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining soft pressure equalization with beta-continuation,
    L-BFGS with log-sum-exp smoothing, and subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
                _update(parent)
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}, score={_score(parent):.10f}")
        _update(parent)

    # --- Phase 1: Main optimization on parent (most time) ---
    print(f"\n[Phase 1] Main optimization on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - time.monotonic()
    min_rem_p1 = max(400, remaining * 0.25)
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=min_rem_p1, label="main")
    _update(a_main, sc_main)

    # --- Phase 2: Extended soft pressure on best ---
    print(f"\n[Phase 2] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        time_budget = min(120, deadline - 300 - time.monotonic())
        if time_budget > 20:
            a_pe, sc_pe_norm = _pressure_equalize_soft(
                best_seq, time.monotonic() + time_budget,
                n_betas=30, gamma_start=0.4, gamma_end=0.05,
                momentum=0.8, n_iters_per_beta=30
            )
            sc_pe = _score(a_pe)
            _update(a_pe, sc_pe)
            if sc_pe < best_score:
                # Follow up with L-BFGS
                a_f = _run_lbfgs(a_pe, deadline, min_time_remaining=250)
                _update(a_f, _score(a_f))

    # --- Phase 3: Try different n values ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 10240, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)  # Larger first

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 40:
            break

        # Initialize by resampling
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, best_seq if best_seq is not None else parent)
        a_init = np.maximum(a_init, 0)
        s_init = a_init.sum()
        if s_init > 1e-10:
            a_init = a_init / s_init
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(30, remaining_for_this * 0.4)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem, label=f"n_{n_try}")
        _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)

        for i in range(4):
            if time.monotonic() > deadline - 50:
                break

            # Create diverse initialization
            if i == 0:
                a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, "tapered")
            elif i == 1:
                a_blend = 0.5 * best_seq + 0.5 * _make_profile(n_best, "broad_taper")
            elif i == 2:
                # Fourier perturbation
                z = np.log(np.maximum(best_seq, 1e-30))
                zh = np.fft.rfft(z)
                for _ in range(8):
                    freq = rng.integers(1, max(2, n_best // 4))
                    if freq < len(zh):
                        delta = 0.05 * rng.uniform(0.5, 2.0) * rng.standard_normal()
                        phase = rng.uniform(0, 2 * np.pi)
                        zh[freq] += delta * np.exp(1j * phase)
                        if freq < len(zh) - 1:
                            zh[-freq] = np.conj(zh[freq])
                z_new = np.fft.irfft(zh, n_best)
                z_new = z_new - z_new.mean()
                a_blend = _softmax(z_new)
            else:
                # Shrink small elements
                a_blend = best_seq.copy()
                threshold = np.percentile(a_blend, 5)
                a_blend[a_blend < threshold] = 0
                a_blend = a_blend / a_blend.sum()

            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()

            min_rem = max(20, (deadline - time.monotonic() - 50) * 0.3)
            a_r, sc_r = _multi_stage_optimize(a_blend, deadline, min_remaining=min_rem,
                                              label=f"restart_{i}")
            _update(a_r, sc_r)

    # --- Phase 5: Coordinate descent polish ---
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_budget = min(90, deadline - 30 - time.monotonic())
        if time_budget > 15:
            a_cd, sc_cd = _coordinate_polish(best_seq, time.monotonic() + time_budget, n_rounds=20)
            _update(a_cd, _score(a_cd))

    # --- Phase 6: Final fine optimization ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # One more round of soft pressure with very small gamma
        time_budget = min(40, deadline - 15 - time.monotonic())
        if time_budget > 10:
            a_fp, sc_fp = _pressure_equalize_soft(
                best_seq, time.monotonic() + time_budget,
                n_betas=10, gamma_start=0.15, gamma_end=0.03,
                momentum=0.85, n_iters_per_beta=40
            )
            _update(a_fp, _score(a_fp))

        # Final L-BFGS
        if time.monotonic() < deadline - 5:
            a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=2)
            _update(a_fin, _score(a_fin))

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.005)
            _update(a_sg, sc_sg)

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Verify with evaluate_sequence
        eval_sc = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: internal_score={sc_final:.10f}, eval_score={eval_sc:.10f}")
        print(f"n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
