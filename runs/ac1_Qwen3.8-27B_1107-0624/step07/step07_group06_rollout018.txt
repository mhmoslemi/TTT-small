"""
Optimized step function sequence minimizer using momentum-based projected gradient,
L-BFGS refinement, and active-set Newton polish. Exploits scale invariance and
FFT-based convolution for efficient high-dimensional optimization.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _lsexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_flip = a[::-1]
    N = _next_pow2(4 * n)
    fa = rfft(a_flip, N)
    fw = rfft(w, N)
    r = irfft(fa * fw, N)
    h = r[n - 1:2 * n - 1]
    W = float(np.dot(a, h))
    return 4.0 * n * a * (h - W)


def _autocorr_grad(a, n, w=None):
    """Compute gradient of weighted sum of autocorrelation w.r.t. a on simplex."""
    c = _full_convolve(a, a)
    if w is None:
        c_max = float(np.max(c))
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    a_flip = a[::-1]
    N = _next_pow2(4 * n)
    fa = rfft(a_flip, N)
    fw = rfft(w, N)
    r = irfft(fa * fw, N)
    h = r[n - 1:2 * n - 1]
    return 4.0 * n * h


def _hessian_matvec(v, active, n):
    """Compute H*v where H[i,j] = 2*active[i+j]."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fv = rfft(v_rev, N)
    fa = rfft(active, N)
    r = irfft(fv * fa, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=500, tol=1e-12):
    """Conjugate gradient solve Hx = b."""
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _momentum_anneal(a_init, deadline, min_time=300.0):
    """Momentum-based projected gradient with log-sum-exp annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    # Initial autocorrelation stats
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(np.max(c))
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    init_beta = max((c_max - gap) * 0.5, c_max * 0.02)
    final_beta = c_max * 1e-7

    # Geometric schedule
    n_steps = 3000
    betas = np.geomspace(init_beta, final_beta, n_steps)

    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    best_a = a_init.copy()
    best_sc = _score(a_init)

    for step in range(n_steps):
        if time.monotonic() > deadline - min_time:
            break

        beta = betas[step]
        a = _softmax(z)
        g = _lsexp_grad(z, beta, n)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()  # Re-center (softmax is invariant to shifts)

        # Periodic scoring
        if step % 100 == 0 or step == n_steps - 1:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                if step % 500 == 0:
                    print(f"    Anneal step {step}/{n_steps}: score={best_sc:.10f}, beta={beta:.2e}")

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _lbfgs_refine(a_init, deadline, min_time=60.0):
    """L-BFGS with aggressive beta continuation on the smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
                                  0.00005, 0.00002, 0.00001, 0.000005, 0.000001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.002 * max(1, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        def fun_local(z_local):
            return _lsexp_obj(z_local, beta, n)

        def jac_local(z_local):
            return _lsexp_grad(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200, 'maxfun': max_iter * 3}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if i % 3 == 0:
                    print(f"      L-BFGS beta_idx={i}: score={current_sc:.10f}")
        except Exception:
            pass

    return current_a


def _active_set_newton(a_init, deadline, max_iter=200)":
    """Active-set Newton method for precise minimization of max autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive tolerance for active set
        tol = max(1e-6 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient of sum_{k in active} c_k
        grad = _hessian_matvec(a, active.astype(np.float64), n)

        # Project to tangent space (sum = 0)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        # Newton step with regularization
        mu = 1e-10
        def hess_vec(v):
            return _hessian_matvec(v, active.astype(np.float64), n) + mu * v

        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=300, tol=1e-12)
        dx = dx - np.mean(dx)

        # Line search with Armijo condition
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try <= f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            if alpha < 1e-12:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if it % 20 == 0:
            print(f"      Newton iter {it}: c_max={c_max:.10f}, active={active.sum()}, score={best_sc:.10f}")

    return best_a, best_sc

            update_best(a_n, sc_n, f"round{round_idx}_newton")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n exploration t={time.monotonic() - t_start:.1f}s")
    if time.monotonic() < deadline - 60:
        a_mult, sc_mult = _try_multiple_n(best_seq, deadline, min_time=30.0)
        update_best(a_mult, sc_mult, "multi_n")

    # === Phase 6: Final polish ===
    print(f"\n[Phase 6] Final polish t={time.monotonic() - t_start:.1f}s")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _final_polish(best_seq, deadline - 2)
        update_best(a_fin, sc_fin, "final_polish")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
