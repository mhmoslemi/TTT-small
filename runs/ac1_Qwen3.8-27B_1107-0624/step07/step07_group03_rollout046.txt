def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _corr_topleft(A_vec, n):
    """
    Given A_vec of length 2n-1 (a window of the convolution index space),
    compute g[i] = sum_j a_or_v[j] * A_vec[i+j] for i=0..n-1.
    Here we compute it for a given vector v (standing in for a).
    """
    # g[i] = sum_j v[j] * A_vec[i+j]
    # This is a cross-correlation: flip v and convolve with A_vec, then extract
    flip_v = v[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fv = np.fft.rfft(flip_v, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fv * fA, N)[:out_len]
    return result[n-1:2*n-1]

def _parametric_exp(n, alpha, gamma):
    """Exponential decay profile: a[i] ∝ exp(-alpha * (i/n)^gamma)"""
    t = np.arange(n) / n
    a = np.exp(-alpha * t**gamma)
    a = np.maximum(a, 1e-30)
    return a / a.sum()

def _parametric_powersine(n, p, q):
    """Power-sine profile: a[i] ∝ sin(pi * (i/n)^p)^q"""
    t = np.arange(n) / n
    t = np.clip(t, 1e-10, 1 - 1e-10)
    a = np.sin(np.pi * t**p)**q
    a = np.maximum(a, 1e-30)
    return a / a.sum()

def _parametric_polynomial(n, gamma, center):

    t = np.arange(n) / n
    width = 0.8
    a = np.maximum(1 - ((t - center) / width)**2, 0)**gamma
    a = np.maximum(a, 1e-30)
    return a / a.sum()

def _parametric_bipolar(n, alpha):

def _fista_smoothed(a_init, deadline, p_norm=200, max_iter=5000):
    """
    FISTA on the P-norm smoothed objective.
    P-norm: M_p(a) = ((1/(2n-1)) * sum_k |c[k]|^p)^(1/p)
    As p -> inf, this approaches max(c).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    t_fista = 1.0
    z_prev = z.copy()
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_curr = _softmax(z_prev)
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        
        # P-norm value and gradient
        # M_p = ((1/(2n-1)) * sum |c|^p)^(1/p)
        # dM_p/da[q] = (1/p) * M_p^(1-p) * (1/(2n-1)) * sum_k p * |c[k]|^(p-1) * sign(c[k]) * d(c[k])/da[q]
        #           = M_p^(1-p) * (1/(2n-1)) * sum_k |c[k]|^(p-1) * sign(c[k]) * d(c[k])/da[q]
        # Since c[k] >= 0 (a >= 0): sign(c[k]) = 1
        # d(c[k])/da[q] = 2 * c_shift... = 2 * a[k-q] (as derived before)
        
        # Let w[k] = |c[k]|^(p-1) / ((2n-1) * M_p^(p-1))  (normalized weights)
        # Then grad_Mp[q] = 2 * sum_k w[k] * a[k-q] = 2 * _corr_topleft(w, n)[q]
