def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:

    c = _full_convolve(a, a)

"""
Optimized step-function coefficient search combining LP-based flattening,
L-BFGS with log-sum-exp smoothing, active-set Newton, and multi-n exploration.
Key innovation: constraint-generation LP for direct convex-direction search.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_A_rows(a, K):
    """Build sparse rows for LP constraint matrix. K is list of row indices into conv."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k_idx, k in enumerate(K):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        rows.extend([k_idx] * len(i_range))
        cols.extend(i_range.tolist())
        vals.extend((a[k - i_range]).tolist())
    A = coo_matrix((vals, (rows, cols)), shape=(len(K), n))
    return A.tocsr()


def _solve_flatten_lp(a, M, K):
    """Solve: max 1^T b s.t. conv(a,b)[k] <= M for k in K, b >= 0."""
    n = len(a)
    A = _build_A_rows(a, K)
    c_obj = -np.ones(n)
    b_rhs = np.full(len(K), M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_rhs,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _lp_flatten(a_init, deadline, max_outer=100, max_inner=30, n_init_active=100, max_active=2000):
    """LP-based flattening with constraint generation."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    print(f"  LP flatten start: n={n}, score={best_sc:.10f}")

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        M = c_max

        # Build initial active set
        sorted_idx = np.argsort(c)[::-1]
        K = sorted_idx[:n_init_active].tolist()

        b_new = None
        # Constraint generation
        for inner in range(max_inner):
            if time.monotonic() > deadline:
                break
            b_new = _solve_flatten_lp(a, M, K)
            if b_new is None:
                break
            b_new = np.maximum(b_new, 0)
            b_sum = b_new.sum()
            if b_sum < 1e-12:
                break
            # Check violated constraints
            conv_ab = _full_convolve(a, b_new)
            violated = conv_ab > M * (1 + 1e-8) + 1e-12
            if not violated.any():

            # Add most violated
            mv = int(np.argmax(conv_ab))
            if mv not in K:
                K.append(mv)
            if len(K) >= max_active:
                break

        if b_new is None or not np.all(np.isfinite(b_new)):
            break
        b_sum = float(b_new.sum())
        if b_sum < 1e-12:
            break
        b_new = b_new / b_sum

        # Check if b is essentially same as a
        diff = np.linalg.norm(a - b_new)
        if diff < 1e-10:
            break

        # Line search
        best_t = 0.0
        best_f = _score(a)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_try = (1.0 - t) * a + t * b_new

            if s_try < 1e-10:

        a = np.maximum(a, 0)

        # Convergence
        if sc - best_sc < 1e-14 and outer > 5:
            break

    print(f"  LP flatten done: score={best_sc:.10f}, iterations used")
    return best_a, best_sc


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time_per_iter = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time_per_iter * 10))))
        try:
            result = minimize(
                _lsobj, z, args=(beta, n),
                jac=_lsgrad,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 60, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _active_set_newton(a_init, deadline, max_iter=200, tol_active=0.001):
    """Damped Newton on active set."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    def _corr_a_A(a_vec, A_vec):
        flip_a = a_vec[::-1]
        out_len = n + len(A_vec) - 1
        N = _next_pow2(out_len)
        fa = np.fft.rfft(flip_a, N)
        fA = np.fft.rfft(A_vec, N)
        result = np.fft.irfft(fa * fA, N)[:out_len]
        return result[n - 1:2 * n - 1]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())
        if n_active > 2 * n:
            top_idx = np.argsort(c)[::-1][:2 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 2 * n

        A_vec = active.astype(np.float64)
        m = n_active
        g = (2.0 / m) * _corr_a_A(a, A_vec)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        mu = 1e-6 * (c_max / max(m, 1)) + 1e-8

        def _hess_matvec(v):
            return (2.0 / m) * _corr_a_A(v, A_vec) + mu * v

        # CG solve
        x = np.zeros(n)
        r = g_t - _hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)

            r_new = r - alpha * Ap

            if np.sqrt(rs_new) < 1e-12:

            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        for cg_it in range(200):
            Ap = _hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            w += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
        dx = dx - np.mean(dx)

        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-dx[neg_mask])) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False

            if np.min(a_try) < -1e-10:

            if s_try < 1e-15:

            t = min(0.1, t_max)

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient with active-set tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3)**2) + np.exp(-20 * (t - 0.7)**2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: LP flattening -> L-BFGS -> Active-set Newton ->
    Multi-n -> Restarts -> Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        _update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # === Phase 1: LP-based flattening ===
    print(f"\n[Phase 1] LP flattening (t={time.monotonic()-t_start:.1f}s)")
    lp_deadline = deadline - 500.0
    a_lp = parent.copy()
    a_lp, sc_lp = _lp_flatten(a_lp, lp_deadline, max_outer=80, max_inner=25,
                              n_init_active=80, max_active=1500)
    _update_best(a_lp, sc_lp, "lp_flatten")

    # Second LP pass from the improved point
    if time.monotonic() < deadline - 450:
        lp_deadline2 = min(deadline - 450, time.monotonic() + 120)
        a_lp2, sc_lp2 = _lp_flatten(best_seq, lp_deadline2, max_outer=40, max_inner=20,
                                    n_init_active=60, max_active=1000)
        _update_best(a_lp2, sc_lp2, "lp_flatten_2")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n[Phase 2] L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_lb = _run_lbfgs(best_seq, deadline, min_time_remaining=250.0)
        _update_best(a_lb, _score(a_lb), "lbfgs_1")

    # Another L-BFGS from a perturbed start
    if time.monotonic() < deadline - 200:
        a_pert = _perturb(best_seq, 0.05, rng)
        a_lb2 = _run_lbfgs(a_pert, deadline, min_time_remaining=150.0)
        _update_best(a_lb2, _score(a_lb2), "lbfgs_perturb")

    # === Phase 3: Active-set Newton ===
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        as_deadline = min(deadline - 130, time.monotonic() + 100)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=200, tol_active=0.001)
        _update_best(a_as, sc_as, "as_newton")

        if time.monotonic() < deadline - 100:
            as_d2 = min(deadline - 90, time.monotonic() + 50)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_d2, max_iter=100, tol_active=0.0005)
            _update_best(a_as2, sc_as2, "as_newton_tight")

    # Subgradient polish
    if time.monotonic() < deadline - 80:
        sg_d = min(deadline - 70, time.monotonic() + 60)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_d, n_max_steps=2000, lr_init=0.015)
        _update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue

        if parent is not None:
            a_init_n = _interpolate_to_n(parent, n_try)
        else:
            a_init_n = _make_profile(n_try, "uniform")

        # Brief LP + L-BFGS on this n
        if time.monotonic() < deadline - 100:
            a_try_lp, sc_try_lp = _lp_flatten(a_init_n, min(deadline - 80, time.monotonic() + 40),
                                              max_outer=20, max_inner=15, n_init_active=50, max_active=800)
            _update_best(a_try_lp, sc_try_lp, f"lp_n{n_try}")

        if time.monotonic() < deadline - 60:
            a_try_lb = _run_lbfgs(best_seq if len(best_seq) == n_try else a_init_n,
                                  deadline, min_time_remaining=50.0)
            _update_best(a_try_lb, _score(a_try_lb), f"lbfgs_n{n_try}")

    # === Phase 5: Diverse restarts ===
    print(f"\n[Phase 5] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            (0.03, "small"),
            (0.08, "mid"),
            (0.12, "large"),
        ]
        for sigma, label in restart_configs:
            if time.monotonic() > deadline - 20:
                break
            p_r = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_r, deadline, min_time_remaining=15.0)
            _update_best(a_r, _score(a_r), f"restart_{label}")

    # === Phase 6: Final polish ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Final active-set Newton with very tight tolerance
        as_final = min(deadline - 3, time.monotonic() + 40)
        a_fin, sc_fin = _active_set_newton(best_seq, as_final, max_iter=150, tol_active=0.0003)
        _update_best(a_fin, sc_fin, "final_as")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(2000, max(50, int(remaining / 0.05)))
                try:
                    result = minimize(
                        _lsobj, z, args=(beta_final, n),
                        jac=_lsgrad,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    _update_best(a_fin3, _score(a_fin3), "final_tiny_beta")
                except Exception:
                    pass

        # One more LP pass
        if time.monotonic() < deadline - 8:
            a_fin_lp, sc_fin_lp = _lp_flatten(best_seq, deadline - 3, max_outer=10,
                                              max_inner=10, n_init_active=40, max_active=500)
            _update_best(a_fin_lp, sc_fin_lp, "final_lp")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
