def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:

    c = _full_convolve(a, a)

    best_a = a_init.copy()
    best_sc = _score(a_init)

    a_cur = _softmax(z)
    v = np.zeros(n)  # momentum
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _conv(a_cur)
        c_max = c.max()
        c_mean = c.mean()

        # Adaptive weights: emphasize peaks that are far above mean
        idx_all = np.argsort(c)[::-1][:k_active]
        peak_vals = c[idx_all]
        excess = peak_vals - c_mean
        excess = np.maximum(excess, 1e-15)
        w = excess / excess.sum()

        # Compute subgradient as weighted average of individual gradients
        grad = np.zeros(n)
        for r in range(k_active):
            grad += w[r] * _grad_conv_fast(a_cur, idx_all[r], n)

        # Project gradient to be orthogonal to all-ones (to maintain sum)
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Normalized step
        step = lr * a_cur.sum() / gnorm
        v = momentum * v - step * grad / (n * a_cur.sum() + 1e-30)
        z_new = z + v
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -200.0, 200.0)

        a_try = _softmax(z_new)
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-15:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                v = 0.0
                if lr < 1e-10:
                    break
            elif stall % 10 == 0:
                lr *= 0.95

        z = z_new
        a_cur = _softmax(z)

    return best_a, best_sc


def _direct_subgrad(a_init, deadline, k_active=3, max_steps=30000):
    """Direct subgradient in a-space with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _conv(a)
        idx = np.argsort(c)[::-1][:k_active]
        # Weight by excess
        peak_vals = c[idx]
        c_mean = c.mean()
        excess = np.maximum(peak_vals - c_mean, 1e-15)
        w = excess / excess.sum()

        grad = np.zeros(n)
        for r in range(k_active):
            grad += w[r] * _grad_conv_fast(a, idx[r], n)
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Try step with backtracking
        alpha = step_size
        improved = False
        for _ in range(10):
            a_try = a - alpha * grad / (np.sqrt(n) * a.sum() + 1e-30)
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-30:
                alpha *= 0.5
                continue
            a_try = a_try / (a_try / a_try.sum())  # keep same sum
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                alpha = min(alpha * 1.05, 5.0 / np.sqrt(n))
                stall = 0
                break
            alpha *= 0.6

        if not improved:
            stall += 1
            if stall > 200:
                break
            if stall % 50 == 0:
                # Perturb to escape
                z = np.log(np.maximum(a, 1e-18))
                z += 0.01 * np.random.standard_normal(n)
                z = z - z.mean()
                a = _softmax(z)
                stall = 0

    return best_a, best_sc


def _newton_active_set(a_init, deadline, k_active=4, max_iter=100):
    """Newton method to equalize top-k peaks of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _conv(a)
        idx = np.argsort(c)[::-1][:k_active]
        peak_vals = c[idx]
        target = peak_vals.mean()
        residual = target - peak_vals  # we want to increase low peaks, decrease high peaks

        # Build system: for each active peak k_r, we want c[k_r] = target
        # d(c[k_r])/d(a[j]) = a[k_r - j] (for valid j)
        # We solve: sum_j G[r,j] * da[j] = residual[r]
        # where G[r,j] = a[idx[r] - j] (for valid j)
        # Plus constraint: sum(da) = 0 (maintain sum)

        # This is an underdetermined system (n unknowns, k_active+1 equations)
        # Use least squares with the sum constraint
        G = np.zeros((k_active + 1, n))
        for r in range(k_active):
            G[r, :] = _grad_conv_fast(a, idx[r], n)
        G[k_active, :] = 1.0  # sum constraint

        b = np.append(residual, 0.0)

        # Solve least squares (with regularization for stability)
        try:
            # Use conjugate gradient or direct solve
            A = G @ G.T + 1e-10 * np.eye(k_active + 1)
            B = G @ b
            da_all = np.linalg.solve(A, B)
            da = G.T @ da_all
        except Exception:
            break

        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a + alpha * da
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try * (a.sum() / s)
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            if alpha < 1e-15:
                break

    return best_a, best_sc


def _peak_shave(a_init, deadline, max_steps=5000):
    """Iteratively shave the highest peak by adjusting the sequence."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _conv(a)
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]
        c_mean = c.mean()

        if peak_val < c_mean * 1.001:
            break  # Nearly flat

        # Gradient direction to reduce this specific peak
        grad = _grad_conv_fast(a, k_peak, n)
        grad = grad - grad.mean()

        # Step in the direction that reduces the peak
        alpha = 0.1 / (np.linalg.norm(grad) + 1e-30)
        step = alpha * a.sum()

        a_try = a - step * grad / n
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            break
        a_try = a_try * (a.sum() / s)

        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            # Try smaller step
            a_try2 = a - 0.1 * step * grad / n
            a_try2 = np.maximum(a_try2, 0.0)
            s2 = a_try2.sum()
            if s2 > 1e-30:
                a_try2 = a_try2 * (a.sum() / s2)
                sc2 = _score(a_try2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = a_try2.copy()
                    a = a_try2
                else:
                    break
            else:
                break

    return best_a, best_sc


def _spectral_perturb(a, magnitude=0.05):
    """Perturb in the frequency domain."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)

    # Random phase perturbation (preserving magnitude approximately)
    phase_pert = magnitude * (np.random.standard_normal(len(A)) * 0.5 +
                              1j * np.random.standard_normal(len(A)) * 0.5)
    A_pert = A * (1.0 + phase_pert)

    # Ensure real and non-negative
    a_pert = np.fft.irfft(A_pert, N)[:n]
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-30:
        return a.copy()
    a_pert = a_pert * (a.sum() / s)
    return a_pert


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur, endpoint=False)
    t_new = np.linspace(0, 1, target_n, endpoint=False)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-15)
    s = a_interp.sum()
    if s > 1e-30:
        a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_good_random(n, rng, alpha=1.0):
    """Generate a random sequence with exponential-like distribution."""
    a = rng.exponential(1.0, n) ** alpha
    a = a + 0.01
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses L-BFGS smoothing, accelerated subgradient, Newton active-set,
    peak shaving, spectral perturbation, and multi-length exploration.
    """
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Load parent ===
    print("=== Loading parent ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw
                try_update(parent, "parent")
                print(f"  Parent: n={len(parent)}, score={best_score_val:.10f}")
    except Exception as e:
        print(f"  Could not load parent: {e}")

    if parent is None:
        parent = np.ones(4096) / 4096
        try_update(parent, "uniform_fallback")

    n = len(parent)
    print(f"Working with n={n}")

    # === Phase 1: L-BFGS multi-scale ===
    print(f"\n=== Phase 1: L-BFGS multi-scale (t={time.monotonic()-t_start:.1f}s) ===")
    current_a = parent.copy()

    c_cur = _conv(current_a)
    c_max_cur = float(c_cur.max())
    print(f"  Initial max_conv={c_max_cur:.6f}, score={_score(current_a):.10f}")

    beta_scales = [0.3, 0.15, 0.075, 0.04, 0.02, 0.01, 0.005, 0.002,
                   0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]

    for i, bs in enumerate(beta_scales):
        if time.monotonic() > deadline - 300:
            break
        beta = bs * c_max_cur
        remaining = deadline - 300 - time.monotonic()
        scale_factor = max(1.0, (n / 8192.0))
        max_iter = int(min(8000, max(200, remaining / (0.003 * scale_factor * 15))))

        print(f"  beta={beta:.2e}, max_iter={max_iter}")
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        print(f"  Score after: {sc:.10f}")

        if sc < _score(current_a) - 1e-15:
            current_a = a_new
            try_update(current_a, f"lbfgs_b{i}")
            c_cur = _conv(current_a)
            c_max_cur = float(c_cur.max())
        if _score(current_a) < 1.5001:
            print("  Reached target, stopping L-BFGS phase.")
            break

    # === Phase 2: Accelerated subgradient ===
    print(f"\n=== Phase 2: Accelerated subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    for k_act in [3, 5, 7]:
        if time.monotonic() > deadline - 200:
            break
        a_sg, sc_sg = _subgrad_accelerated(current_a, min(deadline - 200,
                                                         time.monotonic() + 300),
                                           k_active=k_act, max_steps=15000,
                                           lr_init=0.04)
        try_update(a_sg, f"accel_sg_k{k_act}")
        if a_sg is not None and sc_sg < _score(current_a):
            current_a = a_sg

    # === Phase 3: Newton active-set ===
    print(f"\n=== Phase 3: Newton active-set (t={time.monotonic()-t_start:.1f}s) ===")
    for k_act in [3, 4, 5]:
        if time.monotonic() > deadline - 150:
            break
        a_nw, sc_nw = _newton_active_set(current_a, min(deadline - 150,
                                                       time.monotonic() + 200),
                                         k_active=k_act, max_iter=80)
        try_update(a_nw, f"newton_k{k_act}")
        if a_nw is not None and sc_nw < _score(current_a):
            current_a = a_nw

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak shaving (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_ps, sc_ps = _peak_shave(current_a, deadline - 100, max_steps=5000)
        try_update(a_ps, "peak_shave")
        if a_ps is not None and sc_ps < _score(current_a):
            current_a = a_ps

    # === Phase 5: More L-BFGS (tighter) ===
    print(f"\n=== Phase 5: Tighter L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        c_cur = _conv(current_a)
        c_max_cur = float(c_cur.max())
        for bs in [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]:
            if time.monotonic() > deadline - 60:
                break
            beta = bs * c_max_cur
            remaining = deadline - 60 - time.monotonic()
            max_iter = int(min(5000, max(100, remaining / 0.05)))
            a_new = _lbfgs_run(current_a, beta, n, max_iter)
            sc = _score(a_new)
            if sc < _score(current_a) - 1e-16:
                current_a = a_new
                try_update(current_a, f"tight_lbfgs_{bs}")

    # === Phase 6: Direct subgradient ===
    print(f"\n=== Phase 6: Direct subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        a_ds, sc_ds = _direct_subgrad(current_a, deadline - 80, k_active=3,
                                      max_steps=20000)
        try_update(a_ds, "direct_sg")
        if a_ds is not None and sc_ds < _score(current_a):
            current_a = a_ds

    # === Phase 7: Basotine hopping (spectral perturbation + optimize) ===
    print(f"\n=== Phase 7: Basin hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 50:
        n_hops = 6
        hop_deadline_each = max(5.0, (deadline - 50 - time.monotonic()) / n_hops)
        for h in range(n_hops):
            if time.monotonic() > deadline - 40:
                break
            hop_end = time.monotonic() + hop_deadline_each * 0.8

            if h % 3 == 0:
                a_pert = _spectral_perturb(current_a, magnitude=0.03 + 0.02 * h)
            elif h % 3 == 1:
                # Gaussian in log-space
                z = np.log(np.maximum(current_a, 1e-18))
                z += (0.02 + 0.01 * h) * rng.standard_normal(n)
                z = z - z.mean()
                a_pert = _softmax(z)
            else:
                # Block perturbation
                z = np.log(np.maximum(current_a, 1e-18))
                n_blocks = 5
                block_sz = n // n_blocks
                for _ in range(n_blocks):
                    start = int(rng.integers(0, max(1, n - block_sz)))
                    end = min(n, start + block_sz)
                    z[start:end] += rng.uniform(-0.02, 0.02)
                z = z - z.mean()
                a_pert = _softmax(z)

            # Quick optimization from perturbation
            c_p = _conv(a_pert)
            mc = float(c_p.max())
            beta = mc * 0.005
            max_iter = int(min(2000, max(50, (hop_end - time.monotonic()) / 0.01)))
            a_opt = _lbfgs_run(a_pert, beta, n, max_iter)
            try_update(a_opt, f"hop_{h}")
            if _score(a_opt) < _score(current_a):
                current_a = a_opt

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 50:
        lengths_to_try = [n * 2, n, n // 2]
        lengths_to_try = [l for l in lengths_to_try if 512 <= l <= 32768]
        # Remove duplicates
        seen = set()
        unique_lengths = []
        for l in lengths_to_try:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)

        for target_n in unique_lengths:
            if time.monotonic() > deadline - 30:
                break
            if target_n == len(best_seq) if best_seq is not None else False:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq if best_seq is not None else current_a, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            # Optimize at new length
            if time.monotonic() < deadline - 20:
                c_i = _conv(a_interp)
                mc_i = float(c_i.max())
                for bs in [0.05, 0.01, 0.005]:
                    if time.monotonic() > deadline - 15:
                        break
                    beta = bs * mc_i
                    max_iter = int(min(3000, max(100, (deadline - 15 - time.monotonic()) / 0.02)))
                    a_opt = _lbfgs_run(a_interp, beta, target_n, max_iter)
                    try_update(a_opt, f"n{target_n}_b{bs}")
                    a_interp = a_opt

    # === Phase 9: Final intensive polishing ===
    print(f"\n=== Phase 9: Final polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(8):
            if time.monotonic() > deadline - 3:
                break

            # Alternating: subgradient -> Newton -> L-BFGS
            if rnd % 3 == 0:
                a_f, _ = _subgrad_accelerated(best_seq, deadline - 3,
                                              k_active=3, max_steps=5000,
                                              lr_init=0.02)
                try_update(a_f, f"final_sg_{rnd}")
            elif rnd % 3 == 1:
                a_f, _ = _newton_active_set(best_seq, deadline - 3,
                                            k_active=3, max_iter=50)
                try_update(a_f, f"final_newton_{rnd}")
            else:
                c_b = _conv(best_seq)
                mc_b = float(c_b.max())
                beta = mc_b * 5e-7
                max_iter = int(min(3000, max(100, (deadline - 3 - time.monotonic()) / 0.01)))
                a_f = _lbfgs_run(best_seq, beta, len(best_seq), max_iter)
                try_update(a_f, f"final_lbfgs_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL (internal): score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"FINAL (evaluator): score={sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(1024) / 1024
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
