"""
Multi-scale optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous: n=16384 exploration, growing active-set gradient,
aggressive multi-phase annealing with diverse restarts, and spectral initialization.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assume sum(a)=1. Returns 2n*max_conv."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_active_gradient(a, c, top_k, decay=0.7, T=0.0):
    """
    Compute gradient of the objective w.r.t. a.
    If T > 0: uses smoothed softmax approximation.
    If T == 0: uses weighted active-set (top_k peaks with exponential decay).
    Returns gradient vector of length n.
    """
    n = len(a)
    if T > 0:
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
    else:
        k_indices = np.argsort(c)[::-1][:top_k]
        p = np.zeros_like(c)
        for i, idx in enumerate(k_indices):
            p[idx] = decay ** i
        p_sum = p.sum()
        if p_sum < 1e-300:
            p[np.argmax(c)] = 1.0
        else:
            p = p / p_sum

    # grad[j] = 2 * sum_k p[k] * a[k-j] (with valid bounds)
    # This is 2 * cross-correlation of a and p, first n entries
    # Using FFT: corr(j) = ifft(conj(FFT(p)) * FFT(a)).real[j] for j >= 0
    N_fft = _next_pow2(len(p) + n - 1)
    Fp = np.fft.fft(p, N_fft)
    Fa = np.fft.fft(a, N_fft)
    corr = np.fft.ifft(np.conj(Fp) * Fa).real
    g = 2.0 * corr[:n]
    return g


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,

def _interp_to_length(a, n_new):
    n0 = len(a)

    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _spectral_init(n, rng):
    """
    Generate a non-negative sequence with relatively flat autocorrelation
    using random phase spectrum.
    """
    # Random phase in frequency domain
    N_fft = _next_pow2(2 * n)
    amp = np.ones(N_fft // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, size=N_fft // 2 + 1)
    phase[0] = 0  # DC component
    spec = amp * np.exp(1j * phase)
    # Make it Hermitian
    spec_full = np.zeros(N_fft, dtype=complex)
    spec_full[:N_fft // 2 + 1] = spec
    for i in range(1, N_fft // 2):
        spec_full[N_fft - i] = np.conj(spec[i])
    signal = np.fft.ifft(spec_full).real[:n]
    signal = np.maximum(signal, 0.0)
    s = signal.sum()
    if s < 1e-10:
        signal = rng.random(n)
        s = signal.sum()
    return signal / s


def _random_smooth_init(n, rng, n_bumps=50):
    """Generate non-negative sequence by summing random Gaussian bumps."""
    centers = rng.uniform(0, n, size=n_bumps)
    widths = rng.uniform(n * 0.01, n * 0.1, size=n_bumps)
    heights = rng.uniform(0.5, 1.0, size=n_bumps)
    x = np.arange(n)
    signal = np.zeros(n)
    for c, w, h in zip(centers, widths, heights):
        signal += h * np.exp(-0.5 * ((x - c) / w) ** 2)
    signal += 1e-6  # ensure positive
    return signal / signal.sum()

            if time.monotonic() > phase5_end - 30:
                break
            sigma = 0.02 + 0.02 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            time_per = max(25.0, (phase5_end - time.monotonic()) / (3 - attempt))
            sub_dl = min(phase5_end - 5, time.monotonic() + time_per)
            a_pert_opt, _ = _adam_z_space(a_pert, sub_dl,
                                           T_start=0.0005 * max_cp, T_end=0.000002 * max_cp,
                                           lr_start=0.015, lr_end=0.0005,
                                           max_iter=80000, active_k_start=1, active_k_end=8,
                                           eval_interval=60, use_backtrack=True)
            sc_pert = _score(a_pert_opt)
            print(f"  Perturb[{attempt}] sigma={sigma:.3f}: score={sc_pert:.10f} t={time.monotonic()-t_global_start:.0f}s")
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert_opt.copy()

    # === PHASE 6: Final ultra-fine polish (1650s - deadline) ===
    if time.monotonic() < deadline - 15:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _adam_z_space(best_a, deadline - 12,
                                    T_start=0.0001 * max_cf, T_end=0.0000005 * max_cf,
                                    lr_start=0.003, lr_end=0.0001,
                                    max_iter=100000, active_k_start=1, active_k_end=15,
                                    eval_interval=50, use_backtrack=True)
        sc_final = _score(a_final)
        print(f"Phase 6 final: score={sc_final:.10f}, t={time.monotonic()-t_global_start:.0f}s")
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

        if time.monotonic() < deadline - 5:
            a_cd_final, _ = _fast_cd_polish(best_a, deadline - 3, max_rounds=20)
            sc_cd = _score(a_cd_final)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cd_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_check = _score(final)
    print(f"FINAL: score={sc_check:.10f}, n={len(final)}, time={time.monotonic()-t_global_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
