def solve_direction_lp(a, M, active_set, max_active=500):
    """Solve the LP to find the direction b."""
    n = len(a)
    # Build constraint matrix for active set
    c_ub = np.full(len(active_set), M)
    # A_ub[k, i] = a[i] if 0 <= i < n and 0 <= active_set[k] - i < n
    # Build sparsely
    rows = []
    cols = []
    vals = []
    for k_idx, k in enumerate(active_set):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k_idx)
            cols.append(i)
            vals.append(a[i])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_set), n))
    
    c_vec = -np.ones(n)  # minimize -sum(b)
    
    res = linprog(c_vec, A_ub=A, b_ub=c_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
        # Check all constraints
        c_conv = full_convolve(a, b)
        violated = np.where(c_conv > M * (1 + 1e-8))[0]

"""
Step sequence optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
LP-based valley-filling direction finding (active-set constraint generation),
combined with log-sum-exp L-BFGS polishing, multi-length exploration,
and adaptive peak shaving. Key innovation: the LP direction finder provides
principled "valley-filling" directions that can escape local minima of
gradient-based methods.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lp_direction(a, M, max_active=1000, max_iter_lp=50):
    """Solve the LP to find valley-filling direction using active-set constraint generation."""
    n = len(a)
    # Start with just the peak
    c_aa = _full_convolve(a, a)
    active_set = [int(np.argmax(c_aa))]
    
    b_opt = None
    for lp_iter in range(max_iter_lp):
        # Build constraint matrix for active set
        rows = []
        cols = []
        vals = []
        for k_idx, k in enumerate(active_set):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                rows.append(k_idx)
                cols.append(i)
                vals.append(a[i])
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_set), n))
        c_vec = -np.ones(n)
        b_ub = np.full(len(active_set), M)
        
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            break
        
        b_candidate = res.x
        if b_candidate.sum() < 1e-15:
            break
        
        # Check all constraints
        c_ab = _full_convolve(a, b_candidate)
        tol = M * 1e-8
        violated = np.where(c_ab > M + tol)[0]
        
        if len(violated) == 0:
            b_opt = b_candidate
            break
        else:
            # Add violated constraints (limit total active set size)
            for v in violated:
                v = int(v)
                if v not in active_set:
                    active_set.append(v)
            if len(active_set) > max_active:
                # Keep only the most violated
                viol_vals = c_ab[violated]
                active_set = [int(np.argmax(c_aa))]
                top_violated = violated[np.argsort(viol_vals)[::-1]][:max_active - 1]
                for v in top_violated:
                    v = int(v)
                    if v not in active_set:
                        active_set.append(v)
                # Force resolution with limited set
                b_opt = b_candidate
                break
    
    if b_opt is None:
        # Fallback: return a (which is always feasible)
        b_opt = a.copy()
    
    return b_opt


def _lp_optimize(a_init, deadline, max_outer=500, min_time=5.0):
    """Iterative LP-based valley-filling optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline - min_time:
            break
        
        c_aa = _full_convolve(a, a)
        M = float(c_aa.max())
        sc = float(2.0 * n * M)  # since sum(a) = 1
        
        # Find LP direction
        b = _lp_direction(a, M, max_active=500, max_iter_lp=30)
        b = np.maximum(b, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Line search for step size
        best_t = 0.0
        best_sc_t = sc
        best_a_t = a
        
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(2.0 * n * c_try.max())
            if sc_try < best_sc_t - 1e-12:

                best_t = t
                best_a_t = a_try.copy()
        
        if best_t > 0:
            a = best_a_t
            if best_sc_t < best_sc - 1e-14:
                best_sc = best_sc_t
                best_a = a.copy()
            if outer % 10 == 0:
                elapsed = time.monotonic()
                print(f"    LP iter {outer}: score={best_sc:.10f}, t={best_t:.3f}")

def _smoobj_direct(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-20,
                                  'gtol': 1e-16, 'maxcor': 200,
                                  'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _lbfgs_schedule(a_start, deadline, min_time=30.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - min_time:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break
        est_per_iter = 0.002 * (n / 4096.0) ** 1.5
        max_iter = min(8000, max(100, int(remaining / (est_per_iter * 30))))
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

    return current_a


def _peak_shaving(a_init, deadline, max_iters=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Adaptive active set
        threshold = c_max * 0.999
        active = c >= threshold
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        g = g / g_norm
        
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            step = eta * frac
            a_try = a - step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                eta = min(eta * 1.02, 0.05)
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            eta *= 0.9
            if eta < 1e-9 or stall > 200:
                break
    
    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
