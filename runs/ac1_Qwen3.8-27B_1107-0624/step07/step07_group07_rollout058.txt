import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_multi(c, num_scales=5):
    """Try multiple active set sizes, return list of weight vectors."""
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return [w]
    
    sorted_c = np.sort(c)[::-1]
    n_c = len(c)
    weights = []
    
    for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        n_active = max(2, int(frac * n_c))
        n_active = min(n_active, n_c)
        threshold = sorted_c[min(n_active - 1, n_c - 1)]
        active = c >= threshold
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        weights.append(w)
    
    return weights


def _gradient_direct(a, c, n, w):
    """Compute gradient of w^T c = w^T (a*a) with respect to a."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 200,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _peak_equalization(a_init, deadline, max_iters=30000, K=5):

            # Gradient of conv[k] w.r.t. a:
            # d conv[k] / d a[i] = a_rev[k-i] (where valid)

                # d conv[k] / d a[j] = sum_i [delta(i,j) * a[k-i] + a[i] * delta(k-i,j)]
                # = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n) 
                # = 2 * a[k-j] (if 0 <= k-j <= n-1)
                # But if k-j == j (i.e., k = 2j), then it's just a[j] (not 2*a[j])
                # Wait no: delta(i,j)*a[k-i] gives a[k-j] when i=j
                # a[i]*delta(k-i,j) gives a[k-j] when k-i=j, i.e., i=k-j
                # So d conv[k]/d a[j] = a[k-j] (from first term, if k-j in [0,n-1])
                #                            + a[k-j] (from second term, if k-j in [0,n-1])
                # = 2*a[k-j] if k-j in [0,n-1]

                # The second term gives a[k-j] = a[j]  
                # So it's 2*a[j], which is correct (the cross term is counted once for each factor)
                
                if 0 <= k - j_valid <= n - 1:

        # d conv[k] / d a = 2 * (a_rev shifted by k) (with appropriate truncation)
        # For the "middle" part (n-1 to 2n-2), it's a reversed and shifted version
        
        # Actually, let's use the same trick as before:
        # If w is a weight vector on the convolution, then grad = 2 * (w * a_rev)[n-1:2n-1]
        # adjusted by the mean
        
        w = np.zeros_like(c)
        for k_idx in top_k:
            w[int(k_idx)] = 1.0
        w = w / w.sum()
        
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_targeted(a, rng, c, peak_frac=0.01):
    """Perturb the sequence in a way that targets the peak lags."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    c_max = c.max()
    threshold = c_max * (1 - peak_frac)
    active_lags = np.where(c >= threshold)[0]
    
    # For each active lag, add a small anti-correlation perturbation
    a_rev = a[::-1]
    pert = np.zeros(n)
    for k in active_lags[:20]:  # limit to top 20
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            pert[lo:hi+1] += a_rev[k - lo:k - lo + (hi - lo + 1)]
    
    pert = pert - pert.mean()
    scale = 0.02 * rng.uniform(0.5, 2.0)
    z = z - scale * pert / (np.linalg.norm(pert) + 1e-15)
    z = z + 0.01 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

Algorithm:

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _compute_active_weights(c, frac):
    """Compute uniform weights on active set of size frac*len(c)."""
    c_max = c.max()
    n_c = len(c)
    if c_max < 1e-30:
        w = np.zeros(n_c)
        w[int(np.argmax(c))] = 1.0
        return w
    n_active = max(2, int(frac * n_c))
    n_active = min(n_active, n_c)
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[min(n_active - 1, n_c - 1)]
    active = c >= threshold
    if active.sum() == 0:
        active = np.zeros(n_c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros(n_c)
    w[active] = 1.0 / active.sum()
    return w


def _gradient_given_weights(a, c, n, w):
    """Compute gradient of w^T conv(a,a) w.r.t. a, centered."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    g = 2.0 * n * (h - W)
    return g - g.mean()

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

        if target_n in explored_n or target_n == start_n:
            continue
        explored_n.add(target_n)
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_i = _score(a_interp)
        print(f"    Interpolated: {sc_i:.10f}")
        try_update(a_interp, f"interp_{target_n}")
        
        c_t = _full_convolve(a_interp, a_interp)
        a_opt, _, _ = _lbfgs_run(a_interp, float(c_t.max()) * 0.05, target_n, max_iter=2000)
        try_update(a_opt, f"n{target_n}_lbfgs")
        
        if time.monotonic() < deadline - 80:
            a_ps3, _ = _peak_shaving_momentum(a_opt, deadline - 80, max_iters=8000, base_lr=0.005)
            try_update(a_ps3, f"n{target_n}_ps")

    print(f"  Done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 7: Targeted restarts ===
    print(f"\n=== Phase 7: Targeted restarts ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 80:
        c_best = _full_convolve(best_seq, best_seq)
        
        restart_configs = [
            ("gauss_003", 0.03),
            ("gauss_008", 0.08),
            ("gauss_015", 0.15),
            ("asym_struct", None),
            ("fourier_low", None),
        ]
        
        for name, scale in restart_configs:
            if time.monotonic() > deadline - 40:
                break
            
            n = len(best_seq)
            if name.startswith("gauss"):
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + scale * rng.standard_normal(n)
                z = z - z.mean()
                p_start = _softmax(z)
            elif name == "asym_struct":
                # Add asymmetric structure
                t = np.arange(n)
                z = np.log(np.maximum(best_seq, 1e-15)) - np.log(np.maximum(best_seq, 1e-15)).mean()
                z = z + 0.03 * np.sin(2 * np.pi * 3 * t / n)
                z = z + 0.02 * np.cos(2 * np.pi * 7 * t / n + 1.0)
                p_start = _softmax(z)
            elif name == "fourier_low":
                t = np.arange(n)
                z = np.log(np.maximum(best_seq, 1e-15)) - np.log(np.maximum(best_seq, 1e-15)).mean()
                for _ in range(10):
                    freq = int(rng.integers(1, max(2, n // 15)))
                    amp = 0.004 * rng.uniform(0.5, 3.0)
                    phase = rng.uniform(0, 2.0 * np.pi)
                    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
                p_start = _softmax(z)
            else:
                continue
            
            sc_s = _score(p_start)
            print(f"  [{name}]: start={sc_s:.10f}")
            
            c_rs = _full_convolve(p_start, p_start)
            a_r, _, _ = _lbfgs_run(p_start, float(c_rs.max()) * 0.05, n, max_iter=1500)
            try_update(a_r, f"r_{name}_lbfgs")
            
            if time.monotonic() < deadline - 20:
                a_r2, _ = _peak_shaving_momentum(a_r, deadline - 20, max_iters=5000, base_lr=0.005)
                try_update(a_r2, f"r_{name}_ps")

    print(f"  Done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 8: Ultra-fine L-BFGS ===
    print(f"\n=== Phase 8: Ultra-fine L-BFGS ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 50:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        for beta_frac in [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - 30 - time.monotonic()
            if remaining < 12:
                break
            max_iter = min(10000, max(400, int(remaining / 0.006)))
            a_ultra, _, _ = _lbfgs_run(best_seq, mb * beta_frac, n, max_iter)
            try_update(a_ultra, f"ultra_{beta_frac:.0e}")
    print(f"  Done: {time.monotonic()-t_phase:.1f}s, best={best_score_val:.10f}")

    # === Phase 9: Final intensive cycling ===
    print(f"\n=== Phase 9: Final cycle ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(5):
            if time.monotonic() > deadline - 5:
                break
            
            a_f1, _ = _peak_shaving_momentum(best_seq, deadline - 5,
                                              max_iters=6000, base_lr=0.003,
                                              momentum_decay=0.93)
            try_update(a_f1, f"fc_ps_{cycle}")
            
            a_f2, _ = _peak_equalization(best_seq, deadline - 5, max_iters=4000, K=3)
            try_update(a_f2, f"fc_pe_{cycle}")
            
            if best_seq is not None and time.monotonic() < deadline - 5:
                n = len(best_seq)
                c_b2 = _full_convolve(best_seq, best_seq)
                mb2 = float(c_b2.max())
                max_iter = min(3000, max(200, int((deadline - 5 - time.monotonic()) / 0.006)))
                if max_iter > 100:
                    a_f3, _, _ = _lbfgs_run(best_seq, mb2 * 1e-8, n, max_iter)
                    try_update(a_f3, f"fc_lb_{cycle}")

    print(f"  Done: {time.monotonic()-t_phase:.1f}s")

    # === Final ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(8192, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
