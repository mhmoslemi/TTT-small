def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _variance_objective(z, n, gamma=1.0):
    """Objective that penalizes variance of convolution."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_mean = c.mean()

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    if n_cur == 1:
        return np.full(target_n, a[0])
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

        g = g - g.mean()

        # Line search
        alpha = 1.0 / (n * np.linalg.norm(g) + 1e-10)
        improved = False
        for _ in range(20):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-30:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:
            break

    return best_a, best_sc


def _basin_hop(a_current, deadline, rng, n_hops=8, reserve=30.0):
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)

    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        remaining = deadline - reserve - time.monotonic()
        if remaining < 5:
            break
        hop_time = remaining / (n_hops - h)

        # Create perturbation
        pert_type = h % 4
        if pert_type == 0:
            # Gaussian in log-space
            sigma = 0.03 + 0.02 * h
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(2 + h):
                freq = int(rng.integers(1, max(2, n // 8)))
                amp = 0.01 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 5 + h
            block_sz = max(1, n // n_blocks)
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.03, 0.03)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Mirror/flip perturbation
            a_pert = best_a.copy()
            # Swap random blocks
            for _ in range(3):
                i = int(rng.integers(0, n // 2))
                j = int(rng.integers(n // 2, n))
                a_pert[i], a_pert[j] = a_pert[j], a_pert[i]
            a_pert = a_pert / a_pert.sum()

        # Quick L-BFGS from perturbation
        c_p = _full_convolve(a_pert, a_pert)
        mc = c_p.max()
        beta = mc * 0.005
        max_iter = max(100, int(hop_time * 0.6 / max(0.001, (n / 4096.0) ** 1.3 * 0.003)))
        a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
        sc_opt = _score(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Hop {h}: improved to {best_sc:.10f}")

    return best_a, best_sc
