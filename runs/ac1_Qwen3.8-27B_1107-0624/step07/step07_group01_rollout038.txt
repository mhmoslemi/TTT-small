"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral water-filling equalization (novel frequency-domain approach)
2. KKT linear constraint regularization
3. Adaptive L-BFGS with log-sum-exp smoothing and aggressive beta tightening
4. Projected subgradient with backtracking line search
5. Active-set-aware Newton-like refinement
6. Multi-length exploration with interpolation

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smoobj_kkt(z, beta, n, kkt_lambda, H_matrix):
    """Smoothed objective + KKT linear constraint penalty."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
    # KKT penalty: ||H @ a - mean(H @ a)||^2
    Ha = H_matrix @ a
    penalty = kkt_lambda * np.sum((Ha - Ha.mean()) ** 2)
    return obj + penalty


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-20, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _lbfgs_with_kkt(a_init, beta, n, max_iter, H_matrix, kkt_lambda,
                    ftol=1e-20, gtol=1e-15):
    """L-BFGS with KKT linear constraint penalty."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj_kkt(z, beta, n, kkt_lambda, H_matrix)

    def jac(z):
        a = _softmax(z)
        grad_smo = _smograd(z, beta, n)
        # Gradient of KKT penalty
        Ha = H_matrix @ a
        Ha_centered = Ha - Ha.mean()
        Ht_Ha = H_matrix.T @ Ha_centered
        grad_kkt = kkt_lambda * 2.0 * (Ht_Ha - (Ht_Ha @ a * 0 + Ht_Ha.mean() * a) )
        # Actually: d/dz [lam * ||H@softmax(z) - mean||^2]
        # = lam * 2 * H^T (Ha - mean(Ha)) * diag(a) - correction for softmax
        # The gradient w.r.t. z is: lam * 2 * diag(a) * H^T (Ha - mean(Ha))
        # minus lam * 2 * a * sum_j a_j * (H^T(Ha-mean))_j ... 
        # For softmax, d/dz_j f(a) = a_j * (df/da_j - sum_k a_k df/da_k)
        df_da = kkt_lambda * 2.0 * H_matrix.T @ (Ha - Ha.mean())
        grad_kkt_z = a * (df_da - a @ df_da)
        return grad_smo + grad_kkt_z

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        # Fallback to plain L-BFGS
        return _lbfgs_smooth(a_init, beta, n, max_iter)


def _spectral_equalize(a, n_iter=200, alpha_init=0.1, alpha_decay=0.97):
    """
    Spectral water-filling: iteratively flatten the power spectrum
    to reduce autocorrelation peaks, projecting back to non-negative simplex.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    alpha = alpha_init
    N = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute current autocorrelation
        c = _full_convolve(a, a)
        c_max = c.max()
        c_mean = c.mean()

        # Target: flatten P so that IDFT(P) is as flat as possible
        # Water-filling: reduce peaks in P relative to the mean
        P_target = np.ones_like(P)
        P_target[0] = P[0]  # Keep DC

        # Blend current P with target
        P_new = (1.0 - alpha) * P + alpha * P_target * (P[0] / N)

        # Ensure P_new[0] = P[0] (preserve sum)
        P_new[0] = P[0]

        # Reconstruct A from new magnitudes and old phases
        A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)

        # Compute new time-domain signal
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce alpha and try again
            alpha *= alpha_decay
            if alpha < 1e-8:
                break

    return best_a


def _spectral_notch(a, active_lags, n_iter=100):
    """
    Notch out frequencies that contribute to the active autocorrelation peaks.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    N_half = N // 2 + 1

    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        c = _full_convolve(a, a)
        c_max = c.max()

        # Compute the "notch" profile: for each active lag m,
        # the frequency response is exp(-2*pi*i*m*f/N)
        # We want to reduce P at frequencies where the active lags
        # constructively interfere
        notch = np.ones(N_half)
        for m in active_lags:
            notch = notch * (1.0 - 0.3 * np.cos(2.0 * np.pi * m * np.arange(N_half) / N) ** 2)

        # Suppress frequencies where notch is small
        P_new = P * notch
        P_new[0] = P[0]

        # Normalize to preserve energy
        P_new = P_new * (P.sum() / max(P_new.sum(), 1e-30))

        A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def _subgrad_projected(a_init, deadline, k_active=5, max_steps=20000):
    """
    Projected subgradient method with backtracking line search.
    Works directly in the simplex (not log-space).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    step = 1.0 / n
    stall = 0
    stall_limit = 50

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Find active set
        idx = np.argsort(c)[::-1][:k_active]

        # Compute subgradient: average of active lag gradients
        # grad[m][j] = 2*a[m-j] for valid j
        grad = np.zeros(n)
        for m in idx:
            # For lag m, the gradient w.r.t. a[j] is 2*a[m-j] (if valid)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[m - j]
        grad /= len(idx)

        # Center the gradient (for the sum=1 constraint)
        grad_centered = grad - grad.mean()

        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(30):
            a_try = a + alpha * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if improved:
            step = min(step * 1.05, 5.0 / n)
            stall = 0
        else:
            stall += 1
            step *= 0.9
            if stall > stall_limit and step < 1e-10 / n:
                break

    return best_a, best_sc


def _adaptive_lbfgs_tight(a_start, deadline, reserve_time=60.0, max_beta_ratio=1e-8):
    """
    Aggressive adaptive L-BFGS: keep tightening beta as long as there's time
    and progress is being made.
    """
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Start with a larger beta and tighten aggressively
    beta = max_c * 0.1
    min_beta = max_c * max_beta_ratio
    progress_count = 0
    no_progress_count = 0

    while time.monotonic() < deadline - reserve_time:
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break

        # Scale max_iter based on remaining time and problem size
        scale = max(1.0, (n / 8192.0) ** 1.3)
        est_time_per_iter = 0.003 * scale
        max_iter = min(50000, max(50, int(remaining / 2.0 / est_time_per_iter)))

        if max_iter < 20:
            break

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new
            progress_count += 1
            no_progress_count = 0
            # Tighten beta
            beta = max(beta * 0.5, min_beta)
        else:
            no_progress_count += 1
            if no_progress_count >= 3:
                # Stuck: try a different beta
                beta = max(beta * 0.3, min_beta)
                no_progress_count = 0
                if beta <= min_beta * 1.01:
                    break

        if current_sc < 1.5001:
            break

    return current_a


def _active_set_newton(a_init, deadline, k_active=4, max_iter=200):
    """
    Newton-like method that tries to equalize the active autocorrelation values.
    At each step, solves a small QP to find the best adjustment.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        # Build the "correlation matrix" for the active lags
        # The gradient of c[m] w.r.t. a is: 2 * a[m-j] (shifted)
        # We want to find delta such that:
        #   c[m] + 2*(delta correlated with a at lag m) ≈ c_max for m in idx
        #   sum(delta) = 0 (to preserve normalization)
        #   a + delta >= 0

        # Linearized: c[m] ≈ c[m]_current + 2 * sum_j delta[j] * a[m-j]
        # We want c[m]_new <= c_max_target for all m, and c[m]_new ≈ c_max for m in idx

        # Use a simple approach: compute the "autocorrelation gradient" and
        # take a step that reduces the variance of the active values
        active_vals = c[idx]
        mean_active = active_vals.mean()

        # Compute gradient direction to equalize
        g = np.zeros(n)
        for m in idx:
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * a[m - j]
        g /= k_active
        g -= g.mean()

        # Line search
        alpha = 2.0 / n
        improved = False
        for _ in range(20):
            a_try = a - alpha * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-16:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.6

        if not improved:
            # Try adding noise to escape
            z = np.log(np.maximum(a, 1e-18))
            z += 0.01 * np.random.standard_normal(n)
            z -= z.mean()
            a = _softmax(z)
            if alpha < 1e-15 / n:
                break

    return best_a, best_sc


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur, endpoint=False)
    t_new = np.linspace(0, 1, target_n, endpoint=False)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _compute_kkt_matrix(a, active_lags):
    """Build the H matrix where H[j,i] = 1 iff (j+i) in active_lags."""
    n = len(a)
    active_set = set(active_lags.tolist())
    H = np.zeros((n, n), dtype=np.float64)
    for j in range(n):
        for i in range(n):
            if (j + i) in active_set:
                H[j, i] = 1.0
    return H


def _kkt_linproject(a, active_lags, n_iter=50):
    """
    Project a onto the set {a >= 0, sum=1, H@a = C*1} approximately.
    Using iterative water-filling on the linear constraint.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    active_set = set(active_lags.tolist())

    for it in range(n_iter):
        # Compute Ha (the KKT linear form)
        Ha = np.zeros(n)
        for j in range(n):
            for m in active_lags:
                j_minus_m = j - m
                if 0 <= j_minus_m < n:
                    Ha[j] += a[j_minus_m]
            # Also: a[m-j] for m in active_lags
            # Actually H[j,i] = 1 iff j+i in active_set
            # So Ha[j] = sum_i a[i] * H[j,i] = sum_{i: j+i in active} a[i]
        # Recompute properly
        Ha = np.zeros(n)
        for i in range(n):
            for m in active_lags:
                j = m - i
                if 0 <= j < n:
                    Ha[j] += a[i]

        mean_Ha = Ha.mean()
        residual = Ha - mean_Ha

        # Step to reduce residual
        alpha = 0.5 / max(np.linalg.norm(residual), 1e-15)
        # The gradient of ||Ha - mean||^2 w.r.t. a involves H^T
        # Simplified: just adjust a to reduce the spread
        # a_new[j] should be larger where Ha[j] < mean, smaller where Ha[j] > mean
        adjustment = -alpha * residual
        a_new = a + adjustment
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        if np.linalg.norm(Ha - mean_Ha) / max(np.linalg.norm(Ha), 1e-15) < 1e-10:
            break
        a = a_new

    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral methods, adaptive L-BFGS,
    KKT-guided refinement, and projected subgradient.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 4096

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print(f"=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    n_init = 4096
    candidates = []

    if parent is not None:
        candidates.append(("parent", parent))
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")

    # Structured initializations
    structured_inits = [
        ("chirp_f1", _make_chirp(n_init, freq=1.0, eps=0.3)),
        ("chirp_f2", _make_chirp(n_init, freq=2.0, eps=0.25)),
        ("chirp_f1_5", _make_chirp(n_init, freq=1.5, eps=0.2)),
        ("multi_13", _make_multiband(n_init, bands=[1, 3], eps=0.25)),
        ("multi_12", _make_multiband(n_init, bands=[1, 2], eps=0.3)),
        ("multi_25", _make_multiband(n_init, bands=[2, 5], eps=0.2)),
        ("multi_135", _make_multiband(n_init, bands=[1, 3, 5], eps=0.2)),
    ]

    for name, a_init in structured_inits:
        candidates.append((name, a_init))
        sc = _score(a_init)
        try_update(a_init, name)

    for r in range(2):
        a_r = _make_random_positive(n_init, rng)
        sc = _score(a_r)
        try_update(a_r, f"random_{r}")

    if best_seq is None:
        best_seq = structured_inits[0][1].copy()
        best_score_val = _score(best_seq)
        best_n = len(best_seq)

    print(f"  Phase 1 done. Best: {best_score_val:.10f} (n={best_n})")

    # === Phase 2: Spectral equalization on best ===
    print(f"\n=== Phase 2: Spectral equalization (t={time.monotonic()-t_start:.1f}s) ===")
    a_spec = _spectral_equalize(best_seq, n_iter=300, alpha_init=0.15, alpha_decay=0.96)
    try_update(a_spec, "spectral_eq")

    # Spectral notch on active lags
    c_b = _full_convolve(best_seq, best_seq)
    active = np.argsort(c_b)[::-1][:5]
    a_notch = _spectral_notch(best_seq, active, n_iter=150)
    try_update(a_notch, "spectral_notch")

    # === Phase 3: Main adaptive L-BFGS ===
    print(f"\n=== Phase 3: Main L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    reserve = max(600, int(0.4 * (deadline - time.monotonic())))
    a_p3 = _adaptive_lbfgs_tight(best_seq, deadline, reserve_time=reserve)
    try_update(a_p3, "main_lbfgs")

    # === Phase 4: KKT-guided refinement ===
    print(f"\n=== Phase 4: KKT refinement (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        active = np.argsort(c_b)[::-1][:4].tolist()

        # KKT linear projection
        a_kkt_lin = _kkt_linproject(best_seq, active, n_iter=100)
        try_update(a_kkt_lin, "kkt_linproj")

        # Active set Newton
        a_newton = _active_set_newton(best_seq, deadline - 300, k_active=4, max_iter=300)
        try_update(a_newton[0], "active_newton")

        # L-BFGS with KKT penalty (use a smaller H for efficiency)
        if time.monotonic() < deadline - 200 and n <= 8192:
            H_mat = _compute_kkt_matrix(best_seq, active)
            c_b2 = _full_convolve(best_seq, best_seq)
            mb = c_b2.max()
            a_kkt_lbfgs = _lbfgs_with_kkt(best_seq, mb * 0.001, n,
                                           max_iter=5000, H_matrix=H_mat,
                                           kkt_lambda=1.0)
            try_update(a_kkt_lbfgs, "kkt_lbfgs")

    # === Phase 5: Tight L-BFGS round 2 ===
    print(f"\n=== Phase 5: Tight L-BFGS R2 (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        reserve2 = max(150, int(0.15 * (deadline - time.monotonic())))
        a_p5 = _adaptive_lbfgs_tight(best_seq, deadline, reserve_time=reserve2,
                                      max_beta_ratio=1e-9)
        try_update(a_p5, "lbfgs_r2")

    # === Phase 6: Projected subgradient ===
    print(f"\n=== Phase 6: Projected subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_projected(best_seq, deadline - 150, k_active=5, max_steps=30000)
        try_update(a_sg, "sg_proj_k5")

    if time.monotonic() < deadline - 100:
        a_sg2, sc_sg2 = _subgrad_projected(best_seq, deadline - 100, k_active=3, max_steps=20000)
        try_update(a_sg2, "sg_proj_k3")

    # === Phase 7: Basin hopping ===
    print(f"\n=== Phase 7: Basin hopping (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        n = len(best_seq)
        best_hop_a = best_seq.copy()
        best_hop_sc = best_score_val

        for h in range(8):
            if time.monotonic() > deadline - 60:
                break
            hop_time_left = min(60.0, (deadline - 60 - time.monotonic()) / max(1, 8 - h))

            # Perturb
            pert_type = h % 3
            if pert_type == 0:
                z = np.log(np.maximum(best_hop_a, 1e-18))
                z += (0.03 + 0.02 * h) * rng.standard_normal(n)
                z -= z.mean()
                a_pert = _softmax(z)
            elif pert_type == 1:
                z = np.log(np.maximum(best_hop_a, 1e-18))
                for _ in range(3):
                    freq = int(rng.integers(1, max(2, n // 32)))
                    amp = 0.03 * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2.0 * np.pi)
                    t = np.arange(n)
                    z += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
                z -= z.mean()
                a_pert = _softmax(z)
            else:
                # Block perturbation
                z = np.log(np.maximum(best_hop_a, 1e-18))
                n_blocks = 4 + h
                block_sz = max(1, n // n_blocks)
                for _ in range(n_blocks):
                    start = int(rng.integers(0, max(1, n - block_sz)))
                    z[start:start+block_sz] += rng.uniform(-0.04, 0.04)
                z -= z.mean()
                a_pert = _softmax(z)

            # Quick L-BFGS from perturbation
            c_p = _full_convolve(a_pert, a_pert)
            mc = c_p.max()
            beta = mc * 0.005
            max_it = min(5000, max(200, int(hop_time_left / 0.01)))
            a_opt = _lbfgs_smooth(a_pert, beta, n, max_it)
            sc_opt = _score(a_opt)

            if sc_opt < best_hop_sc:
                best_hop_sc = sc_opt
                best_hop_a = a_opt.copy()

        try_update(best_hop_a, "basin_hop")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        lengths_to_try = [8192, 16384, 2048]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 45:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")
            if sc_interp < best_score_val:
                best_seq = a_interp.copy()
                best_score_val = sc_interp
                best_n = target_n

            # Quick optimization at new length
            if time.monotonic() < deadline - 30:
                a_opt = _adaptive_lbfgs_tight(a_interp, deadline, reserve_time=30.0,
                                               max_beta_ratio=1e-7)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 9: Final aggressive polishing ===
    print(f"\n=== Phase 9: Final polishing (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = c_b.max()

        # Very tight L-BFGS
        for beta_ratio in [1e-6, 1e-7, 1e-8]:
            if time.monotonic() > deadline - 5:
                break
            beta = mb * beta_ratio
            max_it = min(10000, max(200, int((deadline - 5 - time.monotonic()) / 0.004)))
            if max_it < 50:
                break
            a_f = _lbfgs_smooth(best_seq, beta, n, max_it)
            try_update(a_f, f"final_beta_{beta_ratio}")

        # KKT projection + L-BFGS
        if time.monotonic() < deadline - 5:
            active = np.argsort(_full_convolve(best_seq, best_seq))[::-1][:3].tolist()
            a_kkt = _kkt_linproject(best_seq, active, n_iter=50)
            try_update(a_kkt, "final_kkt")
            if time.monotonic() < deadline - 5:
                a_f2 = _lbfgs_smooth(a_kkt, mb * 1e-7, n,
                                      min(5000, max(100, int((deadline-5-time.monotonic())/0.004))))
                try_update(a_f2, "final_kkt_lbfgs")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_s, _ = _subgrad_projected(best_seq, deadline - 2, k_active=3, max_steps=10000)
            try_update(a_s, "final_sg")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
