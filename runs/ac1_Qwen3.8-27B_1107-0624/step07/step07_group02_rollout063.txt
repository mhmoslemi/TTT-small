Combines adaptive log-sum-exp L-BFGS, momentum-based active-set gradient

and adaptive multi-level active set scheduling with momentum.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    rho_star = int(np.max(rho[cond])) if np.any(cond) else 1
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.7, 0.5, 0.3, 0.2, 0.1,
                                  0.05, 0.025, 0.012, 0.006,
                                  0.003, 0.0015, 0.0007, 0.0003, 0.00015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 15))))

        class _Prob:
            def __call__(self, z):
                return _smoobj(z, beta, n)
            def grad(self, z):
                return _smograd(z, beta, n)

        prob = _Prob()
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}, iter={i}")
        except Exception:
            pass

    return current_a


def _active_set_momentum(a_init, deadline, min_time_remaining=20.0,
                         tol_start=0.01, tol_end=0.00003,
                         lr_init=0.04, momentum=0.5, verbose=False):
    """Active-set gradient descent with momentum and adaptive tolerance."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)

    lr = lr_init
    tol_frac = tol_start
    v = np.zeros(n)  # momentum
    no_improve = 0
    max_stall = 300
    stall_count = 0
    iteration = 0

    while time.monotonic() < deadline - min_time_remaining:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        # Update momentum
        v = momentum * v + (1.0 - momentum) * grad_t

        alpha = min(lr, 2.0 / (np.linalg.norm(v) + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * v
            a_try = _simplex_project(a_try)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 25 == 0:
                tol_frac = max(tol_end, tol_frac * 0.7)
                lr *= 0.65
                v *= 0.5
            if no_improve >= max_stall:
                a = best_a.copy()
                v = np.zeros(n)
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.4
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a), _score_normalized(best_a)


def _freq_domain_peak_reduction(a_init, deadline, min_time_remaining=20.0,
                                 verbose=False):
    """Frequency-domain iterative peak reduction with backtracking."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)

    w = np.arange(N // 2 + 1, dtype=np.float64)
    alpha = 0.05
    no_improve = 0
    max_stall = 200

    while time.monotonic() < deadline - min_time_remaining:
        A = np.fft.rfft(a, N)
        C = np.abs(A) ** 2
        c = np.fft.irfft(C, N)[:2 * n - 1]
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        # Find active peaks (multiple levels)
        tol1 = 0.005 * c_max
        tol2 = 0.02 * c_max
        active_strong = c >= (c_max - tol1)
        active_weak = c >= (c_max - tol2)

        # Compute combined direction
        direction = np.zeros(N // 2 + 1, dtype=np.float64)
        n_strong = int(active_strong.sum())
        n_weak = int(active_weak.sum())
        if n_strong > 0:
            for k in np.where(active_strong)[0]:
                direction += np.cos(2.0 * np.pi * w * k / N)
            direction /= n_strong
        if n_weak > n_strong:
            for k in np.where(active_weak)[0]:
                if not active_strong[k]:
                    direction += 0.5 * np.cos(2.0 * np.pi * w * k / N)
            direction /= (n_weak - n_strong)

        # Update C with backtracking
        improved = False
        for _ in range(30):
            C_new = C * (1.0 - alpha * direction)
            C_new = np.maximum(C_new, 0.0)
            # Ensure C_new[0] is reasonable (DC component)
            if C_new[0] < 1e-10:
                C_new[0] = 1e-10
            A_new = np.sqrt(C_new)
            a_raw = np.fft.irfft(A_new, N)[:n]
            a_try = _simplex_project(a_raw)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                alpha = max(0.001, alpha * 0.7)
            if no_improve >= max_stall:
                break
        else:
            alpha = min(0.1, alpha * 1.05)

    return best_a, _score(best_a)


def _cepstrum_fixed_point(a_init, deadline, min_time_remaining=20.0):
    """Cepstrum-domain fixed-point iteration."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_sc = _score_normalized(a)

    # Initialize target spectrum
    c = _full_convolve(a, a)
    C = np.fft.rfft(c, N)
    C = np.maximum(C, 0.0)
    beta = 0.3
    no_improve = 0

    while time.monotonic() < deadline - min_time_remaining:
        A = np.sqrt(C)
        a_raw = np.fft.irfft(A, N)[:n]
        a_new = _simplex_project(a_raw)
        c_new = _full_convolve(a_new, a_new)
        C_new = np.fft.rfft(c_new, N)
        C_new = np.maximum(C_new, 0.0)

        # Blend
        C = (1.0 - beta) * C + beta * C_new

        sc_new = 2.0 * n * float(c_new.max())
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        beta = max(0.05, beta * 0.995)
        if no_improve > 500:
            break

    a = _simplex_project(a)
    return best_a, _score(best_a)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.02
        return a / a.sum()
    elif kind == "edge":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 16))
        amp = 0.05 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try zeroing out the smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining L-BFGS, momentum active-set GD,
    frequency-domain peak reduction, cepstrum fixed-point,
    multi-n exploration, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main optimization ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 600, t_start + 400)
    a_main = _run_lbfgs(parent, p1_deadline, min_time_remaining=100.0, verbose=True)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set GD with momentum (coarse) ===
    print(f"\n[P2] Active-set momentum coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        p2_deadline = min(deadline - 400, time.monotonic() + 200)
        a_pol, sc_pol, _ = _active_set_momentum(best_seq, p2_deadline,
                                                 min_time_remaining=30,
                                                 tol_start=0.008, tol_end=0.0003,
                                                 lr_init=0.04, momentum=0.5)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} [active_coarse] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Frequency-domain peak reduction ===
    print(f"\n[P3] Frequency-domain peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        p3_deadline = min(deadline - 300, time.monotonic() + 180)
        a_freq, sc_freq = _freq_domain_peak_reduction(best_seq, p3_deadline,
                                                       min_time_remaining=30)
        if sc_freq < best_score:
            best_seq = a_freq.copy()
            best_score = sc_freq
            print(f"  NEW BEST: {best_score:.10f} [freq_peak] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Cepstrum fixed-point ===
    print(f"\n[P4] Cepstrum fixed-point (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p4_deadline = min(deadline - 200, time.monotonic() + 120)
        a_cep, sc_cep = _cepstrum_fixed_point(best_seq, p4_deadline,
                                               min_time_remaining=20)
        if sc_cep < best_score:
            best_seq = a_cep.copy()
            best_score = sc_cep
            print(f"  NEW BEST: {best_score:.10f} [cepstrum] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS fine (after all the above) ===
    print(f"\n[P5] L-BFGS fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        p5_deadline = min(deadline - 200, time.monotonic() + 150)
        a_fine = _run_lbfgs(best_seq, p5_deadline, min_time_remaining=50.0)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [lbfgs_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Active-set GD with momentum (fine) ===
    print(f"\n[P6] Active-set momentum fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        p6_deadline = min(deadline - 150, time.monotonic() + 120)
        a_pol2, sc_pol2, _ = _active_set_momentum(best_seq, p6_deadline,
                                                   min_time_remaining=20,
                                                   tol_start=0.002, tol_end=0.00005,
                                                   lr_init=0.015, momentum=0.3)
        if sc_pol2 < best_score:
            best_seq = a_pol2.copy()
            best_score = sc_pol2
            print(f"  NEW BEST: {best_score:.10f} [active_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Shrinkage ===
    print(f"\n[P7] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 120)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 80:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=40.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - len(best_seq)))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 30:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.01 and time.monotonic() < deadline - 25:
            a_opt2, sc_opt2, _ = _active_set_momentum(a_opt, min(deadline - 20, time.monotonic() + 25),
                                                       min_time_remaining=5,
                                                       tol_start=0.003, tol_end=0.0001,
                                                       lr_init=0.012, momentum=0.4)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Diverse restarts ===
    print(f"\n[P9] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        restart_starts = [
            ("pert3", _perturb(best_seq, 0.03, rng)),
            ("pert6", _perturb(best_seq, 0.06, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("edge", _make_profile(n_best, "edge")),
            ("bimo", _make_profile(n_best, "bimodal")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 15:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin, _ = _active_set_momentum(best_seq, deadline - 2,
                                                 min_time_remaining=2,
                                                 tol_start=0.0005, tol_end=0.00001,
                                                 lr_init=0.004, momentum=0.2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_active] t={time.monotonic()-t_start:.1f}s")

        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.0001
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()

            class _FinProb:
                def __call__(self, z):
                    return _smoobj(z, beta_fin, n)
                def grad(self, z):
                    return _smograd(z, beta_fin, n)

            fin_prob = _FinProb()
            max_iter = min(3000, max(50, int(remaining / 0.05)))
            try:
                result = minimize(fin_prob, z, jac=fin_prob.grad,
                                  method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 50})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
