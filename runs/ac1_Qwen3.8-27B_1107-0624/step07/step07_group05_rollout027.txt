"""
Improved optimizer using multi-scale initialization, aggressive L-BFGS with 
extreme beta continuation, active-set engineering, and pressure equalization.
Key improvements over previous:
1. Multi-scale: optimize at small n, upsample, continue at large n
2. Active set engineering: deliberately create multiple active peaks
3. Extreme beta continuation (down to 1e-8)
4. Hybrid L-BFGS / pressure-equalization cycling
5. Non-power-of-2 n exploration

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk(a, c, n, k=5):
    """Gradient using top-k convolution peaks."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    # Select top-k peaks
    top_idx = np.argsort(c)[-k:]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _pressure_equalize_v2(a, deadline, n_iters=300, gamma=0.3, tol_frac=0.005):
    """
    Improved pressure equalization with adaptive tolerance and self-interaction correction.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 2:
            # Expand: use top 3
            idx = np.argsort(c)[-3:]
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 3

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]

        # Self-interaction correction: if 2i is in active set, subtract a_i
        # Because (a*a)_{2i} includes a_i^2 which is the "self" term
        self_mask = np.zeros(2 * n - 1, dtype=bool)
        self_idx = 2 * np.arange(n)
        self_mask[self_idx] = active[self_idx]
        p = p - self_mask * a
        p = np.maximum(p, 1e-30)

        # Water-filling update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Check improvement
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

    current_sc = _score(current_a)

def _active_set_engineer(a, n_peaks=5):
    """
    Try to create n_peaks active peaks in the convolution at the same level.
    This gives more freedom for pressure equalization.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Find current peaks
    peak_threshold = c_max * 0.95
    current_peaks = np.where(c >= peak_threshold)[0]
    
    if len(current_peaks) >= n_peaks:
        return a
    
    # Try to raise some lower peaks to match the max
    # Strategy: identify k values where we can increase (a*a)_k by increasing a_i
    # without increasing (a*a)_{n-1} too much
    candidates = np.where(c < peak_threshold)[0]
    # Sort by how close they are to the max
    candidates = candidates[np.argsort(c[candidates])[::-1][:20]]
    
    best_a = a.copy()
    best_score = _score(a)
    
    for k_try in candidates[:5]:
        # Find which a_i to increase to raise (a*a)_{k_try}
        # (a*a)_{k_try} = sum_i a_i * a_{k_try - i}
        # Increasing a_i by delta increases (a*a)_{k_try} by delta * a_{k_try - i}
        # and (a*a)_{n-1} by a_i * (something)
        
        # Simple approach: increase a at positions that contribute to k_try
        # but not to n-1
        contrib_to_target = np.zeros(n)
        for i in range(n):
            j = k_try - i
            if 0 <= j < n:
                contrib_to_target[i] = a[j]
        
        contrib_to_center = np.zeros(n)
        for i in range(n):
            j = (n - 1) - i
            if 0 <= j < n:
                contrib_to_center[i] = a[j]
        
        # Find positions that help target but hurt center least
        ratio = contrib_to_target / (contrib_to_center + 1e-30)
        top_positions = np.argsort(ratio)[::-1][:10]
        
        # Increase those positions slightly
        a_try = a.copy()
        for pos in top_positions[:5]:
            a_try[pos] *= 1.05
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
    
    return best_a

def _subgrad_descent(a_init, deadline, n_steps=5000, lr=0.02):
    """Momentum-based subgradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros_like(z)  # momentum
    beta = 0.9

    best_sc = _score(a_init)
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        
        # Use top-5 gradient
        k = min(5, len(c))
        grad = _gradient_topk(a, c, n, k=k)
        
        # Adaptive learning rate
        lr_i = lr / (1.0 + 0.1 * i / 1000)
        
        v = beta * v - lr_i * grad
        z = z + v
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

                best_sc = sc
                best_a = a_check.copy()

        best_a = a_final.copy()

def _mirror_descent(a_init, deadline, n_iters=3000, lr=0.05):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Uses the subgradient of 2n*max(conv) with respect to a.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Subgradient: d(2n*max_c)/da_i = 2n * 2 * sum_{k: c_k = max} a_{k-i} / sum_a^2
        # Simplified: just use the active set gradient
        active = (c >= c_max * 0.999)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        
        a_rev = a[::-1]
        h = _full_convolve(active.astype(float), a_rev)
        h = h[n - 1:2 * n - 1]

        lr_i = lr * (1.0 - 0.9 * i / n_iters)
        a = a * np.exp(-lr_i * 4.0 * h)

        a = a / a.sum()
        
        if i % 50 == 0:

            if sc < best_sc:

                best_a = a.copy()

def _make_profile(n, kind="sin_taper"):

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):

            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 3
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        # Self-interaction correction
        self_idx = 2 * np.arange(n)
        self_mask = active[self_idx].copy()
        p = p - self_mask * a
        p = np.maximum(p, 1e-30)
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs_extreme(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    n_top = min(5, len(c_sorted))
    gap = max_c - c_sorted[n_top - 1] if n_top > 1 else max_c * 0.01
    base_beta = max(gap, max_c * 0.005)
    # Extreme schedule: 32 geometric steps
    betas = base_beta / (2.0 ** np.arange(32))
    t_start_local = time.monotonic()
    for i, beta in enumerate(betas):
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        n_ratio = max(1.0, n / 2048.0)
        est_per_iter = 0.004 * n_ratio
        max_iter = min(30000, max(200, int(remaining / (est_per_iter * 8))))
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 200, 'maxls': 500}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 4 == 0:
                print(f"    L-BFGS beta={beta:.2e} (i={i}), score={current_sc:.10f}, "
                      f"elapsed={time.monotonic()-t_start_local:.0f}s")
        except Exception:
            pass
    return current_a, current_sc


def _subgrad_momentum(a_init, deadline, n_steps=5000, lr=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros_like(z)
    beta_m = 0.85
    best_a = a_init.copy()
    best_sc = _score(a_init)
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        k = min(7, len(c))
        grad = _gradient_topk(a, c, n, k=k)
        lr_i = lr / (1.0 + i / 2000.0)
        v = beta_m * v - lr_i * grad
        z = z + v
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _mirror_descent(a_init, deadline, n_iters=3000, lr=0.04):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        a = a / a.sum()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = (c >= c_max * 0.999)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        a_rev = a[::-1]
        h = _full_convolve(active.astype(float), a_rev)
        h = h[n - 1:2 * n - 1]
        lr_i = lr * (1.0 - 0.8 * i / max(1, n_iters))
        a = a * np.exp(-lr_i * 4.0 * h)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        if i % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="sin_taper"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.1
        return a / a.sum()
    elif kind == "bipolar":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.4 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(4 * np.pi * t) + 0.05 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to(a_src, n_target):
    x_old = np.linspace(0, 1, len(a_src))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_src)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _cyclic_optimize(a_init, deadline, min_remaining=30, n_cycles=3):
    """Alternate L-BFGS and pressure equalization for better convergence."""
    current_a = a_init.copy()
    current_sc = _score(current_a)
    for cyc in range(n_cycles):
        if time.monotonic() + min_remaining > deadline:
            break
        # L-BFGS step
        a1, sc1 = _run_lbfgs_extreme(
            current_a, deadline, min_time_remaining=min_remaining
        )
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1
        # Pressure equalization step
        if time.monotonic() + min_remaining > deadline:
            break
        a2, sc2 = _pressure_equalize_v2(
            current_a, min(deadline - min_remaining, time.monotonic() + 40),
            n_iters=150, gamma=0.25, tol_frac=0.002
        )
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2
    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses cyclic L-BFGS / pressure-equalization,
    active set engineering, multi-n exploration, and final polishing.
    """
    global _t0
    _t0 = time.monotonic()
    deadline = _t0 + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  ** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-_t0:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent)
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "sin_taper")
        sc_parent = _score(parent)
        print(f"No parent, using sin_taper n={n_parent}, score={sc_parent:.10f}")
        _update(parent, sc_parent)

    # --- Phase 1: Cyclic optimization on parent ---
    print(f"\n[Phase 1] Cyclic opt on parent n={n_parent} (remaining={deadline-time.monotonic():.0f}s)")
    a_cyc, sc_cyc = _cyclic_optimize(parent, deadline, min_remaining=400, n_cycles=3)
    _update(a_cyc, sc_cyc)
    if sc_cyc < best_score:
        parent = a_cyc

    # --- Phase 2: Subgradient momentum ---
    print(f"\n[Phase 2] Subgradient momentum (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() < deadline - 200:
        a_sg, sc_sg = _subgrad_momentum(
            parent, min(deadline - 100, time.monotonic() + 120),
            n_steps=4000, lr=0.012
        )
        _update(a_sg, sc_sg)

    # --- Phase 3: Mirror descent ---
    print(f"\n[Phase 3] Mirror descent (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() < deadline - 150:
        a_md, sc_md = _mirror_descent(
            parent, min(deadline - 80, time.monotonic() + 80),
            n_iters=3000, lr=0.03
        )
        _update(a_md, sc_md)
        if sc_md < best_score:
            parent = a_md

    # --- Phase 4: Second cyclic pass (finer) ---
    print(f"\n[Phase 4] Second cyclic pass (t={time.monotonic()-_t0:.0f}s)")
    if time.monotonic() < deadline - 120:
        a_c2, sc_c2 = _cyclic_optimize(parent, deadline, min_remaining=80, n_cycles=2)
        _update(a_c2, sc_c2)
        if sc_c2 < best_score:
            parent = a_c2

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-_t0:.0f}s)")
    n_candidates = [4096, 5120, 6144, 7168, 8192, 9216, 10240, 11264, 12288]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates[:5]:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - 80 - time.monotonic()
        if remaining < 25:
            break

        a_init = _interpolate_to(parent, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}")

        # Quick L-BFGS
        a_opt, sc_opt = _run_lbfgs_extreme(a_init, deadline, min_time_remaining=40)
        _update(a_opt, sc_opt)

        # Pressure equalize
        if time.monotonic() < deadline - 40:
            a_pe, sc_pe = _pressure_equalize_v2(
                a_opt, min(deadline - 30, time.monotonic() + 25),
                n_iters=100, gamma=0.2, tol_frac=0.002
            )
            _update(a_pe, sc_pe)

    # --- Phase 6: Diverse restarts ---
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-_t0:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        profiles = ["gaussian", "bipolar", "cos_mod"]
        for pi, prof in enumerate(profiles):
            if time.monotonic() > deadline - 30:
                break
            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            # Small random perturbation
            noise = rng.standard_normal(n_best) * 0.01
            a_blend = a_blend + noise
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_r, sc_r = _cyclic_optimize(a_blend, deadline, min_remaining=20, n_cycles=1)
            _update(a_r, sc_r)

    # --- Phase 7: Final extreme polish ---
    print(f"\n[Phase 7] Final extreme polish (t={time.monotonic()-_t0:.0f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _run_lbfgs_extreme(best_seq, deadline, min_time_remaining=3)
        _update(a_fin, sc_fin)
        if time.monotonic() < deadline - 2:
            a_pe3, sc_pe3 = _pressure_equalize_v2(
                best_seq, deadline - 1, n_iters=80, gamma=0.15, tol_frac=0.001
            )
            _update(a_pe3, sc_pe3)

    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-_t0:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
