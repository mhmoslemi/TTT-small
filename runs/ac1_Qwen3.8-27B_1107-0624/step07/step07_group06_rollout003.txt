"""
Spectral-active-set hybrid optimizer for minimizing peak autocorrelation of
non-negative sequences. Combines aggressive log-sum-exp L-BFGS continuation,
Newton active-set with adaptive tolerance, direct peak-shaving, spectral
phase optimization, diverse restarts, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_vec(v, active, n):
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec_fn, b, n, max_iter=300, tol=1e-12):
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec_fn(p)
        pH = float(np.dot(p, Hp))
        if pH < 1e-30:
            # Try steepest descent step
            alpha = rs / max(pH, 1e-30)
            x = x + alpha * p
            r = r - alpha * Hp
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=200, adaptive_tol=True):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    tol_frac = 0.005

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if adaptive_tol:
            tol_frac = max(0.001, 0.005 * (0.95 ** it))
        tol = max(tol_frac * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        grad = _hess_vec(a, active, n)
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-14:
            break

        mu = 1e-7
        def hess_fn(v):
            return _hess_vec(v, active, n) + mu * v

        dx = _cg_solve(hess_fn, -grad_t, n, max_iter=200, tol=1e-10)
        dx = dx - np.mean(dx)

        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        improved = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:

            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)

        a = a_new

def _peak_shaving(a_init, deadline, n_iter=500):
    """Iteratively shave the peak of the autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.005

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Use weighted active set
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        step = lr / (1.0 + 0.005 * it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.85

    return best_a


def _spectral_flat(a_init, deadline, n_iter=300):
    """Try to flatten the power spectrum via iterative phase adjustment."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2

        # Compute target: weighted flat spectrum
        # DC is fixed at 1, so target for other bins
        non_dc = np.concatenate([power[1:N//2], power[N//2+1:]]) if N > 2 else np.array([0.0])
        if len(non_dc) > 0 and non_dc.max() > 0:
            target = np.mean(non_dc)
        else:
            target = power[0] / N

        # Rescale non-DC bins toward target (soft)
        mix = 0.1 + 0.3 * (it / n_iter)  # gradually increase mixing
        A_new = A.copy()
        for k in range(1, N):
            if k == N // 2 and N % 2 == 0:
                continue
            mag = np.abs(A_new[k])
            if mag > 1e-15:
                new_mag = (1 - mix) * mag + mix * np.sqrt(target)
                phase = np.angle(A_new[k])
                A_new[k] = new_mag * np.exp(1j * phase)

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "kaiser":
        # Kaiser window
        delta = 60  # dB sidelobe
        if delta > 50:
            A = 0.1102 * (delta - 8.7)
        elif delta > 21:
            A = 0.5842 * (delta - 21) ** 0.4 + 0.07886 * (delta - 21)
        else:
            A = 0.0
        alpha = np.arccosh(10 ** (delta / 20.0)) if delta > 0 else 0
        t = np.arange(n) - (n - 1) / 2.0
        if n > 1:
            t = t / ((n - 1) / 2.0)
        else:
            t = np.zeros(n)
        if A > 0:
            a = np.i0(alpha * np.sqrt(np.maximum(1 - t**2, 0))) / np.i0(alpha)
        else:
            a = np.ones(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()


def _perturb_random(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        amp = 0.05 * rng.uniform(0.3, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_swap(a, rng, n_swaps=10):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // 50)
    for _ in range(n_swaps):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < 2 * block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Aggressive L-BFGS from parent ===
    print(f"\n[Phase 1] Aggressive L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 800:
        a_main = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=800)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 2: Newton active-set ===
    print(f"\n[Phase 2] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        n_dl = min(deadline - 600, time.monotonic() + 300)
        a_newton, sc_newton = _newton_active_set(best_seq, n_dl, max_iter=300)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 3: Peak shaving ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        ps_dl = min(deadline - 500, time.monotonic() + 150)
        a_ps = _peak_shaving(best_seq, ps_dl, n_iter=1000)
        update_best(a_ps, _score(a_ps), "peak_shave")
        if time.monotonic() < deadline - 400:
            a_ps2 = _run_lbfgs_aggressive(a_ps, deadline, min_time_remaining=400)
            update_best(a_ps2, _score(a_ps2), "peak_shave_lbfgs")

    # === Phase 4: Spectral flattening ===
    print(f"\n[Phase 4] Spectral flattening (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        sf_dl = min(deadline - 400, time.monotonic() + 120)
        a_sf = _spectral_flat(best_seq, sf_dl, n_iter=500)
        update_best(a_sf, _score(a_sf), "spectral")
        if _score(a_sf) < best_score * 1.005 and time.monotonic() < deadline - 350:
            a_sf2 = _newton_active_set(a_sf, min(deadline - 350, time.monotonic() + 100), max_iter=100)
            update_best(a_sf2, _score(a_sf2), "spectral_newton")

    # === Phase 5: Second L-BFGS pass (refined) ===
    print(f"\n[Phase 5] Second L-BFGS pass (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_pass2 = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=350)
        update_best(a_pass2, _score(a_pass2), "lbfgs_pass2")

    # === Phase 6: Newton refinement ===
    print(f"\n[Phase 6] Newton refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        ndl = min(deadline - 300, time.monotonic() + 200)
        a_nr, sc_nr = _newton_active_set(best_seq, ndl, max_iter=200, adaptive_tol=True)
        update_best(a_nr, sc_nr, "newton_refine")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[Phase 7] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates and n_parent >= 2048:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "kaiser")

        remaining = deadline - 100 - time.monotonic()
        if remaining < 30:
            break
        max_t = min(remaining * 0.4, 120.0)
        if max_t < 15:
            break
        sub_deadline = time.monotonic() + max_t
        a_opt = _run_lbfgs_aggressive(a_init, sub_deadline + 1, min_time_remaining=10)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score * 1.01 and time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _newton_active_set(a_opt, min(deadline - 60, time.monotonic() + 60), max_iter=80)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb_random(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=8)
            elif i == 2:
                p_start = _perturb_swap(best_seq, rng, n_swaps=15)
            elif i == 3:
                p_start = _perturb_random(best_seq, 0.05, rng)
            else:
                p_start = _perturb_fourier(_perturb_random(best_seq, 0.01, rng), rng, n_modes=4)

            remaining = deadline - 20 - time.monotonic()
            if remaining < 10:
                break
            sub_dl = time.monotonic() + min(remaining * 0.5, 40.0)
            a_r = _run_lbfgs_aggressive(p_start, sub_dl + 1, min_time_remaining=5)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

    # === Phase 9: Final combined polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Quick Newton
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _newton_active_set(best_seq, max(deadline - 3, time.monotonic()), max_iter=100)
            update_best(a_fin, sc_fin, "final_newton")

        # Peak shave
        if time.monotonic() < deadline - 2:
            a_fin2 = _peak_shaving(best_seq, deadline - 2, n_iter=500)
            update_best(a_fin2, _score(a_fin2), "final_shave")

        # Very low beta L-BFGS
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = max(mb * 0.000001, 1e-10)
            remaining = deadline - 2 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.005)))

                def fun_final(z_local):
                    return _smoobj(z_local, beta_final, n)

                def jac_final(z_local):
                    return _smograd(z_local, beta_final, n)

                try:
                    result = minimize(
                        fun_final, z, jac=jac_final, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                                 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_ultra")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
