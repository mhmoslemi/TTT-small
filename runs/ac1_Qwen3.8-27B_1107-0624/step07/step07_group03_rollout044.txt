Uses multi-scale active-set Newton, palindromic averaging, diverse n exploration,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _palindromic_avg(a):
    return 0.5 * (a + a[::-1])


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):

    c_max = float(c.max())

    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_matvec(v, A_vec, m, mu, n):
    return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v


def _cg_solve(rhs, A_vec, m, mu, n, max_cg=300):
    x = np.zeros_like(rhs)
    r = rhs - _hess_matvec(x, A_vec, m, mu, n)
    p = r.copy()
    rs = np.dot(r, r)
    if rs < 1e-30:
        return x
    for cg_it in range(max_cg):
        Ap = _hess_matvec(p, A_vec, m, mu, n)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-13 * (1 + np.linalg.norm(rhs)):
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    return x

            break

                    accepted = True
                    best_step_found = True

            best_a = a.copy()

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation and palindromic projection."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.004 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time_per_iter * 6))))

        try:
            result = minimize(
                _lsobj_wrapper,
                z,
                jac=_lsgrad_wrapper,
                args=(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-17,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 120,
                }
            )
            a_new = _softmax(result.x)
            a_new = _palindromic_avg(a_new)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            elif i >= 5:
                break
        except Exception:
            continue

    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / 0.08)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_to_n(a, n_new):

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()

def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements."""

        n_zero = max(1, int(n * frac))

        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()

        sc_try = _score(a_try)

    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                parent = _palindromic_avg(parent)
                parent = parent / parent.sum()
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # Phase 1: Main L-BFGS from parent
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    a_main = _palindromic_avg(a_main)
    a_main = a_main / a_main.sum()
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Multi-tolerance active-set Newton
    print(f"\n[Phase 2] Active-set Newton multi-tol (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        for tol in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.monotonic() > deadline - 500:
                break
            as_deadline = min(deadline - 500, time.monotonic() + 100)
            a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=200, tol_active=tol)
            update_best(a_as, sc_as, f"as_newton_tol{tol}")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=5000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_polish_1")

    # Phase 3: Shrinkage
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 350)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=250.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # Phase 4: Multi-n search with diverse profiles
    print(f"\n[Phase 4] Multi-n diverse search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 6144, 7168, 8192, 9216, 10240, 12288]
    profiles = ["uniform", "tapered", "cos", "gaussian_center"]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue

        if parent is not None and n_parent != n_try:
            a_init = _interpolate_to_n(best_seq if best_seq is not None else parent, n_try)
        else:
            a_init = _make_profile(n_try, profiles[rng.integers(0, len(profiles))])

        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=100.0)
        a_opt = _palindromic_avg(a_opt)
        a_opt = a_opt / a_opt.sum()
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 80:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 60, time.monotonic() + 60),
                                                  max_iter=150, tol_active=0.002)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_specs = [
            ("perturb_0.02", 0.02),
            ("perturb_0.05", 0.05),
            ("perturb_0.10", 0.10),
            ("perturb_0.20", 0.20),
        ]
        for label, sigma in restart_specs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0)
            a_r = _palindromic_avg(a_r)
            a_r = a_r / a_r.sum()
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

        # Fourier perturbation
        if time.monotonic() > 0 and time.monotonic() < deadline - 20:
            p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            a_r = _palindromic_avg(a_r)
            a_r = a_r / a_r.sum()
            sc_r = _score(a_r)
            update_best(a_r, sc_r, "restart_fourier")

    # Phase 6: Final polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active-set Newton
        as_deadline_final = min(deadline - 3, time.monotonic() + 80)
        a_fin, sc_fin = _active_set_newton(best_seq, as_deadline_final,
                                            max_iter=300, tol_active=0.0003)
        update_best(a_fin, sc_fin, "final_as_tight")

        # Second round even tighter
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_newton(best_seq, min(deadline - 2, time.monotonic() + 40),
                                                  max_iter=150, tol_active=0.0001)
            update_best(a_fin2, sc_fin2, "final_as_ultra")

        # Final subgradient with tiny lr
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 2,
                                               n_max_steps=2000, lr_init=0.003)
            update_best(a_fin3, sc_fin3, "final_sg")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 4:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj_wrapper,
                        z,
                        jac=_lsgrad_wrapper,
                        args=(beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_fin4 = _softmax(result.x)
                    a_fin4 = _palindromic_avg(a_fin4)
                    a_fin4 = a_fin4 / a_fin4.sum()
                    update_best(a_fin4, _score(a_fin4), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
