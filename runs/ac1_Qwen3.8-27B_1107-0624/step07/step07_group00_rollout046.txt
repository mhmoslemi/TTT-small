        z = np.clip(z, -50, 50)
    
    a_final = np.exp(z)
    a_final = a_final / a_final.sum()
    c_final = _fft_convolve(a_final, a_final)
    if float(c_final.max()) < best_cmax:
        best_cmax = float(c_final.max())
        best_a = a_final.copy()
    
    return best_a, 2.0 * n * best_cmax

def _optimize_subgradient(a_init, deadline, max_iter=50000, top_k=3):
    """
    Projected subgradient with Polyak step size.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    c_lower = 1.0 / (2 * n - 1)
    
    best_a = a.copy()
    best_cmax = float(_fft_convolve(a, a).max())
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # Multi-lag gradient
        g = _compute_gradient_multi_lag(a, c, top_k=top_k)
        
        # Polyak step
        gap = c_max - c_lower
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-30:
            break
        eta = gap / g_norm_sq
        eta = min(eta, 0.3)
        
        a_new = _project_simplex(a - eta * g)
        a = a_new
    
    return best_a, 2.0 * n * best_cmax

def _coord_descent(a_init, deadline, max_rounds=100, top_lags=5):
    """
    Coordinate descent: for each coordinate j, find the best value
    that reduces the max autocorrelation.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_cmax = float(_fft_convolve(a, a).max())
    
    for round in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        if c_max < best_cmax:
            best_cmax = c_max
        
        # Find active lags (those close to max)
        active_threshold = c_max * 0.95
        active_lags = np.where(c >= active_threshold)[0]
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try different values for a[j]
            best_val = old_val
            best_local = c_max
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.0]:
                new_val = old_val * frac if old_val > 0 else 0.0
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_val
                    break
                a = a / s
                c_new = _fft_convolve(a, a)
                # Only check the active lags for speed
                if len(active_lags) < 2*n:
                    local_max = float(c_new[active_lags].max())
                else:
                    local_max = float(c_new.max())
                if local_max < best_local:
                    best_local = local_max
                    best_val = new_val
            if best_val != old_val:
                a[j] = best_val
                a = a / a.sum()
                improved = True
            else:
                a[j] = old_val
                a = a / a.sum()
        
        if not improved:
            break
    
    final_cmax = float(_fft_convolve(a, a).max())
    if final_cmax < best_cmax:
        best_cmax = final_cmax
        a = a  # already at best
    else:
        # Revert to best
        pass
    
    return a, 2.0 * n * best_cmax

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
