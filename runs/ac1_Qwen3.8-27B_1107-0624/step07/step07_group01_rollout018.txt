import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(x):
    """Project x onto the simplex {a: a>=0, sum(a)=1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
   (theta) = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def _grad_autocorr(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    w has length 2n-1, a has length n.
    Returns gradient of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return h


def _objective_and_grad(a, w=None):
    """Compute objective (max autocorrelation) and gradient.
    If w is None, use the argmax subgradient.

    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if w is None:
        idx = int(np.argmax(c))
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    grad = _grad_autocorr(a, w)
    return c_max, grad, c


def _game_iterate(a_init, deadline, n_epochs=200, momentum=0.9):
    """
    Game-theoretic iteration for min_a max_k (a*a)_k s.t. a in simplex.
    Uses exponential weighting for the dual variable w and
    projected gradient with momentum for the primal variable a.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    w = np.ones(2 * n - 1) / (2 * n - 1)
    v = np.zeros(n)  # momentum

    best_a = a.copy()
    best_score = _score(a)
    eta = 0.5 / np.sqrt(n)
    temp = 1.0 / (10.0 * n)  # temperature for exponential weighting
    stall = 0

    t = 0
    while time.monotonic() < deadline:
        t += 1

        # Compute autocorrelation

            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        w = _softmax(c_norm / temp)

def _frank_wolfe(a_init, w, deadline, max_iter=5000):
    """
    Frank-Wolfe (conditional gradient) method for:
    min_a w^T (a*a) subject to a >= 0, sum a = 1
    where w is the dual variable (distribution over lags).
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s

    best_a = a.copy()
    best_val = float(w @ _full_convolve(a, a))
    gamma = 1.0  # FW step size parameter

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient: grad[i] = sum_k w[k] * (a[i-k] + a[k-i])
        grad = _grad_autocorr(a, w)

        # Find the vertex (standard basis vector e_j) that minimizes grad^T e_j
        # i.e., argmin_j grad[j]
        j = int(np.argmin(grad))

        # FW step: a_new = (1 - step) * a + step * e_j
        step = 2.0 / (it + 2)  # standard FW step size
        a_new = (1 - step) * a + step * 0
        a_new[j] += step
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 0:
            a_new = a_new / s_new

        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        val_new = float(w @ c_new)

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def _active_set_refine(a_init, deadline, k_active=5):

            best_score = score
            best_a = a.copy()

        # Find active lags (top k)

            eta *= 0.9
            if eta < 1e-8:

def _lbfgs_smooth_obj(z, beta, n, target):
    """Objective for L-BFGS: smoothed max of autocorrelation."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_smooth_grad(z, beta, n):
    """Gradient for L-BFGS smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        return _lbfgs_smooth_grad(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        s = a_out.sum()
        if s > 0:
            a_out = a_out / s
        return a_out
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    """Adaptive L-BFGS with decreasing beta schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # More aggressive beta schedule - go smaller
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7, 2e-7,
                              1e-7, 5e-8])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 3:
            break

        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(15000, max(200, int(remaining / (0.003 * scale * 25))))

        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc

        if current_sc < 1.5005:
            break

    return current_a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.25):
    if bands is None:
        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _nesterov_subgrad(a_init, deadline, k_active=5, max_steps=100000):
    """
    Nesterov-accelerated subgradient method with adaptive step size.
    Uses top-k lags for the subgradient with geometric weighting.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = _score(a)

    beta = 0.9  # momentum parameter
    eta = 2.0 / (np.sqrt(n) + 1)
    stall = 0

    for t in range(1, max_steps + 1):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a + beta * (a - y) if t > 1 else a.copy()
        # Actually standard Nesterov: y_t = x_t + beta_t * (x_t - x_{t-1})
        # Simplified: y = (1+beta)/2 * a + (1-beta)/2 * y_prev
        # Let's use a simpler momentum scheme
        y = a + beta * (a - y) if t > 1 else a.copy()

        # Compute subgradient at y
        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())

        # Top-k with geometric weights
        idx = np.argsort(c_y)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        for i, k in enumerate(idx):
            w[k] = (0.7 ** i)
        w = w / w.sum()

        grad = _grad_autocorr(y, w)

        # Step
        a_new = _proj_simplex(y - eta * grad / max(c_max_y, 1e-15))

        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        score_new = 2.0 * n * c_new.max() / (a_new.sum() ** 2)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.002, 5.0 / np.sqrt(n))
        else:
            stall += 1
            if stall > 100:
                eta *= 0.8
                stall = 0
            if eta < 1e-8:
                break

        a = a_new

        # Decay momentum
        beta = min(0.999, beta * 0.9999)

    return best_a, best_score


def _mirror_descent(a_init, deadline, k_active=5):
    """
    Mirror descent (multiplicative weight updates) on the simplex.
    a[i] <- a[i] * exp(-eta * grad[i]) / Z
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)
    eta = 1.0 / np.sqrt(n)
    stall = 0

    for t in range(200000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Subgradient
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k_active
        grad = _grad_autocorr(a, w)

        # Mirror descent update (multiplicative weights)
        z = np.log(np.maximum(a, 1e-30)) - eta * grad / max(c_max, 1e-15)
        z = z - z.max()
        a_new = _softmax(z)

        score_new = _score(a_new)
        if score_new <= score + 1e-14:
            a = a_new
            if score_new < best_score:
                eta = min(eta * 1.001, 5.0 / np.sqrt(n))
        else:
            eta *= 0.9
            if stall > 500:
                eta *= 0.5
            if eta < 1e-8:
                break

    return best_a, best_score


def _sweep_n	best(a_best, deadline, candidate_ns):
    """Try different sequence lengths by interpolation + optimization."""
    results = {}
    for target_n in candidate_ns:
        if time.monotonic() > deadline:
            break
        a_interp = _interpolate_to_n(a_best, target_n)
        sc = _score(a_interp)
        print(f"    n={target_n}: interpolated score = {sc:.10f}")

        # Quick optimization
        if time.monotonic() > deadline - 30:
            continue

        # Game iteration
        a_game, sc_game = _game_iterate(a_interp, time.monotonic() + min(60, (deadline - time.monotonic()) * 0.4))
        if sc_game < sc:
            a_interp = a_game
            sc = sc_game

        # L-BFGS polish
        if time.monotonic() < deadline - 20:
            a_lbfgs = _adaptive_lbfgs(a_interp, time.monotonic() + 60, reserve_time=20)
            sc_lb = _score(a_lbfgs)
            if sc_lb < sc:
                a_interp = a_lbfgs
                sc = sc_lb

        results[target_n] = (a_interp, sc)
        print(f"    n={target_n}: optimized score = {sc:.10f}")

    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining game-theoretic iteration,
    Nesterov subgradient, mirror descent, active-set refinement,
    and L-BFGS polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf
    best_n = 16384

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        try_update(parent, "parent")
        print(f"  Parent: n={len(parent)}, score={_score(parent):.10f}")
        a_start = parent.copy()
    else:
        n_init = 16384
        a_start = _make_chirp(n_init, freq=1.0, eps=0.3)
        try_update(a_start, "chirp_init")
        print(f"  Chirp init: score={_score(a_start):.10f}")

    # Quick game iteration for initial improvement
    t_phase1 = time.monotonic() + 60
    a_g1, sc_g1 = _game_iterate(a_start, t_phase1, n_epochs=100)
    try_update(a_g1, "game_init")

    # Quick L-BFGS
    if time.monotonic() < t_phase1 + 30:
        a_l1 = _adaptive_lbfgs(best_seq, time.monotonic() + 90, reserve_time=30)
        try_update(a_l1, "lbfgs_init")

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: Game-theoretic iteration (main) ===
    print(f"\n=== Phase 2: Game iteration (t={time.monotonic() - t_start:.1f}s) ===")
    t_phase2 = time.monotonic() + min(300, (deadline - time.monotonic()) * 0.35)
    a_g2, sc_g2 = _game_iterate(best_seq, t_phase2, n_epochs=500)
    try_update(a_g2, "game_main")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    t_phase3 = time.monotonic() + min(200, (deadline - time.monotonic()) * 0.2)
    a_ns, sc_ns = _nesterov_subgrad(best_seq, t_phase3, k_active=5, max_steps=80000)
    try_update(a_ns, "nesterov_k5")

    a_ns3, sc_ns3 = _nesterov_subgrad(best_seq, time.monotonic() + 100, k_active=3, max_steps=40000)
    try_update(a_ns3, "nesterov_k3")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror descent (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        t_phase4 = time.monotonic() + min(150, (deadline - time.monotonic()) * 0.15)
        a_md, sc_md = _mirror_descent(best_seq, t_phase4, k_active=5)
        try_update(a_md, "mirror_k5")

    # === Phase 5: Active set refinement ===
    print(f"\n=== Phase 5: Active set refinement (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        t_phase5 = time.monotonic() + min(120, (deadline - time.monotonic()) * 0.15)
        a_as, sc_as = _active_set_refine(best_seq, t_phase5, k_active=5)
        try_update(a_as, "active_k5")

        a_as3, sc_as3 = _active_set_refine(best_seq, time.monotonic() + 60, k_active=3)
        try_update(a_as3, "active_k3")

    # === Phase 6: L-BFGS with tight smoothing ===
    print(f"\n=== Phase 6: L-BFGS polish (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        t_phase6 = deadline - 100
        a_l6 = _adaptive_lbfgs(best_seq, t_phase6, reserve_time=50)
        try_update(a_l6, "lbfgs_main")

    # === Phase 7: Additional game iteration from current best ===
    print(f"\n=== Phase 7: More game iteration (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        t_phase7 = deadline - 80
        a_g7, sc_g7 = _game_iterate(best_seq, t_phase7, n_epochs=300)
        try_update(a_g7, "game_round2")

    # === Phase 8: Length exploration ===
    print(f"\n=== Phase 8: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        candidate_ns = [8192, 10240, 12288, 16384, 20480, 24576, 32768]
        candidate_ns = [n for n in candidate_ns if n != n_best]
        # Sort by closeness to current best
        candidate_ns.sort(key=lambda x: abs(x - n_best))

        for target_n in candidate_ns[:3]:  # Try at most 3 new lengths
            if time.monotonic() > deadline - 40:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")

            if time.monotonic() < deadline - 30:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=30)
                try_update(a_opt, f"n_{target_n}_lbfgs")

            if time.monotonic() < deadline - 30:
                a_opt2, _ = _game_iterate(a_interp, time.monotonic() + 30)
                try_update(a_opt2, f"n_{target_n}_game")

    # === Phase 9: Final aggressive polishing ===
    print(f"\n=== Phase 9: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(8):
            if time.monotonic() > deadline - 5:
                break

            # L-BFGS with very tight beta
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * (1e-7 * (0.5 ** rnd))
            remaining = deadline - 5 - time.monotonic()
            max_iter = min(8000, max(500, int(remaining / 0.005)))
            a_f = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Quick game iteration
            if time.monotonic() < deadline - 5:
                a_g, _ = _game_iterate(best_seq, time.monotonic() + 30)
                try_update(a_g, f"final_game_{rnd}")

            # Nesterov
            if time.monotonic() < deadline - 5:
                a_n, _ = _nesterov_subgrad(best_seq, time.monotonic() + 20, k_active=3, max_steps=5000)
                try_update(a_n, f"final_nesterov_{rnd}")

            # Active set
            if time.monotonic() < deadline - 5:
                a_k, _ = _active_set_refine(best_seq, time.monotonic() + 20, k_active=3)
                try_update(a_k, f"final_active_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
