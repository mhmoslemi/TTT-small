import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute g[i] = sum_j a[j] * A_vec[i+j+n-1] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _hess_matvec_precond(v, A_vec, n, mu):
    """Hessian-vector product for averaging over active set, with regularization."""
    return (1.0 / max(len(A_vec[A_vec > 0], 1), 1)) * _corr_a_A(v, A_vec, n) + mu * v


def _spectral_preconditioner(A_vec, n):
    """Compute diagonal preconditioner based on spectral approximation of Hessian."""
    m = int(A_vec.sum())
    if m == 0:
        return np.ones(n)
    # The Hessian is (1/m) * sum_{k in active} H_k
    # Each H_k has eigenvalues ~ 2*cos(2*pi*j*k/n) for mode j
    # So H has eigenvalues ~ (1/m) * sum_{k in active} 2*cos(2*pi*j*k/n) = (2/m) * |FFT(A_vec)_j| * cos phase
    # Simpler: the eigenvalue for mode j is (2/m) * Re(FFT(A_vec)[j])
    N = _next_pow2(2 * n - 1)
    FA = np.fft.rfft(A_vec, N)
    # Eigenvalue approximation for mode j
    eig_approx = (2.0 / m) * np.abs(FA[:n])
    # Preconditioner is 1/eigvalue (clipped)
    eig_approx = np.maximum(eig_approx, 0.01)
    return 1.0 / eig_approx

                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = best_patience_lr * 0.7
                    beta_m = min(beta_m * 0.95, 0.5)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_v2(a, deadline):
    """Try zeroing out the smallest elements with refinement."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.005, 0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _symmetric_project(a):
    """Project to palindromic sequences."""
    return 0.5 * (a + a[::-1])


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Spectral initialization
    2. L-BFGS with log-sum-exp continuation
    3. Active-set Newton with preconditioned CG
    4. Symmetric Newton variant
    5. Subgradient polishing with momentum
    6. Shrinkage + re-optimization
    7. Multi-n exploration
    8. Diverse restarts
    9. Final ultra-precision polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 0: Spectral initialization
    print(f"\n[Phase 0] Spectral initialization (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        spec_deadline = min(deadline - 200, time.monotonic() + 60)
        a_spec = _spectral_init(parent, spec_deadline, n_max_refine=15)
        update_best(a_spec, _score(a_spec), "spectral_init")

    # Phase 1: Main L-BFGS
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs_v2(best_seq if best_seq is not None else parent, deadline, min_time_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Active-set Newton (asymmetric)
    print(f"\n[Phase 2] Active-set Newton asymmetric (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        as_deadline = min(deadline - 500, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton_v2(best_seq, as_deadline, max_iter=300, tol_active=0.001,
                                            use_precond=True, use_symmetric=False)
        update_best(a_as, sc_as, "as_newton_asym")

    # Phase 3: Active-set Newton (symmetric)
    print(f"\n[Phase 3] Active-set Newton symmetric (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        # Symmetrize current best
        a_sym = _symmetric_project(best_seq)
        a_sym = np.maximum(a_sym, 0)
        a_sym = a_sym / a_sym.sum()
        update_best(a_sym, _score(a_sym), "symmetrized")

        as_deadline = min(deadline - 400, time.monotonic() + 120)
        a_as_sym, sc_as_sym = _active_set_newton_v2(a_sym, as_deadline, max_iter=300, tol_active=0.001,
                                                    use_precond=True, use_symmetric=True)
        update_best(a_as_sym, sc_as_sym, "as_newton_sym")

    # Phase 4: Subgradient with momentum
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_refined(best_seq, sg_deadline, n_max_steps=5000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_momentum")

    # Phase 5: Shrinkage
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage_v2(best_seq, deadline - 250)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score + 0.001:
            a_shr_opt = _run_lbfgs_v2(a_shr, deadline, min_time_remaining=150.0)
            update_best(a_shr_opt, _score(a_shr_opt), "lbfgs_post_shrink")

    # Phase 6: Multi-n search
    print(f"\n[Phase 6] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent and abs(_score(best_seq) - best_score) < 1e-15:
            continue

        a_init = _interpolate_to_n(best_seq, n_try)
        sc_init = _score(a_init)

        a_opt = _run_lbfgs_v2(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _active_set_newton_v2(a_opt, min(deadline - 40, time.monotonic() + 60),
                                                     max_iter=150, tol_active=0.001, use_precond=True)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 7: Diverse restarts
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        restarts = [
            ("perturb_003", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_008", lambda: _perturb(best_seq, 0.05, rng)),
            ("fourier_pert", lambda: _perturb_fourier(best_seq, rng, n_modes=5)),
            ("perturb_015", lambda: _perturb(best_seq, 0.10, rng)),
            ("gaussian_init", lambda: _make_profile(len(best_seq), "gaussian")),
            ("bimodal_init", lambda: _make_profile(len(best_seq), "bimodal")),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 30:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs_v2(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton_v2(a_r, min(deadline - 10, time.monotonic() + 30),
                                                     max_iter=100, tol_active=0.001, use_precond=True)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 8: Final ultra-precision polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tight active-set Newton
        as_dl = min(deadline - 3, time.monotonic() + 120)
        a_fin, sc_fin = _active_set_newton_v2(best_seq, as_dl, max_iter=500, tol_active=0.0003,
                                              use_precond=True)
        update_best(a_fin, sc_fin, "final_as_tight")

        # Even tighter
        if time.monotonic() < deadline - 3:
            as_dl2 = min(deadline - 2, time.monotonic() + 60)
            a_fin2, sc_fin2 = _active_set_newton_v2(best_seq, as_dl2, max_iter=300, tol_active=0.0001,
                                                    use_precond=True)
            update_best(a_fin2, sc_fin2, "final_as_ultra")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 5 - time.monotonic()
            if remaining > 5:
                max_iter = min(5000, max(200, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj_wrapper,
                        z,
                        args=(beta_final, n),
                        jac=_lsgrad_wrapper,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

        # One more Newton pass
        if time.monotonic() < deadline - 2:
            a_final, sc_final = _active_set_newton_v2(best_seq, deadline - 1, max_iter=200,
                                                      tol_active=0.0001, use_precond=True)
            update_best(a_final, sc_final, "absolute_final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
