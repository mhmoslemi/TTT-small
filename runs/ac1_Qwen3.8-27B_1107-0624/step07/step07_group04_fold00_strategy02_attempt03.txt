    k = 1

    """Compute full convolution using FFT with power-of-two padding."""
    n, m = len(a), len(b)
    out_n = n + m -1
    N = _next_pow2(out_n)
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)

    conv = np.fft.irfft(Fc, N)[:out_n]
