    N = _next_pow2(out_len)
    F = np.fft.rfft(v_rev, N) * np.fft.rfft(S_ind, N)
    result = np.fft.irfft(F, N)[:out_len]
    return (2.0 / m) * result[n - 1:2 * n - 1]


class _HankelOp:
    """Linear operator for the QP subproblem Hessian (restricted to support)."""

    def __init__(self, S_ind, m, n, support, eps_reg):
        self.S_ind = S_ind
        self.m = m
        self.n = n
        self.support = support
        self.eps_reg = eps_reg

    def _matvec(self, v):
        v_zeroed = v * self.support
        result = _hankel_matvec_full(v_zero
                                     d, self.S_ind, self.m)
        result = result * self.support
        return result + self.eps_reg * v


def _solve_qp_subproblem(S_ind, m, n, deadline, eps_reg=1e-10, max_outer=80):
    """
    Solve: min (1/m) * sum_{k in S} c_k(a) s.t. sum a = 1, a >= 0
    using active-set method on non-negativity constraints.
    """
    support = np.ones(n, dtype=bool)
    a = np.ones(n) / n

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        n_sup = int(support.sum())
        if n_sup < 2:
            return np.ones(n) / n

        idx = np.where(support)[0]

        # Build linear operator for the reduced system
        op = _HankelOp(S_ind, m, n, support, eps_reg)
        L = LinearOperator((n, n), matvec=op._matvec)

        rhs = np.zeros(n)
        rhs[support] = 1.0

        # Solve Q_sub * v = 1 on the support using CG
        max_cg = min(20 * n_sup, 5000)
        try:
            v, info = cg(L, rhs, tol=1e-10, maxiter=max_cg)
        except Exception:
            v = rhs * support
            v = v / max(v.sum(), 1e-15)

        # Form solution
        a = np.zeros(n)
        a[support] = v[support]
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            return a
        a = a / s

        # Check non-negativity on support
        min_val = np.min(a[support])
        if min_val > -1e-9:
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                return np.ones(n) / n
            a = a / s

            # Check KKT for non-support indices
            Qa = _hankel_matvec_full(a, S_ind, m)
            lan = -np.mean(Qa[support])
            violated = (~support) & (Qa + lan < -1e-7)

            if not violated.any():
                return a

            # Add most violated indices to support
            vio_idx = np.where(violated)[0]
            n_add = min(20, len(vio_idx))
            order = np.argsort(Qa[vio_idx] + lan)[:n_add]
            support[vio_idx[order]] = True
        else:
            # Remove most negative support indices
            neg = support & (a < -1e-9)
            if neg.any():
                neg_idx = np.where(neg)[0]
                n_remove = min(20, len(neg_idx))
                worst = neg_idx[np.argsort(a[neg_idx])[:n_remove]]
                support[worst] = False
            else:
                break

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _active_set_qp_refine(a_init, deadline, tol_active=0.002, max_rounds=15):
    """
    Active-set QP refinement: iteratively identify active set and solve subproblem.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())

        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        # Limit active set size for stability
        if n_active > 4 * n:
            top_idx = np.argsort(c)[::-1][:4 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 4 * n

        S_ind = active.astype(np.float64)
        m = n_active

        # Solve QP subproblem
        a_new = _solve_qp_subproblem(S_ind, m, n, deadline, eps_reg=1e-10)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            sc_old_best = best_sc
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"    QP round {rnd}: {best_sc:.10f} (was {sc_old_best:.10f})")
        else:
            # Blend to stay safe
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        # Check convergence: is the active set correct?
        c_check = _full_convolve
        (a, a)
        c_max_check = float(c_check.max())
        active_check = c_check >= (c_max_check - 0.001 * c_max_check)
        # If no new peaks appeared, we're at a stationary point
        new_peaks = active_check & (~active)
        if not new_peaks.any():
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta-continuation on log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.0075, 0.0037, 0.0018, 0.0009,
                                  0.00045, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1, n / 2048.0)
        max_iter = min(6000, max(30, int(remaining / (est_time * 6))))

        try:
            result = minimize(
                _lsobj_kw,
                z,
                args=(beta, n),
                jac=_lsgrad_kw,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            sc_cur = _score(current_a)
            if sc_new < sc_cur - 1e-14:
                current_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _lsobj_kw(z, beta, n):
    return _lsobj(z, beta, n)


def _lsgrad_kw(z, beta, n):
    return _lsgrad(z, beta, n)


def _hedge_steps(a_init, deadline, n_steps=5000, eta_init=0.5):
    """Multiplicative weights (Hedge) for the max-convolution problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Gradient of c_{k*}: grad_i = 2 * a[k*-i] if 0 <= k*-i <= n-1
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Multiplicative update
        grad_scaled = eta * grad / max(c_max, 1e-15)
        a = a * np.exp(-np.clip(grad_scaled, -1.0, 1.0))
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Adaptive learning rate
        if i % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                eta = min(eta * 1.05, eta_init)
            else:
                eta *= 0.85

        # Decay eta
        eta *= 0.9995

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 5)))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage(a, deadline):
    """Try zeroing out smallest elements at various fractions."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, Active-Set QP, Hedge, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "init_uniform")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Active-Set QP refinement (KEY INNOVATION)
    print(f"\n[Phase 2] Active-Set QP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        qp_deadline = min(deadline - 500, time.monotonic() + 180.0)
        a_qp, sc_qp = _active_set_qp_refine(best_seq, qp_deadline, tol_active=0.002, max_rounds=20)
        update_best(a_qp, sc_qp, "qp_refine_1")

        # Second round with tighter tolerance
        if time.monotonic() < deadline - 450:
            qp_deadline2 = min(deadline - 450, time.monotonic() + 90.0)
            a_qp2, sc_qp2 = _active_set_qp_refine(best_seq, qp_deadline2, tol_active=0.001, max_rounds=15)
            update_best(a_qp2, sc_qp2, "qp_refine_tight")

    # Phase 3: Hedge (multiplicative weights)
    print(f"\n[Phase 3] Hedge/Multiplicative Weights (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        hedge_deadline = min(deadline - 400, time.monotonic() + 120.0)
        a_h, sc_h = _hedge_steps(best_seq, hedge_deadline, n_steps=3000, eta_init=0.3)
        update_best(a_h, sc_h, "hedge")

        # QP after hedge
        if sc_h < best_score + 0.01 and time.monotonic() < deadline - 350:
            qp_deadline3 = min(deadline - 350, time.monotonic() + 60.0)
            a_hq, sc_hq = _active_set_qp_refine(a_h, qp_deadline3, tol_active=0.002, max_rounds=10)
            update_best(a_hq, sc_hq, "hedge+qp")

    # Phase 4: Shrinkage + L-BFGS
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 300)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score + 0.001 and time.monotonic() < deadline - 200:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=100.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")
            if time.monotonic() < deadline - 100:
                a_shr3, sc_shr3 = _active_set_qp_refine(a_shr2, min(deadline - 50, time.monotonic() + 60.0),
                                                         tol_active=0.002, max_rounds=10)
                update_best(a_shr3, sc_shr3, "shrink+lbfgs+qp")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates.sort()

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent and n_try in [2048, 4096, 8192]:
            continue  # already done

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        if sc_init < best_score + 0.05:
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n_{n_try}")

            if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 60:
                a_opt2, sc_opt2 = _active_set_qp_refine(a_opt, min(deadline - 40, time.monotonic() + 50.0),
                                                         tol_active=0.002, max_rounds=8)
                update_best(a_opt2, sc_opt2, f"n_{n_try}_qp")

    # Phase 6: Diverse restarts
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("perturb_s", 0.03),
            ("perturb_m", 0.08),
            ("perturb_l", 0.15),
        ]
        for label, sigma in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=25.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _active_set_qp_refine(a_r, min(deadline - 15, time.monotonic() + 30.0),
                                                     tol_active=0.002, max_rounds=8)
                update_best(a_r2, sc_r2, f"restart_{label}_qp")

        # Fourier perturbation
        if time.monotonic() < deadline - 25:
            p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            a_f = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            update_best(a_f, _score(a_f), "restart_fourier")

    # Phase 7: Final polish - very tight QP
    print(f"\n[Phase 7] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Tight QP
        a_fin, sc_fin = _active_set_qp_refine(best_seq, min(deadline - 3, time.monotonic() + 60.0),
                                               tol_active=0.001, max_rounds=10)
        update_best(a_fin, sc_fin, "final_qp")

        # One more L-BFGS with very small beta
        if time.monotonic() < deadline - 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0001
            remaining = deadline - 5 - time.monotonic()
            if remaining > 3:
                max_iter = min(2000, max(100, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _lsobj_kw, z, args=(beta_final, n),
                        jac=_lsgrad_kw,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs")
                except Exception:
                    pass

        # Final QP after everything
        if time.monotonic() < deadline - 3:
            a_last, sc_last = _active_set_qp_refine(best_seq, deadline - 2,
                                                    tol_active=0.0005, max_rounds=5)
            update_best(a_last, sc_last, "abs_final_qp")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
