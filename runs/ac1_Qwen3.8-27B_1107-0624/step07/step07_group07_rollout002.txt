"""
Step sequence optimizer v2: minimizes 2*n*max(conv(a,a))/sum(a)^2.
New approaches: hinge-loss ADAM, spectral shaping, parametric Fourier init,
improved L-BFGS, gambit peak shaving, multi-length search.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hinge_obj(z, M, n):
    """Objective: sum of squared hinge losses on autocorrelation peaks."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    excess = np.maximum(c - M, 0)
    return float(np.sum(excess * excess))

def _hinge_grad(z, M, n):
    """Gradient of hinge objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    excess = np.maximum(c - M, 0)
    # d/d a_i of sum_k excess_k^2 = 2 * sum_k excess_k * d(a*a)_k / da_i
    # d(a*a)_k / da = (a flipped) convolved with excess
    a_rev = a[::-1]
    w = 2.0 * excess
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    # Gradient w.r.t. z: h * a - a * sum(h * a) (softmax chain rule)
    return h * a - a * np.dot(h, a)

def _adam_optimize(z_init, M, n, deadline, lr_init=0.1, max_steps=50000):
    """ADAM optimizer for the hinge-loss objective in z-space."""
    z = z_init.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_a = _softmax(z)
    best_sc = _score(best_a)
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()
        
        obj = _hinge_obj(z, M, n)
        grad = _hinge_grad(z, M, n)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (i + 1))
        v_hat = v / (1 - beta2 ** (i + 1))
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.0005 * i)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()
        z = np.clip(z, -50, 50)
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _spectral_shaping(a_init, deadline, max_iters=10000, aggression=0.3):
    """
    Spectral shaping: iteratively flatten the power spectrum |A(f)|^2
    while maintaining non-negativity of a.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT of a
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flatten the magnitude
        # Use geometric mean as target
        log_mag = np.log(np.maximum(mag, 1e-30))
        target_log_mag = np.mean(log_mag)
        target_mag = np.exp(target_log_mag)
        
        # Blend: move current magnitude toward target
        new_mag = (1 - aggression) * mag + aggression * target_mag
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_sc

def _fourier_init(n, n_modes=50, rng=None):
    """Generate a random Fourier-based initialization."""
    if rng is None:
        rng = np.random.default_rng()
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.01, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _parametric_optimize(n, n_params=30, deadline, init_a=None):
    """Optimize Fourier coefficients to get a good initialization."""
    if init_a is not None:
        # Extract Fourier coefficients from init
        A = np.fft.rfft(init_a, n)
        # Use first n_params real FFT bins
        params = np.zeros(n_params * 2)  # real and imag parts
        for i in range(min(n_params, len(A))):
            params[2*i] = A[i].real
            params[2*i+1] = A[i].imag
    else:
        params = np.random.default_rng(42).standard_normal(n_params * 2) * 0.01
        params[0] = 1.0  # DC component
    
    def fun(p):
        # Reconstruct a from Fourier params
        A = np.zeros(n, dtype=complex)
        for i in range(min(len(p) // 2, n)):
            A[i] = p[2*i] + 1j * p[2*i+1]
        a = np.fft.irfft(A, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        return _score(a)
    
    # Use L-BFGS on the parametric space
    try:
        result = minimize(fun, params, method='L-BFGS-B',
                         options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-10})
    except Exception:
        result = None
    
    # Reconstruct
    p = params if result is None else result.x
    A = np.zeros(n, dtype=complex)
    for i in range(min(len(p) // 2, n)):
        A[i] = p[2*i] + 1j * p[2*i+1]
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return None, np.inf
    a = a / s
    return a, _score(a)

def _gambit_peak_shaving(a_init, deadline, max_iters=20000):

    to reduce peaks more effectively (sacrificing other lags).
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.01
    stall = 0
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Get active peaks
        thresh = c_max * 0.999
        active = c >= thresh
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Standard gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()  # Project to sum-preserving direction
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Try standard descent
        best_a_try = a
        best_sc_try = best_sc
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:

            a_try = a - step * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:

            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_a_try = a_try.copy()

        # Find a_i that, when increased, would reduce the peak most
        # This is: find i that minimizes sum_k w_k * (a*a)_k derivative wrt a_i INCREASE
        # Actually, the gradient already tells us this. For gambit, try:
        # increase a_i for i where g_i is most negative (these would reduce the peak if increased)
        
        # Actually, let's try a different gambit: perturb in a random direction

        if i % 5 == 0:
            rng_step = np.random.standard_normal(n)
            rng_step = rng_step - rng_step.mean()

            best_a = best_a_try.copy()

            eta = min(eta * 1.02, 0.1)
            stall = 0
        else:
            stall += 1
            eta *= 0.92

def _lbfgs_smooth(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with log-sum-exp smoothing and fine beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_init
    
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001, 5e-5,
                              2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()
        
        def fun(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def jac(z):
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)
        
        max_iter = min(5000, max(100, int(remaining / 0.003)))
        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                            options={'maxiter': max_iter, 'ftol': 1e-18,
