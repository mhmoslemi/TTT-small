    n = len(a_init)
    if K_fracs is None:
        K_fracs = [0.05, 0.1, 0.2, 0.5]

    best_a = a_init.copy()
    best_sc = _score(a_init)
    c = _full_convolve(a_init, a_init)

    for K_frac in K_fracs:
        if time.monotonic() > deadline:
            break

        K = max(1, int(K_frac * (2 * n - 1)))
        top_idx = np.argsort(c)[::-1][:K]
        S_vec = np.zeros(2 * n - 1, dtype=np.float64)
        S_vec[top_idx] = 1.0
        m = K

        N = _next_pow2(n + (2 * n - 1) - 1)
        fS = np.fft.rfft(S_vec, N)

        a = a_init.copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s

        mu = 1e-5 * c.max() + 1e-8
        u_prev = None
        stall_count = 0
        prev_f = float(c.max())

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break

            # Compute gradient
            grad = _hess_matvec_precomputed(a, S_vec, n, m, 0.0, N, fS)
            g_t = grad - np.mean(grad)
            gnorm = np.linalg.norm(g_t)

            if gnorm < 1e-14:
                break

            # CG solve: (H + mu*I) u = g_t
            x = u_prev.copy() if u_prev is not None else np.zeros(n)
            r = g_t - _hess_matvec_precomputed(x, S_vec, n, m, mu, N, fS)
            # Diagonal preconditioner
            diag = _hess_matvec_precomputed(np.ones(n), S_vec, n, m, mu, N, fS)
            diag = np.maximum(diag, 1e-10)
            z_vec = r / diag
            p = z_vec.copy()
            rz = np.dot(r, z_vec)

            for cg_it in range(200):
                Ap = _hess_matvec_precomputed(p, S_vec, n, m, mu, N, fS)
                pAp = np.dot(p, Ap)
                if pAp < 1e-30:
                    break
                alpha = rz / pAp
                x += alpha * p
                r_new = r - alpha * Ap
                z_new = r_new / diag
                rz_new = np.dot(r_new, z_new)
                if np.sqrt(abs(rz_new)) < 1e-12 * (1 + np.linalg.norm(g_t)):
                    break
                p = z_new + (rz_new / rz) * z_vec
                r = r_new
                z_vec = z_new
                rz = rz_new
                if rz < 0:
                    break

            u = x
            u_prev = u.copy()

            # Solve for constraint
            w = np.zeros(n)
            r_w = np.ones(n) - _hess_matvec_precomputed(w, S_vec, n, m, mu, N, fS)
            z_w = r_w / diag
            p_w = z_w.copy()
            rz_w = np.dot(r_w, z_w)
            for cg_it in range(100):
                Ap_w = _hess_matvec_precomputed(p_w, S_vec, n, m, mu, N, fS)
                pAp_w = np.dot(p_w, Ap_w)
                if pAp_w < 1e-30:
                    break
                alpha_w = rz_w / pAp_w
                w += alpha_w * p_w
                r_w_new = r_w - alpha_w * Ap_w
                z_w_new = r_w_new / diag
                rz_w_new = np.dot(r_w_new, z_w_new)
                if np.sqrt(abs(rz_w_new)) < 1e-12:
                    break
                p_w = z_w_new + (rz_w_new / rz_w) * z_w
                r_w = r_w_new
                z_w = z_w_new
                rz_w = rz_w_new
                if rz_w < 0:
                    break

            sum_w = float(w.sum())
            if abs(sum_w) < 1e-15:
                dx = -u
            else:
                lam = float(u.sum()) / sum_w
                dx = -u + lam * w
            dx = dx - np.mean(dx)

            # Line search
            f_old = float(_full_convolve(a, a).max())
            neg_mask = dx < -1e-15
            if neg_mask.any():
                t_max = 0.99 * np.min(a[neg_mask] / (-dx[neg_mask]))
            else:
                t_max = 1.0
            t_max = max(t_max, 1e-10)

            accepted = False
            t = min(1.0, t_max)
            for ls in range(50):
                a_try = a + t * dx
                if np.min(a_try) < -1e-10:
                    t *= 0.5
                    continue
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try <= f_old + 1e-13:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5

            if not accepted:
                # Backtracking gradient step
                t = min(0.1, t_max)
                for ls in range(20):
                    a_try = a - t * g_t
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        t *= 0.5
                        continue
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    f_try = float(c_try.max())
                    if f_try < f_old - 1e-13:
                        a = a_try
                        break
                    t *= 0.5
                else:
                    stall_count += 1
                    if stall_count > 10:
                        break
                    continue

            stall_count = 0
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()

            # Check convergence
            c_new = _full_convolve(a, a)
            if abs(float(c_new.max()) - prev_f) < 1e-14:
                break
            prev_f = float(c_new.max())

            # Update active set every 20 iterations
            if (it + 1) % 20 == 0:
                top_idx = np.argsort(c_new)[::-1][:K]
                S_vec_new = np.zeros(2 * n - 1, dtype=np.float64)
                S_vec_new[top_idx] = 1.0
                if not np.allclose(S_vec, S_vec_new):
                    S_vec = S_vec_new
                    fS = np.fft.rfft(S_vec, N)
                    u_prev = None

    return best_a, best_sc


def _run_lbfgs_improved(a_init, deadline, min_time_remaining=50.0):
    """L-BFGS with adaptive beta continuation, more aggressive schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 2.0, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.025,
                                  0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1, n / 4096.0)
        max_iter = min(10000, max(100, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_momentum(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient with momentum for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    momentum = np.zeros(n)
    beta_mom = 0.9
    patience = 0
    prev_sc = best_sc

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        momentum = beta_mom * momentum - step * grad
        z = z + momentum
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                if sc < prev_sc - 1e-10:
                    lr = min(lr * 1.05, lr_init * 1.5)
            else:
                patience += 1
                beta_mom = max(0.5, beta_mom * 0.95)
                lr *= 0.9
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.5
                    beta_mom = 0.9
            prev_sc = best_sc

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bicount":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(4.0 * np.pi * t) + 0.05 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage_polish(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.02, 0.05, 0.10]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, multi-K active-set Newton,
    subgradient momentum, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent (use ~35% of time)
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = t_start + (deadline - t_start) * 0.35
    a_main = _run_lbfgs_improved(parent, lbfgs_deadline, min_time_remaining=10.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Multi-K active-set Newton (use ~35% of time)
    print(f"\n[Phase 2] Multi-K Active-Set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        as_deadline = t_start + (deadline - t_start) * 0.70
        K_fracs = [0.1, 0.05, 0.2, 0.5]
        a_as, sc_as = _active_set_newton_multi(best_seq, as_deadline,
                                                max_iter=400, K_fracs=K_fracs)
        update_best(a_as, sc_as, "active_set_multi")

        # Second round with different K fracs
        if time.monotonic() < deadline - (deadline - t_start) * 0.15:
            as_deadline2 = t_start + (deadline - t_start) * 0.80
            K_fracs2 = [0.02, 0.05, 0.15]
            a_as2, sc_as2 = _active_set_newton_multi(best_seq, deadline,
                                                      max_iter=300, K_fracs=K_fracs2)
            update_best(a_as2, sc_as2, "active_set_round2")

    # Subgradient momentum polish
    if best_seq is not None and time.monotonic() < deadline - 120:
        sg_deadline = min(deadline - 100, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_momentum(best_seq, sg_deadline, n_max_steps=4000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_momentum")

        # Active-set Newton after subgradient
        if sc_sg < best_score + 0.001 and time.monotonic() < deadline - 80:
            a_as3, sc_as3 = _active_set_newton_multi(best_seq, deadline - 60,
                                                      max_iter=200, K_fracs=[0.05, 0.1])
            update_best(a_as3, sc_as3, "as_post_sg")

    # Phase 3: Shrinkage
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage_polish(best_seq, deadline - 100)
        update_best(a_shr, sc_shr, "shrinkage")

    # Phase 4: Multi-n search
    print(f"\n[Phase 4] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = [n for n in n_candidates if n != n_parent]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 30:
            break

        if best_seq is not None:
            a_init = _interpolate_to_n(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        print(f"  Trying n={n_try} (remaining={remaining:.1f}s)")
        a_opt = _run_lbfgs_improved(a_init, deadline - 40, min_time_remaining=10.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _active_set_newton_multi(a_opt, min(deadline - 20, time.monotonic() + 60),
                                                        max_iter=150, K_fracs=[0.05, 0.1])
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restarts = [
            ("perturb_003", 0.003),
            ("perturb_01", 0.01),
            ("perturb_03", 0.03),
        ]
        for label, sigma in restarts:
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs_improved(p_start, deadline - 10, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _active_set_newton_multi(a_r, min(deadline - 5, time.monotonic() + 30),
                                                        max_iter=100, K_fracs=[0.05])
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 6: Final tight polish
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very tight active-set Newton with small K
        as_final_deadline = min(deadline - 2, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_newton_multi(best_seq, as_final_deadline,
                                                  max_iter=300, K_fracs=[0.01, 0.02, 0.05])
        update_best(a_fin, sc_fin, "final_as_tight")

        # Final subgradient with tiny lr
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_momentum(best_seq, deadline - 2,
                                                 n_max_steps=1000, lr_init=0.003)
            update_best(a_fin2, sc_fin2, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
