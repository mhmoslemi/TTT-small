"""
Novel optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Projected smoothed gradient descent in primal (simplex) space
2. Pair-swap coordinate descent
3. Multi-scale exploration with temperature annealing
4. Perturbation restarts for escaping local minima
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = ind[cond][-1]
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _compute_gradient(a, c, T):
    """
    Compute gradient of T*log(sum exp(c/T)) w.r.t. a.
    Uses the FFT-based formula: grad = 2 * convolve(a, w_rev)[n-1:2n-1][::-1]
    """
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        # Fallback: one-hot at argmax
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    w_rev = w[::-1]
    h = _fft_convolve(a, w_rev)
    grad = 2.0 * h[n-1:2*n-1][::-1]
    return grad


def _smoothed_grad_descent(a_init, deadline, T_start, T_end, max_iter=500000,
                           lr_start=0.01, lr_end=0.001, momentum=0.9,
                           eval_interval=50, use_backtrack=True):
    """
    Smoothed gradient descent on the simplex with momentum and backtracking.
    Works in primal space (not log-space).
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)  # momentum buffer
    lr = lr_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, (it + 1) / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        grad = _compute_gradient(a, c, T)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        step = lr * v
        
        # Clip step to avoid numerical issues
        step_norm = np.linalg.norm(step)
        if step_norm > 0.5:
            step = step * (0.5 / step_norm)
        
        a_new = _project_simplex(a - step)
        
        if use_backtrack and it % 20 == 0:
            # Check if the move improved the smoothed objective
            c_new = _fft_convolve(a_new, a_new)
            c_shifted_new = (c_new - c_new.max()) / T
            e_new = np.exp(c_shifted_new - np.max(c_shifted_new))  # already shifted
            smooth_new = T * np.log(e_new.sum() + 1e-300)
            c_shifted = (c - c.max()) / T
            e_old = np.exp(c_shifted - np.max(c_shifted))
            smooth_old = T * np.log(e_old.sum() + 1e-300)
            
            if smooth_new > smooth_old * 1.01 + 0.01:
                # Revert
                a_new = a.copy()
                v *= 0.5
        
        a = a_new
    
    # Final evaluation
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _pair_swap_cd(a, deadline, max_rounds=1000):
    """
    Coordinate descent using pair swaps: move mass from high-pressure to low-pressure coordinates.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        best_score = 2.0 * n * c_max
        
        # Compute pressure at each coordinate
        # Pressure[j] = sum over active peaks k of a[k-j] (weighted by peak value)
        threshold = c_max * 0.9
        active = np.where(c > threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute pressure
        pressure = np.zeros(n)
        for k in active:
            w = c[k] - threshold
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    pressure[j] += w * best_a[ki]
        
        # Find high-pressure and low-pressure coordinates
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        
        improved = False
        # Try moving mass from top pressure to bottom pressure
        top_idx = np.argsort(pressure)[::-1][:20]
        bot_idx = np.argsort(pressure)[:20]
        
        for j in top_idx:
            if best_a[j] < 1e-10:
                continue
            for k in bot_idx:
                if k == j:
                    continue
                for frac in [0.1, 0.25, 0.5, 0.75]:
                    delta = best_a[j] * frac
                    if delta > best_a[j] * 0.9:  # don't zero out completely
                        delta = best_a[j] * 0.9
                    a_cand = best_a.copy()
                    a_cand[j] -= delta
                    a_cand[k] += delta
                    a_cand = np.maximum(a_cand, 0)
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc_cand = 2.0 * n * float(c_cand.max())
                    if sc_cand < best_score * (1 - 1e-10):
                        best_score = sc_cand
                        best_a = a_cand.copy()
                        improved = True
                        rateom_break = True
                        break
                    else:
                        rateom_break = False
                        break  # Try next k
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try single coordinate scaling
            for j in top_idx[:10]:
                if best_a[j] < 1e-10:
                    continue
                for scale in [0.7, 0.5, 0.3, 0.1]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    a_cand = _project_simplex(a_cand)
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc_cand = 2.0 * n * float(c_cand.max())
                    if sc_cand < best_score * (1 - 1e-10):
                        best_score = sc_cand
                        best_a = a_cand.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, eval_interval=100, clip_norm=0.8):
    """Adam in log-space (similar to previous but simplified)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    
    def _softmax_stable(z):
        z = z - np.max(z)
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return np.ones_like(z) / len(z)
        return e / s
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Smoothed gradient
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s
        
        w_rev = w[::-1]
        h = _fft_convolve(a, w_rev)
        g = 2.0 * h[n-1:2*n-1][::-1]
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

c[k] = sum_{i} a[i] * a[k-i] (valid indices)
∂c[k]/∂a[j]:
- If j appears as the first index (i=j): contribution is a[k-j] (if valid)
- If j appears as the second index (k-i=j, i.e., i=k-j): contribution is a[k-j] (if valid, and i=k-j is in range)
- If j = k-j (i.e., k=2j), these are the same term, so we count it only once: contribution is a[j]

So ∂c[k]/∂a[j] = 2*a[k-j] if 0≤k-j<n and k≠2j, or a[j] if 0≤k-j<n and k=2j, or 0.

For the weighted sum: ∇f[j] = sum_k w[k] * ∂c[k]/∂a[j]

The "2*a[k-j]" term dominates. The correction for k=2j is: instead of 2*a[j], we have a[j], so the correction is -w[2j]*a[j].

In the FFT computation:
h = convolve(a, w_rev) gives us h[m] = sum_i a[i] * w_rev[m-i] = sum_i a[i] * w[2n-2-m+i]

For j in [0, n-1]:
h[2n-2-j] = sum_i a[i] * w[i+j]

So grad_approx[j] = 2 * h[2n-2-j] = 2 * sum_i a[i] * w[i+j]

The true gradient is: sum_k w[k] * (2*a[k-j] if valid else 0) - correction
= 2 * sum_{k: 0≤k-j<n} w[k] * a[k-j] - sum_{j: 2j<n and 2j≤2n-2} w[2j] * a[j]
= 2 * sum_{i: 0≤i<n} w[i+j] * a[i] - sum_j w[2j] * a[j] (for valid 2j)
= 2 * h[2n-2-j] - w[2j] * a[j]

So the correction is small (one term per j). For most j, w[2j] is close to 0 (since the peaks are concentrated), so the correction is negligible. I'll ignore it.

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Primal-space projected smoothed gradient descent with momentum (new)
2. Log-space Adam with adaptive restarts and patience (improved)
3. Pair-swap coordinate descent with FFT-based pressure (new)
4. Symmetric subspace exploration (new)
5. Multi-scale (n=8192 parent, n=4096 alternative) with interpolation
6. Staircase temperature scheduling for better convergence
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = ind[cond][-1]
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    w_rev = w[::-1]
    h = _fft_convolve(a, w_rev)
    grad = 2.0 * h[n - 1:2 * n - 1][::-1]
    return grad


def _primal_grad_descent(a_init, deadline, T_start, T_end, max_iter,
                         lr_start, lr_end, momentum, eval_interval,
                         staircase_steps=5, patience_limit=8):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    b1 = momentum
    patience = 0
    lr_mult = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, (it + 1) / max_iter)
        # Staircase temperature
        step_idx = min(staircase_steps - 1, int(frac * staircase_steps))
        T = T_start * (T_end / T_start) ** (step_idx / staircase_steps)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        c = _fft_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > patience_limit and lr_mult > 0.03:
            lr_mult *= 0.5
            patience = 0

        grad = _compute_grad(a, c, T)

        v = b1 * v + (1.0 - b1) * grad
        step = lr * v
        step_norm = np.linalg.norm(step)
        if step_norm > 0.3:
            step = step * (0.3 / step_norm)

        a_new = _project_simplex(a - step)
        a = a_new

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter, eval_interval, clip_norm, patience_limit=6):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _compute_grad(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_limit and lr_mult > 0.03:
                lr_mult *= 0.5
                patience = 0

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _pair_swap_cd(a_init, deadline, max_rounds=200):
    n = len(a_init)
    best_a = a_init.copy() / a_init.sum()
    best_score = _score_norm(best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        best_score = 2.0 * n * c_max

        threshold = c_max * 0.85
        w = np.zeros_like(c)
        active_mask = c > threshold
        w[active_mask] = c[active_mask] - threshold

        if w.sum() < 1e-15:
            break

        # Compute pressure using FFT
        w_rev = w[::-1]
        h = _fft_convolve(best_a, w_rev)
        pressure = h[n - 1:2 * n - 1][::-1]

        p_max = pressure.max()
        if p_max < 1e-15:
            break

        top_idx = np.argsort(pressure)[::-1][:30]
        bot_idx = np.argsort(pressure)[:30]

        improved = False
        for j in top_idx:
            if best_a[j] < 1e-12:
                continue
            for k in bot_idx:
                if k == j:
                    continue
                for frac in [0.15, 0.3, 0.5, 0.7]:
                    delta = best_a[j] * frac
                    a_cand = best_a.copy()
                    a_cand[j] -= delta
                    a_cand[k] += delta
                    a_cand = np.maximum(a_cand, 0.0)
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc_cand = 2.0 * n * float(c_cand.max())
                    if sc_cand < best_score * (1.0 - 1e-10):
                        best_score = sc_cand
                        best_a = a_cand.copy()
                        improved = True
                        break
                    break

            if improved:
                break

        if not improved:
            # Try single coordinate scaling on high-pressure coords
            any_improved = False
            for j in top_idx[:15]:
                if best_a[j] < 1e-12:
                    continue
                for scale in [0.6, 0.4, 0.2, 0.8, 1.3]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    a_cand = _project_simplex(a_cand)
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc_cand = 2.0 * n * float(c_cand.max())
                    if sc_cand < best_score * (1.0 - 1e-10):
                        best_score = sc_cand
                        best_a = a_cand.copy()
                        any_improved = True
                        break
                if any_improved:
                    break
            if not any_improved:
                break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _symmetric_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                        max_iter, eval_interval):
    """Optimize in the symmetric subspace: a[i] = a[n-1-i]."""
    n = len(a_init)
    # Project to symmetric
    a_sym = 0.5 * (a_init + a_init[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    half = (n + 1) // 2
    # Work with the first half + center (if n odd)
    a_half = a_sym[:half].copy()

    # Optimize a_half, then reconstruct
    z = np.log(np.maximum(a_half, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(half)
    v_buf = np.zeros(half)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_score = _score_norm(a_sym)
    best_a = a_sym.copy()

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        a_half_cur = _softmax_stable(z)
        # Reconstruct full symmetric a
        a_full = np.zeros(n)
        a_full[:half] = a_half_cur
        a_full[n - 1 : n - 1 - half : -1] = a_half_cur
        a_full = a_full / a_full.sum()

        c = _fft_convolve(a_full, a_full)
        g_full = _compute_grad(a_full, c, T)
        # Project gradient back to half
        g_half = 0.5 * (g_full[:half] + g_full[n - 1 : n - 1 - half : -1][::-1])

        W = float(np.dot(a_half_cur, g_half))
        z_grad = a_half_cur * (g_half - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(half)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_half_cur = _softmax_stable(z)
            a_full = np.zeros(n)
            a_full[:half] = a_half_cur
            a_full[n - 1 : n - 1 - half : -1] = a_half_cur
            a_full = a_full / a_full.sum()
            sc = _score_norm(a_full)
            if sc < best_score:
                best_score = sc
                best_a = a_full.copy()

    return best_a, best_score

            T_start=0.0005 * max_cf, T_end=0.000005 * max_cf,
            max_iter=200000, lr_start=0.002, lr_end=0.0002,
            momentum=0.95, eval_interval=50,
            staircase_steps=3, patience_limit=10
        )
        full_sc = _score(a_final)
        print(f"Phase 7 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score_full:
            best_score_full = full_sc
            best_a = a_final.copy()

        # Quick pair-swap at the very end
        if time.monotonic() < deadline - 5:
            a_cd_fin, _ = _pair_swap_cd(best_a, deadline - 3, max_rounds=30)
            sc_cd_fin = _score(a_cd_fin)
            if sc_cd_fin < best_score_full:
                best_score_full = sc_cd_fin
                best_a = a_cd_fin.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
