                       0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                       0.00005, 0.00002, 0.00001, 0.000004, 0.000001])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_iter_time = 0.002 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_iter_time * 20))))

        try:
            obj_fn = lambda z_local: _smooth_obj(z_local, beta, n)
            grad_fn = lambda z_local: _smooth_grad(z_local, beta, n)
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _proj_grad_optimize(a_init, deadline, max_iter=20000):
    """
    Projected gradient on simplex with adaptive active set tracking.
    Uses the quadratic structure of sum_k W_k c_k for efficient gradient.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    alpha = 0.5
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set with tightening tolerance
        tol_frac = max(0.0005, 0.02 * (0.97 ** (it // 50)))
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        if active.sum() > max(5, 2 * n // 10):
            thresh = np.sort(c)[::-1][min(active.sum(), max(5, 2 * n // 10)) - 1]
            active = c >= thresh

        n_act = int(active.sum())
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_act

        # Gradient of the weighted quadratic
        g = _hankel_matvec(a, W, n)
        g_centered = g - np.mean(g)
        g_norm = float(np.linalg.norm(g_centered))
        if g_norm < 1e-15 * n:
            break

        # Quadratic value for line search
        f_curr = float(np.dot(W, c))

        # Backtracking line search
        step_alpha = alpha
        improved = False
        for _ls in range(30):
            a_try = a - step_alpha * g_centered
            a_try = _proj_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(W, c_try))
            # Sufficient decrease condition
            if f_try <= f_curr - 1e-4 * step_alpha * g_norm * float(a @ g_centered):
                a = a_try
                improved = True
                break
            step_alpha *= 0.5

        if improved:
            no_improve = 0
            alpha = min(alpha * 1.02, 2.0)
        else:
            alpha *= 0.8
            no_improve += 1
            if no_improve > 100:
                break

        # Update best periodically
        if it % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()

    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _kkt_polish(a_init, deadline, max_rounds=200):
    """
    KKT-based polishing: iteratively identify the exact active set
    and solve the resulting convex QP subproblem to higher precision.
    Uses the KKT conditions to check and refine.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify tight active set (very small tolerance)
        tol = 1e-10 * c_max + 1e-16
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        # For the active set, solve: min sum_{k in A} c_k / |A| s.t. sum(a)=1, a>=0
        # using projected gradient with tight convergence
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_act

        converged = True
        for inner in range(200):
            g = _hankel_matvec(a, W, n)
            g_centered = g - np.mean(g)
            g_norm = float(np.linalg.norm(g_centered))
            if g_norm < 1e-14 * n:
                break

            # Exact line search for the quadratic (since f(a) is quadratic for fixed W)
            # f(a + t*d) = f(a) + t*(grad^T d) + t^2*(d^T H d)
            # where d = -g_centered, H is the Hessian of the quadratic
            d = -g_centered
            grad_d = float(g @ d)  # = -||g_centered||^2
            # d^T H d = d^T (2*M) d where M_{ij} = W_{i+j}
            Hd = _hankel_matvec(d, W, n)
            dHd = float(d @ Hd)
            if dHd <= 1e-30:
                alpha_ls = min(1e-4, -grad_d / (1e-30 + abs(grad_d)))
            else:
                alpha_ls = -grad_d / dHd
                alpha_ls = min(alpha_ls, 1e-3)

            a_try = a + alpha_ls * d
            a_try = _proj_simplex(a_try)

            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(W, c_try))
            f_curr = float(np.dot(W, c))

            if f_try < f_curr - 1e-18:
                a = a_try
            else:
                # Try smaller step
                a_try = a + 0.5 * alpha_ls * d
                a_try = _proj_simplex(a_try)
                c_try = _full_convolve(a_try, a_try)
                if float(np.dot(W, c_try)) < f_curr - 1e-18:
                    a = a_try
                else:
                    converged = False
                    break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if converged and g_norm < 1e-12 * n:
            break

    return best_a, best_sc


def _dual_ascent(a_init, deadline, max_iter=5000):
    """
    Dual ascent method: work with the dual variable (Lagrange multiplier for active set)
    to refine the solution. The dual problem for max_k c_k is related to finding
    optimal weights w on the convolution indices.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    # Initialize weights on active set
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 0.01 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / active.sum()

    lr = 0.1
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Primal update: solve min a^T H_w a s.t. sum(a)=1, a>=0
        # Using a few projected gradient steps
        for _inner in range(5):
            g = _hankel_matvec(a, w, n)
            g_c = g - np.mean(g)
            g_norm = float(np.linalg.norm(g_c))
            if g_norm < 1e-15:
                break
            step = min(0.1, 0.5 / max(np.linalg.norm(g_c, ord=np.inf), 1e-15))
            a_try = _proj_simplex(a - step * g_c)
            c_try = _full_convolve(a_try, a_try)
            if float(w @ c_try) < float(w @ _full_convolve(a, a)):
                a = a_try
            else:
                a_try = _proj_simplex(a - 0.5 * step * g_c)
                if float(w @ _full_convolve(a_try, a_try)) < float(w @ _full_convolve(a, a)):
                    a = a_try

        # Dual update: increase w_k for k where c_k is large
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Gradient of dual w.r.t. w: c_k - (w^T c)
        dual_grad = c - float(w @ c)
        # Scale
        w_new = w + lr * dual_grad / (c_max + 1e-15)
        w_new = np.maximum(w_new, 0)
        w_sum = w_new.sum()
        if w_sum < 1e-15:
            w = w.copy()
        else:
            w = w_new / w_sum

        # Adaptive learning rate
        if it % 100 == 0 and it > 0:
            lr *= 0.97

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _support_reduction(a, deadline, min_time_remaining=60.0):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_idx = np.argsort(a)

    fractions = [0.01, 0.03, 0.05, 0.08, 0.12, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() + min_time_remaining > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_a = a_try.copy()
            a_idx = np.argsort(best_a)
            # Re-optimize
            if time.monotonic() + min_time_remaining > deadline:
                break
            a_re = _lbfgs_optimize(best_a, deadline, min_time_remaining=min_time_remaining)
            sc_re = _score(a_re)
            if sc_re < best_sc:
                best_sc = sc_re
                best_a = a_re
                a_idx = np.argsort(best_a)
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.02
        return a / a.sum()
    elif kind == "u_shaped":
        t = np.linspace(0, 1, n)
        a = 0.3 + t ** 2 + (1 - t) ** 2
        return a / a.sum()
    elif kind == "flat_center":
        a = np.ones(n)
        center = n // 2
        width = n // 3
        a[center - width:center + width] = 1.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method search: L-BFGS + projected gradient + KKT polish +
    dual ascent + support reduction + multi-n + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"delta={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init_tapered")

    # === Phase 1: L-BFGS from parent (main workhorse) ===
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    a1 = _lbfgs_optimize(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_main")

    # === Phase 2: Projected gradient on simplex ===
    print(f"\n[Phase 2] Projected gradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        pg_deadline = min(deadline - 400, time.monotonic() + 150)
        a2, sc2 = _proj_grad_optimize(best_seq, pg_deadline, max_iter=15000)
        update_best(a2, sc2, "proj_grad")

    # === Phase 3: KKT polish ===
    print(f"\n[Phase 3] KKT polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        kkt_deadline = min(deadline - 350, time.monotonic() + 120)
        a3, sc3 = _kkt_polish(best_seq, kkt_deadline, max_rounds=100)
        update_best(a3, sc3, "kkt_1")

    # === Phase 4: Second L-BFGS pass (from improved point) ===
    print(f"\n[Phase 4] L-BFGS pass 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a4 = _lbfgs_optimize(best_seq, deadline, min_time_remaining=250.0)
        update_best(a4, _score(a4), "lbfgs_2")

    # === Phase 5: KKT polish again ===
    print(f"\n[Phase 5] KKT polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        kkt_deadline = min(deadline - 250, time.monotonic() + 80)
        a5, sc5 = _kkt_polish(best_seq, kkt_deadline, max_rounds=100)
        update_best(a5, sc5, "kkt_2")

    # === Phase 6: Projected gradient from current best ===
    print(f"\n[Phase 6] Proj grad 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        pg_deadline = min(deadline - 200, time.monotonic() + 80)
        a6, sc6 = _proj_grad_optimize(best_seq, pg_deadline, max_iter=10000)
        update_best(a6, sc6, "proj_grad_2")

    # === Phase 7: Support reduction ===
    print(f"\n[Phase 7] Support reduction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a7, sc7 = _support_reduction(best_seq, deadline, min_time_remaining=100.0)
        update_best(a7, sc7, "support_red")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[Phase 8] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384, 2048]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _lbfgs_optimize(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.001 and time.monotonic() < deadline - 50:
            a_kk, sc_kk = _kkt_polish(a_opt, min(deadline - 40, time.monotonic() + 30), max_rounds=50)
            update_best(a_kk, sc_kk, f"n_{n_try}_kkt")

    # === Phase 9: Diverse restarts ===
    print(f"\n[Phase 9] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            (0.02, "perturb_small"),
            (0.08, "perturb_med"),
            (0.2, "perturb_large"),
        ]
        for sigma, label in restart_configs:
            if time.monotonic() > deadline - 25:
                break
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _lbfgs_optimize(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")
            if sc_r < best_score + 0.0005 and time.monotonic() < deadline - 15:
                a_rk, sc_rk = _kkt_polish(a_r, min(deadline - 10, time.monotonic() + 15), max_rounds=50)
                update_best(a_rk, sc_rk, f"restart_{label}_kkt")

        # Different profile restart
        if time.monotonic() < deadline - 20:
            for profile_kind in ["u_shaped", "bimodal", "flat_center"]:
                if time.monotonic() > deadline - 15:
                    break
                p_start = _make_profile(len(best_seq), profile_kind)
                a_r = _lbfgs_optimize(p_start, deadline, min_time_remaining=10.0)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"profile_{profile_kind}")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # KKT with very tight tolerance
        a_fin, sc_fin = _kkt_polish(best_seq, deadline - 1, max_rounds=100)
        update_best(a_fin, sc_fin, "final_kkt")

        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 1e-8
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(2000, max(50, int(remaining / 0.01)))
                try:
                    obj_fn = lambda z_local: _smooth_obj(z_local, beta_fin, n)
                    grad_fn = lambda z_local: _smooth_grad(z_local, beta_fin, n)
                    result = minimize(
                        obj_fn, z, jac=grad_fn,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                 'maxcor': 80}
                    )
                    a_flb = _softmax(result.x)
                    update_best(a_flb, _score(a_flb), "final_lbfgs")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
