"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Adaptive active-set Newton with CG for the Hessian system
- Barzilai-Borwein subgradient as a robust pre-optimizer

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _compute_gradient_active(a, active_vec, n, m):
    g = (2.0 / m) * _corr_a_A(a, active_vec, n)
    return g


def _active_set_newton(a_init, deadline, max_iter=500, tol_active=0.001):

    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):

            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # CG solve for (H + mu*I) u = g_t
        x = np.zeros_like(a)

        p = r.copy()

            rs_new = np.dot(r_new, r_new)
            if np.sqrt(max(rs_new, 0)) < 1e-10 * (1.0 + np.linalg.norm(g_t)):
                cg_converged = True
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new

            rw_new = rw - alphaw * Apw

            if np.sqrt(max(rsw_new, 0)) < 1e-10 * (1.0 + np.linalg.norm(w_vec)):
                break
            pw = rw_new + (rsw_new / rsw) * pw
            rw = rw_new
            rsw = rsw_new
        w = xw

        sum_w = float(w.sum())
        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15

        else:

        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False

                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:

                continue

                if s_try < 1e-15:

                c_try = _full_convolve(a_try, a_try)

                    break

            else:

        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:

        # Barzilai-Borwein step size
        if s_prev is not None and g_prev is not None:
            gs = np.dot(g_prev, s_prev)
            gg = np.dot(g_prev, g_prev)
            ss = np.dot(s_prev, s_prev)

                    lr_bb = gs / ss
                else:
                    lr_bb = ss / gg
                if np.isfinite(lr_bb) and lr_bb > 0:
                    lr = np.clip(lr_bb, 1e-6, 1.0)
                else:
                    lr = max(1e-6, lr * 0.9)
            else:
                lr *= 0.95
        else:
            lr = lr_init / (1.0 + 0.0005 * i)

        s_new = -lr * grad
        z = z + s_new
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        s_prev = s_new
        g_prev = grad
        x_prev = z.copy()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()

            else:

                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))

                    patience = 0

def _run_lbfgs(a_init, deadline, min_time_remaining=100.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.05)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                  0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                                  0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.008 * max(1, n / 4096.0)
        max_iter = min(6000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 60,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(current_a, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        d = np.abs(t - 0.5) * 2.0
        a = 1.0 + 0.5 * d ** 2
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # Phase 2: Active-set Newton (multiple tolerance levels)
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        for tol_level in [0.002, 0.001, 0.0005, 0.0002]:
            if time.monotonic() > deadline - 350:
                break
            as_deadline = min(deadline - 350, time.monotonic() + 180)
            a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=400, tol_active=tol_level)
            update_best(a_as, sc_as, f"as_newton_{tol_level}")

    # Phase 3: Subgradient with BB steps
    print(f"\n[Phase 3] Subgradient BB (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_bb(best_seq, sg_deadline, n_max_steps=10000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_bb")

        # Follow up with Newton
        if sc_sg <= best_score + 0.001 and time.monotonic() < deadline - 250:
            a_sg2, sc_sg2 = _active_set_newton(a_sg, min(deadline - 250, time.monotonic() + 120),
                                                max_iter=300, tol_active=0.001)
            update_best(a_sg2, sc_sg2, "sg_bb_newton")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue

        if parent is not None and n_parent != n_try:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _make_profile(n_try, "uniform")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        if time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 50, time.monotonic() + 80),
                                                  max_iter=200, tol_active=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restarts = [
            ("symm", lambda: _symmetrize(best_seq)),
            ("perturb_005", lambda: _perturb(best_seq, 0.05, rng)),
            ("perturb_010", lambda: _perturb(best_seq, 0.10, rng)),
            ("perturb_020", lambda: _perturb(best_seq, 0.20, rng)),
        ]
        for label, gen_fn in restarts:
            if time.monotonic() > deadline - 25:
                break
            p_start = gen_fn()
            sc_start = _score(p_start)
            update_best(p_start, sc_start, f"restart_{label}_init")

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}_lbfgs")

            if sc_r <= best_score + 0.002 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 10, time.monotonic() + 40),
                                                  max_iter=150, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # Phase 6: Different profile initializations
    print(f"\n[Phase 6] Profile exploration (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_exp = n_parent if n_parent <= 16384 else 8192
        profiles = ["tapered", "edge_heavy", "cos2", "super_tapered"]
        for prof in profiles:
            if time.monotonic() > deadline - 25:
                break
            a_prof = _make_profile(n_exp, prof)
            sc_prof = _score(a_prof)
            update_best(a_prof, sc_prof, f"profile_{prof}")

            if sc_prof < best_score + 0.05:
                a_prof_opt = _run_lbfgs(a_prof, deadline, min_time_remaining=15.0)
                sc_prof_opt = _score(a_prof_opt)
                update_best(a_prof_opt, sc_prof_opt, f"profile_{prof}_opt")

    # Phase 7: Final aggressive polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Multiple Newton rounds with decreasing tolerance
        for tol in [0.001, 0.0005, 0.0002, 0.0001]:
            if time.monotonic() > deadline - 3:
                break
            as_dead = min(deadline - 2, time.monotonic() + 100)
            a_fin, sc_fin = _active_set_newton(best_seq, as_dead, max_iter=300, tol_active=tol)
            update_best(a_fin, sc_fin, f"final_as_{tol}")

        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 5:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00005
            remaining = deadline - 5 - time.monotonic()
            if remaining > 5:
                max_iter = min(5000, max(200, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _lsobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _lsgrad(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 60}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_tiny")
                except Exception:
                    pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
