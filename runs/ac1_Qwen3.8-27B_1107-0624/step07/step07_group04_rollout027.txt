Improved minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. Nesterov-accelerated projected subgradient with symmetry exploitation
2. Active-set CG-Newton with improved tolerances
3. Log-sum-exp L-BFGS with adaptive beta continuation

5. Adaptive shrinkage and dimension reduction

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    i = rho[-1]
    theta = (cssv[i] - 1) / (i + 1)
    return np.maximum(v - theta, 0)


def _make_symmetric(a):
    return (a + a[::-1]) / 2.0


def _hankel_matvec(v, W, n):
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve_hankel(W, n, rhs, max_iter=500, tol=1e-14):
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        if not np.isfinite(alpha):
            break
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x


def _grad_convolution(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad = _grad_convolution(a, w)
    return grad - np.mean(grad) * a * 0  # just return the gradient (softmax handles constraint)

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new

            sc_reopt = _score(a_reopt)
            if sc_reopt < best_sc:
                best_sc = sc_reopt
                best_a = a_reopt.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng, use_symmetry=True):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    if use_symmetry:
        z = _make_symmetric(z)
    z = z - z.mean()
    return _softmax(z)


def _final_polish(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    # Tight Newton
    a_np, sc_np = _newton_active_polish(best_a, min(deadline, time.monotonic() + 120), max_rounds=200)
    if sc_np < best_sc:
        best_sc = sc_np
        best_a = a_np.copy()

    # Ultra-fine L-BFGS
    if time.monotonic() < deadline - 10:
        c_best = _full_convolve(best_a, best_a)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_a, 1e-30))
        z = z - z.mean()
        beta_fin = mb * 5e-8
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            max_iter = min(2000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_local: _smoobj(z_local, beta_fin, n),
                    z,
                    jac=lambda z_local: _smograd(z_local, beta_fin, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                             'maxcor': 60, 'maxls': 200}
                )
                a_flb = _softmax(result.x)
                a_flb = _make_symmetric(a_flb)
                a_flb = np.maximum(a_flb, 0)
                s_flb = a_flb.sum()
                if s_flb > 1e-15:
                    a_flb = a_flb / s_flb
                sc_flb = _score(a_flb)
                if sc_flb < best_sc:
                    best_sc = sc_flb
                    best_a = a_flb.copy()
                    print(f"  Ultra-fine L-BFGS: {sc_flb:.10f}")
            except Exception:
                pass

    # One more Newton pass
    if time.monotonic() < deadline - 2:
        a_np2, sc_np2 = _newton_active_polish(best_a, deadline - 1, max_rounds=100)
        if sc_np2 < best_sc:
            best_sc = sc_np2
            best_a = a_np2.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_tapered(n_parent, 0.5)
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    n = len(parent)
    print(f"\n[Phase 1] Nesterov accelerated subgradient (t={time.monotonic() - t_start:.1f}s)")
    t1_deadline = min(deadline - 600.0, t_start + 300.0)
    a1, sc1 = _nesterov_subgrad(parent, t1_deadline, use_symmetry=True)
    update_best(a1, sc1, "nesterov1")

    print(f"\n[Phase 2] L-BFGS on symmetrized (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2 = _run_lbfgs_sym(best_seq, deadline, min_time_remaining=500.0)
        update_best(a2, _score(a2), "lbfgs_sym")

    print(f"\n[Phase 3] Newton active set polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3, sc3 = _newton_active_polish(best_seq, min(deadline - 400, time.monotonic() + 200), max_rounds=80)
        update_best(a3, sc3, "newton1")

    print(f"\n[Phase 4] Nesterov round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a4, sc4 = _nesterov_subgrad(best_seq, min(deadline - 350, time.monotonic() + 150), use_symmetry=True)
        update_best(a4, sc4, "nesterov2")

    print(f"\n[Phase 5] L-BFGS round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a5 = _run_lbfgs_sym(best_seq, deadline, min_time_remaining=300.0)
        update_best(a5, _score(a5), "lbfgs_sym2")

    print(f"\n[Phase 6] Newton active set polish 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a6, sc6 = _newton_active_polish(best_seq, min(deadline - 250, time.monotonic() + 150), max_rounds=100)
        update_best(a6, sc6, "newton2")

    print(f"\n[Phase 7] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a7, sc7 = _shrinkage_and_reopt(best_seq, deadline, min_time_remaining=150.0)
        update_best(a7, sc7, "shrinkage")

    print(f"\n[Phase 8] Multi-start diversity (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        initializations = [
            ("blackman", _make_blackman_harris(n)),
            ("tapered1", _make_tapered(n, 0.3)),
            ("tapered3", _make_tapered(n, 1.5)),
            ("periodic", _make_periodic(n, 3)),
            ("flat_edges", _make_flat_edges(n)),
        ]
        for name, init in initializations:
            if time.monotonic() > deadline - 100:
                break
            init_sym = _make_symmetric(init)
            init_sym = np.maximum(init_sym, 0)
            s_init = init_sym.sum()
            init_sym = init_sym / s_init

            # Quick Nesterov warmup
            t_ns = min(deadline - 100, time.monotonic() + 40)
            a_ns, _ = _nesterov_subgrad(init_sym, t_ns, use_symmetry=True)

            # L-BFGS polish
            if time.monotonic() < deadline - 80:
                a_ms = _run_lbfgs_sym(a_ns, deadline, min_time_remaining=80.0)
                sc_ms = _score(a_ms)
                update_best(a_ms, sc_ms, f"multi_{name}")

    print(f"\n[Phase 9] Perturbation restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        for i in range(5):
            if time.monotonic() > deadline - 50:
                break
            sigma = [0.01, 0.03, 0.05, 0.1, 0.2][i]
            p_start = _perturb(best_seq, sigma, rng, use_symmetry=True)

            if time.monotonic() > deadline - 40:
                break
            a_r = _run_lbfgs_sym(p_start, deadline, min_time_remaining=40.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"perturb_{i}")

            if sc_r < best_score + 0.001 and time.monotonic() < deadline - 25:
                a_rp, sc_rp = _newton_active_polish(a_r, min(deadline - 20, time.monotonic() + 30), max_rounds=50)
                update_best(a_rp, sc_rp, f"perturb_{i}_np")

    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _final_polish(best_seq, deadline - 2)
        update_best(a_fin, sc_fin, "final")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(4096, 0.5)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_tapered(8192, 0.5)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
