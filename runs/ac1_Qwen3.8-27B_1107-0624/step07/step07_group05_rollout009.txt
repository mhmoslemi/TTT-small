import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_obj(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    topk_sum = float(c_sorted[:k].sum())
    return 2.0 * n * topk_sum / k


def _topk_grad(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    topk_idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[topk_idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    s = e.sum()

def _pressure_equalize(a_init, deadline, min_remaining=5.0, n_iters=500, gamma_init=0.5):

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    gamma = gamma_init
    for it in range(n_iters):
        if time.monotonic() + min_remaining > deadline:
            break
        # Anneal gamma
        gamma = gamma_init * (0.995 ** it)
        gamma = max(gamma, 0.05)

        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        # Adaptive active set
        tol_frac = max(0.001, 0.02 * (0.98 ** it))
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        if n_active < 2:
            idx = np.argsort(c)[-max(2, n // 1000):]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = len(idx)

        # Compute pressure
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]
        p = np.maximum(p, 1e-30)

        # Update
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 50 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

        if time.monotonic() + min_remaining > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        tol = 0.03 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        n_try = min(n, max(10, n // 80))
        top_idx = np.argsort(contrib)[-n_try:]
        top_idx = rng.permutation(top_idx[:max(5, n_try // 4)])

        improved = False
        for i in top_idx:
            if time.monotonic() + min_remaining > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            c_old = c_max

            for factor in [0.6, 0.8, 0.9, 1.1, 1.4, 1.7]:
                a[i] = old_val * factor
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-16:
                    improved = True
                    c_old = c_try
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 5 * t * t) + 1.5
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * t) + 0.15 * np.cos(2 * np.pi * 19 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.3) ** 2) + np.exp(-4.0 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=6, amplitude=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def _multi_optimize(a_init, deadline, min_remaining=20.0):
    """Full pipeline: pressure equalize -> topk L-BFGS -> accelerated MD -> smo L-BFGS."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: Pressure equalization (quick)
    if time.monotonic() < deadline - min_remaining - 30:
        a1, sc1 = _pressure_equalize(current_a, min(deadline - min_remaining - 20, time.monotonic() + 80),
                                     min_remaining=5, n_iters=200, gamma_init=0.4)
        sc1_full = _score(a1)
        if sc1_full < current_sc:
            current_a = a1
            current_sc = sc1_full

    # Stage 2: Top-k L-BFGS
    if time.monotonic() < deadline - min_remaining - 15:
        a2 = _run_topk_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 15)
        sc2 = _score(a2)
        if sc2 < current_sc:
            current_a = a2
            current_sc = sc2

    # Stage 3: Accelerated mirror descent
    if time.monotonic() < deadline - min_remaining - 10:
        a3, sc3 = _accelerated_md(current_a, min(deadline - min_remaining - 5, time.monotonic() + 100),
                                  min_remaining=3, n_steps=3000)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: SMO L-BFGS (fine polishing)
    if time.monotonic() < deadline - min_remaining:
        a4 = _run_smo_lbfgs(current_a, deadline, min_time_remaining=min_remaining)
        sc4 = _score(a4)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search combining pressure equalization, top-k L-BFGS,
    accelerated mirror descent, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_parent = _score(parent)
                print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
                _update(parent, sc_parent)
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent found, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    t_phase1_end = t_start + 400
    a_main, sc_main = _multi_optimize(parent, min(deadline, t_phase1_end), min_remaining=60)
    _update(a_main, sc_main)

    # Extended pressure equalization on best
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_pe, sc_pe = _pressure_equalize(best_seq, min(deadline - 250, time.monotonic() + 150),
                                         min_remaining=10, n_iters=400, gamma_init=0.3)
        sc_pe_full = _score(a_pe)
        _update(a_pe, sc_pe_full)
        if sc_pe_full < best_score:
            a_follow = _run_topk_lbfgs(a_pe, deadline, min_time_remaining=250)
            _update(a_follow, _score(a_follow))

    # --- Phase 2: Multi-n exploration ---
    print(f"\n[Phase 2] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 6144, 12288, 16384]
    n_candidates = [x for x in n_candidates if x != n_parent]
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        remaining_for_this = deadline - 150 - time.monotonic()
        if remaining_for_this < 40:
            break

        print(f"  Trying n={n_try}, remaining={remaining_for_this:.0f}s")
        a_init = _resample(parent, n_try)
        # Add slight perturbation for diversity
        a_init = _fourier_perturb(a_init, rng, n_modes=3, amplitude=0.01)

        min_rem = max(20, remaining_for_this * 0.25)
        a_opt, sc_opt = _multi_optimize(a_init, deadline, min_remaining=min_rem)
        _update(a_opt, sc_opt)

    # --- Phase 3: Diverse restarts ---
    print(f"\n[Phase 3] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        profiles = ["tapered", "chirp", "double_cos", "gauss_bump"]

        for i, prof in enumerate(profiles):
            if time.monotonic() > deadline - 30:
                break

            a_blend = 0.6 * best_seq + 0.4 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            a_pert = _fourier_perturb(a_blend, rng, n_modes=5, amplitude=0.02)

            remaining = deadline - 30 - time.monotonic()
            if remaining < 30:
                break
            min_rem = max(10, remaining * 0.3)
            a_r, sc_r = _multi_optimize(a_pert, deadline, min_remaining=min_rem)
            _update(a_r, sc_r)

    # --- Phase 4: Final intensive polish on best ---
    print(f"\n[Phase 4] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)

        # Coordinate descent
        a_cd, sc_cd = _coordinate_polish(best_seq, deadline, min_remaining=5, n_rounds=100)
        _update(a_cd, sc_cd)

        # More pressure equalization with very small gamma
        if time.monotonic() < deadline - 5:
            a_pe2, sc_pe2 = _pressure_equalize(best_seq, deadline, min_remaining=2,
                                               n_iters=500, gamma_init=0.15)
            _update(a_pe2, sc_pe2)

        # Final top-k L-BFGS with k=1
        if time.monotonic() < deadline - 2:
            a_fin = _run_topk_lbfgs(best_seq, deadline, min_time_remaining=1)
            _update(a_fin, _score(a_fin))

        # Final accelerated MD
        if time.monotonic() < deadline - 1:
            a_md, sc_md = _accelerated_md(best_seq, deadline, min_remaining=0.5, n_steps=1000)
            _update(a_md, sc_md)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
