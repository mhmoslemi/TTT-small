```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
```

```python
def _gs_iterate(a_init, n_iters=200, deadline=None):
    """Gerchberg-Saxton style iteration: alternate non-negativity and flat spectrum."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N)
        phase = np.angle(A)
        A_flat = np.ones_like(A) * np.exp(1j * phase)
        # Ensure DC component is positive
        A_flat[0] = abs(A_flat[0])
        a_new = np.fft.irfft(A_flat, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _chirp_init(n, rng):

    a = a * (0.5 + 0.5 * rng.random(n))
    s = a.sum()
    if s < 1e-15:

    return a / s


def _random_phase_init(n, rng):

def _active_set_newton(a_init, deadline, max_iter=300, tol_active=0.001):
    """Damped Newton method on the active set of the max-convolution problem."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

        if time.monotonic() > deadline:
            break

        c_max = float(c.max())
        tol = tol_active * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        if n_active > 3 * n:
            top_idx = np.argsort(c)[::-1][:3 * n]
            active = np.zeros_like(c, dtype=bool)
            active[top_idx] = True
            n_active = 3 * n

        m = n_active

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        r = g_t - hess_matvec(x)

            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r_new = r - alpha * Ap
            rs_new = np.dot(r_new, r_new)
            if np.sqrt(rs_new) < 1e-12 * (1.0 + np.linalg.norm(g_t)):
                break
            p = r_new + (rs_new / rs) * p
            r = r_new
            rs = rs_new
        u = x

        # Solve for the normalization multiplier
        w_h = np.zeros(n)
        r2 = np.ones(n) - hess_matvec(np.zeros(n))
        p2 = r2.copy()
        rs2 = np.dot(r2, r2)
        for cg_it in range(200):
            Ap2 = hess_matvec(p2)
            pAp2 = np.dot(p2, Ap2)
            if pAp2 < 1e-30:
                break
            alpha2 = rs2 / pAp2
            w_h += alpha2 * p2
            r_new2 = r2 - alpha2 * Ap2
            rs_new2 = np.dot(r_new2, r_new2)
            if np.sqrt(rs_new2) < 1e-12:
                break
            p2 = r_new2 + (rs_new2 / rs2) * p2
            r2 = r_new2
            rs2 = rs_new2
        w = w_h
        sum_w = float(w.sum())

        if abs(sum_w) < 1e-15:
            dx = -u
            dx = dx - np.mean(dx)
        else:
            lam = float(u.sum()) / sum_w
            dx = -u + lam * w
            dx = dx - np.mean(dx)

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = c_max
        accepted = False
        t = min(1.0, t_max)
        for ls in range(40):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 or ls == 39:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            t = min(0.1, t_max)
            for ls in range(20):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13:
                    a = a_try
                    break
                t *= 0.5
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation on log-softmax coordinates."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.008, 0.004, 0.002, 0.001, 0.0005,
                                  0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 2048.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 10))))

        def obj_func(z_local):
            return _lsobj(z_local, beta, n)

        def grad_func(z_local):
            return _lsgrad(z_local, beta, n)

        try:
            result = minimize(
                obj_func,
                z,
                jac=grad_func,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient method with active-set tracking in log-softmax coordinates."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        flip_a = a[::-1]
        wc = _full_convolve(w, flip_a)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.87
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining spectral initialization, GS iterations,
    L-BFGS with log-sum-exp, and active-set Newton refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _tapered_init(n_parent, "sin")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # ============================================================
    # Phase 1: Diverse initialization with GS refinement
    # ============================================================
    print(f"\n[Phase 1] Diverse initialization + GS (t={time.monotonic() - t_start:.1f}s)")

    n_main = n_parent if n_parent > 0 else 8192
    candidates = []

    # Start from parent
    candidates.append(("parent", parent.copy()))

    # Chirp-based
    for i in range(3):
        a_chirp = _chirp_init(n_main, rng)
        candidates.append((f"chirp_{i}", a_chirp))

    # Random phase
    for i in range(3):
        a_rp = _random_phase_init(n_main, rng)
        candidates.append((f"randphase_{i}", a_rp))

    # Multi-sinusoid
    for i in range(3):
        n_sines = rng.integers(4, 16)
        a_ms = _multi_sin_init(n_main, rng, n_sines=int(n_sines))
        candidates.append((f"msin_{i}", a_ms))

    # Tapered variants
    for kind in ["sin", "cos", "tri"]:
        a_t = _tapered_init(n_main, kind)
        candidates.append((f"taper_{kind}", a_t))

    # Random
    for i in range(2):
        a_r = rng.random(n_main) + 0.01
        a_r = a_r / a_r.sum()
        candidates.append((f"random_{i}", a_r))

    # Run GS on each candidate (fast pre-optimizer)
    gs_time_budget = min(120.0, (deadline - t_start) * 0.1)
    gs_deadline = t_start + gs_time_budget
    gs_per_candidate = max(20, int(n_iters := 200) // max(1, len(candidates)))

    for label, a_cand in candidates:
        if time.monotonic() > gs_deadline:
            break
        a_gs = _gs_iterate(a_cand, n_iters=gs_per_candidate, deadline=gs_deadline)
        sc_gs = _score(a_gs)
        update_best(a_gs, sc_gs, f"gs_{label}")

    # ============================================================
    # Phase 2: L-BFGS refinement on top candidates
    # ============================================================
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")

    # Refine the best few from Phase 1
    if best_seq is not None:
        a_l = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0)
        update_best(a_l, _score(a_l), "lbfgs_main")

    # Also try refining a different candidate for diversity
    for label, a_cand in candidates[:4]:
        if time.monotonic() > deadline - 250:
            break
        a_gs = _gs_iterate(a_cand, n_iters=100, deadline=deadline - 250)
        a_l = _run_lbfgs(a_gs, deadline, min_time_remaining=200.0)
        update_best(a_l, _score(a_l), f"lbfgs_{label}")

    # ============================================================
    # Phase 3: Active-set Newton refinement
    # ============================================================
    print(f"\n[Phase 3] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")

    if best_seq is not None and time.monotonic() < deadline - 200:
        as_deadline = min(deadline - 200, time.monotonic() + 100)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=200, tol_active=0.001)
        update_best(a_as, sc_as, "as_newton_1")

        if time.monotonic() < deadline - 180:
            as_deadline2 = min(deadline - 180, time.monotonic() + 60)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_deadline2, max_iter=150, tol_active=0.0005)
            update_best(a_as2, sc_as2, "as_newton_tight")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 150:
        sg_deadline = min(deadline - 150, time.monotonic() + 80)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=3000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_polish")

    # ============================================================
    # Phase 4: Multi-n search
    # ============================================================
    print(f"\n[Phase 4] Multi-n search (t={time.monotonic() - t_start:.1f}s)")

    n_candidates = [4096, 8192, 16384]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    # Sort by proximity to n_parent first (already done ones first)
    n_candidates_sorted = sorted(set(n_candidates), key=lambda x: abs(x - n_parent))

    for n_try in n_candidates_sorted:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_parent:
            continue

        if parent is not None:
            a_init = _interpolate_to_n(parent, n_try)
        else:
            a_init = _tapered_init(n_try, "sin")

        # Quick GS
        a_init_gs = _gs_iterate(a_init, n_iters=50, deadline=deadline - 80)
        a_opt = _run_lbfgs(a_init_gs, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 50:
            a_opt2, sc_opt2 = _active_set_newton(a_opt, min(deadline - 40, time.monotonic() + 40),
                                                  max_iter=80, tol_active=0.001)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # ============================================================
    # Phase 5: Diverse restarts from best
    # ============================================================
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")

    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts_info = [
            ("perturb_01", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_05", lambda: _perturb(best_seq, 0.05, rng)),
            ("perturb_10", lambda: _perturb(best_seq, 0.10, rng)),
            ("fourier_3", lambda: _perturb_fourier(best_seq, rng, n_modes=3)),
            ("fourier_7", lambda: _perturb_fourier(best_seq, rng, n_modes=7)),
            ("gs_restart", None),  # Special: GS from best
        ]

        for label, gen_fn in restarts_info:
            if time.monotonic() > deadline - 25:
                break
            if gen_fn is not None:
                p_start = gen_fn()
            else:
                p_start = _gs_iterate(best_seq, n_iters=100, deadline=deadline - 60)
            sc_start = _score(p_start)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{label}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _active_set_newton(a_r, min(deadline - 10, time.monotonic() + 25),
                                                  max_iter=60, tol_active=0.001)
                update_best(a_r2, sc_r2, f"restart_{label}_as")

    # ============================================================
    # Phase 6: Final intensive polish
    # ============================================================
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")

    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very tight active-set Newton
        as_deadline_final = min(deadline - 2, time.monotonic() + 60)
        a_fin, sc_fin = _active_set_newton(best_seq, as_deadline_final, max_iter=200, tol_active=0.0005)
        update_best(a_fin, sc_fin, "final_as_tight")

        # One more subgradient pass
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 3, n_max_steps=2000, lr_init=0.01)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 5:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00002
            remaining = deadline - 6 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(100, int(remaining / 0.03)))

                def obj_final(z_local):
                    return _lsobj(z_local, beta_final, n)

                def grad_final(z_local):
                    return _lsgrad(z_local, beta_final, n)

                try:
                    result = minimize(
                        obj_final,
                        z,
                        jac=grad_final,
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_lbfgs_minibeta")
                except Exception:
                    pass

        # Final active-set Newton after all polishing
        if time.monotonic() < deadline - 2:
            a_final_as, sc_final_as = _active_set_newton(best_seq, deadline - 1,
                                                         max_iter=100, tol_active=0.0003)
            update_best(a_final_as, sc_final_as, "very_final_as")

    # Final validation and return
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
