    """
    Multiplicative weight update: a[i] <- a[i] * (1 - lr * grad[i] / max_grad)
    where grad[i] = 2 * sum_k w[k] * a[k-i]

    grad = _correlation_gradient(a, w)
    max_grad = np.max(np.abs(grad))
    if max_grad < 1e-30:

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:

    return a_new / s

    a = _norm1(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    # Geometric schedule for beta
    n_betas = 20
    log_beta_start = np.log(beta_start * max(_conv(a).max(), 1e-10))
    log_beta_end = np.log(beta_end * max(_conv(a).max(), 1e-10))
    log_betas = np.linspace(log_beta_start, log_beta_end, n_betas)
    
    iters_per_beta = max(10, max_iter // n_betas)
    lr = lr_init
    total_iter = 0
    
    for bi, log_beta in enumerate(log_betas):
        if time.monotonic() > deadline - reserve:
            break
        beta = np.exp(log_beta)
        
        for it in range(iters_per_beta):
            if time.monotonic() > deadline - reserve:
                break
            
            c = _conv(a)
            c_max = c.max()
            
            # Soft max weights
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            w = e / e.sum()
            
            # Multiplicative update
            a_new = _multiplicative_update(a, w, lr, deadline)
            if np.allclose(a_new, a):
                break
            
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new
            total_iter += 1
            
            # Adaptive lr
            if it % 100 == 0:
                if sc < best_sc + 1e-10:
                    lr *= 1.01
                else:
                    lr *= 0.95
                lr = np.clip(lr, 0.001, 0.5)
    
    return best_a, best_sc

def _lp_dual_direction(a, M, deadline, max_iter=5000, reserve=30.0):
    """
    Solve the dual LP approximately using projected gradient.
    Dual: min M * sum(y) s.t. (A^T y)_i >= 1, y >= 0
    Returns the primal solution g (zero-padded to 2n-1 for the update).
    """
    n = len(a)
    # (A^T y)_i = sum_k y[k] * a[k-i] (same as correlation)
    # We want to find y >= 0 such that corr(y, a)[i] >= 1 for all i.
    # Minimize M * sum(y).
    
    # Initialize y
    y = np.ones(2*n - 1) / (2*n - 1) * (1.0 / (max(a.max(), 1e-10)))
    # Scale so that corr(y, a) is roughly 1
    
    lr = 1.0 / (M * max(a.max()**2 * (2*n-1), 1e-10))
    best_y = y.copy()
    best_obj = M * y.sum()
    
    for it in range(max_iter):
        if time.monotonic() > deadline - reserve:
            break
        
        # Compute constraint violation: v[i] = 1 - corr(y, a)[i]
        corr_val = _correlation_gradient(y, a) / 2.0  # corr(y,a)[i] = sum_k y[k]*a[k-i]
        # Wait, _correlation_gradient(y, a) computes 2*sum_k a[k]*y[k-i]... 
        # Let me recompute directly.
        # We need: v[i] = 1 - sum_k y[k] * a[k-i]
        # sum_k y[k] * a[k-i] is the cross-correlation (y ⋆ a)[i]
        N = _next_pow2(4*n)
        Y = np.fft.rfft(y, N)
        A = np.fft.rfft(a, N)
        corr_val = np.fft.irfft(Y * np.conj(A), N)[:n]
        v = 1.0 - corr_val  # violation (positive means constraint not met)
        
        # Projected gradient step on y:
        # Objective: M * sum(y), gradient: M * ones
        # Constraint: corr_val >= 1, i.e., -corr_val <= -1
        # Lagrangian: M*sum(y) + sum_i μ_i * (1 - corr_val[i])
        # ∂/∂y[k] = M - sum_i μ_i * ∂corr_val[i]/∂y[k]
        # ∂corr_val[i]/∂y[k] = a[k-i] (from the definition corr_val[i] = sum_k y[k]*a[k-i])
        # So ∂/∂y[k] = M - sum_i μ_i * a[k-i]
        # At optimum: y[k] = 0 => M - sum_i μ_i * a[k-i] >= 0
        #             y[k] > 0 => M - sum_i μ_i * a[k-i] = 0
        
        # For a simple method: use the violation as surrogate for μ
        # y[k] <- max(0, y[k] - lr * (M - sum_i v[i]_+ * a[k-i]))
        # Where v[i]_+ = max(v[i], 0)
        
        v_pos = np.maximum(v, 0.0)
        # Compute sum_i v_pos[i] * a[k-i] for each k
        # This is the cross-correlation of v_pos (padded to 2n-1) with a
        v_pos_padded = np.zeros(2*n - 1)
        v_pos_padded[:n] = v_pos
        N2 = _next_pow2(4*n)
        V = np.fft.rfft(v_pos_padded, N2)
        A2 = np.fft.rfft(a, N2)
        grad_y = M * np.ones(2*n-1) - np.fft.irfft(V * np.conj(A2), N2)[:2*n-1]
        
        y = np.maximum(y - lr * grad_y, 0.0)
        
        obj = M * y.sum()
        if obj < best_obj:
            best_obj = obj
            best_y = y.copy()
        
        # Adaptive lr
        if it % 100 == 0:
            cv = np.max(v)
            if cv < 0.01:
                break
            lr *= 0.98 if cv > 0.5 else 1.02
    
    return best_y

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    """L-BFGS on smoothed objective."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()
    
    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _adaptive_lbfgs(a_start, deadline, reserve_time=60.0):
    """Adaptive L-BFGS with decreasing beta."""
    n = len(a_start)
    c_cur = _conv(a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])
    
    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 10:
            break
        
        scale = max(1.0, (n / 4096.0) ** 1.3)
        max_iter = min(8000, max(200, int(remaining / (0.003 * scale * 15))))
        
        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
        
        if current_sc < 1.5010:
            break
    
    return current_a

def _active_equalize(a_init, deadline, k_active=8, max_iter=2000, reserve=30.0):
    """
    Active set equalization: try to equalize the top-k convolution values.
    Uses a Newton-like approach with the constraint sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = _norm1(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline - reserve:
            break
        
        c = _conv(a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]
        
        # Compute the gradient vectors for each active lag
        # grad_i[k] = 2*a[k-i] for the i-th variable
        # We want to find da such that:
        #   c[k_j] + grad[k_j] @ da ≈ t for all j (active lags)
        #   sum(da) = 0 (maintain normalization)
        
        # Build the linear system
        # For each active lag k_j, the gradient is 2*a[k_j - i] for i=0,...,n-1
        # (with a[j]=0 for j outside [0,n-1])
        
        # This is an n-dimensional system with k_active+1 equations.
        # Use least squares with a regularization.
        
        G = np.zeros((k_active, n))
        for j, k in enumerate(idx):
            # G[j, i] = 2 * a[k - i] (with a[j]=0 outside [0,n-1])
            for i in range(n):
                val = k - i
                if 0 <= val < n:
                    G[j, i] = 2.0 * a[val]
        
        # Target: equalize all active lags to their mean
        target = c[idx].mean()
        b_vec = target - c[idx]
        # Add normalization constraint
        G = np.vstack([G, np.ones(n)])
        b_vec = np.append(b_vec, 0.0)
        
        # Solve least squares (regularized)
        reg = 1e-6
        da = np.linalg.lstsq(G + reg * np.eye(len(G)), b_vec, rcond=None)[0]
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(30):
            a_try = a + alpha * da
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try a different approach: reduce the max lag directly
            max_idx = int(np.argmax(c))
            # Decrease a at positions that contribute to c[max_idx]
            contrib = np.zeros(n)
            for i in range(n):
                val = max_idx - i
                if 0 <= val < n:
                    contrib[i] = a[val]
            norm_c = np.linalg.norm(contrib)
            if norm_c > 1e-30:
                a_try = a - 0.01 * contrib / norm_c
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    if _score(a_try) < best_sc:
                        best_sc = _score(a_try)
                        best_a = a_try.copy()
                        a = a_try
        
        # Check convergence
        spread = c[idx].max() - c[idx].min()
        if spread < 1e-10 * c_max:
            break
    
    return best_a, best_sc

def _subgrad_refine(a_init, deadline, k_active=5, max_steps=20000, lr_init=0.05, reserve=30.0):
    """Subgradient method with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline - reserve:
            break
        
        a = _softmax(z)
        c = _conv(a)
        c_max = c.max()
        
        idx = np.argsort(c)[::-1][:k_active]
        w = np.zeros(2*n - 1)
        w[idx] = 1.0 / len(idx)
        
        grad = _correlation_gradient(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Update in log-space
        z = z - lr * grad / (a * n + 1e-30)
        z = z - z.mean()
        z = np.clip(z, -50.0, 50.0)
        
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
            else:
                stall += 1
                if stall > 15:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
                else:
                    lr *= 0.92
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _basin_hop(a_current, deadline, rng, n_hops=5, reserve=60.0):
    """Basin hopping for diversification."""
    n = len(a_current)
    best_a = a_current.copy()
    best_sc = _score(a_current)
    time_per_hop = max(20.0, (deadline - reserve - time.monotonic()) / n_hops)
    
    for h in range(n_hops):
        if time.monotonic() > deadline - reserve:
            break
        hop_deadline = time.monotonic() + time_per_hop * 0.8
        
        # Create perturbation
        pert_type = h % 4
        if pert_type == 0:
            sigma = 0.03 + 0.02 * (h % 3)
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 1:
            # Fourier perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            for _ in range(1 + h % 3):
                freq = int(rng.integers(1, max(2, n // 32)))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2.0 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            z = z - z.mean()
            a_pert = _softmax(z)
        elif pert_type == 2:
            # Block perturbation
            z = np.log(np.maximum(best_a, 1e-18))
            n_blocks = 2 + h % 4
            block_sz = n // n_blocks
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_sz)))
                end = min(n, start + block_sz)
                z[start:end] += rng.uniform(-0.03, 0.03)
            z = z - z.mean()
            a_pert = _softmax(z)
        else:
            # Swap-based perturbation
            a_pert = best_a.copy()
            n_swaps = 10 + 5 * h
            for _ in range(n_swaps):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                a_pert[i], a_pert[j] = a_pert[j], a_pert[i]
            a_pert = _norm1(a_pert)
        
        # Quick optimization from perturbation
        if time.monotonic() < hop_deadline:
            c_p = _conv(a_pert)
            mc = c_p.max()
            beta = mc * 0.01
            max_iter = min(2000, max(100, int((hop_deadline - time.monotonic()) / 0.005)))
            a_opt = _lbfgs_smooth(a_pert, beta, n, max_iter)
            sc_opt = _score(a_opt)
            
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
                print(f"    Basin hop {h}: new best {best_sc:.10f}")
    
    return best_a, best_sc
