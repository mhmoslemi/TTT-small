"""
Active-Set L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous approach:
1. Active-set L-BFGS: fixes the active set of near-max lags, then runs full L-BFGS
   on the sum of those lags (faster convergence than gradient descent).
2. Bisection line search for exact 1D optimization in active-set steps.
3. Adaptive active-set tolerance based on the spectral gap.
4. Multi-n exploration with resampling.
5. Spectral initialization via Gerchberg-Saxton for diverse starting points.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsSmo:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


class _LbfgsActive:
    """L-BFGS objective for a fixed active set S."""
    def __init__(self, active_mask, n):
        self.active_mask = active_mask
        self.n = n
        self.n_active = int(active_mask.sum())

    def _grad_a(self, a):
        """Gradient of sum_{k in S} c_k w.r.t. a."""
        a_rev = a[::-1]
        wc = _full_convolve(self.active_mask.astype(float), a_rev)
        return 2.0 * wc[self.n - 1:2 * self.n - 1]

    def _val_a(self, a):
        """Value of sum_{k in S} c_k."""
        c = _full_convolve(a, a)
        return float(np.dot(self.active_mask, c))

    def __call__(self, z):
        a = _softmax(z)
        return self._val_a(a)

    def grad(self, z):
        a = _softmax(z)
        g_a = self._grad_a(a)
        g_z = a * (g_a - float(np.dot(g_a, a)))
        return g_z


def _run_lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):

    n = len(a_init)

    c_sorted = np.sort(c_cur)[::-1]

    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)

        prob = _LbfgsSmo(beta, n)

            a_new = _softmax(result.x)

            if sc_new < current_sc - 1e-15:

def _active_set_lbfgs(a_init, deadline, min_time_remaining=10.0,
                      tol_start=0.005, tol_end=0.00002, max_rounds=200):

    on the sum of active lags. This converges much faster than gradient descent
    because L-BFGS captures curvature information.
    """

    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(best_a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    tol = tol_start
    stall_count = 0
    max_stall = 40
    round_num = 0

    while time.monotonic() + min_time_remaining < deadline and round_num < max_rounds:
        round_num += 1

        # Compute current autocorrelation
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Determine active set
        threshold = c_max - tol * c_max
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Cap active set size for efficiency (use top-K if too many)
        if n_active > min(500, n):
            c_sorted_idx = np.argsort(c)[::-1]
            active = np.zeros_like(c, dtype=bool)
            active[c_sorted_idx[:500]] = True
            n_active = 500

        # Run L-BFGS on the active-set objective
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 10))))

        prob = _LbfgsActive(active, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxcor': 80, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-14:
                a = a_new
                z = np.log(np.maximum(a, 1e-30))
                z = z - z.mean()
            else:
                # L-BFGS didn't improve; try small perturbation
                z = z + 0.01 * np.random.default_rng(round_num * 7).standard_normal(n)
                z = z - z.mean()
                a = _softmax(z)
        except Exception:
            z = z + 0.005 * np.random.default_rng(round_num * 13).standard_normal(n)
            z = z - z.mean()
            a = _softmax(z)

        # Adaptive tolerance tightening
        if stall_count > max_stall:
            tol = max(tol_end, tol * 0.5)
            # Reset with perturbation
            z = z + 0.02 * np.random.default_rng(round_num * 31).standard_normal(n)
            z = z - z.mean()
            a = _softmax(z)
            stall_count = 0
            if tol <= tol_end * 1.1:
                break
        elif stall_count == 0:
            tol = max(tol_end, tol * 0.95)

    return best_a, _score(best_a)

                z = z + 0.01 * rng.standard_normal(n)
                z = z - z.mean()
                a = _softmax(z)
                tol = max(tol_end, tol * 0.5)
                lr = lr_init * 0.5
                no_improve = 0
                if tol <= tol_end * 1.1:
                    break
        else:
            lr = min(lr_init, lr * 1.01)
            if tol > tol_end:
                tol = max(tol_end, tol * 0.98)

    return best_a, _score(best_a)


def _gs_initialization(n, n_iters=30, seed=0):
    """Gerchberg-Saxton style initialization for flat-spectrum non-negative signal."""
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n)

    # Target: flat power spectrum
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = np.sqrt(n)  # DC component (sum = 1 means a_i ~ 1/n, so A[0] ~ 1)

    # Random phases with Hermitian symmetry
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    phases[0] = 0.0
    A[:] = np.sqrt(1.0 / n) * np.exp(1j * phases)
    # Ensure Hermitian symmetry for real signal
    # DFT: A[k] = conj(A[N-k])
    # For rfft output (length N//2+1), we just use it directly

    a = np.zeros(n)
    for _ in range(n_iters):
        # Inverse FFT to get time domain (take first n samples)
        A_full = np.zeros(N, dtype=complex)
        A_full[:N // 2 + 1] = A
        if N % 2 == 0:
            A_full[N // 2 + 1:] = np.conj(A_full[N // 2 - 1:0:-1])
        else:
            A_full[N // 2 + 1:] = np.conj(A_full[N // 2:0:-1])

        b = np.fft.ifft(A_full).real[:n]

        # Enforce non-negativity
        a = np.maximum(b, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a / s

        # Recompute spectrum
        A_new = np.fft.rfft(a, N)
        # Replace magnitudes with target
        target_mag = np.sqrt(1.0 / n) * np.ones(N // 2 + 1)
        target_mag[0] = 1.0  # DC = sum(a) = 1
        A = target_mag * np.exp(1j * np.angle(A_new))

    return a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage_optimize(a, deadline):
    """Try setting small elements to zero and re-optimizing locally."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    1. L-BFGS with log-sum-exp smoothing (adaptive beta)
    2. Active-set L-BFGS (fast convergence via curvature)
    3. Active-set GD with bisection line search (fine polish)
    4. Multi-n exploration
    5. Spectral (GS) initialization for diversity
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS with smooth objective ===
    p1_deadline = min(deadline, t_start + 300)
    print(f"\n[P1] L-BFGS smooth (t={time.monotonic()-t_start:.1f}s, until t={p1_deadline-t_start:.0f}s)")
    a_main = _run_lbfgs_smooth(parent, p1_deadline, min_time_remaining=5.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} [P1 lbfgs_smooth] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set L-BFGS (main optimization) ===
    p2_deadline = min(deadline, t_start + 700)
    print(f"\n[P2] Active-set L-BFGS (t={time.monotonic()-t_start:.1f}s, until t={p2_deadline-t_start:.0f}s)")
    a_as, sc_as = _active_set_lbfgs(best_seq, p2_deadline, min_time_remaining=10.0,
                                     tol_start=0.004, tol_end=0.00001, max_rounds=300)
    if sc_as < best_score:
        best_seq = a_as.copy()
        best_score = sc_as
        print(f"  NEW BEST: {best_score:.10f} [P2 active_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Fine active-set GD with bisection ===
    p3_deadline = min(deadline, t_start + 1100)
    print(f"\n[P3] Fine active-set GD (t={time.monotonic()-t_start:.1f}s, until t={p3_deadline-t_start:.0f}s)")
    a_gd, sc_gd = _active_set_gd(best_seq, p3_deadline, min_time_remaining=5.0,
                                  tol_start=0.001, tol_end=0.000005, lr_init=0.005)
    if sc_gd < best_score:
        best_seq = a_gd.copy()
        best_score = sc_gd
        print(f"  NEW BEST: {best_score:.10f} [P3 active_gd] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Another round of active-set L-BFGS with tighter tol ===
    p4_deadline = min(deadline, t_start + 1400)
    print(f"\n[P4] Tighter active-set L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < p4_deadline - 20:
        a_as2, sc_as2 = _active_set_lbfgs(best_seq, p4_deadline, min_time_remaining=10.0,
                                           tol_start=0.001, tol_end=0.000005, max_rounds=200)
        if sc_as2 < best_score:
            best_seq = a_as2.copy()
            best_score = sc_as2
            print(f"  NEW BEST: {best_score:.10f} [P4 tight_active_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Shrinkage ===
    print(f"\n[P5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_sh, sc_sh = _shrinkage_optimize(best_seq, deadline - 200)
        if sc_sh < best_score:
            best_seq = a_sh.copy()
            best_score = sc_sh
            print(f"  NEW BEST: {best_score:.10f} [P5 shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 100:
                a_sh2 = _run_lbfgs_smooth(best_seq, min(deadline - 50, time.monotonic() + 80),
                                           min_time_remaining=5.0)
                sc_sh2 = _score(a_sh2)
                if sc_sh2 < best_score:
                    best_seq = a_sh2.copy()
                    best_score = sc_sh2
                    print(f"  NEW BEST: {best_score:.10f} [P5 post_shrink_lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    n_candidates = [n_best]
    for dn in [-512, -256, 256, 512, 1024, -1024]:
        n_try = n_best + dn
        if 1024 <= n_try <= 16384 and n_try != n_best:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_best:
            continue
        a_try = _resample(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: resample score={sc_try:.10f}")
        if time.monotonic() < deadline - 30:
            a_opt = _run_lbfgs_smooth(a_try, min(deadline - 20, time.monotonic() + 40),
                                       min_time_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [P6 multi_n] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Spectral initialization + optimization ===
    print(f"\n[P7] Spectral init (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_spec = len(best_seq)
        for gs_seed in [7, 13, 42]:
            if time.monotonic() > deadline - 30:
                break
            a_gs = _gs_initialization(n_spec, n_iters=40, seed=gs_seed)
            sc_gs = _score(a_gs)
            print(f"  GS seed={gs_seed}: score={sc_gs:.10f}")
            if sc_gs < best_score + 0.05:
                # Worth optimizing
                if time.monotonic() < deadline - 20:
                    a_gs_opt = _run_lbfgs_smooth(a_gs, min(deadline - 10, time.monotonic() + 30),
                                                  min_time_remaining=5.0)
                    sc_gs_opt = _score(a_gs_opt)
                    if sc_gs_opt < best_score:
                        best_seq = a_gs_opt.copy()
                        best_score = sc_gs_opt
                        print(f"  NEW BEST: {best_score:.10f} [P7 gs_seed{gs_seed}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Diverse restarts ===
    print(f"\n[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        n_cur = len(best_seq)
        restart_inits = [
            ("pert2", _perturb(best_seq, 0.02, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("taper", _make_profile(n_cur, "tapered")),
            ("cosmod", _make_profile(n_cur, "cos_mod")),
        ]
        for label, a_r in restart_inits:
            if time.monotonic() > deadline - 15:
                break
            sc_r = _score(a_r)
            if sc_r < best_score + 0.1:
                a_opt = _run_lbfgs_smooth(a_r, min(deadline - 10, time.monotonic() + 20),
                                           min_time_remaining=5.0)
                sc_opt = _score(a_opt)
                if sc_opt < best_score:
                    best_seq = a_opt.copy()
                    best_score = sc_opt
                    print(f"  NEW BEST: {best_score:.10f} [P8 restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final ultra-fine polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_lbfgs(best_seq, deadline - 1, min_time_remaining=1.0,
                                           tol_start=0.0005, tol_end=0.000001, max_rounds=100)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [P9 final_polish] t={time.monotonic()-t_start:.1f}s")

        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            a_fin2, sc_fin2 = _active_set_gd(best_seq, deadline - 1, min_time_remaining=1.0,
                                              tol_start=0.0003, tol_end=0.000001, lr_init=0.002)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [P9 final_gd] t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
