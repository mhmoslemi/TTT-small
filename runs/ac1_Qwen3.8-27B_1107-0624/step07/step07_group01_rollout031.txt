from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-15:

    c = _full_convolve(a, a)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Dual active-set alternating optimization with temperature annealing
2. L-BFGS inner solver in log-space (softmax parameterization)
3. Direct peak-shaving with blame vectors and backtracking line search
4. Nesterov accelerated mirror descent on the simplex
5. Perturbation-based diversification and multi-length exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def _cross_corr(w, a):
    """Compute (w cross-correlate a)[j] = sum_m w[m+j]*a[m] for j=0..len(a)-1.
    w has length 2n-1, a has length n."""
    n = len(a)
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(np.conj(W) * A, N)[:n]
    return result


def _smooth_obj(z, w, n):
    """Compute smoothed objective in z-space: g(softmax(z)) where
    g(a) = sum_k w[k] * (a*a)[k]."""
    a = _softmax_safe(z)
    h = _cross_corr(w, a)
    g = float(np.dot(a, h))
    return g


def _smooth_grad(z, w, n):
    """Gradient of smoothed objective in z-space."""
    a = _softmax_safe(z)
    h = _cross_corr(w, a)
    g = float(np.dot(a, h))
    return 2.0 * a * (h - g)


def _dual_lbfgs(a_init, w, n, max_iter, gtol=1e-14, ftol=1e-18):
    """Run L-BFGS to minimize the weighted quadratic form in log-space."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smooth_obj(z, w, n)

    def jac(z):
        return _smooth_grad(z, w, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50, 'maxls': 60,
                                   'disp': False})
        return _softmax_safe(result.x)
    except Exception:
        return a_init.copy()


def _dual_anneal(a_start, deadline, reserve_time=300.0, n_temps=25):
    """Dual active-set method with temperature annealing."""
    n = len(a_start)
    current_a = a_start.copy()
    best_a = a_start.copy()
    best_sc = _score(current_a)

    c = _full_convolve(current_a, current_a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return best_a, best_sc

    temps = c_max * np.geomspace(1.0, 1e-8, n_temps)

    for ti, temp in enumerate(temps):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        c = _full_convolve(current_a, current_a)
        shifted = (c - c.max()) / temp
        ex = np.exp(shifted)
        w = ex / ex.sum()

        # Time budget per temperature level
        elapsed_in_phase = 0.0
        t0 = time.monotonic()
        max_iter = min(3000, max(50, int(remaining / (n_temps - ti + 1) / 0.01)))

        a_new = _dual_lbfgs(current_a, w, n, max_iter)
        sc = _score(a_new)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            elapsed = time.monotonic() - t0
            print(f"    T={ti}/{n_temps}: score={sc:.10f} (improved) [{elapsed:.1f}s]")

        if sc < current_sc - 1e-15:
            current_a = a_new

        if best_sc < 1.501:
            break

    return best_a, best_sc


def _peak_shave(a_start, deadline, k_active=5, max_steps=50000):
    """Direct peak shaving with blame vectors and backtracking."""
    n = len(a_start)
    a = a_start.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.1 / np.sqrt(n)
    stall = 0
    stall_limit = 200

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        idx = np.argsort(c)[::-1][:k_active]

        # Compute blame vector
        blame = np.zeros(n)
        for p in idx:
            for j in range(n):
                m = p - j
                if 0 <= m < 2 * n - 1:
                    blame[j] += a[m]

        g = blame - blame.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        alpha = step_size
        improved = False
        for _ in range(15):
            a_try = a - alpha * g / np.sqrt(n)
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                step_size = min(alpha * 1.05, 0.5)
                stall = 0
                break
            alpha *= 0.6

        if not improved:
            stall += 1
            step_size *= 0.85
            if stall > stall_limit or step_size < 1e-10:
                break

    return best_a, best_sc


def _mirror_descent(a_start, deadline, max_steps=30000):
    """Nesterov accelerated mirror descent on the simplex."""
    n = len(a_start)
    z = np.log(np.maximum(a_start, 1e-18))
    z = z - z.mean()
    best_a = a_start.copy()
    best_sc = _score(a_start)
    lr = 0.1
    stall = 0
    y = z.copy()

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax_safe(y)
        c = _full_convolve(a, a)
        c_max = c.max()
        idx = np.argsort(c)[::-1][:3]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)

        h = _cross_corr(w, a)
        g = float(np.dot(a, h))
        grad_z = 2.0 * a * (h - g)

        z = y - lr * grad_z / (a * n + 1e-18)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if (i + 1) % 3 == 0:
            t = (1.0 + np.sqrt(1.0 + 4.0 * (i + 1) ** 2)) / 2.0
            y = z + ((t - 1.0) / t) * (z - y)

        if i % 1000 == 0:
            a_check = _softmax_safe(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.7
                    if lr < 1e-10:
                        break

    a_final = _softmax_safe(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _logsumexp_lbfgs(a_start, deadline, reserve_time=100.0):
    """Standard log-sum-exp smoothed L-BFGS with adaptive beta."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)
    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                              0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
                              5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])

    for i, beta in enumerate(betas):
        if time.monotonic() + reserve_time > deadline:
            break
        remaining = deadline - reserve_time - time.monotonic()
        if remaining < 5:
            break

        max_iter = min(5000, max(100, int(remaining / (len(betas) - i) / 0.008)))

        def fun(z, b=beta):
            a_s = _softmax_safe(z)
            c_s = _full_convolve(a_s, a_s)
            c_m = c_s.max()
            shifted = (c_s - c_m) / b
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_m
            return 2.0 * n * (c_m + b * np.log(s))

        def jac(z, b=beta):
            a_s = _softmax_safe(z)
            c_s = _full_convolve(a_s, a_s)
            c_m = c_s.max()
            shifted = (c_s - c_m) / b
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c_s)
                w[int(np.argmax(c_s))] = 1.0
            else:
                w = e / s
            a_rev = a_s[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_s))
            return 4.0 * n * a_s * (h - W)

        z0 = np.log(np.maximum(current_a, 1e-18))
        z0 = z0 - z0.mean()

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 60,
                                       'maxls': 80, 'disp': False})
            a_new = _softmax_safe(result.x)
            sc = _score(a_new)
            if sc < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc
        except Exception:
            pass

        if current_sc < 1.501:
            break

    return current_a


def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.25):

        bands = [1, 3]
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_and_optimize(a_base, rng, deadline, reserve=30.0):

        try_update(a_ps2, "peak_shave_k7")

    # === Phase 5: Mirror descent ===
    print(f"\n=== Phase 5: Mirror Descent (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_md, sc_md = _mirror_descent(best_seq, deadline - 100, max_steps=20000)
        try_update(a_md, "mirror_descent")

    # === Phase 6: Diversification (perturb + optimize) ===
    print(f"\n=== Phase 6: Diversification (t={time.monotonic()-t_start:.1f}s) ===")
    n_hops = 8
    if best_seq is not None and time.monotonic() < deadline - 60:
        time_per_hop = max(5.0, (deadline - 60 - time.monotonic()) / n_hops)
        for h in range(n_hops):
            if time.monotonic() > deadline - 30:
                break
            hop_dl = time.monotonic() + time_per_hop * 0.8
            a_hop = _perturb_and_optimize(best_seq, rng, hop_dl, reserve=5.0)
            try_update(a_hop, f"diverse_{h}")

    # === Phase 7: Try different length ===
    print(f"\n=== Phase 7: Length Exploration (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        cur_n = len(best_seq)
        target_ns = [32768, 8192]
        for tn in target_ns:
            if time.monotonic() > deadline - 40:
                break
            if tn == cur_n:
                continue
            print(f"  Trying n={tn}...")
            a_interp = _interpolate_to_n(best_seq, tn)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")

            if time.monotonic() < deadline - 30:
                # Quick dual anneal at new length
                a_opt = _dual_anneal(a_interp, deadline, reserve_time=20, n_temps=8)
                try_update(a_opt, f"n_{tn}_dual")

    # === Phase 8: Final aggressive polish ===
    print(f"\n=== Phase 8: Final Polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Multiple rounds of tiny steps
        for rnd in range(10):
            if time.monotonic() > deadline - 3:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())

            # Very fine dual step
            temp = mb * 1e-5
            shifted = (c_b - mb) / temp
            ex = np.exp(shifted)
            w = ex / ex.sum()
            max_iter = min(2000, max(50, int((deadline - 3 - time.monotonic()) / 0.005)))
            a_f = _dual_lbfgs(best_seq, w, n, max_iter)
            try_update(a_f, f"final_dual_{rnd}")

            # Tiny peak shave
            if time.monotonic() < deadline - 3:
                a_ps, _ = _peak_shave(best_seq, deadline - 3, k_active=3, max_steps=5000)
                try_update(a_ps, f"final_ps_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
