Actually, one more thought: the problem has a nice structure that I can exploit. The objective `2n * max(c) / (sum a)^2` is scale-invariant, so I can fix `sum a = 1` and just minimize `2n * max(c)`. With `sum a = 1`, the problem is:

minimize `max_k c[k]` subject to `sum a = 1`, `a >= 0`, `c = convolve(a, a)`.

The KKT conditions at the optimum (assuming the active set is `S` and all `a[i] > 0`):
- For `k in S`: `∂c[k]/∂a = lambda * 1 + mu_k` (where `mu_k` is the multiplier for the constraint `c[k] <= M`)
- `lambda` is the multiplier for `sum a = 1`
- At the optimum, `c[k] = M` for all `k in S`

Actually, the KKT conditions are:
- Minimize `M` subject to `c[k] <= M` for all k, `sum a = 1`, `a >= 0`
- Lagrangian: `L = M - sum_k nu_k (M - c[k]) - lambda (sum a - 1) - sum_i eta_i a_i`
- Stationarity: `∂L/∂M = 1 - sum_k nu_k = 0`, so `sum nu_k = 1`
- `∂L/∂a[i] = -sum_k nu_k * 2*a[k-i] - lambda + eta_i = 0`
- Complementary slackness: `nu_k (M - c[k]) = 0`, `eta_i * a[i] = 0`

So at the optimum: for active k (nu_k > 0), `c[k] = M`. For inactive k, `nu_k = 0`.
And `lambda = -sum_{k active} nu_k * 2*a[k-i] + eta_i` for each `i` (with `eta_i = 0` if `a[i] > 0`).

For `a[i] > 0`: `sum_{k in S} nu_k * a[k-i] = -lambda/2` (constant for all i with a[i] > 0).

This means: the weighted sum of "shifted a" over the active set is constant (on the support of a). This is a strong condition!

If the support of `a` is all of `{0, ..., n-1}` (i.e., all `a[i] > 0`), then:
`sum_{k in S} nu_k * a[k-i] = C` for all `i = 0, ..., n-1`.

This is a system of `n` equations in the weights `nu_k` (with the constraint `sum nu_k = 1`). If `|S| = n`, we have a square system. The solution gives the optimal multipliers, and from them we can determine the optimal `a`.

But solving this system exactly is complex. The Newton approach essentially does this iteratively.

"""
Optimized step function minimizer using multi-phase approach:
L-BFGS with log-sum-exp smoothing, active-set CG-Newton QP, 
LP-based peak reduction, multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]

def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs using CG with the Hankel structure."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        if pHp < 0:  # Not PD, can't continue CG
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.5:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _active_set_newton(a_init, deadline, max_rounds=100):
    """CG-Newton polish on the active set with constraint checking."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set: start very tight
        tol_af = 1e-4 * c_max
        active = c >= (c_max - tol_af)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        # Build W vector
        W = np.zeros(2 * n - 1)
        W[active] = 1.0 / n_active
        
        # Gradient of sum of active c[k]
        g = _hankel_matvec(a, W, n)
        g_t = g - np.mean(g)  # Project onto sum=0
        gn = float(np.linalg.norm(g_t))
        if gn < 1e-15 * n:
            break
        
        # Newton step via CG
        Hg = _cg_solve(W, n, g, max_iter=300)
        H1 = _cg_solve(W, n, np.ones(n), max_iter=300)
        H1_norm = float(np.linalg.norm(H1))
        if H1_norm < 1e-20 or not np.isfinite(H1_norm):
            dx = -g_t / (gn + 1e-30)
        else:
            lam = float(np.ones(n) @ Hg) / float(np.ones(n) @ H1)
            dx = -Hg + lam * H1
            dx = dx - np.mean(dx)
            dx_norm = float(np.linalg.norm(dx))
            if dx_norm < 1e-30 or not np.isfinite(dx_norm):
                dx = -g_t / (gn + 1e-30)
        
        # Damped Newton with backtracking + constraint check
        alpha = 1.0
        step_done = False
        for ls in range(60):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            c_max_try = float(c_try.max())
            # Armijo-like condition
            if c_max_try <= c_max - 1e-4 * alpha * gn:
                a = a_try
                step_done = True
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break
        
        if not step_done:
            # Gradient fallback
            alpha = 0.01
            for ls in range(40):
                a_try = a - alpha * g_t
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if float(c_try.max()) < c_max:
                    a = a_try
                    break
                alpha *= 0.5
        
        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

def _peak_shave(a_init, deadline, max_iters=200):
    """Iteratively reduce the peak element of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # Direction to reduce c[k_star]: grad of c[k_star] is 2*a[k_star - i]
        grad_peak = np.zeros(n)
        valid = (k_star >= 0) & (k_star < 2*n - 1)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad_peak[i] = 2.0 * a[j]
        
        # Project onto sum=0
        grad_peak -= np.mean(grad_peak)
        
        # Also consider other near-peak elements to avoid making them worse
        # Add a regularization from the second-highest peak
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.99 * c_max:
            # Multiple peaks, use their combined gradient
            near_peak = c > 0.99 * c_max
            combined_grad = np.zeros(n)
            for k in np.where(near_peak)[0]:
                for i in range(n):
                    j = int(k) - i
                    if 0 <= j < n:
                        combined_grad[i] += 2.0 * a[j]
            combined_grad /= near_peak.sum()
            combined_grad -= np.mean(combined_grad)
            if np.linalg.norm(combined_grad) > 0:
                grad_peak = combined_grad
        
        gnorm = np.linalg.norm(grad_peak)
        if gnorm < 1e-15:
            break
        
        # Line search
        alpha = 0.1
        improved = False
        for ls in range(30):
            a_try = a - alpha * grad_peak / gnorm
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            if float(c_try.max()) < c_max * (1 - 1e-6):
                a = a_try
                improved = True
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break
            alpha *= 0.5
        
        if not improved:
            break
    
    return best_a, best_sc

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    base_beta = max(gap * 0.3, max_c * 0.01)
    
    # More gradual beta schedule
    ratios = np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.17, 0.12, 0.08, 
                       0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.001,
                       0.0005, 0.0002, 0.0001])
    betas = base_beta * ratios
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 20))))
        
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat":
        return np.ones(n) / n
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)

    c = _full_convolve(a, a)

    V = v[::-1]

    N = _next_pow2(out_len)

    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]

def _smoobj_impl(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_impl(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * t * (1 + t))) + 0.1
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-12.0 * (t - 0.3) ** 2) + np.exp(-12.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def do_update(p, sc_val, label):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if np.isfinite(sc_val) and sc_val < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc_val)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                par_sc = _score(parent)
                print(f"Parent: n={len(parent)}, score={par_sc:.10f}")
                do_update(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        parent = _make_profile(8192, "tapered")
        par_sc = _score(parent)
        print(f"Init: n={len(parent)}, score={par_sc:.10f}")
        do_update(parent, par_sc, "init")

    parent_n = len(parent)

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a1 = _run_lbfgs(parent, deadline, min_time_remaining=350)
        sc1 = _score(a1)
        do_update(a1, sc1, "lbfgs_main")

    # Phase 2: Active set Newton (first round)
    print(f"\n[Phase 2] Newton round 1 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        dl = min(deadline - 300, time.monotonic() + 100)
        a2, sc2 = _active_set_newton(best_seq, dl, max_rounds=150)
        do_update(a2, sc2, "newton_1")

    # Phase 3: Peak shaving
    print(f"\n[Phase 3] Peak shave (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        dl = min(deadline - 250, time.monotonic() + 80)
        a3, sc3 = _peak_shave(best_seq, dl, max_iters=300)
        do_update(a3, sc3, "peak_1")

    # Phase 4: Newton round 2
    print(f"\n[Phase 4] Newton round 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        dl = min(deadline - 200, time.monotonic() + 80)
        a4, sc4 = _active_set_newton(best_seq, dl, max_rounds=150)
        do_update(a4, sc4, "newton_2")

    # Phase 5: Peak shave 2
    print(f"\n[Phase 5] Peak shave 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        dl = min(deadline - 150, time.monotonic() + 60)
        a5, sc5 = _peak_shave(best_seq, dl, max_iters=200)
        do_update(a5, sc5, "peak_2")

    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384, 32768]
    n_candidates = [n for n in n_candidates if n != parent_n or n == 8192]
    # Sort by proximity to parent_n
    n_candidates.sort(key=lambda x: abs(x - parent_n))
    # Put parent_n first if present
    if parent_n in n_candidates:
        n_candidates.remove(parent_n)
        n_candidates = [parent_n] + n_candidates

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == parent_n:
            continue  # Already optimized
        a_init = _resample(parent, n_try)
        remaining_for_lbfgs = min(120, deadline - 80 - time.monotonic())
        if remaining_for_lbfgs < 10:
            break
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=max(80, 1800 - (time.monotonic() - t_start)))
        sc_opt = _score(a_opt)
        do_update(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            dl = min(deadline - 40, time.monotonic() + 50)
            a_np, sc_np = _active_set_newton(a_opt, dl, max_rounds=80)
            do_update(a_np, sc_np, f"n_{n_try}_newton")

    # Phase 7: Diverse restarts on best_n
    print(f"\n[Phase 7] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_profiles = ["chirp", "bimodal", "cos"]
        for i, prof in enumerate(restart_profiles):
            if time.monotonic() > deadline - 25:
                break
            n_b = len(best_seq)
            p_start = _make_profile(n_b, prof)
            # Blend with best
            p_start = 0.7 * best_seq + 0.3 * p_start
            p_start = np.maximum(p_start, 0)
            p_start = p_start / p_start.sum()
            p_start = _perturb(p_start, 0.02, rng)
            if time.monotonic() > deadline - 20:
                break
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=20)
            sc_r = _score(a_r)
            do_update(a_r, sc_r, f"restart_{i}")

    # Phase 8: Final ultra-fine polish
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Multiple short Newton passes
        for pass_i in range(3):
            if time.monotonic() > deadline - 3:
                break
            dl = min(deadline - 3, time.monotonic() + 30)
            a_f, sc_f = _active_set_newton(best_seq, dl, max_rounds=100)
            do_update(a_f, sc_f, f"final_np_{pass_i}")
            if time.monotonic() > deadline - 3:
                break
            a_ps, sc_ps = _peak_shave(best_seq, min(deadline - 3, time.monotonic() + 20), max_iters=100)
            do_update(a_ps, sc_ps, f"final_ps_{pass_i}")

    # Final output
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
