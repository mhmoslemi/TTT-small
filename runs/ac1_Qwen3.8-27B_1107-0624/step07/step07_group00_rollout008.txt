def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1
    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses cosine-annealed Adam in log-space with warm restarts, adaptive temperature,
peak-directed coordinate descent, structured simulated annealing, and Fourier smoothing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_grad(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(np.clip(c_shifted, -100.0, 0.0))
    s = float(e.sum())
    if s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _topk_grad(a, c, top_k=5, decay=0.8):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_cosine(a_init, deadline, T_start, T_end, lr_start, lr_end,
                 max_iter=500000, active_k=3, decay=0.7,
                 eval_interval=200, clip_norm=0.8,
                 n_restarts=3, restart_interval=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0
    restart_count = 0

    total_iters = max_iter * (1 + n_restarts)

    while t < total_iters:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / total_iters)

        # Cosine annealing with warm restarts
        cycle = min(frac * (1 + n_restarts), n_restarts)
        cycle_frac = frac * (1 + n_restarts) - int(frac * (1 + n_restarts))
        cos_val = 0.5 * (1.0 + np.cos(np.pi * cycle_frac))
        lr = lr_end + 0.5 * (lr_start - lr_end) * cos_val * current_lr_mult

        a = _softmax_stable(z)

                    m_buf = np.zeros(n)

                js = list(range(max(0, k_star - n + 1), min(k_star, n - 1) + 1))
            else:
                js = list(range(k_star - (n - 1), n))
            if js:
                best_sc = best_score
                best_delta = None
                for j in js:
                    for scale in [1.1, 1.2, 0.8, 0.7, 1.3]:
                        a_cand = best_a.copy()
                        old_val = a_cand[j]
                        a_cand[j] = old_val * scale
                        s = a_cand.sum()
                        if s < 1e-15:
                            continue
                        a_cand /= s
                        c_cand = _fft_convolve(a_cand, a_cand)
                        sc_cand = float(2.0 * n * c_cand.max())
                        if sc_cand < best_sc:
                            best_sc = sc_cand
                            best_delta = (j, scale)
                            best_a_cand = a_cand.copy()
                if best_delta is not None:
                    best_a = best_a_cand
                    best_score = best_sc
                    improved = True

        if not improved:
            break

    return best_a, _score_norm(best_a)


def _simulated_annealing(a_init, deadline, max_iter=50000, T_start=0.01, T_end=0.0001):
    """Structured simulated annealing with swap and local reshuffle moves."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(a, a)
    T = T_start
    rng = np.random.default_rng(12345)
    iter_count = 0
    no_improve = 0

    while iter_count < max_iter:
        if time.monotonic() > deadline:
            break
        iter_count += 1
        T = T_start * (T_end / T_start) ** (iter_count / max_iter)

        # Choose move type
        move_type = rng.random()
        a_cand = a.copy()

        if move_type < 0.5:
            # Swap: move mass from one coordinate to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.1, 0.5)
            a_cand[i] -= amt
            a_cand[j] += amt
            if a_cand[i] < 0:
                a_cand[i] = 0
                a_cand[j] += amt + a[i]
                a_cand[i] = 0
        elif move_type < 0.8:
            # Local reshuffle: redistribute mass in a window
            center = rng.integers(0, n)
            w = rng.integers(3, 11)
            lo = max(0, center - w)
            hi = min(n, center + w + 1)
            window = a_cand[lo:hi].copy()
            total = window.sum()
            if total < 1e-15:
                continue
            # Perturb the window
            noise = rng.standard_normal(hi - lo) * 0.3
            window = window + noise * total
            window = np.maximum(window, 0)
            wsum = window.sum()
            if wsum < 1e-15:
                window = np.ones(hi - lo) / (hi - lo)
                wsum = 1.0
            window = window / wsum * total
            a_cand[lo:hi] = window
        else:
            # Peak-directed: reduce mass at coordinates feeding the top peak
            c_cur = _fft_convolve(a, a)
            k_star = int(np.argmax(c_cur))
            if k_star < n:
                js = list(range(max(0, k_star - n + 1), min(k_star, n - 1) + 1))
            else:
                js = list(range(max(0, k_star - n + 1), n))
            if not js:
                continue
            j_pick = js[rng.integers(0, len(js))]
            amt = a_cand[j_pick] * rng.uniform(0.05, 0.3)
            a_cand[j_pick] -= amt
            # Redistribute to a random location far from the peak
            j_dest = rng.integers(0, n)
            a_cand[j_dest] += amt
            if a_cand[j_pick] < 0:
                a_cand[j_dest] += a_cand[j_pick]
                a_cand[j_pick] = 0

        s = a_cand.sum()
        if s < 1e-15:
            continue
        a_cand = a_cand / s
        sc_cand = _score_norm(a_cand)

        if sc_cand < best_score:
            best_a = a_cand.copy()
            best_score = sc_cand
            no_improve = 0
        if sc_cand < _score_norm(a) or rng.random() < np.exp((_score_norm(a) - sc_cand) / (T * best_score)):
            a = a_cand
        else:
            no_improve += 1
            if no_improve > 5000:
                break

    return best_a, _score_norm(best_a)


def _fourier_smooth(a, bandwidth=0.1):
    """Apply mild low-pass filter in Fourier domain."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    freqs = np.fft.rfftfreq(N)
    # Keep low frequencies
    mask = np.exp(-(freqs / bandwidth) ** 2)
    A_filt = A * mask
    # Preserve DC
    A_filt[0] = A[0]
    a_filt = np.fft.irfft(A_filt, N)[:n]
    a_filt = np.maximum(a_filt, 0)
    s = a_filt.sum()
    if s < 1e-15:
        return a.copy()
    return a_filt / s


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with cosine-annealed Adam, peak-directed CD,
    structured simulated annealing, and Fourier smoothing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    best_a = a.copy()
    best_score = _score_full(best_a)
    n_best = len(best_a)
    print(f"Initial: n={n_best}, score={best_score:.10f}")

    # Phase 1: Adam cosine with warm restarts (0-500s)
    phase1_end = min(deadline, t_start + 500.0)
    c_cur = _fft_convolve(best_a, best_a)
    max_c = float(c_cur.max())
    a_opt, sc = _adam_cosine(best_a, phase1_end,
                             T_start=0.004 * max_c, T_end=0.00003 * max_c,
                             lr_start=0.05, lr_end=0.002,
                             max_iter=400000, active_k=4,
                             eval_interval=150, n_restarts=3)
    full_sc = _score_full(a_opt)
    print(f"Phase 1 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()
        n_best = len(best_a)

    # Phase 2: Peak-directed coordinate descent (500s-800s)
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end:
        a_cd, sc_cd = _peak_directed_cd(best_a, phase2_end, max_rounds=300)
        full_sc = _score_full(a_cd)
        print(f"Phase 2 CD: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()
            n_best = len(best_a)

    # Phase 3: Simulated annealing (800s-1100s)
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end:
        c_for_T = _fft_convolve(best_a, best_a)
        T_sa = float(c_for_T.max()) * 0.05
        a_sa, sc_sa = _simulated_annealing(best_a, phase3_end,
                                           max_iter=80000, T_start=T_sa,
                                           T_end=T_sa * 0.01)
        full_sc = _score_full(a_sa)
        print(f"Phase 3 SA: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_sa.copy()
            n_best = len(best_a)

    # Phase 4: Fine-tuning Adam (1100s-1400s)
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_pol, sc_pol = _adam_cosine(best_a, phase4_end,
                                     T_start=0.001 * max_cf, T_end=0.00001 * max_cf,
                                     lr_start=0.015, lr_end=0.0005,
                                     max_iter=300000, active_k=3,
                                     eval_interval=100, n_restarts=2)
        full_sc = _score_full(a_pol)
        print(f"Phase 4 fine Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_pol.copy()
            n_best = len(best_a)

        # Fourier smoothing + re-optimize
        if time.monotonic() < phase4_end - 60:
            a_smooth = _fourier_smooth(best_a, bandwidth=0.15)
            c_sm = _fft_convolve(a_smooth, a_smooth)
            max_sm = float(c_sm.max())
            a_sm_opt, sc_sm = _adam_cosine(a_smooth, phase4_end - 5,
                                           T_start=0.0005 * max_sm, T_end=0.000005 * max_sm,
                                           lr_start=0.01, lr_end=0.0003,
                                           max_iter=100000, active_k=3,
                                           eval_interval=80, n_restarts=1)
            full_sc_sm = _score_full(a_sm_opt)
            print(f"  Fourier smooth: score={full_sc_sm:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_sm) and full_sc_sm < best_score:
                best_score = full_sc_sm
                best_a = a_sm_opt.copy()
                n_best = len(best_a)

    # Phase 5: Random restarts (1400s-1650s)
    phase5_end = min(deadline, t_start + 1650.0)
    n_fin = len(best_a)
    if time.monotonic() < phase5_end:
        n_restarts = 5
        time_per = (phase5_end - time.monotonic()) / n_restarts
        for attempt in range(n_restarts):
            if time.monotonic() > phase5_end - 30:
                break
            sigma = 0.015 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(phase5_end - 10, time.monotonic() + time_per)
            a_pert_opt, sc_pert = _adam_cosine(a_pert, sub_dl,
                                               T_start=0.001 * max_cp, T_end=0.000005 * max_cp,
                                               lr_start=0.02, lr_end=0.0005,
                                               max_iter=80000, active_k=3,
                                               eval_interval=80, n_restarts=1)
            full_sc_pert = _score_full(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.4f}): score={full_sc_pert:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_pert) and full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()
                n_fin = len(best_a)

    # Phase 6: Final ultra-fine polish (1650s - deadline)
    if time.monotonic() < deadline - 10:
        c_final = _fft_convolve(best_a, best_a)
        max_cfinal = float(c_final.max())
        a_final_opt, sc_final_opt = _adam_cosine(best_a, deadline - 15,
                                                 T_start=0.0002 * max_cfinal,
                                                 T_end=0.000002 * max_cfinal,
                                                 lr_start=0.005, lr_end=0.0001,
                                                 max_iter=200000, active_k=2,
                                                 eval_interval=50, n_restarts=0)
        full_sc_final = _score_full(a_final_opt)
        print(f"Phase 6 final: score={full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_final) and full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final_opt.copy()

        # Quick CD at the very end
        if time.monotonic() < deadline - 5:
            a_cd_final, _ = _peak_directed_cd(best_a, deadline - 3, max_rounds=30)
            sc_cd_final = _score_full(a_cd_final)
            if sc_cd_final < best_score:
                best_score = sc_cd_final
                best_a = a_cd_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score_full(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
