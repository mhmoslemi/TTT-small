import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return _score_norm(a_norm)

def _project_simplex(v):
    """Project vector v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _subgradient(a, k_active):
    """
    Compute the subgradient of max conv(a,a) at a, using active set k_active.
    The gradient of conv(a,a)[k] w.r.t. a[m] is 2*a[k-m] (for valid indices).
    We average over active indices.

    n = len(a)
    g = np.zeros(n)
    for k in k_active:

            g[m_lo:m_hi+1] += 2.0 * a[k - np.arange(m_lo, m_hi + 1)]

def _symmetric_optimize(a_init, deadline, max_iter=50000, top_k=5):
    """
    Optimize with symmetry constraint: a[j] = a[n-1-j].
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    # Force symmetry
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for j in range(half):
        a_sym[j] = (a[j] + a[n-1-j]) / 2
        a_sym[n-1-j] = a_sym[j]
    a_sym = a_sym / a_sym.sum()
    
    best_a = a_sym.copy()
    best_score = _score_norm(a_sym)
    
    # Work with half the variables
    h = half
    z = np.log(np.maximum(a_sym[:h], 1e-15))
    m_buf = np.zeros(h)
    v_buf = np.zeros(h)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = t / max_iter
        lr = 0.03 * (0.01) ** frac
        
        # Build full symmetric a
        a_full = np.zeros(n)
        for j in range(h):
            a_full[j] = a_sym[j]
            a_full[n-1-j] = a_sym[j]
        a_full = np.maximum(a_full, 0)
        a_full = a_full / a_full.sum()
        
        c = _fft_convolve(a_full, a_full)
        k_active = np.argpartition(c, -top_k)[-top_k:]
        
        # Gradient for half variables
        g_half = np.zeros(h)
        for k in k_active:
            for j in range(h):
                other = k - j
                if 0 <= other < n:
                    g_half[j] += 2.0 * a_full[other]
                    other2 = k - (n-1-j)
                    if 0 <= other2 < n:
                        g_half[j] += 2.0 * a_full[other2]
        g_half /= len(k_active)
        
        m_buf = b1 * m_buf + (1-b1) * g_half
        v_buf = b2 * v_buf + (1-b2) * g_half**2
        mh = m_buf / (1-b1**(t+1))
        vh = v_buf / (1-b2**(t+1))
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        
        # Convert back
        a_half = np.exp(z - z.max())
        a_half = a_half / a_half.sum()
        # Scale so full sum = 1
        if h > n // 2:
            # odd length, middle element counted once
            scale = 1.0 / (2.0 * a_half.sum() - a_half[h-1])
        else:
            scale = 1.0 / (2.0 * a_half.sum())
        a_new_half = a_half * scale
        
        a_full_new = np.zeros(n)
        for j in range(h):
            a_full_new[j] = a_new_half[j]
            a_full_new[n-1-j] = a_new_half[j]
        a_full_new = a_full_new / a_full_new.sum()
        
        sc = _score_norm(a_full_new)
        if sc < best_score:
            best_score = sc
            best_a = a_full_new.copy()
            a_sym = best_a.copy()
        else:
            a_sym[:h] = np.maximum(a_full_new[:h], 1e-15)
            a_sym[n-1-np.arange(h)] = np.maximum(a_full_new[n-1-np.arange(h)], 1e-15)
    
    return best_a, best_score

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
