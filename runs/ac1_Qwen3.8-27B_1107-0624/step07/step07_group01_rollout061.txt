    actual_steps = min(max_feasible, max(100, max_time_feasible))
    
    no_improve = 0
    patience = max(500, actual_steps // 5)

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start with structured + parent initializations
2. Adam optimizer in log-space with LSE-smoothed gradient and gradual beta annealing
3. Multi-track parallel exploration at different lengths
4. L-BFGS polishing with adaptive beta
5. Active-set subgradient refinement with increasing k
6. Final micro-polish for precision
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_lse(a, c, n, beta):
    """Gradient of 2n*LSE(c,beta) w.r.t. log-variables z."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_topk(a, c, n, k):
    """Subgradient using top-k active set with uniform weights."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_exp_weighted(a, c, n, alpha):
    """Gradient using exponential weighting: w_k = exp(-alpha*(cmax-ck))/Z."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    shifted = alpha * (c_max - c)
    shifted_shifted = shifted - shifted.min()
    e = np.exp(-shifted_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_track(a_init, n, deadline, reserve, beta_frac_start, beta_frac_end,
                lr_start, lr_end, total_steps, checkpoint_every=500):
    """Run Adam optimization with annealed beta (as fraction of c_max)."""
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0

    best_a = a_init.copy()
    best_sc = _score(a_init)
    no_improve = 0

    for step in range(total_steps):
        if time.monotonic() > deadline - reserve:
            break

        progress = step / max(1, total_steps - 1)
        # Cosine annealing for both beta and lr
        cos_val = 0.5 * (1 + math.cos(math.pi * progress))
        beta_frac = beta_frac_start + (beta_frac_end - beta_frac_start) * (1 - cos_val)
        lr = lr_start + (lr_end - lr_start) * (1 - cos_val)

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_frac * c_max

        if beta > 1e-18:
            g = _grad_lse(a, c, n, beta)
        else:
            g = _grad_topk(a, c, n, 5)

        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()

        if step % checkpoint_every == 0 or step == total_steps - 1:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += checkpoint_every

        if no_improve > max(3000, total_steps // 3):
            # Small perturbation to escape
            z = z + lr * 0.5 * np.random.standard_normal(n)
            z = z - z.mean()
            no_improve = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _lbfgs_smooth(a_init, beta, n, max_iter=5000):
    """L-BFGS in log-space with LSE smoothing."""
    from scipy.optimize import minimize

    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))

    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50,
                                   'maxls': 60, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)

def _make_modulated(n, freq, eps):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands, eps=0.2):
    t = np.arange(n)
    a = np.ones(n)
    for f in bands:
        a += (eps / len(bands)) * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_willett(n, rng):
    """Willett-type construction: random signs smoothed."""
    a = rng.standard_normal(n)
    # Smooth with a wide window
    kernel_size = max(3, n // 256)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.abs(a) + 0.05
    return a / a.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp

            if target_n == current_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated: {sc_interp:.10f}")

            if time.monotonic() < deadline - 40:
                a_opt, _ = _adam_track(a_interp, target_n, deadline, 40.0,
                                       beta_frac_start=0.05, beta_frac_end=0.005,
                                       lr_start=0.3, lr_end=0.05,
                                       total_steps=8000, checkpoint_every=200)
                try_update(a_opt, f"multilength_{target_n}")

    # === Phase 7: Final aggressive micro-polish ===
    print(f"\n=== Phase 7: Final Polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        n = len(best_seq)

        # Ultra-fine Adam
        if time.monotonic() < deadline - 10:
            a_uf, _ = _adam_track(best_seq, n, deadline, 8.0,
                                  beta_frac_start=0.001, beta_frac_end=0.0001,
                                  lr_start=0.05, lr_end=0.005,
                                  total_steps=20000, checkpoint_every=200)
            try_update(a_uf, "ultra_fine_adam")

        # L-BFGS with tiny beta
        if time.monotonic() < deadline - 8:
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            for bf in [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]:
                if time.monotonic() > deadline - 8:
                    break
                beta = mb * bf
                a_f = _lbfgs_smooth(best_seq, beta, n, 5000)
                try_update(a_f, f"final_lbfgs_{bf}")

        # Active set with k=1 (exact subgradient)
        if time.monotonic() < deadline - 5:
            a_s, _ = _active_polish(best_seq, deadline - 3, 3.0,
                                    k_schedule=[1, 2, 3, 5])
            try_update(a_s, "final_active")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
