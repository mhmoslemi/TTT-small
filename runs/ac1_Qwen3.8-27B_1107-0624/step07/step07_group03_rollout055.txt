"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using Active-Set Multiplicative Update (AS-MU)
combined with L-BFGS refinement and active-set Newton polish.

Key innovation: AS-MU uses the KKT optimality condition (gradient should be constant on the support)
to drive a multiplicative update on the simplex, which is more principled than the previous
heuristic approaches. Combined with careful annealing of the active-set temperature.

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def corr_a_w(a, w, n):
    """Compute g[i] = sum_j a[j] * w[i+j] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def as_mu(a_init, deadline, max_iter=500000):
    """
    Active-Set Multiplicative Update method.
    Minimizes max(conv(a,a)) on the probability simplex (sum(a)=1, a>=0).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    
    best_a = a.copy()
    best_sc = score_fn(a)
    
    # Temperature schedule: start warm, cool down
    tau_init = 0.1  # relative to max
    tau_floor = 0.0001
    
    eta = 0.5
    patience = 0
    max_patience = 5000
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve(a, a)
        c_max = float(c.max())
        
        # Soft active set weights
        tau = max(tau_init * (0.9999 ** it), tau_floor)
        shifted = (c - c_max) / (tau * c_max + 1e-15)
        w = np.exp(np.clip(shifted, -700, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient g[i] = 2 * sum_j a[j] * w[i+j]
        g = 2.0 * corr_a_w(a, w, n)

        if support_mask.sum() > 0:
            target = float(g[support_mask].mean())
        else:
            target = float(g.mean())
        
        if target < 1e-15:

        # a[i] *= exp(-eta * (g[i] - target) / target)
        ratio = (g - target) / (target + 1e-15)
        a = a * np.exp(-eta * np.clip(ratio, -5, 5))
        
        # Project: non-negativity and normalization
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            eta *= 0.5
            continue
        a = a / s

            if sc < best_sc - 1e-12:

                best_a = a.copy()
                patience = 0
                eta = min(eta * 1.02, 1.0)

                    eta = max(eta * 0.5, 0.05)
                    patience = 0
                    # Small perturbation

                    z += 0.01 * np.random.randn(n)
                    z -= z.mean()

            eta = max(eta * 0.95, 0.01)
    
    # Final check

    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc


def ls_obj(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def ls_grad(z, beta, n):

    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    g = 2.0 * corr_a_w(a, w, n)
    W = float(np.dot(w, c))
    # Gradient of the smoothed objective w.r.t. z
    # d/dz [2n * (c_max + beta * log(sum(exp((c-c_max)/beta))))]

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Active-Set Multiplicative Update (AS-MU) as the primary engine,
combined with L-BFGS log-sum-exp refinement and active-set Newton polish.
Key innovation: principled KKT-driven multiplicative updates on the simplex
with adaptive temperature annealing for the active set.
"""
import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def corr_a_w(a, w, n):
    """Compute h[i] = sum_j a[j] * w[i+j] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(w) - 1
    N = next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * fw, N)[:out_len]
    return result[n - 1:2 * n - 1]


def as_mu(a_init, deadline, max_iter=1000000, verbose=True):

    Active-Set Multiplicative Update.
    Drives the KKT condition: corr_a_w(a, w, n) should be constant on support.
    Uses softmax temperature annealing for the active set.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = score_fn(a)
    if verbose:
        print(f"    AS-MU start: score={best_sc:.10f}")

    tau = 0.08
    tau_decay = 0.99985
    tau_floor = 0.00005
    eta = 0.4
    patience = 0
    max_patience = 8000
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = full_convolve(a, a)
        c_max = float(c.max())

        # Soft active set
        shifted = (c - c_max) / (tau * c_max + 1e-15)
        w = np.exp(np.clip(shifted, -700, 0))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        h = corr_a_w(a, w, n)  # h[i] = sum_j a[j]*w[i+j]

        ratio = (h - target) / (target + 1e-15)

        s = a.sum()

            eta *= 0.5

def ls_obj(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def ls_grad(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = corr_a_w(a, w, n)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

            break

        est_time = 0.008 * max(1.0, n / 4096.0)
        max_iter = min(6000, max(50, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                lambda z_local: ls_obj(z_local, beta, n),
                z,
                jac=lambda z_local: ls_grad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 50,
                    'maxls': 80,
                }
            )
            a_new = softmax_fn(result.x)
            sc_new = score_fn(a_new)
            if sc_new < current_sc - 1e-15:
                if verbose and sc_new < current_sc - 1e-5:
                    print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}")
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a
