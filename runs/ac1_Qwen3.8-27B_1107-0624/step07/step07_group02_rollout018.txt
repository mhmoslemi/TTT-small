    """Solve min a^T H a s.t. a >= 0, sum(a) = 1
    where H[j,l] = 2*w[l-j] (Toeplitz)
    """
    a = a.copy()
    # Project onto sum(a) = 1
    a = a / a.sum()
    
    for iter in range(max_iter):
        # Compute gradient: g = H * a
        # H is Toeplitz with first column 2*w[-n+1:n] (or appropriate slicing)
        # Actually, (H*v)[j] = 2*sum_l w[l-j]*v[l] = 2*(w_corr_v)[j]
        # where w_corr_v[j] = sum_l w[l-j]*v[l] = correlation of w and v
        
        # Using FFT: correlation(w, v) = ifft(conj(fft(w_extended)) * fft(v_extended))
        # But w has length 2n-1 (the convolution length), and v has length n.
        # We need to be careful with the indexing.
        
        # Actually, let's use the structure from the previous code:
        # h = convolve(w, a_reversed)[n-1:2n-1]
        # This gives h[j] = sum_k w[k] * a_rev[n-1-k+j] = sum_k w[k] * a[k-j]
        # Which is exactly (1/2) * g[j]
        
        a_rev = a[::-1]
        h = _full_convolve(w, a_rev)
        grad = 2.0 * h[n-1:2*n-1]
        
        # Project gradient onto sum=0 subspace (to handle equality constraint)
        grad = grad - grad.mean()
        
        # Check convergence
        if np.linalg.norm(grad) < 1e-12:
            break
        
        # CG step (simplified: just use steepest descent with optimal step)
        # Optimal step: alpha = (grad^T grad) / (grad^T H grad)
        # H * grad:
        a_rev_g = grad[::-1]
        h_g = _full_convolve(w, a_rev_g)
        Hg = 2.0 * h_g[n-1:2*n-1]
        Hg = Hg - Hg.mean()  # Project to sum=0
        
        denom = np.dot(grad, Hg + 1e-15)
        if denom <= 0:
            alpha = 0.01
        else:
            alpha = np.dot(grad, grad) / denom
        
        alpha = min(alpha, 1.0)
        
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
    
    return a

"""
Optimized step function height sequence minimizer.
Combines Adam-accelerated log-sum-exp smoothing, CG-based active set polish
with Hankel-structured Hessian, adaptive tolerance scheduling, shrinkage,
and multi-n exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(z_init, beta, n, deadline, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for the log-sum-exp smoothed objective."""
    z = z_init.copy().astype(np.float64)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_z = z.copy()
    best_f = _smoobj(z, beta, n)
    f = best_f

    while time.monotonic() < deadline:
        t += 1
        grad = _smograd(z, beta, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z_new = z_new - z_new.mean()

        f_new = _smoobj(z_new, beta, n)
        if f_new < f - 1e-15:
            z = z_new
            f = f_new
            if f_new < best_f:
                best_f = f_new
                best_z = z_new.copy()
        else:
            # Reduce learning rate on stagnation
            lr *= 0.95
            if lr < 1e-6:
                break

        if t % 50 == 0 and time.monotonic() > deadline - 5:

            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        improved = False
        for _ in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_end, tol_frac * 0.75)
                lr *= 0.7
            if no_improve >= 300:
                a = best_a.copy()
                tol_frac = max(tol_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                if tol_frac <= tol_end * 1.1:
                    break
        else:
            if tol_frac > tol_end:
                tol_frac = max(tol_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.01)

    return best_a, _score(best_a)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 12))))

        class _Prob:
            def __init__(self, b, nn):
                self.b = b
                self.nn = nn
            def __call__(self, x):
                return _smoobj(x, self.b, self.nn)
            def grad(self, x):
                return _smograd(x, self.b, self.nn)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.004, 0.008, 0.012, 0.018, 0.025, 0.035, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t + 0.05 * np.sin(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _adapt_beta_schedule(c, max_c):
    """Create an adaptive beta schedule based on the convolution structure."""
    c_sorted = np.sort(c)[::-1]
    # Find natural clusters in the peak region
    n_c = len(c)
    thresholds = [0.5, 0.3, 0.1, 0.01]
    betas = []
    for thr in thresholds:
        idx = int(np.searchsorted(c_sorted, max_c * (1 - thr), side='left'))
        idx = min(idx, n_c - 1)
        if idx > 0:
            gap = c_sorted[0] - c_sorted[idx]
            betas.append(max(gap, max_c * 0.005))
    # Add very small betas for final polishing
    betas.extend([max_c * 0.002, max_c * 0.001, max_c * 0.0005,
                  max_c * 0.0002, max_c * 0.0001, max_c * 0.00005, max_c * 0.00002])
    # Sort descending and deduplicate
    betas = sorted(set(betas), reverse=True)
    return betas


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search combining L-BFGS, Adam, CG active set, and multi-n exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS main ===
    print(f"[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam refinement ===
    print(f"[P2] Adam refinement (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        max_c = float(c.max())
        beta_adam = max_c * 0.003
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        adam_deadline = min(deadline - 400, time.monotonic() + 120)
        z_opt, f_opt = _adam_optimize(z, beta_adam, n, adam_deadline, lr=0.003)
        a_adam = _softmax(z_opt)
        sc_adam = _score(a_adam)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"  NEW BEST: {best_score:.10f} [adam] t={time.monotonic()-t_start:.1f}s")
        # Second Adam pass with smaller beta
        if time.monotonic() < deadline - 350:
            beta_adam2 = max_c * 0.001
            z2 = np.log(np.maximum(best_seq, 1e-30))
            z2 = z2 - z2.mean()
            adam_deadline2 = min(deadline - 350, time.monotonic() + 80)
            z_opt2, f_opt2 = _adam_optimize(z2, beta_adam2, n, adam_deadline2, lr=0.002)
            a_adam2 = _softmax(z_opt2)
            sc_adam2 = _score(a_adam2)
            if sc_adam2 < best_score:
                best_seq = a_adam2.copy()
                best_score = sc_adam2
                print(f"  NEW BEST: {best_score:.10f} [adam2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: CG active set (coarse) ===
    print(f"[P3] CG active set coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        cg_deadline = min(deadline - 300, time.monotonic() + 150)
        a_cg, sc_cg = _cg_active_polish(best_seq, cg_deadline,
                                        tol_start=0.006, tol_end=0.0003,
                                        max_cg_iter=300, max_outer=400)
        if sc_cg < best_score:

        if sc_gd < best_score:

    print(f"[P5] CG active set fine (t={time.monotonic()-t_start:.1f}s)")

        cg_deadline2 = min(deadline - 150, time.monotonic() + 100)
        a_cg2, sc_cg2 = _cg_active_polish(best_seq, cg_deadline2,

    print(f"[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")

        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 120)

                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")
                if time.monotonic() < deadline - 60:
                    a_shr3, sc_shr3 = _cg_active_polish(best_seq,
                                                        min(deadline - 55, time.monotonic() + 40),
                                                        tol_start=0.001, tol_end=0.00002,
                                                        max_cg_iter=300, max_outer=300)
                    if sc_shr3 < best_score:
                        best_seq = a_shr3.copy()
                        best_score = sc_shr3
                        print(f"  NEW BEST: {best_score:.10f} [cg_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-n exploration ===
    print(f"[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192, 10240]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 40:
            continue
        a_init = _resample(best_seq, n_try)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=40.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _cg_active_polish(a_opt,
                                                min(deadline - 25, time.monotonic() + 25),
                                                tol_start=0.003, tol_end=0.0001,
                                                max_cg_iter=200, max_outer=200)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_cg] t={time.monotonic()-t_start:.1f}s")

    print(f"[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        restart_starts = [

            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("raised", _make_profile(n_best, "raised_cos")),
            ("quartic", _make_profile(n_best, "quartic_taper")),
            ("asym", _make_profile(n_best, "asym_ramp")),

                best_score = sc_opt

    print(f"[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        # CG with tightest tolerance

                                          tol_start=0.0005, tol_end=0.00001,
                                          max_cg_iter=500, max_outer=600)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_cg] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS with smallest beta
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())

                    self.nn = nn

                    return _smograd(x, self.b, self.nn)

            try:

                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-15, 'maxcor': 80})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)

                    best_score = sc_fin2

            except Exception:

        # Final Adam micro-refinement
        remaining2 = deadline - 1 - time.monotonic()
        if remaining2 > 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
