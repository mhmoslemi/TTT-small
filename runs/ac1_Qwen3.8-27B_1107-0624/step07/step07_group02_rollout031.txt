    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_fla = a[::-1]
    wc = _full_convolve(w, a_fla)
    h = wc[n - 1:2 * n - 1]  # T1[m] = sum_j a[j]*w[m+j]
    W = float(np.dot(a, h))
    return (4.0 * n / beta) * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _lse_obj(z, self.beta, self.n)

    def grad(self, z):
        return _lse_grad_correct(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(30, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.3, 0.15, 0.08, 0.04,
                                   0.02, 0.01, 0.005, 0.002,
                                   0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(50, int(remaining / (est_time * 10))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _projected_grad_descent(a_init, deadline, min_time=30.0):
    """Projected gradient descent in original space with LSE smoothing."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    c_cur = _full_convolve(a, a)
    c_max = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    beta = max(gap, c_max * 0.01)

    lr = 0.1
    stall = 0

    while time.monotonic() + min_time < deadline:
        c = _full_convolve(a, a)
        c_max_i = float(c.max())
        shifted = (c - c_max_i) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_exp = e.sum()
        if s_exp < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_exp

        # grad_c[m] = 2 * sum_j a[j] * w[m+j]
        a_fla = a[::-1]
        wc = _full_convolve(w, a_fla)
        h = wc[n - 1:2 * n - 1] * 2.0  # grad_c

        cur_sc = 2.0 * n * c_max_i
        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            beta *= 0.5
            lr *= 0.5
            stall = 0
            if beta < c_max * 1e-6:
                break

        if stall % 50 == 0 and stall > 0:
            beta *= 0.8
            lr = max(0.01, lr * 0.9)

        a_try = a - lr * h
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-12:
            lr *= 0.5
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < cur_sc - 1e-14:
            a = a_try
        else:
            lr *= 0.7

    return best_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.005,
                       tol_frac_end=0.00002, lr_init=0.08):
    """Active-set gradient descent with correct subgradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Gradient: grad_c[m] = 2 * sum_j a[j] * w[m+j]
        a_fla = a[::-1]
        wc = _full_convolve(w, a_fla)
        h = wc[n - 1:2 * n - 1] * 2.0
        W = float(np.dot(a, h))
        grad = a * (h - W)
        grad_t = grad - np.mean(grad)  # invariance to shifting (sum-preserving)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for _ in range(80):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-12:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.75
            if no_improve >= 500:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.4)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 10:
                    break
        else:
            if tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.96)
            lr = min(lr_init, lr * 1.03)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    """Try zeroing out small elements to improve score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.03
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = (t * (1 - t)) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "inv_u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.4 * (t * (1 - t)) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "noise":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent_polish(a, deadline, n_sweeps=5):
    """Polish by optimally adjusting pairs of elements that share a convolution peak."""
    n = len(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find top positions
        top_k = min(10, len(c))
        top_indices = np.argsort(c)[::-1][:top_k]
        for ki in top_indices:
            if time.monotonic() > deadline:
                break
            # c[ki] = sum_i a[i]*a[ki-i]
            # Try reducing the largest product pair
            valid_i = []
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    valid_i.append(i)
            if not valid_i:
                continue
            valid_i = np.array(valid_i)
            products = a[valid_i] * a[ki - valid_i]
            idx_max = valid_i[np.argmax(products)]
            idx_min = valid_i[np.argmin(products)]
            # Try shifting mass from idx_max to idx_min
            for frac in [0.01, 0.005, 0.002, 0.001]:
                delta = frac * a[idx_max]
                if delta < 1e-15:
                    break
                a_try = a.copy()
                a_try[idx_max] -= delta
                a_try[idx_min] += delta
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    a = a_try
                    best_sc = sc_try
                    best_a = a_try.copy()
                    break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with corrected gradients, projected gradient descent,
    multi-n warm-start, and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent if available
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}", flush=True)
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.10f}", flush=True)

    def _try_update(a_cand, label):
        nonlocal best_seq, best_score
        if a_cand is None:
            return
        sc = _score(a_cand)
        if sc < best_score - 1e-12:
            best_seq = a_cand.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(a_cand)}) [{label}] t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 1: L-BFGS on parent ===
    print(f"\n[P1] L-BFGS main on parent (t={time.monotonic()-t_start:.1f}s)", flush=True)
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    _try_update(a_main, "lbfgs_main")

    # === Phase 2: Projected gradient descent ===
    print(f"\n[P2] Projected GD (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 400:
        a_pgd = _projected_grad_descent(best_seq, min(deadline - 400, time.monotonic() + 200), min_time=100)
        _try_update(a_pgd, "proj_gd")

    # === Phase 3: L-BFGS again after PGD ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 350:
        a_main2 = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0)
        _try_update(a_main2, "lbfgs_r2")

    # === Phase 4: Active-set polish (coarse) ===
    print(f"\n[P4] Active-set coarse (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 250:
        a_pol, sc_pol = _active_set_polish(best_seq, min(deadline - 250, time.monotonic() + 150),
                                           tol_frac_start=0.01, tol_frac_end=0.0002, lr_init=0.06)
        _try_update(a_pol, "polish_coarse")

    # === Phase 5: Active-set polish (fine) ===
    print(f"\n[P5] Active-set fine (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 180:
        a_pol2, sc_pol2 = _active_set_polish(best_seq, min(deadline - 180, time.monotonic() + 120),
                                             tol_frac_start=0.003, tol_frac_end=0.00005, lr_init=0.03)
        _try_update(a_pol2, "polish_fine")

    # === Phase 6: Shrinkage + re-optimize ===
    print(f"\n[P6] Shrinkage (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 120:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 120)
        _try_update(a_shr, "shrinkage")
        if time.monotonic() < deadline - 80:
            a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=50.0)
            _try_update(a_shr2, "lbfgs_post_shrink")

    # === Phase 7: Multi-n warm-start ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)", flush=True)
    n_best = len(best_seq)
    n_candidates = sorted(set([2048, 4096, 6144, 8192, 10240, 12288, 16384]))
    # Sort by proximity to current best n for diversity
    n_sorted = sorted(n_candidates, key=lambda x: abs(x - n_best))

    for n_try in n_sorted:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_best:
            continue
        # Try a few different initializations at this n
        kinds = ["uniform", "tapered", "bimodal"]
        for kind in kinds:
            if time.monotonic() > deadline - 40:
                break
            if kind == "uniform":
                a_init = _resample(best_seq, n_try)
            else:
                a_init = _make_profile(n_try, kind, rng)
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=35.0)
            _try_update(a_opt, f"multi_n_{n_try}_{kind}")

    # === Phase 8: Diverse restarts ===
    print(f"\n[P8] Diverse restarts (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 50:
        n_cur = len(best_seq)
        restarts = [
            ("pert3", _perturb(best_seq, 0.03, rng)),
            ("pert5", _perturb(best_seq, 0.05, rng)),
            ("pert10", _perturb(best_seq, 0.10, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=5)),
            ("u_shape", _make_profile(n_cur, "u_shape", rng)),
            ("inv_u", _make_profile(n_cur, "inv_u_shape", rng)),
        ]
        for label, a_r in restarts:
            if time.monotonic() > deadline - 20:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=15.0)
            _try_update(a_opt, f"restart_{label}")

    # === Phase 9: Coordinate descent polish ===
    print(f"\n[P9] Coordinate descent polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 10:
        a_cd, sc_cd = _coordinate_descent_polish(best_seq, deadline - 5, n_sweeps=3)
        _try_update(a_cd, "coord_descent")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n[P10] Final polish (t={time.monotonic()-t_start:.1f}s)", flush=True)
    if time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 2,
                                           tol_frac_start=0.001, tol_frac_end=0.00001,
                                           lr_init=0.02)
        _try_update(a_fin, "final_polish")

        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00005
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 100, 'maxls': 50})
                a_fin2 = _softmax(result.x)
                _try_update(a_fin2, "final_lbfgs")
            except Exception:
                pass

    # Final validation using evaluate_sequence
    print(f"\n{'=' * 60}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(1024, "uniform")
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"FINAL (evaluate_sequence): {ev_score:.10f}", flush=True)
        except Exception as e:
            print(f"WARNING: evaluate_sequence failed: {e}", flush=True)
            ev_score = np.inf
        sc_final = _score(final)
        print(f"FINAL (internal): score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(1024, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
