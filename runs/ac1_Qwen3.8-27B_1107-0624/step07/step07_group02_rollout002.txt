import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):

    c_cur = _full_convolve(a_init, a_init)

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,

                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)

        prob = _LbfgsProb(beta, n)
        try:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

        except Exception:

def _mirror_descent(a_init, deadline, max_iter=50000):
    """Mirror descent with multiplicative updates and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    eta = 0.1  # initial step size
    patience = 0
    max_patience = 200
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Compute weights (softmax of convolution values)
        shifted = c - c_max
        # Use a wider temperature for smoother weights
        temp = c_max * 0.1
        w = np.exp(np.clip(shifted / temp, -700, 0))
        w /= w.sum()
        
        # Compute gradient-like quantity h[j] = sum_k w[k] * a[k-j]
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        
        # Multiplicative update
        h_mean = h.mean()
        # Clip to avoid extreme updates
        log_update = np.clip(-eta * (h - h_mean), -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Adaptive step size
        if patience == 0:
            eta = min(0.3, eta * 1.05)
        elif patience > 50:
            eta = max(0.01, eta * 0.9)
    
    return best_a, best_sc


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 400
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Use weighted active set (not uniform)
        active_vals = c[active]
        w_active = (active_vals - c_max + tol) / tol  # linear weight, [0, 1]
        w_active = np.maximum(w_active, 0.01)
        w_active /= w_active.sum()
        
        w = np.zeros_like(c)
        w[active] = w_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.75)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 8:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.02)

    return best_a, _score(best_a)


def _coordinate_descent(a_init, deadline, n_rounds=200):
    """Optimize subsets of coordinates at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    
    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        # Pick a random subset of indices to optimize
        n_sub = min(n, max(16, n // 64))
        indices = rng.choice(n, size=n_sub, replace=False)
        indices.sort()
        
        # Current score
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Try small perturbations of the selected coordinates
        a_sub = a[indices].copy()
        best_sub = a_sub.copy()
        best_sub_sc = None
        
        # L-BFGS on the subset
        def subset_obj(z_sub):
            a_trial = a.copy()
            a_trial[indices] = _softmax(z_sub) * a_sub.sum()
            a_trial = a_trial / a_trial.sum()
            c_trial = _full_convolve(a_trial, a_trial)
            return float(c_trial.max())
        
        z_sub = np.log(np.maximum(a_sub, 1e-30))
        z_sub = z_sub - z_sub.mean()
        
        try:
            result = minimize(subset_obj, z_sub, method='Nelder-Mead',
                             options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            if result.fun < c_max * (1 - 1e-10):
                a_new_sub = _softmax(result.x) * a_sub.sum()
                a[indices] = a_new_sub
                a = a / a.sum()
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_sc


def _thue_morse(n):
    """Generate Thue-Morse sequence of length n (values ±1)."""
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i // 2] ^ (i % 2)
    return 2 * t - 1  # Convert 0/1 to -1/+1


def _make_thue_morse_profile(n, alpha=0.7):
    """Create a non-negative profile based on Thue-Morse sequence."""
    t = _thue_morse(n)
    a = 1.0 + alpha * t
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_profile(n, kind="uniform"):

        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)
