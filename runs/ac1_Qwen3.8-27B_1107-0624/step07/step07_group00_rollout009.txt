    if k <= 1:
        return np.ones(1)

    P = G.T @ G

            return float(w @ P @ w)

            return 2.0 * P @ w

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        if patience > 1000:
            # Restart from best with small perturbation
            y = best_a + 0.001 * np.random.randn(n)
            y = _project_simplex(y)
            a = y.copy()
            t_k = 1.0
            patience = 0

        if it % cd_interval == 0 and time.monotonic() < deadline - 30:
            a_cd, sc_cd = _cd_polish(a, deadline - 5, max_rounds=20, top_k=5)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cd.copy()
                a = a_cd.copy()
                y = a.copy()
                t_k = 1.0

    return best_a, best_score


def _cd_polish(a, deadline, max_rounds=100, top_k=5):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        active = np.argsort(c)[::-1][:top_k * 2]

        for m in active:
            m = int(m)
            j_lo = max(0, m - n + 1)
            j_hi = min(m, n - 1)
            if j_lo > j_hi:
                continue
            js = list(range(j_lo, j_hi + 1))
            # Find the coordinate pair (p, q) that most reduces c[m]
            g_m = _grad_c_m(best_a, m)
            active_js = [j for j in js if best_a[j] > 1e-10]
            if len(active_js) < 2:
                continue

            # Try scaling down the highest-contributing coordinate
            contribs = [best_a[j] * best_a[m - j] for j in active_js if 0 <= m - j < n]
            if not contribs:
                continue
            j_max = active_js[np.argmax(contribs)]

            # Try moving mass from j_max to other coordinates
            for scale in [0.8, 0.6, 0.4, 0.2]:
                a_cand = best_a.copy()
                delta = a_cand[j_max] * (1.0 - scale)
                a_cand[j_max] *= scale
                # Redistribute to coordinate with lowest gradient contribution
                j_alt_candidates = [j for j in range(n) if j != j_max and 0 <= m - j < n]
                if not j_alt_candidates:
                    continue
                g_alt = g_m[j_alt_candidates]
                j_alt = j_alt_candidates[np.argmin(np.abs(g_alt))]
                a_cand[j_alt] += delta
                s = a_cand.sum()
                if s < 1e-15:
                    continue
                a_cand /= s
                c_cand = _fft_convolve(a_cand, a_cand)
                sc = 2.0 * n * float(c_cand.max())
                if sc < best_score * (1.0 - 1e-8):
                    best_score = sc
                    best_a = a_cand.copy()
                    improved = True
                    break
        if not improved:
            # Try individual coordinate scaling
            for j in np.random.choice(n, min(n, 200), replace=False):
                if best_a[j] < 1e-12:
                    continue
                for scale in [0.7, 0.5, 1.3, 1.5]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-9):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    return best_a, best_score


def _qp_deflate(a, deadline, max_iters=50):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(best_a, best_a)
        c_max = float(c.max())
        m_star = int(np.argmax(c))

        # Find direction to reduce c[m_star]
        g = _grad_c_m(best_a, m_star)
        g_t = g - g.mean()

        # Line search for optimal step
        best_step = 0.0
        best_sc = best_score
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_cand = best_a - alpha * g_t
            a_cand = _project_simplex(a_cand)
            c_cand = _fft_convolve(a_cand, a_cand)
            sc = 2.0 * n * float(c_cand.max())
            if sc < best_sc:
                best_sc = sc
                best_step = alpha

        if best_step == 0.0:
            break
        best_a = best_a - best_step * g_t
        best_a = _project_simplex(best_a)
        best_score = best_sc

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _multi_start_optimize(a_init, deadline, n_starts=3, budget_per_start=None):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    rng = np.random.default_rng(12345)

    if budget_per_start is None:
        budget_per_start = max(10.0, (deadline - time.monotonic()) / n_starts)

    for s in range(n_starts):
        if time.monotonic() > deadline - 5:
            break
        if s == 0:
            a_cur = a_init.copy()
        else:
            sigma = 0.01 + 0.02 * s
            z = np.log(np.maximum(a_init, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_cur = np.exp(z)
            a_cur = _project_simplex(a_cur)

        sub_dl = min(deadline - 3, time.monotonic() + budget_per_start)
        a_opt, sc = _nesterov_optimize(a_cur, sub_dl, max_iter=50000,
                                       top_k=8, c0=0.3,
                                       steepest_interval=100, cd_interval=300)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Multi-start [{s}]: score={sc:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 8192
        a = np.ones(n) / n
        print("Using default uniform initialization")

    n_parent = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n_parent}, score={best_score:.10f}")

    # Phase 1: Nesterov on parent size (0 - 400s)
    phase1_dl = min(deadline, t_start + 400.0)
    a_opt, sc = _nesterov_optimize(a, phase1_dl, max_iter=300000,
                                   top_k=10, c0=0.5,
                                   steepest_interval=50, cd_interval=400)
    full_sc = _score(a_opt)
    print(f"Phase 1 (n={n_parent}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a_opt.copy()

    # Phase 2: QP deflation + CD (400s - 550s)
    phase2_dl = min(deadline, t_start + 550.0)
    if time.monotonic() < phase2_dl:
        a_def, sc_def = _qp_deflate(best_a, phase2_dl, max_iters=100)
        full_sc_def = _score(a_def)
        print(f"Phase 2 deflation: score={full_sc_def:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_def) and full_sc_def < best_score:
            best_score = full_sc_def
            best_a = a_def.copy()

    phase2b_dl = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2b_dl:
        a_cd, sc_cd = _cd_polish(best_a, phase2b_dl, max_rounds=50, top_k=8)
        full_sc_cd = _score(a_cd)
        print(f"Phase 2b CD: score={full_sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_cd) and full_sc_cd < best_score:
            best_score = full_sc_cd
            best_a = a_cd.copy()

    # Phase 3: Try larger n (600s - 1100s)
    phase3_dl = min(deadline, t_start + 1100.0)
    n_large = max(2 * n_parent, 8192)
    if n_large <= 16384 and time.monotonic() < phase3_dl:
        a_large = _interp_to_length(best_a, n_large)
        sub_dl = min(phase3_dl, time.monotonic() + 300.0)
        a_large_opt, sc_large = _nesterov_optimize(a_large, sub_dl, max_iter=150000,
                                                   top_k=10, c0=0.4,
                                                   steepest_interval=80, cd_interval=500)
        full_sc_large = _score(a_large_opt)
        print(f"Phase 3 n={n_large}: score={full_sc_large:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_large) and full_sc_large < best_score:
            best_score = full_sc_large
            best_a = a_large_opt.copy()

        # CD polish on large
        if time.monotonic() < phase3_dl - 20:
            a_large_cd, sc_lcd = _cd_polish(best_a, phase3_dl - 5, max_rounds=40, top_k=6)
            full_sc_lcd = _score(a_large_cd)
            print(f"Phase 3 CD n={n_large}: score={full_sc_lcd:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_lcd) and full_sc_lcd < best_score:
                best_score = full_sc_lcd
                best_a = a_large_cd.copy()

    # Phase 4: Try smaller n (1100s - 1350s)
    phase4_dl = min(deadline, t_start + 1350.0)
    if n_parent > 2048 and time.monotonic() < phase4_dl:
        n_small = n_parent // 2
        a_small = _interp_to_length(best_a, n_small)
        sub_dl = min(phase4_dl, time.monotonic() + 200.0)
        a_small_opt, sc_small = _nesterov_optimize(a_small, sub_dl, max_iter=100000,
                                                   top_k=8, c0=0.4,
                                                   steepest_interval=60, cd_interval=300)
        full_sc_small = _score(a_small_opt)
        print(f"Phase 4 n={n_small}: score={full_sc_small:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_small) and full_sc_small < best_score:
            best_score = full_sc_small
            best_a = a_small_opt.copy()

    # Phase 5: Multi-start on best n (1350s - 1600s)
    phase5_dl = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase5_dl:
        budget_ms = (phase5_dl - time.monotonic()) / 3.0
        a_ms, sc_ms = _multi_start_optimize(best_a, phase5_dl, n_starts=3,
                                            budget_per_start=budget_ms)
        full_sc_ms = _score(a_ms)
        print(f"Phase 5 multi-start: score={full_sc_ms:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_ms) and full_sc_ms < best_score:
            best_score = full_sc_ms
            best_a = a_ms.copy()

    # Phase 6: Final intensive polish (1600s - deadline)
    final_dl = deadline - 5.0
    if time.monotonic() < final_dl:
        # Nesterov with small c0 for fine-tuning
        a_fin, sc_fin = _nesterov_optimize(best_a, final_dl - 20, max_iter=200000,
                                           top_k=5, c0=0.1,
                                           steepest_interval=30, cd_interval=200)
        full_sc_fin = _score(a_fin)
        print(f"Phase 6 fine: score={full_sc_fin:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_fin) and full_sc_fin < best_score:
            best_score = full_sc_fin
            best_a = a_fin.copy()

        # QP deflation
        if time.monotonic() < final_dl - 10:
            a_def2, sc_def2 = _qp_deflate(best_a, final_dl - 5, max_iters=50)
            full_sc_def2 = _score(a_def2)
            if np.isfinite(full_sc_def2) and full_sc_def2 < best_score:
                best_score = full_sc_def2
                best_a = a_def2.copy()

        # Final CD
        if time.monotonic() < final_dl - 2:
            a_cd_f, sc_cd_f = _cd_polish(best_a, final_dl, max_rounds=30, top_k=4)
            full_sc_cd_f = _score(a_cd_f)
            if np.isfinite(full_sc_cd_f) and full_sc_cd_f < best_score:
                best_score = full_sc_cd_f
                best_a = a_cd_f.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
        s = float(final.sum())
    # Scale to a reasonable magnitude (keep sum around 1 for the score to be meaningful)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, sum={s:.4f}, t={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
