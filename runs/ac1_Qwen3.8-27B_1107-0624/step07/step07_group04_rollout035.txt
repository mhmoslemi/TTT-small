"""
Active-set KKT Newton optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direct Newton on the active-set multipliers with matrix-free Hankel CG solve,
adaptive active set expansion, KKT verification, and multi-n exploration.
Key differentiator from prior approach: exact KKT satisfaction via Newton on lambda,
non-symmetric profile exploitation, and tighter convergence criteria.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _hankel_matvec(v, W, n):
    """Compute H*v where (H)_{j,m} = 2*W[j+m], W has length 2n-1."""
    V = v[::-1]
    out_len = len(V) + len(W) - 1
    N = _next_pow2(out_len)
    F_V = np.fft.rfft(V, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_V * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_solve(W, n, rhs, max_iter=500, tol=1e-14):
    """Solve H*x = rhs using CG with the Hankel matvec."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new):
            break
        if rs_new < tol * tol * rs0:
            rs = rs_new
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x


def _compute_c(a):
    return _full_convolve(a, a)


def _active_set_qp_newton(a_init, active_lags, deadline, max_newton=100, cg_tol=1e-13):
    """
    Solve the active set QP via Newton on multipliers lambda.
    Given active set A, find a (on simplex) and lambda (on simplex) such that:
      - H_w(a) = sum lambda_k H_k has eigenvector 1 (KKT gradient condition)
      - c[k] is equal for all k in A (active constraints tight)
    Returns optimized a and the common max value.
    """
    n = len(a_init)
    m = len(active_lags)
    if m == 0:
        return a_init, 0.0
    if m == 1:
        c = _compute_c(a_init)
        return a_init, float(c[active_lags[0]])

    # Initial lambda: uniform
    lam = np.ones(m) / m
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, 0.0
    a = a / s

    best_a = a.copy()
    best_obj = _score(a)

    for newton_iter in range(max_newton):
        if time.monotonic() > deadline:
            break

        # Build W from current lambda
        W = np.zeros(2 * n - 1)
        for i, k in enumerate(active_lags):
            W[k] = lam[i]

        # Solve H_w * u = 1 using CG
        u = _cg_solve(W, n, np.ones(n), max_iter=500, tol=cg_tol)
        s_val = float(np.sum(u))
        if abs(s_val) < 1e-20 or not np.isfinite(s_val):
            break

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break

        # Compute convolution
        c = _compute_c(a_new)
        t_val = float(c[active_lags[0]])

        # Check active constraint residuals
        residuals = np.array([float(c[k]) - t_val for k in active_lags])
        max_res = float(np.max(np.abs(residuals)))

        if max_res < 1e-12 * max(1.0, abs(t_val)):

            sc = _score(a)

        # And we need a * da/dlam_j dot products

        # da/dlam_j = (-v_j + u * (1^T v_j)) / s^2
        # where v_j = H_w^{-1} (H_j u)

        # (H_j u)_m = 2 * u[k_j - m] (for valid indices)
        # This is a flip of u

        for j in range(m - 1):
            k_j = active_lags[j + 1]  # j+1 because we use 0-indexed for the m-1 off-ref
            Hju = np.zeros(n)
            valid = (k_j >= 0) & (k_j < 2 * n - 1)
            if valid:
                # (H_j u)_m = 2 * u[k_j - m] for 0 <= k_j - m <= n-1
                lo = max(0, k_j - (n - 1))

            one_vj = float(np.sum(v_j))

            for i in range(m - 1):
                k_i = active_lags[i + 1]
                # a^T (H_{k_i} - H_{k_0}) da/dlam_j
                # = 2 * a[ref_k - ...] ... let's just compute directly
                # a^T H_k v = 2 * sum_m a[m] * v[k-m] = 2 * (cross-corr of a and v at lag k)
                # For simplicity, compute via the c-like formula
                # a^T H_k v = 2 * sum_{i: 0<=k-i<n} a[i] * v[k-i]

                def _hv(a_vec, v_vec, k_val, n_val):

                        return 0.0

                aT_Hk0_da = _hv(a_new, da_dlam_j, ref_k, n)
                J[i, j] = 2.0 * (aT_Hki_da - aT_Hk0_da)
                F_vec[i] = float(c[k_i]) - t_val

        try:
            dlam = np.linalg.solve(J + 1e-15 * np.eye(m - 1), -F_vec)
        except np.linalg.LinAlgError:
            try:
                dlam = np.linalg.lstsq(J, -F_vec, rcond=None)[0]
            except Exception:
                break

        # Damped update
        alpha = 1.0
        for _ in range(20):
            lam_trial = lam.copy()
            lam_trial[1:] += alpha * dlam
            lam_trial = np.maximum(lam_trial, 0)
            ls = lam_trial.sum()
            if ls < 1e-15:
                alpha *= 0.5
                continue

                W_trial[k] = lam_trial[i]

            st = a_trial.sum()
            if st < 1e-15:

                continue
            a_trial = a_trial / st
            c_trial = _compute_c(a_trial)
            t_trial = float(c_trial[active_lags[0]])
            res_trial = float(np.max(np.abs(c_trial[active_lags] - t_trial)))
            if res_trial < max_res * 0.5 or alpha < 0.01:
                lam = lam_trial
                break
            alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_obj:
            best_obj = sc
            best_a = a.copy()

def _kkt_check(a, deadline_check=5.0):
    """Check KKT conditions. Returns (is_kkt, violated_lags)."""
    n = len(a)
    c = _compute_c(a)
    c_max = float(c.max())
    tol = 1e-10 * c_max
    active = c >= (c_max - tol)
    # Check gradient condition: the weighted sum of active gradients should be uniform
    # (approximately). This is hard to check exactly without the multipliers.
    # Instead, just return the active set.
    return active, c, c_max


def _refine_active_set(a, c, c_max, expansion_ratio=0.002, max_expand=200):
    """Adaptively determine the active set."""
    n = len(a)
    tol = expansion_ratio * c_max
    active = np.where(c >= (c_max - tol))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > max_expand:
        # Keep only the top max_expand
        c_active = c[active]
        top_idx = np.argsort(c_active)[::-1][:max_expand]
        active = active[top_idx]
    return active


def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS on the log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _compute_c(a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap * 0.5, max_c * 0.005)

    ratios = np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
                       0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    betas = base_beta * ratios

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    def smoobj(z_local):
        a_s = _softmax(z_local)
        c_s = _compute_c(a_s)
        cm = float(c_s.max())
        shifted = (c_s - cm) / beta_local_holder[0]
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta_local_holder[0] * np.log(s))

    def smograd(z_local):
        a_s = _softmax(z_local)
        c_s = _compute_c(a_s)
        cm = float(c_s.max())
        shifted = (c_s - cm) / beta_local_holder[0]
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c_s)
            w[int(np.argmax(c_s))] = 1.0
        else:
            w = e / s
        a_rev = a_s[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c_s))
        return 4.0 * n * a_s * (h - W)

    beta_local_holder = [betas[0]]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta_local_holder[0] = beta
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 10))))

        try:
            result = minimize(
                smoobj, z,
                jac=smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 100, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _newton_polish_v2(a_init, deadline, max_rounds=200):
    """
    Improved Newton polish with:
    - Adaptive active set size
    - Damped Newton with backtracking on lambda
    - KKT verification
    - Non-symmetric handling
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _compute_c(a)
        c_max = float(c.max())

        # Determine active set (adaptive tightness)
        tightness = max(0.0001, 0.001 * (1.0 - rnd / max_rounds))
        active_lags = _refine_active_set(a, c, c_max, expansion_ratio=tightness, max_expand=300)
        m = len(active_lags)

        if m == 1:
            # Single active lag: just need a^T H_k a to be max
            # This is a Rayleigh quotient problem
            break

        # Newton on lambda
        lam = np.ones(m) / m
        converged = False

        for niter in range(80):
            if time.monotonic() > deadline:
                break

            W = np.zeros(2 * n - 1)
            for i, k in enumerate(active_lags):
                W[k] = lam[i]

            a_candidate = np.maximum(a_candidate, 0)

            max_res = float(np.max(np.abs(residuals)))

                converged = True
                a = a_candidate
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                break

            # If lambda is near boundary, use gradient step instead
            lam_min = float(lam.min())
            if lam_min < 1e-8:
                # Gradient step on log-lambda

                a = a_candidate
                continue

            # Compute J efficiently
            ref_k = active_lags[0]
            J = np.zeros((m - 1, m - 1))

            for j in range(m - 1):
                k_j = active_lags[j]  # j-th off-reference
                if j == 0:
                    k_j = active_lags[1]
                else:
                    k_j = active_lags[j + 1] if j + 1 < m else active_lags[-1]

                lo = max(0, k_j - (n - 1))

                one_vj = float(np.sum(v_j))

                for i in range(m - 1):
                    k_i = active_lags[i + 1]

                    else:
                        aT_i = 0.0
                    lo_0 = max(0, ref_k - (n - 1))
                    hi_0 = min(n - 1, ref_k)
                    if lo_0 <= hi_0:

                                                  da_dlam_j[ref_k - hi_0:ref_k - lo_0 + 1][::-1]))

            # Solve

                dlam = np.linalg.solve(J + 1e-12 * np.eye(m - 1), -residuals[:m - 1])
            except np.linalg.LinAlgError:
                try:
                    dlam = np.linalg.lstsq(J, -residuals[:m - 1], rcond=None)[0]
                except Exception:
                    break

            # Damped update
            alpha = 1.0
            for _ in range(15):
                lam_trial = lam.copy()
                lam_trial[1:] += alpha * dlam
                lam_trial = np.maximum(lam_trial, 1e-10)
                lam_trial = lam_trial / lam_trial.sum()

                W_trial = np.zeros(2 * n - 1)
                for i2, k in enumerate(active_lags):
                    W_trial[k] = lam_trial[i2]
                u_trial = _cg_solve(W_trial, n, np.ones(n), max_iter=100, tol=1e-10)
                s_t = float(np.sum(u_trial))

                    continue
                a_trial = np.maximum(u_trial / s_t, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    alpha *= 0.5
                    continue
                a_trial = a_trial / st
                c_trial = _compute_c(a_trial)
                t_tr = float(c_trial[active_lags[0]])
                res_tr = float(np.max(np.abs(c_trial[active_lags] - t_tr)))
                if res_tr < max_res * (1.0 - 0.1 * alpha) or alpha < 0.01:
                    lam = lam_trial
                    a = a_trial
                    break
                alpha *= 0.5
            else:
                a = a_candidate

        if converged:
            break
        if time.monotonic() > deadline:
            break

        # Check for improvement
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        # Asymmetric bump shifted to one side
        a = np.exp(-12.0 * (t - 0.35) ** 2) + 0.02
        return a / a.sum()
    elif kind == "asymmetric2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.6) ** 2) * 0.5 + 0.01
        return a / a.sum()
    elif kind == "one_sided":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t * 0.8) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _subgrad_refine(a_init, deadline, max_steps=3000):
    """Subgradient method with dynamic active set for global refinement."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.01
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _compute_c(a)
        c_max = float(c.max())

        # Weighted gradient using top active lags
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.0001 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 100 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.03)
            else:
                patience += 1
                if patience > 50:
                    lr *= 0.7
                    if lr < 0.001:
                        break
            z = z_new
        else:
            z = z_new

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining active-set KKT Newton, L-BFGS smoothing,
    subgradient refinement, multi-n exploration, and asymmetric profile search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 8192
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent (asymmetric perturbation to break symmetry)
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    # Slight asymmetry perturbation
    a_asym = _perturb(parent, 0.01, rng)
    a1 = _lbfgs_smooth(a_asym, deadline, min_time_remaining=400.0)
    update_best(a1, _score(a1), "lbfgs_asym")

    # Also try from parent directly
    if time.monotonic() < deadline - 350:
        a1b = _lbfgs_smooth(parent, deadline, min_time_remaining=350.0)
        update_best(a1b, _score(a1b), "lbfgs_parent")

    # Phase 2: KKT Newton polish
    print(f"\n[Phase 2] KKT Newton polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a2, sc2 = _newton_polish_v2(best_seq, min(deadline - 200, time.monotonic() + 120), max_rounds=150)
        update_best(a2, sc2, "kkt_newton_1")

    # Phase 3: Subgradient for global moves
    print(f"\n[Phase 3] Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a3, sc3 = _subgrad_refine(best_seq, min(deadline - 150, time.monotonic() + 80), max_steps=2000)
        update_best(a3, sc3, "subgrad_1")

    # Phase 4: Another Newton round after subgradient
    print(f"\n[Phase 4] KKT Newton 2 (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a4, sc4 = _newton_polish_v2(best_seq, min(deadline - 60, time.monotonic() + 60), max_rounds=150)
        update_best(a4, sc4, "kkt_newton_2")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [4096, 16384, 2048]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if abs(n_try - n_parent) < 100:
            continue
        a_init = _resample(parent, n_try)
        a_opt = _lbfgs_smooth(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_np, sc_np = _newton_polish_v2(a_opt, min(deadline - 30, time.monotonic() + 30), max_rounds=80)
            update_best(a_np, sc_np, f"n_{n_try}_kkt")

    # Phase 6: Diverse restarts with asymmetric profiles
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = len(best_seq)
        profiles = ["asymmetric", "asymmetric2", "one_sided", "tapered"]
        for pi, prof in enumerate(profiles):
            if time.monotonic() > deadline - 20:
                break
            if pi < 2:
                p_start = _make_profile(n_best, prof)
            else:
                p_start = _perturb(best_seq, 0.05 * (pi + 1), rng)
            a_r = _lbfgs_smooth(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{prof}")

            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 15:
                a_rp, sc_rp = _newton_polish_v2(a_r, min(deadline - 10, time.monotonic() + 15), max_rounds=60)
                update_best(a_rp, sc_rp, f"restart_{prof}_kkt")

    # Phase 7: Final ultra-fine KKT polish
    print(f"\n[Phase 7] Final KKT polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _newton_polish_v2(best_seq, deadline - 2, max_rounds=300)
        update_best(a_fin, sc_fin, "final_kkt")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
