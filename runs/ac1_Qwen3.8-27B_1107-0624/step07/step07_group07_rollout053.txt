"""
Step sequence optimizer using:
1. Coarse-to-fine L-BFGS with log-sum-exp smoothing
2. Tight active-set gradient descent with backtracking line search and simplex projection
3. Ultra-fine L-BFGS polish with very small beta
4. Peak surgery heuristic for local improvements
5. Multi-restart diversity
Target: minimize 2*n*max(conv(a,a))/sum(a)^2
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in Euclidean distance."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_phase(a_init, deadline, beta_scale, n_betas=15, max_iter_cap=5000):
    """L-BFGS with geometric beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_init

    betas = max_c * beta_scale * (0.5 ** np.arange(n_betas))
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        est_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(max_iter_cap, max(100, int(remaining / (est_time * 30))))

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 100,
                                       'maxls': 100, 'disp': False})
            a_out = _softmax(result.x)
            sc = _score(a_out)
            if sc < current_sc - 1e-15:
                current_a = a_out
                current_sc = sc
                print(f"    L-BFGS beta={beta:.2e}: score={sc:.10f}")
        except Exception:
            pass

    return current_a


def _active_set_gradient_descent(a_init, deadline, eps_start=0.005, eps_end=1e-7,
                                  max_peaks_start=200, max_peaks_end=20,
                                  eta_start=0.05, max_iters=200000):
    """
    Active-set gradient descent with backtracking line search.
    Tightens the active set over time for more precise peak reduction.
    """
    n = len(a_init)
    S = float(a_init.sum())
    a = a_init.copy() / S
    a = a * S  # keep sum = S
    best_a = a.copy()
    best_sc = _score(a)

    total_iters = max_iters
    for i in range(total_iters):
        if time.monotonic() > deadline:
            break

        # Progress-based tightening of parameters
        progress = min(1.0, i / (total_iters * 0.8))
        eps = eps_start * (eps_end / eps_start) ** progress
        max_peaks = max(max_peaks_end, int(max_peaks_start * (1 - progress) + max_peaks_end * progress))

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Tight active set
        active = c >= (c_max - eps * c_max)
        n_active_check = int(active.sum())
        if n_active_check > max_peaks:
            idx = np.argsort(c)[::-1][:max_peaks]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True

        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Compute gradient of max over active set (uniform weights)
        # grad c_k: (grad)_j = 2*a[k-j] if 0 <= k-j < n
        # h[j] = sum_{k in active} a[k-j] for j=0..n-1
        # Using cross-correlation: h[j] = sum_i a[i] * active[i+j]
        N = _next_pow2(n + len(active) - 1)
        A_ft = np.fft.fft(a, N)
        Act_ft = np.fft.fft(active.astype(np.float64), N)
        h = np.fft.ifft(np.conj(A_ft) * Act_ft).real[:n]
        g = (2.0 / n_active) * h

        # Project gradient to tangent space (zero mean)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        g = g / g_norm

        # Backtracking line search
        best_step = 0
        best_a_try = a
        best_sc_try = _score(a)
        eta = eta_start

        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * frac
            a_try = a - step * g
            a_try = _project_simplex(a_try, S)
            sc_try = _score(a_try)
            if sc_try < best_sc_try - 1e-15:
                best_sc_try = sc_try
                best_a_try = a_try
                best_step = step

        if best_step > 0:
            a = best_a_try
            if best_sc_try < best_sc - 1e-14:
                best_sc = best_sc_try
                best_a = a.copy()
            eta = min(eta * 1.02, 0.2)
        else:
            eta *= 0.7
            if eta < 1e-10:
                break

    return best_a, best_sc


def _peak_surgery(a_init, deadline, max_rounds=5000):
    """
    Heuristic peak reduction: identify top contributing pairs for each peak
    and transfer mass from them to less-critical positions.
    """
    n = len(a_init)
    S = float(a_init.sum())
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find top contributors to this peak
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i] for valid i
        lo = max(0, peak_idx - n + 1)
        hi = min(peak_idx, n - 1)
        contributors = []
        for i in range(lo, hi + 1):
            j = peak_idx - i
            contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        if not contributors:
            break

        # Try reducing the top contributor
        prod_top, i_top, j_top = contributors[0]
        if prod_top < 1e-15:
            break

        # Transfer a small amount from the larger of a[i_top], a[j_top]
        # to a position that minimizes contribution to active peaks
        if a[i_top] >= a[j_top]:
            src = i_top
        else:
            src = j_top

        # Find a good destination: position with low contribution to peaks
        # Score each position by how much it contributes to top peaks
        top_peaks = np.argsort(c)[::-1][:10]
        dest_scores = np.zeros(n)
        for pk in top_peaks:
            lo_p = max(0, pk - n + 1)
            hi_p = min(pk, n - 1)
            for pos in range(n):
                partner = pk - pos
                if 0 <= partner < n:
                    dest_scores[pos] += a[partner]

        # Avoid the source, pick position with lowest dest_score
        dest_scores[src] = np.inf
        dest = int(np.argmin(dest_scores))

        # Try several transfer amounts
        for frac in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            amt = a[src] * frac
            a_try = a.copy()
            a_try[src] -= amt
            a_try[dest] += amt
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            a_try = a_try * (S / s_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_sc


def _multi_perturb_restart(a_best, deadline, rng, n_configs=6, time_per=60):
    """Try diverse perturbations and optimize each briefly."""
    results = []
    n = len(a_best)
    time_per_actual = max(5, min(time_per, (deadline - time.monotonic()) / max(1, n_configs)))

    for i in range(n_configs):
        if time.monotonic() > deadline:
            break

        # Different perturbation strategies
        z = np.log(np.maximum(a_best, 1e-15))
        z = z - z.mean()

        if i % 3 == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.02 * (i // 3)
            z = z + scale * rng.standard_normal(n)
        elif i % 3 == 1:
            # Low-frequency Fourier perturbation
            for _ in range(3 + i):
                freq = int(rng.integers(1, max(2, n // 20)))
                amp = 0.005 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2 * np.pi)
                t = np.arange(n)
                z = z + amp * np.cos(2 * np.pi * freq * t / n + phase)
        else:
            # Block perturbation
            block = max(16, n // 32)
            for _ in range(4 + i):
                start = int(rng.integers(0, max(1, n - block)))
                z[start:start + block] += rng.uniform(-0.005, 0.005)

        z = z - z.mean()
        a_pert = _softmax(z)

        # Quick L-BFGS polish
        c_p = _full_convolve(a_pert, a_pert)
        max_p = float(c_p.max())
        betas = max_p * np.array([0.05, 0.01, 0.002, 0.0005, 0.0001])
        current = a_pert

        for beta in betas:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break
            z0 = np.log(np.maximum(current, 1e-15))
            z0 = z0 - z0.mean()

            def fun(z):
                return _smoobj(z, beta, n)

            def jac(z):
                return _smograd(z, beta, n)

            max_iter = min(2000, max(50, int(remaining / 0.01)))
            try:
                result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 50,
                                           'maxls': 50, 'disp': False})
                a_new = _softmax(result.x)
                if _score(a_new) < _score(current):
                    current = a_new
            except Exception:
                pass

        results.append(current)

    return results


def _coordinate_descent_polish(a_init, deadline, max_rounds=20000):
    """Pairwise coordinate descent for final polish."""
    n = len(a_init)
    S = float(a_init.sum())
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick two random indices
        i1 = int((rnd * 7919 + 13) % n)
        i2 = int((rnd * 6271 + 37) % n)
        if i1 == i2:
            i2 = (i2 + 1) % n

        s_pair = a[i1] + a[i2]
        if s_pair < 1e-15:
            continue

        # Try several splits
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7, 0.45, 0.55]:
            a_try = a.copy()
            a_try[i1] = s_pair * frac
            a_try[i2] = s_pair * (1.0 - frac)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses L-BFGS with log-sum-exp, tight active-set gradient descent,
    peak surgery, coordinate descent, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n0 = 8192
        a0 = np.ones(n0) / n0
        print(f"No parent. Using uniform n={n0}, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    n = len(best_seq)
    print(f"\nStarting optimization: n={n}, initial score={best_score_val:.10f}")

    # === Phase 1: Coarse L-BFGS (300s) ===
    print(f"\n=== Phase 1: Coarse L-BFGS (300s) ===")
    t_phase = time.monotonic()
    phase1_deadline = min(deadline, t_start + 300)
    a_p1 = _lbfgs_phase(best_seq, phase1_deadline, beta_scale=0.1, n_betas=10, max_iter_cap=3000)
    try_update(a_p1, "phase1_lbfgs")
    print(f"  Phase 1 done: {time.monotonic()-t_phase:.1f}s, score={_score(a_p1):.10f}")

    # === Phase 2: Active-set gradient descent (700s) ===
    print(f"\n=== Phase 2: Active-set gradient descent (700s) ===")
    t_phase = time.monotonic()
    phase2_deadline = min(deadline, t_start + 1000)
    if time.monotonic() < phase2_deadline:
        a_p2, sc_p2 = _active_set_gradient_descent(
            best_seq, phase2_deadline,
            eps_start=0.003, eps_end=1e-8,
            max_peaks_start=300, max_peaks_end=10,
            eta_start=0.03, max_iters=500000
        )
        try_update(a_p2, "phase2_active_set")
    print(f"  Phase 2 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 3: Fine L-BFGS (300s) ===
    print(f"\n=== Phase 3: Fine L-BFGS (300s) ===")
    t_phase = time.monotonic()
    phase3_deadline = min(deadline, t_start + 1300)
    if time.monotonic() < phase3_deadline:
        a_p3 = _lbfgs_phase(best_seq, phase3_deadline, beta_scale=0.005, n_betas=12, max_iter_cap=4000)
        try_update(a_p3, "phase3_fine_lbfgs")
    print(f"  Phase 3 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 4: Second active-set pass with tighter eps (200s) ===
    print(f"\n=== Phase 4: Tight active-set pass (200s) ===")
    t_phase = time.monotonic()
    phase4_deadline = min(deadline, t_start + 1500)
    if time.monotonic() < phase4_deadline:
        a_p4, sc_p4 = _active_set_gradient_descent(
            best_seq, phase4_deadline,
            eps_start=0.001, eps_end=1e-9,
            max_peaks_start=50, max_peaks_end=5,
            eta_start=0.01, max_iters=500000
        )
        try_update(a_p4, "phase4_tight_as")
    print(f"  Phase 4 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 5: Peak surgery (100s) ===
    print(f"\n=== Phase 5: Peak surgery (100s) ===")
    t_phase = time.monotonic()
    phase5_deadline = min(deadline, t_start + 1600)
    if time.monotonic() < phase5_deadline:
        a_p5, sc_p5 = _peak_surgery(best_seq, phase5_deadline, max_rounds=3000)
        try_update(a_p5, "phase5_surgery")
    print(f"  Phase 5 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 6: Ultra-fine L-BFGS (80s) ===
    print(f"\n=== Phase 6: Ultra-fine L-BFGS (80s) ===")
    t_phase = time.monotonic()
    phase6_deadline = min(deadline, t_start + 1680)
    if time.monotonic() < phase6_deadline:
        a_p6 = _lbfgs_phase(best_seq, phase6_deadline, beta_scale=0.0005, n_betas=8, max_iter_cap=3000)
        try_update(a_p6, "phase6_ultra")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 7: Coordinate descent polish (60s) ===
    print(f"\n=== Phase 7: Coordinate descent (60s) ===")
    t_phase = time.monotonic()
    phase7_deadline = min(deadline, t_start + 1740)
    if time.monotonic() < phase7_deadline:
        a_p7, sc_p7 = _coordinate_descent_polish(best_seq, phase7_deadline, max_rounds=50000)
        try_update(a_p7, "phase7_cd")
    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Phase 8: Final active-set micro-pass ===
    print(f"\n=== Phase 8: Final micro active-set ===")
    t_phase = time.monotonic()
    if time.monotonic() < deadline - 5:
        a_p8, sc_p8 = _active_set_gradient_descent(
            best_seq, deadline - 5,
            eps_start=1e-6, eps_end=1e-10,
            max_peaks_start=10, max_peaks_end=3,
            eta_start=0.003, max_iters=200000
        )
        try_update(a_p8, "phase8_micro")
    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s, score={_score(best_seq):.10f}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(8192)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(8192)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
