Step function coefficient optimizer using QP active-set method, adaptive n-search,

stronger global exploration via structured initializations.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    """Smooth approximation to 2n*max(c)/1 where c = softmax(z)*softmax(z)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hess_vec(v, active, n):
    """Compute H*v where H[i,j] = 2*[i+j in active]. Uses FFT."""
    N = _next_pow2(4 * n)
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active.astype(np.float64), N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve_matvec(matvec_fn, b, tol=1e-10, max_iter=300, reg=1e-10):
    """Conjugate gradient with regularization."""
    n = len(b)
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    bn = float(np.linalg.norm(b))
    if bn < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = matvec_fn(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        if not np.isfinite(alpha) or alpha < 0:
            break
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * bn:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _qp_active_set(a_init, active_lags, n, max_outer=50, max_inner=300, reg=1e-8):
    """Solve min a^T Q a s.t. sum(a)=1, a>=0 where Q[i,j]=[i+j in active_lags].
    Active-set method: iteratively fix negative variables to zero."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    free = np.ones(n, dtype=bool)
    
    for outer in range(max_outer):
        # Build the reduced system
        n_free = free.sum()
        if n_free == 0:
            break
        if n_free == 1:
            a[~free] = 0
            a[free] = 1.0
            break
        
        free_indices = np.where(free)[0]
        
        def matvec_reduced(v):
            """Compute Q_FF @ v where FF are the free-free block."""
            # v has length n_free, corresponds to a[free]
            full_v = np.zeros(n)
            full_v[free_indices] = v
            result_full = _hess_vec(full_v, active_lags, n)
            return result_full[free_indices] + reg * v
        
        # KKT: [2*Q_FF  1] [a_F] = [0]
        #       [1'^T    0] [lam] = [-1]
        # => 2*Q_FF * a_F = -lam * 1
        # => 1'^T a_F = 1
        # Solve: a_F = -(lam/2) * (2*Q_FF)^{-1} * 1 = -(lam/2) * Hinv(1)
        # where H = 2*Q_FF + reg*I (regularized)
        # lam = -2 / (1^T H^{-1} 1)
        
        ones_free = np.ones(n_free)
        Hinv_ones = _cg_solve_matvec(matvec_reduced, ones_free, tol=1e-12,
                                     max_iter=max_inner, reg=0.0)
        denom = float(np.dot(ones_free, Hinv_ones))
        
        if abs(denom) < 1e-30:
            break
        
        lam = -2.0 / denom
        a_F = lam * Hinv_ones
        
        # Place back
        a_new = np.zeros(n)
        a_new[free_indices] = a_F
        
        # Check non-negativity
        min_val = a_F.min()
        if min_val >= -1e-12:
            a[~free] = 0.0
            a[free_indices] = np.maximum(a_F, 0.0)
            a = a / a.sum() if a.sum() > 1e-15 else a
            break
        else:
            # Find the most negative free variable and fix it to 0
            worst_local = int(np.argmin(a_F))
            worst_global = free_indices[worst_local]
            free[worst_global] = False
            continue
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def _newton_active(a_init, deadline, max_iter=100, tol_frac=0.002):
    """Newton method with active set on the sum of active lag values."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = max(tol_frac * c_max, 1e-12)
        active = (c >= (c_max - tol)).astype(np.float64)
        if active.sum() == 0:
            active[int(np.argmax(c))] = 1.0
        
        # Gradient of sum_{k in active} c_k
        grad = _hess_vec(a, active, n)
        # Project to sum=0
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break
        
        # Newton step with regularization
        mu = 1e-6
        def hess_fn(v):
            return _hess_vec(v, active, n) + mu * v
        
        dx = _cg_solve_matvec(hess_fn, -grad_t, tol=1e-10, max_iter=200, reg=0.0)
        dx = dx - np.mean(dx)
        
        # Line search
        f_old = float(c[active.astype(bool)].sum())
        alpha = 1.0
        found = False
        for ls in range(40):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active.astype(bool)].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a = a_try
                found = True
                break
            alpha *= 0.5
        if not found:
            break
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time=30.0, beta_scale=1.0):
    """L-BFGS with log-sum-exp smoothing and beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    base_beta = max(gap, max_c * 0.01) * beta_scale
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01,
                                  0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 2:
            break
        
        est_time = 0.002 * max(1, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 6))))
        
        z_start = z.copy()
        
        def fun(zl):
            return _logsumexp_obj(zl, beta, n)
        
        def jac(zl):
            return _logsumexp_grad(zl, beta, n)
        
        try:
            result = minimize(fun, z_start, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 40, 'maxls': 50})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _subgrad_descent(a_init, deadline, max_steps=3000, lr=0.02, tol_frac=0.001):
    """Subgradient descent with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    patience = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        step = lr / (1.0 + 0.005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, 0.04)
            else:
                patience += 1
                lr *= 0.92
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = 0.015
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.5) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shaped":
        t = np.linspace(-1, 1, n)
        a = 1.0 + 0.8 * t ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "inv_u":
        t = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * t ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 3 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_pow":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.25) ** 2) / 0.02) + np.exp(-((t - 0.75) ** 2) / 0.02) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.standard_normal(n)
        # Smooth via FFT lowpass
        fa = np.fft.rfft(a)
        n_freq = len(fa)
        cutoff = n_freq // 8
        fa[cutoff:] = 0
        a = np.fft.irfft(fa, n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0.01, 1, n)
        a = t ** 0.5
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        mask = (t > 0.3) & (t < 0.7)
        a[mask] = 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate(a_old, n_new):
    x_old = np.linspace(0, 1, len(a_old))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-strategy optimizer: QP active-set + L-BFGS + subgradient
    with diverse initializations and adaptive n exploration."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    
    if parent is not None:
        sc_p = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        parent = _make_profile(8192, "uniform")
        print(f"No parent, uniform n=8192, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: QP active-set on parent ===
    print(f"\n[Phase 1] QP active-set on parent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        n_parent = len(parent)
        qd = min(deadline - 400, time.monotonic() + 120)
        c_p = _full_convolve(parent, parent)
        c_max_p = float(c_p.max())
        active_p = (c_p >= c_max_p - 0.003 * c_max_p)
        a_qp = _qp_active_set(parent, active_p, n_parent, max_outer=40, max_inner=200)
        sc_qp = _score(a_qp)
        update_best(a_qp, sc_qp, "qp_parent")
        if sc_qp < best_score * 0.999:
            # Chain: Newton from QP result
            a_qn, sc_qn = _newton_active(a_qp, min(deadline - 380, time.monotonic() + 60), max_iter=80)
            update_best(a_qn, sc_qn, "qp_parent_newton")
    
    # === Phase 2: L-BFGS with aggressive beta on parent ===
    print(f"\n[Phase 2] L-BFGS smooth on parent (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a_lb = _lbfgs_smooth(best_seq, deadline, min_time=300, beta_scale=0.8)
        sc_lb = _score(a_lb)
        update_best(a_lb, sc_lb, "lbfgs_parent")
    
    # === Phase 3: Multi-n exploration ===
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [4096, 8192, 16384]
    # Add parent n if not already there
    n_parent = len(parent)
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
        n_candidates.sort()
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        a_init = _interpolate(best_seq, n_try) if len(best_seq) != n_try else best_seq.copy()
        
        # L-BFGS
        rem = deadline - 100 - time.monotonic()
        if rem > 30:
            a_opt = _lbfgs_smooth(a_init, deadline, min_time=80, beta_scale=1.0)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"lbfgs_n{n_try}")
            
            # Newton polish
            if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 80:
                a_np, sc_np = _newton_active(a_opt, min(deadline - 80, time.monotonic() + 50), max_iter=60)
                update_best(a_np, sc_np, f"newton_n{n_try}")
    
    # === Phase 4: Diverse parametric initializations ===
    print(f"\n[Phase 4] Parametric inits (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq)
    profiles = ["u_shaped", "inv_u", "bimodal", "sine_pow", "cos_mod",
                "gaussian", "plateau", "power_law", "random_smooth"]
    
    for kind in profiles:
        if time.monotonic() > deadline - 60:
            break
        a_init = _make_profile(n_best, kind, rng)
        rem = deadline - 50 - time.monotonic()
        if rem < 15:
            break
        # Quick L-BFGS
        a_opt = _lbfgs_smooth(a_init, deadline, min_time=min(40, rem - 10), beta_scale=1.5)
        sc_opt = _score(a_opt)
        label = f"init_{kind}"
        if sc_opt < best_score * 1.01:
            update_best(a_opt, sc_opt, label)
            # Further polish if promising
            if sc_opt < best_score * 1.005 and time.monotonic() < deadline - 40:
                a_np, sc_np = _newton_active(a_opt, min(deadline - 40, time.monotonic() + 40), max_iter=50)
                update_best(a_np, sc_np, label + "_newton")
    
    # === Phase 5: Subgradient on best ===
    print(f"\n[Phase 5] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        a_sg, sc_sg = _subgrad_descent(best_seq, min(deadline - 40, time.monotonic() + 80),
                                       max_steps=2000, lr=0.015, tol_frac=0.001)
        update_best(a_sg, sc_sg, "subgrad")
    
    # === Phase 6: QP on best (with refined active set) ===
    print(f"\n[Phase 6] QP refined (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_b = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        c_max_b = float(c_b.max())
        # Try multiple active set widths
        for tol_f in [0.001, 0.003, 0.005, 0.01, 0.02]:
            if time.monotonic() > deadline - 30:
                break
            active_b = (c_b >= c_max_b - tol_f * c_max_b)
            a_qp2 = _qp_active_set(best_seq, active_b, n_b, max_outer=30, max_inner=150)
            sc_qp2 = _score(a_qp2)
            update_best(a_qp2, sc_qp2, f"qp_tol{tol_f}")
    
    # === Phase 7: Newton chain on best ===
    print(f"\n[Phase 7] Newton chain (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        for tol_f in [0.001, 0.002, 0.005]:
            if time.monotonic() > deadline - 25:
                break
            a_nt, sc_nt = _newton_active(best_seq, min(deadline - 25, time.monotonic() + 50),
                                         max_iter=80, tol_frac=tol_f)
            update_best(a_nt, sc_nt, f"newton_tol{tol_f}")
    
    # === Phase 8: Perturbation restarts ===
    print(f"\n[Phase 8] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)")
    for i in range(5):
        if time.monotonic() > deadline - 20:
            break
        sigma = 0.02 * (1 + 0.3 * i)
        a_pert = _perturb(best_seq, sigma, rng)
        rem = deadline - 15 - time.monotonic()
        if rem < 10:
            break
        a_opt = _lbfgs_smooth(a_pert, deadline, min_time=min(15, rem - 5), beta_scale=1.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"restart_{i}")
    
    # === Phase 9: Ultra-fine final polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        n_f = len(best_seq)
        c_f = _full_convolve(best_seq, best_seq)
        mb = float(c_f.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 5e-5
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter = min(2000, max(50, int(remaining / 0.005)))
            
            def fun_f(zl):
                return _logsumexp_obj(zl, beta_final, n_f)
            
            def jac_f(zl):
                return _logsumexp_grad(zl, beta_final, n_f)
            
            try:
                result = minimize(fun_f, z, jac=jac_f, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-15, 'maxcor': 40})
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                update_best(a_fin, sc_fin, "final_ultra")
            except Exception:
                pass
        
        # One more Newton
        if time.monotonic() < deadline - 3:
            a_fn, sc_fn = _newton_active(best_seq, max(deadline - 3, time.monotonic()),
                                         max_iter=50, tol_frac=0.001)
            update_best(a_fn, sc_fn, "final_newton")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
