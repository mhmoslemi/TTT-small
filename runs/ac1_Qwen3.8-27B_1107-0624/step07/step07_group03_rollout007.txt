        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsgrad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _corr_a_A(a, A_vec, n):
    """Compute g[i] = sum_j a[j] * A_vec[i+j] for i=0..n-1 via FFT."""
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]


def _active_set_newton(a_init, deadline, max_iter=500, tol_start=0.01, tol_end=1e-4):
    """Damped Newton method on the active set with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive tolerance
        tol = tol_start + (tol_end - tol_start) * (it / max_iter)
        tol = max(tol, 1e-6)

        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Cap active set size
        max_active = min(3 * n, len(c))
        if n_active > max_active:
            active = np.zeros_like(c, dtype=bool)
            top_idx = np.argsort(c)[::-1][:max_active]
            active[top_idx] = True
            n_active = max_active

        A_vec = active.astype(np.float64)
        m = n_active

        # Gradient of f(a) = (1/m) * sum_{k in active} c[k]
        # c[k] = sum_i a[i] * a[k-i], so dc[k]/da[i] = a[k-i] + a' = ... 
        # Actually, df/da[i] = (1/m) * sum_{k in active} (a[k-i] + a[|k-i|] ... )
        # No wait. c[k] = sum_j a[j]*a[k-j]. dc[k]/da[i] = a[k-i] + a[k-i] if k-i != i, else 2*a[i].
        # More precisely: dc[k]/da[i] = a[k-i] + a[k-i]... no.
        # c[k] = sum_j a[j]*a[k-j]
        # dc[k]/da[i] = a[k-i] (from the first factor when j=i) + a[k-i] (from the second factor when k-j=i, i.e., j=k-i)
        # But if k-i = i, i.e., k = 2i, then it's 2*a[i].
        # Actually: dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (if both terms exist)
        # Wait, c[k] = sum_j a[j]*a[k-j]. 
        # dc[k]/da[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] + a[k-i] (if k-i is in range for both)
        # = 2*a[k-i] (if 0 <= k-i < n)
        # = a[k-i] (if only one term is valid)
        # = 0 (if neither is valid)
        
        # But for our purposes, using the "correlation" form:
        # df/da[i] = (1/m) * sum_{k in active} dc[k]/da[i]
        #           = (2/m) * sum_{k in active} a[k-i]  (approximately, ignoring boundary effects)
        #           = (2/m) * (A * a^R)[i] where a^R is reversed
        # This is what _corr_a_A computes.

        g = (2.0 / m) * _corr_a_A(a, A_vec, n)
        g_t = g - np.mean(g)  # Project to tangent space (sum a = 1)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break

        # Hessian of f on the active set:
        # H[i,j] = (2/m) * A[i+j] + lambda * (delta[i,j] - 1/n) ... 
        # Actually H[i,j] = (2/m) * A_vec[i+j] (the second derivative of the average)
        # Plus we need to handle the equality constraint sum(a) = 1.
        # The reduced Hessian (projected to sum=0) is:
        # H_proj = H - (H*1)(1^T*H)/(1^T*H*1) ... this is getting complicated.
        # Let's just use the Hessian-vector product with CG.

        mu = 1e-8 + 1e-6 * c_max / max(m, 1)

        def hess_matvec(v):
            # H*v where H[i,j] = (2/m) * A_vec[i+j]
            # But we need the projected version. For CG on the tangent space,
            # we can use the unprojected H and handle the constraint separately.
            return (2.0 / m) * _corr_a_A(v, A_vec, n) + mu * v

        # CG solve for (H + mu*I) * u = g_t
        x = np.zeros(n)
        r = g_t - hess_matvec(x)
        p = r.copy()
        rs = np.dot(r, r)
        
        for cg_it in range(300):
            Ap = hess_matvec(p)
            pAp = np.dot(p, Ap)
            if pAp < 1e-30:
                break
            alpha = rs / pAp
            x += alpha * p
            r = r - alpha * Ap
            rs_new = np.dot(r, r)
            if np.sqrt(rs_new) < 1e-12 * max(1.0, np.linalg.norm(g_t)):
                break
            p = r + (rs_new / rs) * p
            rs = rs_new

        u = x  # Newton direction (approximately)
        
        # Handle the sum constraint: we need sum(dx) = 0
        # The Newton step should already approximately satisfy this if g_t is in the tangent space
        # but let's enforce it
        dx = -u + u.mean()  # This ensures sum(dx) = 0... wait, that's backwards
        # Actually: dx = -u, then project: dx = dx - dx.mean()
        dx = -u
        dx = dx - dx.mean()

        # Line search
        neg_mask = dx < -1e-15
        if neg_mask.any():
            t_max = float(np.min(a[neg_mask] / (-dx[neg_mask]))) * 0.99
        else:
            t_max = 1.0
        t_max = max(t_max, 1e-10)

        f_old = float(c.max())
        accepted = False
        t = min(1.0, t_max)
        for ls in range(50):
            a_try = a + t * dx
            if np.min(a_try) < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try < 1e-15:
                t *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            
            # Check improvement on the active set objective
            active_obj_old = float(np.sum(c[active])) / m
            active_obj_try = float(np.sum(c_try[active])) / m
            
            if f_try < f_old - 1e-12 * t * gnorm or ls == 49:
                if f_try <= f_old + 1e-10:
                    a = a_try
                    accepted = True
                    break
            t *= 0.5

        if not accepted:
            # Fallback: steepest descent
            t = min(0.05, t_max)
            for ls in range(30):
                a_try = a - t * g_t
                a_try = np.maximum(a_try, 0.0)
                s_try = float(a_try.sum())
                if s_try < 1e-15:
                    t *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13 * t * gnorm:
                    a = a_try
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

        # Convergence check
        c_new = _full_convolve(a, a)
        if abs(float(c_new.max()) - c_max) < 1e-13 * c_max:
            break

    return best_a, best_sc


def _lp_polish(a_init, deadline):
    """Solve the LP on the active set for exact optimum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 1e-4
    active = c >= (c_max - tol * c_max)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    m = int(active.sum())
    
    if m > 5 * n:
        active = np.zeros_like(c, dtype=bool)
        top_idx = np.argsort(c)[::-1][:5*n]
        active[top_idx] = True
        m = 5 * n

    # The LP is:
    # maximize sum(b)
    # subject to b >= 0
    #            (a * b)[k] <= M for all active k
    #            sum(b) > 0
    
    # Actually, let me reformulate. We want to find the best b on the active set.
    # The objective is to minimize max_{k in active} (a*b)[k] / sum(b)^2 ... no.
    
    # Let me think again. The original objective is 2n * max_k conv(a,a)[k] / (sum(a))^2.
    # After normalizing sum(a) = 1, it's 2n * max_k c[k].
    # On the active set, we want to minimize (1/m) * sum_{k in A} c[k] (approximation of max).
    # But for the LP approach from the SoTA:
    # Given current a and M = max(c), find b that maximizes sum(b) subject to a*b <= M.
    # Then update a = (1-t)*a + t*b_normalized.
    
    # Let me try the LP:
    # max 1^T b
    # st  (a * b)[k] <= c[k] for active k (use current c values as bounds)
    #     b >= 0
    #     sum(b) = 1 (normalization)
    
    # Wait, the SoTA says: maximize sum(b_j) subject to b >= 0 and ||f * g_0||_inf <= M
    # Then rescale g_0 to have the same sum as f.
    
    # So the LP is:
    # max 1^T b
    # st  (f * b)[k] <= M for k = 0,...,2n-2
    #     b >= 0
    # (no normalization constraint on b in the LP; we rescale after)
    
    # The constraint matrix C has entries C[k, j] = f[k-j] (with appropriate bounds)
    # (f * b)[k] = sum_j f[k-j] * b[j] (with f[i] = 0 outside [0, n-1])
    
    # For active constraints only:
    active_indices = np.where(active)[0]
    
    # Build constraint matrix (sparse)
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (f * b)[k] = sum_j a[k-j] * b[j]
    
    # This is a Toeplitz matrix. For n=8192 and m active constraints, 
    # the matrix is m x n. This might be too large for linprog.
    
    # Let me try with a smaller tolerance (fewer active constraints)
    tol_strict = 2e-4
    active_strict = c >= (c_max - tol_strict * c_max)
    if active_strict.sum() < active.sum() * 0.1:
        active_strict = active  # fallback
    
    active_indices = np.where(active_strict)[0]
    m = len(active_indices)
    
    if m > 3000:  # Too many constraints for linprog
        # Use only the top constraints
        c_vals = c[active_strict]
        top_m = np.argsort(c_vals)[::-1][:min(m, 2000)]
        active_indices = active_indices[top_m]
        m = len(active_indices)
    
    # Build the constraint matrix
    # (a * b)[k] = sum_{j=0}^{n-1} a[k-j] * b[j] (with a indexed 0..n-1)
    # C[i, j] = a[active_indices[i] - j] if 0 <= active_indices[i] - j < n
    
    # For efficiency, use the fact that each row is a shift of a
    # C is a submatrix of a Toeplitz matrix
    
    # Actually for linprog, we need c such that A_ub @ x <= b_ub
    # (a * b)[k] <= c[k] (the current values, not M)
    
    # Hmm, but this LP might not have a useful solution. Let me think...
    # The LP finds the direction of steepest ascent of sum(b) while keeping a*b bounded.
    # This is a standard approach in the SoTA.
    
    # Let me just try it with a reasonable number of constraints.
    if m < 50 or m > 5000:
        return a, _score(a)
    
    # Build C matrix (m x n)
    # C[i, j] = a[k_i - j] if 0 <= k_i - j < n
    # Using broadcasting would be too slow for large m and n.
    # Use the fact that C is a banded Toeplitz matrix.
    
    # For each active index k_i, the row is a reversed and shifted version of a.
    # C[i, :] = [0...0, a[k_i-(n-1)], ..., a[k_i], ..., 0...0]
    # More precisely: C[i, j] = a[k_i - j] for j = max(0, k_i-n+1) to min(k_i, n-1)
    
    # For n=8192 and m=2000, this is a 2000x8192 matrix. That's 16M elements.
    # Too large for dense linprog. Let's skip the LP for large n.
    
    if n > 4096 or m > 1000:
        return a, _score(a)
    
    try:
        C = np.zeros((m, n))
        for i, k in enumerate(active_indices):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1) + 1
            C[i, lo:hi] = a[k - (hi-1):k - lo]  # a[k-lo], a[k-lo-1], ..., a[k-(hi-1-lo)]
            # Actually: C[i, j] = a[k-j] for j in [lo, hi-1]
            # So C[i, lo:hi] = a[k-lo : k-(hi-1)] = a[k-lo : k-hi+1]
            # Reversed: a[k-(hi-1-lo) : ... : k-lo]
            # Let me just do it element by element for correctness
            C[i, lo:hi] = a[k - np.arange(lo, hi)]
        
        b_ub = c[active_indices]
        
        # linprog: min -1^T b st C @ b <= b_ub, b >= 0
        res = linprog(
            c=-np.ones(n),
            A_ub=C,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max(5, (deadline - time.monotonic()) * 0.8)}
        )
        
        if res.success and res.x is not None:
            b = res.x
            b = np.maximum(b, 0)
            s_b = b.sum()
            if s_b > 1e-10:
                b = b / s_b
                # Blend
                for t_blend in [0.3, 0.5, 0.7, 0.9]:
                    a_try = (1 - t_blend) * a + t_blend * b
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < _score(a):
                        a = a_try
                        break
    except Exception as e:
        print(f"  LP polish failed: {e}")
    
    return a, _score(a)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.03)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03,
                                  0.015, 0.007, 0.003, 0.001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _lsobj(z_local, beta, n),
                z,
                jac=lambda z_local: _lsgrad(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxcor': 100,
                    'maxls': 200,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-0.5 * t**2)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * t
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS -> Active-Set Newton (adaptive tol) -> LP polish -> Restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        _update_best(parent, _score(parent), "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update_best(parent, _score(parent), "init")

    # Phase 1: L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=400.0)
    _update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Active-set Newton with adaptive tolerance
    print(f"\n[Phase 2] Active-set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        as_deadline = min(deadline - 150, time.monotonic() + 180)
        a_as, sc_as = _active_set_newton(best_seq, as_deadline, max_iter=300, tol_start=0.01, tol_end=1e-4)
        _update_best(a_as, sc_as, "as_newton_1")

        # Second round with even tighter tolerance
        if time.monotonic() < deadline - 130:
            as_deadline2 = min(deadline - 100, time.monotonic() + 100)
            a_as2, sc_as2 = _active_set_newton(best_seq, as_deadline2, max_iter=200, tol_start=0.005, tol_end=5e-5)
            _update_best(a_as2, sc_as2, "as_newton_2")

    # Phase 3: LP polish
    print(f"\n[Phase 3] LP polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_lp, sc_lp = _lp_polish(best_seq, deadline - 80)
