"""
Optimized minimum peak convolution sequence finder.
Uses FFT-based Hankel matrix-vector products for efficient gradient computation,
mirror descent with Nesterov acceleration, L-BFGS with adaptive log-sum-exp smoothing,
and multi-scale exploration. Key improvements: exact HVP via FFT cross-correlation,
adaptive beta continuation, peak-targeting active set, and palindromic structure exploitation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def _hankel_mv(p, x, n):
    """Compute Q @ x where Q[i,j] = p[i+j] for i,j in [0,n).
    Uses FFT cross-correlation: y[i] = sum_j p[i+j]*x[j] = (p *~ x)[i]
    """
    N = _next_pow2(len(p) + len(x) - 1)
    P = np.fft.rfft(p, N)
    X = np.fft.rfft(x, N)
    corr = np.fft.irfft(P * np.conj(X), N)
    return corr[:n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v, target_sum=1.0):
    """Project v onto {a: sum(a)=target_sum, a>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        # Raised cosine with beta parameter
        beta = 1.0
        a = np.ones(n)
        mask1 = t < (1 - beta) / 2
        mask2 = t > (1 + beta) / 2
        mid = ~mask1 & ~mask2
        a[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * t[mask1] / beta - 1)))
        a[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * t[mask2] / beta - 1)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_like":
        t = np.linspace(-1, 1, n)
        a = np.sinc(t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _log_domain_grad(z, p, n, beta=None, c=None, a=None):
    """Compute gradient of the smoothed objective in log domain.
    If beta is None, use hard top-k (p is already the weight).
    Returns (grad_z, objective_value, a).
    """
    if a is None:
        a = _softmax(z)
    if c is None:
        c = _fft_convolve(a, a)
    
    # grad_a = 2 * hankel_mv(w, a, n) where w = p (the weights over lags)
    grad_a = 2.0 * _hankel_mv(p, a, n)
    # grad_z = a * (grad_a - dot(grad_a, a) * 1)
    val = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - val)
    return grad_z, val, a


def _run_mirror_descent(a_init, deadline, min_time_remaining=30.0):
    """Mirror descent with Nesterov momentum in log domain."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    vel = np.zeros(n)
    mu = 0.9
    
    for step in range(20000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive top-k active set
        k_act = min(max(3, int(2.0 * n / 1000)), len(c))
        top_k = np.argsort(c)[::-1][:k_act]
        p = np.zeros(2 * n - 1)
        p[top_k] = 1.0 / k_act
        
        grad_z, _, _ = _log_domain_grad(z, p, n, a=a, c=c)
        gn = np.linalg.norm(grad_z)
        if gn < 1e-18:
            break
        
        # Nesterov momentum
        vel = mu * vel + (1 - mu) * grad_z
        lr = 0.03 / (np.linalg.norm(vel) / gn + 1e-10)
        z_new = z - lr * vel
        z_new = np.clip(z_new, -40.0, 40.0)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if step % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs_ls(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing and adaptive beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.005)

              0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
              0.00002, 0.00001, 0.000005, 0.000002, 0.000001]
    
    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        beta = base_beta * ratio
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(2000, max(20, int(remaining / (est_time * 4))))
        
        try:
            def obj_func(z_local):
                a_local = _softmax(z_local)
                c_local = _fft_convolve(a_local, a_local)
                c_max_local = float(c_local.max())
                shifted = (c_local - c_max_local) / beta
                e = np.exp(np.clip(shifted, -700, 0))
                s_e = e.sum()
                if s_e < 1e-300:
                    return 2.0 * n * c_max_local
                return 2.0 * n * (c_max_local + beta * np.log(s_e))

            def grad_func(z_local):
                a_local = _softmax(z_local)
                c_local = _fft_convolve(a_local, a_local)

                    w = np.zeros_like(c_local)
                    w[int(np.argmax(c_local))] = 1.0
                else:
                    w = e / s_e
                grad_a = 2.0 * _hankel_mv(w, a_local, n)
                val = float(np.dot(grad_a, a_local))
                return a_local * (grad_a - val)

            result = minimize(
                obj_func, z,
                jac=grad_func,
                method='L-BFGS-B',

            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return best_a


def _run_peak_target(a_init, deadline, min_time_remaining=30.0):
    """Directly target and reduce the top peaks of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(500):
        if time.monotonic() + min_time_remaining > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        # Target the top peaks with emphasis
        peaks = c > c_max * 0.99
        if peaks.sum() == 0:
            peaks[int(np.argmax(c))] = True
        p = np.zeros(2 * n - 1)
        p[peaks] = 1.0 / peaks.sum()
        
        # Compute gradient and take a step
        grad_a = 2.0 * _hankel_mv(p, a, n)
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = 0.1 / (gnorm + 1e-10)
        improved = False
        for ls in range(20):
            a_new = _proj_simplex(a - step * grad_a, 1.0)
            c_new = _fft_convolve(a_new, a_new)
            if float(c_new.max()) < c_max * (1 - 1e-8):
                a = a_new
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5
            if step < 1e-14:
                break
        
        if not improved:
            # Try a different perturbation
            z = np.log(np.maximum(a, 1e-30))
            z = z + 0.01 * np.random.randn(n)

def _run_primal_dual_fft(a_init, deadline, min_time_remaining=30.0):
    """Primal-dual with FFT-based HVP for the QP primal step."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    c_cur = _fft_convolve(a, a)
    max_c = float(c_cur.max())
    base_beta = max_c * 0.01
    
    for outer in range(1000):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Dual update
        c = _fft_convolve(a, a)
        beta = base_beta * (0.7 ** min(outer, 100))
        p = _softmax(c / beta)
        
        # Primal step: minimize sum_k p_k * (a*a)[k] s.t. sum(a)=1, a>=0
        # Use CG in the interior (assume a > 0)
        # Q a = -lambda/2 * 1, so solve Q x = 1, then a = x/sum(x)
        # Q[i,j] = p[i+j]
        
        x = np.ones(n) / n
        r = np.ones(n) - _hankel_mv(p, x, n)
        p_cg = r.copy()
        rs = float(r @ r)
        
        for _ in range(50):
            Ap = _hankel_mv(p, p_cg, n)
            pAp = float(p_cg @ Ap)
            if abs(pAp) < 1e-15:
                break
            alpha = rs / pAp
            if alpha < 0:  # Q not PD, use gradient step instead
                break
            x += alpha * p_cg
            r -= alpha * Ap
            rs_new = float(r @ r)
            if rs_new < 1e-16 * rs:
                break
            p_cg = r + (rs_new / rs) * p_cg
            rs = rs_new
        
        if x.min() > 0:
            a_new = x / x.sum()
        else:
            # CG failed (Q not PD), use gradient step
            grad_a = 2.0 * _hankel_mv(p, a, n)
            lr = 0.05 / (np.linalg.norm(grad_a) + 1e-10)
            a_new = _proj_simplex(a - lr * grad_a, 1.0)
        
        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.5 * a + 0.5 * a_new
    
    return best_a, best_sc
