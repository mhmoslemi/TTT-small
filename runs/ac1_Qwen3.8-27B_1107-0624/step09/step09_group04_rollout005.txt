    """Run L-BFGS on the log-sum-exp smoothed objective with decreasing beta."""

        return a_init

    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]

                a_rev = a[::-1]

    return best_a, best_sc


def _asymmetric_perturb(a, rng, strength=0.03):
    """Asymmetric perturbation to break symmetry and escape local minima."""
    n = len(a)
    t = np.arange(n) / n
    ramp = 0.3 * (t - 0.5)
    noise = 0.7 * rng.standard_normal(n)
    perturbation = ramp + noise
    perturbation = perturbation - perturbation.mean()

    z = np.log(np.maximum(a, 1e-30))
    z = z + strength * perturbation
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    """Resample a sequence to a different length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS smoothed optimization,
    active-set peak equalization with momentum, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    n = len(best_seq)

    # === Phase 1: L-BFGS warmup ===
    print(f"\n[P1] L-BFGS warmup (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        a_p1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0, beta_scale=1.0)
        sc_p1 = _score(a_p1)
        print(f"  L-BFGS warmup: {sc_p1:.10f}")
        if sc_p1 < best_score:
            best_seq = a_p1.copy()
            best_score = sc_p1
            print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 2: Active-set equalization (main) ===
    print(f"\n[P2] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    pol2_deadline = min(deadline - 500, time.monotonic() + 500)
    a_eq, sc_eq = _active_set_equalize(best_seq, pol2_deadline, min_time_remaining=10.0)
    print(f"  Active-set: {sc_eq:.10f}")
    if sc_eq < best_score:
        best_seq = a_eq.copy()
        best_score = sc_eq
        print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 3: Asymmetric perturbation + re-optimize ===
    print(f"\n[P3] Asymmetric exploration (t={time.monotonic()-t_start:.1f}s)")
    for trial in range(3):
        if time.monotonic() > deadline - 400:
            break
        strength = 0.02 * (trial + 1)
        a_asym = _asymmetric_perturb(best_seq, rng, strength=strength)
        a_opt = _run_lbfgs(a_asym, deadline, min_time_remaining=400.0, beta_scale=0.5)
        sc_opt = _score(a_opt)
        print(f"  Asym trial {trial} (str={strength:.3f}): {sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n[P4] Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a_fine = _run_lbfgs(best_seq, deadline, min_time_remaining=300.0, beta_scale=0.3)
        sc_fine = _score(a_fine)
        print(f"  Fine L-BFGS: {sc_fine:.10f}")
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 5: Active-set round 2 ===
    print(f"\n[P5] Active-set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        pol5_deadline = min(deadline - 200, time.monotonic() + 200)
        a_eq2, sc_eq2 = _active_set_equalize(best_seq, pol5_deadline, min_time_remaining=10.0)
        print(f"  Active-set 2: {sc_eq2:.10f}")
        if sc_eq2 < best_score:
            best_seq = a_eq2.copy()
            best_score = sc_eq2
            print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([n, 16384, 32768, 65536]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        print(f"  n={n_try}: init={sc_init:.10f}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=100.0, beta_scale=0.7)
        sc_opt = _score(a_opt)
        print(f"  n={n_try}: after L-BFGS={sc_opt:.10f}")
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try})")
            # Quick active-set polish on the new n
            if time.monotonic() < deadline - 60:
                a_eq3, sc_eq3 = _active_set_equalize(best_seq, deadline - 60, min_time_remaining=10.0)
                if sc_eq3 < best_score:
                    best_seq = a_eq3.copy()
                    best_score = sc_eq3
                    print(f"  NEW BEST: {best_score:.10f} (n={n_try})")

    # === Phase 7: Subgradient polish ===
    print(f"\n[P7] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        sg_deadline = min(deadline - 25, time.monotonic() + 80)
        a_sg, sc_sg = _subgradient_polish(best_seq, sg_deadline, min_time_remaining=5.0)
        print(f"  Subgradient: {sc_sg:.10f}")
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  NEW BEST: {best_score:.10f}")

    # === Phase 8: Final ultra-fine ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 15:
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=5.0, beta_scale=0.1)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_equalize(best_seq, deadline - 2, min_time_remaining=2.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_activeset]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
