"""
Spectral flattening (Gerchberg-Saxton style) combined with L-BFGS refinement
and coordinate descent for minimum peak linear convolution sequences.
Key novelty: uses pychography-inspired alternating projections between
spatial domain (non-negativity) and frequency domain (flat magnitude)
to flatten a*a, then refines with gradient methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _gs_flatten(a, n_iter=500, alpha_start=0.1, alpha_end=0.9, gamma=None,
                deadline=None, min_time_remaining=5.0):
    """
    Gerchberg-Saxton style spectral flattening.
    Alternates between:
    - Spatial: enforce a >= 0, normalize
    - Frequency: flatten |FFT(a)| to make a*a more uniform
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break
        
        # Progressively increase aggressiveness
        t = it / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute desired magnitude (flatten)
        # Using power compression: mag_new = mag^(1-alpha) * mean_mag^alpha
        mean_mag = np.mean(mag)
        if mean_mag < 1e-15:
            break
        mag_new = (mag / mean_mag) ** (1.0 - alpha) * mean_mag
        
        if gamma is not None:
            # Additional compression
            mag_new = mag_new ** (1.0 / gamma) * (np.mean(mag_new) ** (1.0 - 1.0/gamma))
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        else:
            # If not improving, try the soft version (blend with old)
            a_new = 0.5 * a + 0.5 * a_new
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = _score(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
        
        a = a_new
    
    return best_a

    current value relative to the peak. This focuses reduction on the peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    N = _next_pow2(2 * n - 1)

            break

        mag = np.abs(A)
        phase = np.angle(A)
        
        # Current convolution (circular, which equals linear due to zero-padding)
        C = mag ** 2
        C_max = C.max()
        if C_max < 1e-15:
            break

        # Soft thresholding: reduce the top part of C

        weight[C > threshold] = 1.0 - 0.5 * (C[C > threshold] - threshold) / (C_max - threshold)

        mag_new = mag * np.sqrt(weight)

        target_energy = np.sum(mag ** 2)
        actual_energy = np.sum(mag_new ** 2)
        if actual_energy > 1e-15:
            mag_new *= np.sqrt(target_energy / actual_energy)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS optimization in log-space with adaptive beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.001)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 4))))

        try:
            result = minimize(
                _lbfgs_obj, z,
                jac=_lbfgs_grad,
                args=(n, beta),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                # Update base_beta based on new gap
                c_new = _full_convolve(a_new, a_new)
                max_c_new = float(c_new.max())
                c_sorted_new = np.sort(c_new)[::-1]
                gap_new = c_sorted_new[0] - c_sorted_new[min(10, len(c_sorted_new) - 1)]
                base_beta = max(gap_new * 0.1, max_c_new * 0.001)
        except Exception:
            pass

    return best_a


def _lbfgs_obj(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))


def _lbfgs_grad(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _coord_descent(a, deadline, min_time_remaining=10.0, max_rounds=100):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate j, c[k] = a[j]*(a[k-j] + a[j-k]) + rest, which is linear in a[j].
    So for each j, we can find the optimal a[j] analytically or by 1D search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    N = _next_pow2(2 * n - 1)
    
    for rnd in range(max_rounds):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        improved = False
        # Cycle through coordinates (random order for diversification)
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() + min_time_remaining > deadline:
                break
            
            # Compute c as a function of a[j]
            # c[k] = a[j] * (a[k-j] if 0<=k-j<n) + a[j] * (a[j-k] if 0<=j-k<n and j!=k-j) + c_rest[k]
            # For each k, the coefficient of a[j] in c[k] is:
            #   coeff[k] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and k-j != j)
            # Actually: c[k] = sum_i a[i]*a[k-i]
            # d(c[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # But if k-j = j (i.e., k=2j), it's 2*a[j]... no wait.
            # c[k] = ... + a[j]*a[k-j] + ... + a[k-j]*a[j] + ...
            # If k-j != j, these are two separate terms, so d(c[k])/d(a[j]) = a[k-j] + a[j-k]
            # If k-j = j (k=2j), it's a[j]^2, so d(c[k])/d(a[j]) = 2*a[j]
            
            # Build coefficient vector
            coeff = np.zeros(2 * n - 1)
            for offset in range(-j, n - j):
                i = j + offset  # this is the index in the "other" factor
                k = i + j  # this gives k = 2j + offset... hmm this is getting confused
                
            # Let me use a cleaner approach:
            # c[k] = sum_i a[i] * a[k-i]
            # The terms involving a[j]: 
            #   i=j: a[j] * a[k-j] (valid if 0<=k-j<n)
            #   k-i=j, i.e., i=k-j: a[k-j] * a[j] (valid if 0<=k-j<n, same term)
            # So d(c[k])/d(a[j]) = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n (when j != k-j)
            # Wait no. c[k] = sum_i a[i]*a[k-i]. 
            # If i = j: contributes a[j]*a[k-j] (if 0 <= k-j < n)
            # If k-i = j, i.e., i = k-j: contributes a[k-j]*a[j] (if 0 <= k-j < n)
            # These are the same term when i=j, i.e., k-j=j, i.e., k=2j.
            # So: d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n and k=2j
            #                          a[k-j] + a[j-k] if 0<=k-j<n and 0<=j-k<n (k != 2j)
            #                          a[k-j] if only 0<=k-j<n
            #                          a[j-k] if only 0<=j-k<n
            #                          0 otherwise
            
            # Simpler: just compute the gradient of c with respect to a[j]
            # Using the fact that c = a * a, d(c[k])/d(a[j]) = a[k-j] + a[j-k] (with bounds)
            
            # Let's just compute it directly for efficiency
            # The "rest" of c[k] (not involving a[j]):
            # c[k] = a[j]*a[k-j] + a[k-j]*a[j] + (other terms not involving a[j])
            #       = a[j] * (a[k-j] + a[j-k]) + c_rest[k]  (when k-j != j)
            # For k = 2j: c[2j] = a[j]^2 + c_rest[2j]
            
            # For the optimization: we want to minimize max_k c[k] by choosing a[j] >= 0
            # With the constraint sum(a) = 1, changing a[j] means we need to rescale others.
            # This is complex. Let's use a simpler approach:
            # Just try a few values of a[j] and pick the best.
            
            a_old_j = a[j]
            a[j] = 0.0
            c_no_j = _full_convolve(a, a)
            # Now c[k] = c_no_j[k] + a[j] * (a[k-j] if valid) + ... 
            # Actually since we set a[j]=0, c_no_j already has no contribution from a[j].
            # Adding back a[j] = v: c[k] = c_no_j[k] + v * (a[k-j] + a[j-k]) for k != 2j
            # and c[2j] = c_no_j[2j] + v^2
            
            # Hmm, this is getting complex. Let me just use a grid search on a[j].
            # Try values: 0, a_old_j*0.25, a_old_j*0.5, a_old_j*0.75, a_old_j, a_old_j*1.25, a_old_j*1.5
            
            candidates = [0.0, a_old_j * 0.25, a_old_j * 0.5, a_old_j * 0.75,
                         a_old_j, a_old_j * 1.25, a_old_j * 1.5, a_old_j * 2.0]
            
            best_val = a_old_j
            best_local_sc = _score(np.maximum(a, 0))  # current score (a[j]=0 for now)
            # Actually let me just evaluate for each candidate
            a[j] = a_old_j  # restore
            current_full_sc = _score(a)
            
            for v in candidates:
                a[j] = v
                a_temp = a.copy()
                # Don't renormalize (that's a separate step)
                sc_temp = _score(a_temp)
                if sc_temp < best_local_sc:
                    best_local_sc = sc_temp
                    best_val = v
            
            a[j] = best_val
            a = np.maximum(a, 0)
            # Renormalize
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
                break
            
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        # Random but smooth
        n_fourier = min(64, n // 4)
        freqs = np.fft.rfftfreq(n)
        # Low-pass filter
        mask = np.zeros(n // 2 + 1)
        mask[:n_fourier] = 1.0
        mask[n_fourier:n_fourier+8] = np.linspace(1, 0, 8) if n_fourier + 8 <= n//2+1 else 1.0
        coeffs = np.random.randn(n // 2 + 1) * mask
        a = np.fft.irfft(coeffs, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _gs_aggressive(a, deadline, min_time_remaining=5.0, n_phases=5):
    """
    Multi-phase GS with progressively increasing aggressiveness.
    Phase 1: gentle flattening (alpha 0.1-0.3)

    ...
    Phase 5: aggressive (alpha 0.7-0.95)
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for phase in range(n_phases):
        if time.monotonic() + min_time_remaining > deadline:
            break
        alpha_s = 0.1 + 0.15 * phase
        alpha_e = 0.25 + 0.15 * phase

            a = a_phase

            sc_w = _score(a_w)
            if sc_w < best_sc:

                best_a = a_w.copy()
                a = a_w
    
    return best_a

"""
Spectral flattening (Gerchberg-Saxton / ptychography-inspired) combined with
L-BFGS refinement for minimizing peak of linear self-convolution under
non-negativity. Key novelty: alternating projections between spatial domain
(non-negativity + normalization) and frequency domain (magnitude flattening)
to flatten a*a, then gradient-based refinement. Hybrid GS+L-BFGS iterations
provide escape from local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def _gs_flatten(a, n_iter=500, alpha_start=0.1, alpha_end=0.9,
                deadline=None, min_time_remaining=5.0, blend_on_worsen=True):
    """
    Gerchberg-Saxton style spectral flattening.
    Alternates between:
    - Spatial: enforce a >= 0, normalize
    - Frequency: compress |FFT(a)| magnitude to flatten a*a
    Uses progressive alpha increase for gradual tightening.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)
    stagnation = 0
    max_stagnation = 50

    for it in range(n_iter):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break
        if stagnation > max_stagnation:
            break

        t = it / max(1, n_iter - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * t

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        mean_mag = np.mean(mag)
        if mean_mag < 1e-15:
            break

        mag_new = (mag / mean_mag) ** (1.0 - alpha) * mean_mag
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            stagnation = 0
            a = a_new
        else:
            stagnation += 1
            if blend_on_worsen:
                a_new = 0.5 * a + 0.5 * a_new
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    sc_blend = _score(a_new)
                    if sc_blend < best_sc - 1e-14:
                        best_sc = sc_blend
                        best_a = a_new.copy()
                        stagnation = 0
                        a = a_new
                    else:
                        a = a_new

    return best_a


def _gs_weighted(a, n_iter=200, deadline=None, min_time_remaining=5.0):
    """
    Weighted GS: reduce magnitudes more where the convolution is currently peaked.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() + min_time_remaining > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        C = mag ** 2
        C_max = C.max()
        if C_max < 1e-15:
            break

        threshold = C_max * 0.85
        weight = np.ones_like(C)
        mask = C > threshold
        weight[mask] = 1.0 - 0.6 * (C[mask] - threshold) / (C_max - threshold + 1e-15)
        weight = np.maximum(weight, 0.3)

        mag_new = mag * np.sqrt(weight)
        target_energy = np.sum(mag ** 2)
        actual_energy = np.sum(mag_new ** 2)
        if actual_energy > 1e-15:
            mag_new *= np.sqrt(target_energy / actual_energy)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def _gs_multi_gamma(a, deadline, min_time_remaining=5.0):
    """Try different gamma values for magnitude compression."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)

    for gamma in [1.5, 2.0, 3.0, 4.0]:
        if time.monotonic() + min_time_remaining > deadline:

        a_cur = np.maximum(a, 0).astype(np.float64).copy()
        s = a_cur.sum()
        if s < 1e-15:
            continue
        a_cur = a_cur / s

        for it in range(100):
            if time.monotonic() + min_time_remaining > deadline:
                break
            A = np.fft.rfft(a_cur, N)
            mag = np.abs(A)
            phase = np.angle(A)
            mean_mag = np.mean(mag)
            if mean_mag < 1e-15:
                break
            mag_new = (mag / mean_mag) ** (1.0 / gamma) * mean_mag
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)

def _lbfgs_obj(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s_e))


def _lbfgs_grad(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W_val = float(np.dot(w, c))
    return 4.0 * n * a * (h - W_val)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with log-sum-exp smoothing and adaptive beta continuation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    current_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = current_sc

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)]
    base_beta = max(gap * 0.1, max_c * 0.001)

    ratios = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
              0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
              0.000005, 0.000002, 0.000001]

    for i, ratio in enumerate(ratios):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * ratio
        est_time = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 4))))

        try:
            result = minimize(
                _lbfgs_obj, z,
                jac=_lbfgs_grad,
                args=(n, beta),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                c_new = _full_convolve(a_new, a_new)
                max_c_new = float(c_new.max())
                c_sorted_new = np.sort(c_new)[::-1]
                gap_new = c_sorted_new[0] - c_sorted_new[min(10, len(c_sorted_new) - 1)]
                base_beta = max(gap_new * 0.1, max_c_new * 0.001)
        except Exception:
            pass

    return best_a


def _hybrid_gs_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """
    Hybrid: alternate between GS steps (for exploration/escape) and
    L-BFGS steps (for exploitation/fine-tuning).
    This is the key innovation - GS provides a different optimization
    landscape that helps escape L-BFGS local minima.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    N = _next_pow2(2 * n - 1)
    alpha = 0.2

    while time.monotonic() + min_time_remaining < deadline:
        # GS phase: do a few GS steps
        for _ in range(10):
            if time.monotonic() + min_time_remaining > deadline:
                return best_a
            A = np.fft.rfft(a, N)
            mag = np.abs(A)
            phase = np.angle(A)
            mean_mag = np.mean(mag)
            if mean_mag < 1e-15:
                return best_a
            mag_new = (mag / mean_mag) ** (1.0 - alpha) * mean_mag
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                return best_a
            a_new = a_new / s_new
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new

        # L-BFGS phase: do a short L-BFGS run
        z = np.log(np.maximum(a, 1e-30))
        z = z - z.mean()
        c_cur = _full_convolve(a, a)
        max_c = float(c_cur.max())
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)]
        beta = max(gap * 0.05, max_c * 0.0005)
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(200, max(20, int(remaining / 0.5)))

        try:
            result = minimize(
                _lbfgs_obj, z,
                jac=_lbfgs_grad,
                args=(n, beta),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxcor': 30}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            pass

        # Increase alpha gradually
        alpha = min(alpha + 0.03, 0.85)

    a_final = _score(a)
    if a_final < best_sc:
        best_sc = a_final
        best_a = a.copy()
    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * t
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.05
        return a / a.sum()
    elif kind == "bump2":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        n_fourier = min(64, n // 4)
        mask = np.zeros(n // 2 + 1)
        mask[:n_fourier] = 1.0
        if n_fourier + 8 <= n // 2 + 1:
            mask[n_fourier:n_fourier + 8] = np.linspace(1, 0, 8)
        coeffs = np.random.randn(n // 2 + 1) * mask
        a = np.fft.irfft(coeffs, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2) + 0.02
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _primal_dual_refine(a_init, deadline, min_time_remaining=30.0):
    """
    Primal-dual: maintain a "dual" distribution over lags that emphasizes
    the peaks, and use it for gradient computation. This focuses optimization
    on the most constraining lags.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for outer in range(200):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Soft dual: weight lags by exp(c / temp)
        temp = c_max * (0.1 * (1.0 + 0.05 * outer))
        w = np.exp((c - c_max) / temp)
        w = w / w.sum()

        # Gradient of weighted sum
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 2.0 * n * a * (h - W_val)

        # Normalized gradient step with projection
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr = 0.05 / (gnorm + 1e-10)

        # Line search
        for ls_factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_try = _proj_simplex(a - lr * ls_factor * grad)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                break
            elif sc_try < c_max + 1e-10:
                a = a_try
                break

    return best_a

    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                par_sc = _score(parent)
                print(f"Parent: n={n_parent}, score={par_sc:.10f}")
                update_best(parent, par_sc, "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: GS Spectral Flattening (300s) ===
    print(f"\n[Phase 1] GS Spectral Flattening (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = t_start + 300

    # GS from parent with progressive alpha
    a_gs1 = _gs_flatten(parent, n_iter=400, alpha_start=0.1, alpha_end=0.8,
                       deadline=phase1_deadline, min_time_remaining=30)
    update_best(a_gs1, _score(a_gs1), "gs_parent")

    # Weighted GS
    if time.monotonic() < phase1_deadline - 60:
        a_gsw = _gs_weighted(a_gs1, n_iter=300, deadline=phase1_deadline,
                           min_time_remaining=20)
        update_best(a_gsw, _score(a_gsw), "gs_weighted")

    # Multi-gamma GS
    if time.monotonic() < phase1_deadline - 40:
        a_gmg = _gs_multi_gamma(a_gs1, deadline=phase1_deadline, min_time_remaining=20)
        update_best(a_gmg, _score(a_gmg), "gs_multi_gamma")

    # GS from different profiles
    if time.monotonic() < phase1_deadline - 30:
        for prof in ["cos", "gaussian"]:
            if time.monotonic() > phase1_deadline - 15:
                break
            a_p = _make_profile(n_parent, prof)
            a_gp = _gs_flatten(a_p, n_iter=200, alpha_start=0.2, alpha_end=0.7,
                             deadline=phase1_deadline, min_time_remaining=10)
            update_best(a_gp, _score(a_gp), f"gs_{prof}")

    # === Phase 2: Hybrid GS+L-BFGS (400s) ===
    print(f"\n[Phase 2] Hybrid GS+L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(deadline - 500, phase1_deadline + 400)

    a_hyb1 = _hybrid_gs_lbfgs(best_seq, phase2_deadline, min_time_remaining=30)
    update_best(a_hyb1, _score(a_hyb1), "hybrid_1")

    # Second hybrid run from perturbed best
    if time.monotonic() < phase2_deadline - 30:
        a_pert = _perturb(best_seq, 0.02, rng)
        a_hyb2 = _hybrid_gs_lbfgs(a_pert, phase2_deadline, min_time_remaining=20)
        update_best(a_hyb2, _score(a_hyb2), "hybrid_perturb")

    # === Phase 3: L-BFGS refinement (300s) ===
    print(f"\n[Phase 3] L-BFGS Refinement (t={time.monotonic()-t_start:.1f}s)")
    phase3_deadline = min(deadline - 300, phase2_deadline + 300)

    a_lb1 = _run_lbfgs(best_seq, phase3_deadline, min_time_remaining=30)
    update_best(a_lb1, _score(a_lb1), "lbfgs_1")

    if time.monotonic() < phase3_deadline - 30:
        a_lb2 = _run_lbfgs(a_lb1, phase3_deadline, min_time_remaining=20)
        update_best(a_lb2, _score(a_lb2), "lbfgs_2")

    # === Phase 4: Primal-Dual (200s) ===
    print(f"\n[Phase 4] Primal-Dual (t={time.monotonic()-t_start:.1f}s)")
    phase4_deadline = min(deadline - 200, phase3_deadline + 200)

    a_pd = _primal_dual_refine(best_seq, phase4_deadline, min_time_remaining=20)
    update_best(a_pd, _score(a_pd), "primal_dual")

    if time.monotonic() < phase4_deadline - 20:
        a_pd2 = _primal_dual_refine(a_pd, phase4_deadline, min_time_remaining=10)
        update_best(a_pd2, _score(a_pd2), "primal_dual_2")

    # === Phase 5: Multi-n exploration (200s) ===
    print(f"\n[Phase 5] Multi-n Exploration (t={time.monotonic()-t_start:.1f}s)")
    phase5_deadline = min(deadline - 100, phase4_deadline + 200)

    n_tries = [32768, 24576, 49152, 16384, 65536]
    for n_try in n_tries:
        if time.monotonic() > phase5_deadline - 25:
            break
        a_n = _resample(best_seq, n_try)
        a_n_gs = _gs_flatten(a_n, n_iter=150, alpha_start=0.3, alpha_end=0.6,
                           deadline=phase5_deadline, min_time_remaining=10)
        sc_n = _score(a_n_gs)
        update_best(a_n_gs, sc_n, f"n_{n_try}")

        if sc_n < best_score + 0.005 and time.monotonic() < phase5_deadline - 15:
            a_n_hyb = _hybrid_gs_lbfgs(a_n_gs, phase5_deadline, min_time_remaining=10)
            update_best(a_n_hyb, _score(a_n_hyb), f"n_{n_try}_hybrid")

    # === Phase 6: Final polish (60s) ===
    print(f"\n[Phase 6] Final Polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Ultra-tight GS
        a_fin = _gs_flatten(best_seq, n_iter=300, alpha_start=0.5, alpha_end=0.95,
                          deadline=deadline - 15, min_time_remaining=5)
        update_best(a_fin, _score(a_fin), "final_gs")

        # Final hybrid
        if time.monotonic() < deadline - 10:
            a_fin_h = _hybrid_gs_lbfgs(best_seq, deadline - 5, min_time_remaining=3)
            update_best(a_fin_h, _score(a_fin_h), "final_hybrid")

        # Final L-BFGS
        if time.monotonic() < deadline - 5:
            a_fin_l = _run_lbfgs(best_seq, deadline - 3, min_time_remaining=2)
            update_best(a_fin_l, _score(a_fin_l), "final_lbfgs")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(8192, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
