def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoothed_obj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    c_top_shifted = (c_top - c_max) / beta
    c_top_shifted = np.clip(c_top_shifted, -700, 0)

    s = e.sum()

def _peak_equalization(a_init, deadline, min_time_remaining=30.0):
    """New method: Solve linear system to equalize top peaks."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    K = 20  # Number of peaks to equalize
    max_iter = 5000
    stall = 0
    max_stall = 100
    
    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        
        # Select top K peaks
        n_2nm1 = 2 * n - 1
        K_act = min(K, n_2nm1)
        top_idx = np.argpartition(c, -K_act)[-K_act:]
        
        # Target: equalize to mean of top K
        c_active = c[top_idx]
        target = c_active.mean()
        
        # Build system: A[i,j] = 2*a[top_idx[i] - j] (with zero-padding)
        # b[i] = target - c[top_idx[i]]
        # Solve A @ d = b with constraint sum(d) = 0
        
        A = np.zeros((K_act, n))
        for i, k in enumerate(top_idx):
            # A[i,j] = 2*a[k-j] if 0 <= k-j < n
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_end >= j_start:
                A[i, j_start:j_end+1] = 2.0 * a[k - j_start:k - j_end + 1][::-1]
        
        b = target - c_active
        
        # Add constraint sum(d) = 0 via Lagrange: minimize ||Ad - b||^2 + lambda*(sum(d))^2
        # Equivalently, augment system with a row of 1s and target 0
        A_aug = np.vstack([A, np.ones(n) * 1e-3])
        b_aug = np.append(b, 0.0)
        
        # Solve least squares
        try:
            d, residuals, rank, sv = np.linalg.lstsq(A_aug, b_aug, rcond=None)
        except Exception:
            break
        
        # Center d to ensure sum(d) ≈ 0
        d = d - d.mean()
        
        # Line search
        alpha = 0.1
        improved = False
        for ls in range(30):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try smaller K
            K = max(5, K // 2)
            alpha = 0.01
            continue
    
    return best_a, best_sc

def _peak_equalize(a_init, deadline, min_time_remaining=30.0):
    """Equalize top-K peaks by solving a linear system for the direction."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    K = 30
    stall = 0
    max_stall = 150
    lr_scale = 1.0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        
        n_2nm1 = 2 * n - 1
        K_act = min(K, n_2nm1)
        top_idx = np.argpartition(c, -K_act)[-K_act:]
        c_active = c[top_idx]
        target = c_active.mean()
        
        # Build system A @ d = b
        # A[i, j] = 2 * a[k_i - j] if 0 <= k_i - j < n
        A = np.zeros((K_act + 1, n))
        for i in range(K_act):
            k = top_idx[i]
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_end >= j_start:
                A[i, j_start:j_end + 1] = 2.0 * a[k - j_start:k - j_end + 1][::-1]
        b = np.zeros(K_act + 1)
        b[:K_act] = target - c_active
        # Last row: sum(d) = 0 (with small weight to allow infeasibility)
        A[K_act, :] = 1e-4
        b[K_act] = 0.0
        
        try:
            d, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
        except Exception:
            break
        
        d = d - d.mean()
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-20:
            break
        
        d = d / d_norm * lr_scale
        
        # Line search
        alpha = 0.5
        improved = False
        for ls in range(40):
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                lr_scale = min(2.0, lr_scale * 1.1)
                break
            alpha *= 0.5
        
        if not improved:
            lr_scale = max(0.01, lr_scale * 0.5)
            if lr_scale < 0.01:
                K = max(10, K // 2)
                lr_scale = 0.1
            stall += 5
    
    return best_a, best_sc

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Methods: L-BFGS with adaptive top-K smoothing, novel peak-equalization
via linear system solve, projected subgradient with BB steps,
Fourier-mode restarts, and multi-scale exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoothed(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _SmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def fval(self, z):
        return _topk_smoothed(z, self.beta, self.K, self.n)[0]

    def fgrad(self, z):
        return _topk_smoothed(z, self.beta, self.K, self.n)[1]

    max_c = float(c_cur.max())

    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    gap = max(gap, max_c * 0.001)

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, float(a[0]) / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()
