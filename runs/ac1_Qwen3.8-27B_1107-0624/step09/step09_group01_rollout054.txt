"""
Spectral-Newton hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral pre-shaping, soft KKT-Newton with BB step sizes, 
momentum subgradient, coordinate descent polish, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _spectral_flatten(a, n_iters=50, beta=0.7, deadline=None):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a_work = a.astype(np.float64).copy()
    s = a_work.sum()
    if s < 1e-15:
        return a_work
    a_work = a_work / s
    
    best_a = a_work.copy()
    best_sc = _score(a_work)
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compute power spectrum
        N = _next_pow2(2 * n - 1)
        A = np.fft.fft(a_work, N)
        P = np.abs(A) ** 2
        
        # Flatten: raise to power beta < 1
        P_flat = P ** beta
        
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_flat + 1e-30) * np.exp(1j * phase)
        
        # Inverse transform
        a_new = np.real(np.ifft(A_new, N)[:n])
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a_work = a_new
        else:
            # Try different betas
            break
    
    return best_a

def _soft_active_weights(c, M, width=5.0):
    """Compute soft weights for active set based on proximity to max."""
    # Gaussian kernel around the peak
    weights = np.exp(-((c - M) ** 2) / (2 * width * width * max(M, 1e-10)))
    # Also boost the actual peaks
    top_k = max(3, len(c) // 100)
    idx = np.argpartition(c, -top_k)[-top_k:]
    weights[idx] += 1.0
    return weights / weights.sum()

def _compute_conv_grad(a, w):
    """Compute gradient of sum(w_k * c_k) w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    # c_k = sum_i a_i * a_{k-i}
    # d(sum_k w_k c_k) / da_i = 2 * sum_k w_k * a_{k-i}
    # This is 2 * conv(w, a)[n-1 : 2n-1] (the central part)
    wc = _full_convolve(w, a)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

        g = g - np.mean(g)  # remove component in null space of sum constraint
        
        # Hessian-vec product: H = 2 * (conv matrix)^T (diag(w)) (conv matrix)
        # For Newton step, we need to solve H * dx = -g + lambda * ones
        # Use CG for the linear system
        dx = _newton_step(a, w, g, n)

    # This is the cross-correlation of w with reversed v, evaluated at specific lags
    
    # Alternative: use the fact that this is a Toeplitz-like operation
    v_rev = v[::-1]
    # (H*v)[i] = 2 * sum_j w[i+j] * v[j]
    # Let's index: for i=0: 2*sum_j w[j]*v[j]
    # for i=1: 2*sum_j w[1+j]*v[j]
    # ...
    # for i=n-1: 2*sum_j w[n-1+j]*v[j]
    
    # We can compute this as: convolve w with v, but that gives sum_j w[j]*v[i-j]
    # What we want is sum_j w[i+j]*v[j] = sum_j w[m]*v[m-i] where m = i+j
    # = (w cross-correlated with v) at lag i
    # = conv(w, v_rev)[n-1+i] ... let me verify
    
    # conv(w, v_rev)[k] = sum_j w[j] * v_rev[k-j] = sum_j w[j] * v[n-1-(k-j)]
    # For k = n-1+i: = sum_j w[j] * v[n-1-n+1+i-j] = sum_j w[j] * v[i-j]
    # Hmm, that's not quite what we want.
    
    # Let me just do it directly for small n or use a different FFT approach
    # Actually, let's use: sum_j w[i+j] * v[j]
    # = sum_m w[m] * v[m-i] where m = i+j, so j = m-i, and m ranges from i to i+n-1
    # This is the cross-correlation of w and v at lag i (with appropriate bounds)
    
    # Using FFT: cross_corr(w, v)[i] = sum_j w[j+i] * v[j] (for valid range)
    # This equals conv(w, v_flipped)[n-1+i] where v_flipped = v[::-1]
    # conv(w, v_flipped)[k] = sum_j w[j] * v_flipped[k-j] = sum_j w[j] * v[n-1-k+j]
    # For k = n-1+i: = sum_j w[j] * v[n-1-n+1+i+j] = sum_j w[j] * v[i+j]
    # Yes! So (H*v)[i] = 2 * conv(w, v_flipped)[n-1+i]
    
    w_concat = _full_convolve(w, v_rev)
    return 2.0 * w_concat[n-1:2*n-1]

def _newton_step(a, w, g, n):
    """Solve the KKT Newton system."""
    # We need to solve: H * dx + mu * 1 = -g
    # [H  1][dx]   [-g]
    # [1^T 0][mu] = [ 0]
    
    # Use augmented system with CG
    # Solve H * y = -g first (ignoring the constraint), then project
    y = _cg_hankel(w, n, -g)
    
    # Enforce sum constraint: dx = y - mean(y) * 1 = y - (sum(y)/n) * 1
    dx = y - np.mean(y) * np.ones(n)
    
    # Check: if dx is in null space of H (all same), use subgradient
    dx_norm = np.linalg.norm(dx)
    if dx_norm < 1e-14:
        g_t = g - np.mean(g)
        return -g_t / (np.linalg.norm(g_t) + 1e-30)
    
    return dx

def _cg_hankel(w, n, rhs, max_iter=200, tol=1e-12):
    """Conjugate gradient for the Hankel system H*x = rhs."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, w, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp) or pHp <= 0:
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = float(r @ r)
        if not np.isfinite(rs_new) or rs_new > rs * 1.05:
            x = x - alpha * p  # rollback
            break
        if rs_new < tol * tol * rs0:
            break
        p = (rs_new / rs) * p + r
        rs = rs_new
    return x

def _subgrad_bb(a_init, deadline, max_steps=5000):
    """Subgradient method with Barzilai-Borwein step sizes and momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    v = np.zeros(n)
    momentum = 0.7
    bb_type = 0  # alternate between BB1 and BB2
    
    a_prev = None
    g_prev = None
    cur_a = None
    cur_c_max = None
    
    no_improve = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive active set
        for tol_frac in [0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03]:
            active = c >= (c_max - tol_frac * c_max)
            if active.sum() >= 3:
                break
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        g = _compute_conv_grad(a, w)
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # BB step size
        if a_prev is not None and g_prev is not None and i > 0:
            s_vec = a - a_prev
            y_vec = g - g_prev
            sy = float(s_vec @ y_vec)
            yy = float(y_vec @ y_vec)
            ss = float(s_vec @ s_vec)
            if yy > 1e-30 and ss > 1e-30:
                if bb_type == 0:
                    bb = sy / yy if sy > 0 else 0.01 / g_norm
                else:
                    bb = sy / ss if sy > 0 else 0.01 / g_norm
                bb = np.clip(bb, 1e-6, 1.0)
                bb_type = 1 - bb_type
            else:
                bb = 0.01 / g_norm
        else:
            bb = 0.01 / g_norm
        
        v = momentum * v - bb * g
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 20 == 19:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
                momentum = min(momentum + 0.01, 0.9)
            else:
                no_improve += 1
                momentum = max(momentum - 0.02, 0.5)
                if no_improve > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    no_improve = 0
                    momentum = 0.7
        
        a_prev = a.copy()
        g_prev = g.copy()
        cur_a = a
        cur_c_max = c_max
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _coordinate_descent(a_init, deadline, max_sweeps=50):
    """Coordinate descent: optimally update one coordinate at a time."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for i in order:
            # Optimize a[i] with others fixed
            # c_k = sum_j a_j a_{k-j}
            # For fixed a_{-i}, c_k = a_i * (sum_{j≠i} a_{k-j}) + (contribution from j≠i and k-j≠i)
            # Let's compute: c_k = a_i*a_{k-i} + a_{k-i}*a_i + sum_{j≠i, k-j≠i} a_j*a_{k-j}
            # = 2*a_i*a_{k-i} + rest_k (where rest_k excludes terms involving a_i)
            # Score = 2n * max_k c_k / s^2
            # This is complex to do exactly, so let's just try a few values
            
            base = a[i]
            # Try a range of values
            for frac in [0.0, 0.25, 0.5, 0.75, 1.5, 2.0, 3.0]:
                a_try = a.copy()
                a_try[i] = base * frac
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = 2.0 * n * _full_convolve(a_try, a_try).max()  # since sum=1
                if sc_try < best_sc - 1e-14:
                    a = a_try
                    best_sc = sc_try
                    best_a = a.copy()
                    improved = True
                    base = a_try[i]
                    break
        
        if not improved and sweep > 5:
            break
    
    return best_a

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

The quadratic form is Q(a) = Σ_k w_k * (a*a)_k where (a*a)_k = Σ_i a_i * a_{k-i}

Expanding: Q = Σ_k w_k Σ_i a_i a_{k-i} = Σ_{i,j} a_i a_j w_{i+j}... 

Wait, let me be more careful. (a*a)_k = Σ_i a_i a_{k-i} where the sum is over valid i (0 ≤ i ≤ n-1 and 0 ≤ k-i ≤ n-1).

So Q = Σ_k w_k Σ_i a_i a_{k-i}

Let me substitute j = k-i: Q = Σ_k w_k Σ_j a_{k-j} a_j = Σ_{j,k} a_j a_{k-j} w_k

Let i = k-j, so k = i+j: Q = Σ_{i,j} a_i a_j w_{i+j}

The Hessian H has entries H[i,j] = ∂²Q/∂a_i∂a_j = w_{i+j} + w_{i+j} = 2*w_{i+j} (since Q is a quadratic form, the Hessian of a^T M a is M + M^T = 2M when M is symmetric, or M + M^T in general).
