```python
def lp_active_set(a, M, max_iter=100):
    n = len(a)
    c = _full_convolve(a, a)
    
    # Start with top constraints
    active = set(np.argsort(c)[-100].tolist())  # top 100 peaks
    
    for _ in range(max_iter):
        # Build constraint matrix for active constraints only
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        rows = sorted(active)
        A = np.zeros((len(rows), n))
        for i, k in enumerate(rows):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                A[i, j] = a[k - j]
        
        # Solve: max sum(b) s.t. A@b <= M, b >= 0
        from scipy.optimize import linprog
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(rows), M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        b = res.x
        
        # Check all constraints
        ab = _full_convolve(a, b)
        violated = np.where(ab > M * (1 + 1e-10))[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints (up to a limit)
        if len(violated) > 200:
            violated = np.argsort(ab[violated])[-200:] + violated[np.argsort(ab[violated])[-200:]]
            # Actually, let me just take the worst violators
            idx = np.argsort(ab)[::-1][:200]
            violated = idx[idx > M * (1 + 1e-10)]
        
        active = set(active)
        for v in violated:
            active.add(int(v))
        
        # Limit active set size
        if len(active) > 2000:
            # Keep only the most constraining ones
            ab_active = ab[list(active)]
            keep = np.argsort(ab_active)[::-1][:2000]
            active = set([list(active)[i] for i in keep])
    
    return b
```

"""
Optimized step function search using raised-cosine initialization,
LP-based active-set direction finding, and Nesterov-accelerated gradient.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _logsumexp_smoothed(z, beta, n):
    """Full log-sum-exp smoothed objective and gradient in log-domain."""
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

def _make_raised_cosine(n):
    """Raised cosine window, normalized to sum to 1."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    # This gives a = 1 at t=0 and t=1, minimum at t=0.5
    # Actually for our purposes, we want a "bell" shape
    # Let's use: a = 0.5 * (1 + cos(pi * (2*t - 1))) = cos^2(pi*(t-0.5))
    a = np.cos(np.pi * (t - 0.5)) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_tapered(n, p=0.6):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** p
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _run_lbfgs_lse(a_init, deadline, min_time_remaining=30.0, beta_init=None):
    """L-BFGS with full log-sum-exp smoothing and adaptive beta."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    
    if beta_init is None:
        beta_init = max(gap, max_c * 0.01)
    
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Beta annealing schedule
    betas = [beta_init, beta_init*0.5, beta_init*0.25, beta_init*0.1,
             beta_init*0.05, beta_init*0.02, beta_init*0.01, beta_init*0.005,
             beta_init*0.002, beta_init*0.001]
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        def obj(z):
            return _logsumexp_smoothed(z, beta, n)[0]
        
        def grad(z):
            return _logsumexp_smoothed(z, beta, n)[1]
        
        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(30, int(remaining / (est_time * 15))))
        
        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 100, 'maxls': 50}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0):
    """Nesterov accelerated subgradient method with adaptive step size."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    y = a.copy()  # Nesterov lookahead
    velocity = np.zeros(n)
    
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.02
    gamma = 0.9  # momentum (Nesterov uses (t-1)/(t+2), we use fixed for stability)
    
    iteration = 0
    t_start = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        
        # Compute gradient at y (lookahead point)
        c = _full_convolve(y, y)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = y.copy()
        
        # Active set: top peaks
        n_active = max(5, min(2*n-1, int((2*n-1) * 0.005)))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros(2*n-1)
        # Weight by how far above the minimum of the active set
        active_vals = c[top_idx]
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[top_idx] = (active_vals - active_vals.min()) / spread + 0.2
        else:
            w[top_idx] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_maxconv(y, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        t_curr = 1 + iteration
        gamma_nest = (t_curr - 1) / (t_curr + 2)
        
        velocity = gamma_nest * velocity - (lr / (gnorm + 1e-15)) * g_centered
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
            velocity = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        # Update Nesterov lookahead
        y = a_new + gamma_nest * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate
        if current_sc < best_sc + 1e-12:
            lr = min(0.05, lr * 1.01)
        else:
            lr = max(0.001, lr * 0.95)
    
    return best_a, best_sc

def _lp_direction(a, M, max_active=500, max_iter=50):
    """Find LP direction: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Uses constraint generation."""
    n = len(a)
    ab = _full_convolve(a, a)
    
    # Start with top constraints
    n_init = min(100, 2*n-1)
    active = set(np.argsort(ab)[::-1][:n_init].tolist())
    
    for iter_idx in range(max_iter):
        rows = sorted(active)
        if len(rows) == 0:
            break
        
        # Build constraint matrix (sparse would be better, but let's use dense for small active sets)
        m = len(rows)
        A = np.zeros((m, n))
        for i, k in enumerate(rows):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            A[i, j_lo:j_hi+1] = a[k - (j_lo:j_hi+1)]
        
        # Solve LP
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(m, M),
                          bounds=[(0, None)]*n, method='highs',
                          options={'max_iteration': 10000})
            if res.status != 0:
                return None
            b = res.x
        except Exception:
            return None
        
        # Check all constraints
        ab_check = _full_convolve(a, b)
        tol = M * 1e-8 + 1e-15
        violated = np.where(ab_check > M + tol)[0]
        
        if len(violated) == 0:
            return b
        
        # Add most violated constraints
        if len(violated) > max_active - len(active):
            worst = np.argsort(ab_check[violated])[::-1]
            worst = violated[worst[:max(0, max_active - len(active))]]
        else:
            worst = violated
        
        for v in worst:
            active.add(int(v))
        
        if len(active) > max_active:
            # Trim: keep most constraining
            ab_active = ab_check[sorted(active)]
            keep = np.argsort(ab_active)[::-1][:max_active]
            active = set([sorted(active)[i] for i in keep])
    
    return None

def _lp_based_optimization(a_init, deadline, min_time_remaining=30.0):
    """Optimize using LP-based direction finding."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    t_start = time.monotonic()
    no_improve = 0
    max_no_improve = 30
    
    blend = 0.3  # Blending parameter
    
    while time.monotonic() + min_time_remaining < deadline and no_improve < max_no_improve:
        c = _full_convolve(a, a)
        M = float(c.max())
        current_sc = 2.0 * n * M
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find LP direction
        b = _lp_direction(a, M, max_active=300, max_iter=20)
        if b is None:
            break
        
        b_sum = float(b.sum())
        if b_sum < 1e-10:
            break
        
        b = b / b_sum  # Normalize to sum 1
        
        # Blend
        a_try = (1 - blend) * a + blend * b
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        
        sc_try = _score(a_try)
        if sc_try < current_sc - 1e-15:
            a = a_try
            # Reduce blend as we converge
            blend = max(0.05, blend * 0.95)
        else:
            # Try smaller blend
            blend = max(0.01, blend * 0.5)
    
    return best_a, best_sc
