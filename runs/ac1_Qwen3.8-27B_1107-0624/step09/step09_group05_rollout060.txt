def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

    c_max = float(c.max())

    s = e.sum()
    if s < 1e-300:

        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

    if s < 1e-300:

        idx = int(np.argmax(c))
        w[idx] = 1.0

    """Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.

    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

    # So just minimize 2n * weighted_sum(c, top-K)
    w = _top_k_weights(c, K, beta)
    val = float(np.dot(w, c))
    return 2.0 * n * val

        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        a = rng.random(n) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "centered":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2) + 0.001
        return a / a.sum()
    elif kind == "wide_gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2) + 0.001
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, top_k=100, verbose=True, tag=""):
    """L-BFGS with adaptive beta and top-K gradient focus."""
    t_start = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute gap for beta scheduling
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    # Beta schedule: more levels, more aggressive
    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.00000025
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(6000, max(200, int(remaining / (est_iter_time * 15))))

        # For very small beta, cap iterations
        if beta < gap * 0.001:
            max_iter = min(max_iter, 1500)

        def obj(z):
            a_local = _softmax(z)
            c_local = _full_convolve(a_local, a_local)
            # Use top-K focused objective
            w = _top_k_weights(c_local, top_k, beta)
            val = float(np.dot(w, c_local))
            return 2.0 * n * val

        def grad(z):
            a_local = _softmax(z)
            c_local = _full_convolve(a_local, a_local)
            w = _top_k_weights(c_local, top_k, beta)
            grad_a = _grad_structured(a_local, w, n)
            W = float(np.dot(grad_a, a_local))
            return a_local * (grad_a - W)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    {tag} beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 5:
                    print(f"    {tag} beta={beta:.2e}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    {tag} beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _peak_reduction(a_init, deadline, min_time_remaining=30.0, verbose=True, tag=""):
    """Aggressive peak reduction using gradient of top-K peaks.
    
    Directly targets the largest correlation peaks and reduces them
    by adjusting the contributing elements.
    """
    t_start = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Find top K peaks
        K = min(50, len(c))
        sorted_idx = np.argsort(c)[::-1][:K]
        
        # Compute gradient from top-K peaks
        w = np.zeros(len(c))
        w[sorted_idx] = 1.0 / K
        grad = _grad_structured(a, w, n)
        
        # Normalize gradient (remove mean to preserve sum)
        grad_centered = grad - np.mean(grad)
        g_norm = np.sqrt(np.sum(grad_centered ** 2))
        if g_norm < 1e-20:
            break
        
        # Polyak-like step size
        lb = 2.0 * n / (2.0 * n - 1.0)
        gap = current_sc - lb
        if gap < 1e-10:
            break
        
        # Adaptive step
        step = min(0.05, gap / (g_norm ** 2 + 1e-10))
        
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
            if verbose and no_improve % 50 == 0:
                elapsed = time.monotonic() - t_start
                print(f"    {tag} peak_red: sc={best_sc:.10f}, t={elapsed:.0f}s")
        else:
            # Try smaller step
            for trial_step in [step * 0.5, step * 0.25, step * 0.1]:
                a_try = a - trial_step * grad_centered
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    break
            else:
                no_improve += 50  # Penalize harder

    return best_a, best_sc


def _perturb_and_refine(a_best, deadline, min_time_remaining=30.0, n_perturb=5, verbose=True):
    """Add structured perturbations and re-optimize."""
    t_start = time.monotonic()
    n = len(a_best)
    best_a = a_best.copy()
    best_sc = _score(a_best)
    
    rng = np.random.default_rng(12345)
    
    for p in range(n_perturb):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        # Different perturbation strategies
        strategy = p % 3
        a_pert = a_best.copy()
        
        if strategy == 0:
            # Smooth random perturbation
            noise = np.cumsum(rng.normal(0, 0.01, n))
            noise -= noise.mean()
            noise /= (np.std(noise) + 1e-10)
            a_pert = a_best * (1 + 0.1 * noise)
        elif strategy == 1:
            # Localized perturbation
            center = rng.integers(n // 4, 3 * n // 4)
            width = n // 8
            mask = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
            a_pert = a_best * (1 + 0.2 * mask * rng.normal(0, 1, n))
        else:
            # Frequency-domain perturbation
            A = np.fft.rfft(a_best)
            mask = np.ones_like(A)
            n_mod = n // 4
            mask[rng.integers(1, len(A), size=n_mod)] = 1 + 0.1 * rng.normal(0, 1, n_mod)
            A_pert = A * mask
            a_pert = np.fft.irfft(A_pert, n)
            a_pert = np.maximum(a_pert, 0)
        
        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert
        
        # Quick L-BFGS from perturbed point
        p_deadline = min(deadline - min_time_remaining, time.monotonic() + 60)
        a_opt, sc_opt = _run_lbfgs(a_pert, p_deadline, min_time_remaining=5,
                                    top_k=50, verbose=False, tag=f"pert{p}")
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    perturbation {p} (strategy={strategy}): {best_sc:.10f}")

    return best_a, best_sc


def _active_set_kkt(a_init, deadline, min_time_remaining=30.0, verbose=True, tag=""):
    """KKT-inspired active-set refinement with adaptive tolerance."""
    t_start = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 500
    tol_frac = 0.01
    min_tol = 0.00005
    eta = 0.03

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            # Tighten tolerance on improvement
            tol_frac = max(min_tol, tol_frac * 0.95)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                tol_frac = max(min_tol, tol_frac * 0.5)
                eta = max(0.005, eta * 0.7)
                if tol_frac <= min_tol * 1.1 and eta <= 0.005 * 1.1:
                    break
                no_improve = 0

        # Identify active set (peaks near max)
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute average gradient from active set
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # KKT: grad should be constant on support
        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        # Update
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [eta * 0.5, eta * 0.25, eta * 0.1]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                continue

        a = a_new

    return best_a, best_sc

Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Top-K peaked gradient for sharper focus on dominant correlation peaks
- Aggressive perturbation-based restarts with structured noise to escape local minima

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _top_k_weights(c, K, beta):
    """Soft top-K weights using sigmoid thresholding."""
    if K >= len(c):
        return _soft_max_weights(c, beta)
    sorted_c = np.sort(c)[::-1]
    threshold = float(sorted_c[K - 1])
    if beta < 1e-15:
        idx = np.argsort(c)[::-1][:K]
        w = np.zeros_like(c)
        w[idx] = 1.0 / K
        return w
    w = 1.0 / (1.0 + np.exp(-(c - threshold) / beta))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
        return w
    return w / s


def _grad_structured(a, weights, n):
    """Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a[j] = 2*(w * a_rev)[n-1+j]."""
    a_rev = a[::-1]
    wc = _full_convolve(weights, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _make_profile(n, kind="tapered", seed=42):
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered_0.5":

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered_1.0":
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)

    elif kind == "tapered_0.3":

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_taper":
        a = np.sin(np.pi * t) ** 0.7 * (1.0 + 0.3 * (2 * t - 1))

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True, tag=""):
    """L-BFGS with adaptive beta and hybrid top-K/soft-max gradient."""
    t_start = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers

    # Adaptive K: start large (smooth), reduce to focus on peaks
    K_start = min(500, n)
    K_end = min(30, n)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Interpolate K from K_start to K_end
        frac = i / max(1, len(betas) - 1)
        K_cur = int(K_start + (K_end - K_start) * frac)
        K_cur = max(5, K_cur)

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(5000, max(200, int(remaining / (est_iter_time * 12))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 1500)

        def lbf(obj_z):
            a_local = _softmax(obj_z)
            c_local = _full_convolve(a_local, a_local)
            w = _top_k_weights(c_local, K_cur, beta)
            val = float(np.dot(w, c_local))
            return 2.0 * n * val

        def lbg(obj_z):
            a_local = _softmax(obj_z)
            c_local = _full_convolve(a_local, a_local)
            w = _top_k_weights(c_local, K_cur, beta)
            grad_a = _grad_structured(a_local, w, n)
            W = float(np.dot(grad_a, a_local))
            return a_local * (grad_a - W)

        try:
            result = minimize(
                lbf, z, jac=lbg, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"    {tag} beta={beta:.2e} K={K_cur} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
            else:
                if verbose and i < 5:
                    print(f"    {tag} beta={beta:.2e} K={K_cur}: no improvement (best={best_sc:.10f})")
        except Exception as e:
            if verbose:
                print(f"    {tag} beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _peak_reduction_momentum(a_init, deadline, min_time_remaining=30.0, verbose=True, tag=""):
    """Peak reduction with momentum for escaping plateaus."""
    t_start = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    momentum = np.zeros(n)
    beta_mom = 0.9
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Top-K gradient
        K = min(30, len(c))
        sorted_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        w[sorted_idx] = 1.0 / K
        grad = _grad_structured(a, w, n)
        grad_centered = grad - np.mean(grad)
        g_norm = np.sqrt(np.sum(grad_centered ** 2))
        if g_norm < 1e-20:
            break

        # Polyak step with momentum
        lb = 2.0 * n / (2.0 * n - 1.0)
        gap = current_sc - lb
        if gap < 1e-10:
            break

        base_step = min(0.03, gap / (g_norm ** 2 + 1e-10))
        momentum = beta_mom * momentum + (1 - beta_mom) * grad_centered
        update = momentum / (np.sqrt(np.sum(momentum ** 2)) + 1e-20) * base_step

        a_new = a - update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            a = a_new
            if verbose and no_improve == 0 and (no_improve % 100 == 0 or no_improve < 5):
                elapsed = time.monotonic() - t_start
                print(f"    {tag} peak_mom: sc={best_sc:.10f}, t={elapsed:.0f}s")
        else:
            # Try base gradient step without momentum
            a_try = a - base_step * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-14:
                    a = a_try
                    momentum = grad_centered
                else:
                    for shrink in [0.5, 0.25, 0.1]:
                        a_sh = a - base_step * shrink * grad_centered
                        a_sh = np.maximum(a_sh, 0.0)
                        s_sh = a_sh.sum()
                        if s_sh < 1e-10:
                            continue
                        a_sh = a_sh / s_sh
                        sc_sh = _score(a_sh)
                        if sc_sh < current_sc - 1e-14:
                            a = a_sh
                            break

    return best_a, best_sc


def _active_set_kkt(a_init, deadline, min_time_remaining=30.0, verbose=True, tag=""):

    t_start = time.monotonic()

        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            tol_frac = max(min_tol, tol_frac * 0.96)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                tol_frac = max(min_tol, tol_frac * 0.5)
                eta = max(0.003, eta * 0.6)
                if tol_frac <= min_tol * 1.05 and eta <= 0.003 * 1.05:
                    break
                no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [eta * 0.5, eta * 0.25, eta * 0.1]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)

                if s_try < 1e-10:

                a_try = a_try / s_try

                    sc_new = sc_try
                    found = True
                    break

                continue

        a = a_new

    return best_a, best_sc


def _perturb_and_refine(a_best, deadline, min_time_remaining=30.0, n_perturb=6, verbose=True):
    """Structured perturbation restarts to escape local minima."""
    t_start = time.monotonic()

            break

        strategy = p % 4
        a_pert = a_best.copy()

        if strategy == 0:
            # Smooth low-frequency perturbation
            noise = np.fft.irfft(np.fft.rfft(np.ones(n) / np.sqrt(n)) *
                                  rng.normal(0, 0.05, n // 2 + 1)[1:] *
                                  (1 + np.arange(n // 2 + 1)[1:]), n)

            noise /= (np.std(noise) + 1e-10)

            sign = 1.0 if rng.random() > 0.5 else -1.0

            delta = 0.05 * (m1 - m2)

        if s_pert < 1e-10:

        a_pert = a_pert / s_pert

        # Quick L-BFGS from perturbed point

            best_a = a_opt.copy()
            if verbose:
                print(f"    pert {p} (strat={strategy}): {best_sc:.10f}")

    return best_a, best_sc

    print(f"\n[P4] KKT active-set (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 50 < deadline:
        p4_end = min(deadline - 300, time.monotonic() + 180)
        a4, sc4 = _active_set_kkt(best_seq, p4_end, min_time_remaining=30, tag="kkt")
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: L-BFGS polish ===
    print(f"\n[P5] L-BFGS polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p5_end = min(deadline - 200, time.monotonic() + 150)
        a5, sc5 = _run_lbfgs(best_seq, p5_end, min_time_remaining=20, verbose=True, tag="polish")
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Perturbation restarts ===
    print(f"\n[P6] Perturbation restarts (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p6_end = min(deadline - 100, time.monotonic() + 250)
        a6, sc6 = _perturb_and_refine(best_seq, p6_end, min_time_remaining=50,
                                       n_perturb=6, verbose=True)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.10f}")

    # === Phase 7: Length variation ===
    print(f"\n[P7] Length variation (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [n_main - 1024, n_main + 1024]
    n_options = [x for x in n_options if x > 2048]
    for n_try in n_options:
        if time.monotonic() + 50 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try} raw: {sc_r:.10f}")
        if time.monotonic() + 35 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p7_end = min(deadline - 40, time.monotonic() + 60)
        a7, sc7 = _run_lbfgs(a_r, p7_end, min_time_remaining=10, verbose=False, tag=f"n{n_try}")
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try} opt: {best_score:.10f}")

    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 15 < deadline:
        p8_end = min(deadline - 3, time.monotonic() + 60)
        a8, sc8 = _run_lbfgs(best_seq, p8_end, min_time_remaining=3, verbose=False, tag="final")
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

        if time.monotonic() + 8 < deadline:
            p8b_end = min(deadline - 2, time.monotonic() + 20)
            a8b, sc8b = _active_set_kkt(best_seq, p8b_end, min_time_remaining=2, tag="final_kkt")
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

    # Final validation
    print(f"\n{'=' * 60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
