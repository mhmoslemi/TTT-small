        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normed(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _variance_value(a):
    c = _full_convolve(a, a)
    return float(np.sum(c * c))


def _variance_grad(a):
    n = len(a)
    c = _full_convolve(a, a)
    a_flip = a[::-1]
    cc = _full_convolve(c, a_flip)
    return 2.0 * cc[n - 1:2 * n - 1]


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smooth_obj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _variance_obj_grad_z(z, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    val = _variance_value(a)
    ga = _variance_grad(a)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return val, grad_z


class _VarianceProb:
    def __init__(self, n):
        self.n = n

    def __call__(self, z):
        return _variance_obj_grad_z(z, self.n)[0]

    def grad(self, z):
        return _variance_obj_grad_z(z, self.n)[1]


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs_var(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    prob = _VarianceProb(n)
    est_time = 0.003 * max(1.0, n / 4096.0)
    remaining = deadline - min_time_remaining - time.monotonic()
    if remaining < 10:
        return current_a
    max_iter = min(10000, max(100, int(remaining / (est_time * 15))))

    try:
        result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxcor': 100, 'maxls': 50})
        a_new = np.exp(result.x - result.x.max())
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < current_sc:
            return a_new
    except Exception:
        pass
    return current_a


def _run_lbfgs_topk(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()

    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.01)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    schedule = [
        (1.0, 0.15), (0.7, 0.12), (0.5, 0.08), (0.3, 0.06),
        (0.2, 0.04), (0.1, 0.03), (0.05, 0.02), (0.03, 0.015),
        (0.01, 0.01), (0.005, 0.006), (0.002, 0.004), (0.001, 0.003),
        (0.0005, 0.002),
    ]

    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        beta = base_beta * beta_mult
        K = max(10, int((2 * n - 1) * K_frac))

        est_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 20))))

        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 100, 'maxls': 50})
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgradient_adaptive(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    n_2nm1 = 2 * n - 1
    active_frac = 0.02
    momentum = 0.75
    lr = 0.03
    lr_min = 1e-5
    lr_max = 0.08
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)

    prev_a = None
    prev_g = None
    prev_a2 = None
    prev_g2 = None

    iteration = 0
    last_lbfgs_time = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_active = max(5, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]

        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break

        # Barzilai-Borwein
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)

        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(30):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7

        if improved:
            prev_a2 = prev_a
            prev_g2 = prev_g
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.003, active_frac * 0.985)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.1, active_frac * 1.4)

        if stall > max_stall:
            active_frac = min(0.1, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.92, momentum + 0.03)
            stall = 0

        if iteration % 800 == 0 and time.monotonic() - last_lbfgs_time > 90:
            if time.monotonic() + min_time_remaining + 60 < deadline:
                lbfgs_dl = time.monotonic() + 50
                a_tmp = _run_lbfgs_topk(a, lbfgs_dl, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                last_lbfgs_time = time.monotonic()

    return best_a, best_sc


def _peak_reduction(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    patience = 0
    max_patience = 500

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Find top peaks
        n_peaks = max(3, int(n * 0.003))
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        # For each peak, compute which a[i] contribute most
        # and try to redistribute
        delta = np.zeros(n)
        for k_peak in peak_indices:
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

            # Reduce top contributors
            n_red = max(2, int(n * 0.002))
            top_idx = np.argsort(contrib)[::-1][:n_red]
            reduction = (c[k_peak] - c_max * 0.999) * 0.01
            if reduction > 0:
                for idx in top_idx:
                    if contrib[idx] > 1e-15 and a[idx] > 1e-15:
                        delta[idx] -= reduction / (2.0 * contrib[idx] + 1e-30) * 0.5

        delta -= delta.mean()

        # Line search
        alpha = 0.5
        improved = False
        for ls in range(25):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            patience += 5

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_off":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.3) ** 2) + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.35) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "smooth_noise":
        rng = np.random.default_rng(12345)
        a = 0.5 + 0.5 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_taper":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.3 + 0.05
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.008
    tol = 0.003
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active

        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break

        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(40):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.00005, tol * 0.7)
                lr *= 0.7
            if no_improve > 500:
                a = best_a.copy()
                tol = max(0.00005, tol * 0.5)
                lr = 0.004
                no_improve = 0
        else:
            if tol > 0.00005:
                tol = max(0.00005, tol * 0.97)
            lr = min(0.01, lr * 1.005)

    return best_a, best_sc


def _two_phase_optimize(a_init, n, deadline, min_time_remaining=120.0):
    """Run variance minimization then top-K L-BFGS."""
    t_remaining = deadline - min_time_remaining - time.monotonic()
    if t_remaining < 20:
        return a_init

    # Phase A: Variance minimization
    var_budget = min(t_remaining * 0.3, 200.0)
    var_deadline = time.monotonic() + var_budget
    a_var = _run_lbfgs_var(a_init, var_deadline, min_time_remaining=10.0)
    sc_var = _score(a_var)
    print(f"    Variance phase: score={sc_var:.10f}, t={time.monotonic()-deadline+deadline-time.monotonic():.1f}s")

    best = a_var if sc_var < _score(a_init) else a_init.copy()

    # Phase B: Top-K L-BFGS
    topk_deadline = time.monotonic() + min(t_remaining - var_budget, deadline - min_time_remaining - time.monotonic())
    a_topk = _run_lbfgs_topk(best, topk_deadline, min_time_remaining=10.0)
    sc_topk = _score(a_topk)
    if sc_topk < sc_var:
        best = a_topk
        print(f"    TopK phase: score={sc_topk:.10f}")
    else:
        print(f"    TopK phase: no improvement (best={sc_var:.10f})")

    return best


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "tapered")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, tapered n={n_parent}, score={best_score:.10f}")

    n_target = len(best_seq)

    # === Phase 1: Multi-profile coarse optimization at smaller n ===
    print(f"\n[P1] Coarse multi-profile (t={time.monotonic()-t_start:.1f}s)")
    n_coarse = min(4096, n_target)
    coarse_deadline = min(deadline - 600, time.monotonic() + 200)

    profiles = ["tapered", "cos_taper", "gauss_center", "gauss_off",
                "bimodal", "smooth_noise", "power_taper"]

    best_coarse = None
    best_coarse_sc = np.inf

    for prof in profiles:
        if time.monotonic() > coarse_deadline:
            break
        a_prof = _make_profile(n_coarse, prof)

        # Quick L-BFGS on variance
        var_dl = min(coarse_deadline, time.monotonic() + 40)
        a_opt = _run_lbfgs_var(a_prof, var_dl, min_time_remaining=2)

        # Then top-K
        topk_dl = min(coarse_deadline, time.monotonic() + 40)
        a_opt = _run_lbfgs_topk(a_opt, topk_dl, min_time_remaining=2)

        sc = _score(a_opt)
        if sc < best_coarse_sc:
            best_coarse_sc = sc
            best_coarse = a_opt.copy()
            print(f"  Profile '{prof}': score={sc:.10f}")

    if best_coarse is not None and best_coarse_sc < best_score:
        # Resample to target n
        a_resampled = _resample(best_coarse, n_target)
        # Quick refine at target n
        refine_dl = min(deadline - 500, time.monotonic() + 80)
        a_refined = _run_lbfgs_topk(a_resampled, refine_dl, min_time_remaining=30)
        sc = _score(a_refined)
        if sc < best_score:
            best_seq = a_refined
            best_score = sc
            print(f"  NEW BEST from coarse: {best_score:.10f}")

    # === Phase 2: Two-phase optimization (variance + topK) on best ===
    print(f"\n[P2] Two-phase on best (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(deadline - 500, time.monotonic() + 300)
    a_p2 = _two_phase_optimize(best_seq, n_target, phase2_deadline + time.monotonic() * 0 + deadline, min_time_remaining=300)
    sc_p2 = _score(a_p2)
    if sc_p2 < best_score:
        best_seq = a_p2
        best_score = sc_p2
        print(f"  NEW BEST: {best_score:.10f} [two-phase]")
    else:
        print(f"  No improvement (best={best_score:.10f})")

    # === Phase 3: Subgradient adaptive ===
    print(f"\n[P3] Adaptive subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        sub_deadline = min(deadline - 300, time.monotonic() + 350)
        a3, sc3 = _subgradient_adaptive(best_seq, sub_deadline, min_time_remaining=60)
        if sc3 < best_score:
            best_seq = a3
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [subgrad]")
        else:
            print(f"  No improvement (best={best_score:.10f})")

    # === Phase 4: Peak reduction ===
    print(f"\n[P4] Peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        pr_deadline = min(deadline - 200, time.monotonic() + 200)
        a4, sc4 = _peak_reduction(best_seq, pr_deadline, min_time_remaining=30)
        if sc4 < best_score:
            best_seq = a4
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [peakred]")

    # === Phase 5: Second top-K L-BFGS from improved point ===
    print(f"\n[P5] TopK L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a5 = _run_lbfgs_topk(best_seq, min(deadline - 150, time.monotonic() + 200),
                             min_time_remaining=40)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_seq = a5
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [topk2]")

    # === Phase 6: Fine polish ===
    print(f"\n[P6] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a6, sc6 = _fine_polish(best_seq, min(deadline - 80, time.monotonic() + 120),
                               min_time_remaining=15)
        if sc6 < best_score:
            best_seq = a6
            best_score = sc6
            print(f"  NEW BEST: {best_score:.10f} [polish]")

    # === Phase 7: Multi-restart with perturbations ===
    print(f"\n[P7] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        sigmas = [0.02, 0.04, 0.06, 0.08, 0.03, 0.05, 0.1]
        for i, sigma in enumerate(sigmas):
            if time.monotonic() > deadline - 60:
                break
            a_pert = _perturb(best_seq, sigma, rng)
            # Quick two-phase on perturbed
            restart_dl = min(deadline - 50, time.monotonic() + 40)
            a_r = _run_lbfgs_topk(a_pert, restart_dl, min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i}, sigma={sigma}]")

    # === Phase 8: Multi-n exploration ===
    print(f"\n[P8] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = [int(n_main * 0.75), int(n_main * 1.25), n_main]
        n_options = [x for x in n_options if x != n_main and x >= 1024]

        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs_topk(a_res, min(deadline - 30, time.monotonic() + 40),
                                    min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}]")

    # === Phase 9: Final polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish]")

        # Final tiny L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-10)
            K_fin = max(10, int((2 * n - 1) * 0.003))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _TopkSmoProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 80})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")
            except Exception:
                pass

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
