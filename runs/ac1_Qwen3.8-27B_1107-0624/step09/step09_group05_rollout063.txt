Let me also consider: what if I use a **power iteration** approach in the frequency domain?

The objective is to minimize max(c) where c = IFFT(|A|^2). The gradient of max(c) w.r.t. |A|^2 is the indicator of the max position (in time domain), which in frequency domain is the DFT of that indicator. So:

1. Compute c = a*a.
2. Find the "hot" positions: h = soft_max_indicator(c).
3. Compute H = FFT(h, N).
4. Update: P = |A|^2. P_new = P - eta * H (project to P >= 0).
5. A_new = sqrt(P_new) * exp(i*angle(A)).
6. a_new = IFFT(A_new)[:n]. Project to a >= 0.

This is a form of gradient descent in the frequency domain. The step size eta needs to be chosen carefully.

Actually, this is similar to the "error reduction" step in Gerchberg-Saxton. Let me combine both:

1. Error reduction in time domain: reduce the peaks of c (by reducing a at positions that contribute to the peaks).
2. Error reduction in frequency domain: flatten |A|^2 (by reducing the spectral peaks).

This is the classic alternating projection approach.

Let me code this up properly:

```python
def _alternating_projection(a, n_iters=1000, t_time=0.5, t_freq=0.3):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    
    # Normalize
    S = a.sum()
    a = a / S
    
    for i in range(n_iters):
        # Time domain: reduce peaks of c
        c = _full_convolve(a, a)
        c_max = c.max()
        c_mean = c.sum() / len(c)  # = 1/(2n-1) since sum(c) = 1
        # Reduce a where it contributes to peaks
        # Gradient of c[k] w.r.t. a[j] = a[k-j] (for valid k-j)
        # So to reduce c at position k, reduce a[j] for j where a[k-j] is large
        # Simple: reduce a proportionally to its contribution to the max
        # contrib[j] = sum over k where c[k] > c_mean of a[k-j] * (c[k] - c_mean)
        # This is O(n^2) which is too slow for large n
        # Instead, use the soft-max gradient
        ...
        
        # Frequency domain: flatten |A|^2
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        P_mean = P.mean()
        P_new = (1-t_freq) * P + t_freq * P_mean
        P_new = np.maximum(P_new, 0)
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new * S  # preserve sum
    
    return a
```

The frequency-domain step is O(N log N) which is very fast. The time-domain step is the expensive part.

Actually, let me think about this more carefully. The frequency-domain step alone (without the time-domain step) might be sufficient if we do enough iterations. Each iteration:

So each iteration is O(N log N) ≈ O(n log n). For n = 16384 and N = 32768, this is very fast (maybe 1ms per iteration). We can do 100,000+ iterations in 1800 seconds.

```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)

def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)

def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)

class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)
```

```python
def _spectral_flatten(a_init, n_iters=5000, start_power=0.3, end_power=0.95, start_blend=0.8, end_blend=0.1, deadline=None, verbose=True):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)
    
    # Precompute
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Interpolate parameters
        frac = i / max(1, n_iters - 1)
        power = start_power + (end_power - start_power) * frac
        blend = start_blend + (end_blend - start_blend) * frac
        
        # Compute spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        
        # Fix DC component
        P_dc = P[0]
        
        # Flatten: P_new = P^power * (P_mean / (P^power).mean())
        P_pos = np.maximum(P, 1e-30)
        P_alpha = P_pos ** power
        P_target = P_alpha * (P.mean() / P_alpha.mean())
        P_target[0] = P_dc  # preserve DC
        
        # Blend
        P_new = blend * P_target + (1.0 - blend) * P
        P_new = np.maximum(P_new, 0)
        
        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new  # normalize to sum 1
        
        # Check score periodically
        if i % 500 == 0 or i == n_iters - 1:
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            if verbose and i % 2000 == 0:
                print(f"    spectral iter {i}: score={sc:.10f}, power={power:.3f}, blend={blend:.3f}")
        
        a = a_new
    
    return best_a, best_sc
```

```python
"""
Multi-phase optimizer combining spectral flattening (frequency-domain) with
time-domain L-BFGS and active-set KKT refinement.
Key innovation: Gerchberg-Saxton-style spectral shaping to escape local minima
of the time-domain optimizer, followed by precise L-BFGS polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb,
                     N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _spectral_flatten(a_init, deadline, min_time=2.0,
                      start_power=0.2, end_power=0.99,
                      start_blend=0.9, end_blend=0.05,
                      n_max_iters=200000, verbose=True):
    """Gerchberg-Saxton style spectral flattening.
    Iteratively flattens |FFT(a)|^2 while preserving non-negativity of a.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy().astype(np.float64)
    S = float(a.sum())
    if S < 1e-15:
        return a_init, np.inf
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)

    # Track for early stopping
    patience = 0
    max_patience = 5000

    for i in range(n_max_iters):
        if time.monotonic() + min_time > deadline:
            break

        frac = min(1.0, i / max(1, n_max_iters - 1))
        power = start_power + (end_power - start_power) * frac
        blend = start_blend + (end_blend - start_blend) * frac

        # Compute spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_dc = P[0]

        # Power compression to flatten
        P_pos = np.maximum(P, 1e-30)
        P_alpha = P_pos ** power
        alpha_mean = P_alpha.mean()
        if alpha_mean < 1e-30:
            break
        P_target = P_alpha * (P.mean() / alpha_mean)
        P_target[0] = P_dc  # preserve DC

        # Blend
        P_new = blend * P_target + (1.0 - blend) * P
        P_new = np.maximum(P_new, 0.0)

        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project to non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        # Check score periodically
        if i % 1000 == 0 or i == n_max_iters - 1:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1000
            if verbose and i % 10000 == 0:
                print(f"    spectral iter {i}: score={sc:.10f}, "
                      f"power={power:.3f}, blend={blend:.3f}, patience={patience}")

        # Early stop if no improvement for a long time
        if patience >= max_patience:
            break

    return best_a, best_sc


def _projected_gradient_c2(a_init, deadline, min_time=2.0,
                           eta_init=0.05, verbose=True):
    """Projected gradient descent on ||a*a||^2 (smooth proxy for max).
    Faster convergence near the optimum since the objective is smooth.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy().astype(np.float64)
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)

    eta = eta_init
    no_improve = 0
    max_no_improve = 20000

    for i in range(5000000):
        if time.monotonic() + min_time > deadline:
            break

        # Compute gradient of ||a*a||^2 w.r.t. a
        c = _full_convolve(a, a)
        a_rev = a[::-1]
        cc = _full_convolve(c, a_rev)
        grad = 4.0 * cc[n - 1:2 * n - 1]

        # Polyak step size
        c2_val = float(np.dot(c, c))
        lb_c2 = 1.0 / (2 * n - 1)
        gap = c2_val - lb_c2
        if gap < 1e-20:
            break
        g_centered = grad - grad.mean()
        g_norm_sq = float(np.dot(g_centered, g_centered))
        if g_norm_sq < 1e-30:
            break
        eta_step = gap / g_norm_sq
        eta_use = min(eta, eta_step * 0.5)
        eta_use = max(eta_use, 1e-8)

        # Gradient step
        a_new = a - eta_use * g_centered
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        # Check improvement
        if i % 500 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 500
            if verbose and i % 50000 == 0:
                print(f"    proj-grad iter {i}: score={sc:.10f}, eta={eta_use:.2e}")

        # Decay
        if i % 10000 == 0:
            eta *= 0.9

        if no_improve >= max_no_improve:
            break

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    if s < 1e-15:
        return a_init, np.inf
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001
    ])
    betas = gap * multipliers

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 20))))
        if beta < gap * 0.001:
            max_iter = min(max_iter, 2000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}")
        except Exception:
            pass

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 500
    tol_frac = 0.005
    min_tol = 0.00005

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-8 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        eta = 0.05
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [0.02, 0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-12:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.8)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _combined_spectral_time(a_init, deadline, min_time=5.0, verbose=True):
    """Alternating spectral flattening and time-domain gradient steps."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy().astype(np.float64)
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)

    # Spectral parameters
    power = 0.4
    blend = 0.5
    # Time-domain step
    eta_td = 0.02

    no_improve = 0
    max_no_improve = 20000

    for i in range(1000000):
        if time.monotonic() + min_time > deadline:
            break

        # Spectral step
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_dc = P[0]
        P_pos = np.maximum(P, 1e-30)
        P_alpha = P_pos ** power
        alpha_mean = P_alpha.mean()
        if alpha_mean < 1e-30:
            break
        P_target = P_alpha * (P.mean() / alpha_mean)
        P_target[0] = P_dc
        P_new = blend * P_target + (1.0 - blend) * P
        P_new = np.maximum(P_new, 0.0)
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_spec = np.fft.irfft(A_new, N)[:n]

        # Time-domain step: reduce peaks of c
        c = _full_convolve(a_spec, a_spec)
        c_max = float(c.max())
        a_rev = a_spec[::-1]
        cc = _full_convolve(c, a_rev)
        grad = 4.0 * cc[n - 1:2 * n - 1]
        g_centered = grad - grad.mean()
        a_td = a_spec - eta_td * g_centered

        # Combine spectral and time-domain results
        a_new = 0.5 * a_spec + 0.5 * a_td
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        # Anneal parameters
        if i % 5000 == 0:
            power = min(0.99, power * 1.02)
            blend = max(0.05, blend * 0.99)
            eta_td *= 0.98

        # Check score
        if i % 2000 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 2000
                # Perturb to escape local min
                if no_improve > 10000:
                    rng = np.random.default_rng(i)
                    a = a + rng.normal(0, 0.001, n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    no_improve = 0
            if verbose and i % 50000 == 0:
                print(f"    combined iter {i}: score={sc:.10f}, "
                      f"power={power:.3f}, blend={blend:.3f}")

        if no_improve >= max_no_improve:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Spectral flattening (fast, frequency-domain)
    P2: Combined spectral + time-domain optimization
    P3: Projected gradient on ||c||^2
    P4: L-BFGS refinement
    P5: Active-set KKT
    P6: L-BFGS round 2
    P7: Multi-n exploration
    P8: Diverse profile search
    P9: Final polish
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Spectral flattening ===
    print(f"\n[P1] Spectral flattening (n={n_main}, t={time.monotonic()-t0:.0f}s)")
    p1_end = t0 + min(400, (budget_s - 10) * 0.22)
    a1, sc1 = _spectral_flatten(best_seq, p1_end, min_time=200,
                                start_power=0.15, end_power=0.99,
                                start_blend=0.95, end_blend=0.03,
                                n_max_iters=500000, verbose=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")
    else:
        print(f"  P1 no improvement (best={best_score:.10f})")

    # === Phase 2: Combined spectral + time-domain ===
    print(f"\n[P2] Combined spectral+time (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 100 < deadline:
        p2_end = min(deadline - 600, time.monotonic() + 300)
        a2, sc2 = _combined_spectral_time(best_seq, p2_end, min_time=100, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")
        else:
            print(f"  P2 no improvement (best={best_score:.10f})")

    # === Phase 3: Projected gradient on ||c||^2 ===
    print(f"\n[P3] Projected gradient (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 400, time.monotonic() + 200)
        a3, sc3 = _projected_gradient_c2(best_seq, p3_end, min_time=60, verbose=True)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: L-BFGS ===
    print(f"\n[P4] L-BFGS (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p4_end = min(deadline - 300, time.monotonic() + 350)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=200, verbose=True)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Active-set KKT ===
    print(f"\n[P5] Active-set KKT (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p5_end = min(deadline - 200, time.monotonic() + 200)
        a5, sc5 = _active_set_refine(best_seq, p5_end, min_time_remaining=60)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: L-BFGS round 2 (after KKT) ===
    print(f"\n[P6] L-BFGS round 2 (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p6_end = min(deadline - 150, time.monotonic() + 250)
        a6, sc6 = _run_lbfgs(best_seq, p6_end, min_time_remaining=80, verbose=True)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n exploration (t={time.monotonic()-t0:.0f}s)")
    n_options = [8192, 12288, 20480, 24576, 32768]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try}: resampled score={sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p7_end = min(deadline - 80, time.monotonic() + 100)
        # Quick spectral + L-BFGS
        a7s, sc7s = _spectral_flatten(a_r, p7_end, min_time=30,
                                       start_power=0.3, end_power=0.99,
                                       start_blend=0.8, end_blend=0.05,
                                       n_max_iters=100000, verbose=False)
        if sc7s < best_score:
            best_score = sc7s
            best_seq = a7s.copy()
            print(f"  n={n_try} spectral: {best_score:.10f}")
        if time.monotonic() + 30 < p7_end:
            a7f, sc7f = _run_lbfgs(a7s, min(p7_end, time.monotonic() + 50),
                                    min_time_remaining=10, verbose=False)
            if sc7f < best_score:
                best_score = sc7f
                best_seq = a7f.copy()
                print(f"  n={n_try} lbfgs: {best_score:.10f}")

    # === Phase 8: Diverse profiles with spectral + L-BFGS ===
    print(f"\n[P8] Diverse profiles (t={time.monotonic()-t0:.0f}s)")
    profiles = ["periodic", "bimodal", "cubic_taper", "gaussian", "quadratic", "raised_cosine"]
    n_for_profiles = len(best_seq)
    for prof in profiles:
        if time.monotonic() + 50 > deadline:
            break
        a_p = _make_profile(n_for_profiles, prof)
        sc_p = _score(a_p)
        if sc_p < best_score + 0.3:
            p8_end = min(deadline - 30, time.monotonic() + 80)
            # Quick spectral flatten
            a_ps, sc_ps = _spectral_flatten(a_p, p8_end, min_time=20,
                                            start_power=0.3, end_power=0.99,
                                            start_blend=0.8, end_blend=0.05,
                                            n_max_iters=50000, verbose=False)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
                print(f"  {prof} spectral: {best_score:.10f}")
            # L-BFGS polish
            if time.monotonic() + 20 < p8_end:
                a_pf, sc_pf = _run_lbfgs(a_ps, min(p8_end, time.monotonic() + 40),
                                         min_time_remaining=10, verbose=False)
                if sc_pf < best_score:
                    best_score = sc_pf
                    best_seq = a_pf.copy()
                    print(f"  {prof} lbfgs: {best_score:.10f}")

    # === Phase 9: Final spectral + L-BFGS polish ===
    print(f"\n[P9] Final polish (t={time.monotonic()-t0:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p9_end = min(deadline - 5, time.monotonic() + 80)
        a9s, sc9s = _spectral_flatten(best_seq, p9_end, min_time=20,
                                       start_power=0.5, end_power=0.999,
                                       start_blend=0.5, end_blend=0.02,
                                       n_max_iters=100000, verbose=False)
        if sc9s < best_score:
            best_score = sc9s
            best_seq = a9s.copy()

        if time.monotonic() + 10 < deadline:
            p9f_end = min(deadline - 3, time.monotonic() + 60)
            a9f, sc9f = _run_lbfgs(best_seq, p9f_end, min_time_remaining=5, verbose=False)
            if sc9f < best_score:
                best_score = sc9f
                best_seq = a9f.copy()

        if time.monotonic() + 10 < deadline:
            p9k_end = min(deadline - 2, time.monotonic() + 30)
            a9k, sc9k = _active_set_refine(best_seq, p9k_end, min_time_remaining=2)
            if sc9k < best_score:
                best_score = sc9k
                best_seq = a9k.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
```
