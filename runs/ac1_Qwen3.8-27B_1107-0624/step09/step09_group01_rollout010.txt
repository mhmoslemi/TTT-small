"""
Multi-scale log-sum-exp optimization with L-BFGS-B, structured restarts,
and adaptive active sets for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses temperature annealing from coarse to fine, multiple n exploration,
and low-frequency perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _lse_value(a, beta, n):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(a, beta, n):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * n * h


def _normalize(a):
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_profile(n, kind="tapered"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "cos":
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "linear":
        a = t + 0.1
    elif kind == "ramp":
        a = np.linspace(0.1, 1.0, n)
    elif kind == "bell":
        a = np.exp(-3.0 * (t - 0.5) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _perturb_freq(a, rng, num_modes=5):
    """Apply low-frequency perturbations in the Fourier domain."""
    n = len(a)
    A = np.fft.rfft(a)
    for _ in range(num_modes):
        k = rng.integers(1, min(n // 2, 50))
        A[k] *= (1.0 + 0.1 * rng.standard_normal())
        if 2 * k - 1 < len(A):
            A[2 * k - 1] *= (1.0 + 0.1 * rng.standard_normal())
    a_new = np.fft.irfft(A, n)
    return np.maximum(a_new, 0)


def _optimize_lse(a0, deadline, betas, max_iter_per_beta=2000, label=""):
    """Optimize using L-BFGS-B with log-sum-exp smoothing at multiple temperatures."""
    n = len(a0)
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def obj(z):
            a_local = _normalize(np.exp(z - z.max()))
            return _lse_value(a_local, beta, n)
        
        def jac(z):
            a_local = _normalize(np.exp(z - z.max()))
            g = _lse_grad(a_local, beta, n)
            # Transform gradient for the log parameterization
            # a = exp(z - max(z)) / sum(exp(z - max(z)))
            # da_i/dz_j = a_i * (delta_ij - a_j)
            # grad_z_j = sum_i grad_a_i * da_i/dz_j = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
            return g * a_local - a_local * float(np.dot(g, a_local))
        
        z0 = np.log(np.maximum(a, 1e-30))
        z0 = z0 - z0.mean()
        
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        try:
            result = minimize(
                obj, z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _normalize(np.exp(result.x - result.x.max()))
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] beta={beta:.1f}: score={sc:.10f}")
            a = a_new
        except Exception as e:
            print(f"    [{label}] beta={beta:.1f}: error {e}")
    
    return best_a, best_sc


def _projected_grad_descent(a0, deadline, beta, steps=5000, label=""):
    """Projected gradient descent with log-sum-exp smoothing."""
    n = len(a0)
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 1e-4
    patience = 0
    
    for step in range(steps):
        if time.monotonic() > deadline:
            break
        
        g = _lse_grad(a, beta, n)
        # Center the gradient (remove component along 'a' for scale invariance)
        g = g - float(np.dot(g, a)) * a
        
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        a_new = _normalize(a_new)
        
        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
            lr = min(lr * 1.01, 1e-2)
        else:
            a = a_new  # Still move, but don't update best
            patience += 1
            lr *= 0.95
            if patience > 100:
                a = _normalize(best_a.copy())
                patience = 0
                lr = 1e-5
    
    return best_a, best_sc
