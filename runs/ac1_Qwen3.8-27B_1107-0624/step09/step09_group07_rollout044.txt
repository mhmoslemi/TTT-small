Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous state-of-the-art:
1. Extended L-BFGS annealing (16 stages) with adaptive K from peak spread
2. Adaptive beta: relaxes (increases) when stuck, tightens when progressing
3. Nesterov-accelerated subgradient with expanding/contracting active set
4. Peak equalization via direct pair surgery on top contributors
5. Aggressive multi-restart with log-spaced sigma schedule
6. Multi-n exploration (8192, 16384, 32768)
7. Duality-residual power metric to guide phase transitions
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoothed_obj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    M = 2 * n - 1
    if K < M:
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
        c_top_min = float(c_top.min())
    else:
        top_idx = np.arange(M)
        c_top = c
        c_top_min = float(c.min())

    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s_e = float(e.sum())

    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros(M)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(M)
        e_full[top_idx] = e
        w = e_full / s_e
        obj = 2.0 * n * (c_max + beta * np.log(s_e))

    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


def _make_smooth_prob(beta, K, n):
    class _Prob:
        def __init__(self):
            self.beta = beta
            self.K = K
            self.n = n

        def __call__(self, z):
            return _topk_smoothed_obj_grad(z, self.beta, self.K, self.n)[0]

        def grad(self, z):
            return _topk_smoothed_obj_grad(z, self.beta, self.K, self.n)[1]
    return _Prob()


def _peak_spread(c, frac=0.02):

    if c_max < 1e-15:
        return 0, 1
    threshold = c_max * (1.0 - frac)
    n_active = int(np.sum(c >= threshold))
    return c_max, max(1, n_active)

def _peak_surgery(a_init, deadline, min_time_remaining=30.0):
    """Directly reduce top peaks by adjusting contributing elements."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    M = 2 * n - 1

    patience = 0

            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Focus on top 3 peaks
        top_k = np.argsort(c)[::-1][:3]
        # Equalize them
        target = float(c[top_k].min())

        for k_peak in top_k:
            if c[k_peak] <= target * 1.001:
                continue

            excess = c[k_peak] - target
            if excess < 1e-15:
                continue

            # Compute contributions
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            for i in range(i_lo, i_hi + 1):
                j = k_peak - i
                if j < n:
                    contrib[i] += a[j]
                    if i != j:
                        contrib[j] += a[i]

            # Reduce top contributors, add to bottom
            n_adj = max(2, int(n * 0.003))
            top_contrib = np.argsort(contrib)[::-1][:n_adj]
            bot_contrib = np.argsort(contrib)[:n_adj]

            delta = np.zeros(n)
            red_amount = excess * 0.3
            for idx in top_contrib:
                if contrib[idx] > 1e-15 and a[idx] > 1e-15:
                    frac = min(0.5, red_amount / (2.0 * contrib[idx] * a[idx] + 1e-30))
                    delta[idx] = -frac * a[idx]
            for idx in bot_contrib:
                delta[idx] = -delta.sum() / max(1, n_adj)

            # Line search
            alpha = 1.0
            improved = False
            for ls in range(25):
                a_try = a + alpha * delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    alpha *= 0.5
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    a = a_try
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                patience += 5

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.15
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2) + 0.03
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_heavy":
        t = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * (4.0 * t * (1.0 - t)) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

        p3_deadline = min(deadline - 350, time.monotonic() + 150)
        a3, sc3 = _peak_surgery(best_seq, p3_deadline, min_time_remaining=30)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [surgery] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        p4_deadline = min(deadline - 300, time.monotonic() + 250)
        a4 = _run_lbfgs_adaptive(best_seq, p4_deadline, min_time_remaining=50)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Aggressive multi-restart ===
    print(f"\n[P5] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        sigmas = [0.02, 0.04, 0.06, 0.08, 0.12, 0.03, 0.1, 0.15, 0.05, 0.2]
        n_restarts = min(10, max(3, int((deadline - 200 - time.monotonic()) / 35)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 100:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            p5_dl = min(deadline - 80, time.monotonic() + 30)
            a_r = _short_lbfgs(a_pert, p5_dl, min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_{i},sigma={sigma}] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  restart_{i}: {sc_r:.10f} (no improve) t={time.monotonic()-t_start:.1f}s")

        # Try different profiles from scratch
        profiles = ["cos_taper", "edge_heavy", "gauss_center", "bimodal", "asym"]
        for prof_kind in profiles:
            if time.monotonic() > deadline - 60:
                break
            a_prof = _make_profile(len(best_seq), prof_kind)
            p5b_dl = min(deadline - 50, time.monotonic() + 25)
            a_p = _short_lbfgs(a_prof, p5b_dl, min_time_remaining=5)
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_seq = a_p.copy()
                best_score = sc_p
                print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = [16384, 32768, 8192]
        n_options = [x for x in n_options if x != n_main]

        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            p6_dl = min(deadline - 30, time.monotonic() + 40)
            a_opt = _short_lbfgs(a_res, p6_dl, min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  n={n_try}: {sc_opt:.10f} (no improve)")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 3, min_time_remaining=3)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")

        # Final tiny L-BFGS
        remaining = deadline - 3 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0001, 1e-10)
            K_fin = max(5, int((2 * n - 1) * 0.003))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _make_smooth_prob(beta_fin, K_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "tapered")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "tapered")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
