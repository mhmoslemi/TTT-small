"""
Multi-strategy optimizer combining Fourier-domain PAPR flattening, adaptive L-BFGS,
and KKT active-set refinement to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key new ideas: Fourier-domain spectral flattening for diversification,
more aggressive beta schedules, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_a(a, beta, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    n = len(a)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    grad_a = _grad_soft_a(a, beta, w)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _ProbObj:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.01

    elif kind == "periodic":

def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _fourier_flatten(a, n_iter=50, alpha=0.3, deadline=None):
    """Fourier-domain spectral flattening (PAPR reduction style).
    Iteratively compresses the dynamic range of |FFT(a)|^2 while
    preserving non-negativity via projection.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    phase = np.angle(A)
    mag = np.abs(A)
    mag_mean = mag.mean()

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compress dynamic range of |A|
        mag_norm = mag / (mag + 1e-15)
        mag_new = mag * (mag_norm ** alpha) * (1.0 / (mag_norm.mean() ** alpha + 1e-15))
        # Alternative: soft thresholding
        # mag_new = mag / (1.0 + mag / (mag_mean * 5.0))
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            A = np.fft.rfft(a_new, N)
            phase = np.angle(A)
            mag = np.abs(A)
        a = a_new
    return best_a, best_sc


def _kkt_fixed_point(a_init, n_iter=200, deadline=None):
    """KKT fixed-point iteration: alternate between
    (1) given w on active peaks, solve for a making cross-correlation constant
    (2) given a, update w to make cross-correlation most constant
    Uses multiplicative weights update.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (s * s)
        s = a.sum()
        sc = 2.0 * n * c_max / (s * s)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        # Active set with tolerance
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute cross-correlation b[j] = (1/|S|) * sum_{k in S} a[k-j]
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        b = wc[n - 1:2 * n - 1]  # shape (n,)

        # Target: make b constant on support
        support = a > 1e-10
        if support.sum() < 2:
            break
        b_mean = float(np.mean(b[support]))

        # Multiplicative weights update: reduce a[j] where b[j] > b_mean
        eta = 0.02
        a_new = a.copy()
        a_new[support] *= np.exp(-eta * (b[support] - b_mean) / (np.std(b[support]) + 1e-15))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= sc + 1e-12:
            # Try smaller eta
            found = False
            for trial_eta in [0.01, 0.005, 0.002, 0.001]:
                a_try = a.copy()
                a_try[support] *= np.exp(-trial_eta * (b[support] - b_mean) / (np.std(b[support]) + 1e-15))
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < sc - 1e-12:
                    a_new = a_try
                    found = True
                    break
            if not found:
                continue
        a = a_new
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False, tag=""):
    """L-BFGS with adaptive beta schedule - improved version."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Compute initial gap
    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    # More aggressive beta schedule with many levels

        1.0, 0.5, 0.25, 0.15, 0.08, 0.04,
        0.02, 0.01, 0.005, 0.003,
        0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 0.00005, 0.000025, 0.00001,
        0.000005, 0.000002, 0.000001, 0.0000005
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Scale iterations

        est_time_per_iter = 0.001 * size_factor

        # For small beta, fewer iterations needed (landscape is sharper)
        beta_ratio = beta / (gap * multipliers[0])

                prob, z, jac=prob.grad, method='L-BFGS-B',

                    'maxls': 100,

        except Exception as e:
            if verbose:

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement with adaptive tolerance."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 500
    tol_frac = 0.01
    min_tol = 0.0001

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        # Identify active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        # Compute average gradient
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        # Support
        support = a > 1e-8 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        # KKT: grad should be constant on support
        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam
        r_std = np.std(residual)
        if r_std < 1e-15:
            break

        # Normalized step
        eta = 0.03
        a_new = a.copy()
        a_new[support] -= eta * residual / (r_std + 1e-15) * lam * 0.1

        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [0.01, 0.005, 0.002, 0.001]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual / (r_std + 1e-15) * lam * 0.1
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue

                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break

                tol_frac = max(min_tol, tol_frac * 0.7)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.95)

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, min_time_remaining=30.0):
    """Subgradient method with Polyak step size for final polishing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    lb = 2.0 * n / (2 * n - 1)
    step_scale = 0.3
    t = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-12:
            best_sc = current_sc
            best_a = a.copy()

        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        gap = current_sc - lb
        if gap < 1e-10:
            break
        g_c = grad - np.mean(grad)
        g_norm_sq = float(np.dot(g_c, g_c))
        if g_norm_sq < 1e-20:
            break
        step = step_scale * gap / g_norm_sq
        step = min(step, 0.05)

        a_new = a - step * g_c
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
        t += 1
        if t % 200 == 0:
            step_scale *= 0.93

    return best_a, best_sc

    print(f"  P1 best: {best_score:.10f}")

    # === Phase 2: Main L-BFGS ===
    print(f"\n[P2] Main L-BFGS (t={time.monotonic()-t_start:.0f}s)")
    p2_budget = min(500, (budget_s - 10) * 0.30)
    p2_end = t_start + p1_budget + p2_budget
    a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=200, verbose=True, tag="P2")
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()
        print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT refinement ===
    print(f"\n[P3] KKT refinement (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 300, time.monotonic() + 200)
        # KKT fixed point
        a3a, sc3a = _kkt_fixed_point(best_seq, n_iter=300, deadline=p3_end - 60)
        if sc3a < best_score:
            best_score = sc3a
            best_seq = a3a.copy()
            print(f"  KKT-FP: {best_score:.10f}")
        # Active set refine
        if time.monotonic() + 30 < p3_end:
            a3b, sc3b = _active_set_refine(best_seq, p3_end, min_time_remaining=10)
            if sc3b < best_score:
                best_score = sc3b
                best_seq = a3b.copy()
                print(f"  Active-set: {best_score:.10f}")

    # === Phase 4: L-BFGS round 2 ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 350)
        a4, sc4 = _run_lbfgs(best_seq, p4_end, min_time_remaining=100, verbose=True, tag="P4")
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n (t={time.monotonic()-t_start:.0f}s)")
    n_options = [32768, 8192, 24576]
    n_options = [x for x in n_options if x != n_main]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        print(f"  n={n_try} resampled: {sc_r:.10f}")
        if time.monotonic() + 60 > deadline:
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p5_end = min(deadline - 60, time.monotonic() + 120)
        a5, sc5 = _run_lbfgs(a_r, p5_end, min_time_remaining=20, verbose=False, tag=f"P5-n{n_try}")
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 6: Fourier diversification + L-BFGS ===
    print(f"\n[P6] Fourier + L-BFGS (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 60 < deadline:
        a_ff2, sc_ff2 = _fourier_flatten(best_seq, n_iter=100, alpha=0.2,
                                         deadline=time.monotonic() + 80)
        if sc_ff2 < best_score:
            best_score = sc_ff2
            best_seq = a_ff2.copy()
            print(f"  Fourier2: {best_score:.10f}")
        # L-BFGS on Fourier result
        if time.monotonic() + 40 < deadline:
            p6_end = min(deadline - 30, time.monotonic() + 80)
            a6, sc6 = _run_lbfgs(a_ff2, p6_end, min_time_remaining=10,
                                  verbose=False, tag="P6")
            if sc6 < best_score:
                best_score = sc6
                best_seq = a6.copy()
                print(f"  P6 L-BFGS: {best_score:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p7_end = min(deadline - 5, time.monotonic() + 100)
        a7, sc7 = _run_lbfgs(best_seq, p7_end, min_time_remaining=5, verbose=False, tag="P7")
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

        if time.monotonic() + 10 < deadline:
            p7b_end = min(deadline - 3, time.monotonic() + 30)
            a7b, sc7b = _subgradient_polish(best_seq, p7b_end, min_time_remaining=3)
            if sc7b < best_score:
                best_score = sc7b
                best_seq = a7b.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
