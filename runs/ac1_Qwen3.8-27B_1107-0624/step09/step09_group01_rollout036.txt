"""
LP-active-set optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses exact LP solves (HiGHS) on active constraint sets with dual-variable
weighting for Newton polishing, multi-profile initialization, and adaptive
K-schedule. Designed to beat 1.505660 by leveraging LP duality for better
directional information than uniform top-K weighting.

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _self_conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_tapered(n, exponent=0.6):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** exponent
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    t = np.linspace(0, 1, n)
    a = 1.0 - eps * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _build_sparse_conv(a, indices):
    """Build K x n sparse matrix: row i gives conv(a, g)_{indices[i]} as linear in g."""
    n = len(a)
    K = len(indices)
    row_arr = np.empty(K * n, dtype=np.int32)
    col_arr = np.empty(K * n, dtype=np.int32)
    val_arr = np.empty(K * n, dtype=np.float64)
    nnz = 0
    for idx_i in range(K):
        k = int(indices[idx_i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        count = j_end - j_start + 1
        if count <= 0:
            continue
        js = np.arange(j_start, j_end + 1)
        row_arr[nnz:nnz + count] = idx_i
        col_arr[nnz:nnz + count] = js
        val_arr[nnz:nnz + count] = a[k - js]
        nnz += count
    return sp.csr_matrix((val_arr[:nnz], (row_arr[:nnz], col_arr[:nnz])), shape=(K, n))


def _solve_lp_for_direction(a, M, active_indices, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns (g, dual_weights)."""
    K = len(active_indices)
    A_mat = _build_sparse_conv(a, active_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    if res.status != 0:
        return None, None
    g = res.x
    s = float(g.sum())
    if s < 1e-15:
        return None, None
    # Dual variables for inequality constraints
    duals = np.zeros(K)
    try:
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            duals = np.asarray(res.ineqlin.marginals, dtype=np.float64)
            duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    if duals.sum() < 1e-30:
        duals = np.ones(K) / K
    return g, duals / duals.sum()

def _hankel_matvec(v, W, n):
    """Compute H @ v where H is the Hankel operator from weights W on self-conv."""
    v_rev = v[::-1]
    out_len = len(v_rev) + len(W) - 1
    N = _next_pow2(out_len)
    F_v = np.fft.rfft(v_rev, N)
    F_W = np.fft.rfft(W, N)
    conv = np.fft.irfft(F_v * F_W, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _cg_hankel(W, n, rhs, max_iter=300, tol=1e-12):
    """CG solve for Hankel system H @ x = rhs."""
    x = np.zeros(n)
    r = rhs.copy()
    p = r.copy()
    rs = float(r @ r)
    rs0 = rs
    if rs0 < 1e-30:
        return x
    for i in range(max_iter):
        Hp = _hankel_matvec(p, W, n)
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30 or not np.isfinite(pHp):
            break
        alpha = rs / pHp
        x_new = x + alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new) or rs_new > rs * 1.1:
            break
        x = x_new
        r = r_new
        p = (rs_new / rs) * p + r
        rs = rs_new
        if rs < tol * tol * rs0:
            break
    return x

def _resample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        W_k = np.zeros(len(c_best))
        W_k[active] = 1.0 / len(active)
        t_end = min(deadline - 80, time.monotonic() + 60)
        a_k, sc_k = _newton_polish(best_seq, W_k, t_end, max_rounds=100)
        update_best(a_k, sc_k, f"polish_K{K_try}")
        c_best = _self_conv(best_seq)
    
    # Phase 5: Alternating LP + Newton cycles
    print(f"\n[Phase 5] Alternating cycles (t={time.monotonic()-t_start:.1f}s)")
    t_p5_end = min(deadline - 150, t_start + 1400)
    for cyc in range(10):
        if time.monotonic() > t_p5_end:
            break
        t_lp = min(t_p5_end, time.monotonic() + 40)
        a_c1, sc_c1, duals_c1 = _lp_phase(best_seq, t_lp, max_K=2000)
        update_best(a_c1, sc_c1, f"cyc_lp_{cyc}")
        
        t_np = min(t_p5_end, time.monotonic() + 40)
        a_c2, sc_c2 = _newton_polish(best_seq, duals_c1, t_np, max_rounds=80)
        update_best(a_c2, sc_c2, f"cyc_np_{cyc}")
    
    # Phase 6: Multi-n exploration
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_options = [n0 // 2, n0, n0 * 2]
    n_options = [n for n in n_options if 4096 <= n <= 65536]
    if 16384 not in n_options:
        n_options.append(16384)
    if 32768 not in n_options:
        n_options.append(32768)
    
    for n_try in sorted(set(n_options)):
        if time.monotonic() > deadline - 60:
            break
        a_res = _resample(best_seq, n_try)
        sc_res = _score(a_res)
        if sc_res < best_score:
            update_best(a_res, sc_res, f"resample_n{n_try}")
        # Quick LP on resampled
        if time.monotonic() < deadline - 30:
            t_res_end = min(deadline - 25, time.monotonic() + 15)
            a_ro, sc_ro, _ = _lp_phase(a_res, t_res_end, max_K=200)
            update_best(a_ro, sc_ro, f"opt_n{n_try}")
    
    # Phase 7: Final ultra-fine polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        c_fin = _self_conv(best_seq)
        sorted_fin = np.argsort(c_fin)[::-1]
        for K_fin in [10, 50, 100]:
            if time.monotonic() > deadline - 3:
                break
            active_fin = sorted_fin[:K_fin]
            W_fin = np.zeros(len(c_fin))
            W_fin[active_fin] = 1.0 / K_fin
            a_fin, sc_fin = _newton_polish(best_seq, W_fin, deadline - 1, max_rounds=50)
            update_best(a_fin, sc_fin, f"final_K{K_fin}")
            c_fin = _self_conv(best_seq)
            sorted_fin = np.argsort(c_fin)[::-1]
    
    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered(256, 0.6)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_tapered(8192, 0.6)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
