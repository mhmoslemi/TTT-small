Uses top-k gradient descent with momentum, adaptive ILS, and multi-resolution initialization.

import time

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_topk_gradient(a, c, k, temperature):
    """
    Compute gradient of soft top-k objective.
    Uses weights w_i = exp((c_i - c_max) / T) for top-k entries.
    Returns gradient of 2n * sum(w_i * c_i) / sum(w_i) (a smoothed peak measure).
    """
    n = len(a)
    c_max = float(c.max())
    
    # Get top-k indices
    if k >= len(c):
        idx = np.arange(len(c))
    else:
        # Efficient: use argpartition for top-k
        part_idx = np.argpartition(c, -k)[-k:]
        idx = part_idx
    
    # Compute weights
    c_topk = c[idx]
    shifted = (c_topk - c_max) / temperature
    w_topk = np.exp(np.clip(shifted, -700, 0.0))
    w_sum = w_topk.sum()
    if w_sum < 1e-300:
        w_topk = np.ones_like(w_topk) / len(w_topk)
        w_sum = 1.0
    else:
        w_topk = w_topk / w_sum
    
    # Build full weight vector
    w = np.zeros(len(c))
    w[idx] = w_topk
    
    # Gradient: (4n/w_sum) * conv(w, a_reversed)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 4.0 * n * h
    
    # Project onto tangent space of sum(a)=1
    grad = grad - np.mean(grad)
    return grad


def _adaptive_topk_optimize(a_init, deadline, min_time_remaining, 
                            tol_start=0.005, tol_end=0.0005,
                            lr_init=0.03, momentum_init=0.85,
                            k_init=20, k_max=100, temp_init=0.01):
    """
    Gradient descent with momentum and adaptive soft-top-k.
    Adapts: learning rate, momentum, k, and temperature based on progress.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    
    lr = lr_init
    momentum = momentum_init
    k = k_init
    temp = temp_init
    no_improve = 0
    max_stall = 200
    stall_count = 0
    patience = 50
    recent_improvement = []
    
    while time.monotonic() < deadline - min_time_remaining:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
            recent_improvement.append(1)
        else:
            no_improve += 1
            recent_improvement.append(0)
        
        # Keep last 100 improvement flags
        if len(recent_improvement) > 100:
            recent_improvement = recent_improvement[-100:]
        
        grad = _soft_topk_gradient(a, c, k, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Adaptive momentum: increase if making consistent progress
        if len(recent_improvement) >= 50:
            improve_rate = np.mean(recent_improvement)
            if improve_rate > 0.3:
                momentum = min(0.92, momentum * 1.005)
            elif improve_rate < 0.05:
                momentum = max(0.7, momentum * 0.99)
        
        # Heavy ball momentum
        v = momentum * v - lr * grad
        alpha = 1.0
        improved = False
        for _ in range(40):
            a_try = _normalize(a + alpha * v)
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            v = np.zeros(n)
            if no_improve % 20 == 0:
                lr *= 0.7
                k = min(k_max, int(k * 1.3) + 1)
                temp *= 0.8
            if no_improve >= patience:
                stall_count += 1
                if stall_count >= 4:
                    # Restart from best with different parameters
                    a = best_a.copy()
                    v = np.zeros(n)
                    lr = lr_init * 0.5
                    momentum = 0.8
                    k = max(5, k // 2)
                    temp = temp_init * 0.5
                    no_improve = 0
                    stall_count = 0
                    patience = max(30, patience * 0.7)
                else:
                    a = best_a.copy()
                    v = np.zeros(n)
                    lr *= 0.6
                    no_improve = 0
        else:
            if no_improve >= patience:
                no_improve = 0
            # Gradually tighten
            if temp > tol_end:
                temp = max(tol_end, temp * 0.99)
            if k < k_max:
                k = min(k_max, k + 1)
            lr = min(lr_init * 2, lr * 1.01)
    
    return best_a, _score(best_a)


def _deflation_step(a, c, strength=0.05):
    """
    Deflation: identify the peak direction and reduce alignment with it.
    The peak at position k* means a is correlated with its k*-shift.
    We subtract a small component in that direction.
    """
    n = len(a)
    k_star = int(np.argmax(c))
    
    # The "peak direction": a shifted by k_star
    # a[j] contributes to peak at k* through a[j]*a[k*-j]
    # The "alignment vector" is essentially a reversed and shifted
    if k_star < n:
        # a_rev_shifted[j] = a[k_star - j] if valid
        start = max(0, k_star - (n - 1))
        end = min(n, k_star + 1)
        length = end - start
        if length <= 0:
            return a.copy()
        a_shifted = np.zeros(n)
        src_start = max(0, n - 1 - (k_star - start))
        src_end = src_start + length
        a_shifted[start:end] = a[n - 1 - (k_star - start):n - src_start]
    else:
        # k_star >= n, no direct alignment
        return a.copy()
    
    # Normalize the alignment vector
    anorm = np.linalg.norm(a_shifted)
    if anorm < 1e-15:
        return a.copy()
    a_shifted = a_shifted / anorm
    
    # Project and subtract
    proj = np.dot(a, a_shifted)
    a_new = a - strength * proj * a_shifted
    return _normalize(a_new)


def _peak_directed_perturb(a, c, rng, sigma):
    """
    Perturb positions that contribute most to the peak of the autocorrelation.
    """
    n = len(a)
    k_star = int(np.argmax(c))
    
    # Find positions with high contribution to the peak
    if 0 <= k_star < 2 * n - 2:
        contrib = np.zeros(n)
        for j in range(n):
            if 0 <= k_star - j < n:
                contrib[j] = a[j] * a[k_star - j]
        
        # Perturb top-contributing positions more
        n_top = max(10, n // 20)
        top_idx = np.argpartition(contrib, -n_top)[-n_top:]
        
        a_new = a.copy()
        # Reduce top contributors, increase bottom ones
        bottom_idx = np.argpartition(contrib, n_top // 4)[:n_top // 4]
        
        a_new[top_idx] *= np.exp(-sigma * rng.random(len(top_idx)))
        a_new[bottom_idx] *= np.exp(sigma * 0.5 * rng.random(len(bottom_idx)))
        return _normalize(a_new)
    else:
        # Fallback: random perturbation
        z = np.log(np.maximum(a, 1e-30))
        z += sigma * rng.standard_normal(n)
        return _softmax(z)


def _block_perturb(a, rng, sigma, block_frac=0.1):
    """Perturb a random block of the sequence."""
    n = len(a)
    block_size = max(50, int(n * block_frac))
    start = int(rng.integers(0, n - block_size))
    a_new = a.copy()
    z = np.log(np.maximum(a_new[start:start + block_size], 1e-30))
    z += sigma * rng.standard_normal(block_size)
    a_new[start:start + block_size] = np.exp(z - np.max(z))
    return _normalize(a_new)


def _mirror_perturb(a, rng, seg_frac=0.1):
    """Mirror a random segment."""
    n = len(a)
    seg_len = max(50, int(n * seg_frac))
    start = int(rng.integers(0, n - seg_len))
    a_new = a.copy()
    a_new[start:start + seg_len] = a_new[start:start + seg_len][::-1]
    z = np.log(np.maximum(a_new, 1e-30))
    z += 0.003 * rng.standard_normal(n)
    return _softmax(z)


def _ils_round(a_init, deadline, rng, min_time_remaining=60.0, temperature=1.0):
    """
    Iterated Local Search with peak-directed perturbations and adaptive temperature.
    """
    n = len(a_init)
    current_a = _normalize(a_init.copy())
    current_sc = _score(current_a)
    best_a = current_a.copy()
    best_sc = current_sc
    
    accept_temp = temperature
    no_improve = 0
    attempt = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        # Choose perturbation strategy
        r = rng.random()
        if r < 0.25:
            c_cur = _full_convolve(current_a, current_a)
            a_perturbed = _peak_directed_perturb(current_a, c_cur, rng, sigma=0.03)
        elif r < 0.45:
            a_perturbed = _block_perturb(current_a, rng, sigma=0.05, block_frac=0.05)
        elif r < 0.60:
            a_perturbed = _mirror_perturb(current_a, rng)
        elif r < 0.75:
            c_cur = _full_convolve(current_a, current_a)
            a_perturbed = _deflation_step(current_a, c_cur, strength=0.03 + 0.04 * rng.random())
        elif r < 0.90:
            z = np.log(np.maximum(current_a, 1e-30))
            z += 0.01 * rng.standard_normal(n)
            a_perturbed = _softmax(z)
        else:
            z = np.log(np.maximum(current_a, 1e-30))
            z += 0.04 * rng.standard_normal(n)
            a_perturbed = _softmax(z)
        
        # Quick local optimization
        opt_deadline = min(deadline - min_time_remaining, time.monotonic() + 5)
        a_opt, sc_opt = _adaptive_topk_optimize(
            a_perturbed, opt_deadline, min_time_remaining=2,
            tol_start=0.003, tol_end=0.0005,
            lr_init=0.015, momentum_init=0.82,
            k_init=10, k_max=50, temp_init=0.005
        )
        
        if sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_a = a_opt.copy()
            current_a = a_opt.copy()
            current_sc = sc_opt
            no_improve = 0
            print(f"    ILS NEW BEST: {best_sc:.10f} (attempt={attempt}) t={time.monotonic()-t_start:.0f}s")
            # Reset temperature
            accept_temp = max(1.0, accept_temp * 0.9)
        elif sc_opt < current_sc - 1e-12:
            # Accept if better than current (local improvement)
            current_a = a_opt.copy()
            current_sc = sc_opt
            no_improve += 1
        else:
            # Metropolis acceptance
            delta = sc_opt - best_sc
            if delta < 0 or rng.random() < np.exp(-delta / max(accept_temp * 1e-6, 1e-10)):
                current_a = a_opt.copy()
                current_sc = sc_opt
            no_improve += 1
        
        if no_improve > 30:
            # Increase temperature to explore more
            accept_temp *= 1.5
            no_improve = 0
        
        attempt += 1
    
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n > new_n:
        # Downsample: average blocks
        ratio = old_n / new_n
        indices = (np.arange(new_n) * ratio).astype(int)
        a_new = a[indices]
    else:
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (2.0 * t - 1.0) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        edge = 0.2
        a = np.where(np.abs(t - 0.5) < edge, 1.0, 0.5 * (1.0 + np.cos(np.pi * (np.abs(t - 0.5) - edge) / (0.5 - edge))))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
