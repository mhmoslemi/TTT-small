"""
Refined multi-phase optimizer focusing on peak equalization and aggressive
smoothing continuation. Key improvements: cycling peak reduction, spectral
perturbation restarts, and tighter L-BFGS convergence with deeper annealing.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]


def _run_lbfgs_deep(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very deep annealing schedule (more stages, lower final beta)."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    current_sc = _score(a)
    current_a = a.copy()
    c_cur = _full_convolve(a, a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(200, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.005)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    schedule = [
        (1.0, 0.15), (0.7, 0.10), (0.5, 0.08), (0.35, 0.06),
        (0.25, 0.05), (0.15, 0.035), (0.10, 0.025), (0.06, 0.015),
        (0.03, 0.01), (0.015, 0.007), (0.008, 0.005), (0.004, 0.003),
        (0.002, 0.002), (0.001, 0.001), (0.0005, 1),
    ]
    for beta_mult, K_frac in schedule:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        beta = base_beta * beta_mult
        if K_frac == 1:
            K = 3
        else:
            K = max(3, int((2 * n - 1) * K_frac))
        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(8000, max(100, int(remaining / (est_time * 15))))
        prob = _TopkSmoProb(beta, K, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxcor': 120, 'maxls': 60}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _cycling_peak_reduction(a_init, deadline, min_time_remaining=30.0):
    """Cycle through top peaks, reducing each in turn. More balanced than single-peak approach."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1
    patience = 0
    max_patience = 500
    cycle_count = 0
    while time.monotonic() + min_time_remaining < deadline:
        cycle_count += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        top_K = max(5, min(20, int(n_2nm1 * 0.005)))
        top_idx_all = np.argpartition(c, -top_K)[-top_K:]
        top_idx_all = top_idx_all[np.argsort(c[top_idx_all])[::-1]]
        peak_to_reduce = top_idx_all[(cycle_count - 1) % min(top_K, len(top_idx_all))]
        i_lo = max(0, int(peak_to_reduce) - (n - 1))
        i_hi = min(n - 1, int(peak_to_reduce))
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = int(peak_to_reduce) - i
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
        n_reduce = max(2, int(n * 0.003))
        top_contrib = np.argsort(contrib)[::-1][:n_reduce]
        reduction_amount = c[peak_to_reduce] * 0.008
        delta = np.zeros(n)
        for idx in top_contrib:
            if contrib[idx] > 1e-15:
                delta[idx] = -reduction_amount / (2.0 * contrib[idx] + 1e-30)
        low_mask = contrib < np.percentile(contrib, 20)
        low_idx = np.where(low_mask)[0]
        if len(low_idx) > 0:
            n_low = min(n_reduce, len(low_idx))
            for idx in low_idx[:n_low]:
                delta[idx] = reduction_amount * 0.4 / (n_low * (a[idx] + 1e-10) + 1e-30)
        delta -= delta.mean()
        alpha = 1.0
        improved = False
        for ls in range(35):
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            patience += 5
    return best_a, best_sc


def _peak_equalization(a_init, deadline, min_time_remaining=20.0):
    """Explicitly minimize the spread of top-K peaks using gradient descent on the spread objective."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1
    lr = 0.01
    K = 8
    no_improve = 0
    momentum = 0.5
    velocity = np.zeros(n)
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-15:
            break
        w = np.zeros(n_2nm1)
        excess = top_vals - top_mean
        pos_excess = np.maximum(excess, 0)
        neg_excess = np.maximum(-excess, 0)
        weight = 0.5 + 0.5 * pos_excess / (top_std + 1e-15) - 0.3 * neg_excess / (top_std + 1e-15)
        weight = np.maximum(weight, 0.1)
        w[top_idx] = weight
        w = w / w.sum()
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                lr = min(0.05, lr * 1.02)
            else:
                for ls in range(15):
                    a_try = a + 0.5 * velocity
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        break
                    a_try = a_try / s_try
                    c_try = _full_convolve(a_try, a_try)
                    sc_try = 2.0 * n * float(c_try.max())
                    if sc_try < current_sc - 1e-16:
                        a = a_try
                        improved = True
                        break
        if not improved:
            velocity = velocity * 0.5
            lr *= 0.7
            if no_improve % 50 == 0:
                K = max(3, K - 1)
    return best_a, best_sc


def _spectral_perturb(a, rng, n_modes=5):
    """Perturb in the frequency domain for more structured exploration."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    A_abs = np.abs(A)
    A_phase = np.angle(A)
    n_real = N // 2 + 1
    for _ in range(n_modes):
        k = rng.integers(1, n_real)
        delta = rng.normal(0, 0.05)
        A_abs[k] = np.maximum(A_abs[k] + delta, 0.01)
        if k < n_real - 1:
            A_abs[n_real - 1 - k] = A_abs[k]
    A_new = A_abs * np.exp(1j * A_phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def _subgradient_improved(a_init, deadline, min_time_remaining=60.0):
    """Improved subgradient with adaptive active set and BB steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    n_2nm1 = 2 * n - 1
    active_frac = 0.005
    momentum = 0.6
    lr = 0.015
    lr_min = 1e-5
    lr_max = 0.08
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    prev_a = None
    prev_g = None
    iteration = 0
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        n_active = max(5, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.05
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-16:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.6
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.002, active_frac * 0.97)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.5)
            active_frac = min(0.05, active_frac * 1.5)
        if stall > max_stall:
            active_frac = min(0.04, active_frac * 2.0)
            lr = max(lr_min, lr * 0.2)
            momentum = min(0.85, momentum + 0.05)
            stall = 0
    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.003
    tol = 0.003
    no_improve = 0
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        if current_sc < best_sc - 1e-16:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        elif n_active > 50:
            top_idx = np.argsort(c)[::-1][:50]
            active = np.zeros_like(active)
            active[top_idx] = True
            n_active = 50
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / n_active
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        alpha = min(lr, 0.3 / (gnorm + 1e-15))
        improved = False
        for ls in range(35):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-16:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(5e-5, tol * 0.6)
                lr *= 0.6
            if no_improve > 400:
                a = best_a.copy()
                tol = max(5e-5, tol * 0.4)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 5e-5:
                tol = max(5e-5, tol * 0.98)
            lr = min(0.008, lr * 1.01)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: deep L-BFGS continuation, cycling peak reduction,
    peak equalization, spectral perturbation restarts, multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_init = 16384
        t = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * t) ** 0.5
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()
        best_score = _score(best_seq)
        print(f"No parent, sine taper n={n_init}, score={best_score:.10f}")
    # Phase 1: Deep L-BFGS
    print(f"\n[P1] Deep L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_lbfgs_deep(best_seq, deadline, min_time_remaining=700)
    sc1 = _score(a1)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_deep] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    # Phase 2: Cycling peak reduction
    print(f"\n[P2] Cycling peak reduction (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a2, sc2 = _cycling_peak_reduction(best_seq, min(deadline - 400, time.monotonic() + 400), min_time_remaining=50)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [cycling] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    # Phase 3: Peak equalization
    print(f"\n[P3] Peak equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a3, sc3 = _peak_equalization(best_seq, min(deadline - 300, time.monotonic() + 300), min_time_remaining=50)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [equalize] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    # Phase 4: L-BFGS round 2
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        a4 = _run_lbfgs_deep(best_seq, min(deadline - 250, time.monotonic() + 300), min_time_remaining=50)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    # Phase 5: Subgradient
    print(f"\n[P5] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a5, sc5 = _subgradient_improved(best_seq, min(deadline - 150, time.monotonic() + 200), min_time_remaining=50)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
    # Phase 6: Spectral perturbation restarts
    print(f"\n[P6] Spectral restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        n_restarts = 6
        restart_time = min(20, (deadline - 100 - time.monotonic()) / n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > deadline - 80:
                break
            a_pert = _spectral_perturb(best_seq, rng, n_modes=3 + i)
            a_r = _run_lbfgs_deep(a_pert, min(deadline - 60, time.monotonic() + max(5, restart_time)), min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [spec_restart_{i}] t={time.monotonic()-t_start:.1f}s")
        # Log perturbation restarts
        for i in range(4):
            if time.monotonic() > deadline - 60:
                break
            sigma = [0.02, 0.04, 0.06, 0.08][i]
            a_pert = _perturb_log(best_seq, sigma, rng)
            a_r = _run_lbfgs_deep(a_pert, min(deadline - 50, time.monotonic() + 15), min_time_remaining=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [log_restart_{i}] t={time.monotonic()-t_start:.1f}s")
    # Phase 7: Multi-n exploration
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        n_main = len(best_seq)
        n_options = [24576, 8192, 32768]
        n_options = [x for x in n_options if x != n_main and x <= 32768]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 40:
                break
            a_res = _resample(best_seq, n_try)
            a_opt = _run_lbfgs_deep(a_res, min(deadline - 20, time.monotonic() + 30), min_time_remaining=5)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    # Phase 8: Final polish
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-9)
            K_fin = 3
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _TopkSmoProb(beta_fin, K_fin, n)
            max_iter = min(5000, max(50, int(remaining / 0.02)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-20,
                                           'gtol': 1e-16, 'maxcor': 100})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            t = np.linspace(0, 1, 256)
            final = np.sin(np.pi * t)
            final = final / final.sum()
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    t = np.linspace(0, 1, 256)
    fallback = np.sin(np.pi * t)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
