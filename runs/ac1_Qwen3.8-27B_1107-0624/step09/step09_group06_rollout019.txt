"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive L-BFGS with log-sum-exp smoothing, subgradient method with
Nesterov momentum, and multi-resolution exploration. Key improvements:
adaptive beta scheduling based on peak spread, more aggressive iteration
counts, warm-starting between beta levels, and randomized local search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(x - theta, 0.0)


def _conv_grad_single(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        idx = k - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] += a[idx]
    if j_lo <= j_hi:
        for j in range(j_lo, j_hi + 1):
            if j != k - j:
                grad[j] += a[k - j]
    return grad


def _conv_grad_multi(a, active_indices):
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += a[idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1) - k
            grad[j_lo2:j_hi2 + 1] += a[idx]
        # Subtract the double-counted diagonal term
        if 0 <= 2 * k <= 2 * (n - 1) and 0 <= k <= n - 1 and 2 * k == 2 * k:
            pass
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=True):
    """L-BFGS with adaptive beta based on convergence and peak spread."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a, current_sc

    c_sorted = np.sort(c_cur)[::-1]
    K_check = min(50, len(c_sorted))
    gap = c_sorted[0] - c_sorted[K_check - 1]
    if gap < max_c * 1e-6:
        gap = max_c * 1e-6
    beta_start = max(gap, max_c * 0.01)

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    beta = beta_start
    total_iters = 0
    min_beta = max_c * 1e-10

    while time.monotonic() + min_time_remaining < deadline and beta > min_beta:
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate time per iteration
        est_time = 0.003 * max(1.0, n / 8192.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            total_iters += result.nit
            a_new = _softmax(result.x)
            sc_new = _score(a_new)

            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    L-BFGS: beta={beta:.2e}, score={current_sc:.12f}, iters={result.nit}")

            # Determine next beta based on convergence
            if result.status == 0 and result.nit < max_iter:
                # Converged: reduce beta more aggressively
                beta *= 0.3
            elif sc_new < current_sc + 1e-12:
                # Some progress: moderate reduction
                beta *= 0.5
            else:
                # Little progress: smaller reduction or increase
                beta *= 0.75

            # Cap beta to not go too small relative to current spread
            c_new = _full_convolve(current_a, current_a)
            c_s = np.sort(c_new)[::-1]
            cur_gap = c_s[0] - c_s[min(30, len(c_s) - 1)]
            beta = max(beta, cur_gap * 0.1)

        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at beta={beta:.2e}: {e}")
            beta *= 0.5

    return current_a, current_sc


def _nesterov_subgrad(a_init, deadline, min_time_remaining=30.0,
                      K_init=10, K_max=200, seed=0, verbose=True):
    """Nesterov-accelerated subgradient with adaptive K and restarts."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.8
    base_lr = 0.003
    lr = base_lr
    K = K_init
    t = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive K
        c_sorted = np.sort(c)[::-1]
        gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if gap < c_max * 0.003 and K < K_max:
            K = min(K_max, int(K * 1.4) + 1)
        elif gap > c_max * 0.05 and K > K_init:
            K = max(K_init, int(K * 0.7))

        # Select active peaks
        c_idx = np.argsort(c)[::-1]
        active = c_idx[:K]

        # Compute subgradient
        grad = _conv_grad_multi(a, active)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-18:
            break

        # Step size
        effective_lr = lr / (gnorm + 1e-18) * 0.15

        # Nesterov step
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
            lr = min(0.01, lr * 1.001)
        else:
            v = 0.5 * v
            lr *= 0.92
            a = a_proj

        # Periodic restart
        if no_improve > 0 and no_improve % 300 == 0:
            stall_count += 1
            if stall_count > 8:
                break
            perturb_scale = 0.0005 * (1.0 + 0.1 * stall_count)
            perturb = perturb_scale * rng.standard_normal(n)
            a_pert = _project_simplex(a + perturb)
            c_pert = _full_convolve(a_pert, a_pert)
            sc_pert = 2.0 * n * float(c_pert.max())
            if sc_pert < best_sc:
                a = a_pert
                best_a = a.copy()
                best_sc = sc_pert
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0

        if t % 1000 == 0 and verbose:
            print(f"    Nested t={t}: best={best_sc:.12f}, K={K}, lr={lr:.6f}")

    return best_a, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        return np.abs(np.random.default_rng(123).standard_normal(n)) / n
    else:
        return np.ones(n) / n


def _local_search(a_init, deadline, min_time_remaining=20.0, seed=42, verbose=True):
    """Random local search with small perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)

    scale = 0.001
    it = 0
    while time.monotonic() + min_time_remaining < deadline:
        it += 1
        # Random perturbation
        pert = scale * rng.standard_normal(n)
        a_new = _project_simplex(a + pert)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            scale = min(0.01, scale * 1.1)
        else:
            scale *= 0.99
            if scale < 1e-7:
                break

        if it % 500 == 0 and verbose:
            print(f"    LocalSearch it={it}: best={best_sc:.12f}, scale={scale:.2e}")

    return best_a, best_sc


def _two_scale_optimize(a_init, deadline, min_time_remaining=60.0, verbose=True):
    """Coarse-to-fine optimization: optimize at half resolution, then refine."""
    n = len(a_init)
    n_coarse = n // 2
    if n_coarse < 64:
        return a_init, _score(a_init)

    # Downsample
    a_coarse = (a_init[0::2] + a_init[1::2]) / 2.0
    s = a_coarse.sum()
    if s > 1e-15:
        a_coarse = a_coarse / s

    # Optimize at coarse scale
    if time.monotonic() + min_time_remaining * 0.6 < deadline:
        a_coarse, sc_coarse = _adaptive_lbfgs(
            a_coarse,
            min(deadline - (deadline - time.monotonic()) * 0.6, deadline),
            min_time_remaining=min_time_remaining * 0.3,
            verbose=False
        )

    # Upsample and use as initialization
    a_fine = np.zeros(n)
    a_fine[0::2] = a_coarse
    a_fine[1::2] = a_coarse
    # Better upsample: interpolate
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n)
    a_fine = np.interp(x_fine, x_coarse, a_coarse)
    a_fine = np.maximum(a_fine, 0.0)
    s = a_fine.sum()
    if s > 1e-15:
        a_fine = a_fine / s

    # Refine at fine scale
    if time.monotonic() + min_time_remaining < deadline:
        a_fine, sc_fine = _adaptive_lbfgs(
            a_fine, deadline,
            min_time_remaining=min_time_remaining * 0.4,
            verbose=verbose
        )

    return a_fine, sc_fine


def _focused_subgrad(a_init, deadline, min_time_remaining=30.0, seed=0, verbose=True):
    """Focused subgradient: only target the top peak and its neighbors."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    momentum = 0.7
    lr = 0.001
    t = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        # Focus on the single top peak and a few neighbors
        peak_idx = int(np.argmax(c))
        # Gradient of the top peak
        k = peak_idx
        grad = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += a[idx]
            for j in range(j_lo, j_hi + 1):
                if j != k - j:
                    grad[j] += a[k - j]

        # Also add gradient from second peak if close
        c_sorted_idx = np.argsort(c)[::-1]
        if len(c_sorted_idx) > 1 and c[c_sorted_idx[1]] > c_max * 0.99:
            k2 = c_sorted_idx[1]
            j_lo2 = max(0, k2 - n + 1)
            j_hi2 = min(n - 1, k2)
            if j_lo2 <= j_hi2:
                idx2 = k2 - np.arange(j_lo2, j_hi2 + 1)
                grad[j_lo2:j_hi2 + 1] += 0.5 * a[idx2]
                for j in range(j_lo2, j_hi2 + 1):
                    if j != k2 - j:
                        grad[j] += 0.5 * a[k2 - j]

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-18:
            break

        effective_lr = lr / (gnorm + 1e-18) * 0.2
        v = momentum * v - effective_lr * grad_t
        a_cand = a + v
        a_proj = _project_simplex(a_cand)

        c_new = _full_convolve(a_proj, a_proj)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_sc:
            a = a_proj
            lr = min(0.005, lr * 1.002)
        else:
            v = 0.3 * v
            lr *= 0.9
            a = a_proj

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search function: adaptive L-BFGS + subgradient + local search."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.12f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.12f}")

    # === Phase 1: Main adaptive L-BFGS (most of the time) ===
    print(f"\n[P1] Adaptive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = deadline - 400.0
    if time.monotonic() < p1_deadline:
        a1, sc1 = _adaptive_lbfgs(best_seq, p1_deadline, min_time_remaining=60.0, verbose=True)
        if sc1 < best_score:
            best_seq = a1.copy()
            best_score = sc1
            print(f"  NEW BEST: {best_score:.12f} [P1-lbfgs] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Focused subgradient ===
    print(f"\n[P2] Focused subgradient (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = deadline - 300.0
    if time.monotonic() < p2_deadline:
        a2, sc2 = _focused_subgrad(best_seq, p2_deadline, min_time_remaining=30.0, seed=seed, verbose=True)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.12f} [P2-focused] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Nesterov subgradient with wider active set ===
    print(f"\n[P3] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = deadline - 200.0
    if time.monotonic() < p3_deadline:
        a3, sc3 = _nesterov_subgrad(best_seq, p3_deadline, min_time_remaining=30.0,
                                    K_init=5, K_max=150, seed=seed + 1, verbose=True)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.12f} [P3-nesterov] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Second L-BFGS pass (with refined state) ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = deadline - 120.0
    if time.monotonic() < p4_deadline:
        a4, sc4 = _adaptive_lbfgs(best_seq, p4_deadline, min_time_remaining=30.0, verbose=True)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.12f} [P4-lbfgs2] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Local search ===
    print(f"\n[P5] Local search (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = deadline - 80.0
    if time.monotonic() < p5_deadline:
        a5, sc5 = _local_search(best_seq, p5_deadline, min_time_remaining=20.0, seed=seed + 2, verbose=True)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.12f} [P5-local] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[P6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = []
    if n_parent in [16384, 12288, 8192, 24576]:
        candidates = [16384, 12288, 20480, 8192, 24576]
    else:
        candidates = [n_parent, 16384, 12288, 8192]
    for nc in candidates:
        if nc == len(best_seq):
            continue
        n_candidates.append(nc)

    for n_try in n_candidates[:3]:
        if time.monotonic() > deadline - 50:
            break
        a_try = _resample(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try}: init score={sc_try:.12f}")
        if sc_try < best_score:
            best_seq = a_try.copy()
            best_score = sc_try

        # Quick L-BFGS at this n
        remaining = deadline - 40 - time.monotonic()
        if remaining > 30:
            a_opt, sc_opt = _adaptive_lbfgs(a_try, deadline - 20, min_time_remaining=10.0, verbose=False)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.12f} (n={n_try}) [P6] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine L-BFGS ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 10 - time.monotonic()
    if remaining > 20:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        c_sorted = np.sort(c_best)[::-1]
        gap = mb - c_sorted[min(20, len(c_sorted) - 1)]
        beta_fin = max(gap * 0.05, mb * 1e-8)

        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        prob = _LbfgsProb(beta_fin, n)
        max_iter = min(5000, max(50, int(remaining / 0.02)))
        try:
            result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxcor': 80, 'maxls': 50})
            a_fin = _softmax(result.x)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_seq = a_fin.copy()
                best_score = sc_fin
                print(f"  NEW BEST: {best_score:.12f} [P7-final] t={time.monotonic()-t_start:.1f}s")
        except Exception:
            pass

    # Final validation and return
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
