    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=50.0, verbose=False, label=""):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < c0_max * 0.002:
        gap = c0_max * 0.002

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.0015 * size_factor
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 25))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 5000)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxcor': 80,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-15:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    {label} beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}")
        except Exception:
            pass

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, min_time_remaining=30, verbose=False, label="CD"):
    """Vectorized coordinate descent with exact normalization-aware evaluation."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_sc = float(2.0 * n * c.max())

    rng = np.random.default_rng(12345)
    pass_count = 0

    while time.monotonic() + min_time_remaining < deadline:
        # Shuffle coordinates
        order = rng.permutation(n)
        improved = 0

        for jj in order:
            j = int(jj)
            aj = a[j]
            if aj < 1e-15:
                continue

            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo > k_hi:
                continue

            # Precompute: affected k indices and corresponding a[k-j]
            k_idx = np.arange(k_lo, k_hi + 1)
            other_idx = k_idx - j
            other_vals = a[other_idx]

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                # Mask out affected
                unaff_vals = np.empty(2 * n - 1)
                unaff_vals[k_lo:k_hi+1] = -1e30
                # Get max of c where not affected
                affected_mask = np.zeros(2 * n - 1, dtype=bool)
                affected_mask[k_lo:k_hi+1] = True
                c_unaff_max = float(c[~affected_mask].max()) if (~affected_mask).any() else 0.0
            else:
                c_unaff_max = 0.0

            two_j_in = (2 * j >= 0) and (2 * j <= 2 * n - 2)

            # Try candidate d values
            candidates = np.array([0.0, -aj * 0.5, -aj * 0.75, -aj, aj * 0.25])
            best_d = 0.0
            best_f = 2.0 * n * float(c.max())  # current score * (sum=1)^2 = max(c)*2n

            current_max_c = float(c.max())
            S_cur = 1.0
            best_f = current_max_c  # f(0) = max(c) / 1 = max(c)

            for d in candidates:
                x_new = aj + d
                if x_new < -1e-14:
                    continue
                x_new = max(0.0, x_new)
                d = x_new - aj
                if abs(d) < 1e-14:
                    continue
                S = 1.0 + d
                if S < 1e-10:
                    continue
                S2 = S * S

                # Affected values: c[k] + 2*d*a[k-j]
                aff_vals = c[k_idx].copy() + 2.0 * d * other_vals
                # Fix quadratic term at k=2j
                if two_j_in:
                    pos = 2 * j - k_lo
                    if 0 <= pos < len(k_idx):
                        aff_vals[pos] = c[2 * j] + 2.0 * d * aj + d * d

                aff_max = float(aff_vals.max())
                unaff_max = c_unaff_max / S2
                f_val = max(aff_max, unaff_max) / S2

                if f_val < best_f - 1e-15:
                    best_f = f_val
                    best_d = d

            if best_d != 0.0:
                a_new = a.copy()
                a_new[j] = max(0.0, aj + best_d)
                S_new = a_new.sum()
                if S_new < 1e-10:
                    continue
                a_new = a_new / S_new
                # Update c using the formula (avoid full reconvolution)
                # Actually for correctness after multiple updates, recompute periodically
                a = a_new
                improved += 1

        # Recompute c periodically (every pass or after many updates)
        c = _full_convolve(a, a)
        current_sc = float(2.0 * n * c.max())
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
        pass_count += 1

        if verbose and pass_count % 10 == 0:
            print(f"    {label} pass={pass_count}, score={best_sc:.10f}, "
                  f"improved={improved}, t={time.monotonic()-t_start_cd:.0f}s")

        # Stop if no improvement in several passes
        if improved == 0 and pass_count > 3:
            break

    return best_a, best_sc


def _cd_with_recompute(a_init, deadline, min_time_remaining=30, verbose=False, label="CD"):
    """Coordinate descent that periodically recomputes the full convolution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_sc = float(2.0 * n * c.max())

    rng = np.random.default_rng(54321)
    pass_count = 0
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        order = rng.permutation(n)
        any_improved = False

        for jj in order:
            j = int(jj)
            aj = a[j]
            if aj < 1e-14:
                continue

            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo > k_hi:
                continue

            k_idx = np.arange(k_lo, k_hi + 1)
            other_vals = a[k_idx - j]

            # Unaffected max (precompute once per pass would be better, but this works)
            if k_lo > 0 or k_hi < 2 * n - 2:
                affected_mask = np.zeros(2 * n - 1, dtype=bool)
                affected_mask[k_lo:k_hi + 1] = True
                c_unaff_max = float(np.max(c[~affected_mask]))
            else:
                c_unaff_max = 0.0

            two_j_in = (2 * j <= 2 * n - 2)

            # Current max
            cur_max = float(c.max())
            best_f = cur_max
            best_d = 0.0

            # Candidates: reduce by 25%, 50%, 75%, 100%, increase by 25%
            for frac in [0.25, 0.5, 0.75, 1.0, -0.25]:
                d = -frac * aj
                x_new = aj + d
                if x_new < 0:
                    continue
                S = 1.0 + d
                if S < 1e-10:
                    continue
                S2 = S * S

                aff = c[k_idx] + 2.0 * d * other_vals
                if two_j_in:
                    pos = 2 * j - k_lo
                    if 0 <= pos < len(aff):
                        aff[pos] = c[2 * j] + 2.0 * d * aj + d * d
                aff_max = float(np.max(aff))
                unaff_max = c_unaff_max / S2
                f_val = max(aff_max, unaff_max) / S2

                if f_val < best_f - 1e-14:
                    best_f = f_val
                    best_d = d

            if best_d != 0.0:
                a[j] = max(0.0, aj + best_d)
                any_improved = True

        # Recompute full convolution
        c = _full_convolve(a, a)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        c = _full_convolve(a, a)
        current_sc = float(2.0 * n * c.max())

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        pass_count += 1
        if verbose and pass_count % 5 == 0:
            print(f"    {label} pass={pass_count}, score={best_sc:.10f}")

        if no_improve >= 5:
            break

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.01
    min_tol = 0.0002

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-10 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        eta = 0.08
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [0.04, 0.02, 0.01, 0.005]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-13:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.85)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.98)

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40.0 * (t - 0.25) ** 2) + np.exp(-40.0 * (t - 0.75) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n)
             + 0.1 * np.sin(2.0 * np.pi * 23.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "optimal_shape":
        # Based on the theoretical continuous optimum (raised cosine squared)
        t = np.linspace(0, 1, n)
        a = (0.5 * (1 + np.cos(2 * np.pi * t))) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb_and_refine(a, n_perturb, deadline, min_time_remaining, rng):
    """Randomly perturb a subset of coordinates and re-optimize."""
    n = len(a)
    a_pert = a.copy()
    idx = rng.choice(n, size=n_perturb, replace=False)
    # Small multiplicative perturbations
    factors = 1.0 + 0.1 * rng.standard_normal(n_perturb)
    a_pert[idx] *= factors
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a, _score(a)
    a_pert = a_pert / s

    # Quick L-BFGS refinement
    a_opt, sc_opt = _run_lbfgs(a_pert, deadline, min_time_remaining, verbose=False)
    return a_opt, sc_opt


t_start_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, coordinate descent,
    KKT refinement, and perturbation-based local search.
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        # Try multiple initializations
        n_init = 16384
        for prof in ["tapered", "optimal_shape", "raised_cosine", "quadratic"]:
            a_try = _make_profile(n_init, prof)
            sc_try = _score(a_try)
            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
        print(f"Best init: n={len(best_seq)}, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Main L-BFGS ===
    print(f"\n[P1] L-BFGS main (n={n_main}, t={time.monotonic()-t_start_global:.0f}s)")
    p1_end = t_start_global + min(400, (budget_s - 10) * 0.25)
    a1, sc1 = _run_lbfgs(best_seq, p1_end, min_time_remaining=150, verbose=True, label="P1")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
        print(f"  P1 result: {best_score:.10f}")

    # === Phase 2: Coordinate descent ===
    print(f"\n[P2] Coordinate descent (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - 300, time.monotonic() + 200)
        # Use a smaller subset for CD if n is large
        if n_main > 8192:
            # Work on a coarser grid for CD, then resample back
            n_cd = min(8192, n_main)
            a_cd = _resample(best_seq, n_cd) if n_cd < n_main else best_seq.copy()
            a_cd, sc_cd = _cd_with_recompute(a_cd, p2_end, min_time_remaining=50, verbose=True, label="CD")
            if sc_cd * (n_main / len(a_cd)) < best_score * 1.01:
                # Resample back to original size
                a_cd_full = _resample(a_cd, n_main)
                sc_cd_full = _score(a_cd_full)
                if sc_cd_full < best_score:
                    best_score = sc_cd_full
                    best_seq = a_cd_full.copy()
                    print(f"  P2 result (resampled): {best_score:.10f}")
        else:
            a_cd, sc_cd = _cd_with_recompute(best_seq, p2_end, min_time_remaining=50, verbose=True, label="CD")
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: L-BFGS round 2 ===
    print(f"\n[P3] L-BFGS round 2 (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 300)
        a3, sc3 = _run_lbfgs(best_seq, p3_end, min_time_remaining=100, verbose=True, label="P3")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Active-set KKT refinement ===
    print(f"\n[P4] Active-set refine (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p4_end = min(deadline - 200, time.monotonic() + 150)
        a4, sc4 = _active_set_refine(best_seq, p4_end, min_time_remaining=50)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: Perturbation-based local search ===
    print(f"\n[P5] Perturbation search (t={time.monotonic()-t_start_global:.0f}s)")
    n_pertursions = 8
    for p in range(n_pertursions):
        if time.monotonic() + 40 > deadline:
            break
        p5_end = min(deadline - 100, time.monotonic() + 60)
        a_p, sc_p = _perturb_and_refine(best_seq, n_main // 10, p5_end,
                                         min(time.monotonic() + 10, deadline - 100), rng)
        if sc_p < best_score:
            best_score = sc_p
            best_seq = a_p.copy()
            print(f"  Pert {p+1}: {best_score:.10f}")

    # === Phase 6: Coordinate descent on full grid ===
    print(f"\n[P6] Full CD (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p6_end = min(deadline - 100, time.monotonic() + 120)
        a6, sc6 = _cd_with_recompute(best_seq, p6_end, min_time_remaining=50, verbose=True, label="CD6")
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  P6 result: {best_score:.10f}")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start_global:.0f}s)")
    n_options = [12288, 16384, 20480, 24576]
    n_options = [x for x in n_options if x != len(best_seq)]
    for n_try in n_options:
        if time.monotonic() + 50 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if time.monotonic() + 40 > deadline:
            # Adjust score for different n
            adjusted = sc_r * (len(best_seq) / n_try)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
            continue
        p7_end = min(deadline - 60, time.monotonic() + 80)
        a7, sc7 = _run_lbfgs(a_r, p7_end, min_time_remaining=30, verbose=False, label=f"n={n_try}")
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()
            print(f"  n={n_try}: {best_score:.10f}")

    # === Phase 8: Final L-BFGS + KKT polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start_global:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p8_end = min(deadline - 5, time.monotonic() + 80)
        a8, sc8 = _run_lbfgs(best_seq, p8_end, min_time_remaining=5, verbose=False, label="P8")
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

        if time.monotonic() + 10 < deadline:
            p8b_end = min(deadline - 3, time.monotonic() + 30)
            a8b, sc8b = _active_set_refine(best_seq, p8b_end, min_time_remaining=3)
            if sc8b < best_score:
                best_score = sc8b
                best_seq = a8b.copy()

    # Final
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
          f"time={time.monotonic()-t_start_global:.1f}s")
    return [float(x) for x in final]
