"""
LP-Alternating + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Uses linear programming alternating method as the primary optimization
driver, which provides a fundamentally different optimization path than gradient methods.
The LP step finds the best 'partner' sequence that fits under the current correlation
bound, enabling escape from local minima that trap gradient-based methods.
Phases: LP alternating (reduced dim) -> upsample -> L-BFGS -> KKT refine -> multi-n.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_value(c, beta):
    c_max = float(c.max())
    if beta < 1e-30:
        return c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return c_max
    return c_max + beta * np.log(s)


def _soft_max_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_soft_structured(a, beta, n):
    c = _full_convolve(a, a)
    w = _soft_max_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return 2.0 * n * _soft_max_value(c, beta)


def _smograd(z, beta, n):
    a = _softmax(z)
    grad_a = _grad_soft_structured(a, beta, n)
    W = float(np.dot(grad_a, a))
    return a * (grad_a - W)


class _Prob:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

        try:
            result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs',
                             options={'time_limit': 30, 'presolve': True})
        except Exception as e:
            if verbose:
                print(f"    LP-alt it={it}: LP error: {e}")
            break

        if result.status != 0:
            if verbose:
                print(f"    LP-alt it={it}: LP status={result.status}, msg={result.message}")
            break

        b = result.x
        S = float(b.sum())

        if S <= 1.0 + 1e-10:
            if verbose and it < 5:
                print(f"    LP-alt it={it}: S={S:.6f} <= 1, no improvement possible")
            # Try with a slightly relaxed bound
            b_ub_relaxed = np.full(2 * n - 1, M * 1.001)
            try:
                result2 = linprog(c_obj, A_ub=A, b_ub=b_ub_relaxed, bounds=bounds,
                                  method='highs', options={'time_limit': 30})
                if result2.status == 0:
                    b = result2.x
                    S = float(b.sum())
                    if S > 1.0 + 1e-10:
                        a_new = b / S
                        sc_new = _score(a_new)
                        if sc_new < best_sc - min_improvement:
                            best_sc = sc_new
                            best_a = a_new.copy()
                            a = a_new
                            continue
            except Exception:
                pass
            break

        a_new = b / S
        sc_new = _score(a_new)

        if sc_new < best_sc - min_improvement:
            a = a_new
            if verbose and it % 10 == 0:
                print(f"    LP-alt it={it}: S={S:.4f}, sc={best_sc:.10f}")
        else:
            # Try damped update
            for damp in [0.5, 0.3, 0.1]:
                a_d = (1 - damp) * a + damp * a_new
                a_d = np.maximum(a_d, 0)
                sd = a_d.sum()
                if sd < 1e-10:
                    continue
                a_d = a_d / sd
                sc_d = _score(a_d)
                if sc_d < best_sc - min_improvement:
                    a = a_d
                    if verbose and it % 10 == 0:
                        print(f"    LP-alt it={it}: damped({damp}), sc={best_sc:.10f}")
                    break
            else:
                no_improve += 5
                if no_improve > max_no_improve:
                    break

    if verbose:
        print(f"    LP-alt done: best_sc={best_sc:.10f} (n={n})")
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, verbose=False):
    """L-BFGS with adaptive beta schedule on softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    best_a = a.copy()
    best_sc = 2.0 * n * c0_max

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    c_sorted = np.sort(c0)[::-1]
    gap = c_sorted[0] - c_sorted[min(100, len(c_sorted) - 1)]
    if gap < c0_max * 0.003:
        gap = c0_max * 0.003

    multipliers = np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.002,
        0.001, 0.0005, 0.00025, 0.000125, 0.0000625,
        0.00003125, 0.000015625, 0.0000078, 0.0000039,
        0.000002, 0.000001, 0.0000005, 0.0000002
    ])
    betas = gap * multipliers

    t_start = time.monotonic()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        size_factor = max(1.0, n / 4096.0)
        est_iter_time = 0.002 * size_factor
        max_iter = min(6000, max(100, int(remaining / (est_iter_time * 25))))

        if beta < gap * 0.001:
            max_iter = min(max_iter, 1500)

        prob = _Prob(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    elapsed = time.monotonic() - t_start
                    print(f"      beta={beta:.2e} ({i+1}/{len(betas)}): "
                          f"score={best_sc:.10f}, nit={result.nit}, t={elapsed:.0f}s")
        except Exception as e:
            if verbose:
                print(f"      beta={beta:.2e}: error {e}")

    return best_a, best_sc


def _active_set_refine(a_init, deadline, min_time_remaining=30.0):
    """KKT-inspired active-set refinement."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    no_improve = 0
    max_no_improve = 200
    tol_frac = 0.005
    min_tol = 0.00005

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-13:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            tol_frac = max(min_tol, tol_frac * 0.5)
            if tol_frac <= min_tol * 1.1:
                break
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad_avg = 2.0 * wc[n - 1:2 * n - 1]

        support = a > 1e-8 * a.max()
        n_support = int(support.sum())
        if n_support < 2:
            break

        lam = float(np.mean(grad_avg[support]))
        residual = grad_avg[support] - lam

        eta = 0.05
        a_new = a.copy()
        a_new[support] -= eta * residual
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new >= current_sc:
            found = False
            for trial_eta in [0.02, 0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[support] -= trial_eta * residual
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < current_sc - 1e-12:
                    a_new = a_try
                    sc_new = sc_try
                    found = True
                    break
            if not found:
                tol_frac = max(min_tol, tol_frac * 0.85)
                continue

        a = a_new
        if no_improve == 0:
            tol_frac = max(min_tol, tol_frac * 0.97)

    return best_a, best_sc


def _frank_wolfe_polish(a_init, deadline, min_time_remaining=10.0):
    """Frank-Wolfe (conditional gradient) method for final polish."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    t_start = time.monotonic()
    for t in range(100000):
        if time.monotonic() + min_time_remaining > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()

        # Subgradient: average gradient over active set
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Center gradient (for simplex constraint)
        grad_centered = grad - np.mean(grad)

        # Frank-Wolfe: v = e_{argmin grad_centered}
        j_min = int(np.argmin(grad_centered))

        # Step size: 2/(t+2)
        gamma = 2.0 / (t + 2)

        a_new = a.copy()
        a_new -= gamma * a_new[j_min]
        # Move mass to position j_min
        a_new[j_min] += gamma
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        a = a_new

        # Early stopping
        if t > 100 and current_sc - best_sc < 1e-12:
            break

    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 11.0 * t / n)
             + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cubic_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2) + 0.005
        return a / a.sum()
    else:
        return np.ones(n) / n


def _random_perturb(a, frac=0.1, seed=0):
    """Add random perturbation and renormalize."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.random(n) * frac
    a_new = a * (1.0 + noise - 0.5 * frac)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using LP alternating as primary driver:
    P1: LP alternating at n_lp (8192 or 4096)
    P2: Upsample to n_full (16384) + L-BFGS
    P3: KKT active-set refinement
    P4: Frank-Wolfe polish
    P5: Multi-n exploration + random restarts
    P6: Final L-BFGS + KKT
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={len(parent)}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        best_seq = _make_profile(16384, "tapered")
        best_score = _score(best_seq)
        print(f"No parent. Tapered n=16384: score={best_score:.10f}")

    n_full = len(best_seq)

    # === Phase 1: LP Alternating ===
    elapsed = time.monotonic() - t0
    print(f"\n[P1] LP Alternating (elapsed={elapsed:.0f}s)")
    p1_budget = min(600, (budget_s - 12) * 0.35)
    p1_end = t0 + p1_budget

    # Try n_lp = 8192 first
    n_lp = 8192
    a_lp = _resample(best_seq, n_lp)
    a_lp_opt, sc_lp = _lp_alternating(a_lp, p1_end, n_lp=n_lp, verbose=True)

    # Upsample back
    a_up = _resample(a_lp_opt, n_full)
    sc_up = _score(a_up)
    print(f"  LP-alt result: n_lp_sc={sc_lp:.10f}, upsampled_sc={sc_up:.10f}")

    if sc_up < best_score:
        best_score = sc_up
        best_seq = a_up.copy()
        print(f"  -> Updated best: {best_score:.10f}")
    else:
        # Try with smaller n_lp
        n_lp2 = 4096
        if time.monotonic() + 120 < deadline:
            a_lp2 = _resample(best_seq, n_lp2)
            p1b_end = min(deadline - (budget_s - 12) * 0.4, time.monotonic() + 200)
            a_lp2_opt, sc_lp2 = _lp_alternating(a_lp2, p1b_end, n_lp=n_lp2, verbose=True)
            a_up2 = _resample(a_lp2_opt, n_full)
            sc_up2 = _score(a_up2)
            print(f"  LP-alt n=4096: sc={sc_up2:.10f}")
            if sc_up2 < best_score:
                best_score = sc_up2
                best_seq = a_up2.copy()

    # === Phase 2: L-BFGS at full dimension ===
    elapsed = time.monotonic() - t0
    print(f"\n[P2] L-BFGS (n={n_full}, elapsed={elapsed:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p2_end = min(deadline - (budget_s - 12) * 0.25, time.monotonic() + 400)
        a2, sc2 = _run_lbfgs(best_seq, p2_end, min_time_remaining=100, verbose=True)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  P2 result: {best_score:.10f}")

    # === Phase 3: KKT Active-Set Refinement ===
    elapsed = time.monotonic() - t0
    print(f"\n[P3] KKT Active-Set (elapsed={elapsed:.0f}s)")
    if time.monotonic() + 60 < deadline:
        p3_end = min(deadline - 200, time.monotonic() + 200)
        a3, sc3 = _active_set_refine(best_seq, p3_end, min_time_remaining=30)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  P3 result: {best_score:.10f}")

    # === Phase 4: Frank-Wolfe Polish ===
    elapsed = time.monotonic() - t0
    print(f"\n[P4] Frank-Wolfe (elapsed={elapsed:.0f}s)")
    if time.monotonic() + 30 < deadline:
        p4_end = min(deadline - 100, time.monotonic() + 100)
        a4, sc4 = _frank_wolfe_polish(best_seq, p4_end, min_time_remaining=20)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  P4 result: {best_score:.10f}")

    # === Phase 5: L-BFGS again after KKT ===
    elapsed = time.monotonic() - t0
    print(f"\n[P5] L-BFGS round 2 (elapsed={elapsed:.0f}s)")
    if time.monotonic() + 40 < deadline:
        p5_end = min(deadline - 80, time.monotonic() + 150)
        a5, sc5 = _run_lbfgs(best_seq, p5_end, min_time_remaining=30, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  P5 result: {best_score:.10f}")

    # === Phase 6: Multi-n Exploration ===
    elapsed = time.monotonic() - t0
    print(f"\n[P6] Multi-n (elapsed={elapsed:.0f}s)")
    n_options = [8192, 12288, 20480, 24576]
    n_options = [x for x in n_options if x != n_full]
    for n_try in n_options:
        if time.monotonic() + 80 > deadline:
            break
        a_r = _resample(best_seq, n_try)
        sc_r = _score(a_r)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  n={n_try}: {best_score:.10f} (resample only)")
        if time.monotonic() + 60 > deadline:
            continue
        p6_end = min(deadline - 50, time.monotonic() + 80)
        a_r_opt, sc_r_opt = _run_lbfgs(a_r, p6_end, min_time_remaining=20, verbose=False)
        if sc_r_opt < best_score:
            best_score = sc_r_opt
            best_seq = a_r_opt.copy()
            print(f"  n={n_try} optimized: {best_score:.10f}")

    # === Phase 7: Random restarts + LP on best ===
    elapsed = time.monotonic() - t0
    print(f"\n[P7] Random restarts (elapsed={elapsed:.0f}s)")
    if time.monotonic() + 60 < deadline:
        for attempt in range(3):
            if time.monotonic() + 60 > deadline:
                break
            a_pert = _random_perturb(best_seq, frac=0.05, seed=seed + attempt * 100)
            n_lp_small = 4096
            a_pert_small = _resample(a_pert, n_lp_small)
            p7_end = min(deadline - 30, time.monotonic() + 80)
            a_pert_opt, sc_pert = _lp_alternating(a_pert_small, p7_end, n_lp=n_lp_small, verbose=False)
            a_pert_full = _resample(a_pert_opt, len(best_seq))
            sc_pert_full = _score(a_pert_full)
            if sc_pert_full < best_score:
                best_score = sc_pert_full
                best_seq = a_pert_full.copy()
                print(f"  Restart {attempt}: {best_score:.10f}")
            # Quick L-BFGS polish
            if time.monotonic() + 30 < deadline:
                p7b_end = min(deadline - 20, time.monotonic() + 40)
                a_pol, sc_pol = _run_lbfgs(a_pert_full, p7b_end, min_time_remaining=10, verbose=False)
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_seq = a_pol.copy()
                    print(f"  Restart {attempt} polished: {best_score:.10f}")

    # === Phase 8: Final KKT + L-BFGS ===
    elapsed = time.monotonic() - t0
    print(f"\n[P8] Final polish (elapsed={elapsed:.0f}s)")
    if time.monotonic() + 20 < deadline:
        p8_end = min(deadline - 5, time.monotonic() + 80)
        a8, sc8 = _run_lbfgs(best_seq, p8_end, min_time_remaining=5, verbose=False)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()

    if time.monotonic() + 10 < deadline:
        p8b_end = min(deadline - 3, time.monotonic() + 30)
        a8b, sc8b = _active_set_refine(best_seq, p8b_end, min_time_remaining=3)
        if sc8b < best_score:
            best_score = sc8b
            best_seq = a8b.copy()

    # Final validation
    print(f"\n{'='*60}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final]
