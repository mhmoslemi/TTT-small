"""
Optimized minimizer for 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-beta continuation L-BFGS with log-sum-exp smoothing
2. Weighted active-set equalization with Nesterov momentum
3. Periodic perturbation to escape local minima
4. Hessian-informed direction using anti-diagonal structure
5. Multi-n resampling exploration
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_gradient(a, c, top_k, soft_weight=False, temperature=1.0):
    """Compute weighted average gradient of top peaks.
    If soft_weight=True, weight peaks by softmax of their values."""
    n = len(a)
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    
    if soft_weight and k > 1:
        # Weight by proximity to max
        w_top = np.exp(-(top_vals - top_vals[0]) / (temperature + 1e-300))
        w_top = w_top / w_top.sum()
    else:
        w_top = np.ones(k) / k
    
    w_S = np.zeros(2 * n - 1)
    w_S[top_idx] = w_top
    a_rev = a[::-1]
    h = _full_convolve(w_S, a_rev)
    grad = 2.0 * h[n - 1:2 * n - 1]
    return grad, top_idx


def _hessian_vector_product(a, v, top_idx, weights):
    """Compute H_S * v where H_S is the weighted sum of Hessians of active peaks.
    Uses the anti-diagonal structure for O(n log n) computation."""
    n = len(a)
    w_S = np.zeros(2 * n - 1)
    w_S[top_idx] = 2.0 * weights
    v_rev = v[::-1]
    h = _full_convolve(w_S, v_rev)
    return h[n - 1:2 * n - 1]


def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    vv = v / target_sum
    u = np.sort(vv)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(vv - theta, 0.0) * target_sum


def _run_lbfgs_continuation(a_init, deadline, min_time_remaining=100.0):
    """Multi-beta continuation L-BFGS optimization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(50, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = max_c * 0.01
    base_beta = max(gap, max_c * 0.01)

    # Beta schedule: start moderate, end very small
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.001953125,
        0.0009765625, 0.0004882812, 0.0002441406
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        # Scale iterations based on beta and time
        if beta < base_beta * 0.03:
            max_iter = min(8000, max(100, int(remaining / 0.05)))
        elif beta < base_beta * 0.1:
            max_iter = min(5000, max(50, int(remaining / 0.06)))
        else:
            max_iter = min(3000, max(30, int(remaining / 0.08)))

        def obj(z_in):
            return _logsumexp_obj(z_in, beta, n)

        def grad(z_in):
            return _logsumexp_grad(z_in, beta, n)

        try:
            result = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-15,
                    'maxcor': 100,
                    'maxls': 50,
                }
            )
            a_new = _softmax_fn(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-16:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Keep old z if no improvement
                pass
        except Exception:
            pass

    return current_a


def _active_set_optimize(a_init, deadline, min_time_remaining=50.0):
    """Improved active-set equalization with momentum and adaptive K."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    # Momentum state
    velocity = np.zeros(n)
    mu = 0.9  # Nesterov momentum

    K = 5
    K_max = min(200, n)
    lr = 0.01
    no_improve = 0
    max_no_improve = 500

    t_start = time.monotonic()

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # Reset with perturbation
            a = best_a.copy()
            velocity = np.zeros(n)
            lr = 0.01
            K = 5
            no_improve = 0
            # Small random perturbation
            rng_reset = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            pert = rng_reset.standard_normal(n) * 0.001
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            if no_improve > max_no_improve * 3:
                break

        # Adaptive K: grow when peaks are close, shrink when stuck
        c_sorted = np.sort(c)[::-1]
        peak_gap = c_sorted[0] - c_sorted[min(K, len(c_sorted) - 1)]
        if peak_gap < c_max * 0.001 and K < K_max:
            K = min(K_max, K + 2)
        elif no_improve > 50 and K > 5:
            K = max(5, K // 2)

        # Compute weighted active set gradient
        temp = c_max * 0.01  # temperature for soft weighting
        grad, top_idx = _active_set_gradient(a, c, K, soft_weight=True, temperature=temp)
        grad = grad - np.mean(grad)  # Project to sum=0 direction
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Nesterov momentum step
        look_ahead = a + mu * velocity
        c_la = _full_convolve(look_ahead, look_ahead)
        grad_la, _ = _active_set_gradient(look_ahead, c_la, K, soft_weight=True, temperature=temp)
        grad_la = grad_la - np.mean(grad_la)

        velocity = mu * velocity - lr * grad_la
        velocity = velocity - np.mean(velocity)

        # Adaptive step size with line search
        vnorm = np.linalg.norm(velocity)
        if vnorm < 1e-18:
            break
        alpha_base = 0.1 / (vnorm + 1e-15)

        improved = False
        for ls_iter in range(40):
            alpha = alpha_base * (0.7 ** ls_iter)
            a_try = a + alpha * velocity
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            # Also accept if no worse (for plateau navigation)
            if sc_try <= current_sc + 1e-13 and ls_iter > 10:
                a = a_try
                improved = True
                break

        if not improved:
            lr *= 0.8
            mu = max(0.5, mu * 0.95)
            if lr < 1e-6:
                lr = 0.005
                velocity = np.zeros(n)
                mu = 0.9
        else:
            lr = min(0.03, lr * 1.02)
            mu = min(0.95, mu * 1.01)

    return best_a, best_sc

    best_sc = 2.0 * n * float(_full_convolve(a, a).max())

    K = 10
    no_improve = 0

    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        # Get active set
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx]
        weights = np.ones(K) / K

        # Compute gradient
        grad = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            k = idx
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += 2.0 * w * a[kj]
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Compute Hessian-vector product for direction quality assessment
        Hv = _hessian_vector_product(a, grad, top_idx, weights)
        curv = float(np.dot(grad, Hv))

        # Newton-like step size
        if curv > 1e-15:
            alpha = gnorm / curv
        else:
            alpha = 1.0 / (gnorm + 1e-15)
        alpha = min(alpha, 0.5)

        # Line search
        improved = False
        for ls in range(30):
            a_try = a - alpha * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            sc_try = 2.0 * n * float(_full_convolve(a_try, a_try).max())
            if sc_try < current_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 10

    return best_a, best_sc


def _perturb_escape(a, rng, strength=0.02):
    """Create a perturbed version to escape potential local minima."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    # Asymmetric perturbation (breaks reflection symmetry)
    t = np.arange(n) / n
    ramp = 0.4 * (t - 0.5)
    freq = rng.integers(1, 20)
    wave = 0.3 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    noise = 0.3 * rng.standard_normal(n)
    pert = strength * (ramp + wave + noise)
    z = z + pert
    z = z - z.mean()
    return _softmax_fn(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "edge_conc":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * t * (1 - t)) + 0.01
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "modulated":
        t = np.arange(n)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 5 * t / n) + 0.1 * np.cos(2 * np.pi * 3 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """Main optimization: multi-phase search for minimum 2n*max(conv)/(sum)^2."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L-BFGS continuation on parent ===
    print(f"\n[P1] L-BFGS continuation (t={time.monotonic()-t_start:.1f}s)")
    a_p1 = _run_lbfgs_continuation(best_seq, deadline, min_time_remaining=600.0)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_seq = a_p1.copy()
        best_score = sc_p1
        print(f"  NEW BEST: {best_score:.10f} [lbfgs_cont]")
    else:
        print(f"  No improvement (score={sc_p1:.10f})")

    # === Phase 2: Active-set optimization with momentum ===
    print(f"\n[P2] Active-set momentum (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        pol_deadline = min(deadline - 300, time.monotonic() + 200)
        a_eq, sc_eq = _active_set_optimize(best_seq, pol_deadline, min_time_remaining=50.0)
        if sc_eq < best_score:
            best_seq = a_eq.copy()
            best_score = sc_eq
            print(f"  NEW BEST: {best_score:.10f} [active_momentum]")
        else:
            print(f"  No improvement (score={sc_eq:.10f})")

    # === Phase 3: Newton active-set refinement ===
    print(f"\n[P3] Newton active-set (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        newt_deadline = min(deadline - 200, time.monotonic() + 100)
        a_newton, sc_newton = _newton_active_set(best_seq, newt_deadline, min_time_remaining=30.0)
        if sc_newton < best_score:
            best_seq = a_newton.copy()
            best_score = sc_newton
            print(f"  NEW BEST: {best_score:.10f} [newton]")
        else:
            print(f"  No improvement (score={sc_newton:.10f})")

    # === Phase 4: Perturbation + re-optimize (escape local minima) ===
    print(f"\n[P4] Perturbation escape (t={time.monotonic()-t_start:.1f}s)")
    n = len(best_seq)
    for trial in range(3):
        if time.monotonic() > deadline - 200:
            break
        strength = 0.005 * (trial + 1)
        a_pert = _perturb_escape(best_seq, rng, strength=strength)
        # Quick L-BFGS on perturbed
        a_opt = _run_lbfgs_continuation(a_pert, deadline, min_time_remaining=200.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (trial={trial}, s={strength:.4f})")
        else:
            # Try active-set on perturbed
            if time.monotonic() < deadline - 100:
                a_eq2, sc_eq2 = _active_set_optimize(a_pert, deadline, min_time_remaining=100.0)
                if sc_eq2 < best_score:
                    best_seq = a_eq2.copy()
                    best_score = sc_eq2
                    print(f"  NEW BEST: {best_score:.10f} (trial={trial}, active)")
                else:
                    print(f"  Trial {trial}: no improvement (lbfgs={sc_opt:.10f}, active={sc_eq2:.10f})")

    # === Phase 5: Second round of active-set on best ===
    print(f"\n[P5] Active-set round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        pol2_deadline = min(deadline - 100, time.monotonic() + 80)
        a_eq3, sc_eq3 = _active_set_optimize(best_seq, pol2_deadline, min_time_remaining=30.0)
        if sc_eq3 < best_score:
            best_seq = a_eq3.copy()
            best_score = sc_eq3
            print(f"  NEW BEST: {best_score:.10f} [active2]")

    # === Phase 6: Fine L-BFGS polish ===
    print(f"\n[P6] Fine L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        a_fine = _run_lbfgs_continuation(best_seq, deadline, min_time_remaining=60.0)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  NEW BEST: {best_score:.10f} [fine_lbfgs]")

    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = sorted(set([len(best_seq), 16384, 24576, 12288, 32768]), reverse=True)
    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if n_try == len(best_seq):
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        # Quick optimization
        a_opt = _run_lbfgs_continuation(a_init, deadline, min_time_remaining=30.0)
        sc_opt = _score(a_opt)
        # Then active set
        if time.monotonic() < deadline - 15:
            a_act, sc_act = _active_set_optimize(a_opt, deadline, min_time_remaining=15.0)
            if sc_act < sc_opt:
                sc_opt = sc_act
                a_opt = a_act
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n]")
        else:
            print(f"  n={n_try}: best={sc_opt:.10f} (no improvement)")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    remaining = deadline - 5 - time.monotonic()
    if remaining > 10:
        a_fin = _run_lbfgs_continuation(best_seq, deadline, min_time_remaining=5.0)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_lbfgs]")

        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_optimize(best_seq, deadline - 2, min_time_remaining=2.0)
            if sc_fin2 < best_score:
                best_seq = a_fin2.copy()
                best_score = sc_fin2
                print(f"  NEW BEST: {best_score:.10f} [final_active]")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
