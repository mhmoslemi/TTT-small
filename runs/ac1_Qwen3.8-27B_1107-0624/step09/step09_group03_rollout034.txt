import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _corr_a_A(a, A_vec, n):

    flip_a = a[::-1]

    fa = np.fft.rfft(flip_a, N)

    result = np.fft.irfft(fa * fA, N)[:out_len]

    if s < 1e-300:

def _cg_solve(matvec, rhs, max_cg=300, tol=1e-14):
    x = np.zeros_like(rhs)
    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

    if m > 200:  # Too large for exact solve

    # (T_k v)[i] = sum_j T_k[i,j] v[j]
    # T_k[i,j] = 1 if j = i+k or j = i-k (within bounds)

    # H = 2 * (T_A + T_A^T) / m + mu * I
    # where T_A v = corr(v, A_vec)
    
    # Form H by applying it to basis vectors
    # This is O(n^2) which is fine for n <= 32768 if m <= 200
    # But forming the full n x n matrix is O(n^2) = 10^9 for n=10000, too much.
    
    # So we should only use exact solve for n <= 2000 or so.
    if n > 4000:
        return None
    
    H = np.zeros((n, n))
    for i in range(n):
        e_i = np.zeros(n)
        e_i[i] = 1.0
        He_i = 2.0 * _corr_a_A(e_i, A_vec, n)
        H[:, i] = He_i + mu * e_i
    
    # Solve H * dx = -g_t, with the constraint sum(dx) = 0
    # Use the消去 constraint: dx = dx_raw - mean(dx_raw)
    
    try:
        dx_raw = np.linalg.solve(H, -g_t)
        dx = dx_raw - np.mean(dx_raw)
    except np.linalg.LinAlgError:
        return None
    
    return dx

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _corr_a_A(a, A_vec, n):
    flip_a = a[::-1]
    out_len = n + len(A_vec) - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(flip_a, N)
    fA = np.fft.rfft(A_vec, N)
    result = np.fft.irfft(fa * fA, N)[:out_len]
    return result[n - 1:2 * n - 1]

    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    r = rhs - matvec(x)
    p = r.copy()
    rs = np.dot(r, r)
    rhs_norm = np.linalg.norm(rhs)
    for _ in range(max_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if pAp < 1e-30:

        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (1 + rhs_norm):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x
