"""
Optimizer combining:
1. New: Alternating multiplicative-weight method with progressive dual sharpening
2. L-BFGS with top-K smoothing for polish
3. Subgradient with Barzilai-Borwein steps for fine-tuning
4. Coarse-to-fine multi-scale strategy
5. Multi-restart for escaping local minima

Key insight: The problem min max(conv(a,a)) s.t. sum(a)=1, a>=0 can be
solved via alternating updates: (a) find a minimizing weighted pressure,
(b) update weights to focus on peaks. The a-update has a fixed point
solving a linear system (Toeplitz), approximated by multiplicative
iteration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1 already."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _pressure(a, y):
    """Compute pressure[m] = sum_j y[m+j] * a[j] for m=0..n-1.
    This is the unnormalized gradient direction for weighted max convolution."""
    n = len(a)
    # pressure = cross-correlation of y and a
    # Using: conv(y[::-1], a)[n-1:2n-1][::-1]
    y_rev = y[::-1]
    conv_result = _full_convolve(y_rev, a)
    # conv_result has length len(y) + n - 1 = (2n-1) + n - 1 = 3n-2
    # We want indices n-1 through 2n-2, reversed
    pressure = conv_result[n - 1:2 * n - 1][::-1]
    return pressure


def _alt_optimize(a_init, deadline, min_time_remaining=60.0):
    """Alternating multiplicative-weight optimizer.
    
    Alternates between:
    - Primal: a[m] <- a[m] * (target/pressure[m])^eta (equalize pressure)
    - Dual: y <- smooth(exp(beta * conv(a,a))) (focus on peaks)
    
    With progressive sharpening of y and adaptive parameters.
    """
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    best_score = 2.0 * n * c_max
    
    # Initialize y: start broad, sharpen over time
    c_norm = (c - c.max()) / max(c_max * 0.01, 1e-30)
    c_norm = np.clip(c_norm, -100, 0)
    beta = 5.0
    y = np.exp(beta * c_norm)
    y = y / y.sum()
    
    eta = 0.4  # multiplicative step size
    inner_iters = 5
    no_improve = 0
    max_no_improve = 200
    stall_cooldown = 0
    
    t_start = time.monotonic()
    iteration = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        
        # --- Primal update: multiplicative weight ---
        for _ in range(inner_iters):
            pressure = _pressure(a, y)
            pressure = np.maximum(pressure, 1e-30)
            mean_p = pressure.mean()
            ratio = mean_p / pressure
            # Damped multiplicative update
            a_new = a * np.power(ratio, eta)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                # Check if this actually helps (quick score check every few iters)
                a = a_new
            else:
                break
        
        # --- Score ---
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_score = 2.0 * n * c_max
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # --- Dual update ---
        c_shifted = (c - c_max) / c_max
        c_shifted = np.clip(c_shifted, -80, 0)
        y_new = np.exp(beta * c_shifted)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y
            # Exponential moving average
            decay = 0.6
            y = (1.0 - decay) * y + decay * y_new
            y = y / y.sum()
        
        # --- Adaptive parameters ---
        # Sharpen y over time (increase beta)
        beta = min(80.0, beta * 1.008)
        # Reduce step size over time
        eta = max(0.05, eta * 0.998)
        # Adjust inner iterations
        if no_improve > 50:
            inner_iters = max(2, inner_iters - 1)
        elif no_improve == 0:
            inner_iters = min(10, inner_iters + 1)
        
        # Stall handling
        if no_improve > max_no_improve:
            # Try a different strategy: reset y to be broader
            if stall_cooldown % 3 == 0:
                c_shifted = (c - c_max) / (c_max * 0.5)
                c_shifted = np.clip(c_shifted, -50, 0)
                y = np.exp(2.0 * c_shifted)
                y = y / y.sum()
                beta = max(5.0, beta * 0.5)
            elif stall_cooldown % 3 == 1:
                # Add noise to a to escape
                a = a * (1.0 + 0.02 * np.random.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = _normalize(a)
            elif stall_cooldown % 3 == 2:
                # Try uniform y
                y = np.ones(2 * n - 1) / (2 * n - 1)
                beta = 5.0
            stall_cooldown += 1
            no_improve = 0
        
        # Progress print
        if iteration % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    [alt] iter={iteration}, score={current_score:.10f}, "
                  f"best={best_score:.10f}, beta={beta:.1f}, eta={eta:.3f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def _alt_optimize_v2(a_init, deadline, min_time_remaining=60.0):

    best_a = a.copy()

def _grad_maxconv(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    # grad[m] = 2 * sum_k w[k] * a[k-m] = 2 * pressure-like with w
    # = 2 * _pressure(a, w) but w here is the full convolution-length weight
    # Actually: grad[m] = 2 * sum_{k} w[k] * a[k-m] (for valid k-m)
    # This is 2 * cross_correlation(w, a)[m]
    w_rev = w[::-1]
    conv_result = _full_convolve(w_rev, a)
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad


def _topk_smooth_obj_grad(z, beta, K, n):
    """Top-K log-sum-exp smoothed objective and gradient in log-space."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
        top_idx = np.arange(len(c))
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z


class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smooth_obj_grad(z, self.beta, self.K, self.n)[1]

        return current_a

    if gap < max_c * 0.001:
        gap = max_c * 0.001

                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 80, 'maxls': 40}
            )
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _subgradient_bb(a_init, deadline, min_time_remaining=60.0):
    """Active-set subgradient with BB steps and momentum."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    n_2nm1 = 2 * n - 1
    active_frac = 0.01
    momentum = 0.7
    lr = 0.02
    lr_min = 1e-5
    lr_max = 0.08
    stall = 0
    max_stall = 100
    velocity = np.zeros(n)
    
    prev_a = None
    prev_g = None
    iteration = 0
    last_lbfgs_time = time.monotonic()
    
    while time.monotonic() + min_time_remaining < deadline:
        iteration += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        n_active = max(3, min(n_2nm1, int(n_2nm1 * active_frac)))
        c_sorted_idx = np.argsort(c)[::-1]
        active_idx = c_sorted_idx[:n_active]
        
        active_vals = c[active_idx]
        w = np.zeros(n_2nm1)
        spread = active_vals.max() - active_vals.min()
        if spread > 1e-18:
            w[active_idx] = (active_vals - active_vals.min()) / spread + 0.1
        else:
            w[active_idx] = 1.0 / n_active
        w = w / w.sum()
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-18:
            break
        
        if prev_a is not None and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g_centered - prev_g
            s_norm = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if s_norm > 1e-18 and abs(sy) > 1e-18:
                lr_bb = s_norm / abs(sy)
                lr = np.clip(lr_bb, lr_min, lr_max)
        
        velocity = momentum * velocity - lr * g_centered
        a_try = a + velocity
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        
        improved = False
        if s_try > 1e-10:
            a_try = a_try / s_try
            for ls in range(20):
                c_try = _full_convolve(a_try, a_try)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < current_sc - 1e-15:
                    improved = True
                    break
                velocity = momentum * velocity * 0.5 - lr * 0.5 * g_centered
                a_try = a + velocity
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    velocity = np.zeros(n)
                    a_try = a
                    break
                a_try = a_try / s_try
                lr *= 0.7
        
        if improved:
            prev_a = a.copy()
            prev_g = g_centered.copy()
            a = a_try
            lr = min(lr_max, lr * 1.02)
            active_frac = max(0.002, active_frac * 0.98)
        else:
            velocity = np.zeros(n)
            lr = max(lr_min, lr * 0.6)
            active_frac = min(0.08, active_frac * 1.3)
        
        if stall > max_stall:
            active_frac = min(0.08, active_frac * 2.0)
            lr = max(lr_min, lr * 0.3)
            momentum = min(0.9, momentum + 0.05)
            stall = 0
        
        if iteration % 400 == 0 and time.monotonic() - last_lbfgs_time > 100:
            if time.monotonic() + min_time_remaining + 80 < deadline:
                lbfgs_deadline = time.monotonic() + 70
                a_tmp = _run_lbfgs(a, lbfgs_deadline, min_time_remaining=5)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_sc:
                    best_a = a_tmp.copy()
                    best_sc = sc_tmp
                    a = a_tmp.copy()
                    velocity = np.zeros(n)
                    prev_a = None
                    prev_g = None
                    last_lbfgs_time = time.monotonic()
    
    return best_a, best_sc


def _fine_polish(a_init, deadline, min_time_remaining=10.0):
    """Very fine active-set polish with small steps."""
    n = len(a_init)
    a = _normalize(a_init.copy())
    best_a = a.copy()
    best_sc = _score(a)
    
    lr = 0.003
    tol = 0.004
    no_improve = 0
    
    while time.monotonic() + min_time_remaining < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-15:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        active = c >= (c_max - tol * c_max)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros(2 * n - 1)
        # Weighted by how close to max
        active_vals = c[active]
        w[active] = 1.0 + (active_vals - active_vals.min()) / max(active_vals.max() - active_vals.min(), 1e-30)
        w = w / w.sum()
        
        g = _grad_maxconv(a, w)
        g_centered = g - g.mean()
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-16:
            break
        
        alpha = min(lr, 0.5 / (gnorm + 1e-15))
        improved = False
        for ls in range(35):
            a_try = a - alpha * g_centered
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < current_sc - 1e-15:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol = max(0.0001, tol * 0.7)
                lr = max(0.0005, lr * 0.7)
            if no_improve > 400:
                a = best_a.copy()
                tol = max(0.0001, tol * 0.5)
                lr = 0.002
                no_improve = 0
        else:
            if tol > 0.0001:
                tol = max(0.0001, tol * 0.97)
            lr = min(0.01, lr * 1.01)
    
    return best_a, best_sc


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bathtub":
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * np.cos(np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bathtub2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * np.cos(np.pi * t) + 0.3 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "power_law":
        t = np.linspace(0.001, 1.0, n)
        mid = n // 2
        dist = np.abs(np.arange(n) - mid) / mid
        a = 1.0 / (1.0 + 2.0 * dist ** 1.5)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    P1: Alternating multiplicative weight (new method)
    P2: L-BFGS top-K polish
    P3: Subgradient BB
    P4: L-BFGS round 2
    P5: Fine polish
    P6: Multi-restart + alternative profiles
    P7: Multi-n exploration
    P8: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 16384
        parent = _make_profile(n_parent, "bathtub")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, bathtub n={n_parent}, score={best_score:.10f}")
    
    n_main = len(best_seq)
    
    # === Phase 1: Alternating multiplicative weight (NEW) ===
    print(f"\n[P1] Alternating MW v1 (t={time.monotonic()-t_start:.1f}s)")
    alt_budget = min(deadline - 600, t_start + 300.0)
    a1, sc1 = _alt_optimize(best_seq, alt_budget, min_time_remaining=600)
    if sc1 < best_score:
        best_seq = a1.copy()
        best_score = sc1
        print(f"  NEW BEST: {best_score:.10f} [alt1] t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # Phase 1b: Alternative v2
    print(f"\n[P1b] Alternating MW v2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        alt_budget2 = min(deadline - 500, time.monotonic() + 120.0)
        a1b, sc1b = _alt_optimize_v2(best_seq, alt_budget2, min_time_remaining=500)
        if sc1b < best_score:
            best_seq = a1b.copy()
            best_score = sc1b
            print(f"  NEW BEST: {best_score:.10f} [alt2] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: L-BFGS ===
    print(f"\n[P2] L-BFGS top-K (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        lbfgs_budget = min(deadline - 400, time.monotonic() + 250.0)
        a2 = _run_lbfgs(best_seq, lbfgs_budget, min_time_remaining=400)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_seq = a2.copy()
            best_score = sc2
            print(f"  NEW BEST: {best_score:.10f} [lbfgs1] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: Subgradient BB ===
    print(f"\n[P3] Subgradient BB (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        sg_budget = min(deadline - 300, time.monotonic() + 200.0)
        a3, sc3 = _subgradient_bb(best_seq, sg_budget, min_time_remaining=300)
        if sc3 < best_score:
            best_seq = a3.copy()
            best_score = sc3
            print(f"  NEW BEST: {best_score:.10f} [subgrad] t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  No improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Second L-BFGS ===
    print(f"\n[P4] L-BFGS round 2 (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        lbfgs_budget2 = min(deadline - 200, time.monotonic() + 150.0)
        a4 = _run_lbfgs(best_seq, lbfgs_budget2, min_time_remaining=200)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_seq = a4.copy()
            best_score = sc4
            print(f"  NEW BEST: {best_score:.10f} [lbfgs2] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Fine polish ===
    print(f"\n[P5] Fine polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        fp_budget = min(deadline - 120, time.monotonic() + 100.0)
        a5, sc5 = _fine_polish(best_seq, fp_budget, min_time_remaining=120)
        if sc5 < best_score:
            best_seq = a5.copy()
            best_score = sc5
            print(f"  NEW BEST: {best_score:.10f} [polish] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multi-restart ===
    print(f"\n[P6] Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        # Perturbed restarts
        sigmas = [0.02, 0.04, 0.06, 0.1, 0.03, 0.08, 0.05, 0.12]
        n_restarts = min(8, max(2, int((deadline - 100 - time.monotonic()) / 40)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            sigma = sigmas[i % len(sigmas)]
            a_pert = _perturb(best_seq, sigma, rng)
            # Try alternating method first (faster)
            alt_d = min(deadline - 50, time.monotonic() + 25.0)
            a_r, sc_r = _alt_optimize(a_pert, alt_d, min_time_remaining=50)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  NEW BEST: {best_score:.10f} [restart_alt_{i}] t={time.monotonic()-t_start:.1f}s")
            elif sc_r < _score(a_pert) + 1e-6 and time.monotonic() < deadline - 40:
                # If alternating helped the perturbed version, try L-BFGS too
                lbfgs_d = min(deadline - 35, time.monotonic() + 20.0)
                a_r2 = _run_lbfgs(a_r, lbfgs_d, min_time_remaining=35)
                sc_r2 = _score(a_r2)
                if sc_r2 < best_score:
                    best_seq = a_r2.copy()
                    best_score = sc_r2
                    print(f"  NEW BEST: {best_score:.10f} [restart_lb_{i}] t={time.monotonic()-t_start:.1f}s")
        
        # Try different profiles
        if time.monotonic() < deadline - 50:
            profiles = ["bathtub", "bathtub2", "power_law", "cos_taper"]
            for prof_kind in profiles:
                if time.monotonic() > deadline - 30:
                    break
                a_prof = _make_profile(n_main, prof_kind)
                # Quick alternating optimization
                alt_d = min(deadline - 25, time.monotonic() + 20.0)
                a_p, sc_p = _alt_optimize(a_prof, alt_d, min_time_remaining=25)
                if sc_p < best_score:
                    best_seq = a_p.copy()
                    best_score = sc_p
                    print(f"  NEW BEST: {best_score:.10f} [profile_{prof_kind}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Multi-n exploration ===
    print(f"\n[P7] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_options = [12288, 24576, 20480]
        n_options = [x for x in n_options if x != n_main]
        for n_try in n_options[:2]:
            if time.monotonic() > deadline - 30:
                break
            a_res = _resample(best_seq, n_try)
            # Quick optimization on new n
            alt_d = min(deadline - 25, time.monotonic() + 25.0)
            a_opt, sc_opt = _alt_optimize(a_res, alt_d, min_time_remaining=25)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} [n={n_try}] t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 8: Final polish ===
    print(f"\n[P8] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3:
        # Fine polish
        a_fin, sc_fin = _fine_polish(best_seq, deadline - 2, min_time_remaining=2)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")
        
        # Tiny L-BFGS
        remaining = deadline - 2 - time.monotonic()
        if remaining > 5:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = max(mb * 0.0003, 1e-8)
            K_fin = max(5, int(n * 0.002))
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _TopkSmoProb(beta_fin, K_fin, n)
            max_iter = min(2000, max(20, int(remaining / 0.05)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-18,
                                           'gtol': 1e-14, 'maxcor': 50})
                a_fin2 = np.exp(result.x - result.x.max())
                a_fin2 = a_fin2 / a_fin2.sum()
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "bathtub")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "bathtub")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
