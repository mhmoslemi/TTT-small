```python
    while k < n:

    fb = np.fft.rfft(b, N)
```

```python
"""
L_p norm minimization with progressive sharpening for minimizing
2n * max(conv(a,a)) / sum(a)^2.
Uses smooth L_p approximation to the max, optimized via L-BFGS-B,
followed by KKT-based refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_obj_and_grad(z, p, n):
    """
    Compute L_p objective and gradient for the L_p norm of the convolution.
    Objective: (sum_k conv[k]^p)^{1/p} / (sum a)^{... }
    Actually, since we use softmax parameterization, sum(a) = 1, so we just minimize
    (sum_k conv[k]^p)^{1/p}.
    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 0.0)  # numerical safety
    
    c_max = float(c.max())
    if c_max < 1e-30:
        return 0.0, np.zeros_like(z)
    
    # Normalize for numerical stability
    c_norm = c / c_max
    obj = c_max * float(np.sum(c_norm**p))**(1.0/p) / (2*n-1)**(1.0/p)
    # Actually, let's just use sum(c^p) as the objective (monotone in the L_p norm)
    # And scale the gradient accordingly
    
    if p == 1:
        w = np.ones_like(c) / len(c)
    else:
        w = p * (c_norm**(p-1)) * c_max**(p-1) / len(c)
    
    # Gradient w.r.t. a:
    # d/d a[i] sum_k w_k * conv[k] = sum_k w_k * 2*a[k-i] (valid indices)
    # = 2 * (w * reverse(a))[i] (convolution, taking the right slice)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad_a = 2.0 * wc[n-1:2*n-1]
    
    # Gradient w.r.t. z (softmax):
    # grad_z = a * (grad_a - dot(a, grad_a))
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z


def _optimize_lp(a_init, deadline, p_schedule=None, max_total_iter=5000):
    """
    Optimize the L_p norm of the convolution with progressive p.
    """
    n = len(a_init)
    
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    iter_offset = 0
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        
        remaining_iters = max_total_iter - iter_offset
        if remaining_iters <= 0:
            break
        
        # Estimate time per iteration
        # For n elements, each gradient eval takes about 2 FFTs of size 2n
        # Rough estimate: 0.5ms per FFT for n=16384, so ~1ms per grad eval
        # L-BFGS uses maybe 2-3 grad evals per iteration
        est_time_per_iter = 0.005 * (n / 16384.0)
        max_iter = min(remaining_iters, max(10, int((deadline - time.monotonic()) / (est_time_per_iter * 3))))
        max_iter = min(max_iter, 1000)
        
        def obj_fn(z_inner, p_inner=p, n_inner=n):
            o, _ = _lp_obj_and_grad(z_inner, p_inner, n_inner)
            return o
        
        def grad_fn(z_inner, p_inner=p, n_inner=n):
            _, g = _lp_obj_and_grad(z_inner, p_inner, n_inner)
            return g
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 50}
            )
            z = result.x
            iter_offset += result.nit if hasattr(result, 'nit') else max_iter
        except Exception as e:

    KKT-based refinement using active set of top-K convolution values.

    if s < 1e-15:
        return a, np.inf

        tol = 0.01 * c_max

        a_rev = a[::-1]

        gn = float(np.linalg.norm(grad_a))

            break

            c_try = _full_convolve(a_try, a_try)

                a = a_try
                improved = True
                break

        if sc < best_sc - 1e-15:
```

```python
"""
L_p norm progressive minimization with KKT refinement for minimizing
2n * max(conv(a,a)) / sum(a)^2. Uses smooth L_p approximations to the max
functional, optimized via L-BFGS-B with FFT-based gradients, followed by
adaptive active-set KKT steps. Explores multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_obj_grad(z, p, n):
    """Compute L_p objective and gradient via softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    n_c = len(c)

    # Objective: sum(c^p) / n_c (monotone in L_p norm)
    log_c = np.log(c)
    log_cp = p * log_c
    log_cp_max = np.max(log_cp)
    sum_cp = np.exp(log_cp_max) * np.sum(np.exp(log_cp - log_cp_max))
    obj = float(sum_cp / n_c)

    # Weight: w[k] = p * c[k]^(p-1) / n_c (numerically stable via log)
    log_w = np.log(float(p)) + (p - 1.0) * log_c - np.log(float(n_c))
    log_w_max = np.max(log_w)
    w = np.exp(log_w - log_w_max)

    # grad_a[i] = 2 * (w * reverse(a))[i] (taking the correct slice)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]

    # grad_z = a * (grad_a - <a, grad_a>)
    dot_val = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot_val)

    return obj, grad_z


def _optimize_lp(a_init, deadline, p_schedule=None, max_iter_per_p=2000):
    """Progressive L_p optimization using L-BFGS-B."""
    n = len(a_init)
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192]

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    for p in p_schedule:
        if time.monotonic() > deadline:
            break

        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter based on remaining time and n
        est_time_per_iter = 0.004 * max(1.0, n / 16384.0)
        this_max_iter = min(max_iter_per_p, max(10, int(remaining / (est_time_per_iter * 3))))
        this_max_iter = min(this_max_iter, 5000)

        def obj_fn(z_inner):
            o, _ = _lp_obj_grad(z_inner, p, n)
            return o

        def grad_fn(z_inner):
            _, g = _lp_obj_grad(z_inner, p, n)
            return g

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',

    if sc1 < best_sc:
        best_a, best_sc = a1, sc1
        print(f"  After L_p: {best_sc:.10f}")

    # Phase 2: KKT refinement
    print(f"\n[Phase 2] KKT refinement (n={len(best_a)})")
    t_p2 = min(deadline - 400.0, t_start + 700.0)
    a2, sc2 = _kkt_refine(best_a, t_p2, max_rounds=300)
    if sc2 < best_sc:
        best_a, best_sc = a2, sc2
        print(f"  After KKT: {best_sc:.10f}")

    # Phase 3: Direct max push
    print(f"\n[Phase 3] Direct max push (n={len(best_a)})")
    t_p3 = min(deadline - 300.0, t_start + 850.0)
    a3, sc3 = _direct_max_push(best_a, t_p3, max_rounds=200)
    if sc3 < best_sc:
        best_a, best_sc = a3, sc3
        print(f"  After push: {best_sc:.10f}")

    # Phase 4: Multi-n exploration
    print(f"\n[Phase 4] Multi-n exploration")
    n_tries = sorted(set([len(best_a) // 2, len(best_a), len(best_a) * 2,
                          8192, 16384, 32768, 65536]))
    n_tries = [n for n in n_tries if 4096 <= n <= 65536]

    for n_try in n_tries:
        if time.monotonic() > deadline - 80:
            break

        a_res = _resample(best_a, n_try)
        sc_res = _score(a_res)
        label = f"n={n_try}"

        if sc_res < best_sc:
            best_a, best_sc = a_res, sc_res
            print(f"  {label} (resample): {best_sc:.10f}")

        # Short L_p optimization on this n
        t_sub = min(deadline - 40.0, time.monotonic() + 80)
        if time.monotonic() < deadline - 50:
            p_short = [16, 64, 256, 1024, 4096]
            a_opt, sc_opt, _ = _optimize_lp(a_res, t_sub, p_schedule=p_short,
                                             max_iter_per_p=500)
            if sc_opt < best_sc:
                best_a, best_sc = a_opt, sc_opt
                print(f"  {label} (L_p): {best_sc:.10f}")

            # Quick KKT
            a_kkt, sc_kkt = _kkt_refine(best_a, min(deadline - 20, time.monotonic() + 30),
                                         max_rounds=50)
            if sc_kkt < best_sc:
                best_a, best_sc = a_kkt, sc_kkt
                print(f"  {label} (KKT): {best_sc:.10f}")

    # Phase 5: Final L_p + KKT on best
    print(f"\n[Phase 5] Final refinement (n={len(best_a)})")
    if time.monotonic() < deadline - 20:
        t_final = min(deadline - 5.0, time.monotonic() + 120)
        p_final = [64, 256, 1024, 4096, 8192, 16384]
        a_f, sc_f, _ = _optimize_lp(best_a, t_final, p_schedule=p_final,
                                     max_iter_per_p=1000)
        if sc_f < best_sc:
            best_a, best_sc = a_f, sc_f
            print(f"  Final L_p: {best_sc:.10f}")

    # Phase 6: Very last KKT
    if time.monotonic() < deadline - 3:
        a_last, sc_last = _kkt_refine(best_a, deadline - 1.0, max_rounds=50)
        if sc_last < best_sc:
            best_a, best_sc = a_last, sc_last
            print(f"  Final KKT: {best_sc:.10f}")

    # Cleanup and return
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256, dtype=np.float64)
    sc_final = _score(final)
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(final)}, score={sc_final:.10f}, time={time.monotonic() - t_start:.1f}s")

    return [float(x) for x in final]
```
