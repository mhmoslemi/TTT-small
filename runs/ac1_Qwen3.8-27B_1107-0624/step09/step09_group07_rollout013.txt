Focused optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K log-sum-exp smoothing with very small K, aggressive L-BFGS,
and multi-restart with diverse perturbations.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _grad_maxconv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

def _topk_smoobj_grad(z, beta, K, n):
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    shifted = (c_top - c_max) / beta
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        e_full = np.zeros(2 * n - 1)
        e_full[top_idx] = e
        w = e_full / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    ga = _grad_maxconv(a, w)
    W = float(np.dot(a, ga))
    grad_z = a * (ga - W)
    return obj, grad_z

class _TopkSmoProb:
    def __init__(self, beta, K, n):
        self.beta = beta
        self.K = K
        self.n = n

    def __call__(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[0]

    def grad(self, z):
        return _topk_smoobj_grad(z, self.beta, self.K, self.n)[1]

        gap = max_c * 0.001
    base_beta = max(gap * 0.5, max_c * 0.003)

    z = z - z.mean()

    prev_g = None

        if current_sc < best_sc - 1e-16:

            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Very focused active set

        c_sorted_idx = np.argsort(c)[::-1]

                a_try = a + velocity

                if s_try < 1e-10:

                    break

def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_taper":
        t = np.linspace(0, 1, n)
        a = 0.4 + 0.6 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        # Flat in the middle, tapered at edges
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(n * 0.2)
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n-1-i] *= (i + 1) / taper
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def _coordinate_descent_step(a, n_steps=100):
    """For n_steps random coordinates, find optimal a[i] by 1D minimization."""
    n = len(a)
    best_sc = _score(a)
    
    for step in range(n_steps):
        i = np.random.randint(n)
        # Find the range of convolution indices affected by a[i]
        # c[k] = sum_j a[j]*a[k-j], so a[i] appears in c[k] when k-j=i, i.e., k=i+j
        # So a[i] affects c[i], c[i+1], ..., c[i+n-1] (capped at 0..2n-2)
        k_lo = max(0, i)
        k_hi = min(2*n-2, i+n-1)
        
        # Current values of these convolution entries
        c = _full_convolve(a, a)
        
        # For each affected c[k], c[k] = a[i]*a[k-i] + rest_k
        # where rest_k = c[k] - a[i]*a[k-i]
        # We want to find a[i] that minimizes max_k (a[i]*a[k-i] + rest_k)
        
        # Collect the coefficients
        coeffs = []
        rests = []
        for k in range(k_lo, k_hi+1):
            j = k - i
            if 0 <= j < n:
                coeffs.append(a[j])
                rests.append(c[k] - a[i] * a[j])
        
        if not coeffs:
            continue
        
        coeffs = np.array(coeffs)
        rests = np.array(rests)
        
        # We want to minimize max_k (coeffs[k]*x + rests[k]) over x >= 0
        # subject to the constraint that the sum changes appropriately
        
        # The optimal x is where the maximum of the lines is minimized
        # This is a 1D linear programming problem
        # The breakpoints are where coeffs[i]*x + rests[i] = coeffs[j]*x + rests[j]
        # i.e., x = (rests[j] - rests[i]) / (coeffs[i] - coeffs[j])
        
        # Find all breakpoints
        breakpoints = [0.0, a[i]]  # Include current value and 0
        m = len(coeffs)
        for p in range(m):
            for q in range(p+1, m):
                if abs(coeffs[p] - coeffs[q]) > 1e-15:
                    bp = (rests[q] - rests[p]) / (coeffs[p] - coeffs[q])
                    if 0 <= bp <= a[i] * 2:
                        breakpoints.append(bp)
        
        if len(breakpoints) < 2:
            continue
        
        breakpoints = sorted(set(round(bp, 10) for bp in breakpoints))
        
        # Evaluate max at each breakpoint and midpoint
        candidates = [a[i]]
        for bp in breakpoints:
            candidates.append(bp)
        
        best_x = a[i]
        best_local_sc = _score(a)
        
        # Evaluate at breakpoints
        for x in candidates:
            x = max(0.0, x)
            a[i] = x
            c_new = _full_convolve(a, a)
            sc = 2.0 * n * c_new.max()
            if sc < best_local_sc:
                best_local_sc = sc
                best_x = x
        
        a[i] = best_x
        # Renormalize
        a = a / a.sum()
    
    return a
